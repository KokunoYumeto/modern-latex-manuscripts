\noindent\textbf{Translation / الترجمة:}

\begin{quote}
In the name of God, the Merciful, the Compassionate.

Praise be to God, who is unique in creating units, and peerless in composing the classes of numbers. And blessings upon the best of His creation, Muhammad, the intercessor for the interceders on the Day of Calling, and upon his family and his sons who guide to the path of salvation and right guidance.

Jamshid ibn Mas'ud ibn Mahmud al-Tabib al-Kashi, called Ghiyath al-Din --- may God improve his condition --- says:

When I had practiced arithmetic and geometric methods until I reached their true principles, and penetrated their subtleties, and uncovered their hidden difficulties and problems and solved their challenges, and derived many rules and controls therein, and extracted what was difficult to extract for many of their practitioners...

I found in the course of these works many rules by which the operations of arithmetic preliminaries may be performed in the easiest manner and the simplest ways and with the least effort and the greatest benefit and the clearest arrangement. So I resolved to record them and to explain them, to serve as a reminder for beloved friends and as guidance for people of understanding.
\end{quote}

\section{\ar{الفصل الرابع: في الضرب وطرقه}}

\begin{arabicbox}
\begin{center}
\arLarge{فصل في الضرب وطرقه}
\end{center}
\end{arabicbox}

\vspace{0.3cm}

\noindent\ar{يقدم الكاشي عدة طرق للضرب. أشهرها هي \textbf{طريقة الشبكة} (\textit{الشبكة} أو ضرب الجدول)، التي يصفها بالتفصيل الكامل. ويقدم أيضاً \textbf{طريقة الأعمدة} وطرق الحساب الذهني.}

\begin{mathalgo}
\textbf{\ar{طريقة ضرب الشبكة:}}
\begin{enumerate}[nosep]
  \item \ar{ارسم شبكة من المربعات بعدد صفوف وأعمدة يساوي عدد أرقام المضروب والمضروب فيه.}
  \item \ar{اكتب المضروب فوق الشبكة والمضروب فيه على يمينها.}
  \item \ar{اقسم كل مربع قطرياً من أعلى يسار إلى أسفل يمين.}
  \item \ar{اضرب كل رقم من المضروب في كل رقم من المضروب فيه، واكتب العشرات فوق القطر والوحدات تحته.}
  \item \ar{اجمع على طول كل قطر من أسفل يمين إلى أعلى يسار، مع النقل عند الحاجة.}
  \item \ar{اقرأ الناتج النهائي على طول الحافتين اليسرى والسفلى.}
\end{enumerate}
\end{mathalgo}

\vspace{0.3cm}

\noindent\textbf{\ar{مثال:}} \ar{لضرب} $524 \times 385$:

\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
& \textbf{5} & \textbf{2} & \textbf{4} \\
\hline
\textbf{3} & \multicolumn{1}{c|}{1/5} & \multicolumn{1}{c|}{0/6} & \multicolumn{1}{c|}{1/2} \\
\hline
\textbf{8} & \multicolumn{1}{c|}{4/0} & \multicolumn{1}{c|}{1/6} & \multicolumn{1}{c|}{3/2} \\
\hline
\textbf{5} & \multicolumn{1}{c|}{2/5} & \multicolumn{1}{c|}{1/0} & \multicolumn{1}{c|}{2/0} \\
\hline
\end{tabular}
\end{center}

\noindent \ar{النتيجة:} $\boxed{524 \times 385 = 201{,}740}$

\section{\ar{الفصل السادس: في استخراج الجذور}}

\begin{arabicbox}
\begin{center}
\arLarge{فصل في استخراج الجذور}
\end{center}
\end{arabicbox}

\vspace{0.3cm}

\noindent\ar{تُعدُّ طريقة الكاشي في استخراج الجذور من أشهر أجزاء المفتاح. يقدم خوارزمية عامة لإيجاد الجذر الـ$n$ لأي عدد، وهي طريقة مكافئة لما يُعرف الآن بـ\textbf{طريقة روفيني-هورنر} (رغم أن الكاشي سبقهما بقرون).}

\begin{mathalgo}
\textbf{\ar{الطريقة العامة لاستخراج الجذر الـ$n$:}}

\ar{بعدد $N$ وعدد صحيح $n$، لإيجاد} $\sqrt[n]{N}$:
\begin{enumerate}[nosep]
  \item \ar{افصل $N$ إلى مجموعات من $n$ أرقام من العلامة العشرية.}
  \item \ar{أوجد أكبر عدد صحيح $a$ بحيث أن $a^n$ لا يتجاوز المجموعة الأولى.}
  \item \ar{اطرح $a^n$ من المجموعة الأولى، ثم انزل بالمجموعة التالية.}
  \item \ar{باستخدام التوسيع ذي الحدين لـ $(a + b)^n$، أوجد الرقم التالي $b$ بقسمة الباقي على $n \cdot a^{n-1}$.}
  \item \ar{كرر حتى الوصول إلى الدقة المطلوبة.}
\end{enumerate}
\end{mathalgo}

\vspace{0.3cm}

\noindent\ar{بالنسبة للـ \textbf{الجذر التربيعي} $(n=2)$، يختزل هذا إلى:}

\begin{mathalgo}
\textbf{\ar{استخراج الجذر التربيعي:}}
\begin{enumerate}[nosep]
  \item \ar{افصل العدد إلى أزواج من الأرقام من العلامة العشرية.}
  \item \ar{أوجد أكبر مربع لا يتجاوز الزوج الأيسر؛ هذا يعطي الرقم الأول من الجذر.}
  \item \ar{اطرح مربعه وانزل بالزوج التالي.}
  \item \ar{اضعف الجذر الحالي، وأوجد رقماً بحيث أن $(20 \times \text{الحالي} + d) \times d$ لا يتجاوز الباقي.}
  \item \ar{كرر حتى معالجة جميع الأزواج.}
\end{enumerate}
\end{mathalgo}

\vspace{0.3cm}

\noindent\textbf{\ar{مثال:}} $\sqrt{144} = 12$
\begin{align*}
1^2 &= 1 \quad \text{(\ar{الرقم الأول})} \\
(20 \times 1 + 2) \times 2 &= 44 \\
44 &= 44 \quad \checkmark \quad \text{(\ar{الرقم الثاني})}
\end{align*}

\chapter{\ar{المقالة الثانية: حساب الكسور}}
\label{ch:treatise2}

\begin{arabicbox}
\begin{center}
\arHuge{المقالة الثانية في حساب الكسور}
\end{center}
\end{arabicbox}

\vspace{0.5cm}

\noindent\ar{هذه المقالة هي الأكثر أهمية تاريخياً في المفتاح، فهي تحتوي على \textbf{أول معالجة منظمة للكسور العشرية} في تاريخ الرياضيات. كما لاحظ يوشكيفتش وروزنفيلد: ``إننا نرى في عمل غياث الدين جمشيد الكاشي في أوائل القرن الخامس عشر، لأول مرة، السيطرة الكاملة على فكرة الكسور العشرية وطريقة ملائمة لكتابتها.''}

\section{\ar{الفصول العشرة للمقالة الثانية}}

\begin{enumerate}[label=\textbf{\ar{فصل} \arabic*:}]
  \item \ar{في تسمية الكسور — التعريفات والمصطلحات للكسور المشتركة.}
  \item \ar{في تسطيح الكسور — التحويل إلى مقامات مشتركة.}
  \item \ar{في جمع الكسور وطرحها.}
  \item \ar{في ضرب الكسور.}
  \item \ar{في قسمة الكسور.}
  \item \ar{في التحويل بين الكسور — بما في ذلك الستيني إلى العشري والعكس.}
  \item \ar{في استخراج جذور الكسور.}
  \item \ar{في \textit{كسر الأعداد} — صنف خاص من الكسور.}

\end{enumerate}

\]
\end{mathalgo}

\section{\ar{الفصل الخامس: في استخراج جذور الأعداد الستينية}}

\noindent\ar{يحتوي هذا الفصل على الطريقة العامة الملحوظة للكاشي في استخراج الجذور من الدرجة $n$ في النظام الستيني. أظهرت دراسة داخل (1960) أن طريقة الكاشي هي في جوهرها نفسها طريقة روفيني-هورنر.}

\begin{mathalgo}
\textbf{\ar{استخراج الجذر في النظام الستيني:}}

\ar{لإيجاد} $\sqrt[n]{N}$ \ar{في النظام الستيني:}
\begin{enumerate}[nosep]
  \item \ar{عبّر عن $N$ في الشكل الستيني:} $N = N_0; N_1, N_2, N_3, \ldots$
  \item \ar{جمّع منازل الستينية بشكل ملائم لدرجة الجذر.}
  \item \ar{طبق الطريقة ذات الحدين تكرارياً:}
    \[
    \sqrt[n]{a^n + r} \approx a + \frac{r}{n \cdot a^{n-1}}
    \]
  \item \ar{في كل خطوة، صحّح التقريب باستخدام حد الخطأ.}
  \item \ar{استمر حتى الوصول إلى الدقة المطلوبة.}
\end{enumerate}
\end{mathalgo}

\vspace{0.3cm}

\noindent\textbf{\ar{مثال — الجذر التربيعي في النظام الستيني:}}

\ar{أوجد} $\sqrt{2}$ \ar{في النظام الستيني. يعطي الكاشي:}
\[
\sqrt{2} \approx 1; 24, 51, 10, 7, 46, 6, 4, 40, \ldots
\]
\ar{وهو ما يقابل:}
\[
1 + \frac{24}{60} + \frac{51}{60^2} + \frac{10}{60^3} + \cdots \approx
1.41421356\ldots
\]

\noindent\ar{هذا التقريب لـ $\sqrt{2}$ دقيق لحدود الحساب. طريقة استخراج الجذور للكاشي كلية العموم وتعمل لأي درجة من الجذور في أي أساس للأعداد.}

\subsection{\ar{قواعد المساحة}}

\noindent\ar{يقدم الكاشي عدة طرق لحساب مساحة المثلث:}

\begin{mathalgo}
\textbf{\ar{مساحة المثلث:}}

\textbf{\ar{الطريقة الأولى — القاعدة والارتفاع:}}
\[
A = \frac{1}{2} \times \text{\ar{القاعدة}} \times \text{\ar{الارتفاع}}
\]

\vspace{0.3cm}
\textbf{\ar{الطريقة الثانية — قانون هيرون:}}
\ar{لمثلث أضلاعه $a$، $b$، $c$ ونصف محيطه} $s = \frac{a+b+c}{2}$:
\[
A = \sqrt{s(s-a)(s-b)(s-c)}
\]

\vspace{0.3cm}
\textbf{\ar{الطريقة الثالثة — ضلعان والزاوية المحصورة:}}
\[
A = \frac{1}{2} ab \sin C
\]
\end{mathalgo}

\vspace{0.3cm}

\noindent\textbf{\ar{مثال:}} \ar{أوجد مساحة مثلث أضلاعه} $13$، $14$، $15$.
\begin{align*}
s &= \frac{13 + 14 + 15}{2} = 21 \\
A &= \sqrt{21(21-13)(21-14)(21-15)} \\
&= \sqrt{21 \times 8 \times 7 \times 6} = \sqrt{7056} = 84
\end{align*}

\noindent\ar{دقة الكاشي: لحساب $2\pi$ بـ 9 منازل ستينية يعني أن الخطأ الأقصى أقل من $1/60^9 \approx 9.92 \times 10^{-17} < 10^{-16}$. أي أن الكاشي حسب $2\pi$ بدقة تصل إلى المنزلة العشرية السادسة عشرة.}

\begin{arabicbox}

\arLarge{فصل في مساحة سطح المقرنص}
\end{center}
\end{arabicbox}

\vspace{0.3cm}

\noindent\ar{\textit{المقرنص} (\ar{مقرنص}) هو عنصر معماري زخرفي في العمارة الإسلامية، يتألف من طبقات من الأجراف المشبّكة التي تخلق تأثير قرص العسل أو المعلقات على الأسقف والزوايا والسطوح الداخلية للقبب. يصف الكاشي أربعة أنواع من المقرنص ويعطي قواعد لحساب مساحة سطحها، حتى يتمكن المعماريون من تحديد كمية المواد المطلوبة.}

\begin{mathalgo}
\textbf{\ar{مساحة سطح المقرنص:}}

\ar{لمقرنص بسيط مؤلف من $n$ طبقة، كل طبقة متوسط مساحتها $a_i$:}
\[
A_{\text{\ar{الكلية}}} = \sum_{i=1}^{n} a_i \times f_i
\]
\ar{حيث $f_i$ هو معامل الشكل المعتمد على نوع خلية المقرنص.}
\end{mathalgo}

\noindent\ar{لقد دُرست هذه الفقرة على نطاق واسع من قبل مؤرخي العمارة الإسلامية. يمثل المقرنص واحداً من أكثر التطبيقات تطوراً للهندسة في الفن والعمارة الإسلامية.}

\section{\ar{الفصل الأول: الصور الست القانونية}}

\noindent\ar{يصنّف الكاشي، متبعاً التقاليد التي أسسها الخوارزمي، جميع المعادلات في ستة أنواع. باستخدام التدوين الحديث مع $x$ كمجهول و $a, b$ كمعاملات موجبة:}

\begin{center}
\begin{longtable}{|c|c|c|}
\hline
\textbf{\ar{الصورة}} & \textbf{\ar{المعادلة}} & \textbf{\ar{الاسم}} \\
\hline
\endfirsthead
\hline
\textbf{\ar{الصورة}} & \textbf{\ar{المعادلة}} & \textbf{\ar{الاسم}} \\
\hline
\endhead
1 & $ax = b$ & \ar{مبسط} \\
\hline
2 & $ax^2 = bx$ & \ar{مربعات تساوي جذوراً} \\
\hline
3 & $ax^2 = b$ & \ar{مربعات تساوي أعداداً} \\
\hline
4 & $ax^2 + bx = c$ & \ar{مربعات وجذور تساوي أعداداً} \\
\hline
5 & $ax^2 + c = bx$ & \ar{مربعات وأعداد تساوي جذوراً} \\
\hline
6 & $bx + c = ax^2$ & \ar{جذور وأعداد تساوي مربعات} \\
\hline
\end{longtable}
\end{center}

\begin{mathalgo}
\textbf{\ar{حل} $ax^2 + bx = c$ \ar{(الصورة الرابعة):}}

\ar{اقسم على} $a$: $x^2 + \frac{b}{a}x = \frac{c}{a}$

\vspace{0.3cm}
\textbf{\ar{أكمل المربع:}}
\[
x^2 + \frac{b}{a}x + \left(\frac{b}{2a}\right)^2 = \frac{c}{a} +
\left(\frac{b}{2a}\right)^2
\]

\vspace{0.3cm}
\textbf{\ar{وبالتالي:}}
\[
\left(x + \frac{b}{2a}\right)^2 = \frac{4ac + b^2}{4a^2}
\]

\vspace{0.3cm}
\textbf{\ar{الحل:}}
\[
x = \frac{-b + \sqrt{b^2 + 4ac}}{2a}
\]
\end{mathalgo}

% -----------------------------------------------------------------------
\section{\ar{الفصل الثاني: طريقة الخطأين}}

\begin{arabicbox}
\begin{center}
\arLarge{فصل في استخراج المجهول بطريقة الخطأين}
\end{center}
\end{arabicbox}

\vspace{0.3cm}

\noindent\ar{\textbf{طريقة الخطأين} (\ar{طريقة الخطأين}) هي طريقة قديمة لحل المسائل الخطية بدون جبر. يقدم الكاشي بياناً واضحاً:}

\begin{mathalgo}
\textbf{\ar{طريقة الخطأين:}}

\ar{لمسألة من الشكل} $ax + b = c$، \ar{اجعل تخمينين $g_1$ و $g_2$، تعطيان نتيجتين $f_1$ و $f_2$ مع أخطاء} $e_1 = c - f_1$ \ar{و} $e_2 = c - f_2$. \ar{فإن:}
\[
x = \frac{g_1 e_2 - g_2 e_1}{e_2 - e_1}
\]

\vspace{0.3cm}
\noindent\textbf{\ar{مثال:}} \ar{كمية بحيث إذا جمعت ثلثها مع ربعها، كانت النتيجة 14. ما هي الكمية؟}

\vspace{0.3cm}
\ar{التخمين} $g_1 = 12$: $\frac{12}{3} + \frac{12}{4} = 4 + 3 = 7$، \ar{الخطأ} $e_1 = 14 - 7 = 7$

\vspace{0.2cm}
\ar{التخمين} $g_2 = 24$: $\frac{24}{3} + \frac{24}{4} = 8 + 6 = 14$، \ar{الخطأ} $e_2 = 14 - 14 = 0$

\[
x = \frac{12 \cdot 0 - 24 \cdot 7}{0 - 7} = \frac{-168}{-7} = 24
\]
\end{mathalgo}

% -----------------------------------------------------------------------
\section{\ar{الفصل الخامس: في حل المعادلات التكعيبية}}

\noindent\ar{يقدم الكاشي طرقاً لحل المعادلات التكعيبية. للمعادلة $x^3 + px = q$، يستخدم طريقة تكرارية:}

\begin{mathalgo}
\textbf{\ar{الحل التكراري لـ} $x^3 + px = q$:}

\ar{أعد الترتيب:} $x = \sqrt[3]{q - px}$

\vspace{0.3cm}
\textbf{\ar{التكرار:}}
\begin{enumerate}[nosep]
  \item \ar{ابدأ بتخمين أولي $x_0$.}
  \item \ar{احسب} $x_{n+1} = \sqrt[3]{q - p x_n}$.
  \item \ar{كرر حتى} $|x_{n+1} - x_n| < \varepsilon$.
\end{enumerate}

\vspace{0.3cm}
\noindent\textbf{\ar{مثال:}} \ar{حل} $x^3 + 2x = 20$

\ar{ابدأ بـ} $x_0 = 2$:
\begin{align*}
x_1 &= \sqrt[3]{20 - 2 \cdot 2} = \sqrt[3]{16} \approx 2.52 \\
x_2 &= \sqrt[3]{20 - 2 \cdot 2.52} = \sqrt[3]{14.96} \approx 2.466 \\
x_3 &= \sqrt[3]{20 - 2 \cdot 2.466} = \sqrt[3]{15.068} \approx 2.472 \\
x_4 &= \sqrt[3]{20 - 2 \cdot 2.472} = \sqrt[3]{15.056} \approx 2.471
\end{align*}

\ar{الحل:} $\boxed{x \approx 2.471}$
\end{mathalgo}

\noindent\ar{هذه الطريقة التكرارية لحل المعادلات التكعيبية هي في جوهرها ما يُعرف الآن بـ\textbf{تكرار النقطة الثابتة}. استخدم الكاشي نهجاً مماثلاً في \textit{رسالة الوتر والجيب} لحساب $\sin(1^\circ)$ بدقة غير مسبوقة.}
