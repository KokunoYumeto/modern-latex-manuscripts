\section*{Translator's Preface}
\addcontentsline{toc}{section}{Translator's Preface}

This bilingual edition presents Bh\={a}skara II's \textit{B\={\i}jaga\d{n}ita} (``Seed Computation'') in parallel Sanskrit-English columns. The text is based on the Benares Sanskrit Series edition of 1927, edited by Sudh\={a}kara Dvived\={\i} with additional notes by Mural\={\i}dhara Jha.

\textbf{Key Terminology:}
\begin{itemize}[leftmargin=2em]
  \item \dev{बीज} (\textit{b\={\i}ja}) --- seed, origin; hence \textit{algebra}
  \item \dev{अव्यक्त} (\textit{avyakta}) --- unseen, unknown quantity
  \item \dev{यावत्तावत्} (\textit{y\={a}vat-t\={a}vat}) --- ``as much as,'' the unknown $x$
  \item \dev{रूप} (\textit{r\={u}pa}) --- absolute number, constant term
  \item \dev{वर्ण} (\textit{var\d{n}a}) --- colour, representing different unknowns
  \item \dev{करणी} (\textit{kara\d{n}\={\i}}) --- surd, irrational quantity
  \item \dev{समीकरण} (\textit{sam\={\i}kara\d{n}a}) --- equation (making equal)
  \item \dev{धन} (\textit{dhana}) --- positive (wealth)
  \item \dev{ऋण} (\textit{\d{r}\d{n}a}) --- negative (debt)
\end{itemize}

The left column presents the original Sanskrit in Devan\={a}gar\={\i} script. The right column provides the English translation with IAST transliteration of key terms.

उत्पादकं यत् प्रवदन्ति बुद्धेः स्थितं सुरैरपि~।\\
अस्य क्रिया तत्कथयाम्यथवा तद्वीजमष्टधा~॥\\[0.2cm]
पूर्वं प्रोक्तं व्यक्तमस्य कथये प्रज्ञा ते च मेऽव्यक्तयुक्तया~।\\

\sectionbicol{घनाणां सङ्कलनम्}{Ghan\={a}\d{n}\={a}\d{m} Sa\d{m}kalana\d{m} (Sum of Cubes)}

\vspace{0.3cm}

\subsectionbicol{घनाणां सङ्कलने करणसूत्रं बुद्धये}{S\={u}tra for Sum of Cubes}

\vspace{0.3cm}

\bicolverse{%
घनाः समं सङ्गुणिता द्वित्रिभिः समन्विताः\\
एकघनाद्विवर्जितास्ते सङ्कलिताः स्युः सदा घनाः~॥ १ ॥
}{%
\textit{ghan\={a}\d{h} sama\d{m} sa\d{n}gu\d{n}it\={a} dvitribhi\d{h} samanvit\={a}\d{h}}\\
\textit{ekaghan\={a}d vivarjit\={a}s te sa\d{m}kalit\={a}\d{h} syu\d{h} sad\={a} ghan\={a}\d{h}} --- Verse 1
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्रार्थ:} पूर्वसङ्ख्यानां घनानां योगः समानम् अस्ति यत् तेषां योगस्य वर्गः। अतः
\[1^3 + 2^3 + 3^3 + \cdots + n^3 = \left(\frac{n(n+1)}{2}\right)^2\]
\textbf{उदाहरणम्:} पञ्चानां घनानां योगः
\begin{align*}
1^3 + 2^3 + 3^3 + 4^3 + 5^3 &= \left(\frac{5 \cdot 6}{2}\right)^2 \\
&= 15^2 = 225
\end{align*}
\textbf{सत्यापनम्:} $1 + 8 + 27 + 64 + 125 = 225$
}{%
\textbf{Rule:} The sum of cubes of the first $n$ natural numbers equals the square of their sum:
\[1^3 + 2^3 + 3^3 + \cdots + n^3 = \left(\frac{n(n+1)}{2}\right)^2\]
\textbf{Example:} Sum of the first 5 cubes:
\begin{align*}
1^3 + 2^3 + 3^3 + 4^3 + 5^3 &= \left(\frac{5 \cdot 6}{2}\right)^2 \\
&= 15^2 = 225
\end{align*}
\textbf{Verification:} $1 + 8 + 27 + 64 + 125 = 225$ \checkmark
}

%% ========================================================================
%% CHAPTER II -- SUBTRACTION OF CUBES
%% ========================================================================
\sectionbicol{घनानां व्यवकलनम्}{Ghan\={a}n\={a}\d{m} Vyavakalana\d{m} (Subtraction of Cubes)}

\vspace{0.3cm}

\bicolverse{%
व्यस्तद्वयोर्घनयोर्व्यस्तयोर्घनयोर्व्यवकलनम्~।\\
समघ्नयोरपि तथा व्यवकलनं सम्भवत्येव~॥ २ ॥
}{%
\textit{vyastadvayor ghanayor vyastayor ghanayor vyavakalanam}\\
\textit{samaghnayor api tath\={a} vyavakala\d{m} sambhavaty eva} --- Verse 2
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्रार्थ:} द्वयोः घनयोः व्यवकलनं तयोः मूलयोः व्यवकलनेन सह घनेन समं भवति। यदि $a$ तथा $b$ द्वौ राशी, तर्हि
\[a^3 - b^3 = (a-b)(a^2 + ab + b^2)\]
\textbf{उदाहरणम्:}
\begin{align*}
(y\bar{a}\ 5)^3 - (y\bar{a}\ 2)^3 &= 125y\bar{a}^3 - 8y\bar{a}^3 \\
&= 117y\bar{a}^3
\end{align*}
अथवा: $(5-3)(25 + 15 + 9) = 2 \cdot 49 = 98$ (रूपेण)
}{%
\textbf{Rule:} The subtraction of cubes of two quantities equals the product of their difference and the sum of their squares plus their product:
\[a^3 - b^3 = (a-b)(a^2 + ab + b^2)\]
\textbf{Example:}
\begin{align*}
(5y\bar{a})^3 - (2y\bar{a})^3 &= 125y\bar{a}^3 - 8y\bar{a}^3 \\
&= 117y\bar{a}^3
\end{align*}
Or: $(5-3)(25 + 15 + 9) = 2 \cdot 49 = 98$ (in constants)
}

\sectionbicol{घनानां गुणनम्}{Ghan\={a}n\={a}\d{m} Gu\d{n}anam (Multiplication of Cubes)}

\vspace{0.3cm}

\bicolverse{%
घनयोर्घनयोर्घनयोर्वा गुणनं सम्भवति~।\\
घनघातः सम्भवत्येवं गुणकारघनयोः समं~॥ ३ ॥
}{%
\textit{ghanayor ghanayor ghanayor v\={a} gu\d{n}ana\d{m} sambhavati}\\
\textit{ghanagh\={a}ta\d{h} sambhavaty eva\d{m} gu\d{n}ak\={a}raghanayo\d{h} sama\d{m}} --- Verse 3
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्रार्थ:} द्वयोः घनयोः गुणनफलं तयोः मूलयोः गुणनस्य घनत्वेन समं भवति।
\[a^3 \times b^3 = (ab)^3\]
\textbf{उदाहरणम्:}
\begin{align*}
(3y\bar{a})^3 \times (2y\bar{a})^3 &= (6y\bar{a})^3 \\
&= 216y\bar{a}^3
\end{align*}
}{%
\textbf{Rule:} The product of two cubes equals the cube of the product of their roots:
\[a^3 \times b^3 = (ab)^3\]
\textbf{Example:}
\begin{align*}
(3y\bar{a})^3 \times (2y\bar{a})^3 &= (6y\bar{a})^3 \\
&= 216y\bar{a}^3
\end{align*}
}

\sectionbicol{घनानां भागहारः}{Ghan\={a}n\={a}\d{m} Bh\={a}gah\={a}ra\d{h} (Division of Cubes)}

\vspace{0.3cm}

\bicolverse{%
घनं घनेन विभजेद् घनमूलयोर्मूलयोरिव~।\\
घनभागः स्तः सम्भवत्येवं भागकारयोः समं~॥ ४ ॥
}{%
\textit{ghana\d{m} ghanena vibhajed ghanam\={u}layor m\={u}layor iva}\\
\textit{ghanabh\={a}ga\d{h} stha\d{h} sambhavaty eva\d{m} bh\={a}gak\={a}rayo\d{h} sama\d{m}} --- Verse 4
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्रार्थ:} घनं घनेन विभज्य लब्धिः मूलयोः भागस्य घनत्वेन समं भवति।
\[\frac{a^3}{b^3} = \left(\frac{a}{b}\right)^3\]
\textbf{उदाहरणम्:}
\begin{align*}
\frac{(8y\bar{a})^3}{(2y\bar{a})^3} &= \left(\frac{8y\bar{a}}{2y\bar{a}}\right)^3 \\
&= (4y\bar{a})^3 = 64y\bar{a}^3
\end{align*}
}{%
\textbf{Rule:} A cube divided by a cube; the quotient is the cube of the quotient of their roots:
\[\frac{a^3}{b^3} = \left(\frac{a}{b}\right)^3\]
\textbf{Example:}
\begin{align*}
\frac{(8y\bar{a})^3}{(2y\bar{a})^3} &= \left(\frac{8y\bar{a}}{2y\bar{a}}\right)^3 \\
&= (4y\bar{a})^3 = 64y\bar{a}^3
\end{align*}
}

\clearpage

%% ========================================================================
\sectionbicol{घनानां वर्गो मूलं च}{Ghan\={a}n\={a}\d{m} Vargo M\={u}la\d{m} ca (Square and Root of Cubes)}

\vspace{0.3cm}

\bicolverse{%
घनस्य वर्गः स्यात् षट्ठी घनस्य मूलं तु मूलघनम्~।\\
वर्गमूलं घनमूलं च कल्पनीयं ततस्ततः~॥ ५ ॥
}{%
\textit{ghanasya varga\d{h} sy\={a}t \d{s}a\d{s}\d{t}h\={\i} ghanasya m\={u}la\d{m} tu m\={u}laghanam}\\
\textit{vargam\={u}la\d{m} ghanam\={u}la\d{m} ca kalpan\={\i}ya\d{m} tatas tata\d{h}} --- Verse 5
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्रार्थ:} घनस्य वर्गः षष्ठी घातः। घनस्य मूलं तु मूलस्य घनः।
\begin{align*}
(a^3)^2 &= a^6 \quad \text{(घनवर्गः = षट्ठी घातः)} \\
\sqrt[3]{a^3} &= a \quad \text{(घनमूलम्)} \\
\sqrt{a^3} &= a^{3/2} \quad \text{(वर्गमूलं घनस्य)}
\end{align*}
}{%
\textbf{Rule:} The square of a cube is the sixth power. The root of a cube is the cube-root.
\begin{align*}
(a^3)^2 &= a^6 \quad \text{(square of cube = 6th power)} \\
\sqrt[3]{a^3} &= a \quad \text{(cube root)} \\
\sqrt{a^3} &= a^{3/2} \quad \text{(square root of cube)}
\end{align*}
}

%% CHAPTER VI -- ADDITION/SUBTRACTION OF UNKNOWNS
%% ========================================================================
\sectionbicol{अव्यक्तसङ्कलनव्यवकलनम्}{Avyakta-sa\d{m}kalana-vyavakalanam (Addition/Subtraction of Unknowns)}

\vspace{0.3cm}

\bicolverse{%
(१) यावत्तावत् कालक नीलक पीतक हरितकादयः~।\\
वर्णाः पीतो हरितश्चेत् द्वयं च अव्यक्तानां कल्पना मतसंज्ञा—\\
राशिसंख्यानां कल्पिता मताचार्येण~॥ ६ ॥
}{%
(1) \textit{y\={a}vat-t\={a}vat k\={a}laka n\={\i}laka p\={\i}taka haritak\={a}daya\d{h}}\\
\textit{var\d{n}\={a}h} --- The unknown quantities are represented by colours:
\textit{y\={a}vat-t\={a}vat} ($x$), \textit{k\={a}laka} ($y$), \textit{n\={\i}laka} ($z$), \textit{p\={\i}taka} ($u$), \textit{haritaka} ($v$), etc.
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र (१):} अज्ञातराशीनाम् प्रतिनिधित्वाय वर्णाः कल्पिताः। यावत्तावत् (= $x$), कालक (= $y$), नीलक (= $z$), पीतक (= $u$), हरितक (= $v$) इत्यादयः।
\textbf{सूत्र (२):} समानजातीयानां राशीनां योगः व्यवकलनं वा क्रियते। भिन्नजातीयाः पृथक् स्थापनीयाः।
\begin{align*}
(y\bar{a}\ 1 + k\bar{a}\ 1) + (y\bar{a}\ 2 + k\bar{a}\ 8) &= y\bar{a}\ 3 + k\bar{a}\ 9 \\
(y\bar{a}\ 5 + k\bar{a}\ 3) - (y\bar{a}\ 2 + k\bar{a}\ 8) &= y\bar{a}\ 3 + k\bar{a}\ \bar{5}
\end{align*}
}{%
\textbf{Rule (1):} Unknown quantities are represented by colours: \textit{y\={a}vat-t\={a}vat} ($x$), \textit{k\={a}laka} ($y$), \textit{n\={\i}laka} ($z$), \textit{p\={\i}taka} ($u$), \textit{haritaka} ($v$).
\textbf{Rule (2):} Like terms are combined; unlike terms are set down separately.
\begin{align*}
(y\bar{a}\ 1 + k\bar{a}\ 1) + (y\bar{a}\ 2 + k\bar{a}\ 8) &= y\bar{a}\ 3 + k\bar{a}\ 9 \\
(y\bar{a}\ 5 + k\bar{a}\ 3) - (y\bar{a}\ 2 + k\bar{a}\ 8) &= y\bar{a}\ 3 + k\bar{a}\ \bar{5}
\end{align*}
}

\vspace{0.5cm}

\bicol{%
\footnotesize\textbf{टीका:} ``यावत्तावत् साकल्येन मनस्यवस्था'' इत्यमरकोषोक्तेः `यावत्तावत्' इति भवितुमर्हति किंत्वद्यपदम्। बीजगणिते च ``यावत्तावद्विनिम्मे'' इत्याख्यासम्भवम्।
}{%
\footnotesize\textbf{Note:} From the Amarakosha: ``\textit{y\={a}vat-t\={a}vat s\={a}kalyena manasyavasth\={a}}'' --- the term \textit{y\={a}vat-t\={a}vat} means ``as much as,'' representing the state of the mind regarding an unknown quantity.
}

\clearpage

%% ========================================================================
\sectionbicol{अव्यक्तगुणनम्}{Avyakta-gu\d{n}anam (Multiplication of Unknowns)}

\vspace{0.3cm}

\subsectionbicol{अव्यक्तवर्गविधम्}{Avyakta-varga-vidham}

\vspace{0.3cm}

\bicolverse{%
अव्यक्ताकृतयोः समस्य हेतोः समेत्य वर्गद्वयं कल्पयेत्~।\\
तदा तयोर्वर्गयोर्वर्गः सम्भवत्येवमेकस्य वर्गः~॥ ७ ॥
}{%
\textit{avyakt\={a}k\d{r}tayo\d{h} samasya heto\d{h} sametya vargadvaya\d{m} kalpayet}\\
\textit{tad\={a} tayor vargayor varga\d{h} sambhavaty ekasya varga\d{h}} --- Verse 7
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} अव्यक्ताकृतयोः गुणने नवः जातिभेदः सम्भवति। अव्यक्तस्य वर्गः \dev{वर्गः} इति, ततो घनः, वर्गवर्गः इत्यादि।
\textbf{उदाहरणम्:}
\begin{align*}
(y\bar{a}\ 5 + k\bar{a}\ 3) \times (y\bar{a}\ 3 + k\bar{a}\ 2) &= y\bar{a}v\ 15 + y\bar{a}\ 19 + k\bar{a}\ 6
\end{align*}
\textbf{गुण्यस्य धनर्णव्यत्यासे:} $y\bar{a}\ 5\ k\bar{a}\ \bar{3}$ गुणितं $y\bar{a}\ 3\ k\bar{a}\ 2$ सह = $y\bar{a}v\ 15 + y\bar{a}\ \bar{1} + k\bar{a}\ \bar{6}$
}{%
\textbf{Rule:} When two unknowns are multiplied, a new species is produced. The square of an unknown is \textit{varga}; higher powers follow.
\textbf{Example:}
\begin{align*}
(5y\bar{a} + 3k\bar{a}) \times (3y\bar{a} + 2k\bar{a}) &= 15y\bar{a}^2 + 19y\bar{a} + 6k\bar{a}
\end{align*}
\textbf{With sign change of multiplicand:} $(5y\bar{a} - 3k\bar{a}) \times (3y\bar{a} + 2k\bar{a})$ $= 15y\bar{a}^2 + y\bar{a} - 6k\bar{a}$
}

\clearpage

%% ========================================================================
\sectionbicol{अव्यक्तभागहारः}{Avyakta-bh\={a}gah\={a}ra\d{h} (Division of Unknowns)}

\vspace{0.3cm}

\bicolverse{%
भागाहरे करणसूत्रं बुद्धये~।\\
भाज्यच्छेदः सृजति प्रच्युतः सन् क्षेपे क्षेपे स्थानक्षेपं क्षेपेण~।\\
दैन्यैर्द्विजैः सङ्गुणो यैः क्षेपैर्भागाहरे लब्धयस्तः स्पृशेत्~
}{%
\textit{bh\={a}g\={a}hare kara\d{n}as\={u}tra\d{m} buddhaye}\\
\textit{bh\={a}jyaccheda\d{h} s\d{r}jati pracyuta\d{h} san k\d{s}epe k\d{s}epe sth\={a}nak\d{s}epa\d{m} k\d{s}epe\d{n}a}\\
--- The division of unknown quantities follows the method for known quantities.
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} अव्यक्तानां भागहारः व्यक्तानामिव एव। यदि भाजकः भाज्यं न सम्पूर्णं विभजेत्, तर्हि भिन्नं सम्भवति।
\textbf{उदाहरणम्:} $y\bar{a}\ 18\ k\bar{a}\ 24$ विभज्य $k\bar{a}\ 6\ y\bar{a}\ 5$ इति
\begin{align*}
\text{भाज्य: }& y\bar{a}v\ 10 + y\bar{a}\ 27 + y\bar{a}\ 18 + k\bar{a}\ 24 \\
\text{भाजक: }& y\bar{a}\ 5 + k\bar{a}\ 6 \\
\text{लब्धि: }& y\bar{a}\ 2 + y\bar{a}\ 3 + k\bar{a}\ 4
\end{align*}
}{%
\textbf{Rule:} Division of unknowns is the same as for knowns. If division is not exact, a fraction results.
\textbf{Example:} Divide $18y\bar{a} + 24k\bar{a}$ by $5y\bar{a} + 6k\bar{a}$:
\begin{align*}
\text{Dividend: }& 10y\bar{a}^2 + 27y\bar{a} + 18y\bar{a} + 24k\bar{a} \\
\text{Divisor: }& 5y\bar{a} + 6k\bar{a} \\
\text{Quotient: }& 2y\bar{a} + 3y\bar{a} + 4k\bar{a}
\end{align*}
}

\sectionbicol{करणीसङ्कलनव्यवकलनम्}{Kara\d{n}\={\i}-sa\d{m}kalana-vyavakalanam (Addition/Subtraction of Surds)}

\vspace{0.3cm}

\bicolverse{%
करणीसंयुक्तिरेभिः सङ्कलिता स्यात् समं करणीभिः~।\\
करणीभिर्विशिष्टाः पृथक् स्थापनीयाः समं करणीभिः~
}{%
\textit{kara\d{n}\={\i}sa\d{m}yuktirebhi\d{h} sa\d{m}kalit\={a} sy\={a}t sama\d{m} kara\d{n}\={\i}bhi\d{h}}\\
\textit{kara\d{n}\={\i}bhir vi\d{s}i\d{s}\d{t}\={a}\d{h} p\d{r}thak sth\={a}pan\={\i}y\={a}\d{h} sama\d{m} kara\d{n}\={\i}bhi\d{h}}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} समानकरणिकाः करण्यः संकलिताः स्युः। भिन्नकरणिकाः पृथक् स्थापनीयाः।
\vspace{0.3cm}
\textbf{उदाहरणम् १:} $\sqrt{3} + \sqrt{3} + 2\sqrt{3} + 5\sqrt{3}$ योजय।
\[\text{योग: } (1+1+2+5)\sqrt{3} = 9\sqrt{3}\]
\textbf{उदाहरणम् २:} $\sqrt{2} + \sqrt{3} + \sqrt{5}$ योजय।
\[\text{फलम्: } \sqrt{2} + \sqrt{3} + \sqrt{5} \quad \text{(पृथक् स्थापितम्)}\]
}{%
\textbf{Rule:} Surds having the same radical are added together; those with different radicals are set down separately.
\vspace{0.3cm}
\textbf{Example 1:} Add $\sqrt{3} + \sqrt{3} + 2\sqrt{3} + 5\sqrt{3}$.
\[\text{Sum: } (1+1+2+5)\sqrt{3} = 9\sqrt{3}\]
\textbf{Example 2:} Add $\sqrt{2} + \sqrt{3} + \sqrt{5}$.
\[\text{Result: } \sqrt{2} + \sqrt{3} + \sqrt{5} \quad \text{(set down separately)}\]
}

\clearpage

%% CHAPTER XII -- MULTIPLICATION OF SURDS
%% ========================================================================
\sectionbicol{करणीगुणनम्}{Kara\d{n}\={\i}-gu\d{n}anam (Multiplication of Surds)}

\vspace{0.3cm}

\bicolverse{%
करणीगुणने करणीगुणने समे समे विपरे विपरे~।\\
गुण्यकरणीगुणककरणी गुणनफलं करणी भवति~
}{%
\textit{kara\d{n}\={\i}gu\d{n}ane kara\d{n}\={\i}gu\d{n}ane same same vipare vipare}\\
\textit{gu\d{n}yakara\d{n}\={\i}gu\d{n}akakara\d{n}\={\i} gu\d{n}anaphala\d{m} kara\d{n}\={\i} bhavati}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} करण्योः गुणने गुण्यकरणी-गुणककरण्योः पदयोः गुणनं नवकरणीं जनयति। गुणकोत्तरयोः गुणनं नवगुणकः।
\vspace{0.3cm}
\textbf{उदाहरणम्:} $3\sqrt{2}$ गुणितं $4\sqrt{3}$
\[\text{फलम्: } (3 \times 4) \times \sqrt{2 \times 3} = 12\sqrt{6}\]
\textbf{सूत्रत:}
\[a\sqrt{b} \times c\sqrt{d} = (ac)\sqrt{bd}\]
}{%
\textbf{Rule:} In multiplication of surds, multiply the radical quantities to obtain the new radical; multiply the coefficients for the new coefficient.
\vspace{0.3cm}
\textbf{Example:} Multiply $3\sqrt{2}$ by $4\sqrt{3}$.
\[\text{Product: } (3 \times 4) \times \sqrt{2 \times 3} = 12\sqrt{6}\]
\textbf{In general:}
\[a\sqrt{b} \times c\sqrt{d} = (ac)\sqrt{bd}\]
}

%% ========================================================================
%% CHAPTER XIII -- DIVISION OF SURDS
%% ========================================================================
\sectionbicol{करणीभजनम्}{Kara\d{n}\={\i}-bhajanam (Division of Surds)}

\vspace{0.3cm}

\bicolverse{%
भजनं करण्योर्भजनं करण्योर्वर्गवत् संस्कारः~।\\
भाज्यकरणीभाजककरणीभागो यथोक्तक्रमेण~
}{%
\textit{bhajana\d{m} kara\d{n}yor bhajana\d{m} kara\d{n}yor vargavat sa\d{msk\={a}ra\d{h}}}\\
\textit{bh\={a}jyakara\d{n}\={\i}bh\={a}jakakara\d{n}\={\i}bh\={a}go yathoktakrame\d{n}a}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} करण्योः भजने संस्कारः वर्गवत्। भाज्यकरणीम् भाजककरण्या विभज्य लब्धिः करणीरूपेण भवति।
\vspace{0.3cm}
\textbf{उदाहरणम्:} $12\sqrt{6}$ विभज्य $3\sqrt{2}$
\[\text{भाग: } \frac{12}{3} \times \sqrt{\frac{6}{2}} = 4\sqrt{3}\]
\textbf{सूत्रत:}
\[\frac{a\sqrt{b}}{c\sqrt{d}} = \frac{a}{c}\sqrt{\frac{b}{d}}\]
}{%
\textbf{Rule:} In division of surds, the operation is like that for squares. Divide the radical of the dividend by the radical of the divisor, and the coefficient of the dividend by the coefficient of the divisor.
\vspace{0.3cm}
\textbf{Example:} Divide $12\sqrt{6}$ by $3\sqrt{2}$.
\[\text{Quotient: } \frac{12}{3} \times \sqrt{\frac{6}{2}} = 4\sqrt{3}\]
\textbf{In general:}
\[\frac{a\sqrt{b}}{c\sqrt{d}} = \frac{a}{c}\sqrt{\frac{b}{d}}\]
}

\clearpage

%% CHAPTER XIV -- SQUARE OF SURDS
%% ========================================================================
\sectionbicol{करणीवर्गः}{Kara\d{n}\={\i}-varga\d{h} (Square of Surds)}

\vspace{0.3cm}

\bicolverse{%
वर्गः करण्याः स्यात् पदस्य वर्गः करणी चेति~।\\
वर्गः करण्याः स्यात् पदस्य वर्गः करणी च भवति~
}{%
\textit{varga\d{h} kara\d{n}y\={a}\d{h} sy\={a}t padasya varga\d{h} kara\d{n}\={\i} ceti}\\
\textit{varga\d{h} kara\d{n}y\={a}\d{h} sy\={a}t padasya varga\d{h} kara\d{n}\={\i} ca bhavati}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} करण्याः वर्गः = गुणकवर्गः $\times$ पदम् (करणी ततः निर्गच्छति)।
\vspace{0.3cm}
\textbf{उदाहरणम्:} $3\sqrt{5}$ इत्यस्य वर्गः
\[\text{वर्ग: } (3)^2 \times 5 = 9 \times 5 = 45\]
यतः $(3\sqrt{5})^2 = 3^2 \times (\sqrt{5})^2 = 9 \times 5 = 45$
\vspace{0.3cm}
\textbf{सूत्रत:}
\[(a\sqrt{b})^2 = a^2 \cdot b\]
}{%
\textbf{Rule:} The square of a surd equals the square of the coefficient times the radical quantity (the radical sign vanishes).
\vspace{0.3cm}
\textbf{Example:} Square of $3\sqrt{5}$.
\[\text{Square: } (3)^2 \times 5 = 9 \times 5 = 45\]
Since $(3\sqrt{5})^2 = 3^2 \times (\sqrt{5})^2 = 9 \times 5 = 45$
\vspace{0.3cm}
\textbf{In general:}
\[(a\sqrt{b})^2 = a^2 \cdot b\]
}

\sectionbicol{करणीमूलम्}{Kara\d{n}\={\i}-m\={u}lam (Square Root of Surds)}

\vspace{0.3cm}

\bicolverse{%
(१) वर्गं करण्या यदि वा करणीस्तु तयो रूपपक्षं वा बहूनाम्~।\\
विशोध्येष्टं पदं ते शेषस्य रूपाणि युतानि निपातयेत्~
}{%
(1) \textit{varga\d{m} kara\d{n}y\={a} yadi v\={a} kara\d{n}\={\i}stu tayo r\={u}papak\d{s}a\d{m} v\={a} bah\={u}n\={a}m}\\
\textit{vi\'{s}odhye\d{s}\d{t}am pada\d{m} te \'{s}e\d{s}asya r\={u}p\={a}\d{n}i yut\={a}ni nip\={a}tayet}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} करणीविशिष्टस्य राशेः वर्गमूलं ज्ञातुं यदि $\sqrt{a + \sqrt{b}} = \sqrt{x} + \sqrt{y}$, तर्हि
\begin{align*}
x + y &= a \\
4xy &= b
\end{align*}
$\textit{atha } x - y = \sqrt{a^2 - b}$ इति। अतः
\begin{align*}
x &= \frac{a + \sqrt{a^2 - b}}{2} \\
y &= \frac{a - \sqrt{a^2 - b}}{2}
\end{align*}
}{%
\textbf{Rule:} To find $\sqrt{a + \sqrt{b}}$, assume $\sqrt{a + \sqrt{b}} = \sqrt{x} + \sqrt{y}$. Then:
\begin{align*}
x + y &= a \\
4xy &= b
\end{align*}
Also $x - y = \sqrt{a^2 - b}$. Therefore:
\begin{align*}
x &= \frac{a + \sqrt{a^2 - b}}{2} \\
y &= \frac{a - \sqrt{a^2 - b}}{2}
\end{align*}
}

%% ========================================================================
\sectionbicol{कुट्टकः}{Ku\d{t}\d{t}aka\d{h} (The Pulverizer)}

\vspace{0.3cm}

\bicolverse{%
भाज्यं हरेण विभजेद् भाजकेनापवर्त्य तौ~।\\
यावत्तावत्प्रचलितं कुट्टकं तत्र योजयेत्~
}{%
\textit{bh\={a}jya\d{m} hare\d{n}a vibhajed bh\={a}jaken\={a}pavartya tau}\\
\textit{y\={a}vatt\={a}vatpracalita\d{m} ku\d{t}\d{t}aka\d{m} tatra yojayet}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} कुट्टकः रैखिकअनिर्णीतसमीकरणस्य $ax \pm c = by$ इत्यस्य विधिः, यत्र $a$, $b$, $c$ ज्ञातसङ्ख्याः, $x$, $y$ ज्ञेयसङ्ख्याः।
\vspace{0.3cm}
\textbf{विधानम्:}
\begin{enumerate}[leftmargin=1.5em,topsep=0pt]
\item भाज्यं ($a$) भाजकेन ($b$) विभज्य लब्धिः तथा शेषं ज्ञातव्यम्
\item परस्परभागहारः (निरन्तरभागहार: continued fractions)
\item \dev{मतित:} (बुद्धिपूर्वक) क्रिया
\item एकस्य विशिष्टोत्तरात् सामान्योत्तरं प्राप्तुम्
\end{enumerate}
\textbf{उदाहरणम्:} $100x + 90 = 63y$
\begin{align*}
x &= 18 + 63t \\
y &= 30 + 100t
\end{align*}
}{%
\textbf{Rule:} The Pulverizer (\textit{ku\d{t}\d{t}aka}) solves linear indeterminate equations $ax \pm c = by$, where $a$, $b$, $c$ are known integers, and $x$, $y$ are unknown integers.
\vspace{0.3cm}
\textbf{Method:}
\begin{enumerate}[leftmargin=1.5em,topsep=0pt]
\item Divide dividend ($a$) by divisor ($b$) to get quotient and remainder
\item Mutual division (continued fractions)
\item \textit{Matita\d{h}} (intelligent) operation
\item Derive general solution from a particular one
\end{enumerate}
\textbf{Example:} $100x + 90 = 63y$
\begin{align*}
x &= 18 + 63t \\
y &= 30 + 100t
\end{align*}
}

%% CHAPTER XVIII -- CYCLIC METHOD
%% ========================================================================
\sectionbicol{चक्रवालम्}{Cakrav\={a}lam (The Cyclic Method)}

\vspace{0.3cm}

\bicol{%
\textbf{चक्रवालविधिः} भास्करस्य बीजगणिते अत्यन्तं महत्त्वपूर्णं योगदानम् अस्ति। एतत् सर्वथा न्यूनोत्तरं प्राप्नोति।
\vspace{0.3cm}
\textbf{चरण १:} $a$, $b$, $k$ चिनोतु यत् $Na^2 + k = b^2$
\textbf{चरण २:} $m$ ज्ञायतां यत् $a \cdot m + b$ विभाज्यते $k$ अनया, तथा $|m^2 - N|$ न्यूनतमम्।
\textbf{चरण ३:}
\begin{align*}
a' &= \frac{am + b}{k} \\
b' &= \frac{bm + Na}{k} \\
k' &= \frac{m^2 - N}{k}
\end{align*}
\textbf{चरण ४:} यदि $k' = 1$, तर्हि $(a', b')$ उत्तरम्। अन्यथा पुनरावृत्तिः।
}{%
\textbf{The Cyclic Method} is Bh\={a}skara's most important contribution to algebra. It always converges to the minimal solution.
\vspace{0.3cm}
\textbf{Step 1:} Choose $a$, $b$, $k$ such that $Na^2 + k = b^2$
\textbf{Step 2:} Find $m$ such that $am + b$ is divisible by $k$, and $|m^2 - N|$ is minimized.
\textbf{Step 3:}
\begin{align*}
a' &= \frac{am + b}{k} \\
b' &= \frac{bm + Na}{k} \\
k' &= \frac{m^2 - N}{k}
\end{align*}
\textbf{Step 4:} If $k' = 1$, then $(a', b')$ is the solution. Otherwise repeat.
}

\sectionbicol{एकवर्णसमीकरणबीजम्}{Ekavar\d{n}a-sam\={\i}kara\d{n}a-b\={\i}jam (Seed of Single-Colour Equations)}

\vspace{0.3cm}

\bicolverse{%
रूपाणि यत्र स्युः समतुल्यानि तत्र यावत् तुल्यानि~।\\
यावन्ति तानि तानि विपरीतानि तुल्यानि तु तुल्यानि~
}{%
\textit{r\={u}p\={a}\d{n}i yatra syu\d{h} samatuly\={a}ni tatra y\={a}vat tuly\={a}ni}\\
\textit{y\={a}vanti t\={a}ni t\={a}ni vipar\={\i}t\={a}ni tuly\={a}ni tu tuly\={a}ni}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} एकवर्णसमीकरणे \dev{संरमण} (पार्श्वान्तरम्) द्वारा अज्ञातराशीनाम् समीकरणं क्रियते।
\begin{enumerate}[leftmargin=1.5em,topsep=0pt]
\item न्यूनपक्षात् अज्ञातांशं महतः पक्षात् अपनयतु
\item महतः पक्षात् रूपाणि न्यूनात् पक्षात् अपनयतु
\item रूपशेषं अज्ञातशेषेन विभजतु
\end{enumerate}
\textbf{उदाहरणम्:} $4y\bar{a} + 7 = 2y\bar{a} + 13$
\begin{align*}
4y\bar{a} - 2y\bar{a} &= 13 - 7 \\
2y\bar{a} &= 6 \\
y\bar{a} &= 3
\end{align*}
\textbf{सत्यापनम्:} $4(3) + 7 = 19$ तथा $2(3) + 13 = 19$ \checkmark
}{%
\textbf{Rule:} In a single-coloured equation, equalize unknowns by transposition (\textit{sa\d{m}rama\d{n}a}):
\begin{enumerate}[leftmargin=1.5em,topsep=0pt]
\item Subtract unknown on smaller side from unknown on greater side
\item Subtract known on greater side from known on smaller side
\item Divide the remainder of known by remainder of unknown
\end{enumerate}
\textbf{Example:} $4y\bar{a} + 7 = 2y\bar{a} + 13$
\begin{align*}
4y\bar{a} - 2y\bar{a} &= 13 - 7 \\
2y\bar{a} &= 6 \\
y\bar{a} &= 3
\end{align*}
\textbf{Check:} $4(3) + 7 = 19$ and $2(3) + 13 = 19$ \checkmark
}

\textbf{उदाहरणानि:}

\textbf{Examples:}

\vspace{0.2cm}
\textbf{(2)} $5y\bar{a} - 8 = 3y\bar{a} + 10$

5y\bar{a} - 3y\bar{a} &= 10 + 8 \\
2y\bar{a} &= 18 \Rightarrow y\bar{a} = 9

\sectionbicol{असकृत्वर्गादिसमीकरणम्}{Asak\d{r}t-varg\={a}di-sam\={\i}kara\d{n}am (Repeated Square Equations)}

\vspace{0.3cm}

\bicolverse{%
वर्गो घनो वर्गवर्गोऽष्टघातो नवमो वर्गघनश्चेति~।\\
समीकरणं समं समं कुर्यात् समतुल्यं तु तुल्यम्~
}{%
\textit{vargo ghano vargavargo '\d{s}\d{t}agh\={a}to navamo vargaghana\'{s} ceti}\\
\textit{sam\={\i}kara\d{n}a\d{m} sama\d{m} sama\d{m} kury\={a}t samatulya\d{m} tu tulyam}

\bicol{%
\textbf{सूत्र:} वर्गघनादिक्रमसमीकरणानि संरमणेन समीकृत्य मूलद्वारा वा अपवर्तनद्वारा वा सोद्धरणानि।
\vspace{0.3cm}
\textbf{(१) द्विघातसमीकरणम्:} $y\bar{a}^2 - 5y\bar{a} + 6 = 0$
\begin{align*}
(y\bar{a} - 2)(y\bar{a} - 3) &= 0
\end{align*}
$\therefore y\bar{a} = 2$ अथवा $y\bar{a} = 3$
\vspace{0.3cm}
\textbf{(२) द्विघातसमीकरणम्:} $2y\bar{a}^2 + 7y\bar{a} - 15 = 0$
\[y\bar{a} = \frac{-7 \pm \sqrt{49 + 120}}{4} = \frac{-7 \pm 13}{4}\]
$\therefore y\bar{a} = \frac{3}{2}$ अथवा $y\bar{a} = -5$
}{%
\textbf{Rule:} For equations with squares, cubes, and higher powers, equalize by transposition, then factor or extract roots.
\vspace{0.3cm}
\textbf{(1) Quadratic:} $y\bar{a}^2 - 5y\bar{a} + 6 = 0$
\begin{align*}
(y\bar{a} - 2)(y\bar{a} - 3) &= 0
\end{align*}
$\therefore y\bar{a} = 2$ or $y\bar{a} = 3$
\vspace{0.3cm}
\textbf{(2) Quadratic:} $2y\bar{a}^2 + 7y\bar{a} - 15 = 0$
\[y\bar{a} = \frac{-7 \pm \sqrt{49 + 120}}{4} = \frac{-7 \pm 13}{4}\]
$\therefore y\bar{a} = \frac{3}{2}$ or $y\bar{a} = -5$
}

\sectionbicol{अनेकवर्णमध्यमाहरणम्}{Aneka-var\d{n}a-madhyam\={a}hara\d{n}am (Elimination of Many Unknowns)}

\vspace{0.3cm}

\bicolverse{%
मध्यमाहरणं तेषां यथा स्थानं प्रयोजयेत्~।\\
तुल्यानां तुल्यतां नीत्वा मध्यमं हरेद् बुधः~
}{%
\textit{madhyam\={a}hara\d{n}a\d{m} te\d{s}\={a}\d{m} yath\={a} sth\={a}na\d{m} prayojayet}\\

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} मध्यमाहरणम् एकस्य वर्णस्य गुणकानां समीकरणं कृत्वा व्यवकलनेन क्रियते।
\vspace{0.3cm}
\textbf{उदाहरणम्:}
\begin{align*}
2x + 3y + 6 &= 0 \quad \text{(१)} \\
3x + 2y + 7 &= 0 \quad \text{(२)} \\
4x + 5y + 9 &= 0 \quad \text{(३)}
\end{align*}
(१) तथा (२) योजनेन:
\begin{align*}
6x + 9y + 18 &= 0 \\
6x + 4y + 14 &= 0 \\
5y + 4 &= 0 \Rightarrow y = -\tfrac{4}{5}
\end{align*}
प्रतिस्थापनेन: $x = -\tfrac{9}{5}$
}{%
\textbf{Rule:} Elimination of the middle (\textit{madhyam\={a}hara\d{n}a}) is performed by equalizing coefficients of one unknown and subtracting.
\vspace{0.3cm}
\textbf{Example:}
\begin{align*}
2x + 3y + 6 &= 0 \quad \text{(1)} \\
3x + 2y + 7 &= 0 \quad \text{(2)} \\
4x + 5y + 9 &= 0 \quad \text{(3)}
\end{align*}
From (1) and (2):
\begin{align*}
6x + 9y + 18 &= 0 \\
6x + 4y + 14 &= 0 \\
5y + 4 &= 0 \Rightarrow y = -\tfrac{4}{5}
\end{align*}
Substituting: $x = -\tfrac{9}{5}$
}

\clearpage

%% ========================================================================
%% CHAPTER XXIII -- BHAVITA (PRODUCT) EQUATIONS
%% ========================================================================
\sectionbicol{भावितम्}{Bh\={a}vitam (The Product)}

\vspace{0.3cm}

\bicolverse{%
भावितं भावितं यत्र द्वयोर्वर्णयोर्गुणः~।\\
स्थाप्यते समतुल्यं तद् भावितं प्रचक्षते~
}{%
\textit{bh\={a}vita\d{m} bh\={a}vita\d{m} yatra dvayor var\d{n}ayor gu\d{n}a\d{h}}\\
\textit{sth\={a}pyate samatulya\d{m} tad bh\={a}vita\d{m} pracak\d{s}ate}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} \dev{भावित} इति द्वयोः अज्ञातराश्योः गुणनफलम्। एतादृशं समीकरणं \dev{भावितसमीकरण} इति उच्यते।
\vspace{0.3cm}
\textbf{उदाहरणम्:} $(2x + 3) \cdot y = 6$ अथवा $2xy + 3y = 6$
\vspace{0.2cm}
\textbf{उत्तरम्:} अनिर्णीतम्। यस्याः $y \neq 0$, तस्य
\[x = \frac{6 - 3y}{2y}\]
\vspace{0.2cm}
यदि $xy = c$ (नियतगुणनफलम्) अथवा $xy = ax + by + c$ तर्हि भावितसमीकरणम्।
}{%
\textbf{Rule:} \textit{Bh\={a}vita} is the product of two unknown quantities. An equation involving such a product is called a \textit{bh\={a}vita-sam\={\i}kara\d{n}a}.
\vspace{0.3cm}
\textbf{Example:} $(2x + 3) \cdot y = 6$, or $2xy + 3y = 6$
\vspace{0.2cm}
\textbf{Solution:} Indeterminate. For any $y \neq 0$:
\[x = \frac{6 - 3y}{2y}\]
\vspace{0.2cm}
Equations of the form $xy = c$ (constant product) or $xy = ax + by + c$ are \textit{bh\={a}vita} equations.
}

\clearpage

%% ========================================================================
%% CHAPTER XXIV -- INDIVIDUAL SUBTRACTION METHOD
%% ========================================================================
\sectionbicol{प्रत्येकसंहारः}{Pratyeka-sa\d{m}h\={a}ra\d{h} (Individual Subtraction)}

\vspace{0.3cm}

\bicolverse{%
प्रत्येकं संहरेद् भाज्यं हरेण प्रत्येकं बुधः~।\\
लब्धानि यानि तान्येव मूलानि प्रत्येकशः~
}{%
\textit{pratyeka\d{m} sa\d{m}hared bh\={a}jya\d{m} hare\d{n}a pratyeka\d{m} budha\d{h}}\\
\textit{labdh\={a}ni y\={a}ni t\={a}ny eva m\={u}l\={a}ni pratyeka\'{s}a\d{h}}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} प्रत्येकसंहारः तर्हि प्रयुज्यते यदा एकेनैव हरेण बहूनि भाज्यानि विभज्यन्ते। प्रत्येकं लब्धिः पृथक् मूलत्वेन गृह्यते।
\vspace{0.3cm}
\textbf{उदाहरणम्:} $100x^2$, $80x^2$, $60x^2$ इति त्रीणि प्रत्येकं $20x^2$ इति विभज्यन्ताम्।
\[\text{लब्धय: } 5,\ 4,\ 3 \quad \text{क्रमश:}\]
}{%
\textbf{Rule:} The method of individual subtraction is used when the same divisor is applied separately to multiple dividends. Each quotient is then a separate root.
\vspace{0.3cm}
\textbf{Example:} Divide $100x^2$, $80x^2$, and $60x^2$ individually by $20x^2$.
\[\text{Quotients: } 5,\ 4,\ 3 \quad \text{respectively}\]
}

\clearpage

%% ========================================================================
%% CHAPTER XXV -- PRADISHA METHOD
%% ========================================================================
\sectionbicol{प्रदिशविद्यया}{Pradi\'{s}a-vidyay\={a} (By the Pradi\'{s}a Method)}

\vspace{0.3cm}

\bicolverse{%
प्रदिशा विद्यया ज्ञेयं मूलं यत्र प्रदिश्यते~।\\
प्रदिशं तत्र योज्यं स्याद् विद्वद्भिरनुमीयते~
}{%
\textit{pradi\'{s}\={a} vidyay\={a} j\~{n}eya\d{m} m\={u}la\d{m} yatra pradi\'{s}yate}\\
\textit{pradi\'{s}a\d{m} tatra yojya\d{m} sy\={a}d vidvadbhir anum\={\i}yate}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} \dev{प्रदिश} विधिः मूलस्य अन्वेषणे प्रज्ञापूर्वकअनुमानं कर्तुं प्रयुक्ता। उच्चघातसमीकरणेषु विशिष्टा।
\vspace{0.3cm}
\textbf{उदाहरणम्:} $13824$ इत्यस्य घनमूलं ज्ञातव्यम्।
\begin{itemize}[leftmargin=1.5em,topsep=0pt]
\item एककस्थानम् $4$ अतः घनमूलस्य एककं $4$
\item $20^3 = 8000$, $30^3 = 27000$
\item अतः मूलं $20$ तथा $30$ मध्ये
\item $\therefore$ घनमूलम् $= 24$ ($24^3 = 13824$)
\end{itemize}
}{%
\textbf{Rule:} The \textit{pradi\'{s}a} method finds roots by inspection and intelligent guessing, especially useful for higher-degree equations.
\vspace{0.3cm}
\textbf{Example:} Find the cube root of $13824$.
\begin{itemize}[leftmargin=1.5em,topsep=0pt]
\item The number ends in $4$, so the cube root ends in $4$
\item $20^3 = 8000$ and $30^3 = 27000$
\item So the root is between $20$ and $30$
\item $\therefore$ Cube root $= 24$ ($24^3 = 13824$)
\end{itemize}
}

\clearpage

%% ========================================================================
%% CHAPTER XXVI -- NEW PRADISHA METHOD
%% ========================================================================
\sectionbicol{नवीनप्रदिशविद्यया}{Nav\={\i}na-pradi\'{s}a-vidyay\={a} (By the New Pradi\'{s}a Method)}

\vspace{0.3cm}

\bicolverse{%
नवीं प्रदिशमास्थाय मूलं यत् प्रदिश्यते~।\\
तत् सुलभं भवेत् प्राज्ञैर् विदुषां नात्र संशयः~
}{%
\textit{nav\={\i}\d{m} pradi\'{s}am \={a}sth\={a}ya m\={u}la\d{m} yat pradi\'{s}yate}\\
\textit{tat sulabha\d{m} bhavet pr\={a}j\~{n}air vidu\d{s}\={a}\d{m} n\={a}tra sa\d{ms}aya\d{h}}
}

\vspace{0.5cm}

\bicol{%
\textbf{सूत्र:} नवी प्रदिशविद्या पूर्वविधेः सुधारिता संस्करणम्। दुष्करतरमूलानाम् अप्य् अन्वेषणे समर्था।
\vspace{0.3cm}

अस्यां संहितायां पूर्वस्याः प्रदिशविधेः सन्ततिः दुष्करतरघातानाम् उच्चकोटिसमीकरणानां च मूलप्रदानाय सन्ततिरस्ति।
}{%
\textbf{Rule:} The new \textit{pradi\'{s}a} method gives an improved technique for finding roots, suitable for even more complex cases.
\vspace{0.3cm}

This chapter extends the previous \textit{pradi\'{s}a} methods with additional techniques for extraction of roots of higher powers and for handling more complex algebraic expressions.
}

\textbf{२. ऋणसंख्याविषये:} भास्करः ऋणसंख्याः पूर्णतया स्वीकरोति, \dev{ऋण} (ऋणम्) इति धनविपरीतम् उच्यते। ऋणैः सह गुणनादिक्रियानां नियमाः स्पष्टतया दत्ताः।

\textbf{2. On negative numbers:} Bh\={a}skara fully accepts negative numbers, calling them \d{r}\d{n}a (debt), opposed to dhana (wealth/positive). He gives clear rules for arithmetic with negatives.

\textbf{३. करणीविषये:} \dev{करणी} (करणी) क्रियाः असङ्ख्येयसंख्यानां कलनस्य साक्षात् साक्षात् सूचयन्ति। भास्करः करणीविशिष्टराशीन् संकलयितुं, व्यवकलयितुं, गुणयितुं, विभजितुं, मूलानि च निष्काषयितुं शक्नोति।
}{%
\textbf{3. On surds:} The \textit{kara\d{n}\={\i}} (surd) operations demonstrate sophisticated understanding of irrational numbers. Bh\={a}skara can add, subtract, multiply, divide, and extract roots of surd expressions.
}

\vspace{0.5cm}

\bicol{%
\textbf{४. चक्रवालविधौ:} $Nx^2 + 1 = y^2$ इत्यस्य समाधानाय \dev{चक्रवाल} विधिः भारतीयगणितस्य अत्युत्तमसाधनानाम् एकतमा। एषः सर्वदा न्यूनोत्तरं प्राप्नोति।
}{%
\textbf{4. On the cyclic method:} The \textit{cakrav\={a}la} method for solving $Nx^2 + 1 = y^2$ is one of the greatest achievements of Indian mathematics. It always converges to the minimal solution.
}

\vspace{0.5cm}

\bicol{%
\textbf{५. समीकरणविषये:} भास्करः समीकरणानि बहुविधानि वर्गीकरोति तेषां प्रत्येकं क्रमबद्धं विधिं ददाति। द्विघातसमीकरणस्य व्याख्या पूर्णा, द्वयोः, एकस्य, निर्णीतानाम् उत्तरानां सहिता।
}{%
\textbf{5. On equations:} Bh\={a}skara classifies equations into several types and gives systematic methods for each. His treatment of quadratic equations is complete, including cases with two, one, or no real solutions.
}

\bilingual{\textbf{संस्कृतम्}\\[0.5em]
अयं ग्रन्थः श्रीभास्कराचार्यविरचितस्य \textbf{बीजगणितस्य} द्वितीयं भागं समाविशति। बीजगणितं गणितशास्त्रस्य मूलभूतं खण्डं यत् समीकरणानां साधनं, अज्ञातराशीनां निर्णयं च व्याख्यायते। पृष्ठक्रमांक ५८--११४ पर्यन्तं चत्वारि प्रधानाध्यायानि समाविष्टानि—एकवर्णसमीकरणम्, अव्यक्तवर्गादिसमीकरणम्, अनेकवर्णसमीकरणम्, तथा अनेकवर्णमध्यमाहरणभेदाः।
}{%
\textbf{English Translation}\\[0.5em]
This volume contains \textbf{Part II} of the \textit{B\={\i}jaga\d{n}ita} (``Seed Computation,'' i.e., Algebra) composed by \textbf{Bh\={a}skara II} (1114--1185 CE). The \textit{B\={\i}jaga\d{n}ita} is the foundational treatise of Indian algebra, treating equations, unknown quantities, and their solutions. Pages 58--114 encompass four principal chapters: equations with one unknown, equations with unknown squares, equations with many unknowns, and variations of the elimination method.}

}{%
A distinctive feature of Bh\={a}skara's exposition is that each topic is introduced by a verse (\textit{s\={u}tra}) containing a rule, followed by worked examples (\textit{ud\={a}hara\d{n}a}) and detailed commentary (\textit{v\={a}rttika}) showing the method of solution step by step. The unknown quantity is denoted by \dev{यावत्तावत्} (\textit{y\={a}vatt\={a}vat}, ``as much as''), abbreviated \dev{या} or \dev{याव}. Equations are solved by the process of \textit{sama-\'{s}odhana} (making equal and clearing).}

}{%

\section*{\dev{एकवर्णसमीकरणम्} $\parallel$ Ekavar\d{n}a-sam\={\i}kara\d{n}a}
\addcontentsline{toc}{section}{Ekavar\d{n}a-sam\={\i}kara\d{n}a (pp.~58--76)}

\bilingual{\lettrine[lines=2, lraise=0.1, nindent=0em, slope=-.5em]{\dev{अ}}{}थैकवर्णसमीकरणम्। अत्र \textbf{यावत्तावत्} एको वर्णः प्रकल्प्यते, ततो रूपैः सह समीकृत्य यावत्तावन्मानं साध्यते। राशिश्च द्विधा—\textit{स्वीकृतपक्षः} (accepting side) तथा \textit{दातृपक्षः} (giving side) इति। तयोः समशोधनेन मानं निर्णीयते।
}{%
\lettrine[lines=2, lraise=0.1, nindent=0em, slope=-.5em]{N}{}ow, equations with one unknown. Here, a single unknown quantity, called \textbf{y\={a}vatt\={a}vat} (\dev{यावत्तावत्}, ``as much as''), is postulated. Combining it with known quantities (\textit{r\={u}pa}, \dev{रूप}), one forms an equation and determines the value of the unknown. Each side of the equation is called either the \textit{accepting side} (what is received) or the \textit{giving side} (what is given). By the process of equal-clearing (\textit{sama-\'{s}odhana}), the value is determined.}

\subsection*{Simple Linear Equation $\parallel$ सरलरेखीयसमीकरणम्}

\bilingual{\textbf{उदाहरणम्} — अत्राश्वमूल्यमज्ञातं तस्य मानं \dev{यावत्तावत्} एकं प्रकल्प्तं \dev{या १}। तदा श्रेणीशकं यदेकस्य यावत्तावन्मूल्यं तदा पणां किमिति कल्पितच्छागपणां प्रमाणमकं लब्धं पणामश्वानां मूल्यम्। \dev{या ६}। अत्र रूपातनये प्रक्षिप्ते जातमाश्चस्य धनं \dev{या ६ रूप २०}। एवं दशानां मूल्यं \dev{या १०}। अत्र रूपाते चाणांते प्रक्षिप्ते जातं द्वितीयस्य धनं \dev{या १० रूप १००}।
}{%
\textbf{Example} --- Here, the price of a horse is unknown; let its value be one \textbf{y\={a}vatt\={a}vat}, written \textbf{y\={a} 1}. If the price of one horse equals the price of so many goats, then the price in coins is determined. For one horse: \textbf{y\={a} 6}. Adding twenty coins (\textit{r\={u}pa}), the first person's wealth becomes: \textbf{y\={a} 6, r\={u}pa 20}. The price of ten horses is \textbf{y\={a} 10}. Adding one hundred coins, the second person's wealth becomes: \textbf{y\={a} 10, r\={u}pa 100}.}

\vs

\bilingual{\textbf{समशोधनम्} — एतौ समशोधने छते एव सभी जाते समशोधनार्थं—
\begin{center}\dev{न्यासः---या ६ रूप २०}\\\dev{या १० रूप १००}\end{center}
}{%
\textbf{Equal-clearing} --- These two are made equal for clearing:
\begin{center}\textsc{Setting down:} \textbf{y\={a} 6, r\={u}pa 20}\\\textbf{y\={a} 10, r\={u}pa 100}\end{center}}

\vs

\bilingual{\textbf{नियमः} — अथैकाव्यक्त शोधयेदन्यपक्षादव्ययव्यक्तान्। छोधिते द्वितीयपदमुक्त्वेषु आचयपक्षतुल्यपक्षः शोधितेषु शेषं रूप ४००। अथ कल्पितकलवर्ण \dev{या ४} रूपैः रूप ४०० उक्ते लब्धमेकस्य यावत्तावतो मानं व्यक्तं १००।
}{%
\textbf{Rule} --- One should clear the unknown from one side, and the known from the other side. When cleared, the remainder on the other side is r\={u}pa 400. Then, dividing r\={u}pa 400 by the coefficient of the unknown (y\={a} 4), the value of one y\={a}vatt\={a}vat is obtained as 100. Thus the price of one horse is 100 (coins).}

\subsection*{Example: Jewels and Wealth / रत्नधनस्योदाहरणम्}

\bilingual{\begin{sanskritverse}
माणिक्यामल्नीलमौक्तिककिमितिः पञ्चाषसममा—\\
देकस्यान्यतरस्य सप्त नव पट् तद्दलसंख्या सखे।\\
रूपाणां नवतिद्विषष्ठिर्नयोस्तौ तुल्यवित्तौ तथा\\
वीजं प्रतिरुज्ञानि कुरुते मौल्यानि शीघ्रं वद ॥३॥
\end{sanskritverse}
}{%
\begin{flushright}
\textit{Verse 3: Jewels and Equal Wealth}
\end{flushright}
``The measures of ruby, sapphire, pearl, and emerald of one are fifty, seven, nine, and half that number, O friend. Their total wealth is one hundred sixty-two coins each. Tell me quickly the prices, O algebraist, that make them equal.''

\vs
\textsc{Solution:} Let the prices of ruby etc. be y\={a}(1) 3, y\={a} 2, y\={a} 1. The two sides become:
\begin{center}\textbf{y\={a} 19, y\={a} 16, y\={a} 7, r\={u}pa 90}\\\textbf{y\={a} 21, y\={a} 18, y\={a} 6, r\={u}pa 62}\end{center}
By equal-clearing, the value of y\={a}vatt\={a}vat is 4. The prices of jewels are 12, 8, 4.}

\vs

\bilingual{\begin{sanskritverse}
एको द्वीति मम द्विहि शतं धनेन\\
त्वत्तो भवामि हि सखे द्विगुणस्ततोऽस्यः।\\
भूते द्वयोर्यदसि वैनम बहुगुणोऽहं\\
त्वत्तस्तयोर्वद धने मम किं प्रमाणे ॥४॥
\end{sanskritverse}
}{%
\begin{flushright}
\textit{Verse 4: The Merchant's Wealth}
\end{flushright}
``If you give me one hundred, O friend, I become twice as rich as you. But if I give you two hundred, I become some multiple. Tell me, what are our respective measures of wealth?''

\vs
\textsc{Solution:} Let the two merchants' wealth be y\={a} 1 and y\={a} 1. After giving 100, the first becomes twice the second: \textbf{y\={a} 1, r\={u}pa 100} = \textbf{y\={a} 2, r\={u}pa 200}. By clearing: \textbf{y\={a} 4, r\={u}pa 300}, whence y\={a} = 75. The wealths are 40 and 170.}

\subsection*{Quadratic and Higher Equations / वर्गसमीकरणम्}

\bilingual{\textbf{उदाहरणम्} — राशिद्वैदशानिशं राशिद्वयनाल्यक्ष कः समो यः स्यात्। राशिकृतिः बहुगुणिता पञ्चत्रिंशद्युता विद्ध ॥६॥
}{%
\textbf{Example} --- ``A certain number when multiplied by twelve and added to its own square equals thirty-five. What is that number?'' Let the number be y\={a} 1. Then: 12y\={a} + y\={a}varga 1 = 35. Clearing gives sides: \textbf{y\={a}v 1, y\={a} 12, r\={u}pa 0} and \textbf{r\={u}pa 35}. Extracting the square root: \textbf{y\={a} 1, r\={u}pa 2} and \textbf{r\={u}pa 3}. From the equation of these, the number is 5.}

\vs

\bilingual{\textbf{वर्गपूरणनियमः} — एषां पूर्वमूलानां च सर्वेषां योगः = \dev{याव ३ या ३१ रूप ८८}। इदमेकाद्वययुतं प्रयोदशवर्गं—
\begin{center}\dev{याव ३ या ३१ रूप ७५}\\\dev{याव ० या ० रूप १६९}\end{center}
}{%
\textbf{Rule for Completion of the Square:} The sum of all these roots = \textbf{y\={a}v 3, y\={a} 31, r\={u}pa 88}. Adding an increment to make a perfect square:
\begin{center}\textbf{y\={a}v 3, y\={a} 31, r\={u}pa 75}\\\textbf{r\={u}pa 169}\end{center}
Equating and multiplying both sides by twelve, taking the double-root from the square 169, the roots are \textbf{y\={a} 6, r\={u}pa 31} and \textbf{r\={u}pa 43}.}

\subsection*{The Lotus and the Swallow / हंसपद्मोदाहरणम्}

\bilingual{\textbf{उदाहरणम्} — अत्र वा प्रथमप्रमाणकलं द्वितीयप्रमाणकलं विमले प्रलम्बये तद्गुणगुणितेन द्वितीयमूललेन तुल्यमेव प्रथममूलधनं स्यात्। अतो द्वितीयस्यायं गुणः २।
}{%
\textbf{Example: The Lotus and the Swallow} --- Here, the difference of the two root-measures, multiplied by the product of the multiplier, equals the second root-wealth. The multiplier of the second is 2. If the first root-wealth equals the product of one multiplier and the second root-wealth, divided by the square of the difference, then the difference of measures is 4. Thus the first and second root-wealths are 8 and 4, yielding a result of 2.}

\vs

\bilingual{\begin{sanskritverse}
एकक्रान्तर्चयनात् फलस्य वर्गं विशोध्य परिशिष्यम्।\\
पञ्चक्रान्तेन दत्तं तुल्यः कालः फले च तथैः ॥८॥
\end{sanskritverse}
}{%
\begin{flushright}
\textit{Verse 8: The Lotus Problem}
\end{flushright}
``Having subtracted the square of the result from the first increased [quantity], the remainder, divided by five, gives equal time and result.''

\vs
\textsc{Solution:} The multiplier is 5. Dividing by the decreased multiplier (4), the square of the small difference (16) divides the second wealth, yielding 4. Adding the square of the difference gives the first wealth as 20. By the rule of proportion, time = 20.}

अव्यक्तवर्गादि यदा द्वयोः पक्षौ तदैकेन निःशेषं किंचित्।\\

\begin{flushright}

\textit{many unknown values may be obtained;}\\
\textit{if the square does not succeed,}\\

\end{flushright}}

\end{sanskritverse}

\textbf{Verse 3:}
\begin{flushright}
\textit{If the root of the unknown side, having a number,}\\
\textit{when made the root of the known side---}\\
\textit{having made that root positive or negative,}\\
\textit{the unknown value is to be assumed twofold.}
\end{flushright}}

\subsection*{Example: Series of Numbers / राशिश्रेण्युदाहरणम्}

\bilingual{\begin{sanskritverse}
स्वार्धपञ्चाशनवमैर्युक्ताः के स्युः समाश्रयाः।\\
अन्यैर्द्वादशहीनाख्य परिशिष्टाख्य तान् वद ॥१४॥
\end{sanskritverse}
}{%
\begin{flushright}
\textit{Verse 14: Numbers with Fractional Relations}
\end{flushright}
``What numbers, when increased by their own half, five-ninths, and [other fractions], become equal? And when diminished by others, twelve less---tell me the remainders.''

\vs
\textsc{Solution:} Let the equal number be y\={a}vatt\={a}vat 1. By the reverse process, the numbers are y\={a} $\frac{2}{5}$, y\={a} $\frac{5}{6}$, y\={a} $\frac{9}{10}$. Dividing by the greatest common measure and multiplying by y\={a}vatt\={a}vat value 150, the numbers are 100, 125, 135.}

\vs

\bilingual{\begin{sanskritverse}
त्रयोदश तथा पञ्च करण्यो मूजयोनितो।\\
भूजाता च चत्वारः फले भूमिं वदाऽखु मे ॥१५॥
\end{sanskritverse}
}{%
\begin{flushright}
\textit{Verse 15: The Excavated Plot}
\end{flushright}
``The square-roots of thirteen and five, diminished by the base, and four as the result---tell me the area, O algebraist.''

\vs
\textsc{Solution:} Let the area be y\={a}vatt\={a}vat 1. By the formula ``altitude times half-base divided [equals the area of a triangle],'' the section-ratio is k\={a} $\frac{5}{13}$. Its square is k\={a} 5. Subtracting from r\={u}pa 5, r\={u}pa $\frac{5}{13}$, the root gives the altitude k\={a} $\frac{5}{13}$.}

\end{sanskritverse}
}{%
\textbf{Verse 17:}
\begin{flushright}
\textit{The difference between the square of the sum}\\
\textit{and four times the product}\\
\textit{equals the square of the difference of the two numbers,}\\
\textit{as between the last two terms.}
\end{flushright}
\textsc{Numerical verification:} For numbers 3 and 5: $(3+5)^2 - 4 \cdot 3 \cdot 5 = 64 - 60 = 4 = (5-3)^2$. This is the algebraic identity: $(a+b)^2 - 4ab = (a-b)^2$.}

\subsection*{The General Rule / सामान्यसूत्रम्}

\bilingual{\begin{sanskritverse}
द्वयोः समीकरणयोर्मूलमेकं यदि विद्यते।\\
तदा तदन्यत्समीकर्य मूलं तद्वदनुष्ठितम्॥
\end{sanskritverse}
}{%
\textbf{General Rule:}
\begin{flushright}
\textit{When the root of one of two equations is known,}\\
\textit{then having equated the other,}\\
\textit{its root is obtained by the same procedure.}
\end{flushright}
\textsc{Explanation:} This is the fundamental principle of elimination. If one variable can be expressed from one equation, substitute this expression into the second equation. This is the \textit{madhyam\={a}hara\d{n}a} (elimination of the middle) method in its general form.}

\subsection*{Example: Doubled Number with Square / द्विगुणितराशिवर्गयुतः}

\bilingual{\begin{sanskritverse}

मूलद्वो जायते बीजगणितं वद तत् त्वम् ॥१॥
\end{sanskritverse}
}{%
\begin{flushright}
\textit{Verse 1: A Number with Its Square}
\end{flushright}
``What number, when doubled and added to its square-root's square (i.e., the number itself), produces two roots? Speak that algebra, O you.''

\vs
\textsc{Solution:} Let the first unknown's doubled quantity with its square be y\={a}v 6, y\={a} 2. For the square, the setting-down is:
\begin{center}\textbf{y\={a}v 6, y\={a} 2, k\={a}v 0}\\\textbf{y\={a}v 0, y\={a} 0, k\={a}v 1}\end{center}
By equal-clearing, the sides become y\={a}v 6, y\={a} 2, k\={a}v 1. Multiplying by the coefficient and adding r\={u}pa, the first root is \textbf{y\={a} 6, r\={u}pa 1}. The second root is \textbf{k\={a} 2}, or k\={a} 20 with residue 49. Equating with the first side y\={a} 6, r\={u}pa 1, the value of y\={a}vatt\={a}vat is $\frac{2}{5}$ or 8.}

\section*{The Cakrav\={a}la (Cyclic) Method $\parallel$ चक्रवालपद्धतिः}
\addcontentsline{toc}{section}{The Cakrav\={a}la Method (Vargaprak\d{r}ti)}

\bilingual{\textbf{चक्रवालपद्धतिः} — \dev{वर्गप्रकृति} इत्युक्ते $x^2 - Ny^2 = 1$ रूपं समीकरणं यत्र $N$ अज्ञातवर्गे कारः (गुणकः), $x$ प्रकृतिः (ज्येष्ठमूलम्), $y$ ह्रस्वम् (कनिष्ठमूलम्)। भास्कराचार्यस्य \textbf{चक्रवालपद्धतिः} अस्मिन् समीकरणे साधने सर्वोत्तमा अल्गोरिद्मिकपद्धतिः अस्ति। इयं पद्धतिः सदैव परिसमाप्तिं गच्छति तथा च न्यूनतमं समाधानं प्रयच्छति।
}{%
\textbf{The Cakrav\={a}la (Cyclic) Method:} The \textit{vargaprak\d{r}ti} is the equation $x^2 - Ny^2 = 1$, where $N$ is the \textit{k\={a}ra\d{n}a} (multiplier/prak\d{r}ti), $x$ is the \textit{jye\d{s}\d{t}ha-m\={u}la} (greater root), and $y$ is the \textit{kani\d{s}\d{t}ha-m\={u}la} (lesser root). Bh\={a}skara's \textbf{Cakrav\={a}la} (``cyclic'') method is the supreme algorithmic procedure for solving this equation. This method always terminates and produces the minimal solution.}

\vs

\bilingual{\textbf{चक्रवालस्य पद्धतिः} —
\begin{enumerate}[label=\arabic*.]
\item मूलस्य ($\sqrt{N}$) समीपतमं पूर्णाङ्कं गृहाण।
\item प्रथमं समीकरणं रचय: $a^2 - N \cdot 1^2 = k$, यत्र $a$ समीपतमपूर्णाङ्कं, $k$ क्षेपः।
\item ततः $b = \dfrac{a + m}{|k|}$ गणय, यत्र $m$ अनेन प्रकारेण वृत्यते यत् $a + m \equiv 0 \pmod{|k|}$ तथा च $m^2 - N$ न्यूनतमं भवति।
\item नवीन समीकरणं रचय: $a_{\text{new}}^2 - N \cdot b^2 = k_{\text{new}}$।
\item यदा $k_{\text{new}} = 1, 2, -1, -2, 4, -4$ भवति, तदा \textit{भावना} (composition) क्रियते।
\item अन्यथा चक्रं पुनः आरभ्यते।
\end{enumerate}
}{%
\textbf{The Cakrav\={a}la Procedure:}
\begin{enumerate}[label=\arabic*.]
\item Take the integer nearest to $\sqrt{N}$.
\item Form the first equation: $a^2 - N \cdot 1^2 = k$, where $a$ is the nearest integer, $k$ is the additive (\textit{k\d{s}epa}).
\item Compute $b = \dfrac{a + m}{|k|}$, where $m$ is chosen such that $a + m \equiv 0 \pmod{|k|}$ and $m^2 - N$ is minimized.
\item Form the new equation: $a_{\text{new}}^2 - N \cdot b^2 = k_{\text{new}}$.
\item When $k_{\text{new}} = 1, 2, -1, -2, 4, -4$, apply \textit{bh\={a}van\={a}} (composition).
\item Otherwise, repeat the cycle.
\end{enumerate}}

\subsection*{Final Composition: The Solution / अन्तिमभावना: समाधानम्}

\bilingual{\textbf{समाधानम्} — यदा क्षेपः $k = -1$ प्राप्यते (चक्रे ११), तदा \textit{भावनया} सह सम्पूर्णं समाधानं प्राप्यते। अन्तिमे समीकरणे $2857^2 - 61 \cdot 366^2 = -1$ भवति। अस्य वर्गीकरणेन ($k = -1$ इत्यस्य द्विवारं प्रयोगेन) $k = 1$ प्राप्यते।

\vspace{0.5em}
\textbf{\dev{अतः न्यूनतमं समाधानम्:}}
\begin{center}
\fbox{\dev{ज्येष्ठमूलम् $x = 1{,}766{,}319{,}049$}}\\[0.3em]
\fbox{\dev{कनिष्ठमूलम् $y = 226{,}153{,}980$}}
\end{center}
\vspace{0.3em}
\textbf{\dev{सत्यापनम्:}}
\begin{center}
$1{,}766{,}319{,}049^2 - 61 \cdot 226{,}153{,}980^2 = 1$
\end{center}
}{%
\textbf{The Solution:} When the additive $k = -1$ is reached (at cycle 11), the complete solution is obtained by \textit{bh\={a}van\={a}} (composition). The final equation yields $2857^2 - 61 \cdot 366^2 = -1$. Squaring this (applying $k = -1$ twice) gives $k = 1$.

\vspace{0.5em}
\textbf{The minimal solution is:}
\begin{center}
\fbox{Greater root $x = 1{,}766{,}319{,}049$}\\[0.3em]
\fbox{Lesser root $y = 226{,}153{,}980$}
\end{center}
\vspace{0.3em}
\textbf{Verification:}
\begin{center}
$1{,}766{,}319{,}049^2 - 61 \cdot 226{,}153{,}980^2 = 1$ $\checkmark$
\end{center}

\vs
\textsc{Historical Note:} This solution was not matched in Europe until the work of Fermat, Brouncker, and Wallis in the 17th century. Bh\={a}skara's Cakrav\={a}la algorithm always terminates---a fact that was only rigorously proved in the 20th century by Krishnaswami Alladi. The method is superior to the European ``chain method'' of Brouncker in both elegance and efficiency.}

\section*{The Bh\={a}vita Method $\parallel$ भावितपद्धतिः}
\addcontentsline{toc}{section}{The Bh\={a}vita Method (Product Equations)}

\bilingual{\textbf{भावितपद्धतिः} — \dev{भावित} इत्युक्ते द्वयोः अज्ञातराश्योः घातः (product)। यदा समीकरणे द्वयोः वर्णयोः गुणनफलं प्रविष्टं भवति, तदा \textit{भावितपद्धतिः} प्रयुज्यते। अस्याः पद्धतेर्मुख्यं प्रयोजनं \textit{वर्गप्रकृतेः} समाधानेषु \textit{भावनायाः} (composition) क्रियायां विद्यते।
}{%
\textbf{The Bh\={a}vita Method:} \textit{Bh\={a}vita} means the product of two unknown quantities. When the product of two unknowns appears in an equation, the \textit{bh\={a}vita} method is applied. The principal application of this method is in the \textit{bh\={a}van\={a}} (composition) operation for solving the \textit{vargaprak\d{r}ti}.}

\vs

\bilingual{\textbf{भावनानियमः} — यदि $(a, b, k)$ तथा $(A, B, K)$ द्वे वर्गप्रकृतिसमाधाने स्तः, अर्थात् $a^2 - Nb^2 = k$ तथा $A^2 - NB^2 = K$, तदा भावनया नवीनं समाधानं प्राप्यते:
\begin{center}
\fbox{$x = aA \pm NbB$, \quad $y = aB \pm Ab$}
\end{center}
\vspace{0.3em}
यस्य क्षेपः नवीनं समीकरणं $x^2 - Ny^2 = kK$ सन्तोषयति।
}{%
\textbf{The Bh\={a}van\={a} Rule:} If $(a, b, k)$ and $(A, B, K)$ are two solutions of the vargaprak\d{r}ti, i.e., $a^2 - Nb^2 = k$ and $A^2 - NB^2 = K$, then by \textit{bh\={a}van\={a}} a new solution is obtained:
\begin{center}
\fbox{$x = aA \pm NbB$, \quad $y = aB \pm Ab$}
\end{center}
\vspace{0.3em}
whose additive satisfies the new equation $x^2 - Ny^2 = kK$.

\vs
\textsc{Special Case:} When $k = K = 1$, the product $kK = 1$, so the composed solution directly solves $x^2 - Ny^2 = 1$. When $k = K = -1$, then $kK = 1$ again, so two solutions with additive $-1$ compose to a solution with additive $+1$. This is the key to terminating the Cakrav\={a}la method.}

\item \textbf{Equations with Many Unknowns} --- solution by the \textit{ku\d{t}\d{t}aka} (pulverizer) method.
\item \textbf{Elimination of the Middle} --- the method of eliminating the middle unknown.
\item \textbf{The Cakrav\={a}la Method} --- the supreme method for solving $x^2 - Ny^2 = 1$.
\item \textbf{The Bh\={a}vita Method} --- product equations and the composition operation.
\end{enumerate}}

\begin{itemize}

\item Produces the minimal solution (produces the minimal solution)

\section{From the Varga-prakṛti Section (p.\ 115)}

\colheaders

\paralleltext{%
अयं भागः वर्गप्रकृतिचक्रवालविधेः प्रवृत्तिं समाविशति । वर्गप्रकृतिसमीकरणं $Nx^2 + k = y^2$ रूपेण भवति । अत्र नवानि सूत्राणि, उदाहरणानि च प्रदर्शितानि ।

This section continues the discussion of the \textit{Varga-prakṛti} (square-nature) equation, i.e., equations of the form $Nx^2 + k = y^2$, known in modern mathematics as Pell's equation. The text presents further rules and worked examples for finding solutions to such indeterminate equations.

\subsection*{Continuation of the Square-nature Method}

\parallelverse{%
समशोधने हृते पक्षौ यावच्चावत्तावकस्य रूपं प्रक्षिप्य प्रथमपक्षमूलं या २ रू १ । परपक्षस्य काव १२ रू १ । वर्गप्रकृत्या मूले क २ उचे ७ वा क २८ उचे ९७ । कनिष्ठं कालकमानम् । उच्चिष्टस्य या २ रू १ समं हृत्वा लब्धं यावच्चावकमानम् ३ वा ४८ । स्वच्छानेनोत्थापने हृते जाती राशी १, ५ वा २०, ७६ इत्यादि ॥
}{%
After equal subtraction (\textit{sama-śodhana}), the two sides being reduced, adding the original constant to the first side, its root is $\sqrt{4y^2 + 1}$. For the second side: $\sqrt{12y^2 - 1}$. By the \textit{varga-prakṛti} [square-nature method], the roots (\textit{mūla}) are $(2, 7)$ or $(28, 97)$. The lesser (\textit{kaniṣṭha}) is the time-value (\textit{kālaka-māna}). The residue (\textit{ucciṣṭa}) from $\sqrt{4y^2 + 1}$ gives the \textit{yāvattāvat} value $3$ or $48$. By self-substitution (\textit{sva-uddhāpana}) and reduction, the series of values $1, 5$ or $20, 76$, etc.
}

\bigskip

\parallelverse{%
अथापरं सूत्रं साध्येऽस्तु ॥\\[0.5em]
(१) द्वितीयपक्षे सति सममेव तु हत्यापवर्त्योन पदे प्रसाध्ये । उच्चिष्टं कनिष्ठेन तदा निहत्याच्छेदं वर्गेण हतोऽपवर्तः ॥ ६ ॥\\[0.3em]
कनिष्ठवर्गेण तदा निहत्योच्चिष्टं ततः पूर्ववदेव शेषम् । इत्यर्थात् ॥
}{%
Now another rule is to be accomplished: Rule 6: ``When the second side exists [and is not unity], having made [the two sides] equal (\textit{sama}), multiplying and reducing (\textit{apavartana}), the remainder (\textit{ucciṣṭa}) [is obtained]; having multiplied the residue by the lesser [root] (\textit{kaniṣṭha-mūla}), dividing by the square [nature] (\textit{varga-prakṛti}), the reduced [equation is obtained]. Then having multiplied by the square of the lesser [root], the residue --- from that, as before, the remainder.'' Thus the meaning.
}

\subsection*{Worked Example (p.\ 115)}

\parallelverse{%
(१) वि०---कल्प्यते समी पक्षौ\\[0.3em]
\hspace*{2em} काव १\\
\hspace*{2em} यावव्। ई १ याव।ई १
}{%
(1) The two sides are assumed equal: \\[0.3em]
\hspace*{2em} [Side 1:] coefficient $12$ \\[0.3em]
\hspace*{2em} [Side 2:] $\sqrt{\text{yāva} \cdot 1} \cdot \sqrt{\text{yāva} \cdot 1}$
}

\paralleltext{%
अत्र प्रथमपक्षस्य मूले क १ द्वितीयपक्षस्य यावव्।ई १ याव।ई १ मूलेन सममिति । तत्र द्वितीयपक्षस्य मूले च का।
}{%
Here the root of the first side is $1$. The second side, having its root equal to $\sqrt{\text{yāva} \cdot 1 \cdot \text{yāva} \cdot 1}$, is equivalent to that. There, for the second side, let the root be \textit{kā}.
}

\vspace{0.5em}
\begin{center}
\fbox{$\displaystyle \text{का} = \sqrt{\text{यावव्।ई १ याव।ई १}} = \text{या}\sqrt{\text{याव।ई १ ई १}}$}
\end{center}
\vspace{0.5em}

\paralleltext{%
अत इदं याव।ई १ ई १ मूलदं ततो वर्गप्रकृतिविषयो यथा का वर्गः ई, गुणः ई, युतो मूलदं इति हत्वा यावच्चावकमानं उच्चिष्टं चास्य याव।ई १ ई १ मूलेन सममिति।
}{%
Therefore this gives the root of ``\textit{yāva} 1, constant 1''; then by the \textit{varga-prakṛti} method, as the square (\textit{kā-varga}) of \textit{kā} is $1$, the multiplier (\textit{guṇa}) is $1$, and when the additive is added, the root is obtained. Having multiplied, the \textit{yāvattāvat} value and the residue from $\sqrt{\text{yāva} \cdot 1 \cdot \text{constant} \cdot 1}$ are equal.
}

\section{Aneka-varṇa-madhyamāharaṇa-bhedāḥ (pp.\ 116--133)}
\label{sec:aneka-varna}

\paralleltext{%
अयं विभागः समग्रस्यान्तिमभागस्य महत्तमः अस्ति । अत्र बहुवर्णसमीकरणानां व्यवस्थितविधयः प्रदर्श्यन्ते । संस्कृते अज्ञातराशयः \textit{वर्णैः} (रङ्गैः) संज्ञायन्ते --- भिन्नेभ्यो वर्णेभ्यो भिन्नाः अज्ञातराशयः संबद्धाः भवन्ति ।
}{%
This is the largest section of the final portion, dealing systematically with equations involving multiple unknown quantities (\textit{aneka-varṇa}, ``many colours''). In Indian algebra, the term \textit{varṇa} (colour) is the traditional designation for an unknown quantity --- different unknowns being assigned different colours such as \textit{yāvattāvat} (black), \textit{nīla} (blue), \textit{pīta} (yellow), \textit{lohita} (red), etc.
}

\subsection{General Rules for Multiple Unknowns (pp.\ 116--117)}

\parallelverse{%
अथ सूत्रं साध्येऽस्तु ॥\\[0.5em]
(१) द्वितीयपक्षे सति सममेव तु हत्यापवर्त्योन पदे प्रसाध्ये । उच्चिष्टं कनिष्ठेन तदा निहत्याच्छेदं वर्गेण हतोऽपवर्तः ॥ ६ ॥\\[0.3em]
कनिष्ठवर्गेण तदा निहत्योच्चिष्टं ततः पूर्ववदेव शेषम् । इत्यर्थात् ॥
}{%
Now the rule is to be accomplished: ``When the second side exists, having made [the two sides] equal (\textit{sama}), multiplying and reducing (\textit{apavartana}), the remainder (\textit{śeṣa}) [is obtained] for the root; having multiplied the residue (\textit{ucciṣṭa}) by the lesser [root] (\textit{kaniṣṭha}), dividing by the square [coefficient], the reduced [value is obtained]. Having multiplied by the square of the lesser [root], the residue --- from that, as before, the remainder.'' Thus the meaning.
}

\subsection*{Worked Example 1 (p.\ 117)}

\parallelverse{%
उदाहरणम् ॥\\[0.3em]
यस्य वर्गेऽद्वितिः पञ्चगुणा वर्गशतोनिता । मूलद्वा जायते राशिः गणितज्ञ वदाशु तम् ॥ १ ॥
}{%
Example 1: ``A number (\textit{rāśi}) whose square minus twice [itself], diminished by a hundred times five [i.e., 500], has two roots (\textit{mūla-dvā}) --- tell me that number quickly, O mathematician (\textit{gaṇita-jña}).''
}

\paralleltext{%
अत्र राशिः=या १ । अस्य वर्गेऽद्वितिः पञ्चगुणा वर्गशतोनिता यावव ५ याव १०० । अयं वर्ग इति कालकवर्गसमं हृत्वा गृहीतं कालकवर्गस्य मूलं का १ । द्वितीयपक्षस्य यावव ५ याव १०० । यावच्चावकगुणापवत्यं वर्गप्रकृत्या मूले क १० उचे २० वा क १७० उचे ३८० ।
}{%
Here the number = \textit{yā} 1. Its square minus twice, five multiplied, diminished by a hundred [times the square]: [coefficient] 5, [square] 100. This being equal to the time-square, taking the root of the time-square as \textit{kā} 1. For the second side: 5 \textit{yāva}, 100 \textit{yāva}. By \textit{yāvattāvat}-multiplier-reduction and the \textit{varga-prakṛti} method, the roots are $(10, 20)$ or $(170, 380)$.
}

\subsection*{Worked Example 2 (p.\ 117)}

\parallelverse{%
उदाहरणम् ॥\\[0.3em]
कयोः स्यादन्तरे वर्गौ वर्गयोगो ययोर्घनः । तौ राशी कथयामित्रो वहुधा बोज्ञवित्तम ॥ २ ॥
}{%
Example 2: ``Two numbers (\textit{rāśī}) whose difference of squares [equals] the sum of their squares --- tell me those numbers, O friend (\textit{mitra}), in many ways, O best of algebraists (\textit{bīja-jña-vittama}).''
}

\paralleltext{%
अत्र राशी या १, का १ । अनयोरन्तरं या १ का १ नीलकवर्गसमं हृत्वा लब्धं यावच्चावकमानं का १ नीलं ई १ । अनेन यावच्चावकद्वितीयाप्य जाती राशी का १ नीलं ई, का १ । अनयोर्वर्गयोगः काव २ नीलका २ नीलव १ । एष घन इति नीलकवर्गयनसमं हृत्वा शोधने हृते जाते प्रथमपक्षे नीलव १ नीलव १ । द्वितीयपक्षे काव २ नीलका २ ।
}{%
Here the numbers are \textit{yā} 1, \textit{kā} 1. Their difference, \textit{yā} 1 minus \textit{kā} 1, being equal to the blue-square (\textit{nīla-kṛti}), the \textit{yāvattāvat} value obtained is \textit{kā} 1, blue (\textit{nīla}) = constant 1. By this, the two \textit{yāvattāvat} values [produce] the numbers: \textit{kā} 1, blue = constant; and \textit{kā} 1. The sum of their squares: $2$ \textit{kāva}, $2$ \textit{nīla-kā}, $1$ \textit{nīla-varga}. This is the cube (\textit{ghana}), equal to the blue-square product; in equal subtraction, the first side becomes blue $1$, blue $1$; the second side: $2$ \textit{kāva}, $2$ \textit{nīla-kā}.
}

\subsection*{Rule for Three Unknowns (pp.\ 119--120)}

\parallelverse{%
(१) साध्यकल्पो यदि वर्गवर्गेषद्वित्र्यादिवर्णस्य हते समं तम् । हृत्वा पदं तस्य तदन्वपक्षे वर्गप्रकृत्यैवकदैव मूले ॥\\[0.3em]
कनिष्ठमाच्छाद्य पदेन तुल्यं उच्चिष्टं द्वितीयेन समं विधायात् ॥ ८ ॥
}{%
Rule 8: ``If the square of the square of an assumed [quantity] (\textit{sādhya-kalpa}), diminished by the square of one, two, three, etc., unknown quantities (\textit{varṇa}), having made [the two sides] equal, taking the square root (\textit{pada}) of that, then by the \textit{varga-prakṛti} method, the roots (\textit{mūle}) [are found]; covering the lesser root (\textit{kaniṣṭha-mūla}) by an equal root, making the residue (\textit{ucciṣṭa}) equal to the second [unknown]...''
}

\subsection*{Worked Example with Three Unknowns (p.\ 121)}

\parallelverse{%
उदाहरणम् ॥\\[0.3em]
त्रिकादिचतुरश्रेणां नच्छेदं क्वापि च यत् कलम् । तदेव त्रिगुणं क्षेत्रमष्टयुगच्छेदं भवेद्व्र ॥ ९ ॥
}{%
Example: ``That fractional part which, when subtracted from the series beginning with three [i.e., 3, 5, 7, ...], nowhere leaves a remainder; the same tripled, the field [becomes] one with remainder eight --- [find it], O sage (\textit{vṛddha}).''
}

\paralleltext{%
अत्र श्रेण्योन्या इः । आदि=३, चयः=२, गच्छः=या १ । आदि=३, चयः=२, गच्छः=का १ । अनयोः (क) कलं=याव १ या २, काव १ का २ । अनयोरन्यत् त्रिगुणं परस्मिं हृत्वा शोधनार्थं
}{%
Here the series (\textit{śreṇī}) is an arithmetic progression. First term = 3, common difference = 2, number of terms = \textit{yā} 1. Another: first = 3, difference = 2, terms = \textit{kā} 1. Of these, the part = \textit{yāva} 1 \textit{yā} 2, \textit{kāva} 1 \textit{kā} 2. One of these, tripled, subtracted from the other, for equalization:
}

\begin{center}
\fbox{$\displaystyle \text{न्यासः} \longrightarrow \text{याव ३ या ६} \qquad \text{काव १ का २}$}
\end{center}

\paralleltext{%
शोधने हृते पक्षौ त्रिगुणीहृत्व नव प्रक्षिप्य प्रथमपक्षस्य मूलं या ३ रू ३ । द्वितीयपक्षस्य काव ३ का ६ रू ९ । नीलकवर्गेण
}{%
The two sides being reduced by equal subtraction, having tripled, adding nine, the root of the first side: $\sqrt{3\text{yā} + 3}$. For the second side: $\sqrt{3\text{kāva} + 6\text{kā} + 9}$. By the blue-square [method]...
}

\subsection*{Rule for Equal-square Unknowns (p.\ 121)}

\parallelverse{%
(९) प्रथमपक्षस्यद्विगुणीतच्छेदस्य मूलं नीलं प्रकल्प्य तद्वर्गेण समं परं पक्षं हृत्वा पूर्वोक्तर्यस्य वासना चातिसरति ॥
}{%
``Having assumed the root of twice the divisor of the first side as the `blue' (\textit{nīla}) [unknown], having divided the other side by its square, the desire (\textit{vāsanā}) is satisfied more than previously stated.''
}

\subsection{The Method of Substitution (pp.\ 125--128)}

\parallelverse{%
अथ सूत्रं सूचयम् ॥\\[0.5em]
(१) सकृपमयकमल्पकं वा विभोगमूलं प्रथमं प्रकल्प्य । योगान्तरशेषकमाजिताच्छेदान्तरशेषकतः पदं स्यात् ॥ १२ ॥
}{%
Now the rule of instruction: Rule 12: ``Having assumed the root of the difference (\textit{vibhoga-mūla}) of [the two sides] as the first [unknown], or a small quantity, [and] the residue of the difference of sums (\textit{yoga-antara-śeṣa}) as the square [of another unknown], the root (\textit{pada}) [is obtained].''
}

\paralleltext{%
वि०---अत्र कल्प्यते योगान्तरशेषमानम् = क्षे$_1$ वर्गान्तरशेषमानम् = क्षे$_2$, वर्गयोगशेषमानम् = क्षे$_3$ विभोगमूलम् = या, योगमूलम् = का
}{%
Here we assume: residue of difference of sums = \textit{kṣe}$_1$, residue of difference of squares = \textit{kṣe}$_2$, residue of sum of squares = \textit{kṣe}$_3$; root of difference = \textit{yā}, root of sum = \textit{kā}.
}

\begin{center}
\fbox{$\displaystyle \text{विभोगः} = \text{या}^2 - \text{क्षे}_1, \quad \text{योगः} = \text{का}^2 - \text{क्षे}_1$}
\end{center}

\begin{center}
\fbox{$\displaystyle \text{अल्पराशिः} = \frac{\text{का}^2 - \text{या}^2}{2}, \quad \text{वृहद्राशिः} = \frac{\text{का}^2 + \text{या}^2 - 2\text{क्षे}_1}{2}$}
\end{center}

\paralleltext{%
अल्पराशिः (छोटसंख्या) वृहद्राशिः (महत्संख्या) इत्येतयोः सूत्रेण लब्धिः । एतत् ``प्रक्षेपकविधिः'' इति प्रसिद्धम् ।
}{%
The lesser number (\textit{alpa-rāśi}) and the greater number (\textit{vṛhada-rāśi}) are thus obtained by the rule. This is known as the ``method of insertion'' (\textit{prakṣepaka-vidhi}).
}

\subsection*{Rule 13 (p.\ 129)}

\parallelverse{%
शेषं ततः शेषकमूलमन्यच्च मूले विद्ध्वादसकृत् सम्मते ॥ ९ ॥\\[0.3em]
समाविते वर्गकृती तु यत्र तन्मूलमादाय च शेषकस्य । इष्टोद्धृतस्येर्विवर्जितस्य दलैन समं तद्विह तदैव कार्यम् ॥ १० ॥
}{%
``Then the remainder (\textit{śeṣa}), the root of the remainder-residue (\textit{śeṣaka-mūla}), and another, having found the roots repeatedly, is approved. Where the square-forms (\textit{varga-kṛtī}) are combined (\textit{samāvita}), taking its root, and of the residue (\textit{śeṣaka}), diminished by what is raised by the desired [quantity], made equal by halves --- that indeed is the operation (\textit{kārya}).''
}

\subsection*{Worked Example on the Product Method (p.\ 130)}

\parallelverse{%
उदाहरणम् ॥\\[0.3em]
तौ राशी वद यत्कृत्योः समस्तगुणयोर्युतिः । मूलद्वा स्याद्‌योगमष्टौ मूलद्वा रूपसंयुतः ॥ १ ॥
}{%
Example: ``Tell those two numbers (\textit{rāśī}) of which the sum of the product of all the multipliers of the squares [equals]; [the root of] twice the sum [is] eight, or twice the root plus unity.''
}

\paralleltext{%
अत्र राशी या १, का १ । अनयोर्वर्गयोः समस्तगुणयोर्युतिः यावव ७ काव ८ । अयं वर्ग इति नीलकवर्गसमं हृत्वा पक्षयोः कालकधनं प्रक्षिप्य नीलकपक्षस्य मूलं नी १ । परपक्षस्य याव ७ काव ८ ।
}{%
Here the numbers are \textit{yā} 1, \textit{kā} 1. Of their squares, the sum of the products of all the multipliers: 7 \textit{yāva}, 8 \textit{kāva}. This being a square, equal to the blue-square, adding the time-value to both sides, the root of the blue side is \textit{nī} 1. For the other side: 7 \textit{yāva}, 8 \textit{kāva}.
}

\subsection*{Summary Verses (p.\ 131)}

\parallelverse{%
अथ सूत्रं सूचयम् ॥\\[0.5em]
(१) सकृपमयकमल्पकं वा विभोगमूलं प्रथमं प्रकल्प्य । योगान्तरशेषकमाजिताच्छेदान्तरशेषकतः पदं स्यात् ॥ १२ ॥\\[0.3em]
वर्गान्तरशेषकसंमितिर्युता क्षेपेण हृत्वोर्ध्वितेन तै ततः । द्विधा पदं तत्पदयुगविभोगाज् मूलं युतमूलमतस्तयोरिति ॥
}{%
``Having assumed the root of the difference as the first [unknown], or a small quantity, the residue of the difference of sums as the square, the root [is obtained] from the residue of the difference of divisors. The difference of squares having a residue equal to the divisor, divided by the additive (\textit{kṣepa}), then by the quotient, [the root] divided in two; from the difference of the pair of roots, the root [of the lesser] from the sum of the roots, thus [proceeding for] those two.''
}

\subsection{Equations with Product Terms (pp.\ 134--138)}

\paralleltext{%
अयं विभागः \textit{भावितसमीकरणानि} (घातफलयुक्तानि) व्यभचरति । अत्र अज्ञातराशीनां घातफलानि सन्ति । इदं विधानं बहुवर्णमध्यमाहरणस्य विधिं विस्तारयति ।
}{%
This section deals with \textit{bhāvita} (product) equations, where the unknowns appear in product form. The method extends the techniques for multiple unknowns to handle more complex algebraic structures involving products (\textit{ghāta} or \textit{bhāvita}) of unknown quantities.
}

\parallelverse{%
अथ वा स्वर्णकल्पनायां मन्दबुद्धीनां पुनः किञ्चित् पठितः । तत् सूचयामि ॥\\[0.5em]
हरमका यस्य हृति सूक्ष्मति सोऽपि द्वितीयपदयुतातः । तेनाहतोऽस्यवर्णो रूपपदेनान्वितः कल्पः ॥ १६ ॥
}{%
``Now, for the assumption of colours (\textit{svarṇa-kalpanā}) for those of dull intellect, something further is stated; that I indicate: Rule 16: The divisor (\textit{hara}) multiplied by [some quantity] whose [result] is fine (\textit{sūkṣma}), added to the second term; having multiplied by that, the unknown (\textit{asya-varṇa}) [is obtained], combined with a root term [as] the assumption (\textit{kalpa}).''
}

\subsection*{Worked Example (p.\ 137)}

\parallelverse{%
उदाहरणम् ॥\\[0.3em]
न यदि पदं रूपाणां शिष्येत्तं तेन हरतोऽपि । नावच्छद्‌गुणो भवति न चेदेवमपि किल तथै ॥ १७ ॥
}{%
Example: ``If no root of the constants remains, then even by that [method] from the divisor; the enclosed multiplier does not become [known], and if not thus indeed, [proceed] accordingly.''
}

\paralleltext{%
हरमतिः । यस्याङ्कस्य हृतिहरमका सती सूक्ष्यतीति निश्शेषा भवति अपि च सोऽप्यङ्को हर्यो रूपपदेन च गुणितो हरमकः सन् सूक्ष्यति तदा तेनाहतोऽस्यवर्णस्तेन रूपाणां घनमूलं न लभ्यते तदा तेन रूपेषु हरतोऽपि तावद्‌गुणो भवेत् तन्मूलं रूपपदं स्यात् ।
}{%
The divisor is to be considered. That number by which the divisor, being multiplied, becomes exhausted (\textit{niśśeṣa}) --- fine (\textit{sūkṣma}) --- that number, combined with the constant term (\textit{rūpa-pada}) multiplied by the divisor, being exhausted, then having multiplied by that, the unknown; if the cube-root of the constants is not obtained by that, then by that from the constants of the divisor, a multiplier of that extent [is obtained]; its root becomes the constant term.
}

\subsection{The Bhāvita Rule (pp.\ 140--141)}

\parallelverse{%
भावितम् ॥\\[0.5em]
वर्णद्वयातिकूपेऽस्य मकरवेष्टनैकतः । एताम्या संयुताबुद्धि कल्पयो र्वैच्छया च तौ ॥ ३ ॥\\[0.3em]
वर्णद्वयो वर्गयोरन्ते क्रियते ते विपर्ययात् । समयोः पक्षयोरेकमाध्यमिकमपास्य तौ वर्णौ रूपाणि च
}{%
The section on ``Product'' (\textit{Bhāvita}): ``Two unknowns (\textit{varṇa-dvaya}), their product having a coefficient (\textit{atikuṭa}), from one side having a single enclosure (\textit{makara-veṣṭana} = bracket), having assumed any sum of their values (\textit{abhuddhi}), [proceeding] with the pair of unknowns reversed (\textit{viparyaya}), [taking] the square of two unknowns, the means (\textit{mādhyamika}) [being] equal, having removed one middle term, the two unknowns and the roots (\textit{rūpāṇi})...''
}

\subsection*{Worked Example on Bhāvita (p.\ 142)}

\parallelverse{%
उदाहरणम् ॥\\[0.3em]
चतुःषष्टिगुणयो राश्योः संयुतिर्द्वियुता तयोः । राशिघातेन तुल्येति ॥
}{%
Example: ``The sum of two numbers multiplied by sixty-four, increased by two --- that [equals] the product (\textit{rāśi-ghāta}) of those two numbers.''
}

\paralleltext{%
तत्र यथोक्तं हृते पक्षौ $\left\{ \begin{array}{l} \text{या ४ का ३ रू २} \\ \text{या।का।भा १} \end{array} \right.$
}{%
There, as stated, the two sides [are]: $\left\{ \begin{array}{l} \text{yā } 4 \text{, kā } 3 \text{, constant } 2 \\ \text{yā. kā. product } 1 \end{array} \right.$
}

\paralleltext{%
वर्णद्वयातिकूपेऽस्य १४ एतदेकेनाहतं हते जाते द्विके १, १४ । एते वर्णद्वया ४, ३ स्वैच्छया युते जाते यावच्चावककालकमाने ५, १८ वा १७, ५ । द्विकेन ५, ११ वा १०, ६ ॥
}{%
Of the two unknowns' coefficient 14, multiplied by one of these, results 1, 14 for the pair. These two unknowns 4, 3, combined at will, produce the \textit{yāvattāvat}-time values: $(5, 18)$ or $(17, 5)$. For the pair: $(5, 11)$ or $(10, 6)$.
}

\subsection{The Method of Mean Elimination (pp.\ 145--147)}

\parallelverse{%
माध्यमाविगमः ॥\\[0.5em]
आसन्नमानस्य हराशमानम् अप्राप्तिगुण्ये सहिते क्षेपेण । पूर्णे स्याताम् अन्तरराशिकाभ्यां तदा हरांशो भवतीश्रमस्य ॥ १ ॥\\[0.3em]
आसन्नमानयोरन्तरमेव भवेत् । अंशस्थाने सदा रूपं चिन्त्येत तथा धीमता ॥ २ ॥\\[0.3em]
सर्वेषामासन्नमानेषु हरांशो भवति षट् । तथोत्तरोत्तरं सूक्ष्मतामासन्नानि भवन्ति हि ॥ ३ ॥
}{%
The method of mean-elimination (\textit{mādhyamāvigama}): ``The divisor-measure (\textit{harā-śamāna}) of the approximate value (\textit{āsanna-māna}), added to the non-obtained quantity (\textit{aprāpti}) and the additive (\textit{kṣepa}), being complete (\textit{pūrṇa}) by two numbers with difference (\textit{antara-rāśi}), then the divisor-fraction (\textit{harāṃśa}) [gives] the effort. The difference of two approximate values alone [suffices]. The numerator [is] always unity (\textit{rūpa}) [at first], thus by the intelligent. For all approximate values, the divisor-fraction is six. Thus gradually (\textit{uttarottaraṃ}), more and more subtle (\textit{sūkṣmataraṃ}) approximate values are obtained.''
}

\paralleltext{%
अयं माध्यमाविगमः ``उभयनिषेधविधिः'' इत्यपि प्रसिद्धः । अत्र मध्यमपदं निष्कास्य समीकरणं सरलीक्रियते ।
}{%
This method of mean-elimination is also known as ``\textit{ubhaya-niṣedha-vidhi}'' (method of double suppression). Here, the middle term is eliminated and the equation is simplified.
}

\subsection{Navīna-mati-viśeṣaḥ (pp.\ 153--157)}

\paralleltext{%
अयं ``नवीनमतिविशेषः'' नामकः विभागः समीकरणसाधनायाः सन्निकृष्टमूलविधानं (आसन्नमूलम्) समाविशति । अत्र अनिर्मितवर्गसंख्यानां मूलानि सन्निकृष्टतया ज्ञातुं पुनरावृत्तिविधिः प्रदर्श्यते --- इदं न्यूटनस्य पद्धतेः पूर्वगामी अस्ति ।
}{%
The section on ``Special Methods for New Understanding'' (\textit{navīna-mati-viśeṣaḥ}) presents the \textit{āsanna-mūla} (approximate root) method --- an iterative technique for finding successively better approximations to irrational square roots, a remarkable precursor to what is now known as Newton's method.
}

\parallelverse{%
नवीनमतिविशेषः ॥\\[0.5em]
निरक्ष पदं यद्‌द्विगुणात् स्यात् फलार्थं धनार्थं तदेवात्र शेषं तदस्यम् । पदद्वयं धनं शेषहृतप्रमन्यत् फलं तद्‌भवत् शेषमूलं धनेन ॥ १ ॥\\[0.3em]
धनार्थं नवं तस्य कृत्या विहीनो गुणः शेषमकोऽस्यवर्णस्य मानम् । मूद्‌गुसेव मते यदा शेषमानं भवेद्‌द्वितीयं तदा लव्धितौ ते ॥ २ ॥
}{%
Rule 1: ``A number (\textit{nirakṣa pada}), when doubled (\textit{dviguṇāt}), becomes the root; for the fruit (\textit{phala}) [result], for wealth (\textit{dhana}) [coefficient], that same residue (\textit{śeṣa}) here --- its two terms (\textit{pada-dvayaṃ}) [are] the wealth divided by the residue (\textit{śeṣa-hṛta-pramāṇa}); the fruit [is] the root of the residue (\textit{śeṣa-mūla}) multiplied by the wealth. Rule 2: The new wealth (\textit{dhana-artha}) is diminished by nine times its square (\textit{kṛtyā vihīna}); the multiplier (\textit{guṇa}), diminished [thus], [gives] the measure (\textit{māna}) of the unknown (\textit{asya-varṇa}). When by the opinion of Mūdgu [i.e., by the method of approximation], the residue-value becomes the second [term], then the quotient (\textit{labdhi}) for both.''
}

\subsection{The Method of Approximate Roots (pp.\ 160--165)}

\paralleltext{%
अयं विभागः ``आसन्नमूल'' विधिं विस्तारेण व्याख्याति । अनिर्मितवर्गसंख्यायाः वर्गमूलं ज्ञातुं इयं पुनरावृत्तिपद्धतिः उपयुज्यते । एषा ``चक्रवालपद्धतिः'' (cyclic method) इति अपि प्रसिद्धा ।
}{%
This section explains the \textit{āsanna-mūla} (approximate root) method in detail. This iterative procedure is used to find the square root of a non-square number. It is also known as the ``\textit{cakravāla-paddhati}'' (cyclic method) of Bhāskara.
}

\parallelverse{%
आसन्नमूलानि सूत्राणि ॥\\[0.5em]
आसन्नमानस्य हराशमानम् अप्राप्तिगुण्ये सहिते क्षेपेण । पूर्णे स्याताम् अन्तरराशिकाभ्यां तदा हरांशो भवतीश्रमस्य ॥ १ ॥
}{%
Rules for approximate roots (\textit{āsanna-mūlāni sūtrāṇi}): ``The divisor-measure of the approximate value, added to the non-obtained quantity and the additive, being complete by two numbers with difference, then the divisor-fraction [gives] the effort.''
}

\paralleltext{%
अस्य गणितीयप्रक्रिया एतादृशी --- अनिर्मितवर्गसंख्या $n$ दत्ता, $\sqrt{n}$ ज्ञातुं:
}{%
The mathematical procedure is as follows: given a non-square number $n$, to find $\sqrt{n}$:
}

\begin{enumerate}[label=\arabic*.]
  \item Choose an initial approximation $a_1$ (the ``approximate value,'' \textit{āsanna-māna})
  \item Compute the residue (\textit{śeṣa}): $r = n - a_1^2$
  \item Form the next approximation: $a_2 = a_1 + \dfrac{r}{2a_1}$
  \item Iterate for successively better approximations
\end{enumerate}

\paralleltext{%
एषा पद्धतिः आसन्नमूलानां क्रमशः सूक्ष्मतराणि मूलानि प्रदातुं शक्नोति ।
}{%
This method can provide successively more subtle (\textit{sūkṣmatara}) approximations to the root.
}

\vspace{0.5em}
\begin{center}
\fbox{%
\begin{minipage}{0.85\textwidth}
\centering
\dev{%
कल्प्यते---\\[0.3em]
(१) $0$, अ, अ', अ'', $\cdots\cdots\cdots$\\[0.2em]
(२) $1$, सो, सो', सो'', $\cdots\cdots\cdots$\\[0.2em]
(३) अ, क$_१$, क$_२$, क$_३$, क$_₄$, $\cdots\cdots\cdots$}

\vspace{0.5em}
\textit{Here:} क$_1$, क$_2$, क$_3$ --- are the approximate measures\\
of the approximate root: $\dfrac{\text{प}}{\text{ल}}$, $\dfrac{\text{प}'}{\text{ल}'}$, $\dfrac{\text{प}''}{\text{ल}''}$, etc.
\end{minipage}%
}
\end{center}

\paralleltext{%
अत्र क$_१$, क$_₂$, क$_₃$,---अयबवदासन्नमूलस्यासन्नमानानि $\frac{\text{प}}{\text{ल}}$, $\frac{\text{प}'}{\text{ल}'}$, $\frac{\text{प}''}{\text{ल}''}$ चेति ।
}{%
Here \textit{kā}$_1$, \textit{kā}$_2$, \textit{kā}$_3$ --- are the approximate values of the approximate root: $\frac{\text{pa}}{\text{la}}$, $\frac{\text{pa}'}{\text{la}'}$, $\frac{\text{pa}''}{\text{la}''}$, etc.
}

\begin{center}
\fbox{$\displaystyle \sqrt{n} = \frac{\frac{\sqrt{n+\text{अ}_₁}}{\text{सो}'} \cdot \text{प} + \text{प}}{\frac{\sqrt{n+\text{अ}_₁}}{\text{सो}''} \cdot \text{ल} + \text{ल}} = \frac{\text{प}(\sqrt{n+\text{अ}_₁}) + \text{सो}''\text{प}}{\text{ल}(\sqrt{n+\text{अ}_₁}) + \text{सो}''\text{ल}}$}
\end{center}
