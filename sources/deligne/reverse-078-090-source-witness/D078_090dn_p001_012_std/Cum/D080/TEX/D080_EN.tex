\begin{center}
{\Large\bfseries Numerical equivalence, cohomological equivalence,\\
and Lefschetz theory for abelian varieties over finite fields}\\[0.6em]
{\large P. Deligne (written up by L. Clozel)}\\[0.35em]
{\itshape To Jean-Pierre Serre, in testimony of our admiration}
\end{center}

\textbf{Abstract.} We determine the group of automorphisms generated, inside the endomorphisms of the cohomology of an abelian variety over a finite field, by the Lefschetz operators and the complex multiplications.

\textbf{Keywords.} Algebraic geometry (abelian varieties, étale cohomology); 14F20, 14K22.

\footnotetext{Written from a note of Deligne and a letter of 12 July 1997.}

\heading{1.}
Let $k$ be an algebraic closure of the field $\Fp$ with $p$ elements. We write $\ell$ for a prime number different from $p$. We fix, over $k$, an isomorphism $\Q_\ell\simeq \Q_\ell(1)$.

Let $A$ be an abelian variety over $k$. Write $Z$, or $Z(A)$, for the group of cycles with rational coefficients on $A$, graded by codimension:
\[
 Z=\bigoplus_i Z^i .
\]
On $Z^i$ one has numerical equivalence and cohomological equivalence, for étale cohomology with coefficients in $\Q_\ell$. Cohomologically equivalent cycles are numerically equivalent.

Let $A_j$ be the simple factors of $A$, let $E_j$ be the centre of $\End(A_j)\otimes\Q$, and let $F$ be the compositum, in an algebraic closure of $\Q$, of the Galois closures of the $E_j$. Thus $F$ is either a totally real field or a CM field. If $\ell$ is a prime, let $D_\ell\subset\Gal(F/\Q)$ be a decomposition group at $\ell$; let $c\in\Gal(F/\Q)$ be complex conjugation.

\begin{theorem*}[1]
If $c\in D_\ell$, then numerical equivalence and cohomological equivalence, for $\ell$-adic étale cohomology, coincide for $A$.
\end{theorem*}

A less precise theorem, in which the condition on $\ell$ depended, for a simple variety, on the choice of a maximal field of complex multiplication, was proved in [2].

The following remarks are evident:
\begin{enumerate}[label=(\alph*)]
\item Once $A$ is fixed, the condition of Theorem 1 is satisfied, by the Chebotarev theorem, for a set $L$ of primes $\ell$ of non-zero analytic or naive density.
\item If the condition of Theorem 1 is satisfied for $A$, it is also satisfied for the powers $A^n$ of $A$. Hence there is a set $L$, of density $>0$, such that the conclusion holds for every power $A^n$. This was observed by Milne [5].
\end{enumerate}

Put
\[
H=\bigoplus_{i=0}^{2g}H^i(A,\Q_\ell)=\bigoplus_{i=0}^{2g}H^i,
\qquad g=\dim(A),
\]
and put $D=\End(A)\otimes\Q$. The group $D^\times$ acts on $H$, as does the group, isomorphic to $\SL(2,\Q)$, supplied by Lefschetz theory. Both preserve algebraic cohomology classes. Let $G\subset\GL(H)$ be the Zariski closure of the group generated by $D^\times$ and $\SL(2)$. The key point in our proof of Theorem 1 is a description of $G$, or rather of the group $G_{\Q_\ell}$ over $\Q_\ell$ obtained from it by extension of scalars. In \S4 we shall show how this description reduces to the following two cases.

First let $E$ be a supersingular elliptic curve and let $A=E^n$. The group $\End(A)^\times$, acting on
\[
H\otimes\Q_\ell=\Lambda^\bullet(H^1\otimes\Q_\ell),
\]
has Zariski closure $\GL(H^1\otimes\Q_\ell)$. Let $V=H^1\otimes\Q_\ell$ and let $V^*$ be its dual.

\begin{theorem*}[2]
The group $G_{\Q_\ell}$ is isomorphic to the group $\CSpin(V\oplus V^*)$ of spin similitudes, where $V\oplus V^*$ is endowed with the natural quadratic form. It acts on
\[
H\otimes\Q_\ell=\bigoplus_i H^i(A,\Q_\ell)
\]
by the sum of its two half-spin representations. In particular, the representation on even-degree classes, respectively odd-degree classes, is irreducible.
\end{theorem*}

At the other extreme, let $A$ be an irreducible variety such that $\End(A)\otimes\Q=D$ is a division algebra, of reduced degree $n$, over an imaginary quadratic field $E$; $A$ has dimension $n$.

Let $c\in\Gal(E/\Q)$ be complex conjugation, and choose an embedding $j$ of $E$ into $\Q_\ell$. The $E$-module structure of $H^1(A,\Q_\ell)$ gives a decomposition
\[
H^1=H'_1\oplus H''_1,
\]
where $\lambda\in E$ acts on $H'_1$, respectively on $H''_1$, by multiplication by $j(\lambda)$, respectively by $j(c\lambda)$. Let $V=V_1\oplus V_2$ be the dual decomposition of $H^1(A,\Q_\ell)$. The Zariski closure of $D^\times$ becomes, over $\Q_\ell$, the group $\GL(V_1)\times\GL(V_2)$.

Endow
\[
H^\bullet(A,\Q_\ell)=\bigoplus_{p,q}\Lambda^p V_1\otimes\Lambda^q V_2
\]
with the grading given by $k:=p-q$:
\begin{equation}\tag{1.1}
\gr_k=\bigoplus_{p-q=k}\Lambda^p V_1\otimes\Lambda^q V_2 .
\end{equation}
The part of degree $-n$ is therefore $\Lambda^0V_1\otimes\Lambda^nV_2$, of dimension $1$, and the part of degree $-n+1$ is
\[
V_1\otimes\Lambda^nV_2\oplus\Lambda^0V_1\otimes\Lambda^{n-1}V_2,
\]
of dimension $2n$.

\begin{theorem*}[3]
\begin{enumerate}[label=(\roman*)]
\item For an isomorphism from $\bigoplus\Lambda^pV_1\otimes\Lambda^qV_2$ to $\Lambda^\bullet(\gr_{-n+1})$ carrying $\gr_k$ to $\Lambda^{n+k}\gr_{-n+1}$, $G_{\Q_\ell}$ identifies with the group generated by $\GL(\gr_{-n+1})$ acting on $\Lambda^\bullet\gr_{-n+1}$ and by the homotheties $x\mapsto\lambda x$.
\item The $\gr_k$ ($-n\le k\le n$) are therefore irreducible, pairwise non-isomorphic representations of $G$. The subspaces $\gr_k$ and $\gr_\ell$ are orthogonal for the cup product if $k+\ell\ne0$, and are in perfect duality if $k+\ell=0$.
\end{enumerate}
\end{theorem*}

\heading{2.}
We explain how to reduce Theorem 1, and even a slightly stronger variant, cf. \S5, to the analysis of the group $G$.

\begin{enumerate}[label=(\alph*)]
\item Let $Z_0$ be the $\Q$-vector subspace of $H^\bullet(A,\Q_\ell)$ which is the image of $Z$, and let $Z_1$ be the $\Q_\ell$-vector subspace generated by $Z_0$. We shall show that the intersection form is non-degenerate on $Z_1$. Its restriction, which is rational, to $Z_0$ therefore has zero kernel: it is non-degenerate and induces a surjection from $Z_1$ to $\Hom(Z_0,\Q_\ell)$. Hence
\[
\dim_{\Q_\ell}(Z_1)\ge \dim_{\Q}(Z_0),
\]
the elements of a basis of $Z_0$ are linearly independent over $\Q_\ell$, and $Z_0\otimes\Q_\ell\simeq Z_1$.
\item Let $\Q_\lambda$ denote a sufficiently large finite Galois extension of $\Q_\ell$. It is then enough to check the non-degeneracy of the intersection form on
\[
Z_\lambda:=Z_1\otimes_{\Q_\ell}\Q_\lambda\subset H_\lambda:=H^\bullet(A,\Q_\lambda).
\]
\item Let $L$ be the Lefschetz operator on $H$ obtained from a polarization of $A$, and let $\Lambda:H^i\to H^{i-2}$ be the associated operator [4]. The exponentials of the nilpotent endomorphisms $L$ and $\Lambda$ of $H$ generate a subgroup $S$, isomorphic to $\SL(2,\Q)$. Let $G$ be the Zariski closure, in $\GL(H)$, of the subgroup generated by $S$ and by $(\End(A)\otimes\Q)^\times$. It preserves $Z_1$, hence $Z_\lambda$; the fact that $\Lambda$ preserves algebraic cycles is due to Lieberman, cf. [4]. Since $G$ contains the group $\Gm$ corresponding to the grading of $H^\bullet(A)$, via the action of $(\Q\cdot1)^\times\subset\End(A)\otimes\Q$, and also the group corresponding to the same grading shifted by $\dim(A)$, via the diagonal matrices in $S$, it contains the homotheties $x\mapsto \lambda x$.
\item Assume that $\Q_\lambda$ is a sufficiently large Galois extension of $\Q_\ell$. We shall check that $H_\lambda$ is a sum of absolutely irreducible, pairwise non-isomorphic representations $W$ of $G$, and that the intersection form defines non-degenerate pairings on $W\times W'$, where $W\mapsto W'$ is an involution of the factors. Assume that there exists $\sigma\in\Gal(\Q_\lambda/\Q_\ell)$ which exchanges the dual factors. Since $Z_\lambda$ is stable under the Galois group, the intersection form is then non-degenerate; this will finish the proof.
\end{enumerate}

Let $E$ be the centre of $\End(A)\otimes\Q$: $E$ is a product of CM fields $E_i$; we agree to count a totally real field as CM. Assume that $E$ has the following property, equivalent to the hypothesis $c\in D_\ell$ of Theorem 1:
\begin{equation}\tag{2.1}
\begin{gathered}
\text{there are a Galois extension }\Q_\lambda/\Q_\ell\text{ and }\sigma\in\Gal(\Q_\lambda/\Q_\ell)\text{ such that,}\
\text{for all }i\text{ and every embedding }\tau:E_i\to\Q_\lambda,
\quad \sigma\tau=\tau\circ c,
\end{gathered}
\end{equation}
where $c$ is complex conjugation on $E_i$. We shall then verify below the existence of $\sigma$ exchanging $W$ and $W'$. Under this hypothesis, Theorem 1 follows.

\heading{3.}
We recall some linear algebra, referring to [3], based on [1]. Let $V$ be a vector space over a field $K$ (here $\Q_\ell$, $\Q_\lambda$, or $\overline{\Q}_\ell$), let $V^*$ be its dual, and put on $W=V\oplus V^*$ the non-degenerate quadratic form
\[
Q(v+\lambda)=\langle\lambda,v\rangle\qquad (v\in V,\ \lambda\in V^*).
\]
Let $C$ be the associated Clifford algebra, the quotient of the tensor algebra $T(V\oplus V^*)$ by the ideal generated by the elements $x\cdot x-Q(x)$, $x\in V\oplus V^*$. It is a $\Z/2\Z$-graded algebra [3]; write this grading as $C=C^+\oplus C^-$. Let $W$ act on the exterior algebra $\Lambda^\bullet V$ of $V$ by
\begin{equation}\tag{3.1}
\begin{aligned}
v\cdot x&=v\wedge x &&(v\in V,
\ x\in\Lambda^\bullet V),\\
\lambda\cdot x&=\lambda\perp x=i(\lambda)x &&(\lambda\in V^*,\ x\in\Lambda^\bullet V).
\end{aligned}
\end{equation}
The defining relation of $C$ is satisfied, and (3.1) therefore extends to a $C$-module structure
\begin{equation}\tag{3.2}
C\longrightarrow \End(\Lambda^\bullet V).
\end{equation}
The morphism (3.2) is an isomorphism: see [1], Thm. 2; alternatively, observe that if $V=V'\oplus V''$, then $C$ is the tensor product, in the $\Z/2$-graded sense, of the analogous algebras $C'$ and $C''$, reducing one to the easy case where $V$ has rank $1$. The image of $C^+$ under (3.2) is the algebra of endomorphisms of $\Lambda^\bullet V$ respecting the $\Z/2$-grading of $\Lambda^\bullet V$.

The group $C^\times$ acts on $C$ by conjugation. The subgroup $H$ normalizing $W\subset C$ respects the $\Z/2$-grading of $C$, and hence consists of even or odd elements. Its action on $W$ defines $H\to\OO(W)$, whose kernel is the centre of $C^\times$. The subgroup $(C^+)^\times\cap H$ of $(C^+)^\times$ which normalizes $W$ is the group of spin similitudes $\CSpin(W)$. It maps to $\SO(W)$:
\[
1\longrightarrow\Gm\longrightarrow\CSpin(W)\longrightarrow\SO(W)\longrightarrow1.
\]
Thus $\CSpin(W)$ is the group of automorphisms of $\Lambda^\bullet V$ respecting the $\Z/2$-grading and the subspace $W$ of $\End(\Lambda^\bullet V)$. Its Lie algebra is the algebra of even endomorphisms $T$ of $\Lambda^\bullet V$ such that, with $W$ identified with a subspace of $\End(\Lambda^\bullet V)$, one has $[T,W]\subset W$. The representation $\Lambda^\bullet V$ of $\CSpin(W)$ is the sum of the two half-spin representations.

It is clear that $\GL(V)$, acting on $\Lambda^\bullet V$, is a subgroup of $\CSpin(W)$. In $\CSpin$ it lifts the Levi subgroup of $\SO(W)$, the intersection of the opposite parabolic subgroups preserving the maximal isotropic subspaces $V$ and $V^*$ of $W$.

The maximal isotropic subspace $V^*\subset W\subset C$ and the line $K\cdot1$ in $\Lambda^\bullet V$ determine one another: $V^*$ is the subspace of $W$ annihilating $1$, and $K\cdot1$ is the subspace of $\Lambda^\bullet V$ annihilated by $V^*$. By the same rule, $V$ corresponds to the line $\Lambda^{\max}V$.

The same story repeats for any maximal isotropic subspace of $W$. The case of interest to us is the case where $V=V_1\oplus V_2$, and where $W$ is decomposed as the sum of two maximal isotropic subspaces by
\[
W=(V_1\oplus V_2^*)\oplus(V_1^*\oplus V_2).
\]
The annihilator of $V_1^*\oplus V_2$ is the line $\Lambda^{\max}V_2\subset\Lambda^\bullet V$. Fix a generator $\Omega$ of it. The inclusion in $W$ of the complementary isotropic subspace $V_1\oplus V_2^*$ extends to a morphism of algebras
\[
\Lambda^\bullet(V_1\oplus V_2^*)\longrightarrow C,
\]
and
\begin{equation}\tag{3.3}
I:\Lambda^\bullet(V_1\oplus V_2^*)\longrightarrow\Lambda^\bullet V,
\qquad \lambda\longmapsto \lambda\Omega
\end{equation}
is an isomorphism. The super-commutator with $x\in W$ stabilizes $\Lambda^\bullet(V_1\oplus V_2^*)\subset C$ and induces a super-derivation. For $x\in V_1^*\oplus V_2$, identified with the dual of $V_1\oplus V_2^*$, this derivation is interior product $x\perp$. Through the isomorphism $I$, the action of $v\in V_1\oplus V_2^*\subset W$ therefore becomes exterior product, while that of $x\in V_1^*\oplus V_2\subset W$, which annihilates $\Omega$, becomes interior product.

The exterior product in $\Lambda^\bullet(V_1\oplus V_2^*)$, transported by $I$, defines a new product, denoted $*$, on $\Lambda^\bullet V$. If $\Lambda^\bullet(V_1\oplus V_2^*)$ is graded by putting $V_1$ in degree $1$ and $V_2^*$ in degree $-1$, the product is compatible with the grading. The isomorphism $I$ transports it to the evident grading of $\Lambda^\bullet V$, shifted by $\dim(V_2)$ so as to put $\Omega$ in degree $0$. Thus the product $*$ is compatible with the shifted evident grading. The natural grading of $\Lambda^\bullet(V_1\oplus V_2^*)$, for which $V_2^*$ has degree $1$, is also compatible with the product. One then obtains on $\Lambda^\bullet V$ the grading $\gr_k$ of (1.1), shifted by $\dim(V_2)$; $1\in\Lambda^\bullet V_2$, of degree $0$ for $\gr_k$, has degree $\dim(V_2)$.

For $v_1\in V_1$ and $v_2\in V_2$, compute
\[
(v_1\wedge v_2)\wedge:\Lambda^\bullet V\longrightarrow\Lambda^\bullet V
\]
using the isomorphism (3.3). One has the diagram
\[
\begin{array}{ccc}
\Lambda^\bullet(V_1\oplus V_2^*) & \xrightarrow{\ I\ } & \Lambda^\bullet V \\
\Big\Vert && \Big\Vert \\
\Lambda^\bullet V_1\otimes\Lambda^\bullet V_2^* & \longrightarrow & \Lambda^\bullet V_1\otimes\Lambda^\bullet V_2 .
\end{array}
\]
and the bottom map is a tensor product of maps. If $a\in\Lambda^\bullet V_1$ and $b\in\Lambda^\bullet V_2$, then
\[
(v_1\wedge v_2)(a\wedge b)=(-1)^{\deg a}(v_1\wedge a)\wedge(v_2\wedge b).
\]
For the isomorphism $\Lambda^\bullet V_2^*\to\Lambda^\bullet V_2$ defining $I$, the map $b\mapsto v_2\wedge b$ is transported to $b^*\mapsto v_2\perp b^*$. It follows that $(v_1\wedge v_2)\wedge$ is transported to the endomorphism $v_1\wedge v_2\perp$ of $\Lambda^\bullet(V_1\oplus V_2^*)$, which is immediately checked to be a derivation. This derivation preserves $V_1\oplus V_2^*$.

\heading{4.}
Computing the group $G$ of \S1, over a sufficiently large extension of $\Q_\ell$, is amusing. Here is the story.

\begin{enumerate}[label=(\alph*)]
\item Let $E$ be the centre of $D=\End(A)\otimes\Q$, a product of CM fields $E_i$; $D$ is central simple over $E$.
\item After extending scalars from $\Q$ to a sufficiently large $\Q_\lambda$, $E$ becomes a sum of copies of $\Q_\lambda$, indexed by the set $\Sigma$ of morphisms of $\Q$-algebras $E\to\Q_\lambda$. After extending scalars from $\Q_\ell$ to $\Q_\lambda$, $H^1(A)$ becomes $\bigoplus H_\sigma$, $\sigma\in\Sigma$, where
\[
H_\sigma=\{\alpha\in H^1(A,\Q_\lambda):[z]\alpha=\sigma(z)\alpha\ (z\in E)\},
\]
where $[z]$ denotes the action of $E$ on $H^1(A)$ obtained from its action on $A$. The algebra $D\otimes\Q_\lambda$ is
\[
\prod_\sigma \End(H_\sigma).
\]
The group $G$, acting on $\Lambda^\bullet(\bigoplus_\sigma H_\sigma)$, is generated by the product of the $\GL(H_\sigma)$ and by a ``diagonal'' $\SL(2)$. Let
\[
L:H^i(A,\Q_\lambda)\longrightarrow H^{i+2}(A,\Q_\lambda)
\]
be the Lefschetz operator, identified with a cohomology class in
\[
H^2(A,\Q_\lambda)=\Lambda^2\Bigl(\bigoplus_\sigma H_\sigma\Bigr).
\]
Lemma 2.1 of [2] implies that
\[
L=\sum_v L_v,
\]
where $v$ runs through the quotient set $\Sigma/\{1,c\}$, complex conjugation acting on $\Sigma$ through its action on $E$, and where
\[
L_v\in\Lambda^2(H_\sigma\oplus H_{c\sigma})
\]
for $v=\{\sigma,c\sigma\}$. It may happen that $\sigma$ is fixed by $c$, in which case $L_v\in\Lambda^2H_\sigma$. Each $L_v$ lies in the closure of the orbit of $L$ under $\prod\GL(H_\sigma)$, and hence in the Lie algebra of $G$.

The same arguments give an analogous decomposition of the operator $\Lambda$; the commutation relations are preserved by the components $L_v$ and $\Lambda_v$. Consequently, $G$ is also generated by the product of the $\GL(H_\sigma)$ and by one copy of $\SL(2)$, defined over $\Q_\lambda$, for each orbit of $c$ in $\Sigma$. Thus $G$ is a product acting on a tensor product.

\item Story for $\{\sigma\}$, $\sigma=c\sigma$. This corresponds to a possible factor $\Q$ of $E$, hence to a factor of $A$ isogenous to $B=E^n$, where $E$ is a supersingular elliptic curve over $k$.

Let $V=H_\sigma=H^1(B,\Q_\lambda)$, a vector space of even dimension. Let $W=V\oplus V^*$. The Lefschetz class $L\in\Lambda^2V$ is an isomorphism from $V$ to $V^*$. We shall show that the group generated by $\GL(V)$ and by the subgroup $\SL(2)$ associated with $L$ is $\CSpin(W)$ in its spin representation on $H^\bullet(B)=\Lambda^\bullet V$.

It is immediate that, for its action on $\Lambda^\bullet V$ by left multiplication, $L$ satisfies
\[
[L,W]\subset W.
\]
Therefore $L$ is an element of $\Lie(\CSpin(W))$; it lies in the Lie algebra of the unipotent radical $N$ of the Levi subgroup defined by $W=V\oplus V^*$ (\S3), and is a non-singular element of it. The conjugates, by $\GL(V)$, of the exponential of $L$ therefore generate $N$. Similarly, $\Lambda$ and its conjugates generate the Lie algebra of the opposite unipotent subgroup. Consequently the group generated by $\GL(V)$ and $\SL(2)$ contains the Spin group; we have already seen that it contains the homotheties (\S2, (c)). Its Zariski closure, over $\Q_\ell$, is therefore $\CSpin(V\oplus V^*)$, proving Theorem 2.

\item Story for $\{\sigma,c\sigma\}$, $\sigma\ne c\sigma$. In this case
\[
H_v=H_\sigma\oplus H_{c\sigma}=V_1\oplus V_2,
\]
and $(\End(A)\otimes\Q_\lambda)^\times$ acts on $H_v$ by $\GL(V_1)\times\GL(V_2)$.

Write, as in \S3,
\[
\Lambda^\bullet(V_1\oplus V_2)\xrightarrow[\ I^{-1}\ ]{\simeq}
\Lambda^\bullet(V_1\oplus V_2^*).
\]
Endow $\Lambda^\bullet(V_1\oplus V_2)$ with the product $*$, which under $I^{-1}$ becomes the exterior product on the right. One can write ([2, Lemma 2.1])
\[
L=\sum e_i\wedge f_i\in\Lambda^2(V_1\oplus V_2),
\]
with $e_i\in V_1$ and $f_i\in V_2$. The isomorphism $I^{-1}$ exchanges the action of $L$ on $\Lambda^\bullet(V_1\oplus V_2)$ with the action on the right-hand side given by
\[
\sum e_i\wedge f_i\perp.
\]
This is a derivation (\S3); in fact, in the bases of $V_1$ and $V_2^*$ given by $(e_i,f_i^*)$, it is the sum of the endomorphisms
\[
x=\begin{pmatrix}0&1\\0&0\end{pmatrix}\in\mathfrak{sl}(2)
\]
of $k^2$, hence the corresponding endomorphism of $k^{2n}$ extended, as a Lie-algebra action, to
\[
\Lambda^\bullet(V_1\oplus V_2^*)=\Lambda^\bullet(k^{2n}).
\]
Thus the action of $L$ on $\Lambda^\bullet(V_1\oplus V_2)$ leaves the product $*$ invariant, in the Lie-algebra sense; it is therefore a derivation.

The element
\[
h=\begin{pmatrix}1&0\\0&-1\end{pmatrix}
\]
of $\mathfrak{sl}(2)$ acts on $\Lambda^iV$ by $i-n=\gr_1(\Lambda^iV)$, where $\gr_1$ is the shifted natural grading (\S3). Since this grading is compatible with the product $*$, the product is invariant under the Borel subalgebra of $\mathfrak{sl}(2)$. But in a representation of $\mathfrak{sl}(2)$, a vector killed by a Borel subalgebra is killed by all of $\mathfrak{sl}(2)$. It follows that $*$ is invariant under $\SL(2)$.

Consider the decomposition of $\Lambda^\bullet V$ given by (1.1):
\[
\gr_k\Lambda^\bullet V=\bigoplus_{p-q=k}\Lambda^pV_1\otimes\Lambda^qV_2.
\]
This decomposition is invariant under $L$ and $\Lambda$. Indeed, $E$, through its embedding in $\Q_\lambda$, acts on $\Lambda^\bullet V$. Let $U(1)$ be the group
\[
\{\lambda\in E:N_{E/F}\lambda=1\},
\]
where $F$ is the subalgebra of $E$ formed by the fixed points of $c$. If $\lambda\in U(1)$, then $\lambda$ acts on $\gr_k$ by the scalar $\lambda^{p-q}\in\Q_\lambda$; it commutes with $L$ and $\Lambda$, and the $\lambda^{p-q}$ are distinct characters of $U(1)$. Summarizing:
\end{enumerate}

\begin{lemma*}[4.1]
Endow $\Lambda^\bullet V$ with the product $*$ and the grading (1.1).
\begin{enumerate}[label=(\roman*)]
\item $\Lambda^\bullet V$ is generated by
\[
\gr_{1-n}=V_1\otimes\Lambda^nV_2\oplus\Lambda^0V_1\otimes\Lambda^{n-1}V_2.
\]
\item The product $*: \gr_i\times\gr_j\to\gr_{i+j}$ is invariant under $\SL(2)$, $\SL(V_1)$, and $\SL(V_2)$.
\item The grading of $\Lambda^\bullet V$ is preserved by $G$.
\end{enumerate}
\end{lemma*}

The image of $G$ in $\GL(\gr_{1-n})$ contains $\GL(V_1)$, $\GL(V_2)\simeq\GL(\Lambda^{n-1}V_2)$, as well as an isomorphism $\Lambda^0V_1\otimes\Lambda^nV_2\to V_1\otimes\Lambda^nV_2$ supplied by $L$ and an inverse isomorphism supplied by $\Lambda$. It is therefore $\GL(\gr_{1-n})$. Property (i) identifies $\Lambda^\bullet V$ with $\Lambda^\bullet(\gr_{1-n})$, compatibly with the action of $\SL(2)\times\SL(V_1)\times\SL(V_2)$. The group generated by $\{zs_1,z^{-1}s_2\}$, where $z$ is a scalar and $s_i\in\SL(V_i)$, in $\GL(V_1)\times\GL(V_2)$ acts on $\Lambda^\bullet V$ by the action of $\GL(\gr_{1-n})$ described in Theorem 3. Thus $G$ is generated by $\GL(\gr_{1-n})$, isomorphic to $\GL(2n)$, and by a one-dimensional torus, acting by global homotheties on $\Lambda^\bullet V$. This proves Theorem 3 (i). It is clear that, already for the action of $\GL(\gr_{1-n})$, the representations obtained are absolutely irreducible and pairwise non-isomorphic. Finally, if $k+\ell\ne0$, the spaces $\gr_k$ and $\gr_\ell$ are not dual representations, even for the action of $\GL(2n)\subset G$. Since the cup product is non-degenerate, $\gr_k$ and $\gr_\ell$ are therefore in perfect duality.

\heading{5.}
Once the description of $G$ is known, the proof of Theorem 1 is as follows. Write
\[
H_\lambda=H^\bullet(A,\Q_\lambda)=\bigotimes_{v\in\Sigma/c}H_v^\bullet,
\]
where $H_v^\bullet$ is as in \S4. Assume $\Q_\lambda$ sufficiently large and Galois over $\Q_\ell$. Under the action of the product, now denoted $G$, of the groups described in \S4, $H_\lambda$ decomposes as a direct sum indexed by functions $\sigma\mapsto k(\sigma)$, $\sigma\in\Sigma$, where
\begin{align}
\tag{5.1} k(\sigma)&\in\Z &&\text{if }\sigma\ne\sigma c;\quad\text{then }k(\sigma c)=-k(\sigma),\\
\tag{5.2} k(\sigma)&\in\Z/2\Z &&\text{if }\sigma=\sigma c\quad\text{(possible supersingular factor).}
\end{align}
In (5.1), $k(\sigma)$ is given by Theorem 3, extended, as in \S4, to the case where $E$ is CM rather than imaginary quadratic. Note that if $\sigma$ is changed to $\sigma c$, the spaces $V_1$ and $V_2$ of \S4 are exchanged, which gives relation (5.1).

Let $K$ be the group of functions satisfying (5.1) and (5.2). Since the vector space over $\Q_\lambda$ generated by algebraic cycles is stable under $G$ and under cup product, it corresponds to a submonoid $\mathrm{Alg}$ of $K$. Moreover,
\begin{align}
\tag{5.3} \mathrm{Alg}&\subset K_0=\left\{k:\sum_{\sigma\bmod c}k(\sigma)\text{ is even}\right\},\\
\tag{5.4} H_\lambda(k)&\perp H_\lambda(\ell)\quad\text{if }k+\ell\ne0,\qquad
H_\lambda(k)\text{ and }H_\lambda(-k)\text{ are in duality},\\
\tag{5.5} \mathrm{Alg}&\text{ is stable under }\Gal(\Q_\lambda/\Q_\ell).
\end{align}
Assume that the Galois group contains an element acting as $-1$ on $K$. Then, on every $\Q_\ell$-rational subspace of $H_\lambda$ invariant under $G$, the duality given by the intersection form is non-degenerate. This is in particular the case if, with the notation of \S1, complex conjugation $c\in\Gal(E/\Q)$ belongs to the decomposition group of $\ell$, proving Theorem 1.

Slightly less is enough: in order that every stable submonoid be a subgroup of $K$ (so that the space of algebraic cycles contains, together with the space associated with $k$, its dual associated with $-k$), it is enough that $K\otimes\Q$ have no fixed line under the action of $\Gal(\Q_\lambda/\Q_\ell)$. In other words, if $\ell$ is unramified in the factors of $E$, it is enough that $\operatorname{Frob}_\ell$, acting on $\Sigma$, satisfy:
\[
\text{for every }\sigma\in\Sigma,\text{ there is }a\in\Z\text{ such that }\operatorname{Frob}_\ell^a(\sigma)=\sigma c.
\]
