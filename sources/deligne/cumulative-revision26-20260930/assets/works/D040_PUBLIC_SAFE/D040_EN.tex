\section*{1}

The purpose of this paper is to construct a duality operation for representations of a reductive group over a finite field. Its effect, very roughly speaking, is to interchange irreducible representations of small degree with ones of large degree (for example, the unit and Steinberg representation.) At level of characters, this operation has been also considered by Alvis~[1], Curtis~[2], and Kawanaka~[4].

We shall now fix some notation. $\bG$ will denote a connected reductive group defined over a finite field $\F_q$, $(W,S)$ its Weyl group, and $\bar S$ the set of orbits of the Frobenius map on $S$. The subsets of $\bar S$ parametrize the classes of parabolic subgroups of $\bG$ which are defined over $\F_q$; let $\cP_I$ be the class corresponding to $I \subset \bar S$. (Thus $\cP_\varnothing$ is the class of Borel subgroups and $\cP_{\bar S}=\{\bG\}$.) We denote by $G$ the group of $\F_q$-rational points of $\bG$ and by $\cP_I$ the set of $\F_q$-defined parabolic subgroups in that class. If $\bP \in \cP_I$, we denote by $P$ its group of $\F_q$-rational points and by $U_P$ the group of $\F_q$-rational points of its unipotent radical. Bold letters denote algebraic subgroups; the corresponding roman letters denote their groups of rational points. Let $K$ be an algebraically closed field of characteristic zero. All $G$-modules will be over $K$.

Let $E$ be a $G$-module. For each $I \subset \bar S$ we define
\begin{equation*}
E_{(I)} = \bigoplus_{\bP \in \cP_I} E^{U_P},
\end{equation*}
where $E^{U_P}$ denotes the set of $U_P$-invariant vectors of $E$. We regard $E_{(I)}$ as a

\newpage
\noindent $G$-module in a natural way. If $I \subset I'$, there is a canonical linear map $\varphi_{I'}^{I} \colon E_{(I)} \to E_{(I')}$. It is defined as follows: if $e \in E^{U_P}$ ($\bP \in \cP_I$), $e$ can be also regarded as an element of $E^{U_{P'}}$, where $\bP' \in \cP_{I'}$ is uniquely defined by the condition $\bP \subset \bP'$. (Note that $U_P \supset U_{P'}$, hence $E^{U_P} \subset E^{U_{P'}}$.)
The map $\varphi_{I'}^{I}$ takes $e \in E^{U_P}$ to $e$ regarded as an element of $E^{U_{P'}}$. It is clear that if $I \subset I' \subset I''$, then
\begin{equation}\tag{1.1}\label{eq:1.1}
\varphi_{I''}^{I'} \varphi_{I'}^{I} = \varphi_{I''}^{I}.
\end{equation}

Now let $\tilde{E}_{(I)} = E_{(I)} \otimes \Lambda^{|I|}(K^I)$. (We regard $\Lambda^{|I|}(K^I)$ as a $G$-module with trivial action.) If $I \subset I'$ and $|I'| = |I| + 1$, we have a natural map $\varepsilon_{I'}^{I} \colon \Lambda^{|I|}(K^I) \xrightarrow{\;\sim\;} \Lambda^{|I'|}(K^{I'})$ given by $\omega \mapsto \omega \wedge f$ where $f$ is the unique element in $I' - I$. We define $\widetilde\varphi_{I'}^{I} \colon \tilde{E}_{(I)} \to \tilde{E}_{(I')}$ by $\widetilde\varphi_{I'}^{I} = \varphi_{I'}^{I} \otimes \varepsilon_{I'}^{I}$.

We now consider the sequence of maps of $G$-modules:
\begin{equation}\tag{1.2}\label{eq:1.2}
0 \to \tilde{E}_{(\varnothing)} \xrightarrow{d_0} \bigoplus_{|I|=1} \tilde{E}_{(I)} \xrightarrow{d_1} \bigoplus_{|I|=2} \tilde{E}_{(I)} \xrightarrow{d_2} \cdots \to \tilde{E}_{(\bar S)} \to 0
\end{equation}
where the maps $d$ have components $\widetilde\varphi_{I'}^{I}$ ($I \subset I'$, $|I'| = |I|+1$). Using \eqref{eq:1.1}, we see that \eqref{eq:1.2} is a complex.

\section*{2}

We have the following:

\medskip\noindent\textbf{THEOREM.}\quad
Assume that $E$ is irreducible. Let $i_0$ be the smallest integer $\geq 0$ such that $\bigoplus_{|I|=i_0} \tilde{E}_{(I)} \neq 0$. Then the sequence
\begin{equation}\tag{2.1}\label{eq:2.1}
\bigoplus_{|I|=i_0} \tilde{E}_{(I)} \xrightarrow{d_{i_0}} \bigoplus_{|I|=i_0+1} \tilde{E}_{(I)} \xrightarrow{d_{i_0+1}} \cdots \to \tilde{E}_{(\bar S)} \to 0
\end{equation}
is exact.

In particular, the homology of the complex \eqref{eq:1.2} is concentrated in a single degree.

\section*{3}

We shall prove that the conclusion of Theorem~2 holds in the case where instead of assuming $E$ irreducible we shall assume that:
\begin{equation}\tag{3.1}\label{eq:3.1}
\text{$E$ admits a direct sum decomposition $E = \bigoplus_{\bP \in \cP_{I_0}} E_{\bP}$ (for some $I_0 \subset \bar S$)}
\end{equation}
such that

\newpage
\begin{enumerate}[label=(\alph*)]
\item $gE_{\bP} = E_{g\bP g^{-1}}$ ($\forall g \in G$, $\bP \in \cP_{I_0}$);
\item for each $\bP \in \cP_{I_0}$, the $P$-module $E_{\bP}$ factors through $P/U_P$ and is irreducible, cuspidal as a $P/U_P$-module.
\end{enumerate}
This would imply the theorem since any irreducible $G$-module is a direct summand of a $G$-module $E$ as in \eqref{eq:3.1}.

In this and the following section, we assume that $E$ is as in \eqref{eq:3.1}.

Given $J \subset \bar S$, we define $\cZ_J$ to be the set of $G$-orbits $\sigma$ on $\cP_J \times \cP_{I_0}$ which have the following property. There exist $\bP \in \cP_{I_0}$ and $\bQ \in \cP_J$, with $(\bQ,\bP) \in \sigma$, such that $\bQ$ contains a Levi subgroup of $\bP$.

If $\bQ \in \cP_J$, we have
\begin{equation*}
E^{U_Q}
= \Bigl(\bigoplus_{\bP \in \cP_{I_0}} E_{\bP}\Bigr)^{U_Q}
= \Bigl(\bigoplus_{\substack{\bP \in \cP_{I_0} \\ (\bQ \cap \bP)U_P = P}} E_{\bP}\Bigr)^{U_Q}
\oplus \Bigl(\bigoplus_{\substack{\bP \in \cP_{I_0} \\ (\bQ \cap \bP)U_P \subsetneq P}} E_{\bP}\Bigr)^{U_Q}.
\end{equation*}
The second summand is zero (since $E_{\bP}$ is cuspidal for $P/U_P$, so that $E_{\bP}^{U_Q \cap P} = 0$). Thus
\begin{equation}\tag{3.2}\label{eq:3.2}
E_{(J)} = \bigoplus_{\bQ \in \cP_J} E^{U_Q}
= \bigoplus_{\bQ \in \cP_J}\Bigl(\bigoplus_{\substack{\bP \in \cP_{I_0} \\ (\bQ \cap \bP)U_P = P}}E_{\bP}\Bigr)^{U_Q}
= \bigoplus_{\sigma \in \cZ_J} E^\sigma
\end{equation}
where
\begin{equation*}
E^\sigma = \bigoplus_{\bQ \in \cP_J}
\Bigl(\bigoplus_{\substack{\bP \in \cP_{I_0} \\ (\bQ,\bP) \in \sigma \\ (\bQ \cap \bP)U_P = P}} E_{\bP}\Bigr)^{U_Q}
\qquad (\sigma \in \cZ_J).
\end{equation*}

Now assume that $J \subset J'$ and $\sigma \in \cZ_J$; we define $\delta_{J'}^{J}(\sigma) = \sigma' = \{(\bQ',\bP) \in \cP_{J'} \times \cP_{I_0} \mid \exists \bQ \in \cP_J,\ \bQ \subset \bQ',\ (\bQ,\bP) \in \sigma\}$. Then $\sigma' \in \cZ_{J'}$ and we say that $\sigma'$ is the face of type $J'$ of $\sigma$. In this case, we define a linear map $\Psi_{\sigma'}^{\sigma} \colon E^\sigma \to E^{\sigma'}$ as follows. If $e = (e_{\bQ,\bP}) \in E^\sigma$, $e_{\bQ,\bP} \in E_{\bP}$ for every $(\bQ,\bP) \in \sigma$, we set $\Psi_{\sigma'}^{\sigma}(e) = (e'_{\bQ',\bP}) \in E^{\sigma'}$ where, for any $(\bQ',\bP) \in \sigma'$, we have
\begin{equation*}
e'_{\bQ',\bP} = \sum_{\substack{\bQ \in \cP_J \\ \bQ \subset \bQ' \\ (\bQ,\bP) \in \sigma}} e_{\bQ,\bP} \in E_{\bP}.
\end{equation*}

Let us admit the following

\medskip\noindent\textbf{LEMMA 3.3.}\quad
For any $\sigma, \sigma'$ as above, the map $\Psi_{\sigma'}^{\sigma} \colon E^\sigma \to E^{\sigma'}$ is an isomorphism.

Note also that given $J \subset J' \subset J''$ and $\sigma \in \cZ_J$, $\sigma' \in \cZ_{J'}$ as above we can

\newpage
\noindent define $\sigma'' \in \cZ_{J''}$ to be the face of type $J''$ of $\sigma$ (or of $\sigma'$) and we have
\begin{equation*}
\Psi_{\sigma''}^{\sigma} = \Psi_{\sigma''}^{\sigma'}\Psi_{\sigma'}^{\sigma}.
\end{equation*}

Using \eqref{eq:3.2}, the complex \eqref{eq:1.2} can be rewritten as
\begin{equation}\tag{3.4}\label{eq:3.4}
0 \to \cdots \to 0 \to
\bigoplus_{|J|=|I_0|}\ \bigoplus_{\sigma \in \cZ_J} E^\sigma \otimes \Lambda^{|J|}(K^J)
\xrightarrow{d} \bigoplus_{|J|=|I_0|+1}\ \bigoplus_{\sigma \in \cZ_J} E^\sigma \otimes \Lambda^{|J|}(K^J)
\xrightarrow{d} \cdots
\end{equation}
where the maps $d$ have components $\Psi_{\sigma'}^{\sigma} \otimes \varepsilon_{J'}^{J}$ ($J \subset J'$, $|J'| = |J|+1$, $\sigma \in \cZ_J$, $\sigma' \in \cZ_{J'}$ the face of type $J'$ of $\sigma$). Let $\rho$ be the unique element in $\cZ_{\bar S}$. We have $E^\rho = E$. Using Lemma~3.3, we may identify $E^\sigma$ with $E$ for any $\sigma \in \cZ_J$, via the isomorphism $\Psi_{\rho}^{\sigma}$; the complex \eqref{eq:3.4} becomes the tensor product of $E$ with the complex
\begin{equation}\tag{3.5}\label{eq:3.5}
0 \to \cdots \to 0 \to
\bigoplus_{|J|=|I_0|} K^{\cZ_J} \otimes \Lambda^{|J|}(K^J)
\to \bigoplus_{|J|=|I_0|+1} K^{\cZ_J} \otimes \Lambda^{|J|}(K^J)
\to \cdots
\end{equation}
whose differential has components $\delta_{J'}^{J} \otimes \varepsilon_{J'}^{J}$ ($J \subset J'$, $|J'| = |J|+1$). This complex has the following simple interpretation. Let $T$ be an $\F_q$-split torus of maximal possible dimension in the adjoint group $\bG_{\mathrm{ad}}$. Let $Y$ be the lattice of one-parameter subgroups of $T$. The root hyperplanes in the real vector space $Y \otimes \R$ give rise to a partition of $Y \otimes \R$ into simplicial cones. These are in a natural 1--1 correspondence with the parabolic subgroups in $\bG$ which are defined over $\F_q$ and whose image in $\bG_{\mathrm{ad}}$ contain $T$. Let us fix $\bP_0 \in \cP_{I_0}$. It corresponds to a cone in $Y \otimes \R$, which spans a linear subspace $L \subset Y \otimes \R$ of codimension $|I_0|$. The cones contained in $L$ are in 1--1 correspondence with those parabolic subgroups $\bQ \subset \bG$ (defined over $\F_q$) which contain a Levi subgroup of $\bP_0$ and whose image in $\bG_{\mathrm{ad}}$ contain $T$, hence they are in 1--1 correspondence with $\bigcup_J \cZ_J$. Thus $\bigcup_{J \ne \bar S} \cZ_J$ can be regarded as the set of simplices in a triangulation of a unit sphere with centre $0$ in $L$; and \eqref{eq:3.5} is the (reduced) chain complex of this triangulation. It follows that its homology is concentrated in a single degree (corresponding to $|J| = |I_0|$) where it is isomorphic to $K$.

\section*{4}

It remains to prove Lemma~3.3. We will give the proof in six steps. Parts of the argument are similar in spirit to arguments in~[3].

\smallskip
\noindent\textbf{Step 1.} Given $\sigma \in \cZ_J$ and $(\bQ,\bP) \in \sigma$, we consider the parabolic subgroup $\bR = (\bQ \cap \bP)U_Q \subset \bQ$. Then $\bR \in \cP_{\widetilde J}$ for some $\widetilde J \subset J$ and $\sigma$ is the face of

\newpage
\noindent type $J$ of some $\widetilde\sigma\in\cZ_{\widetilde J}$. Also $\bR$, $\bP$ have a common Levi subgroup and hence are associated. We first show that
\begin{equation*}
\widetilde\Psi_{\sigma}^{\widetilde\sigma}\colon
\bigoplus_{\bR\in\cP_{\widetilde J}}
  \Bigl(\bigoplus_{\substack{\bP\in\cP_{I_0}\\(\bR,\bP)\in\widetilde\sigma}}E_{\bP}\Bigr)^{U_R}
\longrightarrow
\bigoplus_{\bQ\in\cP_J}
  \Bigl(\bigoplus_{\substack{\bP\in\cP_{I_0}\\(\bQ,\bP)\in\sigma}}E_{\bP}\Bigr)^{U_Q}
\end{equation*}
is an isomorphism.

Now the pairs $(\bR,\bP)$ such that $(\bR,\bP) \in \widetilde\sigma$ are in 1--1 correspondence with the pairs $(\bQ,\bP)$ such that $(\bQ,\bP) \in \sigma$. Hence $\widetilde\Psi_{\sigma}^{\widetilde\sigma}$ is injective. If we fix $\bQ \in \cP_J$, there is a natural 1--1 correspondence between the set $\{\bR \in \cP_{\widetilde J} \mid \bR \subset \bQ\}$ and the set of orbits of $U_Q$ on the set $\{\bP \in \cP_{I_0} \mid (\bQ,\bP) \in \sigma\}$. Hence the space $E^\sigma$ is a direct sum over all pairs $\bR \subset \bQ$ ($\bR \in \cP_{\widetilde J}$, $\bQ \in \cP_J$) of pieces of form $\bigl(\bigoplus_{\bP \in B}E_{\bP}\bigr)^{U_Q}$, where $\bP$ runs through a fixed $U_Q$-orbit. But this piece has the same dimension as $E_{\bP}$. (Note that for $\bP$ in that $U_Q$-orbit we have $P \cap U_Q \subset U_P$ and $U_P$ acts trivially on $E_{\bP}$.) Thus $\dim E^\sigma = \dim E_{\bP} \cdot \#(\cP_{\widetilde J}) = \dim E_{\bP} \cdot \#(\cP_{I_0}) = \dim E$. Similarly, $\dim E^{\widetilde\sigma} = \dim E$. It follows that $\varphi_{\sigma}^{\widetilde\sigma}$ is an isomorphism.

\smallskip
\noindent\textbf{Step 2.} Assume that $|I_0| = |\bar S| - 1$. Let $\cP_J$ be the class of parabolic subgroups opposed to $\cP_{I_0}$, and let $\sigma = \{(\bQ,\bP) \in \cP_J \times \cP_{I_0} \mid \text{$\bQ$ opposed to $\bP$}\}$. We will show that $\varphi_\rho^\sigma \colon E^\sigma \to E^\rho = E$ is an isomorphism. It is certainly non-zero. Thus, if $E$ is irreducible, $\varphi_\rho^\sigma$ is indeed an isomorphism. If $E$ is reducible, it has exactly two composition factors, and $J$ is necessarily equal to $I_0$. For each $\bP \in \cP_{I_0}$, there exists an isomorphism
\begin{equation*}
\eta_{\bP} \colon E_{\bP} \xrightarrow{\;\sim\;} \Bigl(\bigoplus_{\substack{\bP'\\(\bP',\bP) \in \sigma}} E_{\bP'}\Bigr)^{U_P}
\end{equation*}
of $P/U_P$-modules unique up to a scalar. We may assume that $\eta_{g\bP g^{-1}} = g\eta_{\bP} g^{-1}$ ($\forall \bP \in \cP_{I_0}$, $\forall g \in G$). These maps together define an isomorphism of $G$-modules $\eta = \bigoplus_{\bP \in \cP_{I_0}} \eta_{\bP}$:
\begin{equation*}
E = \bigoplus_{\bP \in \cP_{I_0}} E_{\bP}
\xrightarrow{\;\sim\;}
\bigoplus_{\bP\in\cP_{I_0}}\Bigl(\bigoplus_{\substack{\bP'\\(\bP',\bP)\in\sigma}}E_{\bP'}\Bigr)^{U_P}.
\end{equation*}
The composition $\varphi_\rho^\sigma \circ \eta$ is a semisimple $G$-endomorphism of $E$. It clearly has trace zero (its diagonal blocks are zero). Let $\lambda'$, $\lambda''$ be its eigenvalues on the two irreducible pieces $E'$, $E''$ of $E$. Then $\lambda' \dim E' + \lambda'' \dim E'' = 0$. Since $\dim E' \neq 0$, $\dim E'' \neq 0$, $\lambda'$ and $\lambda''$ are either both zero or both non-zero. They cannot be both zero since $\varphi_\rho^\sigma \circ \eta \neq 0$. Hence they are both non-zero. It follows that $\varphi_\rho^\sigma \circ \eta$ and hence $\varphi_\rho^\sigma$ are isomorphisms.

\smallskip
\noindent\textbf{Step 3.} Assume that $|I_0| \leq |\bar S| - 1$. Let $J \subset \bar S$ be such that $|J| = |I_0| + 1$,

\newpage
\noindent and let $\tau \in \cZ_J$. There are precisely two elements $\sigma' \in \cZ_{J'}$, $\sigma'' \in \cZ_{J''}$ such that $J'$, $J'' \subset J$, $|J'| = |J''| = |I_0|$, and such that $\tau$ is the $J$-face of $\sigma'$ and the $J$-face of $\sigma''$. We shall prove that $\varphi_\tau^{\sigma'}$ and $\varphi_\tau^{\sigma''}$ are isomorphisms. One of these maps, say $\varphi_\tau^{\sigma'}$, is of the type considered in Step~1 ($\sigma = \tau$); hence it is known to be an isomorphism. We now turn to $\varphi_\tau^{\sigma''}$. Define
\begin{equation*}
E' = \bigoplus_{\bR' \in \cP_{J'}} E'_{\bR'}
\end{equation*}
where, by definition,
\begin{equation*}
E'_{\bR'} = \Bigl(\bigoplus_{\substack{\bP \in \cP_{I_0} \\ (\bR',\bP)\in\sigma'}} E_{\bP}\Bigr)^{U_{R'}}.
\end{equation*}
Then $E'$ is a $G$-module in a natural way. It satisfies a property similar to property \eqref{eq:3.1} for $E$. Let $\alpha$ be the set of pairs $(\bR'',\bR') \in \cP_{J''} \times \cP_{J'}$ such that $\bR''$, $\bR'$ are both contained in the same $\bQ \in \cP_J$ and are opposed in $\bQ$. We define $E'^\alpha$ and $\Psi^\alpha \colon E'^\alpha \to E'$ in the same way as $E^\sigma$ and $\varphi_\rho^\sigma \colon E^\sigma \to E$. We have a natural commutative diagram
\begin{center}
\includegraphics[width=0.43\textwidth]{assets/D040_p006_diagram_authority_400dpi.png}
\end{center}
(Note that, given $(\bR'',\bP) \in \sigma''$ there is a unique $\bR' \in \cP_{J'}$ such that $(\bR'',\bR') \in \alpha$, $(\bR',\bP) \in \sigma'$; conversely, if $(\bR'',\bR') \in \alpha$, $(\bR',\bP) \in \sigma'$ then $(\bR'',\bP) \in \sigma''$. This gives rise to the isomorphism $E'^{\alpha} \xrightarrow{\sim} E^{\sigma''}$.) Now the right-hand vertical map in the diagram is known to be an isomorphism; moreover, by Step~2, the top map is an isomorphism. It follows that the bottom map $\varphi_\tau^{\sigma''}$ is an isomorphism.

\smallskip
\noindent\textbf{Step 4.} Let $\sigma \in \cZ_J$, $|J| = |I_0|$. (Thus $\sigma$ corresponds to an open cone in $L$.) We show that $\varphi_\rho^\sigma \colon E^\sigma \to E$ is an isomorphism. We can find a sequence $\sigma = \sigma_1, \sigma_2, \dots, \sigma_n = \rho$ of elements corresponding to open cones in $L$ such that (when regarded as cones in $L$) two consecutive ones have a common face of codimension~1; moreover, we take $n$ as small as possible. Let $\tau$ be the element corresponding to the common face of codimension~1 of $\sigma = \sigma_1$ and $\sigma_2$. By induction on $n$, we may assume that $\varphi_\rho^{\sigma_2}$ is an isomorphism. From Step~3, we know that $\varphi_\tau^\sigma$ and $\varphi_\tau^{\sigma_2}$ are isomorphisms. From $\varphi_\rho^{\sigma_2}=\varphi_\rho^\tau\varphi_\tau^{\sigma_2}$, it follows that $\varphi_\rho^\tau$ is an isomorphism. From $\varphi_\rho^\sigma=\varphi_\rho^\tau\varphi_\tau^\sigma$ it follows that $\varphi_\rho^\sigma$ is an isomorphism.

\smallskip
\noindent\textbf{Step 5.} Let $J \subset \bar S$ and $\sigma \in \cZ_J$. We want to show that $\varphi_\rho^\sigma$ is an isomorphism. We can find $\widetilde J \subset J$ and $\widetilde\sigma \in \cZ_{\widetilde J}$ such that $|\widetilde J| = |I_0|$, such that $\sigma$ is the face of type $J$ of $\widetilde\sigma$, and such that $\varphi_\sigma^{\widetilde\sigma}$ is of the type considered in Step~1 (hence an isomorphism).

\newpage
\noindent We have $\varphi_\rho^\sigma\varphi_\sigma^{\widetilde\sigma}=\varphi_\rho^{\widetilde\sigma}$. By Step~4, $\varphi_\rho^{\widetilde\sigma}$ is an isomorphism. It follows that $\varphi_\rho^\sigma$ is an isomorphism.

\smallskip
\noindent\textbf{Step 6.} Let $J \subset J'$ and $\sigma \in \cZ_J$; let $\sigma' \in \cZ_{J'}$ be the face of type $J'$ of $\sigma$. We want to show that $\varphi_{\sigma'}^\sigma$ is an isomorphism. We have $\varphi_\rho^{\sigma'}\varphi_{\sigma'}^\sigma=\varphi_\rho^\sigma$. By Step~5, $\varphi_\rho^\sigma$ and $\varphi_\rho^{\sigma'}$ are isomorphisms. It follows that $\varphi_{\sigma'}^\sigma$ is an isomorphism.

This completes the proof of Lemma~3.3 and hence that of the theorem.

\section*{5}

We have

\medskip\noindent\textbf{COROLLARY (see Alvis~[1]).}\quad
In the setting of the theorem, let $E^\#$ denote the kernel of $d_{i_0}$ in \eqref{eq:2.1}. Then
\begin{enumerate}[label=(\alph*)]
\item $E^\#$ is an irreducible $G$-module,
\item $(E^\#)^\#$ is isomorphic to $E$ as a $G$-module,
\item $(-1)^{i_0} E^\# = \sum_{I \subset \bar S} (-1)^{|I|} E_{(I)}$ in the Grothendieck group of virtual $G$-modules.
\end{enumerate}

Statement (c) is obvious.

Let $E'$ be a $G$-module of the type considered in \eqref{eq:3.1} which has a direct sum decomposition $E' = E_1 \oplus \cdots \oplus E_n$ with $E_i$ irreducible $G$-modules and $E_1 \cong E$. The proof of Theorem~2 shows that $(E')^\#$ can be defined in the same way as $E^\#$, and that it has the following properties:
\begin{equation}\tag{5.1}\label{eq:5.1}
(E')^\# \cong E_1^\# \oplus \cdots \oplus E_n^\#, \qquad E'^\# \cong E'.
\end{equation}
For each $1 \leq i, j \leq n$, we have from (c)
\begin{equation}\tag{5.2}\label{eq:5.2}
\langle E_i, E_j^\# \rangle = \langle E_i^\#, E_j \rangle = \sum_{I \subset \bar S} (-1)^{|I|+i_0} \langle E_i^{U_{P_I}}, E_j^{U_{P_I}} \rangle_{P_I}
\end{equation}
(where $\bP_I \in \cP_I$). It follows that
\begin{equation*}
\langle E_i^\#, E' \rangle = \sum_j \langle E_i^\#, E_j \rangle = \sum_j \langle E_i, E_j^\# \rangle = \langle E_i, (E')^\# \rangle = \langle E_i, E' \rangle
\end{equation*}
and, in particular, $E_i^\# \neq 0$. If $E_i^\#$ is not irreducible for some $i$, then $E_1^\# \oplus \cdots \oplus E_n^\#$ would have more irreducible components than $E_1 \oplus \cdots \oplus E_n$, contradicting \eqref{eq:5.1}. Thus, $E_i^\#$ is irreducible for each $i$. Hence, there exists a permutation $\pi$ of $\{1,2,\dots,n\}$ such that $E_i^\# = E_{\pi(i)}$ for all $i$. From \eqref{eq:5.2}, we have
\begin{equation*}
\langle E_{\pi^2(i)}, E_i \rangle = \langle (E_i^\#)^\#, E_i \rangle = \langle E_i^\#, E_i^\# \rangle = \langle E_{\pi(i)}, E_{\pi(i)} \rangle = 1.
\end{equation*}
It follows that $E_{\pi^2(i)}$ is isomorphic to $E_i$. The corollary is proved.
