\subsection{F-cristaux of Hodge ordinaires of niveau $\leqslant 1$}

\textbf{Prouvons (ii).} The forms $\eta_{ij}$ being fermees, there exists, en vertu of the lemma of Poincare, of the series $\tau_{ij} \in K[[t]]$, determinees a constant pres, such that the one ait (1.4.2.6). The relation (1.4.2.5) entraine then that
\[
\varphi^*\tau_{ij} - p\tau_{ij} - pu_{ij}(\varphi) \in K,
\]
therefore
\begin{equation}\tag{1.4.2.10}
\varphi^*\tau_{ij} - p\tau_{ij} - pu_{ij}(\varphi) = (\varphi^*\tau_{ij})(0) - p\tau_{ij}(0) - pu_{ij}(\varphi)(0).
\end{equation}

Notons $x \mapsto x^\sigma$ the automorphism of Frobenius of $K$. Fixons all of abord the lifting $\varphi$ of Frobenius such that $\varphi(t_i) = t_i^p$, and normalisons $\tau_{ij}$ by the condition
\begin{equation}\tag{1.4.2.11}
(\varphi^*\tau_{ij})(0) - p\tau_{ij}(0) - pu_{ij}(\varphi)(0) = 0
\end{equation}
which s'ecrit again (car $(\varphi^*\tau_{ij})(0) = \tau_{ij}(0)^\sigma$)
\begin{equation}\tag{1.4.2.12}
\tau_{ij}(0) = \sum_{n\geqslant 1} v^n u_{ij}(\varphi)(0)
\end{equation}
(or $vx = pF^{-1}x = px^{\sigma^{-1}}$). In particular, one has (1.4.2.8). Montrons that, si $\psi$ is a lifting quelconque of Frobenius, the condition (1.4.2.7), with $\varphi$ remplace by $\psi$, is satisfaite. Compte tenu of (1.4.2.10) (with $\varphi$ remplace by $\psi$), it revient to the same of verify that the one has
\begin{equation}\tag{1.4.2.13}
(\psi^*\tau_{ij})(0) - p\tau_{ij}(0) - pu_{ij}(\psi)(0) = 0.
\end{equation}

Si $\psi(0) = 0$ (i.e. $\psi$ is compatible a the augmentation $e : W[[t]] \to W$), one has $(\psi^*\tau_{ij})(0) = \tau_{ij}(0)^\sigma$, and (1.4.2.13) equivaut a (1.4.2.12) with $\varphi$ remplace by $\psi$. It is enough therefore of verify that $u_{ij}(\psi)(0) = u_{ij}(\varphi)(0)$. Or the augmentation $e$ is the lifting of Teichm\"uller (1.1.4) of the augmentation $k[[t]] \to k$ simultanement for $\varphi$ and for $\psi$, therefore $F(\varphi)\varphi^*$ and $F(\psi)\psi^*$ induisent the same endomorphism $\sigma$-lineaire $F$ of $e^*H$. Appliquant $e^*$ a (1.4.2.4) ecrit for $\varphi$ and for $\psi$, one obtains therefore
\begin{align*}
F(e^*b_i) &= p(e^*b_i) + p\sum_{1\leqslant j\leqslant r} u_{ij}(\varphi)(0)(e^*a_j) \\
&= p(e^*b_i) + p\sum_{1\leqslant j\leqslant r} u_{ij}(\psi)(0)(e^*a_j),
\end{align*}
of or $u_{ij}(\psi)(0) = u_{ij}(\varphi)(0)$, this which proves (1.4.2.13) in this case.

\[
\psi(t_i) = \varphi_0(t_i) + pw_i \quad (1 \leqslant i \leqslant n),

with $w = (w_1, \ldots, w_n) \in W^n$, or $\varphi_0 : A \to A$ is a lifting of Frobenius such that $\varphi_0(0) = 0$. Appliquant (1.1.3.4), with $\varphi = \varphi_0$, $\psi = \psi$, $x = b_i$, one obtains, compte tenu of (1.4.2.4),

\begin{lemma}[Dwork {\cite[lemma 1]{Dwork1958}}]\label{lem:1.4.3}
Notons $\varphi$ the automorphism $\sigma$-lineaire of $K[[t]]$ (or $t = (t_1, \ldots, t_n)$ and $K$ is the field of the fractions of $W$) given by $t_i \mapsto t_i^p$. Let $f \in K[[t]]$ such that $f(0) = 1$. The conditions suivantes are equivalentes:
\begin{enumerate}[label=(\roman*)]
\item $f \in W[[t]]$,
\item $\displaystyle\frac{\varphi f}{f^p} = 1 + pu$, with $u \in W[[t]]$, $u(0) = 0$.
\end{enumerate}
\end{lemma}

the implication (i) $\Rightarrow$ (ii) is immediate. Recall the proof of (ii) $\Rightarrow$ (i). Ecrivons $f = \sum a_i t^i$ ($a_0 = 1$), and supposons proves that $a_i \in W$ for $|i| < r$ (or $|i| = i_1 + \cdots + i_n$). Notant $(~)_{<j}$ the part of degree total $< j$, one calcule
\begin{align*}
(f^p)_{<r} &= \bigl((\sum_{|i|\leqslant r-1} a_i t^i)^p\bigr)_{<r} + (\sum_{|i|=r} a_i t^i)^p \\
&= \bigl(\sum_{|i|\leqslant r-1} a_i^\sigma t^{pi}\bigr)_{<r} + p\sum_{|i|=r} a_i t^i \mod pW[[t]],
\end{align*}
then, posant $f^p/\varphi f = 1 + p\sum d_i t^i$ ($d_i \in W$),
\begin{multline*}
(\sum_{|i|\geqslant 0} a_i^\sigma t^{pi})(1 + p\sum_{|i|\geqslant 0} d_i t^i)_{<r} \\
= \bigl((\sum_{|i|\leqslant r-1} a_i^\sigma t^{pi})(1 + p\sum_{|i|\geqslant 1} d_i t^i)\bigr)_{<r} \\
= (\sum_{|i|\leqslant r-1} a_i^\sigma t^{pi})_{<r} \mod pW[[t]] \\
= (\sum_{|i|\leqslant r-1} a_i^p t^{pi})_{<r} \mod pW[[t]].
\end{multline*}

Comparant the deux expressions obtained for $(f^p)_{<r}$, one trouve
\[
p\sum_{|i|=r} a_i t^i \in pW[[t]],
\]
therefore $a_i \in W$ for every $i$ such that $|i| = r$, this which proves (ii) $\Rightarrow$ (i).

Before of revenir a the proof of 1.4.2, indiquons deux consequences immediates of \ref{lem:1.4.3}.

\begin{corollary}\label{cor:1.4.4}
Let $g \in K[[t]]$ such that $g(0) = 0$. The conditions suivantes are equivalentes:
\begin{enumerate}[label=(\roman*)]
\item $\exp(g) \in W[[t]]$,
\item $\varphi^*g - pg \in pW[[t]]$.
\end{enumerate}
\end{corollary}

Indeed, posons $f = \exp(g)$. As, for $n > 1$, $p^n/n!$ (and a fortiori $p^n/n$) appartiennent a $pW$, the condition (ii) equivaut a dire that $\varphi f/f^p = 1 + pu$, with $u \in W[[t]]$, $u(0) = 0$.

\begin{corollary}\label{cor:1.4.5}
Supposons $p \neq 2$. Let $g \in K[[t]]$, such that $g(0) \in pW$, and $\varphi^*g - pg \in pW[[t]]$. Then the serie $f = \exp(g)$ is defined, appartient a $W[[t]]$, and satisfies $f(0) \equiv 1 \mod pW$.
\end{corollary}

Indeed, let $h = g - g(0)$. As $p \neq 2$ and that $g(0) \in pW$, $\exp(g(0))$ is defined, therefore also $\exp(g) = \exp(g(0))\exp(h)$. Posons $g(0) = a$ ($a \in W$). one has
\[
\varphi^*h - ph = \varphi^*g - pg - (a^\sigma - pa) \in pW[[t]]
\]
(car $a^\sigma - pa \in pW$ and $\varphi^*g - pg \in pW[[t]]$). Therefore, by \ref{cor:1.4.4} applique a $h$, one has $\exp(h) \in W[[t]]$, and as $\exp(a) \equiv 1 \mod pW$, one en conclut that $f = \exp(g) \in W[[t]]$ and $f(0) \equiv 1 \mod pW$.

\textbf{Fin of the proof of 1.4.2.} It is enough to appliquer \ref{cor:1.4.5} a $\tau_{ij}$: the hypotheses of \ref{cor:1.4.5} are verifiees en vertu of (1.4.2.7) and (1.4.2.8).

Let, as en 1.3.6, $(H, \Fil^\bullet H)$ a $F$-cristal of Hodge on $A$ ($= k[[t]]$) such that $H$ let ordinaire. Supposons $H$ of niveau $\leqslant N$, i.e. (1.3.1) $\Fil^i H = 0$ for $i \geqslant N+1$, of sorte that the decomposition (1.3.6.1) s'ecrit

\begin{corollary}\label{cor:1.4.9}
Under the hypotheses of 1.4.8, there exists a base $(e^i_1, \ldots, e^i_{h_i})_{0 \leqslant i \leqslant N}$ of $H$, and of the elements $q_{i;\alpha,\beta}$ of $A$ such that
\begin{equation}\tag{1.4.9.1}

\end{equation}
\end{corollary}

One can utiliser \ref{cor:1.4.9} for majorer the croissance of the sections horizontales of $H$. That dit, it follows aisement of \ref{cor:1.4.9} that, for every $i$, the sections horizontales of $U_i \otimes_A K\{\{t\}\}$ appartiennent a $U_i \otimes_A L_i$, or
\[
0 = L_{-1} \subset L_0 \subset L_1 \subset \cdots \subset L_i \subset \cdots
\]
is the sequence of under-$A$-modules of $K\{\{t\}\}$ defined by recurrence by
\[
(f \in L_i) \Longleftrightarrow (f \text{ appartient au sous-$A$-module de } K\{\{t\}\} \text{ engendr\'e par les s\'eries } g \text{ telles que } dg = \sum a_\nu d\log u_\nu,

\begin{lemma}\label{lem:2.1.13}
Let $(a', b'_1, \ldots, b'_{20}, c')$ a base of $H$ and $(q'_i)_{1\leqslant i\leqslant 20}$ a family of elements of $A$ verifiant the same conditions (i), (ii), (iii) of \ref{thm:2.1.7} that $(a, (b_i), c)$ and $(q_i)$. \begin{equation}\tag{2.1.13.1}
\begin{pmatrix}
a' \\ c' \\ b'_i
\end{pmatrix}
=
\begin{pmatrix}
\lambda a \\ \lambda^{-1} c \\ \displaystyle\sum_{1\leqslant j\leqslant 20} \theta_{ji} b_j
\end{pmatrix},
\quad
q'_i = \prod_j q_j^{\theta_{ji}/\lambda}. 
\end{equation}
\end{lemma}

All of abord, as $(a', (b'_i), c')$ is a base of $H$ adaptee a the decomposition $H = H^0 \oplus H^1 \oplus H^2$ and that $\langle a', c' \rangle = 1$, there exists $\lambda \in A^*$ and $\theta = (\theta_{ij}) \in \GL(20, A)$ such that
\[
a' = \lambda a, \quad c' = \lambda^{-1} c, \quad b'_i = \sum_{1\leqslant j\leqslant 20} \theta_{ji} b_j.
\]
Grace a \ref{prop:2.1.9}, notons $e = e_q = e_{q'} : A \to W$ the lifting (2.1.8.3). As $\nabla a = \nabla a' = 0$, one has $d\lambda = 0$, therefore $\lambda$ is ``constant'', of value $e^*\lambda \in W$. Let $\varphi'$ the lifting of Frobenius such that $\varphi'(q'_i) = q'_i{}^p$. one has
\[
a' = F(\varphi')\varphi'^* a' = F(\varphi')\varphi'^*(\lambda a) = \lambda^\sigma F(\varphi')\varphi'^* a = \lambda^\sigma a,
\]
therefore $\lambda^\sigma = \lambda$, i.e. $\lambda/\lambda^\sigma \in \mathbb{Z}_p$. of other part,
\[
\nabla b'_i = \sum_j (d\theta_{ji}) \otimes b_j + \sum_j \theta_{ji}(d\log q_j) \otimes a = (d\log q'_i) \otimes a' = (d\log q'_i) \otimes \lambda a,
\]
therefore
\begin{align*}
d\theta_{ji} &= 0 \quad \forall i, j \tag{$\ast$} \\
\intertext{et}
\lambda \, d\log q'_i &= \sum_j \theta_{ji} \, d\log q_j \quad (1 \leqslant i \leqslant 20). \tag{$\ast\ast$}
\end{align*}
The formule ($\ast$) signifie that $\theta_{ji}$ is constant, of value $e^*\theta_{ji} \in W$. But
\[
F(\varphi')\varphi'^* b'_i = \sum_j \theta_{ji}^\sigma F(\varphi')\varphi'^* b_j = \sum_j \theta_{ji}^\sigma \cdot pb_j.
\]
Appliquant $e^*$ and tenant compte of this that $e^*F(\varphi')\varphi'^* b_j = e^*F(\varphi)\varphi^* b_j = e^*(pb_j)$, one obtains $\theta_{ji}^\sigma = \theta_{ji}$ $\forall i, j$, therefore $\theta \in \GL(20, \mathbb{Z}_p)$. Of ($\ast\ast$) one deduces that there exists $C \in W$ such that
\[
q'_i = C \prod_j q_j^{\theta_{ji}/\lambda}.
\]
But as $e^* q_i = e^* q'_i = 1$, one has $C = 1$, this which acheve the proof.

\subsection*{A1. Uniqueness of group structures}

Let $k$ be a perfect field, $W = W(k)$ its ring of Witt vectors, and $\sigma : W \to W$ the absolute Frobenius automorphism of $W$. Let $M$ be a finite-dimensional formal Lie variety over $W$, i.e. $M = \Spf(A)$ with $A$ not-canonically isomorphic to $W[[T_1, \ldots, T_n]]$, $n = \dim M$. Assumed we are given a $W$-morphism of formal Lie varieties
\[
\Phi : M \longrightarrow M^{(\sigma)}
\]
whose reduction modulo $p$ is the absolute Frobenius endomorphism
\[
\Frob : M \otimes_W k \longrightarrow (M \otimes_W k)^{(\sigma)}.
\]

\begin{lemma}[Uniqueness Lemma A1.1]\ \begin{enumerate}[label=(\arabic*)]
\item Given $(M, \Phi)$ as above, there exists at most one structure of commutative formal Lie group over $W$ one the formal Lie variety $M$ for which the given map $\Phi : M \to M^{(\sigma)}$ is a group homomorphism.
\item If this structure exists, it makes $M$ into a toroidal formal group, and the given $\Phi : M \to M^{(\sigma)}$ is the unique group homomorphism lifting Frobenius.
\item If $(M_1, \Phi_1)$ and $(M_2, \Phi_2)$ both admit group structures as in (2) above, then a morphism $f : M_1 \to M_2$ of formal Lie varieties over $W$ is a group homomorphism if and only if the diagram
\[
@@MATH48@@
\]
commutes.
\end{enumerate}
\end{lemma}

\begin{proof}
We begin by proving (2). Thus let $G$ be a finite-dimensional commutative formal Lie group over $W$, together with a homomorphism
\[
\Phi : G \longrightarrow G^{(\sigma)}
\]
which lifts Frobenius. We must show that $G$ is toroidal, and that $\Phi$ is unique. By the rigidity of toroidal groups, it suffices to show that $G \otimes k$ is toroidal, and for this it suffices to show that $\Ker(\Frob)$ is toroidal. For this, we first observe that $\Phi : G \to G^{(\sigma)}$ is finite (because it is finite modulo $p$, being a lifting of Frobenius) and flat (because it is a finite morphism between regular local rings of the same dimension). Therefore $\Ker(\Phi)$ is a finite flat commutative group-scheme over $W$ whose reduction mod $p$ is $\Ker(\Frob)$. According to Fontaine, if we denote by $N$ the contravariant Dieudonne module of $\Ker(\Frob)$, then the lifting $\Ker(\Phi)$ is described by a $W$-submodule $L \subset N$ which satisfies
\begin{enumerate}[label=\alph*)]
\item $L/pL \xrightarrow{\sim} N/FN$
\item $V|_L$ is injective.
\end{enumerate}
But $N$ is killed by $F$, and is of finite length over $W$. Therefore a) implies that $L = N$, and b) then shows that $V$ is injective, and hence bijective one $N$. Therefore $\Ker(\Frob)$ is toroidal, as required.

We next prove (3). By extending scalars $W(k) \to W(\bar{k})$, we may assumed $k$ algebraically closed. Then $M_1$ and $M_2$ become isomorphic to products of $\hat{\mathbb{G}}_m$, and our commutative diagram --- in the category of formal Lie varieties --- becomes
\[
@@MATH49@@
\]
If $f$ is a group homomorphism, then $f = f^{(\sigma)}$, and the diagram commutes. To prove the converse, we argue as follows.

In terms of ``multiplicative'' coordinates $T$ one $(\hat{\mathbb{G}}_m)^i$, $i = 1, 2$, our hypothesis one $f$ is:
\[
f^{(\sigma)}(T^p) = (f(T))^p, \quad f(1) \equiv 1 \mod p.
\]
Iterating, we find
\[
f^{(\sigma^n)}(T^{p^n}) = (f(T))^{p^n},
\]
so in particular
\[
f^{(\sigma^n)}(1) = (f(1))^{p^n} \to 1 \quad \text{as } n \to \infty.
\]
Therefore $f(1) = 1$. Now take logarithms, i.e. let
\[
F : (\hat{\mathbb{G}}_a)^{n_1} \longrightarrow (\hat{\mathbb{G}}_a)^{n_2}
\]
be the unique pointed morphism (over the fraction field of $W$) for which
\[
F(\log T) = \log(f(T));
\]
then we have
\[
F^{(\sigma)}(pX) = pF(X).
\]
Therefore $F$ is linear, and has coefficients in $W_p$. As these coefficients are intrinsically the matrix entries of the tangent map of $f$ at the origin, they lie in $W$ as well, hence $F$ is a linear map with coefficients in $\mathbb{Z}_p = W \cap \mathbb{Q}_p$. Therefore $f(T) = \exp(F(\log T))$ is a homomorphism.

Finally, we obtain (1) as the special case $(M_1, \Phi_1) = (M_2, \Phi_2)$, $f = \mathrm{id}$, of (3). \qedhere
\end{proof}

\begin{corollary}[A1.2]
Assumed that $k$ is algebraically closed, and that $(M, \Phi)$ admits a group structure as above. Then
\begin{enumerate}[label=(\arabic*)]
\item The character group $X(M) = \Hom_{W\text{-gp}}(M, \hat{\mathbb{G}}_m)$ is a free $\mathbb{Z}_p$-module of rank $n = \dim M$, and the natural map (of functors in $\mathbb{Z}_p$-modules)
\[
M \longrightarrow \Hom_{\mathbb{Z}_p}(X(M), \hat{\mathbb{G}}_m)
\]
is an isomorphism.
\item Let $q$ be a function one $M$ (i.e. $q \in A$ where $M = \Spf(A)$) with $q \equiv 1$ modulo the maximal ideal of $A$. Then $q \in X(M)$ if and only if $q$ transforms under $\Phi$ by
\[
\Phi^*(q^{(\sigma)}) = q^p.
\]
\item Let $\omega$ be a (continuous) one-form one $A$, i.e. $\omega \in (\Omega^1_{A/W})^{\mathrm{contin}}$. Then $\omega = dq/q$ with $q \in X(M) = \Hom(M, \mathbb{G}_m)$ if and only if
\[
\Phi^*(\omega^{(\sigma)}) = p\omega.
\]
If $\omega$ satisfies this, the associated $q \in X(M)$ is unique; it is given by the formula
\[
q(X) = \exp\Bigl(\int_0^X \omega\Bigr), \quad 0 = \text{the origin in } M.
\]
\item Let $\tau \in A \hat{\otimes}_W W[1/p]$ be a function one $M \otimes_W W[1/p]$. Then $\tau = \log(q)$ for some $q \in X(M)$ if and only if $\tau$ satisfies
\[
\tau(0) = 0, \quad \Phi^*(\tau^{(\sigma)}) = p\tau.
\]
If $\tau$ satisfies this, then the associated $q$ is unique; it is given by the formula
\[
q = \exp(\tau).
\]
\item Let $q_1, \ldots, q_n$ be $n = \dim M$ elements of $X(M)$, $\omega_1, \ldots, \omega_n$ the corresponding differentials $\omega_i = dq_i/q_i$ and $\tau_1, \ldots, \tau_n$ the corresponding ``functions with denominators'' $\tau_i = \log q_i$. Then the following conditions are equivalent:
\begin{enumerate}[label=\alph*)]
\item $q_1, \ldots, q_n$ form a $\mathbb{Z}_p$ base of $X(M)$.
\item the natural map
\[
M \longrightarrow \Spf(W[[q_1 - 1, \ldots, q_n - 1]])
\]
is an isomorphism.
\item $\omega_1, \ldots, \omega_n$ form an $A$-base of $\Omega^1_{A/W} \otimes \mathcal{O}^{\mathrm{contin}}$.
\item $\omega_1, \ldots, \omega_n$ form a $\mathbb{Z}_p$-base of
\[
\{\omega \in \Omega^1 \mid \Phi^*(\omega^{(\sigma)}) = p\omega\}.
\]
\item $\tau_1, \ldots, \tau_n$ form a $\mathbb{Z}_p$-base of
\[
\{\tau \in \Omega^1 \otimes_W W[1/p] \mid \tau(0) = 0, \; \Phi^*(\tau^{(\sigma)}) = p\tau\}.
\]
\end{enumerate}
\end{enumerate}
\end{corollary}

\begin{proof}
Because $M$ is toroidal, and $k$ is algebraically closed, $M$ is (not-canonically) isomorphic to $(\hat{\mathbb{G}}_m)^n$. This makes (1) obvious.

Assertion (2) is a particular case of part (3) of the uniqueness lemma, namely $M_1 = M$, $M_2 = \hat{\mathbb{G}}_m$. Assertion (3) becomes obvious if we choose a $\mathbb{Z}_p$-basis $q_1, \ldots, q_n$ of $X(M)$, i.e. if we choose an isomorphism
\[
M \xrightarrow{\sim} (\hat{\mathbb{G}}_m)^n,
\]
and write $\omega$ as an $A$-linear combination of the differentials $dq_i/q_i$:
\[
\omega = \sum f_i \, dq_i/q_i.
\]
The condition
\[
\Phi^*(\omega^{(\sigma)}) = p\omega
\]
means precisely that the coefficient functions $f_i$ each satisfy
\[
\Phi^*(f_i^{(\sigma)}) = f_i,
\]
which in turn implies that each coefficient function $f_i$ is simply a constant in $\mathbb{Z}_p$. Therefore
\[
\omega = dq/q \quad \text{for } q = \prod (q_i)^{f_i};
\]
because $q(0) = 1$, we obtain by integration the formula
\[
\exp\Bigl(\int_0^X \omega\Bigr) = q(X).
\]

For assertion (4), first note that the Dieudonne-Dwork integrality criterion (cf. 1.4.4) guarantees that the series $q$, defined as
\[
q \overset{\mathrm{dfn}}{=} \exp(\tau),
\]
is integral (i.e. lies in $A$) and verifies $q(0) = 1$. From the equality
\[
\Phi^*(\tau^{(\sigma)}) = p\tau
\]
(and not simply their congruence modulo $pA$) we see that
\[
\Phi^*(q^{(\sigma)}) = q^p,
\]
and we conclude by (2) that $q$ lies in $X(M)$.

In assertion (5), the equivalence of a), b) and c) is physically obvious for $(\hat{\mathbb{G}}_m)^n$, and the equivalence of a) with d) and e) is obvious from parts 3) and 4) above. \qedhere
\end{proof}

\subsection*{A3. Formal moduli of ordinary abelian varieties; the Serre-Tate \& Dwork group structures}

Let $k$ be an algebraically closed field of characteristic $p > 0$, and $X_0/k$ an ordinary abelian variety over $k$. Let $M$ be its formal deformation space, and $X/M$ the corresponding formal abelian scheme.

One knows that $M$ is a $g^2$-dimensional formal Lie variety over $W = W(k)$. Because $X_0/k$ is ordinary, the formal group $\hat{X}$ of $X/M$ is not-canonically isomorphic to $(\hat{\mathbb{G}}_m)^g$ over $M$. The ``canonical subgroup'' $H_{\mathrm{can}} \subset \hat{X}$ is defined to be the kernel of $[p]$ in $\hat{X}$; it is the unique finite flat subgroup-scheme of $X/M$ which mod $p$ becomes the kernel of the relative Frobenius endomorphism of $X \otimes_W k/M \otimes_W k$. We denote by
\[
F_{\mathrm{can}} : X \longrightarrow X/H_{\mathrm{can}}
\]
the projection onto the quotient by $H_{\mathrm{can}}$. The quotient $X/H_{\mathrm{can}}$ over $M$ is a deformation of $X_0^{(p)}/k$, so its ``classifying map'' is a morphism of formal Lie varieties
\[
\Phi_{\mathrm{can}} : M \longrightarrow M^{(\sigma)},
\]
``defined by'' an isomorphism of formal abelian schemes over $M$
\[
\theta_{\mathrm{can}}^*(X^{(\sigma)}) \xrightarrow{\sim} X/H_{\mathrm{can}}.
\]
Thus $\Phi_{\mathrm{can}}$ is a lifting of the absolute Frobenius endomorphism of $M \otimes k$, and
\[
(F_{\mathrm{can}})_M^{(\Phi)} : X \longrightarrow X/H_{\mathrm{can}} = \Phi_{\mathrm{can}}^* X^{(\sigma)}
\]
is a lifting of the relative (to $M \otimes k$) Frobenius endomorphism of $X \otimes k/M \otimes k$. Therefore this morphism induces one $H^1_\mathrm{dR}$ an $M$-linear map
\[
F(\Phi_{\mathrm{can}}) : \Phi_{\mathrm{can}}^* H^1_\mathrm{dR}(X/M) \longrightarrow H^1_\mathrm{dR}(X/M)
\]
which is none other than the crystalline map $F(\Phi_{\mathrm{can}})$. Because $F_{\mathrm{can}}$ is a physical morphism, the induced map $F$ respects the Hodge filtration of $H^1_\mathrm{dR}$. Therefore we have
\[
F(\Phi_{\mathrm{can}})(\Phi_{\mathrm{can}}^*(\Fil^1)) \subset \Fil^1.
\]

\begin{theorem}[A3.1]
The structure of group imposed one $M$ by the Serre-Tate description of $M$ as
\[
\Ext((X_0^{(p^n)})^{\mathrm{\acute{e}t}}, X_0^{p^n\text{-}\mathrm{conn}})
\]
coincides with the structure of group one $M$ for which the $q_{ij} = \exp(\tau_{ij})$ define an isomorphism of groups
\[
M \xrightarrow{\sim} \prod_{i,j} \hat{\mathbb{G}}_m.
\]
\end{theorem}

\begin{proof}
The morphism $\Phi_{\mathrm{can}} : M \to M^{(\sigma)}$ is also a group homomorphism for the Serre-Tate group structure one $M$. \qedhere
\end{proof}

\subsection*{A4. Formal moduli of ordinary K3 surfaces; sharpening of 2.1.7, 2.1.14}

\textbf{Corollary A4.1 (of 2.1.7, 2.1.14).} With the hypotheses and notations of 2.1.7 and 2.1.14, there exists a unique morphism
\[

\end{proof}
