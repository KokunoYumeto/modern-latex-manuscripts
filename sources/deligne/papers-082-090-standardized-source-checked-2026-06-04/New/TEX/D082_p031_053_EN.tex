\section*{5. Application to values of multilogarithms: tools}

Fix an embedding \(\sigma\) of \(k\) in \(\C\). For \(\eta\in\mu_N\), we shall again write \(\eta\) for the complex number \(\sigma(\eta)\). Let \(\tau\) denote the action of \(\Gm\) given by multiplication by \(\lambda^d\) in degree \(d\).

\heading{5.1.} We shall use the results of [1] 5.17--5.20, recalled below. Recall that the set \(\Pi(\C)\) of \(\C\)-valued points of \(\Pi\) is the set of group-like elements of
\[
 \C\langle\!\langle e_0,(e_\eta)_{\eta\in\mu_N}\rangle\!\rangle .
\]
(A) There is an element \(d_{\mathrm{ch}}\in\Pi(\C)\) whose coefficients of \(e_0\) and \(e_1\) are zero and such that, for \((s_1,\eta_1)\ne(1,1)\), the coefficient of
\[
 e_0^{s_1-1}e_{\eta_1}\cdots e_0^{s_m-1}e_{\eta_m}
\]
is
\[
 (-1)^m\zeta(s_1,\ldots,s_m;\eta_1^{-1},\eta_1\eta_2^{-1},\ldots,\eta_{m-1}\eta_m^{-1}).
\]
This element is unique ([1] 2.4.5).

(B) Let \(IU_\omega\) be the subgroup of \((\Pi,\circ)\) which is the image of \(U_\omega\) by
\[
(3.4.6):U_\omega\longrightarrow(\Pi,\circ).
\]
There is \(v_\sigma\in\Pi(\Q)\) such that
\[
\tag{5.1.1}
d_{\mathrm{ch}}\in IU_\omega(\C)\circ \tau(2\pi i)(v_\sigma).
\]
The element \(v_\sigma\) of \(\Pi(\Q)\) is determined only up to left multiplication, for the law \(\circ\), by an element of \(IU_\omega(\Q)\). We shall show in A.8 that it can be chosen fixed by the composite of \(\tau(-1)\) and the automorphism of \(\Pi\) induced by the automorphism \(e_0\mapsto e_0\), \(e_\eta\mapsto e_{\eta^{-1}}\) of \(L\). We shall use only the fact, already proved in [1], that if \(N=1\) or \(2\) it can be chosen fixed by \(\tau(-1)\). We shall make that choice.

\heading{5.2.} Both \(IU_\omega\) and \(d_{\mathrm{ch}}\) lie in the subgroup of \((\Pi,\circ)\) defined by \(c(e_0)=c(e_1)=0\) (notation (1.4.2)), and in particular in the subgroup \(D^1\Pi\). The ideal generated by \(c(e_0)\), which defines \(D^1\Pi\), has as a \(\Q\)-basis the elements \(c(me_0^k)c(e_0)\), where \(m\) ranges over the monomials not ending in \(e_0\). One has
\[
c(me_0^k)c(e_0)=(k+1)c(me_0^{k+1})+\hbox{remainder},
\]
where the remainder is a sum of \(c(m'e_0^k)\), with \(m'\) not ending in \(e_0\). It follows that the affine algebra of \(D^1\Pi\) has as a \(\Q\)-basis the restrictions to \(D^1\Pi\) of the \(c(m)\), where \(m\) ranges through the monomials not ending in \(e_0\).

The depth filtration of \(\OO(D^1\Pi)\) is associated (1.3) with its grading by the \(D\)-degree. Applying 1.8 to \((D^1\Pi,\circ)\), one deduces that the vector scheme \(D^1\Pi/D^2\Pi\) admits as coordinate system the \(c(m)\), for \(m\) a monomial of \(D\)-degree one not ending in \(e_0\). More precisely, \(c(m)\) in \(\OO(\Pi)\), restricted to \(D^1\Pi\), factors through a linear function on \(D^1\Pi/D^2\Pi\), and these functions form a coordinate system.

For \(x\in IU_\omega\), (3.10.5) with \(d=-1\) gives
\[
\tag{5.2.1}
c(e_0^{k-1}e_\eta+(-1)^ke_0^{k-1}e_{\eta^{-1}})(x)=0.
\]
Projecting (5.1.1) onto \(D^1\Pi/D^2\Pi\) (the quotient for the law \(\circ\)), one deduces from (5.2.1) that, for \(x\in IU_\omega\circ\tau(t)(v_\sigma)\),
\[
\tag{5.2.2}
c(e_0^{k-1}e_\eta+(-1)^ke_0^{k-1}e_{\eta^{-1}})(x)
=t^k c(e_0^{k-1}e_\eta+(-1)^ke_0^{k-1}e_{\eta^{-1}})(v_\sigma).
\]
Taking \(x=d_{\mathrm{ch}}\) and \(t=2\pi i\), one obtains
\[
\tag{5.2.3}
c(e_0^{k-1}e_\eta+(-1)^ke_0^{k-1}e_{\eta^{-1}})(v_\sigma)
=-\frac{\zeta(k,\eta^{-1})+(-1)^k\zeta(k,\eta)}{(2\pi i)^k}
\]
if \((k,\eta)\ne(1,1)\). Moreover, \(c(e_1)(v_\sigma)=0\). The right-hand side of (5.2.3) is therefore rational. More precisely, for \(z=\exp(2\pi it)\), \(0\le t\le1\), and \(t\ne0,1\) if \(k=1\), one has (the Fourier-series expansion of the Bernoulli polynomials, Bourbaki FVR VI \S2 ex. 12c)
\[
\tag{5.2.4}
\left(\sum\frac{z^n}{n^k}+(-1)^k\sum\frac{z^{-n}}{n^k}\right)/(2\pi i)^k=-B_k(t)/k!.
\]
Let \(c_{k,\eta}\) be the rational number (5.2.3) such that \(c(e_0^{k-1}e_\eta+(-1)^ke_0^{k-1}e_{\eta^{-1}})\) is equal to \(c_{k,\eta}t^k\) on \(IU_\omega\circ\tau(t)(v_\sigma)\).

\begin{corollary*}[5.3]
(i) If \(k\) is even, \(c_{k,1}\ne0\).\\
(ii) If \(k\) is odd and \(\eta\ne\pm1\), then \(c_{k,\eta}\ne0\).
\end{corollary*}

\begin{proofdel}
(i) follows from \(\zeta(k)\ne0\), and (ii) follows from the fact that, for \(k\) odd, the Bernoulli polynomial \(B_k\) vanishes in the interval \([0,1]\) only at \(0,\frac12\), and \(1\) (Bourbaki, FVR VI \S2 exercise 4c).
\end{proofdel}

\heading{5.4.} The map \(t\mapsto\tau(t)(v_\sigma)\) from \(\Gm\) to \(\Pi\) extends to a map from \(\A^1\) to \(\Pi\), still denoted \(t\mapsto\tau(t)(v_\sigma)\), or, in this paragraph,
\[
\tag{5.4.1}
d:\A^1\longrightarrow\Pi.
\]
If \(N=1,2\), it factors through
\[
\tag{5.4.2}
\bar d:\A^1/\mu_2\longrightarrow\Pi.
\]
The affine algebra of \(\A^1\) is \(\Q[t]\). That of \(\A^1/\mu_2\) is the subalgebra \(\Q[t^2]\): \(t^2\) will be viewed as a coordinate on \(\A^1/\mu_2\).

If \(N\ge3\) and \(c_{1,\eta}\ne0\) (respectively \(N=1,2\)), the map (5.4.1) (respectively (5.4.2)) has the retraction
\[
r:x\longmapsto t=\frac{c(e_\eta-e_{\bar\eta})(x)}{c_{1,\eta}}
\]
(respectively
\[
\tag{5.4.3}
\bar r:x\longmapsto t^2=\frac{2c(e_0e_1)(x)}{c_{2,1}}).
\]
It is therefore a closed embedding. Its image will be denoted \(D\), and \(t\) (respectively \(t^2\)) will denote the coordinate on \(D\) given by (5.4.1) (respectively (5.4.2)).

The product for the group law \(\circ\) of \(\Pi\) induces a morphism
\[
\tag{5.4.4}
IU_\omega\times D\longrightarrow\Pi,\qquad x,y\longmapsto x\circ y.
\]
This morphism identifies \(IU_\omega\times D\) with the closed subscheme, denoted \(IU_\omega\circ D\), of the \(x\) satisfying
\[
x\circ d(r(x))^{-1}\in IU_\omega
\]
(respectively
\[
\tag{5.4.5}
x\circ\bar d(\bar r(x))^{-1}\in IU_\omega).
\]
The inverse map
\[
\tag{5.4.6}
IU_\omega\circ D\longrightarrow IU_\omega\times D
\]
is \(x\mapsto(x\circ d(r(x))^{-1},d(r(x)))\) (respectively \(x\mapsto(x\circ\bar d(\bar r(x))^{-1},\bar d(\bar r(x)))\)).

Let \(\Delta\) be the coproduct of the affine algebra \(\OO(D^1\Pi)\) defined by the group law \(\circ\). If \(F\in\OO(D^1\Pi)\) has restriction \(f\) to \(\OO(IU_\omega\circ D)\), the image of \(f\) by (5.4.6) is \(\Delta F\) restricted to \(IU_\omega\times D\):
\[
\tag{5.4.7}
\begin{tikzcd}[column sep=4.0em,row sep=2.2em]
\OO(D^1\Pi) \arrow[r,"\Delta"] \arrow[d] &
\OO(D^1\Pi\times D^1\Pi) \arrow[d]\\
\OO(IU_\omega\circ D) \arrow[r] &
\OO(D^1\Pi\times D).
\end{tikzcd}
\]

We shall still write \(t\) (respectively \(t^2\)) for the inverse image in \(\OO(IU_\omega\circ D)\), by (5.4.6), of the coordinate \(t\) (respectively \(t^2\)) on \(D\). By 5.3, one has:

\begin{lemma*}[5.5]
For \(n\ge1\), even if \(N=1,2\), \(t^n\) is the restriction to \(IU_\omega\circ D\) of an element of codegree \(n\) in \(D^1\OO(\Pi)\).
\end{lemma*}

By A.4, the following conjecture is a consequence of conjecture A.2, which is a variant of Grothendieck's period conjecture.

\begin{theorem*}[Conjecture 5.6]
The complex point \(d_{\mathrm{ch}}\) of \(IU_\omega\circ D\) is a generic point, i.e.
\[
\tag{5.6.1}
f\longmapsto f(d_{\mathrm{ch}}):\OO(IU_\omega\circ D)\longrightarrow\C
\]
is injective.
\end{theorem*}

In other words, every algebraic relation with rational coefficients between the
\[
\zeta(s_1,\ldots,s_m;\varepsilon_1,\ldots,\varepsilon_m),\qquad
\varepsilon_i\in\mu_N,\quad (s_1,\varepsilon_1)\ne(1,1),
\]
is a consequence of the fact that \(d_{\mathrm{ch}}\) belongs to \((IU_\omega\circ D)(\C)\).

\heading{5.7.} If \(T\) is a closed subscheme of \(\Pi\), we define the depth filtration of its affine algebra \(\OO(T)\) as the quotient filtration induced by the depth filtration of \(\OO(\Pi)\). If \(T\subset D^1\Pi\), it is induced by quotient from that of \(\OO(D^1\Pi)\), and satisfies \(D^0\OO(T)=\Q\).

Example. By 5.5, the depth filtration of \(\OO(D)\) is given by
\[
\tag{5.7.1}
D^0=\Q,\qquad D^1=\OO(D).
\]
If \(\mathfrak t\) is a graded subspace of \(L\), its depth filtration is defined as the filtration induced by that of \(L\). If \(\mathfrak t\) is a Lie subalgebra of \(L\) (respectively of \((L,\{\ ,\ \})\)), and \(T\) is the corresponding subgroup of \(\Pi\) (respectively of \((\Pi,\circ)\)), the compatibility [1]A, a consequence of 1.6 (ii) applied to \(T\) and \(\Pi\) (respectively to \((\Pi,\circ)\)), shows that the depth filtration of \(\OO(T)\) is obtained from that of \(\mathfrak t\) by 1.5.

\begin{proposition*}[5.8]
The depth filtration of \(\OO(IU_\omega\circ D)\), identified by (5.4.6) with \(\OO(IU_\omega)\otimes\OO(D)\), is the tensor product of the depth filtration of \(\OO(IU_\omega)\) and that of \(\OO(D)\) (given by (5.7.1)).
\end{proposition*}

\begin{proofdel}
We give the proof only for \(N\ge3\). The argument is the same for \(N=1\) or \(2\), except that \(t\) is to be replaced by \(t^2\). We first prove that the isomorphism of \(\OO(IU_\omega\circ D)\) with \(\OO(IU_\omega)\otimes\OO(D)\) sends
\[
\tag{5.8.1}
D^p\OO(IU_\omega\circ D)
\longrightarrow
D^p\OO(IU_\omega)\oplus\bigoplus_{n>0}D^{p-1}\OO(IU_\omega)\,t^n.
\]
Let \(f\in D^p\OO(IU_\omega\circ D)\). By the definition of the filtration \(D\), \(f\) is the restriction to \(IU_\omega\circ D\) of an element \(F\in D^p\OO(D^1\Pi)\). The coproduct for the law \(\circ\)
\[
\Delta:\OO(D^1\Pi)\longrightarrow\OO(D^1\Pi\times D^1\Pi)
=\OO(D^1\Pi)\otimes\OO(D^1\Pi),\qquad h\longmapsto h(x\circ y),
\]
is compatible with the gradings by \(D\)-degree. Since \(D^1L\) has \(D\)-degree \(>0\), the augmentation ideal has \(D\)-codegrees \(>0\), and \(\Delta F-F\otimes1\), which lies in \(\OO(D^1\Pi)\otimes\mathfrak m\), lies in \(D^{p-1}\OO(D^1\Pi)\otimes\mathfrak m\): one has
\[
\tag{5.8.2}
\Delta F=F\otimes1+\hbox{remainder in }D^{p-1}\OO(D^1\Pi)\otimes\mathfrak m.
\]
Restrict (5.8.2) to \(IU_\omega\times D\subset D^1\Pi_\omega\times D^1\Pi_\omega\) and apply 5.4.7. This gives (5.8.1).

It remains to show that (5.8.1) is onto. By 5.5, the spaces \(D^{p-1}\OO(IU_\omega\circ D)\,t^n\) are contained in \(D^p\OO(IU_\omega\circ D)\). Filter \(D^p\OO(IU_\omega\circ D)\) by
\[
D^p\OO(IU_\omega\circ D)\supset
D^{p-1}\OO(IU_\omega\circ D)t\supset
D^{p-1}\OO(IU_\omega\circ D)t^2\supset\cdots .
\]
By (5.8.1) for \(p\) and \(p-1\), (5.8.1) is compatible with this filtration and with the \(t\)-adic filtration of the right-hand side. The associated graded map is surjective, since it is given by restriction morphisms from \(IU_\omega\circ D\) to \(IU_\omega\). Since it suffices to check the surjectivity of (5.8.1) separately in each degree, and since these filtrations are finite in each degree, the surjectivity of (5.8.1) follows from that of its graded map.
\end{proofdel}

\heading{5.9.} Let \(E\) be a \(\Q\)-vector space, bigraded in positive degrees. Assume that the homogeneous components for the first degree, called simply the degree, are finite-dimensional. The second degree will be called the \(D\)-degree, and the corresponding filtration the depth filtration. Let \(E^\wedge\) be the projective limit of affine spaces
\[
E^\wedge:=\limproj(E/E_{\mathrm{degree}\ge p}).
\]
Its affine algebra, endowed with its grading by the degree and with the depth filtration associated with its grading by \(D\)-degree, is of type \((*)\) of 1.11, and \(E\) is the graded tangent space of \(E^\wedge\) at \(0\).

Let
\[
F:IU_\omega\circ D\longrightarrow E^\wedge
\]
be a morphism of schemes compatible with the degree gradings and with the depth filtrations of affine algebras. By restriction to \(IU_\omega\) and passage to graded tangent spaces at the origin, it induces
\[
\tag{5.9.1}
dF|_{\mathfrak{iu}_\omega}:\mathfrak{iu}_\omega\longrightarrow E.
\]
We shall use 5.8 in the following form.

\begin{corollary*}[5.10]
Assume that (5.9.1) is an isomorphism of filtered graded vector spaces, and let
\[
F_1:IU_\omega\circ D\longrightarrow E^\wedge\times D
\]
be the morphism whose projections are \(F\) and \(IU_\omega\circ D\xrightarrow{5.4.6}IU_\omega\times D\to D\). Then
\[
F_1^*:\OO(E^\wedge\times D)\longrightarrow\OO(IU_\omega\circ D)
\]
is an isomorphism of filtered graded algebras.
\end{corollary*}

In 5.10, the depth filtration of \(\OO(E^\wedge\times D)=\OO(E^\wedge)\otimes\OO(D)\) is defined as the tensor product of those of \(\OO(E^\wedge)\) and \(\OO(D)\).

\begin{proofdel}
By 1.12 and 1.14, the restriction of \(F\) to \(IU_\omega\) induces an isomorphism of filtered graded algebras from \(\OO(E^\wedge)\) to \(\OO(IU_\omega)\). The diagram
\[
\tag{5.10.1}
\begin{tikzcd}[column sep=3.0em,row sep=2.0em]
IU_\omega\circ D \arrow[rr,"F_1"] \arrow[dr] &&
E^\wedge\times D \arrow[dl]\\
& D &
\end{tikzcd}
\]
is commutative. The graded algebra of \(\OO(IU_\omega\circ D)\) for the \(t\)-adic filtration is
\[
\OO((\hbox{fibre over }0)\times D)=\OO(IU_\omega\times D).
\]
By 5.8, the filtration induced on the graded algebra by the depth filtration is simply the depth filtration of \(\OO(IU_\omega\times D)=\OO(IU_\omega)\otimes\OO(D)\). The same holds for \(E^\wedge\times D\). Finally, (5.10.1) ensures that the graded map of \(F_1^*\) is the inverse image by
\[
(\hbox{fibre over }0\hbox{ of }IU_\omega\circ D\to D)\times D
\longrightarrow E^\wedge\times D.
\]
It is therefore an isomorphism of filtered spaces. Since the \(t\)-adic filtration is finite in each degree, 5.10 follows.
\end{proofdel}

\section*{6. \(N=3,4,8\)}

In this section assume that \(N=3,4\) and \(M=1\), or that \(N=8\) and \(M=2\).

\begin{theorem*}[6.1]
Fix \(w,d\ge0\). Every multilogarithm value relative to \(\mu_N\)
\[
\tag{6.1.1}
\zeta(s_1,\ldots,s_r;\eta_1,\ldots,\eta_r)\qquad(\eta_i\in\mu_N)
\]
of weight \(\sum s_i=w\) and depth \(r\le d\), with \((s_1,\eta_1)\ne(1,1)\), is a linear combination with rational coefficients of the quantities
\[
\tag{6.1.2}
\zeta(s_1,\ldots,s_r;\eta_1,\ldots,\eta_r)(2\pi i)^p
\]
for \(p+\sum s_i=w\), \(r\le d\) if \(p=0\), and \(r\le d-1\) if \(p>0\), with \(\eta_1\in\zeta\mu_M\) and \(\eta_2,\ldots,\eta_r\in\mu_M\). Moreover, conjecture 5.6 implies that the quantities (6.1.2) are \(\Q\)-linearly independent.
\end{theorem*}

\begin{proofdel}
By 4.11 one has \(U_\omega\overset{\sim}{\longrightarrow}IU_\omega\). The composite map
\[
\tag{6.1.3}
IU_\omega\circ D\xrightarrow{O_1}D^1\Pi\xrightarrow{O_2}D^1\Pi_M^{\prime\prime}
\]
induces on affine algebras a morphism compatible with the depth filtrations: this is true for \(O_1\) by definition, and for \(O_2\) because \(O_2\) is even compatible with the gradings by the \(D\)-degree. The scheme \(D^1\Pi_M^{\prime\prime}\) is of the type \(E^\wedge\) considered in 5.9, since the exponential identifies it with \(L^{\prime\prime\wedge}\), and 4.11 (ii) tells us that the hypothesis of 5.10 is satisfied. The morphism
\[
\tag{6.1.4}
IU_\omega\circ D\longrightarrow D^1\Pi^{\prime\prime}\times D
\]
therefore induces on affine algebras an isomorphism of filtered graded algebras.

The scheme \(D^1\Pi_M^{\prime\prime}\) is the scheme of group-like elements of
\[
\Q\langle\!\langle e_0,(e_\eta)_{\eta\in\zeta^{-1}\mu_M}\rangle\!\rangle,
\]
with coefficient of \(e_0\) zero. Its affine algebra has as a graded and \(D\)-graded basis the \(c(m)\), where \(m\) is a monomial in \(e_0\) and the \(e_\eta\) (\(\eta\in\zeta^{-1}\mu_M\)) not ending in \(e_0\) (cf. 5.2). The inverse image of \(c(m)\) by (6.1.3) is simply the restriction to \(IU_\omega\circ D\) of \(c(m)\in\OO(\Pi)\).

The isomorphism (6.1.4) thus means that \(\OO(IU_\omega\circ D)\) has as a \(\Q\)-basis the \(c(m)t^p\), where \(m\) is a monomial in \(e_0\) and the \(e_\eta\) (\(\eta\in\zeta\mu_M\)) not ending in \(e_0\), that
\[
\operatorname{codegree}(c(m)t^p)=\operatorname{degree}(m)+p,
\]
and that \(D^d\OO(IU_\omega\circ D)\) has as a basis those \(c(m)t^p\) for which
\[
p=0\hbox{ and }m\hbox{ has }D\hbox{-degree }\le d,
\]
or
\[
p>0\hbox{ and }m\hbox{ has }D\hbox{-degree }\le d-1.
\]
It remains only to evaluate at \(d_{\mathrm{ch}}\in(IU_\omega\circ D)(\C)\). The quantities (6.1.1) are obtained, up to sign, by evaluating the \(c(m)\), where \(m\) is a monomial in \(e_0\) and the \(e_\eta\) (\(\eta\in\mu_N\)) of degree \(w\) and \(D\)-degree \(\le d\). The quantities (6.1.2) are obtained, up to sign, by evaluating the \(c(m)t^p\) of the above basis of \(D^p(\OO(IU_\omega\circ D))_w\). Finally, the linear independence of the quantities (6.1.1) is equivalent to the injectivity of (5.6.1).
\end{proofdel}

\section*{7. \(N=2\)}

In this section \(N=2\), \(M=1\), \(\ell=2\), and \(A\) denotes the set of odd integers \(\ge1\), ordered by \(\ge\), which we write \(\prec\):
\[
\cdots\prec5\prec3\prec1.
\]

\heading{7.1.} Lyndon words in the alphabet \(A\) will be called odd Lyndon words. To such an odd Lyndon word \(n=(n_1,\ldots,n_d)\) we attach \(\zeta(n_1,\ldots,n_d;-1,1,\ldots,1)\), written simply \(\zeta(n;-1,1,\ldots)\), and to a formal linear combination with integer coefficients \(\ge0\),
\[
\sum\lambda(n)[n],
\]
of odd Lyndon words, we attach the product
\[
\prod\zeta(n;-1,1,\ldots)^{\lambda(n)}.
\]
The weight of \(n=(n_1,\ldots,n_d)\) is the sum of the \(n_i\), its depth is its length \(d\), and the weight and depth of \(\sum\lambda(n)[n]\) are defined additively.

\begin{theorem*}[7.2]
Fix \(w,d\ge0\). Every multilogarithm value relative to \(\mu_2\)
\[
\tag{7.2.1}
\zeta(s_1,\ldots,s_r;\eta_1,\ldots,\eta_r)\qquad(\eta_i\in\mu_2=\{\pm1\})
\]
of weight \(\sum s_i=w\) and depth \(r\le d\) is a linear combination with rational coefficients of
\[
\tag{7.2.2}
\prod\zeta(n;-1,1,\ldots)^{\lambda(n)}(2\pi i)^{2p}
\]
where \(\sum\lambda(n)[n]\) is a formal linear combination with coefficients \(\ge0\) of odd Lyndon words and \(p\ge0\), satisfying
\[
2p+\operatorname{weight}\Bigl(\sum\lambda(n)[n]\Bigr)=w,
\]
and, if \(p=0\), \(\operatorname{depth}(\sum\lambda(n)[n])\le d\), while, if \(p>0\), \(\operatorname{depth}(\sum\lambda(n)[n])\le d-1\). Moreover, conjecture 5.6 implies that the quantities (7.2.2) are \(\Q\)-linearly independent.
\end{theorem*}

\begin{proofdel}
Let \(E\) be the vector space freely generated by vectors \(e_{[n]}\) indexed by the odd Lyndon words. It is bigraded by giving \(e_{[n]}\) degree equal to the weight of \(n\), and \(D\)-degree equal to the depth of \(n\).

The quotient \(D^1\Pi_M^{\prime\prime}\) of \(D^1\Pi\) is the scheme of group-like elements of \(\Q\langle\!\langle e_0,e_{-1}\rangle\!\rangle\) whose coefficient of \(e_0\) is zero. To each odd Lyndon word \(n=(x_1,\ldots,x_d)\) attach the monomial
\[
m(n):=e_0^{n_1-1}e_{-1}e_0^{n_2-1}e_{-1}\cdots e_0^{n_d-1}e_{-1}.
\]
Its degree is the weight of \(n\), and its \(D\)-degree is its depth. The \(c(m(n))\) are therefore the coordinates of a morphism of schemes
\[
\tag{7.2.3}
D^1\Pi^{\prime\prime}\longrightarrow E^\wedge,
\]
compatible with degrees and \(D\)-degrees.

The composite map
\[
\tag{7.2.4}
IU_\omega\circ D\xrightarrow{O_1}D^1\Pi\xrightarrow{O_2}D^1\Pi^{\prime\prime}\xrightarrow{O_3}E^\wedge
\]
induces on affine algebras a morphism compatible with the depth filtrations: this is true for \(O_1\) by definition, while \(O_2\) and \(O_3\) are even compatible with the gradings by \(D\)-degree.

Passing to tangent spaces at the origin, the restriction of (7.2.4) to \(IU_\omega\) defines a filtered graded morphism
\[
\tag{7.2.5}
\mathfrak{iu}_\omega\longrightarrow E.
\]
The same arguments as in 6.1 reduce 7.2 to the following statement.

\begin{proposition*}[7.3]
The morphism (7.2.5) is an isomorphism of filtered graded vector spaces.
\end{proposition*}

To prove 7.3 we must return to 4.9. By 4.11 one has \(u_\omega\overset{\sim}{\longrightarrow}\mathfrak{iu}_\omega\), and the depth filtration of \(\mathfrak{iu}_\omega\) identifies with the descending central series. Let \(J\) be, as in 4.6, the image of \(u_\omega^{\mathrm{ab}}\) in \((L^{\prime\prime}\times L')\langle1\rangle\), and consider the morphisms
\[
\tag{7.3.1}
\Lib(J)\longrightarrow \Gr_Z(u_\omega)=\Gr_D(\mathfrak{iu}_\omega)\longrightarrow (L^{\prime\prime}\times L',\{\ ,\ \}).
\]
As in 4.10, it suffices to show that
\[
\tag{7.3.2}
\Lib(J)\xrightarrow{7.3.1}L^{\prime\prime}\times L'\longrightarrow L^{\prime\prime}\longrightarrow E
\]
is an isomorphism.

The morphisms (7.3.2) are obtained by tensoring with \(\Q\) from
\[
\tag{7.3.3}
\Lib(J_\Z)\longrightarrow L^{\prime\prime}_\Z\times L'_\Z\longrightarrow L^{\prime\prime}_\Z\longrightarrow E_\Z,
\]
where \(J_\Z,L'_\Z\), and \(L^{\prime\prime}_\Z\) are as in 4.8 and 4.1, and where \(E_\Z\) is generated over \(\Z\) by the \(e_{[n]}\), for \(n\) an odd Lyndon word. As in 4.10, to prove that (7.3.2) is an isomorphism, it is enough to prove that the reduction modulo \(\ell\)
\[
\tag{7.3.4}
\Lib(J_\Z/\ell)\xrightarrow{O_1}L^{\prime\prime}_\Z/\ell\xrightarrow{O_2}E_\Z/\ell
\]
of (7.3.3) is an isomorphism.

The Lie algebra \(L^{\prime\prime}_\Z/\ell\) is freely generated over \(\Z/\ell\) by the \(\ad(e_0)^{n-1}(e_{-1})\) (\(n\ge1\)), and \(\Lib(J_\Z/\ell)\) is the subalgebra freely generated by the \(\ad(e_0)^{n-1}(e_{-1})\) for odd \(n\). The morphism \(O_2\) has as coordinates the maps
\[
L^{\prime\prime}_\Z/\ell\longrightarrow \Z/\ell\langle e_0,e_{-1}\rangle
\xrightarrow{c(m(n))}\Z/\ell
\]
for odd Lyndon words \(n\).

Each odd Lyndon word \(w\) defines a Lie polynomial in variables \(x_n\) (\(n\) odd, \(n\ge1\)). Applying this Lie polynomial to the \(\ad(e_0)^{n-1}(e_{-1})\), one obtains \(P_w\), and the \(P_w\) form a basis of \(\Lib(J_\Z/\ell)\). By 2.4, in each fixed degree and \(D\)-degree, the matrix \(c(m(n))(P_w)\) (with \(n\) and \(w\) odd Lyndon words of the fixed degree and \(D\)-degree) is triangular, with \(1\)'s on the diagonal. It is therefore invertible, and this concludes the proof of 7.2.
\end{proofdel}

\heading{7.3.} The generating system (7.2.2) should not be taken too seriously. Others are possible. Here is an example. Endow the set of odd integers \(\ge3\) with its usual order \(\le\), and replace in (7.2.2) the \(\zeta(n;-1,1,\ldots)\), for \(n\) an odd Lyndon word, by the
\[
\zeta(n_1,\ldots,n_d;-1,1,\ldots,1)
\]
for which \((n_d,\ldots,n_1)\) is a Lyndon word in this alphabet. The proof is the same, except that 2.4 is replaced by the following lemma, in which the monomials
\[
e^{m_1}fe^{m_2}f\cdots e^{m_d}fe^{m_{d+1}}
\]
of given degrees in \(e\) and \(f\) are ordered according to the lexicographic order of
\[
\left(\sum_{1}^{d}m_i,\sum_{1}^{d-1}m_i,\ldots,m_1\right).
\]

\begin{lemma*}[7.4]
Let \(w=(n_1,\ldots,n_d)\) be a Lyndon word in the alphabet of integers \(\ge0\), ordered by \(\le\), and let \(P_w\) be the corresponding Lie polynomial, applied to the \((\ad e)^n(f)\). Then
\[
\tag{7.4.1}
P_w=\pm e^{n_d}fe^{n_{d-1}}f\cdots e^{n_1}f+\hbox{remainder},
\]
where the remainder is a linear combination of monomials with the same degrees in \(e\) and \(f\) and strictly smaller in the order of 7.3.
\end{lemma*}

\begin{proofdel}
Take the alphabet to be \(\{e,f\}\), with \(f<e\). The Lyndon words are \(e\), and the Lyndon words in the \(fe^a\), ordered lexicographically. This order is the order \(\le\) on \(a\), and one identifies the Lyndon words in the \(fe^a\) with the Lyndon words in the alphabet \((\mathbb N,\le)\). If \(w'=(a_1,\ldots,a_d)\) is such a word, with image the word \(w\) in \(e\) and \(f\), the Lie polynomial \(P_w\) in \(e\) and \(f\) coincides with the Lie polynomial \(P_{w'}\) applied to the \((-\ad e)^a(f)\). By 2.4,
\[
P_w=fe^{a_1}\cdots fe^{a_d}+\hbox{remainder},
\]
where the remainder is a linear combination of monomials strictly larger in lexicographic order (for \(f<e\)), and with the same degrees in \(e\) and \(f\). When these degrees are fixed, this order on the monomials
\[
e^{b_0}fe^{b_1}\cdots fe^{b_d}
\]
is the opposite of the lexicographic order of
\[
\left(\sum_1^d b_i,\sum_2^d b_i,\ldots,b_d\right).
\]
Apply instead \(P_{w'}\) to the \((\ad e)^a(f)\). One obtains
\[
P_{w'}[(\ad e)^a(f)]=\pm fe^{a_1}\cdots fe^{a_d}+\hbox{remainder},
\]
where the remainder is a sum of \(e^{b_0}fe^{b_1}\cdots fe^{b_d}\) smaller in the lexicographic order of
\[
\left(\sum_1^d b_i,\sum_2^d b_i,\ldots,b_d\right).
\]
In a Lie polynomial in \(e\) and \(f\), a monomial and the monomial obtained by reversing the order of the letters occur with equal or opposite coefficients. Lemma 7.4 follows from the preceding discussion by reversing the order of the letters.
\end{proofdel}

\section*{8. \(N=6\)}

In this section \(N=6\), \(M=1\), \(\ell=3\), and \(A\) denotes the set of integers \(\ge2\), ordered by \(\ge\), which we write \(\prec\):
\[
\cdots\prec4\prec3\prec2.
\]
We shall use the notations \(L_S,\Pi_S\) of 4.1.

\begin{proposition*}[8.1]
The morphisms from \(L\) to \(L_{\{1,\zeta\}}\) and to \(L_{\{1,\zeta^{-1}\}}\) induce surjective morphisms
\[
\tag{8.1.1}
L\longrightarrow L_{\{1,\zeta\}}\times_{L_{\{1\}}}L_{\{1,\zeta^{-1}\}},
\]
\[
\tag{8.1.2}
\Pi\longrightarrow \Pi_{\{1,\zeta\}}\times_{\Pi_{\{1\}}}\Pi_{\{1,\zeta^{-1}\}}.
\]
\end{proposition*}

\begin{proofdel}
For every set \(E\), let \(\Lie(E)\) denote the Lie algebra freely generated by \(E\). The surjectivity of (8.1.2) follows from that of (8.1.1), and that of (8.1.1) is a special case of the following lemma.
\end{proofdel}

\begin{lemma*}[8.2]
If \(E\supset A\cup B\), then
\[
\tag{8.2.1}
\Lib(E)\longrightarrow \Lie(A)\times_{\Lib(A\cap B)}\Lib(B)
\]
is surjective.
\end{lemma*}

\begin{proofdel}
One reduces to the case \(E=A\cup B\). The surjectivity of (8.2.1) is equivalent to that of
\[
\tag{8.2.2}
\Ker(\Lib(E)\to\Lib(A\cap B))
\longrightarrow
\Ker(\Lib(A)\to\Lib(A\cap B))\times
\Ker(\Lib(B)\to\Lib(A\cap B)).
\]
For \(F\subset E\), if \(G\) is the \(\Lib(F)\)-module freely generated by \(E-F\) (the sum of \(E-F\) copies of the enveloping algebra of \(\Lib(F)\)), \(\Lib(E)\) is the semidirect product of \(\Lib(F)\) and the Lie algebra freely generated by the vector space \(G\) (Bourbaki Lie II \S2). Apply this to \(A\cap B\subset E\), \(A\cap B\subset A\), and \(A\cap B\subset B\). The surjectivity (8.2.2) is reduced, for the Lie algebra \(L(G)\) freely generated by a vector space \(G\), to the surjectivity of
\[
L(G_1\oplus G_2)\longrightarrow L(G_1)\times L(G_2).
\]
Indeed, the target is the largest quotient of \(L(G_1\oplus G_2)\) in which \(G_1\) and \(G_2\) commute.
\end{proofdel}

\heading{8.3.} Put \(X=\Gm-\{1,\zeta\}\). The inclusion of \(X\) in \(X\) sends the set \(S\) of tangential base points of \(X\), identified with \(\{0\}\cup\mu_N\), to a set of base points of \(X\), still denoted \(S\). See 1.6 for a similar construction. Let \(\bar S\) be the subset \(\{0,1,\zeta\}\) of \(S\).

The motivic fundamental groupoid of \((X,\bar S)\) is a quotient of that of \((X,S)\). In \(\omega\)-realization, one obtains the constant groupoid \(\Pi_{\{1,\zeta\}}\), a quotient of the constant groupoid \(\Pi\). It is the largest quotient in which the \(e_\eta\in\Lie\Pi_{\eta,\eta}\), for \(\eta\in\mu_N-\{1,\zeta\}\), vanish. Since the action of \((\Pi,\circ)\) on the groupoid \(\Pi\) respects these \(e_\eta\), it passes to the quotient, and \((\Pi,\circ)\) acts on the \(\omega\)-realization of \(P(X,\bar S)\), hence also, by restriction, on that of \(P(X,S)\).

Let \(\bar\Pi\) be the quotient
\[
\bar\Pi:=\Pi_{\{1,\zeta\}}\times_{\Pi_{\{1\}}}\Pi_{\{1,\zeta^{-1}\}}
\]
of \(\Pi\) (8.1).

\begin{proposition*}[8.4]
(i) The group law \(\circ\) of \(\Pi\) passes to the quotient and defines a group law \(\circ\) on \(\bar\Pi\).\\
(ii) The quotient \((\bar\Pi,\circ)\) of \((\Pi,\circ)\) is the quotient acting faithfully on \(\omega(P(X,S))\).
\end{proposition*}

\begin{proofdel}
Let \(\Pi^{\{1,\zeta\}}_{b,a}\) be the copy \(\omega(P_{b,a}(X))\) of \(\Pi_{\{1,\zeta\}}\), and let \(x_{b,a}\) be the copy in \(\Pi^{\{1,\zeta\}}_{b,a}\) of \(x\in\Pi_{\{1,\zeta\}}\). Let \(x\in\Pi\), and let \(x'\) and \(x''\) be its images in \(\Pi_{\{1,\zeta\}}\) and \(\Pi_{\{1,\zeta^{-1}\}}\). The action of \(x\) sends \(1_{1,0}\) to \(x'_{1,0}\) and \(1_{\zeta,0}\) to \(([\zeta]x'')_{\zeta,0}\). These images determine the action: \(e_0\in\Lie\Pi^{\{1,\zeta\}}_{0,0}\) is fixed; \(e_1\in\Lie\Pi^{\{1,\zeta\}}_{1,1}\) is fixed, and \(e_1\in\Lie\Pi^{\{1,\zeta\}}_{0,0}\) is obtained from it by transport along \(1_{1,0}\), that is, by
\[
y\longmapsto 1_{1,0}^{-1}y\,1_{1,0}:\Pi^{\{1,\zeta\}}_{1,1}\longrightarrow\Pi^{\{1,\zeta\}}_{0,0},
\]
and is therefore transformed into \((\ad x')^{-1}(e_1)\). Similarly, \(e_\zeta\in\Lie\Pi^{\{1,\zeta\}}_{\zeta,\zeta}\) is transformed into \((\ad[\zeta](x''))^{-1}(e_\zeta)\). This determines the action of \(x\) on \(\Lie\Pi^{\{1,\zeta\}}_{0,0}\), and hence on \(\Pi^{\{1,\zeta\}}_{0,0}\) and on the groupoid.

Thus two elements of \(\Pi\) have the same action if and only if they have the same image in the quotient \(\bar\Pi\) of \(\Pi\). This proves (i) and (ii).
\end{proofdel}

\heading{8.5.} The motivic fundamental groupoid \(P(X,S)\) is mixed Tate and has good reduction everywhere. The action of \(U_\omega\) on its \(\omega\)-realization therefore factors through \(\bar U_\omega\). The torsor \(P_{\zeta,0}(Y)\) is a quotient of \(P_{\zeta,0}(X,S)\); the action of \((\Pi,\circ)\) on \((\Pi_M)_{\zeta,0}\) therefore factors through \(\bar\Pi\): one has a commutative diagram

\tag{8.5.1}
\begin{tikzcd}[column sep=3.4em,row sep=2.1em]

\bar U_\omega \arrow[r] & (\bar\Pi,\circ) \arrow[r] & \Pi_M^{\prime\prime}\ltimes\Pi_M .
\end{tikzcd}

The role played in Sections 6 and 7 by \(d_{\mathrm{ch}},v_\sigma,IU_\omega,D,IU_\omega\circ D\) will here be played by their images

\bar d_{\mathrm{ch}},\quad \bar v_\sigma,\quad I\bar U_\omega,\quad \bar D,\quad I\bar U_\omega\circ\bar D

in \(\bar\Pi\), and we shall need analogues in \(\bar\Pi\) of the results of Section 5.

The action of \(\bar U_\omega\) on \((\Pi_M)_{\zeta,0}\) is faithful (4.11 (ii)). The second line of (8.5.1) is therefore injective, and

\tag{8.5.2}
\bar U_\omega\overset{\sim}{\longrightarrow}I\bar U_\omega\subset(\bar\Pi,\circ).

For \(\eta\in\{1,\zeta,\zeta^{-1}\}\), the functions \(c(e_0^{n-1}e_\eta)\) on \(\Pi\) factor through \(\bar\Pi\) (and indeed already through \(\Pi_{\{1,\zeta\}}\) or through \(\Pi_{\{1,\zeta^{-1}\}}\)). As in 5.4, the morphism

For \(n\ge1\), \(t^n\) is the restriction to \(I\bar U_\omega\circ\bar D\) of an element of codegree \(n\) in \(D^1\OO(\bar\Pi)\).

\heading{8.8.} The Lyndon words in the alphabet \(A\) will be called Lyndon words in the integers \(\ge2\). To such a Lyndon word \(n=(n_1,\ldots,n_d)\), attach \(\zeta(n_1,\ldots,n_d;\zeta,1,\ldots,1)\), written simply \(\zeta(n;\zeta,1,\ldots)\), and, to a formal linear combination with integer coefficients \(\ge0\) of such words,
\[
\sum\lambda(n)[n],
\]
the product
\[
\prod\zeta(n;\zeta,1,\ldots)^{\lambda(n)}.
\]
Weight and depth are defined as in 7.1.

\begin{theorem*}[8.9]
Fix \(w,d\ge0\). Every product
\[
\tag{8.9.1}
\zeta(s'_1,\ldots,s'_{r'};\eta'_1,\ldots,\eta'_{r'})
\zeta(s''_1,\ldots,s''_{r''};\eta''_1,\ldots,\eta''_{r''})
\]
of weight \(\sum s'_i+\sum s''_i=w\) and depth \(r'+r''\le d\), such that all products \(\eta'_1\cdots\eta'_i\) lie in \(\{1,\zeta\}\), all products \(\eta''_1\cdots\eta''_i\) lie in \(\{1,\zeta^{-1}\}\), and \((s'_1,\eta'_1),(s''_1,\eta''_1)\ne(1,1)\), is a rational linear combination of
\[
\tag{8.9.2}
\prod\zeta(n;\zeta,1,\ldots)^{\lambda(n)}(2\pi i)^p,
\]
where \(\sum\lambda(n)[n]\) is a formal linear combination with coefficients \(\ge0\) of Lyndon words in the integers \(\ge2\) and \(p\ge0\), satisfying
\[
p+\operatorname{weight}\Bigl(\sum\lambda(n)[n]\Bigr)=w,
\]
and, if \(p=0\), \(\operatorname{depth}(\sum\lambda(n)[n])\le d\), whereas, if \(p>0\), \(\operatorname{depth}(\sum\lambda(n)[n])\le d-1\). Moreover, conjecture 5.6 implies that the quantities (8.9.2) are \(\Q\)-linearly independent.
\end{theorem*}

The conditions imposed on the products (8.9.1) are expressed more simply in the language of coefficients of monomials. Up to sign, the products (8.9.1) are obtained by evaluating at \(d_{\mathrm{ch}}\) a product
\[
\tag{8.9.3}
c(e_0^{s'_1-1}e_{\varepsilon'_1}e_0^{s'_2-1}e_{\varepsilon'_2}\cdots)
\,c(e_0^{s''_1-1}e_{\varepsilon''_1}e_0^{s''_2-1}e_{\varepsilon''_2}\cdots).
\]
The condition on the \(\eta'\) and \(\eta''\) is equivalent to requiring the \(\varepsilon'_i\) to lie in \(\{1,\zeta^{-1}\}\) and the \(\varepsilon''_i\) to lie in \(\{1,\zeta\}\), so that (8.9.3) factors through \(\bar\Pi\) (the first factor through \(\Pi_{\{1,\zeta^{-1}\}}\), the second through \(\Pi_{\{1,\zeta\}}\)), and the remaining conditions ensure that (8.9.3) has codegree \(w\) and lies in \(D^d\OO(\bar\Pi)\).

With this observation, the proof of 8.9 is identical with that of 7.2, \(D^1\Pi\) being replaced by \(D^1\bar\Pi\).

\section*{Appendix: Betti/\(\omega\) comparison}

\heading{A.1.} Let \(k\) be a number field, \(S\) a finite set of prime numbers, \(\MT(\OO_S)\) the category of mixed Tate motives over \(k\) with good reduction outside \(S\), \(\omega\) its fibre functor (3.1.1), \(G_\omega=\Gm\ltimes U_\omega\) the group scheme of automorphisms of \(\omega\), \(\sigma\) an embedding of \(k\) in an algebraic closure \(\C\) of \(\R\), and \(M\mapsto M_\sigma\) the corresponding ``Betti realization'' fibre functor.

The definition of \(M\mapsto M_\sigma\) uses the topology of \(\C\), but not the choice of a square root of \(-1\). In order to reason by transport of structure, we prefer not to express ourselves in terms of a complex embedding (i.e. into the algebraic closure \(\C\) of \(\R\)). The de Rham realization \(M_{\mathrm{DR}}\) of \(M\) is \(M_\sigma\otimes k\), and the Betti/de Rham comparison isomorphism gives
\[
\tag{A.1.1}
\omega(M)\otimes\C\overset{\sim}{\longrightarrow}M_\sigma\otimes\C.
\]
By this isomorphism, \(M_\sigma\) will be regarded as a new \(\Q\)-structure on \(\omega(M)\otimes\C\). It is known that there exists \(a_\sigma\in G_\omega(\C)\) such that, for every \(M\),
\[
\tag{A.1.2}
a_\sigma\omega(M)=M_\sigma.
\]
The \(a_\sigma\) satisfying (A.1.2) form a \(G_\omega(\Q)\)-torsor \(T_{\sigma/\omega}\subset G_\omega(\C)\). Testing (A.1.2) on \(\Q(1)\), one sees that the projection of \(T_{\sigma/\omega}\) into \(\Gm(\C)=\C^*\) is \(2\pi i\,\Q^*\).

Let \(\tau\) be the inclusion of \(\Gm\) in \(G_\omega\) defined by the grading of \(\omega\). If \(G_\omega\) acts on an object, for example on \(U_\omega\) by inner automorphisms, we shall also denote by \(\tau(\lambda)\) the action of \(\tau(\lambda)\). For \(\lambda\in2\pi i\,\Q^*\), the \(u\in U_\omega(\C)\) such that \(u\tau(\lambda)\) belongs to \(T_{\sigma/\omega}\) form a \(U_\omega(\Q)\)-torsor, with \(x\in U_\omega(\Q)\) acting by
\[
\tag{A.1.3}
\hbox{right multiplication by }\tau(\lambda)[x].
\]
Indeed, \(u\tau(\lambda)x=u\tau(\lambda)[x]\cdot\tau(\lambda)\).

\begin{theorem*}[Conjecture A.2]
\(a_\sigma\) is a generic point of \(G_\omega\).
\end{theorem*}

This is the analogue, for mixed Tate motives, of a conjecture of Grothendieck for pure motives.

\begin{proposition*}[A.3]
Conjecture A.2 implies conjecture 5.6.
\end{proposition*}

\begin{proofdel}
To conform to the notation of Section 5, assume that \(\sigma\) is an embedding of the cyclotomic field \(k\) in \(\C\). The element \(d_{\mathrm{ch}}\) (5.1 (A)) of \(\Pi(\C)\) depends on \(\sigma\). We denote it here by \(d_{\mathrm{ch},\sigma}\). Viewed as an element of \((\Pi,\circ)\), it acts on the \(\omega\)-realization of the motivic fundamental groupoid of \((X,\{0\}\cup\mu_N)\).

Recall the proof [1] 5.19 of 5.1 (B). If one takes \(a_\sigma\) of A.1 in the form \(u_\sigma\tau(2\pi i)\), with \(u_\sigma\in U_\omega(\C)\), whose image is \((3.4.6)(u_\sigma)\) in \(\Pi(\C)\), then both \((3.4.6)(u_\sigma)\tau(2\pi i)\) and \(d_{\mathrm{ch},\sigma}\tau(2\pi i)\) transform the \(\omega\)-realization of the fundamental groupoid into its \(\sigma\)-realization. If \(v_\sigma\) is defined by
\[
d_{\mathrm{ch},\sigma}\tau(2\pi i)=(3.4.6)(u_\sigma)\tau(2\pi i)v_\sigma
\]
(equality of actions, or in \((\Gm\ltimes\Pi)(\C)\)), then \(v_\sigma\in\Pi(\Q)\) and (5.1.1) takes the form
\[
\tag{A.3.1}
d_{\mathrm{ch},\sigma}=(3.4.6)(u_\sigma)\circ\tau(2\pi i)(v_\sigma).
\]
Conjecture A.2 asserts that \((u_\sigma,2\pi i)\) is \(\Q\)-Zariski dense in \(U_\omega\times\A^1\). It implies that the image \(d_{\mathrm{ch},\sigma}\) of \((u_\sigma,2\pi i)\) under the morphism
\[
(x,t)\longmapsto(3.4.6)(x)\circ\tau(t)(v_\sigma)
\]
from \(U_\omega\times\A^1\) onto \(IU_\omega\circ D\) is Zariski dense.
\end{proofdel}

\heading{A.4.} We have no tool for attacking either conjecture A.2 or 5.6. For proving that a subgroup scheme of \(U_\omega\) coincides with \(U_\omega\), the following proposition sometimes has the same effect. It follows from what is known about \(U_\omega^{\mathrm{ab}}\) ([1] (A.14.1) and [1] 2.1.3) and from the injectivity of Borel's regulator [1] 1.6.10.

\begin{proposition*}[A.4]
Let \(H\) be a subgroup scheme of the group scheme \((U_\omega)_\C\) over \(\C\). In order for it to be equal to \((U_\omega)_\C\), it is enough that:
\begin{enumerate}[label=(\alph*)]
\item it be graded (i.e. stable under the action of \(\Gm\));
\item in degree one, its graded Lie algebra coincide with that of \(U_\omega\);
\item for every complex embedding \(\sigma\), it contain \(a_\sigma\bar a_\sigma^{-1}\tau(-1)\).
\end{enumerate}
\end{proposition*}

Note that \(a_\sigma\bar a_\sigma^{-1}\tau(-1)\) lies in \(U_\omega(\C)\) and is independent of the choice of \(a_\sigma\). For \(a_\sigma=u_\sigma\tau(\lambda)\), with \(u_\sigma\in U_\omega(\C)\) and \(\lambda\in2\pi i\,\Q^*\), it is \(u_\sigma\tau(-1)(\bar u_\sigma)\).

\heading{A.5.} Return to the general hypotheses 3.1. In [2], Racinet defines a graded subscheme \(\DMRD_\sigma\) of \(\Pi\), containing \(d_{\mathrm{ch},\sigma}\). It is defined by ``standard'' relations between the coefficients of \(d_{\mathrm{ch},\sigma}\). He proves that its subscheme \(\DMRD_0\), obtained by setting (5.4.3) to zero, is a subgroup scheme of \((\Pi,\circ)\), independent of \(\sigma\), and that, for a suitable \(w_\sigma\in\Pi(\Q)\), \(\DMRD_\sigma\) is the image of \(\DMRD_0^\sigma\times\A^1\) by
\[
(x,t)\longmapsto x\circ\tau(t)(w_\sigma).
\]
The element \(d_{\mathrm{ch},\sigma}\) lies in the image of \(\DMRD_0^\sigma(\C)\times\{2\pi i\}\), and therefore \(d_{\mathrm{ch},\sigma}\circ\tau(-1)(d_{\mathrm{ch},\sigma})^{-1}\) lies in \(\DMRD_0\). This holds for every complex embedding, and A.4 applies to the inverse image of \(\DMRD_0\) in \(U_\omega\). We conclude:

\begin{corollary*}[A.5]
Racinet's subgroup scheme \(\DMRD_0\) contains \(IU_\omega\).
\end{corollary*}

\heading{A.6.} Return to the hypotheses of A.1, and suppose that \(\sigma(k)\subset\C\) is stable under complex conjugation. This is the case if \(\sigma\) is a real embedding, if \(k\) is CM, for instance cyclotomic, or if \(k\) is Galois over \(\Q\). By hypothesis, there is an automorphism of complex conjugation, denoted \(c\), of the triple \((k,\sigma,\C)\). By transport of structure, it acts on \(G_\omega\), and \(T_{\sigma/\omega}\) is stable under the involution \(c\) of \(G_\omega(\C)\). Note that \(c\) acts through its action on \(G_\omega\) and on \(\C\).

\begin{proposition*}[A.7]
There exists \(a_\sigma\in T_{\sigma/\omega}\) such that
\[
\tag{A.7.1}
c(a_\sigma)=a_\sigma\tau(-1).
\]
If \(a_\sigma=u\tau(\lambda)\), with \(u\in U_\omega(\C)\) and \(\lambda\in2\pi i\,\Q^*\), then (A.7.1) is equivalent to \(u\) being fixed by \(c\).
\end{proposition*}

\begin{proofdel}
Fix \(\lambda\in2\pi i\,\Q^*\), and let \(T^\lambda_{\sigma/\omega}\) be the set of \(u\in U_\omega(\C)\) such that \(u\tau(\lambda)\) belongs to \(T_{\sigma/\omega}\). It is a \(U_\omega(\Q)\)-torsor for the action (A.1.3). The involution \(c\) of \(U_\omega(\C)\) stabilizes \(T^\lambda_{\sigma/\omega}\): if \(u\tau(\lambda)\) belongs to \(T_{\sigma/\omega}\), then \(c(u\tau(\lambda))\) and \(c(u\tau(\lambda))\tau(-1)\) also belong to it, and
\[
c(u\tau(\lambda))\tau(-1)=c(u)\tau(-\lambda)\tau(-1)=c(u)\tau(\lambda).
\]
The involution \(c\) of \(T^\lambda_{\sigma/\omega}\) makes \(T^\lambda_{\sigma/\omega}\) an equivariant \(U_\omega(\Q)\)-torsor, relative to the involution
\[
\tau(-1)\circ c:x\longmapsto\tau(-1)c(x)\tau(-1)^{-1}
\]
of \(U_\omega(\Q)\). Indeed,
\[
c(u\cdot\tau(\lambda)x\tau(\lambda)^{-1})
=c(u)\cdot\tau(-\lambda)c(x)\tau(-\lambda)^{-1}
=c(u)\cdot\tau(\lambda)\tau(-1)[c(x)]\tau(\lambda)^{-1}.
\]
Every equivariant \(U_\omega(\Q)\)-torsor is trivial, as follows by dévissage from the vanishing
\[
H^1(\mathbb Z/2,V)=0
\]
for a \(\Q\)-vector space \(V\) with an action of \(\mathbb Z/2\). There is therefore \(u\in T^\lambda_{\sigma/\omega}\) fixed by \(c\), and \(u\tau(\lambda)\) satisfies (A.7.1).
\end{proofdel}

\begin{corollary*}[A.8]
The element \(v_\sigma\) of (5.1.1) can be chosen fixed by the composite of \(\tau(-1)\) and the automorphism of \(\Pi\) induced by complex conjugation of \(k\).
\end{corollary*}

\begin{proofdel}
Choose \(a_\sigma\in T_{\sigma/\omega}\) of the form \(a_\sigma=u\tau(2\pi i)\) and satisfying (A.7.1). Let \(u_1\) be the image of \(u\) in \(IU_\omega(\C)\), and define \(v_\sigma\) by (A.3.1):
\[
d_{\mathrm{ch}}=u_1\cdot\tau(2\pi i)[v_\sigma].
\]
Complex conjugation of \(k\) induces an involution of \(\Pi\), deduced from \(e_a\mapsto e_{\bar a}\) on \(L\). This involution, together with complex conjugation of \(\C\), defines the involution \(c\) of \(\Pi(\C)\). Both \(d_{\mathrm{ch}}\) and \(u_1\) are fixed by this involution; hence \(\tau(2\pi i)[v_\sigma]\) is also fixed. Since
\[
c(\tau(2\pi i)[v_\sigma])=\tau(-2\pi i)[c(v_\sigma)]
=\tau(2\pi i)[\tau(-1)[c(v_\sigma)]],
\]
\(v_\sigma\) is fixed by \(\tau(-1)\circ c\).
\end{proofdel}
