\noindent\textsc{Inventiones math.} \textbf{8}, 238--243 (1969)

\bigskip
\begin{center}
\Large\textbf{Vari\'et\'es ab\'eliennes ordinaires sur un corps fini}
\end{center}

\begin{center}
\textsc{Pierre Deligne} (Bures-sur-Yvette)
\end{center}

\bigskip
On donne une description terre \`a terre de la cat\'egorie des vari\'et\'es
ab\'eliennes ordinaires sur un corps fini $\mathbf{F}_q$.
Le r\'esultat obtenu a \'et\'e inspir\'e par Ihara [2] ch~V (voir aussi [3]).

\smallskip
\textbf{1.} Soient $p$ un nombre premier, $\mathbf{F}_p$ le corps
$\mathbf{Z}/(p)$, $\overline{\mathbf{F}}_p$ une cl\^oture alg\'ebrique de
$\mathbf{F}_p$ et, pour toute puissance $q$ de $p$, soit $\mathbf{F}_q$
le sous-corps \`a $q$ \'el\'ements de $\overline{\mathbf{F}}_p$.
Pour toute extension alg\'ebrique $k$ de $\overline{\mathbf{F}}_p$, on
d\'esigne par $W_0(k)$ l'anneau de valuation discr\`ete hens\'elien
essentiellement de type fini sur $\mathbf{Z}$, absolument non ramifi\'e,
de corps r\'esiduel $k$; soient $W(k)$ l'anneau des vecteurs de Witt
sur $k$, i.e.\ le compl\'et\'e de $W_0(k)$, $W = W(\overline{\mathbf{F}}_p)$
et $\varphi$ un plongement de $W$ dans le corps $\mathbf{C}$ des nombres
complexes. On d\'esigne par $\mathbf{Z}(1)$ le sous-groupe $2\pi i\,\mathbf{Z}$
de $\mathbf{C}$. L'application exponentielle d\'efinit un isomorphisme
entre $\mathbf{Z}(1) \otimes \mathbf{Z}_l$ et
$\mathbf{Z}_l(1)(\mathbf{C}) = \varprojlim \mu_{l^n}(\mathbf{C})$.

On d\'esigne par $A^*$ la vari\'et\'e ab\'elienne duale d'une vari\'et\'e
ab\'elienne $A$. Pour tout corps $k$, on d\'esigne par $\bar{k}$ une
cl\^oture alg\'ebrique de $k$.

\smallskip
\textbf{2.} Soit $A$ une vari\'et\'e ab\'elienne de dimension $g$, d\'efinie
sur un corps $k$ de caract\'eristique $p$. Rappelons que $A$ est dite
\emph{ordinaire} si les conditions \'equivalentes suivantes sont
v\'erifi\'ees:
\begin{enumerate}
\item[(I)] $A$ a $p^g$ points d'ordre divisant $p$ \`a valeur dans $\bar{k}$.
\item[(II)] La \guillemotleft matrice de Hasse-Witt\guillemotright\
$F^*\colon H^1(A^{(p)}, \mathscr{O}_{A^{(p)}}) \to H^1(A, \mathscr{O}_A)$
est inversible.
\item[(III)] La composante neutre du sch\'ema en groupe $A_p$ noyau de
la multiplication par $p$ est de type multiplicatif, (donc
g\'eom\'etriquement isomorphe \`a une puissance de $\mu_p$).
\end{enumerate}

Si $k = \mathbf{F}_q$, si $F$ est l'endomorphisme de Frobenius de $A$
et $Pc_A(F;x)$ son polyn\^ome caract\'eristique, ces conditions
\'equivalent encore \`a
\begin{enumerate}
\item[(IV)] La moiti\'e au moins des racines de $Pc_A(F;x)$ dans
$\overline{\mathbf{Q}}_p$ sont des unit\'es $p$-adiques. En d'autres
termes, si $n = \dim A$, la r\'eduction modulo $p$ du polyn\^ome
$Pc_A(F;x)$ n'est pas divisible par $x^{n+1}$.
\end{enumerate}

\smallskip
\textbf{3.} Soit $A$ une vari\'et\'e ab\'elienne ordinaire sur
$\overline{\mathbf{F}}_p$. On d\'esigne par $\tilde{A}$ le rel\`evement
canonique de Serre--Tate [4] de $A$ sur $W$. Rappelons que
$\tilde{A}$ d\'epend fonctoriellement de $A$ et est caract\'eris\'e
par le fait que le groupe $p$-divisible $T_p(\tilde{A})$ sur $W$
attach\'e \`a $\tilde{A}$ [5] est le produit des groupes $p$-divisibles
(uniquement d\'etermin\'es d'apr\`es 2.~(III)) relevant respectivement
la composante neutre et le plus grand quotient \'etale de $T_p(A)$.
Le rel\`evement canonique $\tilde{A}$ est encore l'unique rel\`evement
de $A$ tel que tout endomorphisme de $A$ se rel\`eve \`a $\tilde{A}$.
On d\'esigne par $T(A)$ l'homologie enti\`ere de la vari\'et\'e
ab\'elienne complexe $A_c$ d\'eduite de $\tilde{A}$ et de $\varphi$
par extension des scalaires de $W$ \`a $\mathbf{C}$:
\[
T(A) = H_1(\tilde{A} \otimes_W \mathbf{C}).
\]
On sait que $\tilde{A}$ se descend de fa\c{c}on unique \`a
$W_0(\overline{\mathbf{F}}_p)$, de sorte que $A_c$ ne d\'epend que de
$A$ et de la restriction de $\varphi$ \`a $W_0(\overline{\mathbf{F}}_p)$.
Le $\mathbf{Z}$-module libre $T(A)$ est de rang $2.\dim(A)$; il est
fonctoriel en $A$. De plus, si $l$ est un nombre premier $\neq p$,
on a fonctoriellement
\begin{equation}
T(A) \otimes \mathbf{Z}_l = T_l(A). \tag{3.1}
\end{equation}

Le rel\`evement canonique de la vari\'et\'e ab\'elienne $A^*$ duale de
$A$ est la duale de $\tilde{A}$, de sorte que $(A_c)^* = \widetilde{A^*}$
et que $T(A)$ et $T(A^*)$ sont en dualit\'e parfaite \`a valeurs dans
$\mathbf{Z}(1)$:
\begin{equation}
T(A) \otimes T(A^*) \longrightarrow \mathbf{Z}(1) \tag{3.2}
\end{equation}
(il est indispensable d'utiliser $\mathbf{Z}(1)$ de pr\'ef\'erence \`a
$\mathbf{Z}$ pour obtenir une th\'eorie invariante par conjugaison
complexe). Les accouplements (3.2) sont compatibles, via (3.1), aux
accouplements
\[
T_l(A) \otimes T_l(A^*) \longrightarrow \mathbf{Z}_l(1);
\]
un morphisme $\phi\colon A \to A^*$ d\'efinit une polarisation de $A$
si et seulement si $\tilde{\phi}\colon \tilde{A} \to \widetilde{A^*}$
d\'efinit une polarisation de $A_c$. Posons
$T'_p(A) = \Hom(\mathbf{Q}_p/\mathbf{Z}_p, A(\overline{\mathbf{F}}_p))$
et $T''_p(A) = \Hom_{\mathbf{Z}_p}(T'_p(A^*), \mathbf{Z}(1) \otimes \mathbf{Z}_p)$.
Ces $\mathbf{Z}_p$-modules sont des foncteurs covariants de $A$.

Par d\'efinition du rel\`evement canonique, le groupe $p$-divisible
$T_p(\tilde{A})$ est somme du groupe pro-\'etale constant $T'_p(A)$ et
du dual de Cartier de $T'_p(A^*)$.\footnote{On d\'esigne par $T_p$
le module de Tate $p$-adique, et par $T'_p$ la composante \'etale du
groupe $p$-divisible.} Pour tout morphisme $u\colon A \to B$, le
morphisme induit $u\colon T_p(A) \to T_p(B)$ s'identifie \`a la somme
de $u|T'_p(A)\colon T'_p(A) \to T'_p(B)$ et du transpos\'e de Cartier
de ${}^t u|T'_p(B^*)\colon T'_p(B^*) \to T'_p(A^*)$. Sur $\mathbf{C}$,
on a canoniquement $\mathbf{Z}(1)/(p^n) \simeq \boldsymbol{\mu}_{p^n}$,
d'o\`u un isomorphisme de foncteurs:
\begin{equation}
T_p(A) = T(A) \otimes \mathbf{Z}_p = T'_p(A) \oplus T''_p(A). \tag{3.3}
\end{equation}

\smallskip
\textbf{4.} Rappelons que si $\varphi\colon X \to Y$ est une isog\'enie
entre vari\'et\'es ab\'eliennes complexes, la suite exacte d'homotopie
se r\'eduit \`a une suite exacte courte
\[
0 \longrightarrow H_1(X) \longrightarrow H_1(Y) \longrightarrow
\Ker(\varphi) \longrightarrow 0.
\]
Les vari\'et\'es ab\'eliennes quotients de $X$ par un sous-groupe fini,
et ces sous-groupes finis de $X$, correspondent bijectivement aux
sous-r\'eseaux de $H_1(X) \otimes \mathbf{Q}$ contenant $H_1(X)$.

Soit $A$ une vari\'et\'e ab\'elienne ordinaire sur
$\overline{\mathbf{F}}_p$. Si $n$ est un entier premier \`a $p$, les
sous-sch\'emas en groupe finis d'ordre $n$ de $A$, de $\tilde{A}$ et de
$A_c$ se correspondent bijectivement, et correspondent encore aux
sur-r\'eseaux $R$ de $T(A)$ tels que $[R:T(A)] = n$.

Posons $V'_p = T'_p(A) \otimes \mathbf{Q}_p$ et
$V''_p(A) = T''_p(A) \otimes \mathbf{Q}_p$. Les sous-sch\'emas en groupes
finis d'ordre $p^k$ de $A$ sont produits d'un sous-groupe \'etale et
d'un sous-groupe infinit\'esimal. Les sous-groupes \'etales d'ordre
$p^k$ de $A$ correspondent \`a ceux des sous-groupes d'ordre $p^k$ de
$A_c$ tels que le sur-r\'eseau $R$ correspondant de $T(A)$ soit
contenu dans $T'_p(A) + V''_p(A)$. Par dualit\'e, les sous-groupes
infinit\'esimaux de $A$ correspondent aux sur-r\'eseaux $R$ de $T(A)$,
$p$-isog\`enes \`a $T(A)$, i.e.\ tels que $[R:T(A)]$ soit une puissance
de $p$, et contenus dans $T''_p(A) + V'_p(A)$.

Au total, les sous-groupes finis de $A$, ou les vari\'et\'es ab\'eliennes
quotient de $A$, correspondent bijectivement aux sur-r\'eseaux $R$ de
$T(A)$ tels que
\begin{equation}
R \otimes \mathbf{Z}_p = (R \otimes \mathbf{Z}_p \cap V'_p) +
(R \otimes \mathbf{Z}_p \cap V''_p). \tag{4.1}
\end{equation}

\smallskip
\textbf{5.} En particulier, $A^{(p)}$, quotient de $A$ par le plus
grand sous-groupe infinit\'esimal de $A$ tu\'e par $p$ (pour $A$
ordinaire), est d\'efini par le sur-r\'eseau $T(A)^{(p)}$ de $T(A)$,
$p$-isog\`ene \`a $T(A)$, tel que
\[
T(A)^{(p)} \otimes \mathbf{Z}_p = T'_p(A) + V''_p(A).
\]

\smallskip
\textbf{6.} Soient $A$ une vari\'et\'e ab\'elienne sur $\mathbf{F}_q$
et $F\colon x \mapsto x^q$ son endomorphisme de Frobenius. Rappelons
que $A$ est uniquement d\'etermin\'e par le couple $(\bar{A}, F)$
d\'eduit de $(A, F)$ par extension des scalaires de $\mathbf{F}_q$ \`a
$\overline{\mathbf{F}}_q$; l'endomorphisme $F$ de $\bar{A}$ se factorise
par le Frobenius relatif $\Fr(q)\colon \bar{A} \to \bar{A}^{(q)}$ et un
isomorphisme $F'\colon \bar{A}^{(q)} \to \bar{A}$. Si $A$ est ordinaire,
on d\'esignera par $T(A)$ le $\mathbf{Z}$-module $T(\bar{A})$ muni de
l'endomorphisme $F$ induit par l'endomorphisme de Frobenius de
$\bar{A}$. En vertu de 5., de ce qui pr\'ec\`ede, et de (3.3), les
r\'eseaux $T(A)$ et $F(T(A))$ sont $p$-isog\`enes et on a
\begin{equation}
F(T'_p(A)) = T'_p(A), \tag{6.1}
\end{equation}
\begin{equation}
F(T''_p(A)) = q\,T''_p(A). \tag{6.2}
\end{equation}

\smallskip
\textbf{7.}
\begin{theorem}
Le foncteur $A \mapsto (T(A), F)$ est une \'equivalence de cat\'egories
entre la cat\'egorie des vari\'et\'es ab\'eliennes ordinaires sur
$\mathbf{F}_q$ et la cat\'egorie des $\mathbf{Z}$-modules libres de type
fini $T$ munis d'un endomorphisme $F$ v\'erifiant les conditions
suivantes
\begin{enumerate}
\item[(a)] $F$ est semi-simple, et ses valeurs propres sont de valeur
absolue complexe $q^{\frac{1}{2}}$,
\item[(b)] la moiti\'e au moins des racines dans $\overline{\mathbf{Q}}_p$
du polyn\^ome caract\'eristique de $F$ sont des unit\'es $p$-adiques;
en d'autres termes, si $T$ est de rang $d$, la r\'eduction mod~$p$
du polyn\^ome $Pc_T(F;x)$ n'est pas divisible par
$x^{[d/2]+1}$,
\item[(c)] il existe un endomorphisme $V$ de $T$ tel que $FV = q$.
\end{enumerate}
Si la condition (a) est v\'erifi\'ee, les conditions (b) et (c)
\'equivalent \`a
\begin{enumerate}
\item[(d)] Le module $T \otimes \mathbf{Z}_p$ admet une d\'ecomposition,
stable par $F$, en deux sous-$\mathbf{Z}_p$-modules $T'_p$ et $T''_p$
de m\^eme dimension, tels que $F|T'_p$ soit inversible et $F|T''_p$
divisible par $q$.
\end{enumerate}
\end{theorem}

\begin{proof}
(A) Prouvons que (a)${}+{}$(b)${}+{}$(c) $\Rightarrow$ (d). Si $\alpha$
est une valeur propre complexe de $F$, $\bar{\alpha}$ en est une autre,
de m\^eme multiplicit\'e, et $\alpha\bar{\alpha} = q$. Si l'on excepte
celles \'egales \`a $\pm q^{\frac{1}{2}}$, les valeurs propres de $F$ dans
$\mathbf{C}$, donc dans $\mathbf{Q}_p$, se rangent en paires de racines
$\alpha$ et $q/\alpha$. Les racines $\alpha$ et $q/\alpha$ ne pouvant
\^etre simultan\'ement des unit\'es $p$-adiques, il r\'esulte de (b)
que $\pm q^{\frac{1}{2}}$ n'est pas valeur propre de $F$, que la
moiti\'e des valeurs propres de $F$ dans $\mathbf{Q}_p$ est form\'ee
d'unit\'es $p$-adiques, soit $\alpha_1, \dots, \alpha_{d/2}$, et que
l'autre moiti\'e est form\'ee de $\beta_1 = q/\alpha_1, \dots,
\beta_{d/2} = q/\alpha_{d/2}$. Soient $T_p = T \otimes \mathbf{Z}_p$,
$V_p = T \otimes \mathbf{Q}_p$, $V'_p$ le sous-espace de $V_p$ noyau de
$\prod_i (F - \alpha_i)$ et $V''_p$ le noyau de l'endomorphisme
$\psi = \prod_i (F - \beta_i)$. On a $V_p = V'_p \oplus V''_p$.
Soient $T'_p$ la projection de $T_p$ sur $V'_p$ et
$T''_p = T_p \cap V''_p$. Puisque $\psi$ annule $V'_p$ et respecte
$T$, il envoie $T_p$ dans $T_p \cap V''_p \subset T''_p$. Par ailleurs,
$\det(\psi|V'_p) = \prod_{i,j}(\alpha_i - \beta_j)$ est une unit\'e
$p$-adique, donc $\psi(T_p) = T_p$ et $T_p \cap V'_p = \psi T_p$, de
sorte que $T_p = T'_p \oplus T''_p$.

(B) Pleine fid\'elit\'e. Soient $A$ et $B$ deux vari\'et\'es ab\'eliennes
sur $\mathbf{F}_q$, et soit $\psi$ la fl\`eche
\[
\psi\colon \Hom(A, B) \longrightarrow \Hom_{\mathbf{Z}}(T(A), T(B)).
\]
En vertu du th\'eor\`eme de Tate [7] et de (3.1), la fl\`eche
\[
\psi_l\colon \Hom(A, B) \otimes \mathbf{Z}_l \longrightarrow
\Hom_{\mathbf{Z}}(T(A), T(B)) \otimes \mathbf{Z}_l
\]
est un isomorphisme pour $(l, p) = 1$, de sorte que $\psi \otimes
\mathbf{Q}$ est un isomorphisme. On sait que $\Hom(A, B)$ est sans
torsion, donc $\psi$ injectif. Soit $u\colon A \to B$ un morphisme tel
que $T(u)$ soit divisible par $n$. Le morphisme induit
$u_c\colon A_c \to B_c$ est alors divisible par $n$, donc aussi
$\tilde{u}\colon \tilde{A} \to \tilde{B}$ au point g\'en\'erique de $W$.
Le noyau de la multiplication par $n$ est plat sur $W$;
s'annule donc sur ce noyau, $\tilde{u}$ et $u$ sont divisibles par $n$,
et $\psi$ est bijectif.

(C) N\'ecessit\'e. Que $(T(A), F)$ v\'erifie (a) r\'esulte de Weil;
la condition (d), qui implique (b) et (c), r\'esulte de 6.

(D) Isog\'enies. Soient $(T_0, F)$ v\'erifiant (a)--(d) et $T$ un
r\'eseau dans $T_0 \otimes \mathbf{Q}$, stable par $F$, v\'erifiant
encore (d). Supposons que $(T_0, F)$ soit l'image d'une vari\'et\'e
ab\'elienne $\bar{A}$ sur $\overline{\mathbf{F}}_q$ et prouvons que
$(T, F)$ provient d'une vari\'et\'e ab\'elienne isog\`ene. Rempla\c{c}ant
$T$ par $\frac{1}{n}T$, qui lui est isomorphe, on peut supposer que
$T \supset T_0$. La condition (d) implique que $T$ v\'erifie (4.1) et
$T$ d\'efinit un sous-groupe $H$ de $\bar{A}$, qui est d\'efini sur
$\mathbf{F}_q$, et $(T, F) = T(\bar{A}/H)$.

(E) Surjectivit\'e. Le foncteur $T$ induit un foncteur $T_{\mathbf{Q}}$
de la cat\'egorie des vari\'et\'es ab\'eliennes ordinaires \`a isog\'enie
pr\`es sur $\mathbf{F}_q$ dans la cat\'egorie des $\mathbf{Q}$-espaces
vectoriels de rang fini, munis d'un automorphisme $F$ v\'erifiant
(a)--(b). En vertu de (D), il suffit de prouver que ce foncteur
$T_{\mathbf{Q}}$ est essentiellement surjectif. Il suffit m\^eme de
montrer que tout objet simple $(V, F)$ de la cat\'egorie image est
atteint. D'apr\`es Honda [1] (voir aussi [6]), il existe une vari\'et\'e
ab\'elienne $\bar{A}$ sur $\overline{\mathbf{F}}_q$, telle que le polyn\^ome
caract\'eristique du Frobenius $F_A$ de $A$ soit une puissance de celui
de $F$. La troisi\`eme caract\'erisation 1.\ des vari\'et\'es ab\'eliennes
ordinaires montre que $\bar{A}$ est ordinaire. De plus,
$(T(A) \otimes \mathbf{Q}, F)$ est somme de copies de $(V, F)$, donc,
d'apr\`es (B), la vari\'et\'e ab\'elienne \`a isog\'enie pr\`es
$A \otimes \mathbf{Q}$ est somme de copies d'une vari\'et\'e ab\'elienne
$B$ v\'erifiant $T(B) \otimes \mathbf{Q} = (V, F)$.
\end{proof}

\smallskip
\textbf{8.} Soit $(T, F)$ un couple v\'erifiant les hypoth\`eses du
th\'eor\`eme, $2g$ le rang de $T$. \`A la vari\'et\'e ab\'elienne
sur $\mathbf{F}_q$ correspondante et $A_c$ la vari\'et\'e ab\'elienne
complexe qui s'en d\'eduit (3.). On a
\[
T = H_1(A_c)
\]
de sorte que $T \otimes \mathbf{R}$ s'identifie \`a l'alg\'ebre de Lie de
$A_c$ et est en tant que tel muni d'une structure complexe. Voici,
d'apr\`es J.-P.\ Serre, comment reconstruire cette structure complexe
en terme de $T$, $F$ et de la restriction de $\varphi$ \`a
$W_0(\overline{\mathbf{F}}_p)$.

\begin{proposition}
La structure complexe d\'efinie plus haut sur $T \otimes \mathbf{R}$ est
caract\'eris\'ee par les propri\'et\'es suivantes :
\begin{enumerate}
\item[(I)] L'endomorphisme $F$ est $\mathbf{C}$-lin\'eaire.
\item[(II)] Si $v$ est la valuation de la cl\^oture alg\'ebrique
$\overline{\mathbf{Q}}$ de $\mathbf{Q}$ dans $\mathbf{C}$ prolongeant la
valuation de $W_0(\overline{\mathbf{F}}_p)$, les valuations des $g$
valeurs propres de cet endomorphisme sont strictement positives.
\end{enumerate}
\end{proposition}

La condition (I) est \'evidente, et la condition (II) r\'esulte de ce
que l'action de $F$ sur l'alg\'ebre de Lie de $\tilde{A}$ est congrue
\`a z\'ero mod $p$. L'unicit\'e de la structure complexe v\'erifiant
(I) et (II) r\'esulte ais\'ement de la condition (b) v\'erifi\'ee
par $(T, F)$.
