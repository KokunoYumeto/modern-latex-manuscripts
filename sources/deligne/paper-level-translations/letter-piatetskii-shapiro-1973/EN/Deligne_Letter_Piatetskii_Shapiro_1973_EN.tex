\begin{center}
{\large\bfseries Deligne's Letter to Piatetski-Shapiro}
\end{center}

\begin{flushright}
Princeton, March 25, 1973
\end{flushright}

Dear Piatetski-Shapiro,

Except at \(2\), I have now a good understanding at the bad primes of the \(\ell\)-adic representations attached to modular forms (for \(\GL(2,\Q)\)).

\begin{center}
\textbf{(A). Isogenies}
\end{center}

\textbf{Definition.} The category of elliptic curves up to isogeny is obtained from that of elliptic curves by inverting isogenies. That is:
\begin{enumerate}[label=(\alph*)]
\item an elliptic curve \(E\) defines an elliptic curve up to isogeny \(E\otimes\Q\);
\item
\[
\Hom(E\otimes\Q,F\otimes\Q)=\Hom(E,F)\otimes\Q .
\]
\end{enumerate}
Hence, if \(\mathcal F\) is a functor from elliptic curves to some category, and if \(\mathcal F\) sends every isogeny to an isomorphism, then \(\mathcal F\) makes sense for elliptic curves up to isogeny.

\textbf{Notations.}
\[
T_{\ell}(E)=\Prolim_n E_{\ell^n}.
\]
For \(\ell\) prime to \(p\), and \(E/k\) over an algebraically closed field of characteristic \(p\), this is a free \(\Zl\)-module of rank \(2\). Put
\[
V_{\ell}(E)=T_{\ell}(E)\otimes_{\Zl}\Ql ,
\]
which makes sense for elliptic curves up to isogeny. Also set
\[
\wh{T}_{p'}(E)=\Prolim_{(p,n)=1}E_n,\qquad
\wh{V}_{p'}(E)=\wh{T}_{p'}(E)\otimes_{\Z}\Q .
\]
This is a module of rank \(2\) over
\[
\Afp=\prod_{\ell\neq p,\infty}'\Ql ,
\]
and it too makes sense for elliptic curves up to isogeny. Finally
\[
\Zl(1)=\Prolim_n\mu_{\ell^n},\qquad
\Ql(1)=\Zl(1)\otimes\Ql,\qquad
M(1)=M\otimes\Ql(1)\ \text{or}\ M\otimes\Zl(1).
\]

The operator \(d\) is defined by
\[
d(E)=\Z,\qquad
f:E\to F\Longrightarrow d(f):d(E)\to d(F)\ \text{is multiplication by}\ \deg(f)
\]
with \(d(f)=0\) if \(f=0\).

\textbf{Theorem.}
\begin{enumerate}[label=(\alph*)]
\item \(d(E)\otimes\Q\) makes sense for elliptic curves up to isogeny. Notation:
\[
d(E\otimes\Q)=d(E)\otimes\Q .
\]
\item For \(E_0\) an elliptic curve up to isogeny, the \(e_n\)-pairings and their behavior under isogeny enable one to define
\[
\bigwedge\nolimits^2 V_{\ell}(E_0)=d(E_0)\otimes\Ql(1).
\]
\end{enumerate}

At \(p\), let us look at the case of supersingular curves in characteristic \(p\). There a substitute for \(T_p(E)\) is the formal group of \(E\), a height \(2\), dimension \(1\), formal group. The formalism of \(d\) has the following analogue:
\begin{enumerate}[label=(\alph*)]
\item A height \(2\), dimension \(1\), formal group defines \(d_p(F)\), a rank-one module over \(\Zp\):
\[
d_p(F)=\{u:F\to F^*\}\quad(F^*=\text{Pontrjagin dual}),\qquad {}^t u=-u .
\]
\item If \(\wt F/S\) is a deformation of \(F\) over \(S\), with \(S\) local complete, and if \(T_p(\wt F)\) is the corresponding local system on \(S\otimes_{\Zp}\Qp\), then
\[
\bigwedge\nolimits^2 T_p(\wt F)\simeq d_p(F)(1).
\]
\item For \(E\) a supersingular elliptic curve, with corresponding formal group \(\wh E\),
\[
d(E)\otimes\Zp\simeq d_p(\wh E).
\]
\end{enumerate}

\textbf{Recovering \(E\).} Let \(E_0\) be a supersingular elliptic curve up to isogeny. An elliptic curve \(E\) with an isomorphism
\[
\beta:E\otimes\Q\xrightarrow{\sim}E_0
\]
defines
\begin{enumerate}[label=(\alph*)]
\item a ``lattice'' \(\wh T_{p'}(E)\) in \(\wh V_{p'}(E_0)\);
\item a ``lattice'' \(d(E)\) in \(d(E_0)\).
\end{enumerate}
The \(p\)-aspect of (b) is determined by (a):
\[
d(E)\otimes\Zl(1)\simeq \bigwedge\nolimits^2T_{\ell}(E)
\subset \bigwedge\nolimits^2V_{\ell}(E)=d(E_0)\otimes\Ql(1).
\]

\textbf{Lemma.} It amounts to the same thing to give either
\begin{enumerate}[label=(\alph*)]
\item \((E,\beta)\);
\item the lattices
\[
\wh T_{p'}(E)\subset\wh V_{p'}(E_0),\qquad
d(E)\otimes\Zp\subset d(E_0)\otimes\Qp .
\]
\end{enumerate}
The reason why no more information is required at \(p\) is that, for supersingular \(E\), the only degree \(p^k\) isogeny of source \(E\) is \(E\to E^{(p^k)}\).

\textbf{Variant.} Let \(F_0\) be a height \(2\), dimension \(1\), formal group law up to isogeny. Then to give \(F\) defining \(F_0\) amounts to giving
\[
d_p(F)\subset d_p(F_0).
\]

\begin{center}
\textbf{(B). The Fundamental Local Construction}
\end{center}

Let:
\[
\begin{array}{ll}
\overline{\Qp} & \text{be an algebraic closure of }\Qp,\\
\overline{\Fp} & \text{be the algebraic closure of }\Fp,\text{ the residue field of }\overline{\Qp},\\
k & \text{be an algebraic closure of }\Fp,\text{ provided with a class modulo}\\
  & \text{integral powers of Frobenius of allowed isomorphisms }k\simeq\overline{\Fp};\\
  & \text{the class is denoted }\Hom(k,\overline{\Fp}),\\
V_0 & \text{be a two-dimensional vector space over }\Qp,\\
E_0 & \text{be a supersingular elliptic curve up to isogeny over }k,\\
F_0 & \text{be the formal group law up to isogeny over }k\text{ defined by }E_0 .
\end{array}
\]
To be able to use transport of structure, I prefer not to take \(k=\overline{\Fp}\) nor \(V_0=\Qp^2\).

\textbf{1. First construction.} Let
\[
\sigma\in\Hom(k,\overline{\Fp}),\qquad
\beta\in\Isom\!\left(d(E_0)\otimes\Qp(1),\bigwedge\nolimits^2V_0\right).
\]
Here \(\Qp(1)\) is relative to \(\Qp\):
\[
\Qp(1)=\Qp\otimes_{\Zp}\Prolim_n
\{\text{group of }p^n\text{-roots of unity in }\overline{\Qp}\}.
\]
Then \(\sigma\) and \(\beta\) define
\[
E_0(\sigma,\beta)=(\sigma(E_0),\sigma(\beta)),
\]
where \(\sigma(E_0)\) is an elliptic curve up to isogeny over \(\overline{\Fp}\), and \(\sigma(\beta)\) is an isomorphism of one-dimensional \(\Qp\)-vector spaces
\[
d(\sigma(E_0))\otimes\Qp(1)\xrightarrow[\sim]{\sigma}
d(E_0)\otimes\Qp(1)\xrightarrow[\sim]{\beta}\bigwedge\nolimits^2V_0 .
\]

Let \(\phi:k\to k\) be Frobenius, \(x\mapsto x^p\). Then, for any elliptic curve \(E/k\), \(\phi(E)\) is \(E^{(p)}\), and one has the Frobenius isogeny \(F:E\to E^{(p)}\). The diagram

\[

  \node (mr) at (4.2,1.05) {$\mathbf Z$};
  \node (bl) at (-1.25,-0.15) {$d(E)$};
  \node (br) at (2.9,-0.15) {$d(E^{(p)})$};
  \draw[<->] (tl) -- node[above] {$\phi$} (tr);
  \draw[double distance=1.5pt] (tl) -- (ml);
  \draw[double distance=1.5pt] (tr) -- (mr);
  \draw[double distance=1.5pt] (ml) -- node[below] {$p$} (mr);
  \draw[double distance=1.5pt] (bl) -- (tl);
  \draw[->] (tr) -- (br);
  \draw[->] (bl) -- node[below] {$F$} (br);
\end{tikzpicture}
\]

is commutative. In particular, \(F\) is an isomorphism \(F:E_0\xrightarrow{\sim}\phi(E_0)\), and the diagram

\[
\begin{tikzpicture}[x=1.05cm,y=0.85cm,>=Latex,baseline=(current bounding box.center)]
  \node (a) at (0,1.4) {$d(E_0)$};
  \node (b) at (3.7,1.4) {$d(\phi(E_0))$};
  \node (c) at (0,0) {$d(E)$};
  \node (d) at (3.7,0) {$d(\phi(F_0))$};
  \draw[->] (a) -- node[above] {$\phi$} (b);
  \draw[double distance=1.5pt] (a) -- (c);
  \draw[->] (b) -- node[right] {$p$} (d);
  \draw[->] (c) -- node[below] {$F$} (d);
\end{tikzpicture}
\]

is commutative. \(F\) hence induces an isomorphism
\[
\tag{1}
F\ \text{or}\ \sigma F:(\sigma(E_0),\sigma(\beta))
\xrightarrow{\sim}
(\sigma\phi(E_0),\sigma(p^{-1}\beta)).
\]

\textbf{Definition.} \(D_p\) is the quotient of
\[
\Isom(k,\overline{\Fp})\times
\Isom\!\left(d(E_0)\otimes\Qp(1),\bigwedge\nolimits^2V_0\right)
\]
by the equivalence relation
\[
(\sigma,\beta)\sim(\sigma\phi^k,p^{-k}\beta),\qquad k\in\Z .
\]
It is a one-dimensional vector space over \(\Qp\).

Equation (1) defines an isomorphism between \(E_0(\sigma,\beta)\) and \(E_0(\sigma',\beta')\) for \((\sigma,\beta)\sim(\sigma',\beta')\), and those isomorphisms form a transitive system. They therefore allow us to define \((E_0(\delta),\beta(\delta))\) for \(\delta\in D_p\), where
\begin{enumerate}[label=(\alph*)]
\item \(E_0(\delta)\) is a supersingular elliptic curve up to isogeny over \(\overline{\Fp}\);
\item \(\beta(\delta)\) is an isomorphism
\[
d(E_0(\delta))\otimes\Qp(1)\xrightarrow{\sim}\bigwedge\nolimits^2V_0 .
\]
\end{enumerate}

\textbf{Remark.} The following groups act on \(D_p\), by transport of structure.
\begin{enumerate}[label=(\alph*)]
\item \(W(\overline{\Qp}/\Qp)\), via its actions both on \(\Isom(k,\overline{\Fp})\) and on \(\Qp(1)\). If the isomorphism
\[
W(\overline{\Qp}/\Qp)^{\mathrm{ab}}\simeq\Qp^\times
\]
is normalized, up to sign, so that Frobenius \(\phi\) corresponds to the inverse of a uniformizing parameter, then
\[
\sigma\delta=d(\sigma)^{-1}\delta .
\]
\item \(\GL(V_0)\), via its action on \(\bigwedge^2V_0\). One has
\[
g\cdot\delta=\det(g)\delta .
\]
\item \(\Aut(E_0)\), via its action on \(d(E_0)\). Put
\[
H=\Aut(E_0).
\]
It is the multiplicative group of a quaternion algebra ramified at \(p\) and at \(\infty\). One has
\[
g\cdot\delta=\Nrd(g)^{-1}\delta ,
\]
where \(\Nrd\) is the reduced norm.
\end{enumerate}

\textbf{Remark.} Assume that a lattice \(\Lambda\simeq\Zp\) has been chosen in \(\bigwedge^2V_0\). Then \((E_0,\beta)/\overline{\Fp}\) as above defines an elliptic curve up to \(p'\)-isogeny over \(\overline{\Fp}\), corresponding to the lattice
\[
\beta^{-1}(\Lambda)(-1)\subset d(E_0)\otimes\Qp .
\]
Here \(p'\)-isogeny means an isogeny of degree prime to \(p\), cf. Lemma A.

\textbf{Variant.} Start with \(F_0\) instead of \(E_0\). \(D_p\) is defined as before, using \(d_p(F_0)\) instead of \(d(E_0)\otimes\Qp\), and is the same as before via the isomorphism
\[
d(E_0)\otimes\Qp\simeq d_p(F_0),\qquad F_0=\wh E_0 .
\]
This time \(D_p\) is acted on by \(W(\overline{\Qp}/\Qp)\), \(\GL(V_0)\), and \(\Aut(F_0)=H(\Qp)\). The formulas are the same as before.

An element \(\delta\in D_p\) defines
\begin{enumerate}[label=(\alph*)]
\item \(F_0(\delta)\), a supersingular formal group law up to isogeny over \(\overline{\Fp}\);
\item an isomorphism
\[
\beta(\delta):d(F_0(\delta))\otimes\Qp(1)\xrightarrow{\sim}\bigwedge\nolimits^2V_0 .
\]
\end{enumerate}
If \(\Lambda\) has been chosen in \(\bigwedge^2V_0\), one obtains
\begin{enumerate}[label=(\alph*)]
\item \(F(\delta)\), a supersingular formal group law over \(\overline{\Fp}\);
\item an isomorphism
\[
\beta(\delta):d(F(\delta))\otimes\Zp(1)\xrightarrow{\sim}\Lambda .
\]
\end{enumerate}

\textbf{2. Second construction.} I now have to define vanishing-cycle varieties and vanishing-cycle groups. Eventually, for \(K^0\) an open compact subgroup of \(\SL(V_0)\), and for \(\delta\in D_p\), a scheme \(V(K^0,\delta)\) over \(\Qp\) will be defined. For \(K\) an open compact subgroup of \(\GL(V_0)\) such that \(K^0=K\cap\SL(V_0)\), isomorphisms
\[
V(K^0,\delta)\leftrightarrow V(K^0,\alpha\delta),
\qquad \alpha\in\det K^0\subset\Zp^\times
\]
are defined. For variable \(\delta\), the \(V(K^0,\delta)\) will thus be a local system of schemes on \(D_p\). The \(V(K^0,\delta)\) are not of finite type, but of the type usual in vanishing-cycle theory. We shall not need this, however.

Here is a description of the \(\overline{\Qp}\)-valued points of \(V(K^0,\delta)\).
\begin{enumerate}[label=(\alph*)]
\item \(\delta\) provides us with \((E_0(\delta),\beta(\delta))\) over \(\overline{\Fp}\).
\item A point of \(V(K^0,\delta)\) is an isomorphism class of systems consisting of:
\begin{enumerate}[label=(\arabic*)]
\item an elliptic curve up to isogeny over \(\overline{\Qp}\), denoted \(\mathcal E\);
\item an isomorphism \(\alpha:V_p(\mathcal E)\xrightarrow{\sim}V_0\), given modulo \(K^0\);
\item an isomorphism of the reduction of \(\mathcal E\) with \(E_0(\delta)\),
\[
\psi:\mathcal E_{\overline{\Fp}}\xrightarrow{\sim}E_0(\delta);
\]
\item the commutativity of the diagram
\[
\begin{tikzcd}
\bigwedge^2 V_p(\mathcal E) \arrow[r,"\bigwedge^2\alpha"] \arrow[d,equal] &
\bigwedge^2 V_0\\
d(\mathcal E)(1) \arrow[r,"\psi"'] &
d(E_0(\delta))(1) \arrow[u,"\beta(\delta)"']
\end{tikzcd}
\tag{1}
\]
\end{enumerate}
\end{enumerate}
If \(K\) is as above, it amounts to the same thing to give only modulo \(K\), Eq. (1) being required only modulo \(\det(K)\).

To construct \(V(K,\delta)\) actually, let us start with:
\begin{enumerate}[label=(\alph*)]
\item \(K\), an open compact subgroup of \(\GL(V_0)\);
\item \(E_0\), a supersingular curve up to isogeny over \(\overline{\Fp}\), in practice \(E_0(\delta)\);
\item an isomorphism
\[
\beta:d(E_0)\otimes\Qp(1)\xrightarrow{\sim}\bigwedge\nolimits^2V_0,
\]
in practice \(\beta(\delta)\);
\item \(\overline{\beta}=\beta\ \bmod\ \det(K)\subset\Zp^\times\). Only \(\overline{\beta}\), not \(\beta\), will be used.
\end{enumerate}
The construction will involve auxiliary data: a \(K\)-stable lattice \(L\) in \(V_0\).
\begin{enumerate}[label=(\alph*)]
\item \(\beta^{-1}(\bigwedge^2L)(-1)\) is a lattice in \(d(E_0)\otimes\Qp\) and defines an elliptic curve up to \(p'\)-isogeny \(E'\), with \(E'\otimes\Q=E_0\). Choose an elliptic curve \(E\) with level \(n\) structure \(\alpha_n\), \(n\geq3\), \((n,p)=1\), with \(E\otimes\Z_{(p)}=E'\). The construction will be shown to be independent of \(L,E',E,\alpha_n\). Let \(M_n\) be the modular scheme for elliptic curves with level \(n\) structure, and let \(e\in M_n(\overline{\Fp})\) be the point defined by \((E,\alpha_n)\).
\item \(M(E,\alpha_n)\) is the spectrum of the henselization of the local ring at \(e\) of \(M_n\otimes_{\Z}W(\overline{\Fp})\).
\item The completion of \(M(E,\alpha)\) is isomorphic to \(W(\overline{\Fp})[[t]]\).
\item Assume that \((E_1,\alpha_n^1)\) and \((E_2,\alpha_n^2)\) are two systems as above, and that \(\phi:E_1\to E_2\) is a \(p'\)-isogeny. There is then one and only one isomorphism
\[
\overline{\phi}:M(E_1,\alpha_n^1)\longrightarrow M(E_2,\alpha_n^2)
\]
fitting in a commutative diagram. In the following diagram, \(\star\) means \(\Spec(\overline{\Fp})\), and \(\wt E_1,\wt E_2\) are the pullbacks over \(M(E_1,\alpha_n^1)\), \(M(E_2,\alpha_n^2)\) of the universal curves over \(M_n\).

\end{enumerate}

This can be better expressed by saying that \(M(E_1,\alpha_n^1)\) is the parameter space of the universal deformation of the elliptic curve up to \(p'\)-isogeny \(E'/\overline{\Fp}\), universal with respect to deformations over henselian local \(W(\overline{\Fp})\)-algebras. This allows us to write simply \(M(E')\) for \(M(E_1,\alpha_n^1)\), and \(\wt E'\) for the elliptic curve up to \(p'\)-isogeny over \(M(E')\) defined by any \(E_1\).

For \(n\) large enough, \(K\subset\GL(L)\) is the inverse image in \(\GL(L)\) of a suitable subgroup \(\overline K\subset\GL(L/p^nL)\). Let \(K(\overline{\Fp})\) be the field of fractions of \(W(\overline{\Fp})\), and let \(\Zp\) be the ring of integers in \(\Qp\). The group scheme \(E'_{p^n}\) over \(M(E')\) is finite étale over \(M(E')\otimes K(\overline{\Fp})\). Hence
\[
\Isom(E'_{p^n},L/p^nL)
\]
is a finite étale Galois covering of \(M(E')\otimes K(\overline{\Fp})\), with Galois group \(\GL(L/p^nL)\). Rather consider
\[
\Isom_{M(E')\otimes\Qp}(E'_{p^n},L/p^nL).
\]
This time one gets a disconnected covering of \(M(E')\otimes\Qp\). A piece of it is picked as follows: via \(\beta\), one has
\[
\bigwedge\nolimits^2E'_{p^n}=d(E')_p\otimes\Z/p^n(1)=\bigwedge\nolimits^2L/p^nL,
\]
and one considers only isomorphisms of determinant \(1\). Similarly, \(\beta\) enables one to pick a component of
\[
K\backslash\Isom_{M(E')\otimes\Qp}(E'_{p^n},L/p^nL).
\]
We call this component \(V_L(K,E_0,\beta)\). It is also the quotient by \(\overline K\cap\SL(L/p^nL)\) of the chosen component of
\[
\Isom_{M(E')}(E'_{p^n},L/p^nL)\otimes_{W(\overline{\Fp})}\Qp .
\]

\textbf{Summary.} \(V_L(K,E_0,\beta)\) depends on \(K\subset\GL(V_0)\), compact open; \(E_0\), up to isogeny over \(\overline{\Fp}\); \(\beta:d(E')\otimes\Qp(1)\xrightarrow{\sim}\bigwedge^2V_0\); and a lattice \(L\) in \(V_0\), stable under \(K\). It does not depend on the whole of \(\beta\), but only on \(\overline{\beta}=\beta\bmod\det(K)\). For \(K'\subset K\), one has a map
\[
V_L(K',E',\beta)\longrightarrow V_L(K,E',\beta).
\]

The construction of \(V_L(K,E_0,\beta)\) is independent of \(L\). More precisely, the system consisting of
\begin{enumerate}[label=(\alph*)]
\item \(V_L(K,E',\beta)\);
\item the elliptic curve up to isogeny \(\wt E_0\) over \(V_L(K,E',\beta)\);
\item the universal isomorphism
\[
\alpha:V_p(\wt E'_0)\xrightarrow{\sim}V_0
\]
given modulo \(K\), deduced from \(\alpha:\wt E'_{p^n}\xrightarrow{\sim}L/p^nL\) modulo \(K\), or from \(\alpha:T_p(\wt E')\xrightarrow{\sim}L\) modulo \(K\),
\end{enumerate}
is independent of \(L\).

Let \(L_1\) and \(L_2\) be two \(K\)-invariant lattices, and let \(E_1\) and \(E_2\) be the corresponding elliptic curves up to \(p'\)-isogeny over \(\overline{\Fp}\). We are to define an isomorphism

\[

  \node (B) at (7.2,1.2) {$K\backslash\underline{\operatorname{Isom}}\!\left(\begin{matrix}V_p(\widetilde E_2)\\[-2pt]\cup\\[-2pt]T_p(\widetilde E_2)\end{matrix},\begin{matrix}V_0\\[-2pt]\cup\\[-2pt]L_2\end{matrix}\right)$};
  \node (M1) at (0,-0.45) {$M(E_1)$};
  \node (M2) at (7.2,-0.45) {$M(E_2)$};
  \draw[->] (A) -- node[above] {$\sim$} (B);
  \draw[->] (A) -- (M1);
  \draw[->] (B) -- (M2);
\end{tikzpicture}
\]

For simplicity, assume \(L_1\subset L_2\). Then, over the first ``Isom'', \(\wt E_1\) is provided with a subgroup isomorphic to \(L_2/L_1\), call it \(H\). Let \(W^1(K,E_0,\beta)\) be the normalization of \(M(E_1)\) in the first ``Isom''. Due to the normality of \(W\) and to the fact that elliptic curves have only finitely many subgroup schemes of a given order, the subgroup scheme \(H\) of \(\wt E_1\) on \(W^1(K,E_0,\beta)\otimes K(\overline{\Fp})\) extends as a subgroup scheme \(H\) of \(\wt E_1\) on \(W^1(K,E_0,\beta)\). The quotient \(\wt E_1/H\) is a deformation of \(E_2\), hence there is a map
\[
W^1(K,E_0,\beta)\longrightarrow M(E_2).
\]
Further, \(V_p(\wt E_1/H)=V_p(\wt E_1)\), and, over
\[
K\backslash\Isom\!\left(
\begin{matrix}V_p(\wt E_1)\\[-2pt]\cup\\[-2pt]T_p(\wt E_1)\end{matrix},
\begin{matrix}V_0\\[-2pt]\cup\\[-2pt]L_1\end{matrix}\right),
\]
there is an isomorphism
\[
\alpha:V_p(\wt E_1)\to V_0
\]
carrying \(T_p(\wt E_1)\) to \(L_1\), and \(T_p(\wt E_1/H)\) to \(L_2\). If \(W_2\) is defined similarly to \(W_1\), one has

\[

This defines the dotted maps which are needed for the isomorphism.
By extending scalars to \(\Qp\) and taking one component, one gets the isomorphisms expressing that \(V_L(K,E_0,\beta)\) is independent of \(L\). For \(\delta\in D_p\), set
\[
V(K,\delta)=V_L(K,E_0(\delta),\beta(\delta)).
\]

\textbf{Summary.} \(V(K,\delta)\) is a scheme over \(\Qp\). It depends on the compact open \(K\) in \(\GL(V_0)\), and on \(\delta\in D_p\), given modulo multiplication by elements of \(\det(K)\subset\Zp^\times\). For smaller and smaller \(K\), the \(V(K,\delta)\) form a projective system. Over \(V(K,\delta)\) is given an elliptic curve up to isogeny \(\wt E\), provided with an isomorphism, given modulo \(K\),
\[
\alpha:V_p(\wt E)\xrightarrow{\sim}V_0.
\]
In a sense, \(\wt E\) is a deformation of \(E_0(\delta)\); in particular \(d(E_0(\delta))=d(\wt E)\). The isomorphism \(\alpha\) is compatible with \(\beta(\delta)\):
\[
\bigwedge\nolimits^2V_p(\wt E)
=d(E_0(\delta))\otimes\Qp(1)
\xrightarrow{\det\alpha=\beta(\delta)}
\bigwedge\nolimits^2V_0
\quad\bmod\ \det K.
\]
In fact, the statement that \(\wt E\) is a deformation of \(E_0\) can be made more precise by introducing a suitable \(\wt E/V(K,\delta)/\Zp\).

\textbf{Variant.} Starting with \(F_0\) instead of \(E_0\), one can construct analogues of the \(V(K,\delta)\), called \(\wh V(K,\delta)\), with complete local rings replacing henselian local rings.

\textbf{3. Third construction.} We are interested in the \(\ell\)-adic cohomology groups
\[
H^1(V(K,\delta),\Ql).
\]
These groups are finite-dimensional and locally constant as a function of \(\delta\), as \(V(K,\delta)\) itself is. If \(K'\) is a distinguished subgroup of \(K\), one clearly has
\[
H^1(V(K,\delta),\Ql)=
H^1(V(K',\delta),\Ql)^{K\cap\SL(V_0)/(K'\cap\SL(V_0))}
\]
(invariants). Note that \(V(K,\delta)\) also depends only on \(K\cap\SL(V_0)\) and \(\delta\).

The local fundamental object is the bundle \(\kappa/D_p\), with
\[
\kappa_{\delta}=\Lim_K H^1(V(K,\delta),\Ql).
\]
There is a notion of locally constant section of \(\kappa_\delta/D_p\). It is a function \(\varphi(\delta)\), where \(\delta\in D_p\) and \(\varphi(\delta)\in\kappa_\delta\), locally lying in some \(H^1(V(K,\delta),\Ql)\) and locally constant. The space of locally constant sections is denoted
\[
\Gamma(D_p,\kappa).
\]

By transport of structure, the local fundamental object admits actions of
\[
W(\overline{\Qp}/\Qp),\qquad \GL(V_0),\qquad \Aut(E_0)=H,
\]
the actions over \(D_p\) being those already described. The actions of \(W(\overline{\Qp}/\Qp)\) and \(\GL(V_0)\) are continuous, the latter with respect to the notion of locally constant section.

\textbf{Proposition.} The action of \(\Aut(E_0)\) extends as a continuous action of \(H(\Qp)\) on \(\kappa/D_p\).

I don’t have a satisfactory proof. My idea of proof would be to go back to the \(W\) introduced earlier and express \(\kappa\) in terms of the special fiber of stable models of \(W\) over suitable ramified extensions of \(W(\overline{\Fp})\).

Intuitively, one may argue that \(V(K,\delta)\) and \(\wh V(K,\delta)\) could have the same cohomology, that \(\Aut(F_0)=H(\Qp)\) acts on \(\wh V(K,\delta)\), hence on \(H^1(\wh V(K,\delta),\Ql)\), and that it would be the hell if the action were not continuous [sic].

\begin{center}
\textbf{(C). Statement of the Local Results}
\end{center}

The proofs will be of a global nature. It will be easier to work not with \(\Ql\)-cohomology, but with \(\overline{\Ql}\)-cohomology, obtained by extending \(\Ql\) to an algebraic closure \(\overline{\Ql}\). By abuse of language, we will again denote by \(\kappa/D_p\) the admissible bundle over \(D_p\), with fiber
\[
\kappa_{\delta}=\Lim_K H^1(V(K,\delta),\Ql)\otimes_{\Ql}\overline{\Ql}.
\]

The groups \(W(\overline{\Qp}/\Qp)\), \(\GL(V_0)\), and \(H(\Qp)\) act admissibly on \(\kappa/D_p\). The actions on \(D_p\) have been computed. Of course, the three actions commute with one another. If \(a\in\Qp^\times\), the actions of the elements \(a\in\GL(V_0)\) and \(a\in H(\Qp)\) are inverse to one another. In terms of a fixed \(\delta_0\in D_p\), this could as well be expressed by saying:
\begin{enumerate}[label=(\alph*)]
\item the subgroup of
\[
W(\overline{\Qp}/\Qp)\times\GL(V_0)\times H(\Qp)
\]
formed by the elements such that
\[
d(\sigma)^{-1}\det(g)\Nrd(h)=1
\]
acts on \(\kappa_{\delta_0}\);
\item the subgroup \((1,a,a)\) acts trivially.
\end{enumerate}

Mnemotechnik way to check the action on \(V(K,\delta)/D_p\) is to view a point of \(\Prolim V(K,\delta)/D_p\) as consisting of
\begin{enumerate}[label=(\alph*)]
\item \(\delta\in D_p\);
\item \(\wt E\), an elliptic curve up to isogeny over \(\overline{\Qp}\);
\item \(\alpha:V_p(\wt E)\xrightarrow{\sim}V_0\);
\item \(\psi\), the specialization map \(\wt E\to\delta(E_0)\),
\end{enumerate}
subject to a compatibility between \(\alpha\), \(\psi\), and \(\beta(\delta)\).

Let \(\chi:\Qp^\times\to\overline{\Ql}^{\times}\) be a quasi-character with open kernel. We denote by \(\Gamma(D_p,\kappa)\) the \(\overline{\Ql}\)-vector space of locally constant sections of \(\kappa/D_p\), and by \(\Gamma_{\chi}(D_p,\kappa)\) the subspace of those sections such that, for any \(a\in\Qp^\times\), with image \(z(a)\) in the center of \(\GL(V_0)\),
\[
z(a)f=\chi(a)f,\qquad\text{i.e.}\qquad
z(a)f(\delta)=\chi(a)f(a^2\delta).
\]

\textbf{Theorem.}
\begin{enumerate}[label=(\roman*)]
\item \(\Gamma_{\chi}(D_p,\kappa)\) is a direct sum of triple tensor products
\[
\Gamma_{\chi}(D_p,\kappa)=
\bigoplus_{\phi\in\Phi}V_{\phi}\otimes V'_{\phi}\otimes W_{\phi},
\]
where:
\begin{enumerate}[label=(\alph*)]
\item \(V_{\phi}\) is an admissible irreducible representation of \(\GL(V_0)\), with central character \(\chi\), belonging to the discrete series, i.e. special or supercuspidal;
\item \(V'_{\phi}\) is an admissible irreducible, finite-dimensional, representation of \(H(\Qp)\), on which the center acts through \(\chi^{-1}\);
\item \(W_{\phi}\) is a two-dimensional, or one-dimensional, continuous \(\overline{\Ql}\)-adic representation of \(W(\overline{\Qp}/\Qp)\). If it is two-dimensional, then \(W(\overline{\Qp}/\Qp)\) acts on \(\bigwedge^2W_{\phi}\) by the character \(\chi\). If \(W_{\phi}\) is one-dimensional, then the action of \(W(\overline{\Qp}/\Qp)\) on \(\bigwedge^2(W_{\phi}\oplus W_{\phi}(1))\) again coincides with \(\chi\).
\end{enumerate}
\item \(V'_{\phi}\) runs once and only once through the representations in (i).
\item The same holds for \(V_{\phi}\), and \(V_{\phi}\) and \(V'_{\phi}\) correspond by the Weil representation construction, suitably normalized.
\item \(W_{\phi}\) is one-dimensional if and only if \(V'_{\phi}\) is also, if and only if \(V_{\phi}\) is special. If \(V'_{\phi}\) is \(\mu\circ\Nrd\), then \(W_{\phi}\) is the character of \(W(\overline{\Qp}/\Qp)\) defined by \(\mu\circ\cl\), where
\[
\cl:W(\overline{\Qp}/\Qp)\to\Qp^\times
\]
is the map of class field theory.
\item If \(V_{\phi}\) is defined by a quasi-character of a quadratic extension of \(\Qp\), with values in \(\overline{\Ql}^{\times}\), then, with a suitable normalization, \(W_{\phi}\) is induced by that same character.
\end{enumerate}

Except for \(p=2\), this gives a complete description of \(\Gamma_{\chi}(D_p,\chi)\).

For \(\mu\) a quasi-character of \(\Qp^\times\) and \(\delta_0\in D_p\), multiplication by the function \(\mu(\delta\delta_0^{-1})\) on \(D_p\) provides an isomorphism
\[
\Gamma_{\chi}(D_p,\kappa)\longrightarrow
\Gamma_{\chi\mu^{-2}}(D_p,\kappa).
\]
If \(V_{\phi}\otimes V'_{\phi}\otimes W_{\phi}\) occurs in \(\Gamma_{\chi}(D_p,\kappa)\), then
\[
\bigl(V_{\phi}\otimes\mu^{-1}\circ\det\bigr)\otimes
\bigl(V'_{\phi}\otimes\mu\circ\Nrd\bigr)\otimes
(W_{\phi}\otimes\mu\circ\cl)
\]
occurs in \(\Gamma_{\chi\mu^{-2}}(D_p,\kappa)\).

\begin{center}
\textbf{(D). Global Theory}
\end{center}

We consider:
\[
\begin{array}{ll}
K & \text{an open compact subgroup of }\GL(2,\Af),\\
X^{\pm} & \text{the Poincare upper and lower half-plane, i.e.}\\
& X^{\pm}=\Isom_{\mathbb R}(\Z^2\otimes\mathbb R,\C)/\C^{\times}
\subset\Hom(\Z^2,\C)/\C^{\times},\\
M_K^{\circ}(\C) & =K\backslash X^{\pm}\times\GL(2,\Af)/\GL(2,\Q),\\
k & \text{an integer } \geq0,\\
\mu & \text{the representation }\Sym^k\text{ of the dual of the obvious representation of }\GL(2,\Q),\\
\mathcal F_{\mu}^{\Q} & \text{the corresponding local system on }M_K^{\circ}(\C),\\
M_K(\C) & \text{the Satake compactification of }M_K^{\circ}(\C),\quad
j:M_K^{\circ}(\C)\hookrightarrow M_K(\C),\\
\mathcal F_{\mu}^{\Q} & =j_*\mathcal F_{\mu}^{\Q}\text{ on }M_K(\C),\\
\kappa(\mu) & =\Lim_K H^1(M_K(\C),\mathcal F_{\mu}^{\Q}),
\end{array}
\]
an admissible representation of \(\GL(2,\Af)\) defined over \(\Q\). Put
\[
\kappa_{\C}(\mu)=\kappa(\mu)\otimes\C,\qquad
\kappa_{\Ql}(\mu)=\kappa(\mu)\otimes\Ql,\qquad
\mathcal F_{\mu}^{\Ql}=\mathcal F_{\mu}\otimes\Ql .
\]

One has a decomposition
\[
\kappa_{\C}(\mu)=\kappa^{k+1,0}\oplus\kappa^{0,k+1}
\]
of \(\kappa_{\C}\) into two complex conjugate subspaces. The space \(\kappa^{k+1,0}\) is a complex admissible representation of \(\GL(2,\Af)\), defined over \(\Q\), and hence isomorphic to the complex conjugate representation \(\kappa^{0,k+1}\). It corresponds to holomorphic modular cusp forms of weight \(k+2\), with \(k+2\geq2\). For some explicit admissible representation \(D_{k+2}\) of \(\GL(2,\mathbb R)\), in the discrete series,
\[
\kappa^{k+1,0}=
\Hom_{\GL(2,\mathbb R)}
\bigl(D_{k+2},L^0(\GL(2,\mathbb A)/\GL(2,\Q))\bigr).
\]
Further, the \(M_K(\C)\) are naturally defined over \(\Q\), with their \(\GL(2,\Af)\)-action under which \(g\) maps \(M_K\) onto \(M_{gKg^{-1}}\), and \(\mathcal F_{\mu}^{\Ql}\) is an \(\ell\)-adic sheaf defined over \(\Q\). Hence \(\kappa_{\Ql}(\mu)\) carries a \(\Gal(\overline{\Q}/\Q)\)-action, commuting with \(\GL(2,\Af)\), acting by transport of structure. After extending \(\Ql\) to \(\overline{\Ql}\), one has a decomposition
\[
\kappa_{\overline{\Ql}}(\mu)=
\bigoplus_{f\in F}\left(\bigotimes_p V_{f,p}\right)\otimes W_f,
\]
where the \(V_{f,p}\) are irreducible representations of \(\GL(2,\Qp)\), and \(W_f\) is a two-dimensional representation of \(\Gal(\overline{\Q}/\Q)\).

By Eichler--Shimura--Kuga--Deligne--Ihara--Piatetski-Shapiro--Langlands, if \(V_{f,p}\) is of the principal series, then \(W_f|_{\Gal(\overline{\Qp}/\Qp)}\) is the sum of the two corresponding characters. If \(V_{f,p}\) is special, then \(W_f|_{\Gal(\overline{\Qp}/\Qp)}\) is the corresponding special \(\ell\)-adic representation of \(\Gal(\overline{\Qp}/\Qp)\).

I can now prove:

\textbf{(A).} If \(V_{f,p}\) is supercuspidal, then \(W_f|_{\Gal(\overline{\Qp}/\Qp)}\) is irreducible, and, with the notation of Theorem C, if \(V_{f,p}\sim V_{\phi}\), then \(W_f\sim W_{\phi}\).

Hence \(V_{f,p}\) determines, by a local rule,
\[
W_{f,p}=W_f|_{\Gal(\overline{\Qp}/\Qp)},
\]
and the rule is the obvious one, up to a normalization, if \(V_{f,p}\) corresponds by the Weil construction to a character of a quadratic extension.

The method is to use the theory of vanishing cycles to prove that a space to be described below is a quotient of \(\kappa(\mu)\otimes\Ql\). This is accurate only for \(k\neq0\); I shall not bother much about \(k=0\) and the phenomena related to the special representation. We keep the notation of \S B, except that now \(V_0=\Qp^2\). Consider
\[
\Isom\!\left(V_{p'}(E_0),(\Afp)^2\right)\times D_p
\]
and the right action of \(H\) on it, by composition on the first factor and by the inverse of the already defined action on \(D_p\). On this space we have the following \(H\)-equivariant local system:
\[
\Sym^k(V_{\ell}(E)^*)\otimes \Lim_K H^1(V(K,\delta),\Ql).
\]
On the second factor, a right action is required; one takes the inverse of the one already constructed. The space and local system are acted upon by \(\GL(2,\Af)\): on the space, \(\GL(2,\Af)\) acts by composition on the first factor and acts on \(D_p\) in the manner already described; on the local system, \(\GL(2,\Af)\) acts trivially on the first factor and acts on the second in the manner already described.

Let \(H^0\) be the space of sections of this local system; this is a representation of \(\GL(2,\Af)\). There is also an action of \(W(\overline{\Qp}/\Qp)\) via its action on \(V(K,\delta)/D_p\).

I will now relate this \(H^0\) to the spectrum of \(H(\mathbb A)/H(\Q)\) and \(\Gamma(D_p,\kappa)\). For simplicity, extend scalars from \(\Ql\) to \(\C\). This does not lose the Galois action; to see this, I have to use that the Galois action is continuous for the discrete topology of \(\overline{\Ql}\), which I can do directly. Let me also choose an isomorphism
\[
V_{p'}(E_0)\simeq(\Afp)^2,
\]
hence
\[
H(\Ql)\simeq\GL(2,\Ql),\qquad \ell\neq p.
\]
Then
\[
\begin{aligned}
H^0
&=\left\{\text{functions }H(\Afp)\to
\Sym^k(V_{\ell}(E)^*)\otimes\Gamma(D_p,\kappa)\right.\\
&\hspace{4em}\left.\text{with }f(x\gamma)=f(x)\gamma,\ \gamma\in H(\Q)\right\}\\
&=\left\{\text{functions }H(\mathbb A)\to
\Sym^k(V_{\ell}(E)^*)\otimes\Gamma(D_p,\kappa)\right.\\
&\hspace{4em}\left.\text{with }f(g_{\infty}g_px\gamma)=g_{\infty}g_pf(x),\
g_{\infty}\in H(\mathbb R),\ g_p\in H(\Qp)\right\}\\
&=\bigoplus_{\chi\in(\mathbb A^{\times}/\Q^{\times})^\wedge}
\Hom_{H(\mathbb R)\times H(\Qp)}
\left(
\Sym^k(V)\otimes\Gamma_{\chi_p^{-1}}(D_p,\kappa)^{\vee},
L_{\chi}^{0}(H(\mathbb A)/H(\Q))
\right).
\end{aligned}
\]

This can be expressed as follows. In \(L^0(H(\mathbb A)/H(\Q))\), one takes only those elements which at \(\infty\) transform as a vector of a given representation of \(H(\mathbb R)\). The representation of \(H(\Af)\) so obtained being written
\[
\bigoplus_{f_0\in F_0}\bigotimes_p V_{f_0,p},
\]
the following representation occurs in \(\kappa(\mu)\):
\[
\bigoplus_{f_0\in F}
\left(\bigotimes_{\ell\neq p}V_{f_0,\ell}\right)
\otimes
\bigoplus_{V'_{\phi}\sim V_{f_0,p}}
(V_{\phi}\otimes W_{\phi}).
\]

Now, by comparison with Jacquet--Langlands, \S 16, plus the fact that supercuspidal representations cannot occur outside \(\kappa(\mu)\), this being given by Piatetski-Shapiro or Langlands plus the vanishing-cycle theory, one gets (a), (b), (ii), (iii), and the two-dimensionality in (c) of Theorem C. One also gets statement (A). By checking this general rule against modular forms attached to \(L\)-functions with Grössencharacters of imaginary quadratic fields, and the remark in C, one gets statement (v). The proof of (c) in characteristic \(2\), in cases not covered by (v), uses entirely different ideas, which I cannot explain here.
