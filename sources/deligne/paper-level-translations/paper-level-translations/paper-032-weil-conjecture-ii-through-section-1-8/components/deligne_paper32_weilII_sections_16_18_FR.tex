\section*{(1.6) Autour de Jacobson--Morosov}

\paragraph{Proposition (1.6.1).} Soit $N$ un endomorphisme nilpotent d'un objet $V$ d'une catégorie abélienne. Il existe une unique filtration finie croissante $M$ de $V$ telle que
\[
        NM_i\subset M_{i-2},
        \qquad
        N^k:\Gr^M_kV\xrightarrow{\sim}\Gr^M_{-k}V.
\]
On prouve l'existence par récurrence sur $d$ tel que $N^{d+1}=0$; pour $d=0$ on prend la filtration triviale. Pour le pas de récurrence on pose $M_d=V$, $M_{d-1}=\Ker N^d$, $M_{-d}=\operatorname{Im}N^d$, et l'on relève la filtration construite par récurrence sur $\Ker N^d/\operatorname{Im}N^d$.

\paragraph{Remarque (1.6.2).} La caractérisation précédente montre la compatibilité au passage à la catégorie duale.

\paragraph{Définition (1.6.3).} La partie primitive $P_i(V)$ de $\Gr^M_iV$ est le noyau du morphisme induit par $N$ de $\Gr^M_iV$ dans $\Gr^M_{i-2}V$.

\paragraph{(1.6.4).} On obtient des isomorphismes
\begin{equation}\tag{1.6.4.1}
        \Gr^M_iV\simeq \bigoplus_{\substack{j\ge |i|\\ j\equiv i\ (2)}} P_{-j}.
\end{equation}
Via ces isomorphismes, $N:\Gr^M_iV\to\Gr^M_{i-2}V$ est l'inclusion évidente pour $i>1$ et la projection évidente pour $i-2<-1$.

\paragraph{Lemme (1.6.5).} Le morphisme $N:(V,M)\to(V,M\text{ décalé de }2)$ est strictement compatible aux filtrations.

\paragraph{Corollaire (1.6.6).} L'inclusion $\Ker N\subset V$ induit des isomorphismes
\[
        \Gr^M_i(\Ker N)\xrightarrow{\sim}P_i.
\]

\paragraph{(1.6.7).} Pour un espace vectoriel de dimension finie, $M$ est décrite bloc par bloc par les blocs de Jordan de $N$. Si $\dim V=d+1$ et si la base est indexée par $d,d-2,\ldots,-d$, avec $Ne_i=e_{i-2}$ et $Ne_{-d}=0$, alors $M_i$ est engendré par les $e_j$ pour $j\le i$.

\paragraph{(1.6.8).} En caractéristique zéro, $M$ s'interprète par le théorème de Jacobson--Morosov: si $u:SL_2\to GL(V)$ relève $N$, et si $V_j$ est le sous-espace de poids $j$, alors $M_i=\sum_{j\le i}V_j$.

\paragraph{Proposition (1.6.9).} Pour $V=V'\otimes V''$ et $N=N'\otimes1+1\otimes N''$, la filtration $M$ est le produit tensoriel des filtrations; pour le dual $((V')^\vee,-{}^tN')$, elle est la filtration duale. Ainsi $\Gr^M$ est compatible au produit tensoriel et au passage au dual.

\paragraph{(1.6.10)--(1.6.12).} Sur $\Gr^M(V)$ il existe une unique représentation de $SL_2$ dont le nilpotent infinitésimal est $N$ et dont le tore agit par le degré $M$. En choisissant, pour chaque $d$, une représentation irréductible $S^d$ et des isomorphismes de Clebsch--Gordan
\[
        S^d\otimes S^{d'}\simeq \bigoplus_{j\in P(d,d')}S^j,
        \qquad (S^d)^\vee\simeq S^d,
\]
on obtient
\begin{equation}\tag{1.6.11.5}
        P_{-j}\simeq \Hom_{SL_2}(S^j,\Gr^M(V)).
\end{equation}
Pour les produits tensoriels et les duaux:
\begin{align}
        P_{-j}&\simeq \bigoplus_{j\in P(j',j'')} P'_{-j'}\otimes P''_{-j''},\tag{1.6.12.1}\\
        P_{-j}((V')^\vee)&\simeq P_{-j}(V')^\vee.\tag{1.6.12.2}
\end{align}

\paragraph{Proposition (1.6.13).} Soient $V$ muni d'une filtration finie croissante $W$ et $N$ un endomorphisme nilpotent respectant $W$. Il existe au plus une filtration finie croissante $M$ telle que $NM_i\subset M_{i-2}$ et que les puissances de $N$ induisent les isomorphismes requis sur $\Gr^W$. La preuve se fait par récurrence sur la longueur de $W$; les identités qui déterminent $M$ sont celles de l'argument original: pour $i\ge0$, $M_{-i}$ est donné par sa restriction à $W_{-1}$, par $N^iM_i$, et $M_i$ est le noyau de $N^{i+1}:V\to V/M_{-i-2}$.

\paragraph{(1.6.14).} Dans les applications on travaille avec une action nilpotente d'une algèbre de Lie de dimension $1$. Le choix de $N\ne0$ permet d'appliquer ce qui précède, mais les filtrations obtenues ne dépendent pas de ce choix. En écrivant $Y(k)=Y\otimes\mathfrak n^{\otimes k}$, les isomorphismes prennent la forme
\begin{align}
        \Gr^M_{i+k}\Gr^W_iV(k)&\simeq\Gr^M_{i-k}\Gr^W_iV,\tag{1.6.14.1}\\
        \Gr^M_i(\Ker N)&\simeq P_i,\tag{1.6.14.2}\\
        \Gr^M_iV&\simeq\bigoplus_{\substack{j\ge |i|\\j\equiv i(2)}}P_{-j}\Bigl(\frac{i+j}{2}\Bigr).\tag{1.6.14.3}
\end{align}

\section*{(1.7) Monodromie locale}

\paragraph{(1.7.1).} Soient $R$ un anneau de valuation discrète hensélien, $K$ son corps des fractions, $k$ son corps résiduel, $\overline K$ une clôture algébrique de $K$ et $\overline k$ le corps résiduel normalisé. Le groupe d'inertie est le noyau
\begin{equation}\tag{1.7.1.1}
0\to I\to\Gal(\overline K/K)\to\Gal(\overline k/k)\to0.
\end{equation}
Pour $\ell$ distinct de la caractéristique résiduelle,
\[
0\to P\to I\xrightarrow{t_\ell}\widehat{\mathbb Z}^{(\ell')}(1)\to0.
\]

\paragraph{(1.7.2).} Si $\rho:I\to GL(V)$ est une représentation $\ell$-adique et si sa restriction à un sous-groupe d'indice fini est unipotente, il existe une unique représentation nilpotente telle que
\begin{equation}\tag{1.7.2.1}
        \rho(\sigma)=\exp(\overline\rho(t_\ell(\sigma))).
\end{equation}
On l'appelle le logarithme de la partie unipotente de la monodromie locale, ou encore un morphisme nilpotent $N:V(1)\to V$; la formule devient
\begin{equation}\tag{1.7.2.2}
        \rho(\sigma)=\exp(Nt_\ell(\sigma)).
\end{equation}

\paragraph{(1.7.3)--(1.7.7).} Si le corps résiduel est fini, le groupe de Weil $W(\overline K/K)$ est l'image inverse de $W(\overline k/k)$. Pour une représentation $\ell$-adique de ce groupe, le logarithme $N$ est défini. Deux relèvements d'un même Frobenius ont des valeurs propres qui coïncident à multiplication près par des racines de l'unité. Si les poids sont entiers, il existe une unique filtration par le poids $M$, stable par le groupe de Weil, avec $\Gr^M_i$ pur de poids $i$; si $V$ est pur, l'inertie agit par un quotient fini.

\paragraph{(1.7.8)--(1.7.12).} Pour un diviseur à croisements normaux $D=\bigcup_{i\in I}D_i$ dans un schéma régulier, la même construction s'applique aux faisceaux modérément ramifiés sur $X-D$. Localement, avec des équations $t_i=0$, on considère les revêtements $X_n=X[t_i^{1/n}]$ et l'action de $(\mu_n)^I$. On obtient des faisceaux $\mathcal F[E]$ sur les intersections et des endomorphismes nilpotents commutants $N_i\in\End(\mathcal F[E])(-1)$. Pour $J\subset K\subset I$, on a les identifications transitives
\begin{equation}\tag{1.7.9.1}
        \mathcal F[D_J][D^	imes_{J\cup K}]\simeq\mathcal F[D_{J\cup K}],
\end{equation}
avec le carré de compatibilité correspondant. La construction se rend canonique au moyen du fibré normal de l'intersection.

\section*{(1.8) Monodromie locale des faisceaux purs}

Les conventions (0.7) sont en vigueur. Soient $X_0$ une courbe lisse absolument irréductible sur $\Fq$ et $j:U_0\hookrightarrow X_0$ un ouvert de complément fini $S_0$.

\paragraph{Lemme (1.8.1).} Si $\mathcal F_0$ est un faisceau lisse ponctuellement $\iota$-pur de poids $\beta$ sur $U_0$, alors, pour tout $x\in |S_0|$ et toute valeur propre $\alpha$ de $F_x$ sur $j_*\mathcal F_0$, on a
\[
        w_{N(x)}(\alpha)\le\beta.
\]
La preuve repose sur la formule
\begin{equation}\tag{1.8.1.1}
\prod_{x\in |U_0|}\det(1-F_xt^{\deg x},\mathcal F_0)^{-1}
\prod_{x\in |S_0|}\det(1-F_xt^{\deg x},j_*\mathcal F_0)^{-1}
=
\frac{\det(1-Ft,H^1_c(X,j_*\mathcal F))}
{\det(1-Ft,H^2_c(X,j_*\mathcal F))},
\end{equation}
puis sur les puissances tensorielles de $\mathcal F_0$.

\paragraph{Théorème (1.8.4).} Avec les notations de (1.8.1) et (1.8.3), si $\mathcal F_0$ est ponctuellement $\iota$-pur de poids $\beta$, alors
\[
        \Gr^M_i(\mathcal F_{0,\overline\eta})
\]
est $\iota$-pur de poids $\beta+i$ comme représentation du groupe de Weil local. La preuve réduit à $\beta=0$, rend la monodromie unipotente par un revêtement fini, puis combine (1.8.1) avec les formules primitives de (1.6.14).

\paragraph{Corollaire (1.8.5).} Si $\mathcal F_0$ admet une filtration $W$ dont les gradués sont ponctuellement $\iota$-purs de poids $i$, la filtration de monodromie locale relative à $W$ existe et coïncide avec la filtration par le poids.

\paragraph{Corollaires (1.8.7)--(1.8.12).} La même conclusion vaut pour un diviseur lisse par restriction à des courbes transverses. En particulier, si $j:U_0\hookrightarrow X_0$ et si $\mathcal F_0$ est lisse $\iota$-mixte de poids ponctuels $\le\beta$ sur $U_0$, alors $j_*\mathcal F_0$ est encore $\iota$-mixte de poids ponctuels $\le\beta$. Sur un schéma normal, tout faisceau lisse $\iota$-mixte est extension successive de faisceaux lisses ponctuellement $\iota$-purs; et si un faisceau lisse $\iota$-mixte sur un schéma connexe est pur en un point, il est ponctuellement pur.

\paragraph{(1.8.14)--(1.8.15).} Le dictionnaire avec les variations de structures de Hodge suggère le problème suivant: définir une classe de bonnes variations de structures de Hodge mixtes sur $D^*$ pour lesquelles la filtration de monodromie relative existe, est compatible aux orbites nilpotentes, et transforme une orbite nilpotente en variation de structures de Hodge mixtes.
