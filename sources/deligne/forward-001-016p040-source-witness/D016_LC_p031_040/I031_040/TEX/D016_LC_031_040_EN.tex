\paragraph{Lemma 3.9.} Let \(F\) be an endomorphism of \(V\), let \(n\) be an integer, and let \(F_n\) be the endomorphism of
\[
V_n=V\otimes\C^n
\]
such that, for a basis \(e_0,\ldots,e_{n-1}\) of \(\C^n\),
\[
F_n(v\otimes e_i)=v\otimes e_{i+1}\quad(0\le i<n-1),
\qquad
F_n(v\otimes e_{n-1})=F(v)\otimes e_0.
\]
Then
\[
\det(1-F_nt)=\det(1-Ft^n).
\]
By the principle of continuation of algebraic identities, one may suppose \(F\) diagonal in a suitable basis of \(V\). One is immediately reduced to the case \(\dim(V)=1\), where \(F\) is multiplication by a number \(a^n\), \(a\ne0\). In the basis \(f_i=a^{-i}(v\otimes e_i)\), one has \(F_n(f_i)=af_{i+1}\), for \(i\in\Z/n\Z\). The eigenvectors of \(F_n\) are the
\[
\sum_i\chi(i)f_i
\]
where \(\chi\) runs through the characters of \(\Z/n\Z\), and the determinant identity follows.

\paragraph{3.10.} Let \(K\) be a global field, an \(\mathfrak A\)-field in the sense of [15]. For every place \(v\) of \(K\), write \(K_v\) for the local field which is the completion of \(K\) at \(v\). The objects defined above for local fields will be denoted, for \(K_v\), with a subscript \(v\). Let \(\mathbb A\) be the adele ring of \(K\), let \(\|\ \|\) be the absolute value satisfying
\[
d(ax)=\|a\|\,dx
\]
for a Haar measure \(dx\) on \(\mathbb A\), and let \(\omega_s\) be the quasi-character \(\|x\|^s\) of \(\mathbb A^*\). If \(x_v\) is the component of \(x\in\mathbb A\) in \(K_v\), then
\[
\|x\|=\prod_v\|x_v\|_v.
\]
Let \(dx\) denote the unique Haar measure on \(\mathbb A\) such that
\[
\int_{\mathbb A/K}dx=1
\]
(the Tamagawa measure), let \(d^*x\) be a Haar measure on \(\mathbb A^*\), and let \(\psi\) be a character trivial on \(\mathbb A/K\), with local components \(\psi_v\). For \(f\) a \(C^\infty\) function with compact support on \(\mathbb A\), put
\[
\widehat f(y)=\int_{\mathbb A}f(x)\psi(xy)\,dx.
\]

\paragraph{3.11.} Tate's global functional equation [14] says, for \(\chi\) a quasi-character of \(\mathbb A^*/K^*\),
\[
\tag{3.11.1}
\int_{\mathbb A^*}\widehat f(x)\chi'(x)\,d^*x
=
\int_{\mathbb A^*}f(x)\chi(x)\,d^*x,
\qquad
\chi'=\omega_1\chi^{-1},
\]
the integrals being defined by analytic continuation in the families \(\chi\omega_s\). Put
\[
\tag{3.11.2}
L(\chi)=\prod_vL(\chi_v)
\]
by analytic continuation, and, for \(dx=\otimes_vdx_v\),
\[
\tag{3.11.3}
\varepsilon(\chi)=\prod_v\varepsilon(\chi_v,\psi_v,dx_v).
\]
The product (3.11.3) is independent of the choice of \(\psi\) and of the decomposition of \(dx\), as follows from (3.3.2) and (3.3.3). If, for almost all \(v\),
\[
\int_{\mathcal O_v}dx_v=1,
\]
then almost all its factors are equal to \(1\). Formulas (3.3.1) and (3.11.1) give the global functional equation of Hecke \(L\)-functions:
\[
\tag{3.11.4}
\varepsilon(\chi)L(\chi')=L(\chi).
\]

\paragraph{3.12.} The results stated below are proved in [1] and [19]. Let \(K\) be a global field, \(\overline K\) an algebraic closure of \(K\), and \(V\) a complex representation of \(\Gal(\overline K/K)\), always assumed continuous: the action of \(\Gal(\overline K/K)\) factors through a finite Galois group \(\Gal(K'/K)\). For every place \(v\) of \(K\), choose a place of \(\overline K\) above it, and let \(V_v\) be the representation obtained from \(V\) by restriction to the subgroup
\[
W(\overline K_v/K_v)
\]
of the decomposition group \(\Gal(\overline K_v/K_v)\).

\begin{enumerate}[label=(\Alph*)]
\item For \(\chi\) a quasi-character of \(\mathbb A^*/K^*\), with local components \(\chi_v\), the infinite product
\[
L(V,\chi)=\prod_vL(V_v\otimes\chi_v)
\]
can be defined by analytic continuation in \(\chi\) in the families \(\chi\omega_s\). It converges for \(\Re(s)\) large and extends to a meromorphic function of \(s\).

\item Let \(V^*\) denote the dual of \(V\). One has
\[
L(V,\chi)=\varepsilon(V,\chi)\,L(V^*,\chi'),
\qquad \chi'=\omega_1\chi^{-1},
\]
with \(\varepsilon(V,\chi)\in\C^*\), of the form \(Ae^{Bs}\) as \(\chi\) varies in the families \(\chi\omega_s\).

\item Let \(L\) be a finite extension of \(K\) in \(\overline K\), let \(N_{L/K}\) be the norm, let \(V_L\) be a representation of \(\Gal(\overline K/L)\), and put
\[
V_K=\Ind_{\Gal(\overline K/L)}^{\Gal(\overline K/K)}(V_L).
\]
For \(\chi\) a quasi-character of \(\mathbb A_K^*/K^*\), one has
\[
L(V_K,\chi)=L(V_L,\chi\circ N_{L/K}),
\]
and similarly
\[
\varepsilon(V_K,\chi)=\varepsilon(V_L,\chi\circ N_{L/K}).
\]
\item One also has
\[
\varepsilon(V'\oplus V'',\chi)=\varepsilon(V',\chi)\varepsilon(V'',\chi),
\]
which makes it possible to define \(\varepsilon(V,\chi)\) when \(V\) is only a virtual representation.
\end{enumerate}
The assertions (A) and (B) are proved by reduction to the case \(\dim V=1\), via Brauer's theorem. Assertion (C) follows from (B) and from the compatibility of norm and transfer recalled above.

\section*{4. Existence of local constants}

\paragraph{Theorem 4.1.} There exists a unique function \(\varepsilon\), satisfying conditions (1) to (4) below, which associates a number
\[
\varepsilon(V,\psi,dx)\in\C^*
\]
to each isomorphism class of sextuples \((K,\overline K,\psi,dx,V,\rho)\), consisting of a local field \(K\), an algebraic closure \(\overline K\) of \(K\), a nontrivial additive character \(\psi:K\to\C^*\), a Haar measure \(dx\) on \(K\), a finite-dimensional complex vector space \(V\), and a representation
\[
\rho:W(\overline K/K)\longrightarrow\GL(V).
\]
\begin{enumerate}[label=(\arabic*)]
\item For every exact sequence of representations
\[
0\to V'\to V\to V''\to0,
\]
one has
\[
\varepsilon(V,\psi,dx)=\varepsilon(V',\psi,dx)\varepsilon(V'',\psi,dx).
\]
This condition shows that \(\varepsilon(V,\psi,dx)\) depends only on the class of \(V\) in the Grothendieck group of representations of \(W(\overline K/K)\), and allows \(\varepsilon(V,\psi,dx)\) to be interpreted when \(V\) is only a virtual representation.

\item
\[
\varepsilon(V,\psi,a\,dx)=a^{\dim V}\varepsilon(V,\psi,dx).
\]
In particular, for virtual \(V\) of dimension \(0\), \(\varepsilon(V,\psi,dx)\) is independent of \(dx\); it is denoted \(\varepsilon(V,\psi)\).

\item If \(L\) is a finite separable extension of \(K\) in \(\overline K\), and \(V_K\) is the virtual representation of \(W(\overline K/K)\) induced by a virtual representation of dimension zero \(V_L\) of \(W(\overline K/L)\), then
\[
\varepsilon(V_K,\psi)=\varepsilon(V_L,\psi\circ\Tr_{L/K}).
\]

\item If \(\dim V=1\), and \(\rho\) is defined by a quasi-character \(\chi\) of \(K^*\), then
\[
\varepsilon(V,\psi,dx)=\varepsilon(\chi,\psi,dx).
\]
\end{enumerate}

\paragraph{4.2.} Let
\[
v=\sum_{i,j}n_{ij}(H_i,\chi_{ij})\in R^+_0(W(\overline K/K))
\]
(1.10). Assume that the \(H_i\) are pairwise nonconjugate. Let \(K_i\) be the extension of \(K\) defined by \(H_i\), and again denote by \(\chi_{ij}\) the quasi-character of \(K_i^*\) defined by \(\chi_{ij}\). For any choice of Haar measures \(dx_i\) on the \(K_i\), and for \(\psi\) an additive character of \(K\), put
\[
\tag{4.2.1}
\varepsilon(v,\psi)=
\prod_{i,j}\varepsilon(\chi_{ij},\psi\circ\Tr_{K_i/K},dx_i)^{n_{ij}}.
\]
It follows from (3.3.2) that \(\varepsilon(v,\psi)\) is independent of the choice of the \(dx_i\).

\paragraph{Lemma 4.3.} With the notation 1.9.8, for \(v\in R^+_0(W(\overline K/K))\),
\[
\varepsilon(v,\psi(ax))=\det(v)(a)\,\varepsilon(v,\psi).
\]
It is enough to check this for \(v=(H,\chi)-(H,1)\), where \(H\) corresponds to an extension \(L\) of \(K\). Let \(v_0\) be the virtual representation \([\chi]-[1]\) of \(W(\overline K/L)\). Formula (3.3.3) gives the assertion over \(L\). It is known ([10], XII, §4) that the inclusion of \(K^*\) into \(L^*\) is identified with the transfer
\[
W(\overline K/K)^{\ab}\longrightarrow W(\overline K/L)^{\ab}.
\]
Since \(b(v)=\Ind(v_0)\), the lemma follows from 1.2.

\paragraph{Lemma 4.4.} The theorem is true in the archimedean case.

For \(K\simeq\C\), define \(\varepsilon\) by (1) and (4), and (2) is automatic. For \(K\simeq\R\), one checks from 2.9 that
\[
b:R^+(W(\overline K/K))\longrightarrow R(W(\overline K/K))
\]
is surjective, with kernel generated by the \(\chi_s-\chi_t\) where
\[
\chi_s=(K^*,\omega_s)-[\omega_s]-[\chi^{-1}\omega_{s+1}].
\]
For a virtual representation \(V\), and \(v\in R^+_0(W(\overline K/K))\) such that
\[
b(v)=V-\dim(V)[1],
\]
put
\[
\tag{4.4.1}
\varepsilon(V,\psi,dx)=\varepsilon(v,\psi)\,\varepsilon(1,\psi,dx)^{\dim V}.
\]
We check that the right-hand side is independent of the choice of \(v\), i.e. that, for \(v\) satisfying \(b(v)=0\), one has \(\varepsilon(v,\psi)=1\). By 4.3, it is enough to check this for \(\psi(x)=\exp(2\pi ix)\). For this choice of \(\psi\), the assertion follows from the preceding determination of \(\Ker(b)\), and from the fact that the right-hand sides of (3.4.1), (3.4.2) are independent of \(s\). It is clear that \(\varepsilon\) defined by (4.4.1) satisfies (1) to (4).

We now restrict to the nonarchimedean case.

\paragraph{4.5.} Let \(K\) be a nonarchimedean local field, \(\overline K\) a separable closure of \(K\), and \(K_{nr}\subset\overline K\) the maximal unramified extension of \(K\). Let \(J\) be an open normal subgroup of the inertia group
\[
I=\Gal(\overline K/K_{nr})
\]
and let \(L\) be the corresponding extension of \(K_{nr}\). For \(\sigma\in I/J\), put
\[
t_{I/J}(\sigma)=\inf\bigl(v_L(\sigma(x)-x)\mid x\text{ integral in }L\bigr),
\]
and define functions \(a_{I/J}\) and \(sw_{I/J}\) by
\[
\tag{4.5.1}
a_{I/J}(\sigma)=-t_{I/J}(\sigma)\quad(\sigma\ne e),
\]
\[
sw_{I/J}(\sigma)=1-t_{I/J}(\sigma)\quad(\sigma\ne e),
\]
\[
\sum_\sigma a_{I/J}(\sigma)=\sum_\sigma sw_{I/J}(\sigma)=0.
\]
By Artin (see [10], VI, §2, th. 1), \(a_{I/J}\) is the character of a representation \(A_{I/J}\) of \(I/J\), the Artin representation. By loc. cit., prop. 2, \(sw_{I/J}\) is likewise the character of a representation \(Sw_{I/J}\), the Swan representation. These representations are defined only up to isomorphism. \(A_{I/J}\) is the sum of \(Sw_{I/J}\) and the augmentation representation of \(I/J\). For \(I\supset J\supset K\), one has (loc. cit., prop. 3)
\[
\tag{4.5.2}
(A_{I/K})^{J/K}\iso A_{I/J},
\qquad
(Sw_{I/K})^{J/K}\iso Sw_{I/J}.
\]
If \(d\) is the valuation of the discriminant \(\delta_{L/K_{nr}}\), i.e. the valuation of the different \(\mathfrak D_{L/K}\) ([10], III, §3, prop. 6), and \(r\) denotes a regular representation, then (loc. cit., prop. 4)
\[
\tag{4.5.3}
A_{I/K}|_{J/K}=d\,r_{J/K}+A_{J/K}.
\]
Let \(V\) be a representation of \(W(\overline K/K)\), trivial on \(J\subset I\), with character \(\psi\). Define the Artin and Swan conductors of \(V\) by
\[
\tag{4.5.4}
a(V)=\dim\Hom_{I/J}(A_{I/J},V)=sw(V)+\dim V-\dim V^I,
\]
\[
sw(V)=\dim\Hom_{I/J}(Sw_{I/J},V)
=\frac1{|I/J|}\sum_\sigma sw_{I/J}(\sigma)\psi(\sigma).
\]
By 4.5.2, these conductors are independent of the choice of \(J\). For \(V\) of dimension \(1\), defined by a quasi-character \(\chi\), one has (loc. cit., prop. 5)
\[
\tag{4.5.5}
a(V)=\text{the conductor }a(\chi)\text{ of }\chi.
\]
The preceding notation is therefore compatible with that introduced in 3.4.

With the notation 1.9.8, one has:

\paragraph{Lemma 4.5.} For \(v\in R^+_0(W(\overline K/K))\) and \(\chi\) an unramified quasi-character of \(K^*\),
\[
\varepsilon(v\cdot\chi,\psi)=\chi(\pi)^{a(v)}\varepsilon(v,\psi).
\]
It suffices to verify this for \(v=(H,\lambda)-(H,\mu)\), where \(H\) defines an extension \(L\) of \(K\), and \(\lambda,\mu\) are two quasi-characters of \(L^*\). Let \(d\) be the valuation of the different of \(L/K\), let \(f\) be the residue-degree, let \(I\subset W(\overline K/K)\) and \(J\subset W(\overline K/L)\) be the inertia groups, and choose \(K\subset J\) small enough. We compute \(a(v)\). The adjunction formula for induced representations, (4.5.3), and (4.5.5) give, for every quasi-character \(\nu\) of \(H\), i.e. of \(L^*\),
\[
\tag{4.5.1'}
a((H,\nu))
=\dim\Hom_{I/K}(A_{I/K},\Ind_H^W(\nu))
=f\,\dim\Hom_{J/K}(A_{J/K}+d\,r_{J/K},[\nu]|_J)
=f\,a(\nu)+fd.
\]
Thus
\[
a(v)=f(a(\lambda)-a(\mu)).
\]
For \(\nu\) a quasi-character of \(L^*\), \(\chi_1\) an unramified quasi-character of \(L^*\), and \(n=n(\psi\circ\Tr_{L/K})\) as in 3.4, one has by (3.4.3.1) or (3.4.3.2)
\[
\frac{\varepsilon(\nu\chi_1,\psi\circ\Tr,dx)}
     {\varepsilon(\nu,\psi\circ\Tr,dx)}
=
\chi_1(\pi_L)^{a(\nu)+n}.
\]
Hence
\[
\varepsilon(v\cdot\chi,\psi)\varepsilon(v,\psi)^{-1}
=
\chi\circ N_{L/K}\bigl(\pi_L^{a(\lambda)-a(\mu)}\bigr)
=
\chi(\pi)^{f(a(\lambda)-a(\mu))},
\]
which is the announced formula.

\paragraph{Terminology 4.6.} Let \(K_1\) be a Galois extension, finite or not, of \(K\), with Galois group \(G\). Let \(\alpha\) be a quasi-character of \(K^*\), and \(\psi\) a nontrivial additive character of \(K\). We shall say that there exists an \(\alpha,\psi\)-theory of constants for \(K_1/K\) if there exists \(\varepsilon_{\alpha,\psi}\) of the following type:
\begin{enumerate}[label=(\arabic*)]
\setcounter{enumi}{-1}
\item For \(H\) an open subgroup of \(G\), defining a finite extension \(L\) of \(K\), for \(V\) a representation of \(H\), and \(dx\) a Haar measure on \(L\),
\[
\varepsilon_{\alpha,\psi}(V,dx)\in\C^*
\]
is defined.
\item For every exact sequence of representations
\[
0\to V'\to V\to V''\to0,
\]
one has
\[
\varepsilon_{\alpha,\psi}(V,dx)=
\varepsilon_{\alpha,\psi}(V',dx)\varepsilon_{\alpha,\psi}(V'',dx).
\]
Thus \(\varepsilon_{\alpha,\psi}\) still makes sense for virtual representations.
\item One has
\[
\varepsilon_{\alpha,\psi}(V,a\,dx)=a^{\dim V}\varepsilon_{\alpha,\psi}(V,dx).
\]
In particular, for virtual \(V\) of dimension \(0\), \(\varepsilon_{\alpha,\psi}(V,dx)\) is independent of \(dx\); it is denoted \(\varepsilon_{\alpha,\psi}(V)\).
\item For \(H_1\subset H_2\subset G\), defining \(L_1\supset L_2\), and \(V\) a virtual representation of dimension \(0\) of \(H_1\), one has
\[
\varepsilon_{\alpha,\psi}(\Ind_{H_1}^{H_2}V)=\varepsilon_{\alpha,\psi}(V).
\]
\item For \(V\) of dimension \(1\), defined by a character \(\chi\) of \(L^*\),
\[
\varepsilon_{\alpha,\psi}(V,dx)
=
\varepsilon(\chi\,\alpha\circ N_{L/K},\,\psi\circ\Tr_{L/K},\,dx).
\]
\end{enumerate}

\paragraph{Lemma 4.7.} Let \(K_1/K\) and \(\alpha\) be as above, and let \(\psi\) be a nontrivial additive character of \(K\). There exists at most one \(\alpha,\psi\)-theory of constants for \(K_1/K\). For one to exist, it is necessary and sufficient that, for every
\[
v\in\Ker(R^+_0(G)\longrightarrow R(G)),
\]
one have
\[
\varepsilon(v\cdot\alpha,\psi)=1.
\]
Let \(H\subset G\), \(L\), and \(V\) be as above. By 1.5, there exists \(v\in R^+_0(H)\) whose image in \(R(H)\) is \(V-\dim(V)[1]\). Necessarily
\[
\tag{4.7.1}
\varepsilon_{\alpha,\psi}(V,dx)
=
\varepsilon(v\cdot(\alpha\circ N_{L/K}),\,\psi\circ\Tr_{L/K})\,
\varepsilon(\alpha\circ N_{L/K},\,\psi\circ\Tr_{L/K},\,dx)^{\dim V},
\]
which gives uniqueness. The hypothesis ensures that \(\varepsilon_{\alpha,\psi}\), defined by this formula, is independent of the choice of \(v\); one easily checks axioms (0) to (4).
