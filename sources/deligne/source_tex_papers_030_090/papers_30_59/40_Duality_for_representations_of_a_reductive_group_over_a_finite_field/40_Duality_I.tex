\section*{1}

The purpose of this paper is to construct a duality operation for representations of a reductive group over a finite field. Its effect, very roughly speaking, is to interchange irreducible representations of small degree with ones of large degree (for example, the unit and Steinberg representation). At the level of characters, this operation has been also considered by Alvis~[1], Curtis~[2], and Kawanaka~[4].

We shall now fix some notation. $G$ will denote a connected reductive group defined over a finite field $\F_q$, $(W,S)$ its Weyl group, and $\tilde{S}$ the set of orbits of the Frobenius map on $S$. The subsets of $\tilde{S}$ parametrize the classes of parabolic subgroup of $G$ which are defined over $\F_q$; let $\cP_I$ be the class corresponding to $I \subset \tilde{S}$. (Thus $\cP_\varnothing$ is the class of Borel subgroups and $\cP_{\tilde{S}} = \{G\}$.) We denote by $G$ the group of $\F_q$-rational points of $G$ and by $\cP_I$ the set of parabolic subgroups in $\cP_I$ which are defined over $\F_q$. Similarly, if $P \in \cP_I$, we denote by $P$ its group of $\F_q$-rational points and by $U_P$ the group of $\F_q$-rational points of its unipotent radical. Let $K$ be an algebraically closed field of characteristic zero. All $G$-modules will be over $K$.

Let $E$ be a $G$-module. For each $I \subset \tilde{S}$ we define
\begin{equation*}
E_{(I)} = \bigoplus_{P \in \cP_I} E^{U_P},
\end{equation*}
where $E^{U_P}$ denotes the set of $U_P$-invariant vectors of $E$. We regard $E_{(I)}$ as a $G$-module in a natural way. If $I \subset I'$, there is a canonical linear map $\varphi_I^{I'} \colon E_{(I)} \to E_{(I')}$. It is defined as follows: if $e \in E^{U_P}$ ($P \in \cP_I$), $e$ can be also regarded as an element of $E^{U_{P'}}$, where $P' \in \cP_{I'}$ is uniquely defined by the condition $P \subset P'$. (Note that $U_{P'} \supset U_P$, hence $E^{U_P} \subset E^{U_{P'}}$.)
The map $\varphi_I^{I'}$ takes $e \in E^{U_P}$ to $e$ regarded as an element of $E^{U_{P'}}$. It is clear that if $I \subset I' \subset I''$, then
\begin{equation}\label{eq:1.1}
\varphi_{I'}^{I''} \varphi_I^{I'} = \varphi_I^{I''}.
\end{equation}

Now let $\tilde{E}_{(I)} = E_{(I)} \otimes \Lambda^{|I|}(K^I)$. (We regard $\Lambda^{|I|}(K^I)$ as a $G$-module with trivial action.) If $I \subset I'$ and $|I'| = |I| + 1$, we have a natural map $\varepsilon_I^{I'} \colon \Lambda^{|I|}(K^I) \to \Lambda^{|I'|}(K^{I'})$ given by $\omega \mapsto \omega \wedge f$ where $f$ is the unique element in $I' - I$. We define $\psi_I^{I'} \colon \tilde{E}_{(I)} \to \tilde{E}_{(I')}$ by $\psi_I^{I'} = \varphi_I^{I'} \otimes \varepsilon_I^{I'}$.

We now consider the sequence of maps of $G$-modules:
\begin{equation}\label{eq:1.2}
0 \to \tilde{E}_{(\varnothing)} \to \bigoplus_{|I|=1} \tilde{E}_{(I)} \to \bigoplus_{|I|=2} \tilde{E}_{(I)} \to \cdots \to \tilde{E}_{(\tilde{S})} \to 0
\end{equation}
where the maps $d$ have components $\psi_I^{I'}$ ($I \subset I'$, $|I'| = |I|+1$). Using \eqref{eq:1.1}, we see that \eqref{eq:1.2} is a complex.

\section*{2}

We have the following:

\begin{theorem}
Assume that $E$ is irreducible. Let $i_0$ be the smallest integer $\geq 0$ such that $\bigoplus_{|I|=i_0} \tilde{E}_{(I)} \neq 0$. Then the sequence
\begin{equation}\label{eq:2.1}
\bigoplus_{|I|=i_0} \tilde{E}_{(I)} \xrightarrow{d_{i_0}} \bigoplus_{|I|=i_0+1} \tilde{E}_{(I)} \to \cdots \to \tilde{E}_{(\tilde{S})} \to 0
\end{equation}
is exact.
\end{theorem}

In particular, the homology of the complex \eqref{eq:1.2} is concentrated in a single degree.

\section*{3}

We shall prove that the conclusion of Theorem~2 holds in the case where instead of assuming $E$ irreducible we shall assume that:
\begin{equation}\label{eq:3.1}
\text{$E$ admits a direct sum decomposition $E = \bigoplus_{P \in \cP_{I_0}} E_P$ (for some $I_0 \subset \tilde{S}$)}
\end{equation}
such that
\begin{enumerate}[label=(\alph*)]
\item $gE_P = E_{gPg^{-1}}$ ($\forall g \in G$, $P \in \cP_{I_0}$);
\item for each $P \in \cP_{I_0}$, the $P$-module $E_P$ factors through $P/U_P$ and is irreducible, cuspidal as a $P/U_P$-module.
\end{enumerate}
This would imply the theorem since any irreducible $G$-module is a direct summand of a $G$-module $E$ as in \eqref{eq:3.1}.

In this and the following section, we assume that $E$ is as in \eqref{eq:3.1}.

Given $J \subset \tilde{S}$, we define $\cZ_J$ to be the set of $G$-orbits $\sigma$ on $\cP_J \times \cP_{I_0}$ which have the following property. There exists $P \in \cP_{I_0}$, $Q \in \cP_J$, $(Q,P) \in \sigma$ such that $Q$ contains a Levi subgroup of $P$.

If $Q \in \cP_J$, we have
\begin{equation*}
E^{U_Q} = \Bigl(\bigoplus_{P \in \cP_{I_0}} E_P\Bigr)^{U_Q} = \bigoplus_{\substack{P \in \cP_{I_0} \\ (Q \cap P)U_P = P}} E_P^{U_Q} \oplus \bigoplus_{\substack{P \in \cP_{I_0} \\ (Q \cap P)U_P \subsetneq P}} E_P^{U_Q}.
\end{equation*}
The second summand is zero (since $E_P$ is cuspidal for $P/U_P$, so that $E_P^{U_P^g \cap P} = 0$). Thus
\begin{equation}\label{eq:3.2}
E_{(J)} = \bigoplus_{Q \in \cP_J} E^{U_Q} = \bigoplus_{Q \in \cP_J} \bigoplus_{\substack{P \in \cP_{I_0} \\ (Q \cap P)U_P = P}} E_P^{U_Q} = \bigoplus_{\sigma \in \cZ_J} E^\sigma
\end{equation}
where
\begin{equation*}
E^\sigma = \bigoplus_{\substack{(Q,P) \in \cP_J \times \cP_{I_0} \\ (Q,P) \in \sigma \\ (Q \cap P)U_P = P}} E_P^{U_Q} \qquad (\sigma \in \cZ_J).
\end{equation*}

Now assume that $J \subset J'$ and $\sigma \in \cZ_J$; we define $\varphi_J^{J'}(\sigma) = \sigma' = \{(Q',P) \in \cP_{J'} \times \cP_{I_0} \mid \exists Q \in \cP_J,\ Q \subset Q',\ (Q,P) \in \sigma\}$. Then $\sigma' \in \cZ_{J'}$ and we say that $\sigma'$ is the face of type $J'$ of $\sigma$. In this case, we define a linear map $\psi_J^{J'} \colon E^\sigma \to E^{\sigma'}$ as follows. If $e = (e_{Q,P}) \in E^\sigma$, $e_{Q,P} \in E_P$ $\forall (Q,P) \in \sigma$, we set $\psi_J^{J'}(e) = (e'_{Q',P}) \in E^{\sigma'}$ where, for any $(Q',P) \in \sigma'$, we have
\begin{equation*}
e'_{Q',P} = \sum_{\substack{Q \in \cP_J \\ Q \subset Q' \\ (Q,P) \in \sigma}} e_{Q,P} \in E_P.
\end{equation*}

Let us admit the following

\begin{lemma}\label{lem:3.3}
For any $\sigma, \sigma'$ as above, the map $\psi_J^{J'} \colon E^\sigma \to E^{\sigma'}$ is an isomorphism.
\end{lemma}

Note also that given $J \subset J' \subset J''$ and $\sigma \in \cZ_J$, $\sigma' \in \cZ_{J'}$ as above we can define $\sigma'' \in \cZ_{J''}$ to be the face of type $J''$ of $\sigma$ (or of $\sigma'$) and we have
\begin{equation*}
\psi_{J'}^{J''} \psi_J^{J'} = \psi_J^{J''}.
\end{equation*}

Using \eqref{eq:3.2}, the complex \eqref{eq:1.2} can be rewritten as
\begin{equation}\label{eq:3.4}
\cdots \to \bigoplus_{|J|=|I_0|+1} \bigoplus_{\sigma \in \cZ_J} E^\sigma \otimes \Lambda^{|J|}(K^J) \to \bigoplus_{|J|=|I_0|} \bigoplus_{\sigma \in \cZ_J} E^\sigma \otimes \Lambda^{|J|}(K^J) \to \cdots
\end{equation}
where the maps $d$ have components $\psi_J^{J'} \otimes \varepsilon_J^{J'}$ ($J \subset J'$, $|J'| = |J|+1$, $\sigma \in \cZ_J$, $\sigma' \in \cZ_{J'}$ the face of type $J'$ of $\sigma$). Let $\rho$ be the unique element in $\cZ_{\tilde{S}}$. We have $E^\rho = E$. Using Lemma~\ref{lem:3.3}, we may identify $E^\sigma$ with $E$ for any $\sigma \in \cZ_J$, via the isomorphism $\psi_{\tilde{S}}^J$; the complex \eqref{eq:3.4} becomes the tensor product of $E$ with the complex
\begin{equation}\label{eq:3.5}
0 \to \cdots \to \bigoplus_{|J|=|I_0|+1} K^{\cZ_J} \otimes \Lambda^{|J|}(K^J) \to \bigoplus_{|J|=|I_0|} K^{\cZ_J} \otimes \Lambda^{|J|}(K^J) \to \cdots
\end{equation}
whose differential has components $\psi_J^{J'} \otimes \varepsilon_J^{J'}$ ($J \subset J'$, $|J'| = |J|+1$). This complex has the following simple interpretation. Let $T$ be an $\F_q$-split torus of maximal possible dimension in the adjoint group $G_{\mathrm{ad}}$. Let $Y$ be the lattice of one-parameter subgroups of $T$. The root hyperplanes in the real vector space $Y \otimes \R$ give rise to a partition of $Y \otimes \R$ into simplicial cones. These are in a natural 1--1 correspondence with the parabolic subgroups in $G$ which are defined over $\F_q$ and whose image in $G_{\mathrm{ad}}$ contain $T$. Let us fix $P_0 \in \cP_{I_0}$. It corresponds to a cone in $Y \otimes \R$, which spans a linear subspace $L \subset Y \otimes \R$ of codimension $|I_0|$. The cones contained in $L$ are in 1--1 correspondence with those parabolic subgroups $Q \subset G$ (defined over $\F_q$) which contain a Levi subgroup of $P_0$ and whose image in $G_{\mathrm{ad}}$ contain $T$, hence they are in 1--1 correspondence with $\bigcup_{J \supset I_0} \cZ_J$. Thus $\bigcup_{J \supset I_0} \cZ_J$ can be regarded as the set of simplices in a triangulation of a unit sphere with centre $0$ in $L$; and \eqref{eq:3.5} is the (reduced) chain complex of this triangulation. It follows that its homology is concentrated in a single degree (corresponding to $|J| = |I_0|$) where it is $\pm K$.

\section*{4}

It remains to prove Lemma~\ref{lem:3.3}. We will give the proof in six steps. Parts of the argument are similar in spirit to arguments in~[3].

\smallskip
\noindent\textbf{Step 1.} Given $\sigma \in \cZ_J$ and $(Q,P) \in \sigma$, we consider the parabolic subgroup $R = (Q \cap P)U_Q \subset Q$. Then $R \in \cP_{\tilde{J}} \in \cZ_{\tilde{J}}$ for some $\tilde{J} \subset J$ and $\sigma$ is the face of type $J$ of some element of $\cZ_{\tilde{J}}$. Also $R$, $P$ have a common Levi subgroup and hence are associated. We first show that
\begin{equation*}
\varphi \colon \bigoplus_{\substack{R \in \cP_{\tilde{J}} \\ (R,P) \in \cZ_{\tilde{J}} \\ R \subset Q}} E_P^{U_R} \longrightarrow E_P^{U_Q}
\end{equation*}
is an isomorphism.

Now the pairs $(R,P)$ such that $(R,P) \in \cZ_{\tilde{J}}$ are in 1--1 correspondence with the pairs $(Q,P)$ such that $(Q,P) \in \sigma$. Hence $\varphi$ is injective. If we fix $Q \in \cP_J$, there is a natural 1--1 correspondence between the set $\{R \in \cP_{\tilde{J}} \mid R \subset Q\}$ and the set of orbits of $U_Q$ on the set $\{P \in \cP_{I_0} \mid (Q,P) \in \sigma\}$. Hence the space $E^\sigma$ is a direct sum over all pairs $R \subset Q$ ($R \in \cP_{\tilde{J}}$, $Q \in \cP_J$) of pieces of form $\bigl(\bigoplus_{P \in B}\bigr)^{U_Q}$, where $P$ runs through a fixed $U_Q$-orbit. But this piece has the same dimension as $E_P$. (Note that for $P$ in that $U_Q$-orbit we have $P \cap U_Q \subset U_P$ and $U_P$ acts trivially on $E_P$.) Thus $\dim E^\sigma = \dim E_P \cdot \#(\cP_J) = \dim E_\rho \cdot \#(\cP_{I_0}) = \dim E$. Similarly, $\dim E^{\sigma'} = \dim E$. It follows that $\varphi$ is an isomorphism.

\smallskip
\noindent\textbf{Step 2.} Assume that $|I_0| = |\tilde{S}| - 1$. Let $\cP_J$ be the class of parabolic subgroups opposed to $\cP_{I_0}$, and let $\sigma = \{(Q,P) \in \cP_J \times \cP_{I_0} \mid \text{$Q$ opposed to $P$}\}$. We will show that $\psi_J^{\tilde{S}} \colon E^\sigma \to E^\rho = E$ is an isomorphism. It is certainly non-zero. Thus, if $E$ is irreducible, $\psi_J^{\tilde{S}}$ is indeed an isomorphism. If $E$ is reducible, it has exactly two composition factors, and $J$ is necessarily equal to $I_0$. For each $P \in \cP_{I_0}$, there exists an isomorphism
\begin{equation*}
\eta_P \colon E_P \xrightarrow{\;\sim\;} \bigoplus_{(P',P) \in \sigma} E_{P'}^{U_P}
\end{equation*}
of $P/U_P$-modules unique up to a scalar. We may assume that $\eta_{gPg^{-1}} = g\eta_P g^{-1}$ ($\forall P \in \cP_{I_0}$, $\forall g \in G$). These maps together define an isomorphism of $G$-modules $\eta = \bigoplus_{P \in \cP_{I_0}} \eta_P$:
\begin{equation*}
E = \bigoplus_{P \in \cP_{I_0}} E_P \xrightarrow{\;\sim\;} \Bigl(\bigoplus_{(P',P) \in \sigma} E_{P'}\Bigr)^{U_P}.
\end{equation*}
The composition $\psi_J^{\tilde{S}} \circ \eta$ is a semisimple $G$-endomorphism of $E$. It clearly has trace zero (its diagonal blocks are zero). Let $\lambda'$, $\lambda''$ be its eigenvalues on the two irreducible pieces $E'$, $E''$ of $E$. Then $\lambda' \dim E' + \lambda'' \dim E'' = 0$. Since $\dim E' \neq 0$, $\dim E'' \neq 0$, $\lambda'$ and $\lambda''$ are either both zero or both non-zero. They cannot be both zero since $\psi_J^{\tilde{S}} \circ \eta \neq 0$. Hence they are both non-zero. It follows that $\psi_J^{\tilde{S}} \circ \eta$ and hence $\psi_J^{\tilde{S}}$ are isomorphisms.

\smallskip
\noindent\textbf{Step 3.} Assume that $|I_0| < |\tilde{S}| - 1$. Let $J \subset \tilde{S}$ be such that $|J| = |I_0| + 1$, and let $\tau \in \cZ_J$. There are precisely two elements $\sigma' \in \cZ_{J'}$, $\sigma'' \in \cZ_{J''}$ such that $J'$, $J'' \subset J$, $|J'| = |J''| = |I_0|$, and such that $\tau$ is the $J$-face of $\sigma'$ and the $J$-face of $\sigma''$. We shall prove that $\psi_{I_0}^{J'}$ and $\psi_{I_0}^{J''}$ are isomorphisms. One of these maps, say $\psi_{I_0}^{J'}$, is of the type considered in Step~1 ($\sigma = \tau$); hence it is known to be an isomorphism. We now turn to $\psi_{I_0}^{J''}$. Define
\begin{equation*}
E' = \bigoplus_{R \in \cP_J} E'_R
\end{equation*}
where, by definition,
\begin{equation*}
E'_R = \Bigl(\bigoplus_{\substack{P \in \cP_{I_0} \\ (R \cap P)U_P = P}} E_P\Bigr)^{U_R}.
\end{equation*}
Then $E'$ is a $G$-module in a natural way. It satisfies a property similar to property \eqref{eq:3.1} for $E$. Let $\alpha$ be the set of pairs $(R'',R') \in \cP_{J''} \times \cP_{J'}$ such that $R''$, $R'$ are both contained in the same $Q \in \cP_J$ and are opposed in $Q$. We define $E'^\alpha$ and $\psi_{J''}^{J'} \colon E'^\alpha \to E'$ in the same way as $E^\sigma$ and $\psi_J^{J'} \colon E^\sigma \to E^{\sigma'}$. We have a natural commutative diagram
\begin{equation*}
\begin{CD}
E'^{\alpha} @>{\psi_{J''}^{J'}}>> E' (= E^{\sigma'}) \\
@VV{\cong}V @AA{\psi_{I_0}^{J'}}A \\
E^{\sigma''} @>{\psi_{I_0}^{J''}}>> E^{\sigma} (= E^\tau)
\end{CD}
\end{equation*}
(Note that, given $(R'',P) \in \sigma''$ there is a unique $R' \in \cP_{J'}$ such that $(R'',R') \in \alpha$, $(R',P) \in \sigma'$; conversely, if $(R'',R') \in \alpha$, $(R',P) \in \sigma'$ then $(R'',P) \in \sigma''$. This gives rise to the isomorphism $E'^{\alpha} \xrightarrow{\sim} E^{\sigma''}$.) Now $\psi_{J''}^{J'}$ is known to be an isomorphism; moreover, by Step~2, $\psi_{J''}^{J'}$ is an isomorphism. It follows that $\psi_{I_0}^{J''}$ is an isomorphism.

\smallskip
\noindent\textbf{Step 4.} Let $\sigma \in \cZ_J$, $|J| = |I_0|$. (Thus $\sigma$ corresponds to an open cone in $L$.) We show that $\psi_{I_0}^{J} \colon E^\sigma \to E$ is an isomorphism. We can find a sequence $\sigma = \sigma_1, \sigma_2, \dots, \sigma_n = \rho$ of elements corresponding to open cones in $L$ such that (when regarded as cones in $L$) two consecutive ones have a common face of codimension~1; moreover, we take $n$ as small as possible. Let $\tau$ be the element corresponding to the common face of codimension~1 of $\sigma = \sigma_1$ and $\sigma_2$. By induction on $n$, we may assume that $\psi_{I_0}^{J_2} \colon E^{\sigma_2} \to E$ is an isomorphism. From Step~3, we know that $\psi_{I_0}^{J}$ and $\psi_{J}^{J_2}$ are isomorphisms. From $\psi_{I_0}^{J_2} = \psi_{J}^{J_2} \circ \psi_{I_0}^{J}$, it follows that $\psi_{I_0}^{J}$ is an isomorphism. From $\psi_{I_0}^{\tilde{S}} = \psi_{J}^{\tilde{S}} \circ \psi_{I_0}^{J}$ it follows that $\psi_{I_0}^{\tilde{S}}$ is an isomorphism.

\smallskip
\noindent\textbf{Step 5.} Let $J \subset \tilde{S}$ and $\sigma \in \cZ_J$. We want to show that $\psi_{I_0}^{J}$ is an isomorphism. We can find $I \subset J$ and $\delta \in \cZ_I$ such that $|I| = |I_0|$, such that $\sigma$ is the face of type $J$ of $\delta$, and such that $\psi_{I_0}^{I}$ is of the type considered in Step~4 (hence an isomorphism). We have $\psi_{I}^{J} \circ \psi_{I_0}^{I} = \psi_{I_0}^{J}$. By Step~4, $\psi_{I_0}^{I}$ is an isomorphism. It follows that $\psi_{I_0}^{J}$ is an isomorphism.

\smallskip
\noindent\textbf{Step 6.} Let $J \subset J'$ and $\sigma \in \cZ_J$; let $\sigma' \in \cZ_{J'}$ be the face of type $J'$ of $\sigma$. We want to show that $\psi_{J}^{J'}$ is an isomorphism. We have $\psi_{I_0}^{J'} \circ \psi_{J}^{J'} = \psi_{I_0}^{J}$. By Step~5, $\psi_{I_0}^{J}$ and $\psi_{I_0}^{J'}$ are isomorphisms. It follows that $\psi_{J}^{J'}$ is an isomorphism.

This completes the proof of the Lemma~\ref{lem:3.3} and hence that of the theorem.

\section*{5}

We have

\begin{corollary}[see Alvis~{[1]}]
In the setting of the theorem, let $E^\#$ denote the kernel of $d_{i_0}$ in \eqref{eq:2.1}. Then
\begin{enumerate}[label=(\alph*)]
\item $E^\#$ is an irreducible $G$-module,
\item $(E^\#)^\#$ is isomorphic to $E$ as a $G$-module,
\item $(-1)^{i_0} E^\# = \sum_{I \subset \tilde{S}} (-1)^{|I|} E_{(I)}$ in the Grothendieck group of virtual $G$-modules.
\end{enumerate}
\end{corollary}

Statement (c) is obvious.

Let $E'$ be a $G$-module of the type considered in \eqref{eq:3.1} which has a direct sum decomposition $E' = E_1 \oplus \cdots \oplus E_n$ with $E_i$ irreducible $G$-modules and $E_1 \cong E$. The proof of Theorem~2 shows that $(E')^\#$ can be defined in the same way as $E^\#$, and that it has the following properties:
\begin{equation}\label{eq:5.1}
(E')^\# \cong E_1^\# \oplus \cdots \oplus E_n^\#, \qquad E'^\# \subset E'.
\end{equation}
For each $1 \leq i, j \leq n$, we have from (c)
\begin{equation}\label{eq:5.2}
\langle E_i^\#, E_j^\# \rangle = \langle (E_i^\#)^\#, E_j \rangle = \sum_{I \subset \tilde{S}} (-1)^{|I|+i_0} \langle E_{i(I)}, E_{j(I)} \rangle
\end{equation}
(where $P_I \in \cP_I$). It follows that
\begin{equation*}
\langle E_i^\#, E' \rangle = \sum_j \langle E_i^\#, E_j^\# \rangle = \sum_j \langle E_i, E_j^\# \rangle = \langle E_i, (E')^\# \rangle = \langle E_i, E' \rangle
\end{equation*}
and, in particular, $E_i^\# \neq 0$. If $E_i^\#$ is not irreducible for some $i$, then $E_1^\# \oplus \cdots \oplus E_n^\#$ would have more irreducible components than $E_1 \oplus \cdots \oplus E_n$, contradicting \eqref{eq:5.1}. Thus, $E_i^\#$ is irreducible for each $i$. Hence, there exists a permutation $\pi$ of $\{1,2,\dots,n\}$ such that $E_i^\# = E_{\pi(i)}$ for all $i$. From \eqref{eq:5.2}, we have
\begin{equation*}
\langle E_{\pi^{-1}(i)}, E_i \rangle = \langle (E_i^\#)^\#, E_i \rangle = \langle E_i^{\#\#}, E_i^\# \rangle = \langle E_{\pi(i)}, E_{\pi(i)} \rangle = 1.
\end{equation*}
It follows that $E_{\pi^{-1}(i)}$ is isomorphic to $E_i$. The corollary is proved.
