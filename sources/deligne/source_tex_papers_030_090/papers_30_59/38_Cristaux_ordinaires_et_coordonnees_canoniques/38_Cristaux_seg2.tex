\subsection{F-cristaux de Hodge ordinaires de niveau $\leqslant 1$}

\textbf{Prouvons (ii).} Les formes $\eta_{ij}$ \'etant ferm\'ees, il existe, en vertu du lemme de Poincar\'e, des s\'eries $\tau_{ij} \in K[[t]]$, d\'etermin\'ees \`a une constante pr\`es, telles que l'on ait (1.4.2.6). La relation (1.4.2.5) entra\^ine alors que
\[
\varphi^*\tau_{ij} - p\tau_{ij} - pu_{ij}(\varphi) \in K,
\]
donc
\begin{equation}\tag{1.4.2.10}
\varphi^*\tau_{ij} - p\tau_{ij} - pu_{ij}(\varphi) = (\varphi^*\tau_{ij})(0) - p\tau_{ij}(0) - pu_{ij}(\varphi)(0).
\end{equation}

Notons $x \mapsto x^\sigma$ l'automorphisme de Frobenius de $K$. Fixons tout d'abord le rel\`evement $\varphi$ de Frobenius tel que $\varphi(t_i) = t_i^p$, et normalisons $\tau_{ij}$ par la condition
\begin{equation}\tag{1.4.2.11}
(\varphi^*\tau_{ij})(0) - p\tau_{ij}(0) - pu_{ij}(\varphi)(0) = 0
\end{equation}
qui s'\'ecrit encore (car $(\varphi^*\tau_{ij})(0) = \tau_{ij}(0)^\sigma$)
\begin{equation}\tag{1.4.2.12}
\tau_{ij}(0) = \sum_{n\geqslant 1} v^n u_{ij}(\varphi)(0)
\end{equation}
(o\`u $vx = pF^{-1}x = px^{\sigma^{-1}}$). En particulier, on a (1.4.2.8). Montrons que, si $\psi$ est un rel\`evement quelconque de Frobenius, la condition (1.4.2.7), avec $\varphi$ remplac\'e par $\psi$, est satisfaite. Compte tenu de (1.4.2.10) (avec $\varphi$ remplac\'e par $\psi$), il revient au m\^eme de v\'erifier que l'on a
\begin{equation}\tag{1.4.2.13}
(\psi^*\tau_{ij})(0) - p\tau_{ij}(0) - pu_{ij}(\psi)(0) = 0.
\end{equation}

Si $\psi(0) = 0$ (i.e. $\psi$ est compatible \`a l'augmentation $e : W[[t]] \to W$), on a $(\psi^*\tau_{ij})(0) = \tau_{ij}(0)^\sigma$, et (1.4.2.13) \'equivaut \`a (1.4.2.12) avec $\varphi$ remplac\'e par $\psi$. Il suffit donc de v\'erifier que $u_{ij}(\psi)(0) = u_{ij}(\varphi)(0)$. Or l'augmentation $e$ est le rel\`evement de Teichm\"uller (1.1.4) de l'augmentation $k[[t]] \to k$ simultan\'ement pour $\varphi$ et pour $\psi$, donc $F(\varphi)\varphi^*$ et $F(\psi)\psi^*$ induisent le m\^eme endomorphisme $\sigma$-lin\'eaire $F$ de $e^*H$. Appliquant $e^*$ \`a (1.4.2.4) \'ecrit pour $\varphi$ et pour $\psi$, on obtient donc
\begin{align*}
F(e^*b_i) &= p(e^*b_i) + p\sum_{1\leqslant j\leqslant r} u_{ij}(\varphi)(0)(e^*a_j) \\
&= p(e^*b_i) + p\sum_{1\leqslant j\leqslant r} u_{ij}(\psi)(0)(e^*a_j),
\end{align*}
d'o\`u $u_{ij}(\psi)(0) = u_{ij}(\varphi)(0)$, ce qui prouve (1.4.2.13) dans ce cas.

Si maintenant $\psi$ est un rel\`evement quelconque de Frobenius, on peut \'ecrire
\[
\psi(t_i) = \varphi_0(t_i) + pw_i \quad (1 \leqslant i \leqslant n),
\]
avec $w = (w_1, \ldots, w_n) \in W^n$, o\`u $\varphi_0 : A \to A$ est un rel\`evement de Frobenius tel que $\varphi_0(0) = 0$. Appliquant (1.1.3.4), avec $\varphi = \varphi_0$, $\psi = \psi$, $x = b_i$, on obtient, compte tenu de (1.4.2.4),
\begin{equation*}
\sum_{1\leqslant j\leqslant r} p u_{ij}(\psi)a_j = \sum_{1\leqslant j\leqslant r} p u_{ij}(\varphi_0)a_j + \sum_{|n|>0} p^{[n]}(w^n)F(\varphi_0)\varphi_0^*(\nabla(D)^n b_i). \tag{$\ast$}
\end{equation*}

Calculons $\nabla(D)^n b_i$ : si $D_k = \partial/\partial t_k$, on a, d'apr\`es (1.4.2.2), (1.4.2.6),
\[
\nabla(D_k)b_i = \langle D_k, \sum_j d\eta_{ij} \rangle \otimes a_j = \sum_j (D_k \eta_{ij})a_j,
\]
d'o\`u, comme $\nabla a_j = 0$,
\[
\nabla(D)^n b_i = \sum_j (D^n \eta_{ij})a_j
\]
pour tout multi-exposant $n$ tel que $|n| > 0$. Reportant dans ($\ast$), on obtient
\[
pu_{ij}(\psi) = pu_{ij}(\varphi_0) + \sum_{|n|>0} p^{[n]}(w^n)(D^n \eta_{ij}),
\]
et par suite la formule (1.4.2.13) \`a v\'erifier s'\'ecrit
\begin{equation*}
(\psi^*\tau_{ij})(0) - p\tau_{ij}(0) - pu_{ij}(\varphi_0)(0) - \sum_{|n|>0} p^{[n]}(w^n)(D^n \tau_{ij})(0) = 0. \tag{$\ast\ast$}
\end{equation*}

Notons que $\psi = \varphi_0 \circ \alpha$, o\`u $\alpha$ est l'endomorphisme de $W$-alg\`ebre de $A$ tel que $\alpha(t_i) = t_i + pw_i$, donc que
\[
(\psi^*\tau_{ij})(0) = (\alpha^*\varphi_0^*\tau_{ij})(0) = (\varphi_0^*\tau_{ij})(pw)^{\sigma}.
\]
D'autre part, $\varphi_0$ v\'erifie (1.4.2.13) comme on l'a vu plus haut, car $\varphi_0(0) = 0$, et $(\varphi_0^*\tau_{ij})(0) = \tau_{ij}(0)^\sigma$, donc
\[
p\tau_{ij}(0) = \tau_{ij}(0)^\sigma - pu_{ij}(\varphi_0)(0).
\]
On peut donc r\'ecrire ($\ast\ast$) sous la forme
\[
\tau_{ij}(pw)^\sigma = \tau_{ij}(0)^\sigma + \sum_{|n|>0} p^{[n]}w^n(D^n\tau_{ij})(0)^\sigma,
\]
i.e.
\[
\tau_{ij}(pw) = \tau_{ij}(0) + \sum_{|n|>0} p^{[n]}w^n(D^n\tau_{ij})(0).
\]
Or on reconna\^it ici le d\'eveloppement de Taylor de $\tau_{ij}(pw)$. Cela \'etablit ($\ast\ast$), donc (1.4.2.13) dans tous les cas. Pour d\'emontrer la derni\`ere assertion de 1.4.2, nous aurons besoin du r\'esultat suivant de Dwork :

\begin{lemma}[Dwork {\cite[lemma 1]{Dwork1958}}]\label{lem:1.4.3}
Notons $\varphi$ l'automorphisme $\sigma$-lin\'eaire de $K[[t]]$ (o\`u $t = (t_1, \ldots, t_n)$ et $K$ est le corps des fractions de $W$) donn\'e par $t_i \mapsto t_i^p$. Soit $f \in K[[t]]$ tel que $f(0) = 1$. Les conditions suivantes sont \'equivalentes :
\begin{enumerate}[label=(\roman*)]
\item $f \in W[[t]]$,
\item $\displaystyle\frac{\varphi f}{f^p} = 1 + pu$, avec $u \in W[[t]]$, $u(0) = 0$.
\end{enumerate}
\end{lemma}

L'implication (i) $\Rightarrow$ (ii) est imm\'ediate. Rappelons la d\'emonstration de (ii) $\Rightarrow$ (i). Ecrivons $f = \sum a_i t^i$ ($a_0 = 1$), et supposons prouv\'e que $a_i \in W$ pour $|i| < r$ (o\`u $|i| = i_1 + \cdots + i_n$). Notant $(~)_{<j}$ la partie de degr\'e total $< j$, on calcule
\begin{align*}
(f^p)_{<r} &= \bigl((\sum_{|i|\leqslant r-1} a_i t^i)^p\bigr)_{<r} + (\sum_{|i|=r} a_i t^i)^p \\
&= \bigl(\sum_{|i|\leqslant r-1} a_i^\sigma t^{pi}\bigr)_{<r} + p\sum_{|i|=r} a_i t^i \mod pW[[t]],
\end{align*}
puis, posant $f^p/\varphi f = 1 + p\sum d_i t^i$ ($d_i \in W$),
\begin{multline*}
(\sum_{|i|\geqslant 0} a_i^\sigma t^{pi})(1 + p\sum_{|i|\geqslant 0} d_i t^i)_{<r} \\
= \bigl((\sum_{|i|\leqslant r-1} a_i^\sigma t^{pi})(1 + p\sum_{|i|\geqslant 1} d_i t^i)\bigr)_{<r} \\
= (\sum_{|i|\leqslant r-1} a_i^\sigma t^{pi})_{<r} \mod pW[[t]] \\
= (\sum_{|i|\leqslant r-1} a_i^p t^{pi})_{<r} \mod pW[[t]].
\end{multline*}

Comparant les deux expressions obtenues pour $(f^p)_{<r}$, on trouve
\[
p\sum_{|i|=r} a_i t^i \in pW[[t]],
\]
donc $a_i \in W$ pour tout $i$ tel que $|i| = r$, ce qui prouve (ii) $\Rightarrow$ (i).

Avant de revenir \`a la d\'emonstration de 1.4.2, indiquons deux cons\'equences imm\'ediates de \ref{lem:1.4.3}.

\begin{corollary}\label{cor:1.4.4}
Soit $g \in K[[t]]$ tel que $g(0) = 0$. Les conditions suivantes sont \'equivalentes :
\begin{enumerate}[label=(\roman*)]
\item $\exp(g) \in W[[t]]$,
\item $\varphi^*g - pg \in pW[[t]]$.
\end{enumerate}
\end{corollary}

En effet, posons $f = \exp(g)$. Comme, pour $n > 1$, $p^n/n!$ (et a fortiori $p^n/n$) appartiennent \`a $pW$, la condition (ii) \'equivaut \`a dire que $\varphi f/f^p = 1 + pu$, avec $u \in W[[t]]$, $u(0) = 0$.

\begin{corollary}\label{cor:1.4.5}
Supposons $p \neq 2$. Soit $g \in K[[t]]$, tel que $g(0) \in pW$, et $\varphi^*g - pg \in pW[[t]]$. Alors la s\'erie $f = \exp(g)$ est d\'efinie, appartient \`a $W[[t]]$, et v\'erifie $f(0) \equiv 1 \mod pW$.
\end{corollary}

En effet, soit $h = g - g(0)$. Comme $p \neq 2$ et que $g(0) \in pW$, $\exp(g(0))$ est d\'efini, donc aussi $\exp(g) = \exp(g(0))\exp(h)$. Posons $g(0) = a$ ($a \in W$). On a
\[
\varphi^*h - ph = \varphi^*g - pg - (a^\sigma - pa) \in pW[[t]]
\]
(car $a^\sigma - pa \in pW$ et $\varphi^*g - pg \in pW[[t]]$). Donc, par \ref{cor:1.4.4} appliqu\'e \`a $h$, on a $\exp(h) \in W[[t]]$, et comme $\exp(a) \equiv 1 \mod pW$, on en conclut que $f = \exp(g) \in W[[t]]$ et $f(0) \equiv 1 \mod pW$.

\textbf{Fin de la d\'emonstration de 1.4.2.} Il suffit d'appliquer \ref{cor:1.4.5} \`a $\tau_{ij}$ : les hypoth\`eses de \ref{cor:1.4.5} sont v\'erifi\'ees en vertu de (1.4.2.7) et (1.4.2.8).

\subsubsection{}
Soit $H$ un $F$-cristal de Hodge, comme en 1.3.5, la connexion $\nabla$ induit, gr\^ace \`a (1.3.5.1), un homomorphisme $A$-lin\'eaire
\begin{equation}\tag{1.4.6.1}
\gr\nabla : \gr^i H \longrightarrow \Omega^1_{A/W} \otimes \gr^{i-1} H 
\end{equation}
(o\`u $\gr^i = \Fil^i/\Fil^{i+1}$). En particulier, dans la situation de 1.4.1, $\nabla$ induit un homomorphisme $A$-lin\'eaire
\begin{equation}\tag{1.4.6.2}
\gr\nabla : \Fil^1 H \longrightarrow \Omega^1_{A/W} \otimes U, 
\end{equation}
puisque $U \xrightarrow{\sim} H/\Fil^1 H$ (1.4.1.3), d'o\`u un homomorphisme $A$-lin\'eaire
\begin{equation}\tag{1.4.6.3}
\gr\nabla : T_{A/W} \longrightarrow \Hom_A(\Fil^1 H, U), 
\end{equation}
o\`u $T_{A/W} = (\Omega^1_{A/W})^\vee$.

\begin{corollary}\label{cor:1.4.7}
Sous les hypoth\`eses de 1.4.1, supposons que (1.4.6.3) soit un isomorphisme, et que $p$ soit diff\'erent de $2$. Alors les \'el\'ements $q_{ij} - 1$ ($1 \leqslant i \leqslant s$, $1 \leqslant j \leqslant r$) d\'efinis en (1.4.2.9) forment avec $p$ un syst\`eme r\'egulier de param\`etres de $A = W[[t]]$, i.e. le $W$-homomorphisme $W[[x_{ij}]]_{(1\leqslant i\leqslant s, 1\leqslant j\leqslant r)} \to A$ envoyant $x_{ij}$ sur $q_{ij} - 1$ est un isomorphisme, et si l'on note $\varphi$ le rel\`evement de Frobenius d\'efini par $\varphi(q_{ij}) = q_{ij}^p$, on a
\begin{equation}\tag{1.4.7.1}
F(\varphi)\varphi^*b_i = pb_i 
\end{equation}
pour tout $i$ (avec les notations de 1.4.2).
\end{corollary}

Notons $\mf = (p, t_1, \ldots, t_n)$ l'id\'eal maximal de $A = W[[t]]$. Comme $q_{ij}(0) \equiv 1 \mod pW$, les $q_{ij}$ sont des unit\'es de $A$ congrues \`a $1 \mod \mf$. Pour prouver la premi\`ere assertion, il suffit de montrer que les formes $\eta_{ij} = d\log q_{ij}$ forment une base (sur $A$) de $\Omega^1_{A/W}$.

Tout homomorphisme $f : \Fil^1 H \to U$ est d\'etermin\'e par une matrice $(f_{ij})$ telle que $f(b_i) = \sum_j f_{ij}a_j$. Notons $D_{ij}$ la base de $T_{A/W}$ telle que
\[
(\gr\nabla)(D_{ij})_{k\ell} =
\begin{cases}
0 & \text{si } (k,\ell) \neq (i,j), \\
1 & \text{si } (k,\ell) = (i,j).
\end{cases}
\]
Pour tout $D \in T_{A/W}$, on a, d'apr\`es (1.4.2.2),
\[
(\gr\nabla)(D)b_i = \sum_j \eta_{ij}(D)a_j,
\]
donc
\[
(\gr\nabla)(D_{ij})_{k\ell} = \eta_{k\ell}(D_{ij}).
\]
Les $\eta_{ij}$ forment donc une base de $\Omega^1_{A/W}$, \`a savoir la base duale de la base $D_{ij}$. D'autre part, la relation $\varphi^*q_{ij} = q_{ij}^p$ entra\^ine $\varphi^*\log q_{ij} = p\log q_{ij}$, mais comme $\log q_{ij} = \tau_{ij}$, (1.4.2.7) entra\^ine $u_{ij}(\varphi) = 0$, donc (1.4.7.1) r\'esulte de (1.4.2.4).

\subsubsection{}
Soit, comme en 1.3.6, $(H, \Fil^\bullet H)$ un $F$-cristal de Hodge sur $A$ ($= k[[t]]$) tel que $H$ soit ordinaire. Supposons $H$ de niveau $\leqslant N$, i.e. (1.3.1) $\Fil^i H = 0$ pour $i \geqslant N+1$, de sorte que la d\'ecomposition (1.3.6.1) s'\'ecrit
\begin{equation}\tag{1.4.8.1}
H = \bigoplus_i H^i, \quad H^i = U_i \cap \Fil^i H, \quad \rg(H^i) = h_i. 
\end{equation}
Supposons d'autre part $p \neq 2$. Appliquant 1.4.2 aux $F$-cristaux de Hodge ordinaires (de niveau $\leqslant 1$) $(U_{i+2}/U_i, (\Fil^\bullet H \cap U_{i+2})/(\Fil^\bullet H \cap U_i))$, on obtient :

\begin{corollary}\label{cor:1.4.9}
Sous les hypoth\`eses de 1.4.8, il existe une base $(e^i_1, \ldots, e^i_{h_i})_{0 \leqslant i \leqslant N}$ de $H$, et des \'el\'ements $q_{i;\alpha,\beta}$ de $A$ tels que
\begin{equation}\tag{1.4.9.1}
\begin{cases}
q_{i;\alpha,\beta}(0) \in 1 + pW, \\[1ex]
\nabla e^0_1 = 0 \quad (1 \leqslant i \leqslant h_0), \\[1ex]
\displaystyle\nabla e^m_\alpha = \sum_{1 \leqslant j \leqslant h_{m-1}} (d\log q_{m-1;\alpha,j}) \otimes e^{m-1}_j \quad (0 < m \leqslant N, \ 1 \leqslant \alpha \leqslant h_m).
\end{cases} 
\end{equation}
\end{corollary}

On peut utiliser \ref{cor:1.4.9} pour majorer la croissance des sections horizontales de $H$. Rappelons \cite[3.1]{Katz1973} que celles-ci ``convergent dans le polydisque unit\'e ouvert'', et plus pr\'ecis\'ement que le module $(H \otimes_A K\{\{t_1, \ldots, t_n\}\}, \nabla)$ poss\`ede une base de sections horizontales : il s'agit d'une propri\'et\'e valable pour tout $F$-cristal, ind\'ependamment de toute hypoth\`ese d'ordinarit\'e. Cela dit, il r\'esulte ais\'ement de \ref{cor:1.4.9} que, pour tout $i$, les sections horizontales de $U_i \otimes_A K\{\{t\}\}$ appartiennent \`a $U_i \otimes_A L_i$, o\`u
\[
0 = L_{-1} \subset L_0 \subset L_1 \subset \cdots \subset L_i \subset \cdots
\]
est la suite de sous-$A$-modules de $K\{\{t\}\}$ d\'efinie par r\'ecurrence par
\[
(f \in L_i) \Longleftrightarrow (f \text{ appartient au sous-$A$-module de } K\{\{t\}\} \text{ engendr\'e par les s\'eries } g \text{ telles que } dg = \sum a_\nu d\log u_\nu,
\]
avec $a_\nu \in L_{i-1}$, et $u_\nu \in A$, $u_\nu(0) \in 1 + pW$).

Dans la situation de 1.4.8, on ne peut esp\'erer toutefois d'\'enonc\'e analogue \`a 1.4.7 (l'hypoth\`ese que (1.4.6.3) est un isomorphisme n'a pas de g\'en\'eralisation en niveau $\geqslant 2$).

\section{Applications g\'eom\'etriques}

On conserve les notations $A$, $A^\circ$ de 1.1.1.

\subsection{Coordonn\'ees canoniques}

\subsubsection{A) Vari\'et\'es ab\'eliennes} (cf.\ \cite[\S 8]{Katz1973}).

\setcounter{subsubsection}{0}
\renewcommand{\thesubsubsection}{\thesubsection.\arabic{subsubsection}}

\subsubsection{}
Soit $X/A$ un sch\'ema ab\'elien formel de dimension relative $g$. Le $A$-module
\[
H = H^1_\mathrm{dR}(X/A)
\]
muni de la connexion de Gauss-Manin $\nabla$, est un $F$-cristal sur $A^\circ$, de rang $2g$. On sait (voir par exemple \cite[Addendum]{MazurMessing}) que la suite spectrale de Hodge
\[
E^{i,j}_1 = H^j(X, \Omega^i_{X/A}) \Longrightarrow H^\bullet_\mathrm{dR}(X/A)
\]
d\'eg\'en\`ere en $E_1$, et que le terme $E^{i,j}_1$ est libre sur $A$, de formation compatible \`a tout changement de base. Il en r\'esulte (1.3.5) que $H$, muni de la filtration de Hodge
\[
H = \Fil^0 H \supset \Fil^1 H \quad (= H^0(X, \Omega^1_{X/A})) \supset 0
\]
est un $F$-cristal de Hodge sur $A^\circ$. Notons aussi que, si $X_0 = X \otimes_A A^\circ$, la suite spectrale de Hodge et la suite spectrale conjugu\'ee de $X_0/A^\circ$ d\'eg\'en\`erent (en $E_1$ et $E_2$ resp.) et que leurs termes initiaux sont libres et de formation compatible \`a tout changement de base. En particulier, le gradu\'e associ\'e \`a la filtration conjugu\'ee $\gr_\bullet(H \otimes_A A^\circ)$ est libre. D'autre part, si $e^\circ : A^\circ \to k$ est l'augmentation, le $F$-cristal $e^\circ_0 H$ induit sur $k$ (1.1.4) n'est autre que $H^1(X \otimes k/W)$, car si $e : A \to W$ est l'augmentation, $e^*H = H^1_\mathrm{dR}(X \otimes W/W) = H^1(X \otimes k/W)$ par d\'efinition de la cohomologie cristalline.

\subsubsection{}
Supposons la vari\'et\'e ab\'elienne $X \otimes k$ ordinaire, ce qui signifie, au choix, que $(X \otimes k)(k) \supset (\mathbb{Z}/p\mathbb{Z})^g$, ou que Frobenius sur $H^1(X \otimes k, \mathcal{O})$ est bijectif, ou que $H^1(X \otimes k/W)$ a m\^emes polygones de Newton et de Hodge. D'apr\`es ce qu'on vient de rappeler, le $F$-cristal $H$ est alors ordinaire au sens de 1.3.3, et comme il est de Hodge de niveau $\leqslant 1$, on peut lui appliquer 1.4.2. Notons que, dans la d\'ecomposition $H = U \oplus \Fil^1 H$, $U$ et $\Fil^1 H$ sont libres de rang $g$. Il existe donc une base $(a_i)_{1\leqslant i\leqslant g}$ de $U$, une base $(b_i)_{1\leqslant i\leqslant g}$ de $\Fil^1 H$, et des s\'eries $\tau_{ij} \in K[[t]]$ ($1\leqslant i,j\leqslant g$, $1\leqslant j\leqslant g$) v\'erifiant les conditions (1.4.2.1) \`a (1.4.2.8). De plus, si $p \neq 2$, on dispose des s\'eries $q_{ij} = \exp(\tau_{ij}) \in A$, telles que $q_{ij}(0) \equiv 1 \mod pW$.

\subsubsection{}
Supposons que $X/A$ soit la d\'eformation formelle verselle de $X \otimes k$, donc que $A = W[[t_1, \ldots, t_{g^2}]]$. Alors (1.4.6.3) est un isomorphisme, qui s'identifie \`a l'isomorphisme de Kodaira-Spencer
\[
T_{A/W} \xrightarrow{\sim} H^1(X, T_{X/A}) \xrightarrow{\sim} \Hom(H^0(X, \Omega^1_{X/A}), H^1(X, \mathcal{O}))
\]
(cf.\ (IV 2.3)). Donc, si $p \neq 2$, les \'el\'ements $q_{ij} - 1 \in A$ forment, en vertu de \ref{cor:1.4.7}, des coordonn\'ees ``canoniques'' sur $A$ (i.e. $A \xrightarrow{\sim} W[[q_{ij} - 1]]$), et, si l'on note $\varphi$ le rel\`evement de Frobenius donn\'e par $\varphi(q_{ij}) = q_{ij}^p$, on a, avec les notations de 2.1.2,
\[
F(\varphi)\varphi^*a_i = a_i, \quad F(\varphi)\varphi^*b_i = pb_i \quad (1 \leqslant i \leqslant g).
\]
Soit $e : A \to W$ le rel\`evement de Teichm\"uller de $e^\circ$ correspondant \`a $\varphi$, i.e. tel que $e(q_{ij}) = 1$. Il n'est pas difficile de montrer --- nous \'etablirons un r\'esultat analogue ci-dessous pour les surfaces K3 --- que $e$ est ind\'ependant du choix de $(a, b, q)$, et qu'on a un isomorphisme canonique (d\'efini \`a l'aide de $(a, b, q)$ mais n'en d\'ependant pas) entre $S$ et $(\mathbb{G}_m^{\wedge g^2})_W$, o\`u $\mathbb{G}_m^{\wedge}$ est le tore formel sur $\mathbb{Z}_p$ de groupe de caract\`eres $\Hom_{\mathbb{Z}_p}(P_0, P_1)$, $P_0$ (resp.\ $P_1$) d\'esignant le sous-$\mathbb{Z}_p$-module (libre de rang $g$) de $H^1(X \otimes k/W)$ form\'e des $x$ tels que $Fx = x$ (resp.\ $Fx = px$). Cet isomorphisme envoie $e$ sur l'\'el\'ement neutre de $(\mathbb{G}_m^{\wedge g^2})_W$. On peut montrer (voir Appendice et Expos\'e suivant, par N.~Katz) que cette structure de groupe formel sur $S$ co\^incide avec celle d\'efinie, \`a la Serre-Tate, par le rel\`evement formel versel du groupe $p$-divisible associ\'e \`a $X \otimes k$. En particulier, la vari\'et\'e ab\'elienne $e^*X/W$ est le rel\`evement canonique de $(X \otimes k)/k$ \cite[\S V \S 3]{Messing}. Rappelons que ce rel\`evement correspond au rel\`evement trivial (i.e. comme produit) du groupe $p$-divisible associ\'e \`a $X \otimes k$, ou, ce qui revient au m\^eme, au rel\`evement de $\Fil^1 H^1_\mathrm{dR}(X \otimes k/k)$ en le facteur direct $P_1 \otimes W$ de pente $1$ de $H^1(X \otimes k/W)$ (cf.\ 1.3.4).

\subsubsection{Variante.}
Soit $X/A$ une courbe propre et lisse \`a fibres g\'eom\'etriquement connexes de genre $g > 1$. Alors $H = H^1_\mathrm{dR}(X/A)$, muni de la connexion de Gauss-Manin $\nabla$ et de la filtration de Hodge $\Fil^1 H = H^0(X, \Omega^1_{X/A})$, est un $F$-cristal de Hodge de niveau $1$. Si l'on suppose $X \otimes k$ ordinaire, on peut appliquer 1.4.2, et l'on obtient les r\'esultats de Katz \cite[\S 8]{Katz1973} : si $(a_i)_{1\leqslant i\leqslant g}$ est une base de sections horizontales fixes par $F$ du sous-cristal unit\'e $U$, on peut en effet prendre comme base $(b_i)_{1\leqslant i\leqslant g}$ de $\Fil^1 H$ la base duale de $(a_i)$ pour la dualit\'e de Poincar\'e ($(a, b)$ est alors une base symplectique de $H$), car la compatibilit\'e de la dualit\'e \`a $F$ et \`a $\nabla$ montre aussit\^ot que les conditions de (1.2.4 (i)) sont v\'erifi\'ees.

\subsubsection{B) Surfaces K3}

\setcounter{subsubsection}{4}

\subsubsection{}
Soit $X/A$ un sch\'ema formel propre et lisse tel que $X_k = X \otimes k$ soit une surface K3. Alors, d'apr\`es (IV 2.2, 2.3), le $A$-module
\[
H = H^2_\mathrm{dR}(X/A),
\]
muni de la connexion de Gauss-Manin $\nabla$, et de la filtration de Hodge
\[
H = \Fil^0 H \supset \Fil^1 H \supset \Fil^2 H \supset 0
\]
est un $F$-cristal de Hodge sur $A^\circ$. Les sous-modules $\Fil^i H$ sont libres de rangs $22$, $21$, $1$ pour $i = 0, 1, 2$, et le gradu\'e associ\'e $\gr^\bullet H$ est libre. De plus, le cup-produit
\[
\langle \cdot, \cdot \rangle : H \otimes H \longrightarrow H^4_\mathrm{dR}(X/A) \cong A
\]
est une dualit\'e parfaite, horizontale pour $\nabla$, i.e. v\'erifiant
\[
\langle \nabla x, y \rangle + \langle x, \nabla y \rangle = \nabla\langle x, y \rangle,
\]
et compatible \`a $F$, i.e. telle que, pour tout rel\`evement de Frobenius $\varphi$, on ait
\[
\langle F(\varphi)\varphi^* x, F(\varphi)\varphi^* y \rangle = F(\varphi)\varphi^*\langle x, y \rangle.
\]
Noter que $F(\varphi)\varphi^* H^4_\mathrm{dR}(X/A) = p^2 \eta$, o\`u $\eta$ est un automorphisme, i.e. $H^4_\mathrm{dR}(X/A)(2)$ est un $F$-cristal unit\'e (de rang $1$), qu'on identifiera au $F$-cristal trivial $A$ au moyen d'une base horizontale fixe par $F$. Rappelons d'autre part que l'on a
\[
\Fil^1 H = (\Fil^2 H)^\perp.
\]
Rappelons enfin que, si $e^\circ : A^\circ \to k$ est l'augmentation, le $F$-cristal $e^\circ_0 H$ (1.1.4) n'est autre que la cohomologie cristalline $H^2(X_k/W)$.

\subsubsection{}
Supposons maintenant $X_k$ ordinaire, ce qui signifie, par d\'efinition, que le polygone de Newton de $H^2(X_k/W)$ co\'incide avec le polygone de Hodge, i.e. a pour pentes $(0, 1, 2)$ avec les multiplicit\'es $(1, 20, 1)$. Il revient au m\^eme de dire que Frobenius sur $H^2(X_k, \mathcal{O})$ est non nul. Du fait que $e^\circ_0 H = H^2(X_k/W)$ et que $\gr^\bullet H$ est libre, le $F$-cristal $H$ est alors ordinaire au sens de 1.3.3, et admet par cons\'equent une filtration par des sous-$F$-cristaux
\[
0 \subset U_0 \subset U_1 \subset U_2 = H
\]
tels que $U_i/U_{i-1}$ soit de la forme $V_i(-i)$, o\`u $V_i$ est un $F$-cristal unit\'e de rang $1$, $20$, $1$, pour $i = 0, 1, 2$. De plus, comme $U_i$ rel\`eve $\Fil_i H^\circ$, la compatibilit\'e de $F$ \`a la dualit\'e entra\^ine que l'on a $U_1 = U_0^\perp$. Enfin (1.3.6) les filtrations $(U_i)$ et $(\Fil^i H)$ sont oppos\'ees, i.e. on a
\[
H = H^0 \oplus H^1 \oplus H^2,
\]
avec
\[
H^0 = U_0, \quad H^1 = U_1 \cap \Fil^1 H, \quad H^2 = \Fil^2 H, \quad H^1 \xrightarrow{\sim} U_1/U_0.
\]

\begin{theorem}\label{thm:2.1.7}
On suppose $p \neq 2$. Soit $X/A = W[[t_1, \ldots, t_{20}]]$ la d\'eformation formelle universelle d'une surface K3 ordinaire $X_k/k$ (IV 1.3). Alors, avec les notations de 2.1.5, 2.1.6, il existe une base $(a, b_1, \ldots, b_{20}, c)$ de $H$ adapt\'ee \`a la d\'ecomposition $H = H^0 \oplus H^1 \oplus H^2$, et telle que $\langle a, c \rangle = 1$, et des \'el\'ements $q_i \in A$ ($1 \leqslant i \leqslant 20)$ poss\'edant les propri\'et\'es suivantes :
\begin{enumerate}[label=(\roman*)]
\item $q_i(0) \equiv 1 \mod pW$ et les $q_i$ d\'efinissent un isomorphisme $A \xrightarrow{\sim} W[[q_1 - 1, \ldots, q_{20} - 1]]$ (i.e. les formes $d\log q_i$ forment une base de $\Omega^1_{A/W}$).
\item $\nabla a = 0$,
\[
\nabla b_i = (d\log q_i) \otimes a \quad (1 \leqslant i \leqslant 20),
\]
\[
\nabla c = -\sum_{1\leqslant i\leqslant 20} (d\log q_i) \otimes b_i^\vee
\]
o\`u $(b_i^\vee)$ d\'esigne la base de $H^1$ duale de $(b_i)$ (pour la restriction \`a $H^1$ de la forme cup-produit).
\item Si $\varphi : A \to A$ d\'esigne le rel\`evement de Frobenius tel que $\varphi(q_i) = q_i^p$, alors
\begin{align*}
F(\varphi)\varphi^* a &= a, \\
F(\varphi)\varphi^* b_i &= pb_i \quad (1 \leqslant i \leqslant 20), \\
F(\varphi)\varphi^* c &= p^2 c.
\end{align*}
\end{enumerate}
\end{theorem}

D'apr\`es 2.1.6, le $F$-cristal $U_1$, muni de la filtration $\Fil^\bullet H \cap U_1$, est un $F$-cristal de Hodge ordinaire de niveau $1$. D'autre part, comme $X/A$ est la d\'eformation formelle universelle de $X_k$, l'application
\[
\gr\nabla : T_{A/W} \longrightarrow \Hom_A(H^1, H^0) = \Hom_A(\gr^1 H, \gr^0 H)
\]
est un isomorphisme d'apr\`es (IV 2.4). En vertu de 1.4.2 et \ref{cor:1.4.7} appliqu\'es \`a $U_1$, il existe donc une base $(a, b_1, \ldots, b_{20})$ de $U_1$ adapt\'ee \`a la d\'ecomposition $U_1 = H^0 \oplus H^1$, et des \'el\'ements $q_i \in A$ v\'erifiant les conditions (i) \`a (iii) de \ref{thm:2.1.7}. Il reste \`a montrer que, si l'on choisit la base $c$ de $H^2$ telle que $\langle a, c \rangle = 1$, les derni\`eres formules de (ii) et (iii) sont satisfaites. Tout d'abord, par la condition de transversalit\'e (1.3.5.1), on a
\[
\nabla c = \alpha \otimes c + \sum_i \beta_i \otimes b_i.
\]
La relation
\[
0 = \nabla\langle a, c \rangle = \langle \nabla a, c \rangle + \langle a, \nabla c \rangle = \langle a, \nabla c \rangle
\]
entra\^ine $\alpha = 0$. Puis, de
\[
0 = \nabla\langle b_i, c \rangle = \langle \nabla b_i, c \rangle + \langle b_i, \nabla c \rangle
\]
et de la deuxi\`eme formule de (ii) on tire $\beta_i = -d\log q_i$, d'o\`u la derni\`ere formule de (ii). D'autre part, de la formule $F(\varphi)\varphi^* a = a$ et de la relation $\langle F(\varphi)\varphi^* a, F(\varphi)\varphi^* c \rangle = p^2\langle a, c \rangle = p^2$ on d\'eduit $F(\varphi)\varphi^* c = p^2 c$, ce qui ach\`eve la d\'emonstration de \ref{thm:2.1.7}.

\subsubsection{}
Soit $X_k/k$ une surface K3 ordinaire. Posons
\begin{equation}\tag{2.1.8.1}
P_0 = H^2(X_k/W)^{F=1}, \quad P_1 = H^2(X_k/W)^{F=p}, \quad P_2 = H^2(X_k/W)^{F=p^2}, 
\end{equation}
o\`u la notation $M^{F=\alpha}$ d\'esigne $\{x \in M \mid Fx = \alpha x\}$. D'apr\`es 1.3.4, $P_i$ est un $\mathbb{Z}_p$-module libre de rang $1$, $20$, $1$ pour $i = 0, 1, 2$, et le $F$-cristal $H^2(X_k/W)$ admet une d\'ecomposition unique
\begin{equation}\tag{2.1.8.2}
H^2(X_k/W) = ((W, \sigma) \otimes P_0) \oplus ((W, p\sigma) \otimes P_1) \oplus ((W, p^2\sigma) \otimes P_2). 
\end{equation}
Supposons de plus $p \neq 2$. Avec les notations de \ref{thm:2.1.7}, d\'esignons par
\begin{equation}\tag{2.1.8.3}
e_q : A \longrightarrow W 
\end{equation}
le rel\`evement de Teichm\"uller de l'augmentation $e^\circ : A^\circ \to k$ correspondant au rel\`evement $\varphi$ consid\'er\'e en (iii), donc donn\'e par $e_q(q_i) = 1$. Alors $X$ induit un sch\'ema formel $e_q^*X/W$ et l'on a
\begin{equation}\tag{2.1.8.4}
e_q^* H = H^2_\mathrm{dR}(e_q^*X/W) = H^2(X_k/W). 
\end{equation}
De plus, d'apr\`es (\ref{thm:2.1.7} (iii)), $e_q^* a$ est une base de $P_0$, $(e_q^* b_i)_{1\leqslant i\leqslant 20}$ une base de $P_1$, et $e_q^* c$ une base de $P_2$, et $\langle e_q^* a, e_q^* c \rangle = 1$. En particulier, la filtration de Hodge de $H^2_\mathrm{dR}(e_q^*X/W)$ est donn\'ee par
\begin{equation}\tag{2.1.8.5}
\Fil^1 H^2_\mathrm{dR}(e_q^*X/W) = (W \otimes P_1) \oplus (W \otimes P_2), \quad \Fil^2 H^2_\mathrm{dR}(e_q^*X/W) = W \otimes P_2. 
\end{equation}

Nous allons en d\'eduire :

\begin{proposition}\label{prop:2.1.9}
Sous les hypoth\`eses de \ref{thm:2.1.7}, le rel\`evement $e_q$ (2.1.8.3) est ind\'ependant de la famille $q = (q_i)$ d\'efinie en \ref{thm:2.1.7}.
\end{proposition}

On \'ecrira donc $e$ au lieu de $e_q$. Par analogie avec la th\'eorie du rel\`evement canonique des vari\'et\'es ab\'eliennes ordinaires \cite[\S V \S 3]{Messing}, on dira que le sch\'ema formel $e^*X/W$, qu'on notera simplement $X_W$, est le \emph{rel\`evement canonique} de $X_k$.

\subsubsection{}
Pour d\'emontrer \ref{prop:2.1.9}, nous utiliserons une description ``lin\'eaire'' des rel\`evements formels sur $W$ d'une surface K3 sur $k$.

Soit $Y/W$ un sch\'ema formel propre et plat tel que $Y_k = Y \otimes k$ soit une K3. Alors $H^2_\mathrm{dR}(Y/W)$ s'identifie canoniquement \`a $H = H^2(Y_k/W)$, et $\Fil^2 H^2_\mathrm{dR}(Y/W)$ est une droite (donc facteur direct de rang $1$) de $H$, isotrope (i.e. contenue dans son orthogonal), relevant $\Fil^2 H^2_\mathrm{dR}(Y_k/k)$.

\begin{theorem}\label{thm:2.1.11}
Supposons $p \neq 2$. Soit $Y_0$ une surface K3 sur $k$. Posons $H = H^2(Y_0/W)$. L'application qui \`a un sch\'ema formel propre et plat $Y/W$ relevant $Y_0$ associe la droite $\Fil^2 H^2_\mathrm{dR}(Y/W)$ est une bijection de l'ensemble des rel\`evements de $Y_0$ sur $W$ (i.e. des points \`a valeurs dans $W$ du sch\'ema modulaire formel de $Y_0$) sur l'ensemble des rel\`evements de $\Fil^2 H^2_\mathrm{dR}(Y_0/k)$ en une droite isotrope de $H$.
\end{theorem}

Notons tout d'abord que les deux types de rel\`evements envisag\'es (rel\`evements de $Y_0$ et rel\`evements de $\Fil^2 H^2_\mathrm{dR}(Y_0/k)$) sont rigides, i.e. n'ont pas d'automorphismes infinit\'esimaux. Soit $Y_n$ un rel\`evement de $Y_0$ sur $W_{n+1}$. L'ensemble des rel\`evements de $Y_n$ sur $W_{n+2}$ est un torseur sous $H^1(Y_0, T_{Y_0/k})$. D'autre part, soit $L_n$ une droite isotrope de $H \otimes W_{n+1}$ relevant $L_0 = \Fil^2 H^2_\mathrm{dR}(Y_0/k)$. Alors l'ensemble des rel\`evements de $L_n$ en une droite isotrope de $H \otimes W_{n+2}$ est un torseur sous $\Hom(L_0, L_0^\perp/L_0)$ : sans condition d'isotropie, on trouverait un torseur sous $\Hom(L_0, H_0/L_0)$ (o\`u $H_0 = H \otimes k = H^2_\mathrm{dR}(Y_0/k)$) ; comme $p \neq 2$, la condition d'isotropie impos\'ee au rel\`evement conduit au r\'esultat indiqu\'e (si $H \otimes W_{n+2} = L_{n+1} \oplus M_{n+1}$, avec $L_{n+1}$ isotrope relevant $L_n$, une droite isotrope relevant $L_n$ est engendr\'ee par un vecteur $x + p^{n+1}y$, o\`u $x$ est une base de $L_n$, et l'on doit avoir $2\langle x, y \rangle = 0 \mod p$). Or $L_0^\perp = \Fil^1 H^2_\mathrm{dR}(Y_0/k)$, donc
\[
\Hom(L_0, L_0^\perp/L_0) = \Hom(\Fil^2 H^2_\mathrm{dR}(Y_0/k), \gr^1 H^2_\mathrm{dR}(Y_0/k)),
\]
et, d'apr\`es (IV 2.4), on a un isomorphisme canonique (donn\'e par le cup-produit)
\[
H^1(Y_0, T_{Y_0/k}) \xrightarrow{\sim} \Hom(\Fil^2 H^2_\mathrm{dR}(Y_0/k), \gr^1 H^2_\mathrm{dR}(Y_0/k)).
\]
La conclusion de \ref{thm:2.1.11} en d\'ecoule facilement.

\begin{remark}
On notera l'analogie de \ref{thm:2.1.11} avec la th\'eorie des rel\`evements des groupes $p$-divisibles et des vari\'et\'es ab\'eliennes (rel\`evement de la filtration de Hodge dans le ``cristal'' extension universelle) \cite{Messing}.
\end{remark}

\textbf{D\'emonstration de \ref{prop:2.1.9}.} Compte tenu de \ref{thm:2.1.11}, il suffit d'observer que $e_q^*X$, donc $e_q^* H = H^2_\mathrm{dR}(e_q^*X/W)$, est enti\`erement caract\'eris\'e par la seconde \'egalit\'e de (2.1.8.5).

Nous allons voir maintenant qu'\`a l'aide de \ref{thm:2.1.7} on peut munir canoniquement la vari\'et\'e modulaire formelle $S = \Spf(A)$ d'une structure de groupe formel sur $W$, d'origine le rel\`evement canonique $X_W$ (\ref{prop:2.1.9}). Nous aurons besoin pour cela du lemme suivant :

\begin{lemma}\label{lem:2.1.13}
Soient $(a', b'_1, \ldots, b'_{20}, c')$ une base de $H$ et $(q'_i)_{1\leqslant i\leqslant 20}$ une famille d'\'el\'ements de $A$ v\'erifiant les m\^emes conditions (i), (ii), (iii) de \ref{thm:2.1.7} que $(a, (b_i), c)$ et $(q_i)$. Il existe alors $\lambda \in \mathbb{Z}_p^*$ et $\theta = (\theta_{ij}) \in \GL(20, \mathbb{Z}_p)$ tels que
\begin{equation}\tag{2.1.13.1}
\begin{pmatrix}
a' \\ c' \\ b'_i
\end{pmatrix}
=
\begin{pmatrix}
\lambda a \\ \lambda^{-1} c \\ \displaystyle\sum_{1\leqslant j\leqslant 20} \theta_{ji} b_j
\end{pmatrix},
\quad
q'_i = \prod_j q_j^{\theta_{ji}/\lambda}. 
\end{equation}
\end{lemma}

Tout d'abord, comme $(a', (b'_i), c')$ est une base de $H$ adapt\'ee \`a la d\'ecomposition $H = H^0 \oplus H^1 \oplus H^2$ et que $\langle a', c' \rangle = 1$, il existe $\lambda \in A^*$ et $\theta = (\theta_{ij}) \in \GL(20, A)$ tels que
\[
a' = \lambda a, \quad c' = \lambda^{-1} c, \quad b'_i = \sum_{1\leqslant j\leqslant 20} \theta_{ji} b_j.
\]
Gr\^ace \`a \ref{prop:2.1.9}, notons $e = e_q = e_{q'} : A \to W$ le rel\`evement (2.1.8.3). Comme $\nabla a = \nabla a' = 0$, on a $d\lambda = 0$, donc $\lambda$ est ``constant'', de valeur $e^*\lambda \in W$. Soit $\varphi'$ le rel\`evement de Frobenius tel que $\varphi'(q'_i) = q'_i{}^p$. On a
\[
a' = F(\varphi')\varphi'^* a' = F(\varphi')\varphi'^*(\lambda a) = \lambda^\sigma F(\varphi')\varphi'^* a = \lambda^\sigma a,
\]
donc $\lambda^\sigma = \lambda$, i.e. $\lambda/\lambda^\sigma \in \mathbb{Z}_p$. D'autre part,
\[
\nabla b'_i = \sum_j (d\theta_{ji}) \otimes b_j + \sum_j \theta_{ji}(d\log q_j) \otimes a = (d\log q'_i) \otimes a' = (d\log q'_i) \otimes \lambda a,
\]
donc
\begin{align*}
d\theta_{ji} &= 0 \quad \forall i, j \tag{$\ast$} \\
\intertext{et}
\lambda \, d\log q'_i &= \sum_j \theta_{ji} \, d\log q_j \quad (1 \leqslant i \leqslant 20). \tag{$\ast\ast$}
\end{align*}
La formule ($\ast$) signifie que $\theta_{ji}$ est constant, de valeur $e^*\theta_{ji} \in W$. Mais
\[
F(\varphi')\varphi'^* b'_i = \sum_j \theta_{ji}^\sigma F(\varphi')\varphi'^* b_j = \sum_j \theta_{ji}^\sigma \cdot pb_j.
\]
Appliquant $e^*$ et tenant compte de ce que $e^*F(\varphi')\varphi'^* b_j = e^*F(\varphi)\varphi^* b_j = e^*(pb_j)$, on obtient $\theta_{ji}^\sigma = \theta_{ji}$ $\forall i, j$, donc $\theta \in \GL(20, \mathbb{Z}_p)$. De ($\ast\ast$) on d\'eduit qu'il existe $C \in W$ tel que
\[
q'_i = C \prod_j q_j^{\theta_{ji}/\lambda}.
\]
Mais comme $e^* q_i = e^* q'_i = 1$, on a $C = 1$, ce qui ach\`eve la d\'emonstration.

\begin{theorem}\label{thm:2.1.14}
Supposons $p \neq 2$. Soient $X_k$ une surface K3 ordinaire sur $k$, et $S = \Spf(A)$ la vari\'et\'e modulaire formelle sur $W$ correspondante. Soit $\mathbb{G}_m^{\wedge}$ le tore formel sur $\mathbb{Z}_p$ de groupe de caract\`eres $\Hom_{\mathbb{Z}_p}(P_0, P_1)$, avec les notations de 2.1.8. Il existe un isomorphisme canonique entre $S$ et $(\mathbb{G}_m^{\wedge 20})_W$, par lequel le rel\`evement canonique $X_W \in S$ (\ref{prop:2.1.9}) correspond \`a l'\'el\'ement neutre de $(\mathbb{G}_m^{\wedge 20})_W$.
\end{theorem}

Soit $\alpha = (a, b, (q_i))$ comme en \ref{thm:2.1.7}. La famille $(q_i)$ fournit un isomorphisme $u_\alpha : S \xrightarrow{\sim} (\hat{\mathbb{G}}_m)^{20}_W = \Spf(W[T_i - 1])$, $T_i \mapsto q_i$, tandis que la base $(a, b)$ fournit un isomorphisme $v_\alpha : \mathbb{G}_m^{\wedge 20} \xrightarrow{\sim} \Hom_{\ldots}(P_0, P_1)$, tel que $v_\alpha(x_1, \ldots, x_{20})(a) = \sum x_i b_i$. Si $\alpha' = (a', b', (q'_i))$ est un autre choix, alors, il existe $\lambda \in \mathbb{Z}_p^*$ et $\theta = (\theta_{ij}) \in \GL(20, \mathbb{Z}_p)$ v\'erifiant (2.1.13.1). Notons
\[
f : (\hat{\mathbb{G}}_m)^{20} \longrightarrow (\hat{\mathbb{G}}_m)^{20}
\]
l'isomorphisme donn\'e par $T_i \mapsto \prod_j T_j^{\theta_{ji}/\lambda}$. L'isomorphisme $\Hom(f, \mathbb{G}_m^{\wedge}) : \mathbb{Z}_p^{20} \to \mathbb{Z}_p^{20}$ induit par $f$ sur les groupes de caract\`eres est la multiplication par la matrice $\theta/\lambda$. Les formules (2.1.13.1) entra\^inent qu'on a des carr\'es commutatifs
\[
\begin{CD}
S @>{u_\alpha}>> (\hat{\mathbb{G}}_m)^{20} @>{v_\alpha}>> \Hom_{\mathbb{Z}_p}(P_0, P_1) \\
@| @V{f}VV @VV{\Hom(f, \mathbb{G}_m^{\wedge})}V \\
S @>{u_{\alpha'}}>> (\hat{\mathbb{G}}_m)^{20} @>{v_{\alpha'}}>> \Hom_{\mathbb{Z}_p}(P_0, P_1)
\end{CD}
\]
En d'autres termes, l'isomorphisme $S \xrightarrow{\sim} (\mathbb{G}_m^{\wedge 20})_W$ donn\'e par $(u_\alpha, v_\alpha)$ est ind\'ependant de $\alpha$, c'est l'isomorphisme canonique annonc\'e. Comme par d\'efinition $X_W$ correspond \`a l'augmentation $q_i \mapsto 1$, \ref{thm:2.1.14} est d\'emontr\'e.

\begin{definition}\label{def:2.1.15}
Nous dirons qu'un syst\`eme $\alpha = (a, b, c, q)$ v\'erifiant les conditions de \ref{thm:2.1.7} est un \emph{syst\`eme de coordonn\'ees} (ou \emph{param\`etres}) canoniques sur $S$.
\end{definition}

Noter que le choix de $a$ correspond \`a celui d'une base de $P_0$ (donc est d\'efini \`a un facteur $\in \mathbb{Z}_p$ pr\`es) et d\'etermine celui de $c$, et que, d'autre part, une fois $a$ choisi, la donn\'ee de $b$ correspond \`a celle d'une base de $P_1$ (qui est donc d\'efinie \`a un facteur $\GL(20, \mathbb{Z}_p)$ pr\`es) et d\'etermine l'isomorphisme $(q_i) : S \xrightarrow{\sim} (\hat{\mathbb{G}}_m)^{20}$.

\textbf{Probl\`emes 2.1.16.}
\begin{enumerate}[label=\alph*)]
\item Peut-on donner une d\'efinition ``intrins\`eque'' de la structure de groupe formel sur $S$ construite en \ref{thm:2.1.14}, par exemple comme repr\'esentant un foncteur convenable ?\footnote{(ajout\'ee en janvier 1981), cette question vient d'\^etre r\'esolue affirmativement par Nygaard.}
\item Etendre la th\'eorie des coordonn\'ees canoniques au cas $p = 2$. Cela pr\'esente plusieurs difficult\'es. Tout d'abord, bien entendu, trouver un substitut ad\'equat \`a la d\'efinition des $q_{ij}$ de 1.4.2. D'autre part, dans le cas des K3, il y a des difficult\'es suppl\'ementaires, li\'ees au fait qu'on ignore, pour $p = 2$, si la forme quadratique $\langle x, x \rangle$ sur $H^2_\mathrm{dR}(X_k/k)$ est identiquement nulle (rappelons que, si $X$ est une surface K3 sur le corps des complexes, la forme $\langle x, x \rangle$ sur $H^2(X, \mathbb{Z})$ est paire \cite{Shafarevitch}). On peut montrer toutefois \cite{Deligne1976} que, pour $Y_0/k$ ordinaire, ou plus g\'en\'eralement si le noyau de la forme $\langle x, x \rangle$ sur $H^2_\mathrm{dR}(Y_0/k)$ n'est pas egal \`a $\Fil^1 H^2_\mathrm{dR}(Y_0/k)$, l'application envisag\'ee en \ref{thm:2.1.11}, associant \`a un rel\`evement formel $Y$ la droite isotrope $\Fil^2 H^2_\mathrm{dR}(Y/W) \subset H^2(Y_0/W)$ est surjective et \`a fibres finies : pour $Y_0$ ordinaire, la droite $W \otimes P_2$ de (2.1.8.5) d\'efinit alors une famille finie de ``rel\`evements canoniques'' de $Y_0$.
\end{enumerate}

\subsection{Rel\`evements de faisceaux inversibles}

\subsubsection{}
Soit $X/S = \Spf(A)$ la d\'eformation formelle universelle d'une surface K3 $X_k/k$, et soit $L^\circ$ un faisceau inversible non trivial sur $X_k$. On sait (IV 1.6) qu'il existe un plus grand sous-sch\'ema formel ferm\'e $T = Z(L^\circ) \subset S$ tel que $L^\circ$ se prolonge en un faisceau inversible $L$ sur $X \times_S T$\footnote{On note $X_{S'}$ le sch\'ema formel d\'edu de $X$ par un changement de base $S' \to S$.} et que $T$ est d\'efini par une \'equation $f = 0$ telle que $p$ ne divise pas $f$. Supposons $p \neq 2$ et $X_k$ ordinaire. On va montrer qu'on peut alors expliciter $f$ en termes de la classe de Chern cristalline
\begin{equation}\tag{2.2.1.1}
c_1(L^\circ) \in H^2(X_k/k). 
\end{equation}
Rappelons (IV 2.9) qu'avec les notations de (2.1.8.1) on a
\begin{equation}\tag{2.2.1.2}
c_1(L^\circ) \in P_1. 
\end{equation}
En particulier, si $(a, b, c, q)$ est un syst\`eme de coordonn\'ees canoniques sur $S$ (\ref{def:2.1.15}), et $e : A \to W$ est l'augmentation donn\'ee par $q_i \mapsto 1$, on peut \'ecrire
\begin{equation}\tag{2.2.1.3}
c_1(L^\circ) = \sum_{1\leqslant i\leqslant 20} x_i (e^*b_i) 
\end{equation}
avec $x_i \in \mathbb{Z}_p$. On a alors le r\'esultat suivant, dont la d\'emonstration va occuper le reste de l'expos\'e :

\begin{theorem}\label{thm:2.2.2}
Avec les hypoth\`eses et notations de 2.2.1, $T$ est d\'efini par l'\'equation
\begin{equation}\tag{2.2.2.1}
\prod_{1\leqslant i\leqslant 20} q_i^{x_i} = 1. 
\end{equation}
En d'autres termes, si, gr\^ace \`a la base $e^*a$ de $P_0$, on identifie $c_1(L^\circ)$ \`a un caract\`ere $\chi$ du tore formel $S$ (\ref{thm:2.1.14}), $T$ est le sous-groupe formel d\'efini par
\begin{equation}\tag{2.2.2.2}
T = \Ker(\chi). 
\end{equation}
En particulier, si $p$ ne divise pas $c_1(L^\circ)$ (ce qui signifie encore \cite[1.4]{Ogus1979} que $p$ ne divise pas la classe de $L^\circ$ dans $\NS(X_k)$), $T$ est un sous-tore formel lisse sur $W$ de dimension relative $19$.
\end{theorem}

\subsubsection{}
Indiquons d'abord comment on peut, heuristiquement, se persuader de la validit\'e de \ref{thm:2.2.2}. Si l'on \'etait sur le corps des complexes, on saurait que $T$ est le lieu o\`u la section horizontale de $H^2_\mathrm{dR}$ passant par $c_1(L^\circ)$ reste de type $(1,1)$, i.e. dans $\Fil^1 H^2_\mathrm{dR}$. Or, soit $x \in H^2_\mathrm{dR}(X/S) \otimes K[[q_i - 1]]$ la section horizontale telle que $e^*x = c_1(L^\circ)$. Un calcul imm\'ediat, \`a partir de (\ref{thm:2.1.7} (ii)), montre que l'on a
\begin{equation}\tag{2.2.3.1}
x = \bigl(-\sum_{1\leqslant i\leqslant 20} x_i \log q_i\bigr)a + \sum_{1\leqslant i\leqslant 20} x_i b_i. 
\end{equation}
Donc, heuristiquement, $-\sum x_i \log q_i = 0$, ``i.e.'' $\prod q_i^{x_i} = 1$ est l'\'equation cherch\'ee. Naturellement, cet argument est insuffisant, mais on peut l'adapter en utilisant la classe de Chern cristalline pour contr\^oler, pas \`a pas, l'obstruction au prolongement de $L^\circ$.

Nous aurons besoin pour cela de quelques rappels sur les classes de Chern et les obstructions, compl\'etant (IV 2.8, 2.9).

\subsubsection{}
Soient $i : Y_0 \hookrightarrow Y$ une immersion ferm\'ee de sch\'emas (ou sch\'emas formels) d\'efinie par un id\'eal $I$, et $E^\circ$ un faisceau inversible sur $Y_0$, de classe $c_1^\mathrm{Zar}(E^\circ) \in H^1(Y_0, \mathcal{O}_{Y_0}^*)$. La suite exacte de faisceaux ab\'eliens sur $Y$
\begin{equation}\tag{2.2.4.1}
0 \longrightarrow (1+I)^* \longrightarrow \mathcal{O}_Y^* \longrightarrow \mathcal{O}_{Y_0}^* \longrightarrow 0 
\end{equation}
montre que l'obstruction \`a $\sigma^*(E^\circ, i)$ prolonger $E^\circ$ en un faisceau inversible sur $Y$ est l'image de $c_1^\mathrm{Zar}(E^\circ)$ par le cobord de la suite exacte de cohomologie d\'edu te de (2.2.4.1)
\begin{equation}\tag{2.2.4.2}
\sigma^*(E^\circ, i) = \partial \, c_1^\mathrm{Zar}(E^\circ) \in H^2(Y, (1+I)^*). 
\end{equation}
Si $Y \to Z$ est un morphisme, on peut consid\'erer le complexe de de Rham ``multiplicatif''
\[
\Omega^*_{Y/Z, \log} : \mathcal{O}_Y^* \xrightarrow{d\log} \Omega^1_{Y/Z} \xrightarrow{d} \Omega^2_{Y/Z} \xrightarrow{d} \cdots
\]
et (2.2.4.1) se raffine en une suite exacte de complexes
\begin{equation}\tag{2.2.4.3}
0 \longrightarrow (1+I)^*_{Y/Z} \longrightarrow \Omega^*_{Y/Z, \log} \longrightarrow \Omega^*_{Y_0/Z, \log} \longrightarrow 0 
\end{equation}
o\`u
\[
(1+I)^*_{Y/Z} = ((1+I)^* \xrightarrow{d\log} \Omega^1_{Y/Z} \xrightarrow{d} \Omega^2_{Y/Z} \xrightarrow{d} \cdots).
\]
L'obstruction $\sigma^*(E^\circ, i)$ est l'image, par la fl\`eche naturelle, de la classe
\begin{equation}\tag{2.2.4.4}
c^*(E^\circ, i, Y/Z) \in H^2(Y, (1+I)^*_{Y/Z}) 
\end{equation}
obtenue \`a partir de $c_1^\mathrm{Zar}(E^\circ)$ comme cobord de la suite exacte de cohomologie associ\'ee \`a (2.2.4.3).

\subsubsection{}
Supposons que $I^2 = 0$. Alors on a
\[
(1+I)^* = 1+I \xrightarrow{\log} I \xrightarrow{\exp} 1+I,
\]
et, de la m\^eme mani\`ere, $(1+I)^*_{Y/Z}$ s'identifie \`a
\[
I \xrightarrow{d} I \otimes \Omega^1_{Y/Z} \xrightarrow{d} I \otimes \Omega^2_{Y/Z} \xrightarrow{d} \cdots
\]
Notons
\begin{equation}\tag{2.2.5.1}
\sigma(E^\circ, i) \in H^2(Y, I), \quad C(E^\circ, i, Y/Z) \in H^2(Y, I\Omega^*_{Y/Z}) 
\end{equation}
les images de $\sigma^*(E^\circ, i)$ et $c^*(E^\circ, i, Y/Z)$ d\'efinies par ces identifications. L'obstruction $\sigma(E^\circ, i)$ est encore l'image de $C(E^\circ, i, Y/Z)$ par la fl\`eche naturelle.

Supposons de plus que $Z = (Z, \mathcal{K}, \delta)$ soit un PD-sch\'ema o\`u $p$ est nilpotent \cite{Berthelot}, ou que $Z$ soit une base $P$-adique (avec $p \in P$) au sens de \cite[7.17]{BerthelotOgus} : le cas qui nous int\'eresse en fait est celui o\`u $Z$ est un $W$-sch\'ema formel $p$-adiquement complet, muni des puissances divis\'ees standard sur $P\mathcal{O}_Z$. Alors, si $Y$ est lisse sur $Z$, et si les puissances divis\'ees triviales sur $I$ (i.e. telles que $\gamma_i(x) = 0$ pour $i > 1$) sont compatibles \`a celles de $Z$, on a
\[
H^2(Y, I\Omega^*_{Y/Z}) = H^2(Y_0/Z, J_{Y_0/Z})
\]
(o\`u $J_{Y_0/Z}$ d\'esigne, comme d'habitude, l'id\'eal cristallin noyau de $\mathcal{O}_{Y_0/Z} \to \mathcal{O}_{Y_0}$), et la classe $C(E^\circ, i, Y/Z)$ de (2.2.5.1) n'est autre que la classe de Chern cristalline de $L^\circ$ relativement \`a $Y_0/Z$, d\'efinie dans \cite{BerthelotIllusie} :
\begin{equation}\tag{2.2.5.2}
C(E^\circ, i, X/Z) = c_1(E^\circ)_{Y_0/Z} \in H^2(X_0/Z, J_{Y_0/Z}). 
\end{equation}
Nous appliquerons ces remarques au cas o\`u l'on a des carr\'es cart\'esiens
\begin{equation}\tag{2.2.5.3}
\begin{CD}
Y @>{i}>> Y' @>>> X \\
@VVV @VVV @VVV \\
Z @>{j}>> Z' @>>> S,
\end{CD} 
\end{equation}
$X/S$ \'etant la d\'eformation formelle universelle de $X_k$, $Z$ d\'esignant un $W$-sch\'ema formel affine $p$-adiquement complet, $j$ d\'esignant une immersion ferm\'ee d\'efinie par un id\'eal $J$ de carr\'e nul, de sorte que $I = f^*(J)$. Comme $R^1f_{0*}\mathcal{O}_{Y_0} = 0$, donc $H^1(Y_0, \mathcal{O}) = 0$, les fl\`eches canoniques
\[
H^2(Y, I) \longrightarrow H^2(Y, \mathcal{O}), \quad H^2(Y, I\Omega^*_{Y/Z}) \longrightarrow H^2_\mathrm{dR}(Y/Z)
\]
sont injectives, nous consid\'ererons donc $\sigma(E^\circ, i)$ (resp.\ $C(E^\circ, i, Y/Z)$) comme un \'el\'ement de $H^2(Y, \mathcal{O})$ (resp.\ $H^2_\mathrm{dR}(Y/Z)$). D'autre part, pour $p \neq 2$, ou si $P\mathcal{O}_Z = 0$, les puissances divis\'ees triviales sur $I$ sont compatibles aux puissances divis\'ees standard sur $P\mathcal{O}_Z$. D'apr\`es ce qu'on vient de voir, l'obstruction $\sigma(E^\circ, i)$ est alors donn\'ee par la recette suivante :

\begin{lemma}\label{lem:2.2.6}
Dans la situation de (2.2.5.3), si $p \neq 2$, ou si $P\mathcal{O}_Z = 0$, l'obstruction $\sigma(E^\circ, i)$ \`a prolonger $E^\circ$ en un faisceau inversible sur $Y$ est l'image, par l'application canonique $H^2_\mathrm{dR}(Y/Z) \to H^2(Y, \mathcal{O}_Y)$ de la classe de Chern cristalline de $E^\circ$ relativement \`a $Z$, $c_1(E^\circ)_{Y_0/Z} \in H^2_\mathrm{dR}(Y/Z)$.
\end{lemma}

En d'autres termes, sous les hypoth\`eses de \ref{lem:2.2.6}, $E^\circ$ se prolonge en un faisceau inversible sur $Y$ si et seulement si $c_1(E^\circ)_{Y_0/Z} \in \Fil^1 H^2_\mathrm{dR}(Y/Z)$.

\subsubsection{}
L'int\'er\^et de la recette \ref{lem:2.2.6} est qu'on dispose d'un principe de calcul pour la classe de Chern $c_1(E^\circ)_{Y_0/Z}$, que nous allons rappeler.

Sous les hypoth\`eses de \ref{lem:2.2.6}, on peut consid\'erer la classe de Chern
\begin{equation}\tag{2.2.7.1}
c_1(E^\circ)_{Y_0/W} \in H^2(Y_0/W), 
\end{equation}
et son image
\begin{equation}\tag{2.2.7.2}
c_1(E^\circ)_{Y_0/W} \in H^0(Z_0/W, R^2f_{0*}(\mathcal{O}_{Y_0/W})) 
\end{equation}
par l'application canonique
\[
H^2(Y_0/W) \longrightarrow H^0(Z_0/W, R^2f_{0*}(\mathcal{O}_{Y_0/W})).
\]
D'autre part, pour tout PD-\'epaississement $Z_1$ de $Z_0$ on peut consid\'erer la classe de Chern
\begin{equation}\tag{2.2.7.3}
c_1(E^\circ)_{Y_0/Z_1} \in H^2(Y_0/Z_1). 
\end{equation}
Le lien entre les classes (2.2.7.2) et (2.2.7.3) est le suivant : le cristal $R^2f_{0*}(\mathcal{O}_{Y_0/W})$ a une ``valeur'' $R^2f_{0*}(\mathcal{O}_{Y_0/W})(Z_1)$ sur $Z_1$, et, pour $Z_1$ affine, la section
\[
c_1(E^\circ)_{Y_0/W}(Z_1) \in H^0(Z_1, R^2f_{0*}(\mathcal{O}_{Y_0/W})(Z_1)) = H^2(Y_0/Z_1)
\]
d\'efinie par (2.2.7.2) est (2.2.7.3). Supposons maintenant que l'on dispose d'une factorisation de la fl\`eche $Z_0 \to S$ de (2.2.5.3) en
\begin{equation}\tag{2.2.7.4}
Z_0 \xrightarrow{j'} Z' \longrightarrow S, 
\end{equation}
o\`u $Z'$ est un $W$-sch\'ema formel lisse, $p$-adiquement complet, et $j'$ une immersion ferm\'ee. Soit $Z'^\wedge = D_{Z_0}(Z')$ la PD-enveloppe, $p$-compl\'et\'ee, de $Z_0$ dans $Z'$. D'apr\`es Berthelot \cite{Berthelot} (ou \cite[\S 7]{BerthelotOgus}), on sait que le cristal $R^2f_{0*}(\mathcal{O}_{Y_0/W})$ est d\'ecrit par un $\mathcal{O}_{Z'^\wedge}$-module \`a connexion int\'egrable relativement \`a $W$. Dans le cas pr\'esent, comme on a des carr\'es cart\'esiens
\begin{equation}\tag{2.2.7.5}
\begin{CD}
Y @>>> Y'^\wedge @>>> X \\
@VVV @VVV @VVV \\
Z @>>> Z'^\wedge = D_{Z_0}(Z') @>>> S,
\end{CD} 
\end{equation}
on voit que ce module n'est autre que $H^2_\mathrm{dR}(X/S) \otimes \mathcal{O}_{Z'^\wedge}$, muni de la connexion de Gauss-Manin. La classe (2.2.7.2) s'interpr\`ete comme une section horizontale de ce module. Quand, dans la situation de (2.2.5.3), $Z$ est un sous-sch\'ema ferm\'e de $Z'$, $Z$ s'envoie dans $Z'^\wedge$ par la propri\'et\'e universelle des PD-enveloppes, et la section de $H^2_\mathrm{dR}(X/S) \otimes \mathcal{O}_Z = H^2_\mathrm{dR}(Y/Z)$ induite par la classe (2.2.7.2) n'est autre que $c_1(E^\circ)_{Y_0/Z}$. Nous verrons plus loin comment exploiter l'horizontalit\'e de la section (2.2.7.2) de $H^2_\mathrm{dR}(X/S) \otimes \mathcal{O}_{Z'^\wedge}$ pour la calculer lorsque $E^\circ$ prolonge le faisceau donn\'e $L^\circ$ sur $X_k$, en utilisant des plongements $j'$ (2.2.7.4) bien choisis.

\subsubsection{}
A cet effet, nous aurons besoin d'un dernier ingr\'edient, un peu technique, concernant les PD-enveloppes. Notons
\begin{equation}\tag{2.2.8.1}
W\ll t \gg = W\ll t_1, \ldots, t_n \gg 
\end{equation}
la PD-enveloppe, $p$-compl\'et\'ee, de l'id\'eal $(t_1, \ldots, t_n)$ de $W[[t]] = W[[t_1, \ldots, t_n]]$ : c'est le sous-anneau de $K[[t]]$ form\'e des s\'eries $\sum a_i t^i/i!$ avec $a_i \in W$ tendant vers $0$. Consid\'erons maintenant l'id\'eal $t^r = (t_1^r, \ldots, t_n^r)$ de $W[[t]]$, et la PD-enveloppe, $p$-compl\'et\'ee, de $t^r$ dans $W[[t]]$ :
\begin{equation}\tag{2.2.8.2}
D_{(t^r)}(W[[t]]). 
\end{equation}
Utilisant la compatibilit\'e de la formation des PD-enveloppes aux extensions plates \cite[I 2.7.4]{Berthelot} dans le cas du morphisme fini et plat $W[[t]] \to W[[t]]$, $t_i \mapsto t_i^r$, on obtient pour (2.2.8.2) la description suivante : le groupe additif sous-jacent est celui des s\'eries formelles \`a coefficients dans $W$, tendant vers $0$, en les $t^{(m)}$, o\`u $t^{(m)} = t^s(t^r)^{[q]}$, si $m = rq+s$, $0 \leqslant s < r$, et la multiplication est la multiplication \'evidente donn\'ee formellement par $(t^r)^{[q]} = t^{rq}/q!$

L'application
\begin{equation}\tag{2.2.8.3}
D_{(t^r)}(W[[t]]) \longrightarrow W\ll t \gg 
\end{equation}
d\'efinie par l'inclusion $(t^r) \subset (t)$ envoie l'\'el\'ement de base $t^{(m)}$ de multi-degr\'e $m$ sur $(m!/q!)t^{[m]}$ en particulier est injective.

\subsubsection{}
\textbf{D\'emonstration de \ref{thm:2.2.2}.} Comme $L^\circ$ est non trivial, on sait (IV 3.4) que $c_1(L^\circ) \neq 0$. Soit $p^m$ la plus grande puissance de $p$ divisant $x = c_1(L^\circ)$, de sorte que $x = p^m y$, avec $p \nmid y$. Donc $y$ fait partie d'une base du groupe de caract\`eres de $S$, et l'on peut supposer les coordonn\'ees $(a, b, q)$ choisies de mani\`ere que
\[
y = (1, 0, \ldots, 0), \quad \text{i.e. } x = p^m (e^* b_1).
\]
Posons
\[
T' = \Spf(W[[q_i - 1]]/(q_1^{p^m} - 1)).
\]
Il s'agit de d\'emontrer que $T = T'$. On va proc\'eder en quatre \'etapes.

\textbf{a) Prolongement de $L^\circ$ sur $X_{\Spec(k[q_1-1]/(q_1^{p^m}-1))}$.} On peut le faire de deux m\'ethodes. La plus rapide consiste \`a observer que, d'apr\`es \cite[1.4]{Ogus1979}, puisque $p^m$ divise $c_1(L^\circ)$, $p^m$ divise aussi la classe de $L^\circ$ dans $\NS(X_k)$. Si $F$ d\'esigne le Frobenius de $X_k$, $L^\circ$ est donc l'image inverse par $F^m$ d'un faisceau inversible $L'$, et par suite $L^\circ$ se prolonge \`a tout $k$-voisinage infinit\'esimal d'ordre $\leqslant p^m - 1$. On peut aussi proc\'eder directement, en prolongeant $L^\circ$ pas \`a pas sur $X_{Y_n}$, o\`u $Y_n = \Spec(k[t_1]/t_1^n)$, $q_1 - 1 = t_1$. Supposons en effet $m > 0$ et $L^\circ$ prolong\'e en un faisceau inversible $L_{Y_n}$ sur $X_{Y_n}$ pour $n < p^m$, et montrons que $L_{Y_n}$ se prolonge en un faisceau inversible sur $X_{Y_{n+1}}$. Notons $c(L_{Y_n})_{Y_{n+1}} \in H^2_\mathrm{dR}(X_{Y_{n+1}}/Y_{n+1})$ la classe de Chern cristalline de $L_{Y_n}$. D'apr\`es \ref{lem:2.2.6}, il s'agit de voir que l'image de cette classe dans $H^2(X_{Y_{n+1}}, \mathcal{O})$ est nulle. Or $Y_n$ est le sous-sch\'ema formel de $\Spf(W[[t_1]])$ d\'efini par l'id\'eal $(p, t_1^n)$, et, d'apr\`es 2.2.7, si $D_n$ d\'esigne la PD-enveloppe, $p$-compl\'et\'ee, de cet id\'eal dans $W[[t_1]]$, i.e. $D_n = D_{(p, t_1^n)}(W[[t_1]])$, $c(L_{Y_n})_{Y_{n+1}}$ est induit par la section horizontale $c(L_{Y_n})_{D_n} \in H^2_\mathrm{dR}(X/S) \otimes D_n$ du type (2.2.7.2). Comme, d'apr\`es 2.2.8, l'application $D_n \to D_1 = W\ll t_1 \gg$ est injective, $c(L_{Y_n})_{D_n}$ est d\'etermin\'e par son image dans $H^2_\mathrm{dR}(X/S) \otimes D_1$. Or cette image n'est autre que la section horizontale $c(L^\circ)_{D_1}$ d\'efinie par $L^\circ$ : cela r\'esulte de la fonctorialit\'e de la classe (2.2.7.1), compte tenu du fait que l'on a un diagramme commutatif
\[
\begin{CD}
Y_n @>>> \Spf(W[[t_1]]) \\
@VVV @VVV \\
\Spec(k) @>>> \Spec(W)
\end{CD}
\]
et que $L_{Y_n}$ prolonge $L^\circ$. Comme $c(L^\circ)$ est la classe horizontale qui prolonge $c_1(L^\circ) = p^m e^* b_1$, un calcul imm\'ediat (utilisant \ref{thm:2.1.7}) montre que l'on a
\[
c(L^\circ)_{D_1} = -(p^m \log q_1)a_{D_1} + p^m (b_1)_{D_1}
\]
(o\`u $(a_{S'}, b_{S'})$ d\'esigne l'image inverse de $(a, b)$ sur $S'$ pour $S' \to S$). Donc par l'injectivit\'e de $D_n \to D_1$,
\[
c(L_{Y_n})_{D_n} = -(\log q_1^{p^m})a_{D_n} + p^m (b_1)_{D_n}
\]
(noter que $q_1^{p^m} - 1$ appartient \`a l'id\'eal engendr\'e par $p$ et $t_1^n = (q_1 - 1)^n$, de sorte que le $\log$ a un sens gr\^ace aux puissances divis\'ees sur $(p, t_1)$). L'application $D_n \to \mathcal{O}_{Y_{n+1}}$ d\'efinie par la propri\'et\'e universelle des PD-enveloppes envoie $x^{[i]}$ (pour $x \in (p, t_1^n)$) sur $x \mod (p, t_1^{n+1})$ pour $i = 1$ et sur $0$ pour $i \geqslant 2$. En particulier, elle envoie $\log q_1^{p^m}$ sur $(q_1^{p^m} - 1) \mod (p, t_1^{n+1}) = 0$, puisque $n < p^m$. En d'autres termes, l'image $c(L_{Y_n})_{Y_{n+1}} \in H^2_\mathrm{dR}(X/S) \otimes \mathcal{O}_{Y_{n+1}}$ de $c(L_{Y_n})_{D_n}$ appartient \`a $\Fil^1$, donc, d'apr\`es ce qu'on a dit plus haut, $L_{Y_n}$ se prolonge en un faisceau inversible sur $X_{Y_{n+1}}$.

\textbf{b) Prolongement de $L^\circ$ sur $X_Z$, o\`u $Z = \Spf(W[[q_1-1]]/(q_1^{p^m} - 1))$.} Notons $L_Y$ le prolongement, obtenu en a), de $L^\circ$ sur $X_Y$. Posons
\[
Z_n = \Spec(W_n[[q_1-1]]/(q_1^{p^m} - 1)).
\]
Supposons $L_{Z_n}$ prolong\'e en un faisceau inversible sur $X_{Z_n}$, et montrons que $L_{Z_n}$ se prolonge en un faisceau inversible sur $X_{Z_{n+1}}$. D'apr\`es \ref{lem:2.2.6} (qui s'applique gr\^ace \`a l'hypoth\`ese $p \neq 2$), il suffit de v\'erifier que la classe de Chern $c(L_{Z_n})_{Z_{n+1}} \in H^2_\mathrm{dR}(X_{Z_{n+1}}/Z_{n+1})$ appartient \`a $\Fil^1$. Or celle-ci est induite par la classe $c(L_{Z_n})_D \in H^2_\mathrm{dR}(X_D/D)$, o\`u $D$ est la PD-enveloppe, $p$-compl\'et\'ee, de $Y_n$ dans $W[[q_1-1]]$. Si $D_I(~)$ d\'esigne la PD-enveloppe, $p$-compl\'et\'ee, de l'id\'eal $I$, on a
\begin{align*}
D &= D_{(p^n, q_1^{p^m} - 1)}(W[[q_1-1]]) \\
&= D_{(p^n, q_1^{p^m} - 1)}(W[[q_1-1]]) \\
&= D_{(p, (q_1-1)^{p^m})}(W[[q_1-1]]) \\
&= D_{(p, t_1^{p^m})}(W[[t_1]]) \\
&= D_{(t_1^{p^m})}(W[[t_1]]),
\end{align*}
o\`u $t_1 = q_1 - 1$. D'apr\`es 2.2.8, l'application $D \to D_{(t_1)}(W[[t_1]]) = W\ll t_1 \gg$ d\'efinie par l'inclusion $(p^n, q_1^{p^m} - 1) \subset (p, q_1 - 1)$ est donc injective, et l'on voit, par le m\^eme argument qu'en a), que
\[
c(L_{Z_n})_D = -(\log q_1^{p^m})a_D + p^m (b_1)_D.
\]
Par suite, $c(L_{Z_n})_{Z_{n+1}}$, image de $c(L_{Z_n})_D$ par l'application $D \to \mathcal{O}_{Z_{n+1}}$ d\'efinie par la propri\'et\'e universelle des PD-enveloppes (qui envoie $x^{[i]}$ (pour $x \in (p^n, q_1^{p^m} - 1)$) sur l'image de $x$ dans $\mathcal{O}_{Z_{n+1}}$ pour $i = 1$ et sur $0$ pour $i > 1$), est donn\'ee par
\begin{align*}
c(L_{Z_n})_{Z_{n+1}} &= -(q_1^{p^m} - 1)a_{Z_{n+1}} + p^m (b_1)_{Z_{n+1}} \\
&= p^m (b_1)_{Z_{n+1}}
\end{align*}
(car $q_1^{p^m} - 1 = 0$ sur $Z_{n+1}$). Donc $c(L_{Z_n})_{Z_{n+1}}$ appartient \`a $\Fil^1 H^2_\mathrm{dR}(X_{Z_{n+1}}/Z_{n+1})$, et par cons\'equent $L_{Z_n}$ se prolonge en un faisceau inversible sur $X_{Z_{n+1}}$. Donc $L^\circ$ se prolonge en un faisceau inversible $L_Z$ sur $X_Z$.

\textbf{c) Prolongement de $L^\circ$ sur $X_{T'_n}$.} Posons $t_i = q_i - 1$. Soit $n = (n_2, \ldots, n_{20})$ une suite d'entiers $\geqslant 1$, posons
\[
T'_n = \Spf(W[[t]]/(q_1^{p^m} - 1, t_2^{n_2}, \ldots, t_{20}^{n_{20}}))
\]
(donc $T'_{(1,\ldots,1)} = Z$). Supposons $L_Z$ prolong\'e en un faisceau inversible $L_{T'_n}$ sur $X_{T'_n}$. Alors, par un argument analogue \`a celui utilis\'e en b), on voit que, si $2 \leqslant i \leqslant 20$, et $n+1_i = (n_2, \ldots, n_{i-1}, n_i+1, n_{i+1}, \ldots, n_{20})$, $L_{T'_n}$ se prolonge en un faisceau inversible sur $X_{T'_{n+1_i}}$. On en d\'eduit, par r\'ecurrence, que $L_Z$ se prolonge en un faisceau inversible $L_{T'_n}$ sur $X_{T'_n}$.

\textbf{d) Fin de la d\'emonstration.} Pour prouver que $T' = T$, il reste \`a v\'erifier que, si $T'' \supset T'_n$ est un sous-sch\'ema formel ferm\'e de $S$, d\'efini par un id\'eal $I$, tel que $L_{T'_n}$ se prolonge en un faisceau inversible sur $X_{T''}$, alors $T'' = T'$. Il revient au m\^eme de montrer que, si $T'' \supsetneqq T'$ et $I^2 = 0$, l'obstruction \`a prolonger $L_{T'_n}$ en un faisceau inversible sur $X_{T''}$ est non nulle, i.e. que la classe de Chern $c(L_{T'_n})_{T''} \in H^2_\mathrm{dR}(X_{T''}/T'')$ n'appartient pas \`a $\Fil^1$. Or, le m\^eme calcul que pr\'ec\'edemment fournit
\[
c(L_{T'_n})_{T''} = -(q_1^{p^m} - 1)a_{T''} + p^m (b_1)_{T''}.
\]
Comme $T'' \supsetneqq T'$, l'image de $q_1^{p^m} - 1$ dans $T''$ est non nulle, donc $c(L_{T'_n})_{T''} \notin \Fil^1$, ce qui ach\`eve la d\'emonstration.

\subsection*{A1. Uniqueness of group structures}

Let $k$ be a perfect field, $W = W(k)$ its ring of Witt vectors, and $\sigma : W \to W$ the absolute Frobenius automorphism of $W$. Let $M$ be a finite-dimensional formal Lie variety over $W$, i.e. $M = \Spf(A)$ with $A$ non-canonically isomorphic to $W[[T_1, \ldots, T_n]]$, $n = \dim M$. Suppose we are given a $W$-morphism of formal Lie varieties
\[
\Phi : M \longrightarrow M^{(\sigma)}
\]
whose reduction modulo $p$ is the absolute Frobenius endomorphism
\[
\Frob : M \otimes_W k \longrightarrow (M \otimes_W k)^{(\sigma)}.
\]

\begin{lemma}[Uniqueness Lemma A1.1]\ \begin{enumerate}[label=(\arabic*)]
\item Given $(M, \Phi)$ as above, there exists at most one structure of commutative formal Lie group over $W$ on the formal Lie variety $M$ for which the given map $\Phi : M \to M^{(\sigma)}$ is a group homomorphism.
\item If this structure exists, it makes $M$ into a toroidal formal group, and the given $\Phi : M \to M^{(\sigma)}$ is the unique group homomorphism lifting Frobenius.
\item If $(M_1, \Phi_1)$ and $(M_2, \Phi_2)$ both admit group structures as in (2) above, then a morphism $f : M_1 \to M_2$ of formal Lie varieties over $W$ is a group homomorphism if and only if the diagram
\[
\begin{CD}
M_1 @>{f}>> M_2 \\
@V{\Phi_1}VV @VV{\Phi_2}V \\
M_1^{(\sigma)} @>{f^{(\sigma)}}>> M_2^{(\sigma)}
\end{CD}
\]
commutes.
\end{enumerate}
\end{lemma}

\begin{proof}
We begin by proving (2). Thus let $G$ be a finite-dimensional commutative formal Lie group over $W$, together with a homomorphism
\[
\Phi : G \longrightarrow G^{(\sigma)}
\]
which lifts Frobenius. We must show that $G$ is toroidal, and that $\Phi$ is unique. By the rigidity of toroidal groups, it suffices to show that $G \otimes k$ is toroidal, and for this it suffices to show that $\Ker(\Frob)$ is toroidal. For this, we first observe that $\Phi : G \to G^{(\sigma)}$ is finite (because it is finite modulo $p$, being a lifting of Frobenius) and flat (because it is a finite morphism between regular local rings of the same dimension). Therefore $\Ker(\Phi)$ is a finite flat commutative group-scheme over $W$ whose reduction mod $p$ is $\Ker(\Frob)$. According to Fontaine, if we denote by $N$ the contravariant Dieudonn\'e module of $\Ker(\Frob)$, then the lifting $\Ker(\Phi)$ is described by a $W$-submodule $L \subset N$ which satisfies
\begin{enumerate}[label=\alph*)]
\item $L/pL \xrightarrow{\sim} N/FN$
\item $V|_L$ is injective.
\end{enumerate}
But $N$ is killed by $F$, and is of finite length over $W$. Therefore a) implies that $L = N$, and b) then shows that $V$ is injective, and hence bijective on $N$. Therefore $\Ker(\Frob)$ is toroidal, as required.

We next prove (3). By extending scalars $W(k) \to W(\bar{k})$, we may suppose $k$ algebraically closed. Then $M_1$ and $M_2$ become isomorphic to products of $\hat{\mathbb{G}}_m$, and our commutative diagram --- in the category of formal Lie varieties --- becomes
\[
\begin{CD}
(\hat{\mathbb{G}}_m)^{n_1} @>{f}>> (\hat{\mathbb{G}}_m)^{n_2} \\
@V{\Frob}VV @VV{\Frob}V \\
(\hat{\mathbb{G}}_m)^{n_1} @>{f^{(\sigma)}}>> (\hat{\mathbb{G}}_m)^{n_2}
\end{CD}
\]
If $f$ is a group homomorphism, then $f = f^{(\sigma)}$, and the diagram commutes. To prove the converse, we argue as follows.

In terms of ``multiplicative'' coordinates $T$ on $(\hat{\mathbb{G}}_m)^i$, $i = 1, 2$, our hypothesis on $f$ is :
\[
f^{(\sigma)}(T^p) = (f(T))^p, \quad f(1) \equiv 1 \mod p.
\]
Iterating, we find
\[
f^{(\sigma^n)}(T^{p^n}) = (f(T))^{p^n},
\]
so in particular
\[
f^{(\sigma^n)}(1) = (f(1))^{p^n} \to 1 \quad \text{as } n \to \infty.
\]
Therefore $f(1) = 1$. Now take logarithms, i.e. let
\[
F : (\hat{\mathbb{G}}_a)^{n_1} \longrightarrow (\hat{\mathbb{G}}_a)^{n_2}
\]
be the unique pointed morphism (over the fraction field of $W$) for which
\[
F(\log T) = \log(f(T));
\]
then we have
\[
F^{(\sigma)}(pX) = pF(X).
\]
Therefore $F$ is linear, and has coefficients in $W_p$. As these coefficients are intrinsically the matrix entries of the tangent map of $f$ at the origin, they lie in $W$ as well, hence $F$ is a linear map with coefficients in $\mathbb{Z}_p = W \cap \mathbb{Q}_p$. Therefore $f(T) = \exp(F(\log T))$ is a homomorphism.

Finally, we obtain (1) as the special case $(M_1, \Phi_1) = (M_2, \Phi_2)$, $f = \mathrm{id}$, of (3). \qedhere
\end{proof}

\begin{corollary}[A1.2]
Suppose that $k$ is algebraically closed, and that $(M, \Phi)$ admits a group structure as above. Then
\begin{enumerate}[label=(\arabic*)]
\item The character group $X(M) = \Hom_{W\text{-gp}}(M, \hat{\mathbb{G}}_m)$ is a free $\mathbb{Z}_p$-module of rank $n = \dim M$, and the natural map (of functors in $\mathbb{Z}_p$-modules)
\[
M \longrightarrow \Hom_{\mathbb{Z}_p}(X(M), \hat{\mathbb{G}}_m)
\]
is an isomorphism.
\item Let $q$ be a function on $M$ (i.e. $q \in A$ where $M = \Spf(A)$) with $q \equiv 1$ modulo the maximal ideal of $A$. Then $q \in X(M)$ if and only if $q$ transforms under $\Phi$ by
\[
\Phi^*(q^{(\sigma)}) = q^p.
\]
\item Let $\omega$ be a (continuous) one-form on $A$, i.e. $\omega \in (\Omega^1_{A/W})^{\mathrm{contin}}$. Then $\omega = dq/q$ with $q \in X(M) = \Hom(M, \mathbb{G}_m)$ if and only if
\[
\Phi^*(\omega^{(\sigma)}) = p\omega.
\]
If $\omega$ satisfies this, the associated $q \in X(M)$ is unique ; it is given by the formula
\[
q(X) = \exp\Bigl(\int_0^X \omega\Bigr), \quad 0 = \text{the origin in } M.
\]
\item Let $\tau \in A \hat{\otimes}_W W[1/p]$ be a function on $M \otimes_W W[1/p]$. Then $\tau = \log(q)$ for some $q \in X(M)$ if and only if $\tau$ satisfies
\[
\tau(0) = 0, \quad \Phi^*(\tau^{(\sigma)}) = p\tau.
\]
If $\tau$ satisfies this, then the associated $q$ is unique ; it is given by the formula
\[
q = \exp(\tau).
\]
\item Let $q_1, \ldots, q_n$ be $n = \dim M$ elements of $X(M)$, $\omega_1, \ldots, \omega_n$ the corresponding differentials $\omega_i = dq_i/q_i$ and $\tau_1, \ldots, \tau_n$ the corresponding ``functions with denominators'' $\tau_i = \log q_i$. Then the following conditions are equivalent :
\begin{enumerate}[label=\alph*)]
\item $q_1, \ldots, q_n$ form a $\mathbb{Z}_p$ base of $X(M)$.
\item the natural map
\[
M \longrightarrow \Spf(W[[q_1 - 1, \ldots, q_n - 1]])
\]
is an isomorphism.
\item $\omega_1, \ldots, \omega_n$ form an $A$-base of $\Omega^1_{A/W} \otimes \mathcal{O}^{\mathrm{contin}}$.
\item $\omega_1, \ldots, \omega_n$ form a $\mathbb{Z}_p$-base of
\[
\{\omega \in \Omega^1 \mid \Phi^*(\omega^{(\sigma)}) = p\omega\}.
\]
\item $\tau_1, \ldots, \tau_n$ form a $\mathbb{Z}_p$-base of
\[
\{\tau \in \Omega^1 \otimes_W W[1/p] \mid \tau(0) = 0, \; \Phi^*(\tau^{(\sigma)}) = p\tau\}.
\]
\end{enumerate}
\end{enumerate}
\end{corollary}

\begin{proof}
Because $M$ is toroidal, and $k$ is algebraically closed, $M$ is (non-canonically) isomorphic to $(\hat{\mathbb{G}}_m)^n$. This makes (1) obvious.

Assertion (2) is a particular case of part (3) of the uniqueness lemma, namely $M_1 = M$, $M_2 = \hat{\mathbb{G}}_m$. Assertion (3) becomes obvious if we choose a $\mathbb{Z}_p$-basis $q_1, \ldots, q_n$ of $X(M)$, i.e. if we choose an isomorphism
\[
M \xrightarrow{\sim} (\hat{\mathbb{G}}_m)^n,
\]
and write $\omega$ as an $A$-linear combination of the differentials $dq_i/q_i$ :
\[
\omega = \sum f_i \, dq_i/q_i.
\]
The condition
\[
\Phi^*(\omega^{(\sigma)}) = p\omega
\]
means precisely that the coefficient functions $f_i$ each satisfy
\[
\Phi^*(f_i^{(\sigma)}) = f_i,
\]
which in turn implies that each coefficient function $f_i$ is simply a constant in $\mathbb{Z}_p$. Therefore
\[
\omega = dq/q \quad \text{for } q = \prod (q_i)^{f_i};
\]
because $q(0) = 1$, we obtain by integration the formula
\[
\exp\Bigl(\int_0^X \omega\Bigr) = q(X).
\]

For assertion (4), first note that the Dieudonn\'e-Dwork integrality criterion (cf. 1.4.4) guarantees that the series $q$, defined as
\[
q \overset{\mathrm{dfn}}{=} \exp(\tau),
\]
is integral (i.e. lies in $A$) and verifies $q(0) = 1$. From the equality
\[
\Phi^*(\tau^{(\sigma)}) = p\tau
\]
(and not simply their congruence modulo $pA$) we see that
\[
\Phi^*(q^{(\sigma)}) = q^p,
\]
and we conclude by (2) that $q$ lies in $X(M)$.

In assertion (5), the equivalence of a), b) and c) is physically obvious for $(\hat{\mathbb{G}}_m)^n$, and the equivalence of a) with d) and e) is obvious from parts 3) and 4) above. \qedhere
\end{proof}

\subsection*{A2. Sharpenings of 1.4.2}

\textbf{Corollary A2.1 of 1.4.2.} With the hypotheses and notations of 1.4.2, suppose that there exists a lifting $\varphi_{\mathrm{can}}$ of Frobenius on $\Spf(A)$ such that
\[
F(\varphi_{\mathrm{can}})(\varphi_{\mathrm{can}}^*(\Fil^1)) \subset \Fil^1.
\]
Let $\bar{e}$ denote the $W$-valued point of $\Spf(A)$ which is the $\varphi_{\mathrm{can}}$-Teichm\"uller representative of the augmentation $A \to k$. Then in formulaire 1.4.2.1--7 we have the following further precisions :
\[
\nabla a_i = 0
\]
\[
\nabla b_i = \sum_j \eta_{ij} \otimes a_j
\]
\[
F(\varphi_{\mathrm{can}})(\varphi_{\mathrm{can}}^*(a_i)) = a_i
\]
\[
F(\varphi_{\mathrm{can}})(\varphi_{\mathrm{can}}^*(b_i)) = pb_i
\]
\[
\varphi_{\mathrm{can}}^*(\eta_{ij}) = p\eta_{ij}, \quad du_{ij} = 0
\]
\[
\tau_{ij}(0) = 0
\]
\[
q_{ij} = \exp(\tau_{ij}) \text{ is defined, lies in } A, \text{ and } q_{ij}(0) = 1
\]
\[
\varphi_{\mathrm{can}}^*(q_{ij}) = q_{ij}^p.
\]

\textbf{Further Corollary A2.2.} (Analogue of (4.7))
\begin{enumerate}[label=(\arabic*)]
\item Given $\varphi_{\mathrm{can}}$ as above, suppose in addition that (1.4.6.3) : $T_{A/W} \to \Hom_A(\Fil^1, U)$ is an isomorphism. Then $(\Spf(A), \varphi_{\mathrm{can}})$ admits a group structure, and the morphism
\[
\Spf(A) \longrightarrow \prod_{i,j} \hat{\mathbb{G}}_m
\]
defined by the $q_{i,j}$ is an isomorphism of groups.
\item If $p \neq 2$ then $\varphi_{\mathrm{can}}$ is the unique lifting of Frobenius such that
\[
F(\varphi_{\mathrm{can}})(\varphi_{\mathrm{can}}^*(\Fil^1)) \subset \Fil^1.
\]
\end{enumerate}

\begin{proof}
The formulaire is simply obtained from the one given in 1.4.2.1--7 by noting that the $u_{ij}(\varphi_{\mathrm{can}})$ all vanish. The first assertion of the further corollary has already been proven when $p \neq 2$, but the condition $p \neq 2$ was used only to assure that $\exp(\tau_{ij}(\bar{e}))$ make sense. As our hypotheses on $\varphi_{\mathrm{can}}$ give $\tau_{ij}(0) = 0$, this problem will not arise, and the proof given goes through tel quel. The unicity of $\varphi_{\mathrm{can}}$ in case $p \neq 2$ results from the observation that the series $q_{ij} = \exp(\tau_{ij})$ are then definible without reference to a particular choice of $\varphi$, and furnish an isomorphism
\[
M \xrightarrow{\sim} \Spf(W[[q_{ij} - 1]]);
\]
our $\varphi_{\mathrm{can}}$ is the lifting of Frobenius given by
\[
\varphi_{\mathrm{can}}(q_{ij}) = (q_{ij})^p. \qedhere
\]
\end{proof}

\subsection*{A3. Formal moduli of ordinary abelian varieties ; the Serre-Tate \& Dwork group structures}

Let $k$ be an algebraically closed field of characteristic $p > 0$, and $X_0/k$ an ordinary abelian variety over $k$. Let $M$ be its formal deformation space, and $X/M$ the corresponding formal abelian scheme.

One knows that $M$ is a $g^2$-dimensional formal Lie variety over $W = W(k)$. Because $X_0/k$ is ordinary, the formal group $\hat{X}$ of $X/M$ is non-canonically isomorphic to $(\hat{\mathbb{G}}_m)^g$ over $M$. The ``canonical subgroup'' $H_{\mathrm{can}} \subset \hat{X}$ is defined to be the kernel of $[p]$ in $\hat{X}$ ; it is the unique finite flat subgroup-scheme of $X/M$ which mod $p$ becomes the kernel of the relative Frobenius endomorphism of $X \otimes_W k/M \otimes_W k$. We denote by
\[
F_{\mathrm{can}} : X \longrightarrow X/H_{\mathrm{can}}
\]
the projection onto the quotient by $H_{\mathrm{can}}$. The quotient $X/H_{\mathrm{can}}$ over $M$ is a deformation of $X_0^{(p)}/k$, so its ``classifying map'' is a morphism of formal Lie varieties
\[
\Phi_{\mathrm{can}} : M \longrightarrow M^{(\sigma)},
\]
``defined by'' an isomorphism of formal abelian schemes over $M$
\[
\theta_{\mathrm{can}}^*(X^{(\sigma)}) \xrightarrow{\sim} X/H_{\mathrm{can}}.
\]
Thus $\Phi_{\mathrm{can}}$ is a lifting of the absolute Frobenius endomorphism of $M \otimes k$, and
\[
(F_{\mathrm{can}})_M^{(\Phi)} : X \longrightarrow X/H_{\mathrm{can}} = \Phi_{\mathrm{can}}^* X^{(\sigma)}
\]
is a lifting of the relative (to $M \otimes k$) Frobenius endomorphism of $X \otimes k/M \otimes k$. Therefore this morphism induces on $H^1_\mathrm{dR}$ an $M$-linear map
\[
F(\Phi_{\mathrm{can}}) : \Phi_{\mathrm{can}}^* H^1_\mathrm{dR}(X/M) \longrightarrow H^1_\mathrm{dR}(X/M)
\]
which is none other than the crystalline map $F(\Phi_{\mathrm{can}})$. Because $F_{\mathrm{can}}$ is a physical morphism, the induced map $F$ respects the Hodge filtration of $H^1_\mathrm{dR}$. Therefore we have
\[
F(\Phi_{\mathrm{can}})(\Phi_{\mathrm{can}}^*(\Fil^1)) \subset \Fil^1.
\]

\begin{theorem}[A3.1]
The structure of group imposed on $M$ by the Serre-Tate description of $M$ as
\[
\Ext((X_0^{(p^n)})^{\mathrm{\acute{e}t}}, X_0^{p^n\text{-}\mathrm{conn}})
\]
coincides with the structure of group on $M$ for which the $q_{ij} = \exp(\tau_{ij})$ define an isomorphism of groups
\[
M \xrightarrow{\sim} \prod_{i,j} \hat{\mathbb{G}}_m.
\]
\end{theorem}

\begin{proof}
The morphism $\Phi_{\mathrm{can}} : M \to M^{(\sigma)}$ is also a group homomorphism for the Serre-Tate group structure on $M$. \qedhere
\end{proof}

\subsection*{A4. Formal moduli of ordinary K3 surfaces ; sharpening of 2.1.7, 2.1.14}

\textbf{Corollary A4.1 (of 2.1.7, 2.1.14).} With the hypotheses and notations of 2.1.7 and 2.1.14, there exists a unique morphism
\[
\Phi_{\mathrm{can}} : \Spf(A) \longrightarrow \Spf(A)^{(\sigma)}
\]
lifting Frobenius for which the induced crystalline map $F(\Phi_{\mathrm{can}})$ on $H^2_\mathrm{dR}(X/A)$
\[
F(\Phi_{\mathrm{can}}) : \Phi_{\mathrm{can}}^* H^2_\mathrm{dR}(X/A) \longrightarrow H^2_\mathrm{dR}(X/A)
\]
preserves the Hodge filtration, i.e. satisfies
\[
F(\Phi_{\mathrm{can}})\Phi_{\mathrm{can}}^*(\Fil^2) \subset \Fil^2, \quad F(\Phi_{\mathrm{can}})\Phi_{\mathrm{can}}^*(\Fil^2) \subset \Fil^1.
\]
The group structure on $\Spf(A)$ defined by $q_1, \ldots, q_{20}$ is the unique one for which $\Phi_{\mathrm{can}}$ is a group homomorphism.

\begin{proof}
The proof of 2.1.7 shows that any $\Phi_{\mathrm{can}}$ whose associated $F(\Phi_{\mathrm{can}})$ preserves the Hodge filtration satisfies
\[
\Phi_{\mathrm{can}}^*(q_i) = q_i^p \quad \text{for } i = 1, \ldots, 20.
\]
Therefore $\Phi_{\mathrm{can}}$ is a group homomorphism ; as it is completely specified by its effect on the $q_i$, it is unique. Part (iii) of 2.1.7 shows that such a $\Phi_{\mathrm{can}}$, preserving the Hodge filtration, does in fact exist. \qedhere
\end{proof}
