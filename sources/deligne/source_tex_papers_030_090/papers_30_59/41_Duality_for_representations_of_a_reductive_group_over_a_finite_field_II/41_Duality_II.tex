\section*{6}

Let $l$ be a prime number, $(l,q)=1$, and let $\bar{\mathbb{Q}}_l$ be an algebraic closure of $\mathbb{Q}_l$. Let $\sR(G)$ be the Grothendieck group of virtual $G$-modules over $\bar{\mathbb{Q}}_l$. We denote $D_G \colon \sR(G) \to \sR(G)$ the homomorphism such that
\begin{equation*}
D_G(E) = \sum_i (-1)^i E_i,
\end{equation*}
for any finite-dimensional $G$-module $E$ over $\bar{\mathbb{Q}}_l$.

Let $T$ be a maximal torus of $G$, defined over $\mathbb{F}_q$ and let $T$ be the group of its $\mathbb{F}_q$-rational points. For any one dimensional $T$-module $\theta$ (over $\bar{\mathbb{Q}}_l$) we have defined in [5, 1.20] a virtual representation $R_T^G(\theta)$. Let $\sigma(G)$ be the $\mathbb{F}_q$-rank of $G$.

We shall prove the following:

\begin{theorem}
\begin{equation*}
D_G(R_T^G(\theta)) = (-1)^{\sigma(G)-\sigma(T)} R_T^G(\theta).
\end{equation*}
\end{theorem}

Let us recall the definition of $R_T^G(\theta)$ or, more generally, that of ``twisted induction'' from Levi subgroups (cf. [5, 7]).

Let $P$ be a parabolic subgroup of $G$, with unipotent radical $U$ and let $L$ be a Levi subgroup of $P$. We assume that $L$ is defined over $\mathbb{F}_q$, but we make no assumption on $P$. Let $L$ be the group of $\mathbb{F}_q$-rational points of $L$. Let $F \colon G \to G$ be the Frobenius map corresponding to the $\mathbb{F}_q$-rational structure of $G$. The variety
\begin{equation*}
X_{L,P} = \{g \in G \mid g^{-1}F(g) \in P\}
\end{equation*}
has a $G \times L$ action
\begin{equation*}
(g_0,l) \colon g \mapsto g_0 g l^{-1}.
\end{equation*}
Hence we have a $G \times L$-action $((g_0,l)^*)^{-1}$ of $G \times L$ on the cohomology groups $H_c^i(X_{L,P})$. (Here $H_c^i(\ )$ denotes \'etale cohomology with compact support with coefficients in $\bar{\mathbb{Q}}_l$.) Let $\pi$ be a finite-dimensional $L$-module over $\bar{\mathbb{Q}}_l$. Then $(H_c^i(X_{L,P}) \otimes \pi)_L$, with $L$ acting diagonally, is in a natural way a $G$-module. By definition,
\begin{equation*}
R_{L\subset P}^G(\pi) = \sum_i (-1)^i (H_c^i(X_{L,P}) \otimes \pi)_L \in \sR(G).
\end{equation*}

In particular, if $B$ is a Borel subgroup containing $T$, with unipotent radical $V$, the virtual representation $R_{T\subset B}^G(\theta)$ is defined. It is independent of $B$ [5, 4.3] and is, by definition, $R_T^G(\theta)$.

Let $\langle\ ,\ \rangle_G \colon \sR(G) \times \sR(G) \to \mathbb{Z}$ be the bilinear form defined by $\langle E, E' \rangle_G = \dim \Hom_G(E,E')$ for any two finite-dimensional $G$-modules $E, E'$ over $\bar{\mathbb{Q}}_l$.

The following result is a generalization of the orthogonality formula [5, 6.8].

\begin{theorem}
\begin{equation*}
\langle R_T^G(\theta), R_{L\subset P}^G(\pi) \rangle_G = \sum_{\substack{n \in G \\ n^{-1}Tn \subset L}} \langle R_{n^{-1}Tn \cap L}^{L}({}^n\theta), \pi \rangle_L,
\end{equation*}
where, for $n$ in the sum, ${}^n\theta$ denotes the representation of $n^{-1}Tn$ defined by ${}^n\theta(n^{-1}tn) = \theta(t)$, $t \in T$.
\end{theorem}

\section*{7}

We shall now give the proof of Theorem 6 assuming Theorem 7. If $G=T$, the theorem is obvious. Assume now that $G \neq T$ and that the theorem is already proved for groups of dimension $< \dim G$. We shall distinguish two cases.

\textbf{Case 1.} $T$ is contained in a proper parabolic subgroup $P$ of $G$, defined over $\mathbb{F}_q$. Let $L$ be a Levi subgroup of $P$, containing $T$ and defined over $\mathbb{F}_q$. Let $L$, $P$ denote the groups of $\mathbb{F}_q$-rational points of $L$, $P$. According to [5, 8.2] we have $R_T^G(\theta) = \Ind(R_T^L(\theta))$, where $\Ind \colon \sR(L) \to \sR(G)$ is the homomorphism defined by lifting a representation of $L$ to $P$ and then inducing it from $P$ to $G$. According to [1], we have $\Ind \circ D_L = D_G \circ \Ind$. Since the theorem is assumed to be true for $L$, we have
\begin{align*}
D_G(R_T^G(\theta)) &= D_G \Ind(R_T^L(\theta)) = \Ind(D_L(R_T^L(\theta))) \\
&= \Ind((-1)^{\sigma(L)-\sigma(T)} R_T^L(\theta)) = (-1)^{\sigma(G)-\sigma(T)} R_T^G(\theta),
\end{align*}
since $\sigma(G) = \sigma(L)$.

\textbf{Case 2.} For any proper parabolic subgroup $P$ defined over $\mathbb{F}_q$, we have $T \not\subset P$.

Given such $P$, let $L$ be a Levi subgroup of $P$ defined over $\mathbb{F}_q$ and let $P$, $L$, $\Ind$ be defined as in Case 1. Our hypothesis shows that the set $\{n \in G \mid n^{-1}Tn \subset L\}$ is empty. By Theorem 7, we then have $\langle R_{L\subset P}^G(\pi), R_T^G(\theta) \rangle_G = 0$, for any $L$-module $\pi$ of finite dimension. An argument entirely similar to that in the proof of [5, 8.2] shows that $R_{L\subset P}^G(\pi) = \Ind \pi$. Thus we have
\begin{equation*}
\langle \Ind \pi, R_T^G(\theta) \rangle_G = 0
\end{equation*}
for any $L$-module $\pi$ of finite dimension or, equivalently,
\begin{equation*}
(R_T^G(\theta))^U = 0,
\end{equation*}
where $M \mapsto M^U$ is the homomorphism $\sR(G) \to \sR(L)$ such that, whenever $M$ is a $G$-module, $M^U$ is the space of vectors of $M$, invariant under $U$ ($=$ the unipotent radical of $P$). From the definition of $D_G$, we see then that $D_G(R_T^G(\theta)) = (-1)^{\sigma(G)} R_T^G(\theta)$. It remains to use the fact that, for our $T$, we have $\sigma(G) - \sigma(T) \equiv 0 \pmod 2$.

\section*{9}

Let $X$ be a scheme of finite type over an algebraic closure of $\mathbb{F}_q$ and let $f \colon X \to X$ be a morphism. We set
\begin{equation*}
\sP(f, X) = \sum_i (-1)^i \Tr(f^*, H_c^i(X)).
\end{equation*}

In the proof of Theorem 7, we shall need the following:

\begin{lemma}
Assume that $f$ is an automorphism (of finite order) of $X$, and let $T$ be a torus acting on $X$ such that $f$ commutes with each element of $T$. Then $\sP(f, X) = \sP(f, X^T)$, where $X^T$ is the fixed point set of $T$ on $X$.
\end{lemma}

\begin{proof}
Write $f = f_u \cdot f_s = f_s \cdot f_u$, where $f_s$ (resp.\ $f_u$) is a power of $f$ and has order prime to $p$ (resp.\ power of $p$). According to [5, 3.2], we have $\sP(f, X) = \sP(f_u, X^{f_s})$, $\sP(f, X^T) = \sP(f_u, (X^T)^{f_s}) = \sP(f_u, (X^{f_s})^T)$. Thus, we are reduced to the case where $f$ has order a power of $p$. We can find an element $t \in T$ such that $X^t = X^T$. Assume that we know that
\begin{equation}\label{eq:9.1}
\sP(f, X) = \sP(ft, X).
\end{equation}
Using again [5, 3.2], we have $\sP(ft, X) = \sP(f, X^t) = \sP(f, X^T)$; hence, by \eqref{eq:9.1}, we have $\sP(f, X) = \sP(f, X^T)$, as desired. It remains to prove \eqref{eq:9.1}.

We choose $\mathbb{F}_q$-rational structures on $T$ and $X$ with Frobenius maps $F \colon T \to T$, $F \colon X \to X$, such that $F(t'x) = F(t')F(x)$, for all $t' \in T$, $x \in X$, $F(f) = f$ and $F(tx) = tF(x)$ for all $x \in X$. Using the trace formula for Frobenius maps, we see that \eqref{eq:9.1} would be a consequence of
\begin{equation}\label{eq:9.2}
|X^{F^nf}| = |X^{F^nft}| \quad \text{for all integers } n \geq 1.
\end{equation}
Given $n \geq 1$, we can find $t_0 \in T$ such that $t_0^{-1}F^n(t_0) = t$. Then the map $x \mapsto t_0^{-1}x$ is a bijection $X^{F^nf} \xrightarrow{\sim} X^{F^nft}$. This completes the proof of the lemma.
\end{proof}

\section*{10}

We shall now prove Theorem 7. As in the proof of [7, Theorem 8] we see that the left-hand side of (7.1) is equal to
\begin{equation}\label{eq:10.1}
|T|^{-1} |L|^{-1} \sum_{\substack{t \in T \\ l \in L \\ j \geq 0}} (-1)^j \Tr((t,l), H_c^j(S)) \Tr(t, \theta) \Tr(l^{-1}, \pi),
\end{equation}
where
\begin{equation*}
S = \{(x, x', y) \in V \times U \times G \mid xF(y) = yx'\}
\end{equation*}
and $T \times L$ acts on $S$ by
\begin{equation*}
(t,l) \colon (x,x',y) \mapsto (txt^{-1}, lx'l^{-1}, tyl^{-1}).
\end{equation*}

Let $C = \{n \in G \mid n^{-1}Tn \subset L\}$ and let $C_1, C_2, \dots, C_m$ be the orbits of $L$ acting on $C$ by right translation. For each $i$, $1 \leq i \leq m$, let $G_i = F^{-1}(V) C_i F^{-1}(P) = F^{-1}(V) C_i F^{-1}(U)$. Then $G_i$ ($1 \leq i \leq m$) are the double cosets of $G$ with respect to $F^{-1}(B)$ and $F^{-1}(P)$.

We set $S_i = \{(x,x',y) \in S \mid y \in G_i\}$. The $S_i$ form a partition of $S$, stable under the action of $T \times L$. Corresponding to this partition, we have a decomposition of expression \eqref{eq:10.1} into the sum of $m$ terms, the terms being obtained from \eqref{eq:10.1} by replacing $S$ by $S_i$ ($1 \leq i \leq m$). Let \eqref{eq:10.1}$(i)$ be the term corresponding to $i$. Let
\begin{equation*}
\tilde{S}_i = \{(x,x',v,n,u) \in V \times U \times F^{-1}(V) \times C_i \times F^{-1}(U) \mid xF(vnu) = vlux'\}.
\end{equation*}

The map $\tilde{S}_i \to S_i$ defined by $(x,x',v,n,u) \mapsto (x,x',vnu)$ is a locally trivial fibration all of whose fibres are affine spaces of fixed dimension. Moreover, this map is $T \times L$-equivariant, for the action of $T \times L$ on $\tilde{S}_i$ given by
\begin{equation}\label{eq:10.2}
(t,l) \colon (x,x',v,n,u) \mapsto (txt^{-1}, lx'l^{-1}, tvt^{-1}, lnl^{-1}, lul^{-1}).
\end{equation}
It follows that expression \eqref{eq:10.1}$(i)$ is equal to
\begin{equation}\label{eq:10.3}
|T|^{-1} |L|^{-1} \sum_{\substack{t \in T \\ l \in L \\ j \geq 0}} (-1)^j \Tr((t,l), H_c^j(\tilde{S}_i)) \Tr(t, \theta) \Tr(l^{-1}, \pi).
\end{equation}

We now make the change of variable $xF(v) \mapsto x$, $x'F(u)^{-1} \mapsto x'$. Then $\tilde{S}_i$ becomes
\begin{equation*}
\{(x,x',v,n,u) \in V \times U \times F^{-1}(V) \times C_i \times F^{-1}(U) \mid xF(n) = vnx'\}.
\end{equation*}
The action of $T \times L$ is given by the same formula \eqref{eq:10.2} as in the old coordinates. Next, we note that the same formula \eqref{eq:10.2} defines an action of the group
\begin{equation*}
H_i = \{(t,l) \in Z \times Z' \mid Z^{-1}F(l) = F(n_i)^{-1} t^{-1}F(t) F(n_i) t\}
\end{equation*}
on $\tilde{S}_i$, where $Z$ (resp.\ $Z'$) is the identity component of the centre of $n_iLn_i^{-1}$ (resp.\ of $L$) and $n_i$ is any element of $C_i$. (Clearly, $H_i$ does not depend on the choice of $n_i$ in $C_i$.)

We shall now determine the fixed point set $(\tilde{S}_i)^{H_i^0}$ of the torus $H_i^0$ ($=$ identity component of $H_i$) on $\tilde{S}_i$.

By Lang's theorem, the homomorphism $H_i \to Z$, $((t,l) \mapsto t)$ is onto, hence its restriction $H_i^0 \to Z$ must be also onto since $Z$ is connected. Similarly, the homomorphism $H_i^0 \to Z'$, $((t,l) \mapsto l)$ is onto. Therefore, if $(x,x',v,n,u) \in (\tilde{S}_i)^{H_i^0}$, we must have
\begin{align*}
txt^{-1} = x, \quad &tvt^{-1} = v \quad \text{for all } t \in Z, \\
lx'l^{-1} = x', \quad &lul^{-1} = u \quad \text{for all } l \in Z'.
\end{align*}
Since $L$ is the centralizer of $Z'$ and $n_iLn_i^{-1}$ is the centralizer of $Z$, it follows that $x, v \in V \cap n_iLn_i^{-1}$ and $x', u \in U \cap L = \{1\}$. We then have $xF(n) = vn$, hence $F(n) = x^{-1}vn \in n_iLn_i^{-1}n$. Since $n \in C_i$, we have $F(n) \in n_iL$, hence $F(C_i) = C_i$. In particular, $(\tilde{S}_i)^{H_i^0}$ is empty, unless $F(C_i) = C_i$. Assume now that $F(C_i) = C_i$. Then $n_i \in C_i$ may be chosen so that $n_i \in G^{F}$. The equation defining $H_i$ may now be written: $F(ln_i^{-1}t^{-1}n_i) = ln_i^{-1}t^{-1}n_i$. It follows that
\begin{equation*}
H_i^0 = \{(t,l) \in Z \times Z' \mid l = n_i^{-1}tn_i\}
\end{equation*}
and
\begin{equation*}
(\tilde{S}_i)^{H_i^0} = \{(x,1,v,n_i,1) \mid x, v \in V \cap n_iLn_i^{-1},\ y \in L,\ y^{-1}F(y) = n_i^{-1}v^{-1}xn_i\}.
\end{equation*}

Since the action of $T \times L$ on $\tilde{S}_i$ commutes with the action of the torus $H_i^0$, expression \eqref{eq:10.3} is equal, by Lemma 9, to
\begin{equation*}
|T|^{-1} |L|^{-1} \sum_{\substack{t \in T \\ l \in L \\ j \geq 0}} (-1)^j \Tr((t,l), H_c^j((\tilde{S}_i)^{H_i^0})) \Tr(t, \theta) \Tr(l^{-1}, \pi).
\end{equation*}

In the case where $F(C_i) = C_i$, the map $(\tilde{S}_i)^{H_i^0} \to \{y \in L \mid y^{-1}F(y) \in n_i^{-1}Vn_i\}$ given by $(x,1,v,n_iy^{-1},1) \mapsto y$ is a locally trivial fibration all of whose fibres are affine spaces of the same dimension. The action of $(t,l)$ on $(\tilde{S}_i)^{H_i^0}$ corresponds to the action $y \mapsto lyn_i^{-1}tn_i$. Hence expression \eqref{eq:10.1}$(i)$ is equal to
\begin{equation*}
|T|^{-1} |L|^{-1} \sum_{\substack{t \in T \\ l \in L \\ j \geq 0 \\ F(C_i) = C_i}} (-1)^j \Tr((t,l), H_c^j(\{y \in L \mid y^{-1}F(y) \in n_i^{-1}Vn_i\})) \Tr(t, \theta) \Tr(l^{-1}, \pi)
\end{equation*}
which is clearly equal to the right hand side of (7.1). This completes the proof of Theorem 6.
