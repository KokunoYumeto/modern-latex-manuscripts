\section{Isogenies}

Recall that for a Drinfel'd $A$-module $(\phi, E)$ over $k$ and any field 
extension $k^*$ of $k$, the group $E(k^*)$ has a given $A$-module structure. 
For two Drinfel'd $A$-modules over $k$ and $u \in k\{\tau\}$ with $u \neq 0$, 
the additive $u\colon \phi \to \phi'$ is an isogeny if and only if the additive 
$u\colon E(\bar{k}) \to E'(\bar{k})$ is $A$-linear. This follows from the fact 
that two polynomials are equal if and only if their associated polynomial maps 
$\bar{k} \to \bar{k}$ are equal. The kernel of $u\colon E(k^*) \to E'(k^*)$ is 
the $A$-module $\ker(u)(k^*)$ where $\ker(u)$ is the scheme kernel of 
$u\colon \phi \to \phi'$.

\subsection*{4.1 Separable isogenies}

An isogeny $u\colon \phi \to \phi'$ is separable when 
$u(X) = b_0X + b_1X^p + \cdots + b_sX^{p^s}$ where $b_0 \neq 0$. Then $u$ is 
determined by $\ker(u)(\bar{k})$, a finite $A$-submodule of $E(\bar{k})$, up to 
constant factor. For a finite $A$-submodule $H \subset E(\bar{k})$, we form 
$P_H(X) = \prod_{h \in H}(X - h)$, and there is a unique Drinfel'd $A$-module 
$\phi'$ such that $\phi'_a P_H = P_H \phi_a$ for all $a \in A$.

If $a \cdot \ker(u)(\bar{k}) = 0$ for $a \neq 0$, then we can form the isogeny 
$v$ with kernel $u(\ker(\phi_a)(\bar{k}))$ and $vu = \phi_a$. Note that such 
a nonzero $a \in A$ with $a \cdot \ker(u)(\bar{k}) = 0$ always exists for 
$u \neq 0$ since it is a finite $A$-module. When $u$ is separable, we can 
always choose $v$ to be separable too. By the same construction, if 
$u\colon \phi \to \phi'$ and $w\colon \phi \to \phi''$ are separable isogenies 
such that $w(\ker(u)(\bar{k})) = 0$, then there is a separable isogeny 
$v\colon \phi' \to \phi''$ with $w = vu$ where 
$\ker(v)(\bar{k}) = u(\ker(w)(\bar{k}))$.

\subsection*{4.2 Purely inseparable isogenies}

These are of the form $\tau^i \in k\{\tau\}$ or $X \mapsto X^{p^i}$. Since 
$\phi_a \tau^i = \tau^i \phi_a$ in $k\{\tau\}$ is equivalent to all coefficients 
of $\phi_a$ being in $\mathbb{F}_{p^i}$, it follows that a purely inseparable 
isogeny exists only when the characteristic $v$ of $\phi$ is unequal to 
$\infty$, and in this case $p^i$ is a power of $q_v$ where $q_v$ is the 
cardinality of the residue class field of $v$. The finite group scheme 
$\ker(\tau^i) = \Spec(k[t]/(t^{p^i}))$. This case corresponds to a purely 
local kernel.

\subsection*{4.3 Remarks}

Let $H$ be a finite subgroup scheme of $\mathbb{G}_a$, and let $\phi$ be a 
Drinfel'd module structure on $\mathbb{G}_a$. Then $H$ is the kernel of some 
isogeny $u\colon \phi \to \phi'$ if and only if $H$ is stable under the action 
of $A$ and
\[
\ker(u) = \begin{cases}
0 & \text{if the characteristic} = \infty \\[2mm]
\Spec(k[t]/(t^{q_v^h})) & \text{if the characteristic} = v \text{ and } 
q_v^h = \Card(F_v) \,.
\end{cases}
\]

For any isogeny $u\colon \phi \to \phi'$ there exists an isogeny 
$v\colon \phi' \to \phi$ and $a \in A$ with $\phi_a = vu$.

Let $\delta(u)$ denote the degree of the additive polynomial 
$u \in \End_k(\mathbb{G}_a)$. The function $\delta\colon \End_k(\mathbb{G}_a) \to \mathbb{Z}$ 
also prolongs under extension of scalars to
\[
\begin{tikzcd}
\End_k(\mathbb{G}_a) \arrow[r] \arrow[d] & \End_{\bar{k}}(\mathbb{G}_a) \arrow[d, "\cong"] \\
\End_k(E) \otimes_A F \arrow[r] & \End_{\bar{k}}(E) \otimes_A F
\end{tikzcd}
\]

\subsection*{4.4 Remarks}

We study the $A$-algebra $\End_{k\text{alg}}(E)$ with $\delta$ using the 
following properties of $\delta\colon \End_{k\text{alg}}(E) \otimes_{\mathbb{F}_q} F_\infty \to \mathbb{Q}$:
\begin{enumerate}[label=(\arabic*)]
\item $\delta(u) \geqslant 0$ and $\delta(u) = 0$ if and only if $u = 0$.
\item $\delta(au) = |a|\delta(u)$ for $a \in F$.
\item $\delta(u+v) \leqslant \max(\delta(u), \delta(v))$.
\item $\delta(uv) = \delta(v)\delta(u)$.
\end{enumerate}
Moreover, $\End(E) \subset \End_{k\text{alg}}(E) \otimes_A F_\infty$ will be seen 
to be a discrete $A$-module in this normed vector space over $F_\infty$, see 
(4.9)(2). The main step will be to show that $\End(E)$ is a finitely generated 
$A$-module. For this we use the following two lemmas.

\begin{lemma}
Let $A^n \subset X \subset F_\infty^n$ where $X$ is a discrete $A$-module. Then 
$X$ is finitely generated.
\end{lemma}

\begin{proof}
For all $i$ observe that $X + \pi^{-i}A^n$ has finite index, e.g. 
$k[[t]]^n + t^{-i}k[[t]]^n \subset k((t))^n$ has index $\Card(k)^{ni}$. There 
exists an $i$ with $X \cap \pi^{-i}A^n = 0$ since $X$ is discrete, and thus 
$X/A^n$ embeds in $(F_\infty/\pi^{-i}A)^n$ which is finite. Since $(X : A^n)$ 
is finite and $A^n$ is finitely generated, it follows that $X$ is a finitely 
generated $A$-module.
\end{proof}

\begin{proposition}
For a finite dimensional subspace $V$ over $F$ of $\End(E) \otimes_A F$, it 
follows that $V \cap \End(E)$ is a finitely generated $A$-module which is 
projective.
\end{proposition}

\begin{proof}
For $V_\infty = F_\infty \cdot V$, it follows that $X = \End(E) \cap V = \End(E) \cap V_\infty$, and we can assume that $V_\infty$ generates $V$ and so $V_\infty$ or replace $V$ by a smaller subspace.

Let $x_1, \ldots, x_n$ be a basis of $V_\infty$ with $x_i \in X$. Then with this 
basis $A^n \subset X \subset F_\infty^n = V_\infty$, and we are reduced to the 
previous lemma since $\End(E)$ is a discrete subspace of 
$\End(E) \otimes_A F_\infty$. To see that $X = V \cap \End(E)$ is projective, 
we have only to remark that it is flat since it is torsion free over a 
Dedekind domain and finitely generated flat modules are projective over a 
Noetherian ring.
\end{proof}

\begin{corollary}
The $A$-module $\End(E)$ is projective.
\end{corollary}

\begin{proof}
Let $W = \End(E) \otimes_A F$ and $W = \bigoplus_i W_i$ where $[W_i : F]$ is 
finite. Then $X_i = \End(E) \cap W_i$ is projective by (4.6) and the 
restrictions of the projections $W \to W_i$ to $X_i = \End(E) \to X_i$ define 
a morphism $f\colon \End(E) \to \bigoplus_i X_i$ onto a projective module with 
$\ker(f) = 0$. Hence $\End(E)$ is projective.
\end{proof}

For estimates on the rank we use the following lemma.

\begin{lemma}
Let $a \in A$ be prime to the characteristic of $E$. Then 
$\End(E) \otimes_A A/a \to \End(E_a)$ is injective.
\end{lemma}

\begin{proof}
If $v \in \End(E)$ and $v(\ker(\phi_a)) = 0$, then as in (4.1), it follows that 
$v = v'\phi_a$, and $v|_{E_a} = 0$ implies $v \in \End(E)a$.
\end{proof}

Now we summarize all the basic results in the following theorem and remarks.

\begin{theorem}
Let $E$ be a Drinfel'd module over an algebraically closed field $\bar{k}$ of 
rank $r$. Then
\begin{enumerate}[label=(\arabic*)]
\item $\End(E)$ is a projective $A$-module of rank $\leqslant r^2$, and
\item $\End_{k\text{alg}}(E) \otimes_A F_\infty$ is a field in which $\End(E)$ 
embeds as a discrete $A$-module of this normed space over $F_\infty$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) The fact that $\End(E)$ is projective is contained in (4.7) and the 
injectivity of $\End(E) \otimes_A A/a \to \End(E_a)$ coming from (4.8) bounds 
the rank by $r^2$ since $E_a$ is an $A/a$-module of rank $r$ for $a$ prime 
to the characteristic.

(2) The existence of $\delta$ on $\End_{k\text{alg}}(E) \otimes_A F$ proves that 
it is a field, and since $F_\infty/F$ is a separable extension 
$\End_{k\text{alg}}(E) \otimes_A F_\infty$ is also a field.
\end{proof}

The subspace where $\delta = 0$ on $\End_{k\text{alg}}(E) \otimes_A F_\infty$ is 
zero since $\dim_F \End(E) \otimes_A F = \rank(\End_{k\text{alg}}(E))$.

\begin{remark}
With the notations of the previous theorem, the ring $\End_{k\text{alg}}(E)$ is 
commutative for $E$ of characteristic $\infty$ and further its rank 
$\leqslant r$.
\end{remark}

\begin{remark}
For a place $v$ of $F$ we denote by $D_v(E) = \bigcup_n E_{v^n}(\bar{k})$ and 
then the Tate module would be by definition
\[
T_v(E) = \Hom(F_v/A_v, D_v(E)) \,.
\]
Then $\End(E) \otimes_A A_v \subset \End T_v(E) = \End D_v(E)$, and the 
cokernel is without torsion as an $A_v$-module.
\end{remark}

\section{Drinfel'd modules over a base scheme}

Recall that locally free sheaf of rank 1 and invertible sheaf are the same 
notions, for which we use also the term line bundle.

\begin{definition}
Let $S$ be a scheme in characteristic $p$. A Drinfel'd module over $S$ of 
rank $r$ is an invertible sheaf $L$ and a morphism of rings 
$\phi\colon A \to \End(L)$ such that locally over open sets where $L$ is 
trivial, $\phi_a(x) = \sum_{i=0}^m a_i x^{p^i}$ where $p^m = |a|_\infty^r$ and 
$a_m$ is a unit. We say that $\phi_a$ is strictly of degree $p^m$.
\end{definition}

In order to analyze the condition that $a_m$ is a unit, we use the following 
lemma.

\begin{lemma}
Let $R$ be a ring with $p = 0$ and $\phi, \psi, f \in R\{\tau\}$ where 
$\phi = \sum a_i \tau^i$ with $a_N$ a unit, $\psi = \sum b_i \tau^i$ with $b_N$ 
a unit, and $f\phi = \psi f$, $\phi = f \psi$. If $f = \sum_{i=0}^M c_i \tau^i$. 
Then the leading coefficient $c_M^{p^N}$ is either a unit or zero, and so $f$ 
is either zero or strictly of degree $\leqslant M$.
\end{lemma}

\begin{proof}
Comparing the coefficients of $\tau^{M+N}$ in the relation 
$\psi f = f \phi$, we deduce
\[
c_M^{p^N} = c_M a_N^{p^M} \quad\text{or}\quad c_M^{p^N}(c_M^{p^N-1} - a_N b_N^{-1})^{p^M} = 0 \,.
\]
Since $b_N$ is a unit, this can be written
\[
c_M(c_M^{p^{N-1}} - a_N b_N^{-1})^{p^M} = 0 \,.
\]
Thus either $c_M = 0$ or $c_M$ is a unit with $c_M^{p^{N-1}} = a_N/b_N$.
\end{proof}

The next lemma is a version of the Weierstrass preparation theorem.

\begin{lemma}
Let $R$ be a ring with $p = 0$. If $\phi = \sum_{i=0}^N a_i \tau^i$ where $a_N$ 
is a unit and $a_0, \ldots, a_{N-1}$ are nilpotent, then after a change of 
coordinates with $u(T) = 1 + \sum_{i=1}^\infty c_i T^i$ such that
\[
c\phi c^{-1}(u(T)) = \sum_{i=0}^N b_i \tau^i \cdot (1 - w\tau)
\]
with $b_N$ a unit.
\end{lemma}

The coefficients $c_N, c_{N-1}, \ldots, c_1, c_0$ are chosen by decreasing 
induction.

\begin{remark}
An alternative form of the definition of a Drinfel'd module over a scheme $S$ 
in characteristic $p$ is to give $E/S$ a group scheme locally isomorphic to 
$\mathbb{G}_a$ and a morphism of rings $\phi\colon A \to \End_S(E)$ with 
$\deg(\phi_a) = |a|_\infty^r$ locally over $S$.

Using lemma (5.3), we can put an $\mathcal{O}_S$-linear structure on $E$ such 
that the action of some $\phi_a$ for $|a|_\infty > 1$ is locally given by a 
polynomial expression with highest coefficient a unit and of degree 
$|a|_\infty^r$. Using lemma (5.2) and the relation 
$\phi_a \phi_b = \phi_b \phi_a$, it follows that all $\phi_b$ are given locally 
by polynomials of strict degree $|b|_\infty^r$. Questions of 
$\mathcal{O}_S$-linearity need only be checked over closed subschemes 
$\Spec(R) \subset S$ where $R$ is a local Artin ring.
\end{remark}

\subsection*{5.5 Analogue of characteristic}

The function $a \mapsto \partial_0\phi_a$ defines a morphism $A \to \mathcal{O}_S$ 
of rings and hence a morphism of schemes $S \to \Spec(A)$. For a closed point 
$s \in S$, the Drinfel'd module $L_s$ over the field $F(s)$ has characteristic 
$s(s) \in \Spec(A)$.


\section{Level structure and the moduli space}

Let $I$ be an ideal in $A$, let $V(I)$ denote the set of prime ideals of $A$ 
containing $I$, and let $E/S$ be a Drinfel'd module of rank $r$ over $S$ with 
characteristic morphism $\theta\colon S \to \Spec(A)$. Then form the finite flat 
group scheme $E_I/S = \bigcap_{a \in I} \ker_a(E \to E)$ of rank equal to 
$\Card(A/I)^r$. The scheme $E_I$ is \'etale outside the characteristics which 
divide $I$, i.e. over $\theta^{-1}(\Spec(A) - V(I)) \subset S$.

In the most elementary sense an $I$-level structure on a Drinfel'd module $E$ 
should be an isomorphism $\alpha\colon (I^{-1}/A)^r \to E_I$ over $S$, and in 
fact, this definition works very well away from characteristics dividing $I$. 
In order to deal smoothly with characteristics dividing $I$, we are led to the 
following definition of Drinfel'd which has also an analogue for elliptic 
curves with level structures.

\begin{definition}
An $I$-level structure on a Drinfel'd module $E/S$ of rank $r$ is an 
$A$-linear morphism
\[
\alpha\colon (I^{-1}/A)^r \longrightarrow E_I
\]
such that for all $i$ in $I^{-1}/A$ the corresponding sections $\alpha(i)$ of 
$E_I$ have the property that as divisors on $E$
\[
\sum_{i \in (I^{-1}/A)^r} (\alpha(i)) = (E_I) \,.
\]
\end{definition}

Locally $E$ is isomorphic to $\mathbb{G}_a$, and in this case $E_I$ is 
defined as the kernel of a polynomial map $P\colon \mathbb{G}_a \to \mathbb{G}_a$ 
where
\[
P(X) = \prod_{i \in (I^{-1}/A)^r} (X - \alpha(i)) \,.
\]

Let $F_I^r$ denote the contravariant functor from schemes to sets which assigns 
to a scheme $S$ the set of isomorphism classes of Drinfel'd modules over $S$ of 
rank $r$ with $I$-level structure.

\begin{theorem}
Let $I$ be an ideal in $A$ with $\Card V(I) \geqslant 1$. The functor $F_I^r$ 
is representable by an affine scheme of finite type over $A$.
\end{theorem}

\begin{proof}
For $x \in V(I)$ it suffices to show that the functor restricted to the 
category of schemes over $\Spec(A) - \{x\}$ is representable. Then for a 
scheme $S$ over $\Spec(A) - \{x\}$, and a Drinfel'd module $E$ over $S$ with 
an $I$-level structure, the choice of nonzero elements for the $I$-level 
structure gives a trivialization of $E$. Over these local pieces $E$ is given 
by coordinates of $\phi_a$ for each $a \in A$ and elements $\alpha(i)$ for 
$i \in (I^{-1}/A)^r$ subject to the relations:
\begin{enumerate}[label=(\alph*)]
\item $\phi_a \phi_b = \phi_b \phi_a$.
\item The leading coefficient of $\phi_a$ is invertible.
\item $\phi_a(\alpha(i)) = 0$.
\item $P(X) = \prod_{i \in (I^{-1}/A)^r} (X - \alpha(i))$.
\end{enumerate}
All of these relations are affine in nature and can be represented by an 
affine scheme.
\end{proof}

The book of Katz--Mazur [1985] carries the construction of moduli spaces of 
elliptic curves using Drinfel'd's definition of level structure.


\section*{Analytic theory of the affine modules of Drinfel'd}

The group of points of an elliptic curve over the complex numbers is of the 
form $\mathbb{C}/L$ where $L$ is a lattice over $\mathbb{Z}$ (so of rank 2) in 
$\mathbb{C}$. There is a similar assertion for Drinfel'd modules of rank $r$. 
The group of $\mathbb{C}_\infty$-valued points is of the form 
$\mathbb{C}_\infty/L$ where $L$ is an $A$-lattice (discrete $A$-module) of 
rank $r$ in $\mathbb{C}_\infty$.

The description of Drinfel'd modules in terms of lattices gives a calculation 
of the moduli space of Drinfel'd modules over $\mathbb{C}_\infty$ in terms of a 
quotient $\GL(r,A) \backslash \Hom_{F_\infty}(F_\infty^r, \mathbb{C}_\infty)$. This 
is similar to the quotient $\GL(2,\mathbb{Z}) \backslash \Hom_\mathbb{R}(\mathbb{R}^2, \mathbb{C})$ 
which classifies elliptic curves plus a differential form. In the last section 
an ad\'elic description of the points of this moduli space is given.


\section{Exponential function associated to a lattice}

Let $X$ be a subset of a complete nonarchimedean value field $K$ such that 
$0 \in X$ and $B(a,r) \cap X$ is finite for any ball $B(a,r) \subset K$ around 
$a$ of radius $r > 0$. Then the Euler product
\[
e_X(t) = t \prod_{Y \in X - \{0\}} \Bigl(1 - \frac{t}{Y}\Bigr)
\]
defines an entire function $e_X\colon K \to K$ with zeros on $X$, because the 
hypothesis implies $\lim_{Y \in X - \{0\}} t/Y = 0$ from which we deduce that 
$e_X(t)$ converges uniformly on any ball.

By changing $t$ by scalar factors, we can assume that the power series 
expansion of an entire function has the form $f(t) = \sum_{n \geqslant 0} a_n t^n$ 
where $|a_i| \leqslant 1$, $|a_i| < 1$ for $i < r$, and $|a_r| = 1$; when $f$ 
is nonconstant, also $r > 0$. By solving congruences modulo powers of the 
maximal ideal of $F$, we can factor
\[
f(t) = (t^r + c_1 t^{r-1} + \cdots + c_r)(b_0 + b_1 t + \cdots) \,.
\]
If, in addition, $K$ is algebraically closed, then it follows that every 
nonconstant entire function has at least one zero. Thus the Euler product 
$e_X(t)$ is the unique entire function, up to a constant factor, with one 
simple zero at each point of $X$ and no other zeros.

\begin{definition}
A subgroup $L$ of the additive group of a complete value field $K$ is called 
a lattice provided the intersection $L \cap B(a,r)$ is finite for any ball 
$B(a,r)$.
\end{definition}

When $K$ is of characteristic $p$, then $pL = 0$ and the torsion group $L$ is 
a limit of finite subgroups $H$, i.e.
\[
L = \varinjlim_{H \text{ finite} \subset L} H \,.
\]

If the above Euler expansion $e_X(t) = t \prod_{Y \in X - \{0\}}(1 - t/Y)$ is 
additive, i.e. $e_X(x+y) = e_X(x) + e_X(y)$ for any $x,y \in K$, then $X$ is 
clearly a lattice and $e_X(t)$ is its exponential function.

\begin{proposition}
Let $L$ be a lattice in a complete value field $K$ of characteristic $p$. 
Then $e_L(x+y) = e_L(x) + e_L(y)$ for any $x, y \in K$.
\end{proposition}

\begin{proof}
Since $L = \varinjlim_{H \text{ finite} \subset L} H$, it follows that uniformly 
on any ball
\[
e_L(t) = \varinjlim_{H \text{ finite} \subset L} e_H(t) \,.
\]
Each $e_H(t)$ is additive since it equals $c \cdot \prod_{h \in H}(t-h)$ which 
is additive by 1(1.3). Thus $e_L(x+y) = e_L(x) + e_L(y)$ since it is a limit 
of functions $e_H(t)$ satisfying this property.
\end{proof}

\begin{remark}
If $L$ is a lattice in a complete value field $K$, and if $\lambda \in K^*$, 
then $\lambda L$ is a lattice also and the exponential function $e_L(t)$ of $L$ 
determines the exponential function of $\lambda L$ by 
$e_{\lambda L}(t) = \lambda e_L(\lambda^{-1}t)$. The lattice $\lambda L$ is 
called the dilation of $L$ by $\lambda \in K^*$.

Two lattices $L$ and $L'$ are in the same dilation class provided $L' = \lambda L$ 
for $\lambda \in K^*$. We always normalize the exponential function so that
\[
e_L(t) = t + \text{(higher order terms)} \,.
\]
\end{remark}


\section{Characterization of Drinfel'd modules over $\mathbb{C}_\infty$}

Lattices $\Gamma \subset \mathbb{C}_\infty$ such that $a\Gamma \subset \Gamma$ for 
$a \in A \subset \mathbb{C}_\infty$ with the induced scalar action by $a \in A$ 
are also $A$-modules, which we call $A$-lattices. A dilation $\lambda\Gamma$ of 
an $A$-lattice $\Gamma$ is again an $A$-lattice for $\lambda \in \mathbb{C}_\infty^*$, 
and we can speak of dilation classes of $A$-lattices.

Given an $A$-lattice $\Gamma$ in $\mathbb{C}_\infty$, we form the ring morphism 
$\phi_\Gamma\colon A \to \End_{\mathbb{C}_\infty}(\mathbb{G}_a)$ where 
$\phi_{\Gamma,a}$ for $a \in A$ is defined by the following commutative diagram
\[
\begin{tikzcd}
0 \arrow[r] & \Gamma \arrow[r] \arrow[d, "a"'] & \mathbb{C}_\infty \arrow[r, "e_\Gamma"] \arrow[d, "a"'] & \mathbb{C}_\infty \arrow[r] \arrow[d, "\phi_{\Gamma,a}"] & 0 \\
0 \arrow[r] & \Gamma \arrow[r] & \mathbb{C}_\infty \arrow[r, "e_\Gamma"'] & \mathbb{C}_\infty \arrow[r] & 0
\end{tikzcd}
\]

For $\lambda\Gamma$, the relation 
$\phi_{\lambda\Gamma}(e_{\lambda\Gamma}(t)) = e_{\lambda\Gamma}(at)$ becomes in 
view of (1.3) simply $\phi_{\lambda\Gamma}(\lambda e_\Gamma(\lambda^{-1}t)) = \lambda 
e_\Gamma(a\lambda^{-1}t)$, i.e. $\lambda^{-1}\phi_{\lambda\Gamma}(\lambda e_\Gamma(t)) 
= e_\Gamma(at)$. This implies that $\phi_{\lambda\Gamma} = \phi_\Gamma$ for 
$\lambda \in \mathbb{C}_\infty^*$.

\begin{theorem}
The function, which assigns to each $A$-lattice $\Gamma$, the ring morphism 
$\phi_\Gamma\colon A \to \End_{\mathbb{C}_\infty}(\mathbb{G}_a)$ defined above, is 
a bijection from the set of $A$-lattice dilation classes in 
$\mathbb{C}_\infty$ determined by $\Gamma$ which are projective of rank $r$ 
onto the set of isomorphism classes of affine $A$-modules $\phi_\Gamma$ over 
$\mathbb{C}_\infty$ of rank $r$ with $\partial\phi_{\Gamma,a} = a$.
\end{theorem}

\begin{proof}
First we calculate $\ker(\phi_{\Gamma,a})$. This is isomorphic to 
$\ker(\phi_{\Gamma,a} e_\Gamma)/\Gamma = \ker(e_\Gamma \cdot a)/\Gamma$, and 
$\ker(e_\Gamma \cdot a)/\Gamma$ is isomorphic to $a^{-1}\ker(e_\Gamma)/\Gamma 
= a^{-1}\Gamma/\Gamma$. Hence $\ker(\phi_{\Gamma,a})$ is isomorphic to 
$(A/a)^r$ since $\Gamma$ is projective of rank $r$, and moreover 
$\Card(\ker\phi_{\Gamma,a}) = \Card(A/a)^r$. Thus $\Gamma \mapsto \phi_\Gamma$ 
is a well defined function from dilation classes of $A$-lattices projective of 
rank $r$ to isomorphism classes of affine $A$-modules over $\mathbb{C}_\infty$ 
of rank $r$. The property $\partial\phi_{\Gamma,a} = a$ follows from 
$e_\Gamma(t) = t + \text{(higher terms)}$, and $|\phi_{\Gamma,a}| = |a|_\infty^r$ 
from the calculation preceding the statement of the theorem.

To show $\Gamma \mapsto \phi_\Gamma$ is a bijection, we consider an affine module 
$\phi\colon A \to \End_{\mathbb{C}_\infty}(\mathbb{G}_a)$ of rank $r$ and 
construct an additive entire function $e\colon \mathbb{C}_\infty \to \mathbb{C}_\infty$ 
such that $\phi_a(e(x)) = e(ax)$ and $e(t) = t + \text{(higher terms)}$. Then 
the corresponding $A$-lattice $\Gamma$ is the set of $x \in \mathbb{C}_\infty$ 
with $e(x) = 0$.

The first step is to show that there is a unique formal solution 
$e(T) \in k[[T]]$ such that $\phi_a(e(x)) = e(ax)$ for a given $a \in A$ with 
$\phi_a(T) = a_0 T + a_1 T^p + \cdots + a_m T^{p^m}$ and $a \cdot \partial_0 \neq 0, 1$. 
The condition $\phi_a(e(x)) = e(ax)$ is equivalent to 
$\phi_a(e(a^{-1}x)) = e(x) = \sum e_i x^{p^i}$. Equating coefficients of $x^n$, 
we derive from the relation
\[
e_n(a^{p^n} - a) = \sum_{i+j=n, i \geqslant 1} a_i e_j^{p^i}
\]
the formula for the coefficients $e_n$ starting with $e_0 = 1$:
\[
e_n(1 - a^{p^n-1}) = \sum_{i=1}^n a_i e_{n-i}^{p^i} \,.
\]
This shows both the existence and uniqueness of $e(t)$.

The second step is to show that for this $e(t)$, the relation 
$\phi_b(e(x)) = e(bx)$ holds for any $b \in A$. Consider
\[
\phi_b(e(b^{-1}ax)) = \phi_b \phi_a e_a^{-1}(x) = (\phi_a(\phi_b e_b^{-1})a^{-1})(x) \,.
\]
From the expression it follows that $e$ and $\phi_b e_b^{-1}$ both satisfy the 
relation $\phi_a e_a^{-1} = e$, and from the uniqueness assertion in step one, 
it follows that $\phi_b e = eb$ for any $b \in A$.

Next we have to show that $e(x) = \sum_{n \geqslant 0} e_n x^{p^n}$ is an entire 
function. Assume in the inductive definition of $e(t)$ that $|a| > 1$ and write 
the above recurrence formula for $e_n$ as
\[
e_n(a^{p^n} - a) = \sum_{i=1}^s a_i e_{n-i}^{p^i} \quad\text{for } n \geqslant s \,.
\]
Then $|a||e_n|^{p^n} \leqslant \max_{1 \leqslant i \leqslant s}\{|a_i|^{p^{-n}} r_{n-i}\}$ 
where $r_j = |e_j|^{p^j}$. For a such that $1/|a| < \theta < 1$ and any 
$n \geqslant \text{fixed } n_0$, each term $|a_i|^{p^{-n}} < 1 + \theta$, and it 
follows that
\[
r_n < \theta \cdot \max_{1 \leqslant i \leqslant s} r_{n-i} \,.
\]
Since $r_n < \theta^{n-n_0} \cdot \max\{r_0, \ldots, r_{n_0-1}\}/(1-\theta)$, we see 
that $r_n = |e_n|^{p^n} \to 0$ which proves that $e(x)$ is an entire function.

Finally, we calculate $\phi_a$ for this $e(x)$ where $\Gamma$ equals the subset 
of $x \in \mathbb{C}_\infty$ with $e(x) = 0$. We have
\[
\phi_a(x) = ax \prod_{\gamma \in a^{-1}\Gamma/\Gamma - \{0\}} \Bigl(1 - \frac{x}{e(\gamma)}\Bigr)
\]
since $\phi_a(x) = 0$ on $a^{-1}\Gamma/\Gamma$. Thus the degree of $\phi_a$ equals 
$\Card(a^{-1}\Gamma/\Gamma) = \Card(\Gamma/a\Gamma) = (\Card(A/a))^r$. This proves 
the theorem.
\end{proof}

For our future considerations $I$-level structures will play a basic role where 
$I$ is a nonzero ideal in $A$. This leads to the following definition.

\begin{definition}
Let $Y$ be a projective $A$-module of rank $r$. For a nonzero ideal 
$I \subset A$ an $I$-level structure is an isomorphism 
$\alpha\colon Y/IY \to (A/I)^r$ of $A$-modules (or free $A/I$-modules).
\end{definition}

An isomorphism $Y/IY \to (A/I)^r$ is equivalent to an isomorphism 
$I^{-1}Y/Y \to (I^{-1}/A)^r$ since $Y = I^{-1} \otimes_A Y$. In the case 
$Y = \Gamma$ an $A$-lattice of rank $r$ and the affine $A$-module 
$E = (\phi_\Gamma\colon A \to \End_{\mathbb{C}_\infty}(\mathbb{G}_a))$ corresponding 
to $\Gamma$ by (2.1), we calculated $\ker(\phi_{\Gamma,a}) = a^{-1}\Gamma/\Gamma$ 
and thus the subfunctor $E_I = \bigcap_{a \in I} \ker(\phi_{\Gamma,a}) \subset E$ 
has the form
\[
E_I(\mathbb{C}_\infty) = \bigcap_{a \in I} a^{-1}\Gamma/\Gamma = I^{-1}\Gamma/\Gamma \,.
\]
Thus we immediately deduce the following assertion.

\begin{proposition}
Using $E_I(\mathbb{C}_\infty) = I^{-1}\Gamma/\Gamma$, we have that an $I$-level 
structure on the affine $A$-module $E = \phi_\Gamma$ is the same as an $I$-level 
structure on the projective $A$-module $\Gamma$.
\end{proposition}


\section{Discrete modules in a vector space over a local field}

The following preliminaries are needed in the next section in order to 
parametrize $A$-lattices in $\mathbb{C}_\infty$ as homogeneous spaces.

\begin{notations}
Let $A$ be a discrete subring of a local field $K$ with field of fractions 
$F \subset K$. We assume that $K/A$ is a compact abelian group.
\end{notations}

\begin{examples}
\begin{enumerate}[label=(\arabic*)]
\item $A = \mathbb{Z} \subset F = \mathbb{Q} \subset K = \mathbb{R}$.
\item $A = \mathbb{Z} + \mathbb{Z}\omega \subset F = \mathbb{Q}(\omega) \subset K = \mathbb{C}$ 
for $d > 0$, $d \in \mathbb{Z}$ square free.
\item $A = \mathbb{F}_q[t] \subset F = \mathbb{F}_q(t) = \mathbb{F}_q(t^{-1}) \subset K = \mathbb{F}_q((t^{-1}))$.
\item $A = \mathbb{F}_q[C - \infty] \subset F = \mathbb{F}_q(C - \infty) = \mathbb{F}_q(C) \subset K = F_\infty$ 
where $C$ is an affine curve and $F_\infty$ is the completion of $F$ at $\infty$.
\end{enumerate}
Of course (3) is a special case of (4), and (4) is the case of interest in 
this part.
\end{examples}

For any finite dimensional vector space $V$ over $K$, the topology is well 
defined and given by a norm. A subgroup $H$ in $V$ is discrete provided there 
exists a neighbourhood $N'$ of $0$ in $V$ with $N' \cap H = 0$. If 
$N + N \subset N'$, then $N + x \cap N + y = \emptyset$ for $x, y \in H$ if 
and only if $x \neq y$.

For an $A$-module $H$, the rank of $H$, denoted $\rank_A H$, is 
$\dim_F(F \otimes_A H)$.

\begin{proposition}
Let $H$ be a discrete $A$-module contained in a finite dimensional 
$K$-vector space $V$. Then
\[
\rank_A(H) \leqslant \dim_K(V) \,.
\]
\end{proposition}

\begin{proof}
Let $x_1, \ldots, x_n$ denote a set of elements in $H$ forming a basis of the 
vector space $H \cdot K \subset V$. Then
\[
L = Ax_1 \oplus \cdots \oplus Ax_n \subset H \subset V \,.
\]
Since $H$ is discrete in $V$, there exists a neighbourhood $N'$ of $0 \in V$ 
with $H \cap N' = \emptyset$ and a neighbourhood $N$ with $N + N \subset N'$. 
Now $N + L$ is a neighbourhood of $0$ in $V/L$ intersecting $H/L$ only at $0$. 
Thus $H/L$ is a discrete subgroup of $V/L$, and of the compact abelian group
\[
V/L = Kx_1 \oplus \cdots \oplus Kx_n / Ax_1 \oplus \cdots \oplus Ax_n \,.
\]
Thus $H/L$ is finite and $n = \dim_K(F \otimes_A L) = \dim_F(F \otimes_A H) = \dim_K(V)$ 
which in turn implies $\rank_A(H) = \dim_F(F \otimes_A H) \leqslant \dim_K(V)$. 
This proves the proposition.
\end{proof}

\begin{proposition}
Let $H$ be a projective $A$-submodule contained in a finite dimensional 
$K$-vector space $V$. Then $H$ is discrete in $V$ if and only if 
$\delta\colon K \otimes_A H \to V$ is injective.
\end{proposition}

\begin{proof}
Assume $H$ is discrete in $V$. Then $H \subset \im(\delta) \subset V$ is 
discrete and so $\rank_A(H) \leqslant \dim_K(\im(\delta))$ by (3.3). On the 
other hand, for any $H$, we have
\[
\rank_A(H) = \dim_K(K \otimes_A H) \geqslant \dim_K(\im(\delta)) \,.
\]
Hence $\dim_K(K \otimes_A H) = \dim_K(\im(\delta))$ and so $\delta$ is 
injective.

Conversely, assume that $\delta$ is injective. There exists $H'$ with 
$H \oplus H'$ free, and a free module $A^n \subset K^n$ is a discrete 
$A$-submodule. Hence $H$ is discrete in $K \otimes_A H$, and since $\delta$ 
is injective, $H$ is discrete in $V$. This proves the proposition.
\end{proof}

\begin{remark}
Let $\mathbb{C}$ be the completion of the algebraic closure of $K$. Then for 
every projective $A$-module $P$ with $\rank_A(P) \leqslant [\mathbb{C} : K]$ 
there exists an $A$-monomorphism $P \to \mathbb{C}$.
\end{remark}

such that $P$ is discrete in $\mathbb{C}$, i.e. by (3.4), the induced 
$K \otimes_k P \to \mathbb{C}$ is injective.

In example (1), $2 = [\mathbb{C} : K]$ and free modules $P$ of rank 
$\leqslant 2$ over $\mathbb{Z}$ can be embedded as discrete subgroups of 
$\mathbb{C} = \mathbb{C}$. In examples (3) and (4), 
$[\mathbb{C}_\infty : F_\infty] = \infty$ and all projective modules embed as 
$A$-lattices of $\mathbb{C}_\infty$.


\section{Moduli spaces as homogeneous spaces (local theory)}

Now we return to the notations of the article where $F = \mathbb{F}_q(C)$, 
$A = \mathbb{F}_q[C - \infty]$, $F_\infty$ completion of $F$ at $\infty$, and 
$\mathbb{C}_\infty$ the completion of the algebraic closure $\bar{F}_\infty$ of 
$F_\infty$. Let $K$ denote $F_\infty$ a local field as referred to in \S 3 
where $A \subset K$ is discrete and $K/A$ compact. We denote points in 
projective space $\mathbb{P}_{r-1}(\mathbb{C}_\infty)$ by their homogeneous 
coordinates $y_1 : \cdots : y_r$.

We begin by parametrizing $F$-vector subspaces or equivalently free 
$A$-lattices in $\mathbb{C}_\infty$.

\begin{proposition}
The function $f\colon \mathbb{C}_\infty^r \setminus (\Hom_{F_\infty}(F_\infty^r, \mathbb{C}_\infty) - \{0\}) \to \mathbb{P}_{r-1}(\mathbb{C}_\infty)$ 
given by $f(u) = u(e_1) : \cdots : u(e_r)$ where $e_i = (0, \ldots, 1, \ldots, 0)$ 
is a bijection which restricts to a bijection
\[
\mathbb{C}_\infty^r \setminus \Mon_F(F \otimes_A Y, \mathbb{C}_\infty) \longrightarrow \mathbb{P}_{r-1}(\mathbb{C}_\infty) - \bigcup(\text{$F_\infty$-rational hyperplanes}) \,.
\]
\end{proposition}

\begin{proof}
A nonzero element $u$ at $\Hom_{F_\infty}(F_\infty \otimes_A Y, \mathbb{C}_\infty)$ 
is determined by its values $(u(e_1), \ldots, u(e_r))$ on the canonical basis 
elements $e_i \in F_\infty^r$. A dilation of $u$ yields a dilation of the 
$r$-tuple $(u(e_1), \ldots, u(e_r)) = (x_1, \ldots, x_r)$. Next, 
$(a_1, \ldots, a_r) \in \ker(u) \subset F_\infty^r$ if and only if 
$\sum_{i=1}^r a_i x_i = 0$, i.e. for $(a_1, \ldots, a_r) \neq 0$, 
$(x_1, \ldots, x_r)$ is on the $F_\infty$-rational hyperplane with equation
\[
a_1 x_1 + \cdots + a_r x_r = 0 \,.
\]
This proves the proposition.
\end{proof}

For an $A$-module $Y$ projective of rank $r$, we have embeddings and 
isomorphisms $Y \to F_\infty \otimes_A Y \cong F_\infty^r$ and 
$\GL_A(Y) \to \GL_{F_\infty}(F_\infty \otimes_A Y) \cong \GL(r, F_\infty)$ 
coming from the tensor product. By taking the image of $Y$ in 
$F_\infty \otimes_A Y$, and using (3.4), the following horizontal arrows are 
well defined functions.

\begin{proposition}
Let $Y$ be a projective module of rank $r$ over $A$. The following diagram is 
commutative and the horizontal arrows are bijections:
\[
\begin{tikzcd}
\Mon_{F_\infty}(F_\infty \otimes_A Y, \mathbb{C}_\infty)/\GL_A(Y) \arrow[r] \arrow[d] & \left\{\substack{\text{$A$-sublattices of $\mathbb{C}_\infty$}\\\text{isomorphic to $Y$}}\right\} \arrow[d] \\
\mathbb{C}_\infty^* \backslash \Mon_{F_\infty}(F \otimes_A Y, \mathbb{C}_\infty)/\GL_A(Y) \arrow[r] & \left\{\substack{\text{$A$-sublattices of $\mathbb{C}_\infty$}\\\text{isomorphic to $Y$}\\\text{modulo dilations in $\mathbb{C}_\infty^*$}}\right\}
\end{tikzcd}
\]
\[
\begin{tikzcd}
\Mon_{F_\infty}(F_\infty \otimes_A Y, \mathbb{C}_\infty)/\GL(r,F) \arrow[r] \arrow[d] & \left\{\substack{\text{$r$-dimensional}\\\text{$F$-vector subspaces}\\\text{of $\mathbb{C}_\infty$}}\right\} \arrow[d] \\
\mathbb{C}_\infty^* \backslash \Mon_{F_\infty}(F \otimes_A Y, \mathbb{C}_\infty)/\GL(r,F) \arrow[r] & \left\{\substack{\text{$r$-dimensional $F$-vector}\\\text{subspaces of $\mathbb{C}_\infty$}\\\text{modulo dilations in $\mathbb{C}_\infty^*$}}\right\}
\end{tikzcd}
\]
\end{proposition}

Now putting (4.1) and (4.2) together along with (2.1), we obtain the 
following assertion. Let $\mathcal{P}_A^r$ denote the set of isomorphism classes 
of projective $A$-modules of rank $r$.

\begin{proposition}
Using the previous notations, we have the following bijections
\[
(\mathbb{P}_{r-1}(\mathbb{C}_\infty) - \bigcup(\text{$F_\infty$-rational hyperplanes}))/\GL_A(Y)
\]
\[
\uparrow
\]
\[
\mathbb{C}_\infty^* \backslash \Mon_{F_\infty}(F_\infty \otimes_A Y, \mathbb{C}_\infty)/\GL_A(Y) \longrightarrow \left\{\substack{\text{$A$-sublattices of $\mathbb{C}_\infty$}\\\text{isomorphic to $Y$}\\\text{modulo dilations in $\mathbb{C}_\infty^*$}}\right\},
\]
and taking disjoint unions
\[
\bigsqcup_{Y \in \mathcal{P}_A^r} (\mathbb{P}_{r-1}(\mathbb{C}_\infty) - \bigcup(\text{$F_\infty$-rational hyperplanes}))/\GL_A(Y)
\]
\[
\uparrow
\]
\[
\bigsqcup_{Y \in \mathcal{P}_A^r} \mathbb{C}_\infty^* \backslash \Mon_{F_\infty}(F_\infty \otimes_A Y, \mathbb{C}_\infty)/\GL_A(Y) \longrightarrow \left\{\substack{\text{$A$-sublattices of $\mathbb{C}_\infty$ of rank $r$}\\\text{modulo dilations in $\mathbb{C}_\infty^*$}}\right\}
\]
\[
\downarrow
\]
\[
\left\{\substack{\text{isomorphism classes of affine}\\\text{$A$-modules $\phi_a$ over $\mathbb{C}_\infty$ of}\\\text{rank $r$ with $\partial\phi_a = a$}}\right\}.
\]
\end{proposition}

\begin{remark}
In the case $r = 1$, $\mathbb{P}_0(\mathbb{C}_\infty) - \bigcup(\text{$F_\infty$-rational hyperplanes})$ 
is a point, and (4.3) is a statement about bijections between sets with one 
point. We will be interested particularly in the case $r = 2$ in latter parts. 
If $a_1 x_1 + a_2 x_2 = 0$ where $a_1, a_2 \in F_\infty$, then 
$x_1 : x_2 = a_2 : a_1 \in \mathbb{P}^1(F_\infty)$. Thus we have
\[
\mathbb{P}_1(\mathbb{C}_\infty) - (\text{$F_\infty$-rational lines}) = \mathbb{P}_1(\mathbb{C}_\infty) - \mathbb{P}_1(F_\infty) \,.
\]
\end{remark}

The next step is to modify (4.3) to include the case of $A$-lattices with 
$I$-level structure. For a projective $A$-module $Y$ of rank $r$ the 
projection $Y \to Y/IY$ includes a group morphism 
$\GL_A(Y) \to \GL_A(Y/IY)$ which has kernel $\GL_A(Y,I)$. In the following 
two statements we have a restatement of (4.2) and a modification taking into 
account $I$-level structures. Fix an $I$-level structure 
$\alpha_0\colon I^{-1}Y/Y \to (I^{-1}/A)^r$.
\begin{enumerate}[label=(\arabic*)]
\item Each monomorphism $u\colon F_\infty \otimes_A Y \to \mathbb{C}_\infty$ 
determines a lattice $u(Y) \subset \mathbb{C}_\infty$ and an $I$-level structure 
$\alpha = \alpha_0((u|_Y)^{-1} \bmod I)$ on $u(Y)$ where 
$u|_Y\colon Y \to u(Y) \subset \mathbb{C}_\infty$. Moreover, all $A$-sublattices 
of $\mathbb{C}_\infty$, isomorphic to $Y$ together with an $I$-level structure, 
come by this construction.
\item Two monomorphisms $u, u'\colon F_\infty \otimes_A Y \to \mathbb{C}_\infty$ 
determine the same lattice in $\mathbb{C}_\infty$ if and only if there exists 
$h \in \GL_A(Y)$ with $u' = uh$. They determine the same lattice in 
$\mathbb{C}_\infty$ with $I$-level structure if and only if there exists 
$h \in \GL_A(Y,I)$ with $u' = uh$.
\end{enumerate}

This leads to the following modification of the previous proposition.

\begin{theorem}
Let $I$ be a nonzero ideal in $A$, and use the previous notations. We have 
the following bijections:
\[
\bigsqcup_{Y \in \mathcal{P}_A^r} (\mathbb{P}_{r-1}(\mathbb{C}_\infty) - \bigcup(\text{$F_\infty$-rational hyperplanes}))/\GL_A(Y,I)
\]
with $I$-level structure modulo $\Aut$ of $Y$, and
\[
\bigsqcup_{Y \in \mathcal{P}_A^r} \mathbb{C}_\infty^* \backslash \Mon_{F_\infty}(F_\infty \otimes_A Y, \mathbb{C}_\infty)/\GL_A(Y,I) \longrightarrow \left\{\substack{\text{$A$-sublattices of $\mathbb{C}_\infty$}\\\text{of rank $r$ with an $I$-level}\\\text{structure modulo dilations}\\\text{in $\mathbb{C}_\infty^*$}}\right\}
\]
with $\alpha\colon I^{-1}Y/Y \to (I^{-1}/A)^r$ modulo $\Aut$ of $Y$, and
\[
\left\{\substack{\text{isomorphism classes of affine $A$-modules $\phi_a$}\\\text{over $\mathbb{C}_\infty$ with $I$-level structure and of}\\\text{rank $r$ with $\partial\phi_a = a$}}\right\}
\]
or more briefly the bijection
\[
\bigsqcup_{(\alpha,Y) \in \mathcal{P}_A^r(I)} \Omega^r(\mathbb{C}_\infty)/\GL_A(Y,I) \cong \mathcal{M}_I^r(\mathbb{C}_\infty) \,.
\]
Here $\Omega^r(\mathbb{C}_\infty) = \mathbb{P}_{r-1}(\mathbb{C}_\infty) - \bigcup(\text{$F_\infty$-rational hyperplanes})$, 
also $\mathcal{M}_I^r$ is the moduli functor for affine $A$-modules rank $r$ 
with $I$-level structure, and $\mathcal{P}_A^r(I)$ is the set of ordered pairs 
$(\alpha,Y)$ with $Y \in \mathcal{P}_A^r$ and $\alpha\colon I^{-1}Y/Y \to (I^{-1}/A)^r$ 
is an isomorphism.
\end{theorem}


\section{Moduli spaces as homogeneous spaces (ad\'elic theory)}

The basic bijection describing $\mathcal{M}_I^r(\mathbb{C}_\infty)$ in (4.5) 
involves only the local field at $\infty$. In this section we will describe 
$\mathcal{M}_I^r(\mathbb{C}_\infty)$ using all the primes of $F$ together with 
$\mathcal{M}_I^r(\mathbb{C}_\infty)$. This ad\'elic description is closely 
related to the ad\'elic description of vector bundles on a curve which we 
recall now.

\subsection*{5.1 Remarks on vector bundles}

Over any ring space $X$, a vector bundle $E$ of rank $n$ is a (locally free) 
sheaf on $X$ locally isomorphic to $\mathcal{O}_X^n$. If $X$ is a nonempty open 
set of the curve $C$ over $\mathbb{F}_q$, for example $X = C - \{\infty\}$, 
then a vector bundle of rank $n$ can be described as a family $(V, L_v)_{v \in X}$ 
where $V$ is an $n$-dimensional vector space over $F = \mathbb{F}_q(C)$ and each 
$L_v$ is a free module of rank $n$ contained in $V$ over the local ring 
$\mathcal{O}_{(v)}$ at $v$. We assume that there exists a basis 
$x_1, \ldots, x_n$ of $V$ over $F$ and a finite set $S \subset X$ with
\[
L_v = \mathcal{O}_{(v)}x_1 + \cdots + \mathcal{O}_{(v)}x_n
\]
for all $v \in X - S$. The basis $x_1, \ldots, x_n$ is called a trivialization 
of $V$ on $X - S$. If $y_1, \ldots, y_n$ is any other basis of $V$, then
\[
y_j = \sum_i a_{i,j} x_i \quad\text{and}\quad x_i = \sum_j b_{j,i} y_j
\]
where $(a_{i,j}), (b_{i,j}) \in \GL(n, \mathcal{O}_{(v)})$ for all but a finite 
number of $v$. Hence every basis of $V$ is a trivialization of $V$ over some 
$X - T$ where $T \subset X$ is finite.

To an $\mathcal{O}_X$-sheaf $E$, locally isomorphic to $\mathcal{O}_X^n$, we assign 
$V = E_\eta$ where $\eta$ is the general point of $C$. The 
$\mathcal{O}_{(v)}$-submodule $L_v \subset V$ is defined by trivializing 
$E|(X-S)$ on an open set $X-S$ around $v$ with $x_1, \ldots, x_n \in V$ and 
requiring for all $w \in X - S$ with $S$ finite that
\[
L_w = \mathcal{O}_{(w)}x_1 + \cdots + \mathcal{O}_{(w)}x_n \,.
\]

These lattices which came up in the above description of a vector bundle can 
be identified with certain homogeneous spaces. Let $F_v$ denote the completion 
of $F$ at $v \in X$ with valuation ring $\mathcal{O}_v$. Then using matrix 
transform between basis of a lattice in a vector space $F^n$ or $F_v^n$, we 
have the following commutative diagram of bijections:
\[
\begin{tikzcd}
\GL(n,F)/\GL(n,\mathcal{O}_{(v)}) \arrow[r, "\cong"] \arrow[d] & \{\mathcal{O}_{(v)}\text{-lattices in }F^n\} \arrow[d] \\
\GL(n,F_v)/\GL(n,\mathcal{O}_v) \arrow[r, "\cong"] & \{\mathcal{O}_v\text{-lattices in }F_v^n\}
\end{tikzcd}
\]

\subsection*{5.2 Homogeneous space description of trivialized vector bundles}

A vector bundle $E$ with a trivialization is described by $(F^n, L_v)$ for 
$v \in X$ where $L_v = \mathcal{O}_{(v)}^n \subset F^n$ for all $v \in X - S$ 
where $S$ is finite. Thus the vector bundle trivialized on $X - S$ is 
determined by a finite set of lattices $L_v \subset F^n$ for $v \in S$. This 
leads to the following bijections:
\[
\begin{tikzcd}
\prod_{v \in S} \GL(n,F_v)/\GL(n,\mathcal{O}_v) \arrow[r, "\cong"] & \left\{\substack{\text{trivialized over $X-S$}\\\text{$n$-dimensional vector bundles on $X$}\\\text{all up to isomorphism}}\right\} \\
\prod_{v \in S} \GL(n,F_v) \times \prod_{v \in X-S} \GL(n,\mathcal{O}_v)/\GL(n,\mathcal{O}_v) \arrow[u, "\cong"]
\end{tikzcd}
\]
where the vertical arrow is a homeomorphism of locally compact spaces.

At points of $S$ we can consider a $\Lambda$-level structure on the vector 
bundle $E$ relative to a positive divisor $\Lambda$ supported on $S$. This is 
just an isomorphism of the sheaf $E|_\Lambda(-\Lambda) = E \otimes_{\mathcal{O}_C} (\mathcal{O}_C/\mathcal{O}_C(-\Lambda)) \to (\mathcal{O}_C/\mathcal{O}_C(-\Lambda))^n$. 
In terms of the description of $E$ with the data $(F^n, L_v)$ this is an 
isomorphism
\[
L_v/\mathfrak{m}_v^{\lambda_v} L_v \xrightarrow{\;\cong\;} (\mathcal{O}_v/\mathfrak{m}_v^{\lambda_v})^n
\]
for each $v \in S$ where $\mathfrak{m}_v$ is the maximal ideal of 
$\mathcal{O}_{(v)}$ and hence $\mathcal{O}_v$. If we denote by
\[
\GL(n,\mathcal{O}_v,\Lambda_v) = \ker(\GL(n,\mathcal{O}_v) \to \GL(n,\mathcal{O}_v/\mathfrak{m}_v^{\lambda_v}))
\]
and $K(n,\Lambda) = \prod_{v \in X} \GL(n,\mathcal{O}_v,\Lambda_v)$, then the 
previous diagram becomes modified as follows for $\supp(\Lambda) \subset S$:
\[
\begin{tikzcd}
\prod_{v \in S} \GL(n,F_v)/\GL(n,\mathcal{O}_v,\Lambda_v) \arrow[r] & \left\{\substack{\text{trivialized over $X-S$}\\\text{$n$-dimensional vector bundles}\\\text{on $X$ with $\Lambda$-level structure}}\right\} \\
\prod_{v \in S} \GL(n,F_v) \times \prod_{v \in X-S} \GL(n,\mathcal{O}_v)/K(n,\Lambda) \arrow[u]
\end{tikzcd}
\]

Now we consider two special cases for $X$ namely $X = C$ and 
$X = C - \{\infty\}$ and remove the condition that the vector bundle is 
trivialized.

\subsection*{5.3 Vector bundles on $C$}

The product in the numerator of $\Vect_S(X)$ or $\Vect_S(X;\Lambda)$ is one of 
the terms in the inductive limit which defines the ad\`ele group 
$\GL(n,\mathbb{A}_F)$ where
\[
\GL(n,\mathbb{A}_F) = \varinjlim_{S ||} \prod_{v \in S} \GL(n,F_v) \times \prod_{v \in C-S} \GL(n,\mathcal{O}_v) \,.
\]
Hence considering vector bundles with $\Lambda$-level structure and 
trivialization over some open set $C - S$ of $C$ where 
$\supp(\Lambda) \subset S$, we obtain the bijection
\[
\begin{tikzcd}
\GL(n,\mathbb{A}_F)/K(n,\Lambda) \arrow[r, "\cong"] & \left\{\substack{\text{vector bundles of dimension $n$}\\\text{with a $\Lambda$-level structure and}\\\text{trivialization off the $\supp(\Lambda)$}\\\text{all up to isomorphism}}\right\}
\end{tikzcd}
\]
Since the various trivializations are all related by the action of 
$\GL(n,F)$ on basis, we have the following quotient of the previous bijection
\[
\begin{tikzcd}
\GL(n,F) \backslash \GL(n,\mathbb{A}_F)/K(n,\Lambda) \arrow[r, "\cong"] & \left\{\substack{\text{isomorphism classes of}\\\text{$n$-dimensional vector bundles}\\\text{on $C$ with $\Lambda$-level structure}}\right\}
\end{tikzcd}
\]

\subsection*{5.4 Vector bundles on $C - \{\infty\}$}

The product in the numerator of $\Vect_S(X)$ or $\Vect_S(X;\Lambda)$ for 
$X = C - \{\infty\}$ is one of the terms in the inductive limit which defines 
the finite ad\`ele group $\GL(n,\mathbb{A}_F^f)$ where 
$\mathbb{A}_F^f = \widehat{\prod}_{v \neq \infty} F_v$ and 
$\widehat{A} = \varprojlim_{J} A/J$, the limit being taken over ideals 
$J \subset A$. A divisor $\Lambda$ can be described as an ideal $I \subset A$, 
and the group $K(n,\Lambda)$ is given by
\[
K(n,\Lambda) = \ker(\GL(n,\widehat{A}) \to \GL(n,\widehat{A}/I))
\]
which we also denote by $\GL(n,\widehat{A},I)$ to avoid confusion with the 
previous case discussed in (5.3). As before we obtain in the injective limit 
the bijection
\[
\begin{tikzcd}
\GL(n,F) \backslash \GL(n,\mathbb{A}_F^f)/\GL(n,\widehat{A},I) \arrow[r, "\cong"] & \left\{\substack{\text{vector bundles of dimension $n$ on}\\\text{$C - \{\infty\}$ with a $\Lambda$-level structure}\\\text{and trivialization off the $\supp(\Lambda)$}\\\text{all up to isomorphism}}\right\}
\end{tikzcd}
\]

On the affine $C - \{\infty\}$ with coordinate ring 
$A = H^0(C - \{\infty\}, \mathcal{O}_C)$ a vector bundle of dimension $n$ is 
just a projective module $Y$ of rank $n$ over $A$.

Now we combine the previous bijection with
\[
\mathbb{C}_\infty^* \backslash \Mon_{F_\infty}(F_\infty \otimes_A Y, \mathbb{C}_\infty)/\GL_A(Y) \cong \Omega^r(\mathbb{C}_\infty) = \mathbb{P}_{r-1}(\mathbb{C}_\infty) - \bigcup\{\text{$F_\infty$-rational hyperplanes}\} \,.
\]
We have a bijection
\[
\begin{tikzcd}
\left\{\substack{\text{vector bundles $Y$ on $C-\infty$ of}\\\text{rank $r$ with a trivialization}\\\text{and a $\Lambda$-level structure all up}\\\text{to isomorphism}}\right\} \times \Omega^r(\mathbb{C}_\infty) \arrow[r, "\cong"] & \left\{\substack{\text{isomorphism classes of $A$-lattices}\\\text{in $\mathbb{C}_\infty$ of rank $r$ with}\\\text{$I$-level structure modulo dilations}}\right\}
\end{tikzcd}
\]

The trivialization of $Y$ is a basis of $Y \otimes_A F$ over $F$ which defines 
an embedding $Y \to F_\infty^r$ as a discrete $A$-submodule. When this 
embedding is composed with a monomorphism $F_\infty^r \to \mathbb{C}_\infty$ yields 
an $A$-sublattice of $\mathbb{C}_\infty$ of rank $r$ with an $I$-level structure 
modulo dilations in $\mathbb{C}_\infty^*$. Hence by (4.5) we have a map
\[
(\GL(r,\mathbb{A}_F^f)/\GL(r,\widehat{A},I)) \times \Omega^r(\mathbb{C}_\infty) \to \mathcal{M}_I^r(\mathbb{C}_\infty)
\]
where the fibres correspond to the various bases of $Y \otimes_A F$ over $F$. 
Hence factoring out by the action of $\GL(r,F)$ yields the ad\'elic 
description of $\mathcal{M}_I^r(\mathbb{C}_\infty)$.

\begin{theorem}
The above map induces a bijection
\[
\GL(r,F) \backslash \GL(r,\mathbb{A}_F^f) \times \Omega^r(\mathbb{C}_\infty)/\GL(r,\widehat{A},I) \xrightarrow{\;\cong\;} \mathcal{M}_I^r(\mathbb{C}_\infty) \,.
\]
\end{theorem}

\begin{remark}
The quotient space descriptions in (5.5) and (5.6) generalize to any open 
compact subgroup $H \subset \GL(r,\widehat{A})$ to give the $\mathbb{C}_\infty$-valued 
points $\mathcal{M}_H^r(\mathbb{C}_\infty)$ of a moduli scheme $\mathcal{M}_H^r$:
\[
\mathcal{M}_H^r(\mathbb{C}_\infty) = \GL(r,F) \backslash \GL(r,\mathbb{A}_F^f) \times \Omega^r(\mathbb{C}_\infty)/H \quad\text{ad\'elic version}
\]
\[
= \bigsqcup_{Y \in \mathcal{P}_A^r} \Omega^r(\mathbb{C}_\infty)/(x_Y^{-1}Hx_Y \cap \GL(r,F)) \quad\text{local version} \,.
\]
The moduli scheme $\mathcal{M}_H^r$ arises from $E$-level structures on 
Drinfel'd modules.
\end{remark}


\section*{The rigid analytic moduli space}

Now our aim is to describe the rigid analytic structure on the 
$\mathbb{C}_\infty$-valued points of the moduli space. These points were 
parametrized by a quotient of $\Omega^r(\mathbb{C}_\infty)$ by a discrete group 
in (2.4.5). The rigid analytic structure on the moduli space is a quotient 
of the rigid analytic structure on $\Omega^r(\mathbb{C}_\infty)$.

In order to define the rigid analytic structure on $\Omega^r(\mathbb{C}_\infty)$, 
we make use of a natural mapping
\[
\lambda\colon \Omega^r(\mathbb{C}_\infty) \longrightarrow I(F_\infty^r)_\mathbb{Q}
\]
onto the rational points $I(F_\infty^r)_\mathbb{Q}$ of the geometric realization 
$I(F_\infty^r)_\mathbb{R}$ of the building $I(F_\infty^r)$ of the group 
$\PGL(r)$ over the local field $F_\infty$. The admissible open sets of the 
rigid analytic structure are inverse images by $\lambda^{-1}$ of certain open 
neighborhoods of skeletons in $I(F_\infty^r)_\mathbb{R}$.

In this discussion we indicate how both $\Omega^r(\mathbb{C}_\infty)$ and the 
building $I(F_\infty^r)$ are $p$-adic analogues of real symmetric spaces.


\section{Norms on vector spaces over a local field}

\begin{notations}
Let $K$ be a local field with valuation ring $R$, maximal ideal 
$\mathfrak{p} = R\pi$, and $q = \Card(k)$ where $k = R/\mathfrak{m}$. We 
normalize $|a|$ on $K$ as $|a| = q^{-\ord(a)}$ so that $|\pi| = q^{-1}$ and 
$|\pi^i| = q^{-i}$. We consider finite dimensional vector spaces over $K$, 
and thus the closed unit ball and unit sphere are compact. If $V$ is of 
dimension $m$ over $K$, then a lattice is any $R$-submodule $M$ free of rank 
$m$.
\end{notations}

\begin{definition}
A norm on a vector space $V$ over $K$ is a function $\alpha\colon V \to \mathbb{R}$ 
satisfying:
\begin{enumerate}[label=(\alph*)]
\item $\alpha(x) \geqslant 0$ and $\alpha(x) = 0$ if and only if $x = 0$.
\item $\alpha(ax) = |a|\alpha(x)$ for $a \in K$ and $x \in V$.
\item $\alpha(x+y) \leqslant \sup(\alpha(x), \alpha(y))$ for $x, y \in V$.
\end{enumerate}
A norm is called integral provided $\alpha(V) = |K| = \{0\} \cup q^\mathbb{Z}$ 
and rational provided $\alpha(V) \subset \mathbb{Q}$.
\end{definition}

If $\alpha(x) < \alpha(y)$, then axiom (c) can be strengthened to the equality
\[
\alpha(x+y) = \alpha(y) = \sup(\alpha(x), \alpha(y)) \,.
\]
If $\alpha$ is a norm and $t > 0$, then the dilation $t\alpha$ is a norm, and 
we denote by $N(V)$ the set of dilation classes of norms on $V$.

\begin{example}[(I)]
For a basis $x_0, \ldots, x_m$ of $V$ and real numbers $r_0 > 0, \ldots, r_m > 0$ 
the function
\[
\alpha(a_0 x_0 + \cdots + a_m x_m) = \sup(r_0|a_0|, \ldots, r_m|a_m|)
\]
is a norm on $V$. In fact, every norm can be described by this formula. This 
is proved by induction on $m$ by considering a nonzero linear form 
$\lambda\colon V \to F$ and choosing $x_0 \neq 0$ in $V$ such that 
$x \mapsto |\lambda(x)|/\alpha(x)$ defined on the compact projective space 
$\mathbb{P}(V)$ takes its maximum at $x_0$. Then $V \cap Fx_0 \oplus \ker(\lambda)$ 
for which on $\ker(\lambda)$ the inductive hypothesis applies. See Goldman and 
Iwahori for the details.
\end{example}

\begin{example}[(II)]
The norm $\alpha_M$ associated to a lattice $M$ in $V$ is given by the formula 
$\alpha_M(x) = \inf\{1/|a| : ax \in M\}$.

Observe that $M$ is the unit $\alpha_M$-ball in $V$ of all $x \in V$ with 
$\alpha_M(x) \leqslant 1$, and for $x \neq 0$ we have $\alpha_M(x) = q^m$ where 
$\mathfrak{p}^m = \{a \in F : ax \in M\}$. For two lattices $L$ and $M$ the 
inclusion $L \subset M$ holds if and only if $\alpha_L(x) \geqslant \alpha_M(x)$ 
for all $x \in V$.

For nonzero $c \in F$ we have $\alpha_{cM}(x) = (1/|c|)\alpha_M(x)$ for all 
$x \in V$, and in particular, $\alpha_{\pi M}(x) = q\alpha_M(x)$. Finally, if 
$x_0, \ldots, x_m$ is a basis of a lattice $M$ over $R$, then
\[
\alpha_M(a_0 x_0 + \cdots + a_m x_m) = \sup\{|a_0|, \ldots, |a_m|\} \,.
\]
Thus this example is a special case of (1.4).
\end{example}

Going back to $\alpha(a_0 x_0 + \cdots + a_m x_m) = \sup(r_0|a_0|, \ldots, r_m|a_m|)$, 
we see that replacing $x_i$ by $\pi^{s_i}x_i$ for $s(i)$ replaces $r_i$ by 
$q^{s_i}r_i$. In particular, $x_i$ replaced by $\pi^{-1}x_i$ leads to $r_i$ 
replaced by $qr_i$. Hence, by rescaling the basis vectors, we can always 
require the constants $r_i$ to be in an interval of the form $[t, qt)$ or 
$(t, qt]$ for some given $t > 0$. So when $\alpha(V) = \{0\} \cup q^\mathbb{Z}$, 
we can choose the $x_i$ with each $r_i = 1$, and this gives the next 
proposition.

\begin{proposition}
The following are equivalent for $t > 0$ and a norm $\alpha$ on a vector 
space $V$:
\begin{enumerate}[label=(\arabic*)]
\item $\alpha = t \cdot \alpha_H$ for some lattice $H$,
\item $\alpha(V) = |F| \cdot t$, and
\item $\alpha(a_0 x_0 + \cdots + a_m x_m) = t\sup(|a_0|, \ldots, |a_m|)$ for some 
basis $x_0, \ldots, x_m$ of $V$.
\end{enumerate}
\end{proposition}

Now we wish to study to what extent the representation
\[
\alpha(a_0 x_0 + \cdots + a_m x_m) = \sup(r_0|a_0|, \ldots, r_m|a_m|)
\]
for a norm $\alpha$ is unique by renormalizing and reordering so that
\[
q \geqslant r_0 \geqslant r_1 \geqslant \cdots \geqslant r_{m-1} \geqslant r_m \geqslant 1 \,.
\]
Then the basis elements are unique up to multiplication by $\pi^{s_i}$ and 
cyclic permutation. The requirement that $r_m \geqslant 1$ removes this 
ambiguity, and further after a dilation of $\alpha$ we can assume that 
$r_0 = q$. The set of values $\alpha(V) = \{0\} \cup q^{\mathbb{Z}}r_0 \cup \cdots \cup q^{\mathbb{Z}}r_m$, 
and thus the numbers $r_i$ are uniquely determined by $\alpha$. They make up 
the set $\alpha(V) \cap (1,q]$ when we require $r_m \geqslant 1$.

With these notations, consider the lattices $L_i$ for $i = m, m-1, \ldots, 0$ 
where
\[
L_i = Rx_0 + \cdots + Rx_i + \mathfrak{p}x_{i+1} + \cdots + \mathfrak{p}x_m \quad\text{for } r_i > r_{i+1},
\]
and $L_m = Rx_0 + \cdots + Rx_m$.

\begin{lemma}
The lattice $L_i$ is the open ball $B(0, qr_i)$.
\end{lemma}

\begin{proof}
For $x = \sum a_j x_j \in V$ clearly $x \in L_i$ if and only if $|a_j| \leqslant 1$ 
if $j \leqslant i$ and $|a_j| < q$ if $j > i$, or equivalently, $r_j|a_j| \leqslant r_j$ 
for $j \leqslant i$ and $r_j|a_j| \leqslant qr_j$ for $j > i$.

Now assume that $r_i > r_{i+1}$, and we see that both inequalities are 
equivalent to $r_i|a_i| \leqslant r_i q$. Hence, $x \in L_i$ if and only if 
$\alpha(x) < qr_i$. If $r_i = r_{i+1}$, then $L_i = L_{i+1}$, and so the 
result holds by induction from $m$ to $0$.
\end{proof}

Finally, we obtain the following structure theorem.

\begin{theorem}
Let $\alpha$ be a norm on a vector space $V$ with $\alpha(V) \cap (1,q] = S$, 
and for $r \in S$, let $L(r) = B(0,qr)$ an open ball. Then 
$\dim V \geqslant \Card(S)$ and $\alpha = \sup_{r \in S}(r\alpha_{L(r)})$.
\end{theorem}

\begin{proof}
As in (1.3), we can choose a basis $x_0, \ldots, x_m$ of $V$ with
\[
\alpha(a_0 x_0 + \cdots + a_m x_m) = \sup(r_0|a_0|, \ldots, r_m|a_m|)
\]
and $q \geqslant r_0 \geqslant r_1 \geqslant \cdots \geqslant r_m \geqslant 1$. The $r_i$ exhaust 
the set $S$ with possible repetitions and thus $\dim V \geqslant \Card(S)$.

By (1.6) the lattice $L_i = L(r_i)$ for $r_i > r_{i+1}$ is given by
\[
L_i = Rx_0 + \cdots + Rx_i + \mathfrak{p}x_{i+1} + \cdots + \mathfrak{p}x_m \,.
\]
Then $\alpha_{L_i}$ is given by
\[
\alpha_{L_i}(a_0 x_0 + \cdots + a_m x_m) = \sup(|a_0|, \ldots, |a_i|, q^{-1}|a_{i+1}|, \ldots, q^{-1}|a_m|) \,,
\]
and since $r_i q^{-1}|a_i| \leqslant r_{i+1}|a_{i+1}|$, it follows that
\[
\alpha(x) = \sup(r_0\alpha_{L_0}(x), \ldots, r_m\alpha_{L_m}(x)) = \sup_{r \in S}(r\alpha_{L(r)}(x)) \,.
\]
This proves the theorem.
\end{proof}


\section{The building for $\PGL(V)$ over a local field}

We continue with the notations (1.1) in this section. The dilation (or 
homothety) class $\{L\}$ of a lattice $L$ is the set of all lattices 
$\lambda L$ where $\lambda \in F^*$ in the vector space $V$. Observe that 
$\{L\}$ is the set of all $\pi^i L$ for $i$ in the integers.

Let $\PGL(V)$ denote $\GL(V)/(\text{scalars})$ for a vector space $V$.

\begin{definition}
Let $V$ be a vector space over the local field $K$. The building $I(V)$ for 
the group $\PGL(V)$ over $K$ is the simplicial complex whose vertices are 
dilation classes $\{L\}$ of lattices $L$ in $V$, and whose simplexes 
$\{v_0, \ldots, v_n\}$ are sets of vertices where after reordering 
$v_i = \{L_i\}$ with $L_0 \supset L_1 \supset \cdots \supset L_n \supset \pi L_0$.
\end{definition}

Observe that the ordering of the vertices $v_0, \ldots, v_n$ such that 
representatives $L_i \in v_i$ can be chosen with 
$L_0 \supset L_1 \supset \cdots \supset L_n \supset \pi L_0$ is unique up to the 
action of the cyclic group on $n+1$ elements inside the symmetric group.

The simplicial complex $I(V)$ has dimension equal to $\dim V - 1$, and each 
simplex is contained in a top dimensional simplex (one whose dimension equals 
$\dim I(V)$). For $\dim V = 2$, the building $I(V)$ is one dimensional, a 
graph. In fact, $I(V)$ is simply connected and hence a tree. Each vertex 
$v = \{L\}$ is contained in $(m+1)$ $1$-simplexes $\{v,v'\}$ corresponding to 
the $m+1$ lattices $L'$ with $L \supset L' \supset \pi L$.

This building $I(V)$ is a special case of the buildings (immeubles) which 
have been associated to general semisimple groups over a local field by 
Bruhat and Tits. These simplicial complexes are contractible, and the 
vertices are in natural bijective correspondence with the cosets of 
$\PGL(V)/(\text{maximal compact subgroup } K)$. Here $K$ is the image of 
$\GL(L)$ in $\PGL(V)$ for a lattice $L$ of $V$. This description together 
with other features suggest that $I(V)$ is an analogue of the symmetric 
space for real Lie groups.

Now recall some generalities on geometric realizations as applied to $I(V)$. 
The geometric realization $I(V)_\mathbb{R}$ is the subset of 
$(t_v) \in \prod_{v \in I(V)} [0,1]$ such that $\{v : t_v \neq 0\}$ is a 
simplex of $I(V)$ and $\sum_v t_v = 1$. For each simplex 
$\sigma = \{v_0, \ldots, v_n\}$ of $I(V)$ its geometric realization 
$|\sigma| \subset I(V)_\mathbb{R}$ is the subset of $(t_v) \in I(V)_\mathbb{R}$ 
with $t_v = 0$ for $v \notin \sigma$. As a subset of 
$\prod_{v \in I(V)} [0,1]$, $|\sigma|$ is compact, we give $I(V)_\mathbb{R}$ 
the inductive (weak) topology where $M$ is closed in $I(V)_\mathbb{R}$ if and 
only if $M \cap |\sigma|$ is closed in $|\sigma|$ for each simplex $\sigma$ 
of $I(V)$. We will make use of the dense subset $I(V)_\mathbb{Q} \subset I(V)_\mathbb{R}$ 
consisting of $(t_v)$ with each $t_v \in \mathbb{Q}$. The set 
$I(V)_\mathbb{Z} \subset I(V)_\mathbb{Q}$ of $(t_v)$ with each $t_v \in \mathbb{Z}$ 
can be identified with the set of vertices of $I(V)$.

The function which links the considerations of this section with those of the 
previous section is $\theta\colon I(V)_\mathbb{R} \to N(V)$ from the geometric 
realization of the building to the dilation classes of norms on $V$ as 
follows: Let $\sigma = \{v_0, \ldots, v_n\}$ be a simplex and 
$t = (t_v) \in |\sigma|$ be a point. We can choose an ordering 
$v_0, \ldots, v_n$ such that $t_n > 0$ and lattices $L_i \in v_i$ with
\[
L_0 \supset L_1 \supset \cdots \supset L_n \supset \pi L_0 \,. \quad \text{Then } \beta(t) = \alpha
\]
where $\alpha = \sup(q^{-t_0}\alpha_{L_0}, \ldots, q^{-t_n}\alpha_{L_n})$.

\begin{theorem}
The function $\theta\colon I(V)_\mathbb{R} \to N(V)$ is a well-defined bijection 
which carries the vertices $I(V)_\mathbb{Z}$ onto the set of classes containing 
integral norms and $I(V)_\mathbb{Q}$ onto the set of classes containing rational 
norms.
\end{theorem}

\begin{proof}
The function $\theta$ is a bijection by (1.7), the structure theorem for 
norm. In that theorem we proved that each norm $\alpha$ has a unique 
representation $\alpha = \sup(r_0\alpha_{L_0}, \ldots, r_m\alpha_{L_m})$ where 
$q = r_0 \geqslant r_1 \geqslant \cdots \geqslant r_m \geqslant 1$ up to dilation. Let 
$t_0 = 1$, $t_i = \log_q(r_i/r_{i+1})$. Then $(t_0, \ldots, t_m)$ determines 
the unique point in $|I(\{L_0, \ldots, L_m\})|$ which maps to $\alpha$ under 
$\theta$. The remaining statements are clear from the formulas relating the 
$t_i$'s and $r_i$'s. This proves the theorem.
\end{proof}


\section{Metric on the building}

We continue with the notations of (1.1) in this section.

\begin{definition}
Let $V$ be a vector space over the local field $K$. For two norms 
$\alpha, \beta$ on $V$, we define the distance $\rho(\alpha,\beta)$ between 
$\alpha$ and $\beta$ by the following equivalent formulae:
\begin{align*}
\rho(\alpha,\beta) &= \log_q\Bigl(\sup_{x \in V, x \neq 0} \alpha(x)/\beta(x)\Bigr) + \log_q\Bigl(\sup_{x \in V, x \neq 0} \beta(x)/\alpha(x)\Bigr) \\
&= \log_q\sup_{x \in V, x \neq 0} \alpha(x)\beta(x)^{-1} = \log_q \inf_{x \in V, x \neq 0} \alpha(x)/\beta(x) \,.
\end{align*}
\end{definition}

\begin{remark}
From the first formula for $\rho(\alpha,\beta)$ we see that 
$\rho(\alpha,\beta) = \rho(\beta,\alpha)$, and from the second form it follows 
that $\rho(\alpha,\beta) \geqslant 0$, and $\rho(\alpha,\beta) = 0$ if and only if 
$\beta = t\alpha$ for some $t > 0$. Moreover, for $t_1 > 0$ and $t_2 > 0$ we 
have $\rho(t_1\alpha, t_2\beta) = \rho(\alpha,\beta)$, and thus 
$\rho(\{\alpha\}, \{\beta\})$ is well defined on dilation classes of norms on 
$V$. Finally, it is easy to check the triangle inequality
\[
\rho(\alpha,\gamma) \leqslant \rho(\alpha,\beta) + \rho(\beta,\gamma)
\]
using the relations of the form
\[
\sup_{x \in V, x \neq 0} \alpha(x)/\beta(x) = \sup_{x \in V, x \neq 0} \alpha(x)/\beta(x) \cdot \sup_{x \in V, x \neq 0} \beta(x)/\gamma(x) \,.
\]
Hence $\rho$ is a metric on the space $N(V)$ of dilation classes of norms on 
$V$.

Note from the definition of $\rho$ that if $t_1\alpha \leqslant \beta \leqslant t_2\alpha$ 
on $V$, then $\rho(\alpha,\beta) \leqslant \log_q(t_1/t_2)$.
\end{remark}

\begin{example}
Let $M$ be a lattice in $V$. Then the integral norm $\alpha_M$ is defined as 
in (1.4), and
\[
M = \{x \in V : \alpha_M(x) \leqslant 1\} \quad\text{and}\quad \mathfrak{p}M = \{x \in V : \alpha_M(x) < 1\} \,.
\]
Thus $M - \mathfrak{p}M$ is the unit $\alpha_M$-sphere of all $x$ with 
$\alpha_M(x) = 1$ in $V$. Moreover, since every $x \in V$, $x \neq 0$, is 
proportional to some $x'$ with $\alpha_M(x') = 1$, we deduce that
\[
\rho(\alpha_M, \alpha_N) = \log_q\Bigl(\sup_{x \in M-\mathfrak{p}M} \alpha_N(x)\Bigr) - \log_q\Bigl(\inf_{x \in M-\mathfrak{p}M} \alpha_N(x)\Bigr) \,.
\]
\end{example}

For two lattices $M$ and $N$ in $V$, there exists a natural number $r$ such 
that $M \supset N \supset \pi^r M$ and $N \supset M \supset \pi^r N$ so that 
$\rho(\alpha_M, \alpha_N) \leqslant 2r$. For a more precise calculation, we 
reorganize the hypothesis in the next proposition.

\begin{proposition}
Let $M \supset N \supset \pi^r M$ be lattices in $V$ where $r$ is minimal with 
this property. Then $r = \rho(\alpha_M, \alpha_N)$.
\end{proposition}

\begin{proof}
From $\alpha_M \leqslant \alpha_N \leqslant q^r\alpha_M$ we deduce that 
$\rho(\alpha_M, \alpha_N) \leqslant r$. Since $N \supsetneq \pi^{r-1}M$, we have 
$x \in (M - \mathfrak{p}M) \cap N$ and therefore $\alpha_N(x) \leqslant 1$. Thus 
$\log_q(\inf_{x \in M-\mathfrak{p}M} \alpha_N(x)) \leqslant 0$. Since 
$\pi^{r+1}N \subset M$ and $\pi^{r+1}N \not\subset \mathfrak{p}M$, there exists 
$x \in M - \mathfrak{p}M$ with $x \in \pi^{r+1}N$ and so $\alpha_N(x) \geqslant q^r$. 
Thus $\log_q(\sup_{x \in M-\mathfrak{p}M} \alpha_N(x)) \geqslant r$. By the example 
(3.3) it follows that $\rho(\alpha_N, \alpha_M) \geqslant r$ and hence 
$\rho(\alpha_M, \alpha_N) = r$.
\end{proof}

\begin{corollary}
A set of lattices $M_0, \ldots, M_r$, or equivalently, a set of integral 
norm classes $\alpha_{M_0}, \ldots, \alpha_{M_r}$ (for example 
$\alpha_i = \alpha_{M_i}$), determine a simplex in the building 
$I(V)_\mathbb{Z}$ or $N(V)$ if and only if $\rho(\alpha_i, \alpha_j) \leqslant 1$ 
for $i \neq j$.
\end{corollary}

\begin{proof}
Arrange, after reordering and dilation, the lattices as 
$M_0 \supset M_1 \supset \cdots \supset M_{r-1} \supset M_r \supset \pi^{-1} M_0$. Then 
$1 = \rho(\alpha_{M_i}, \alpha_{M_{i+1}}) \leqslant 1$ by (3.4), so 
$\rho(\alpha_i, \alpha_j) \leqslant 1$, and the classes $\{M_i\}$ form a simplex.

Conversely, $\rho(\alpha_i, \alpha_j) \leqslant 1$ for $i \neq j$ when the 
classes determine a simplex again by (3.4).
\end{proof}


\section{The mapping from the $p$-adic symmetric space to the building}

Now we return to the basic situation of the function field $F = \mathbb{F}_q(C)$ 
of the smooth curve $C/\mathbb{F}_q$, the local field $F_\infty$ at $\infty$ 
on the curve, and $\mathbb{C}_\infty$ the completion of the algebraic closure 
of $\bar{F}_\infty$.

The simple critical observation is the following: For 
$z = (z_1, \ldots, z_r) \in \mathbb{C}_\infty^r$ the function on $F_\infty^r$
\[
\alpha_z\colon (a_1, \ldots, a_r) \longmapsto |z_1 a_1 + \cdots + z_r a_r|
\]
is a norm on the $F_\infty$-vector space $F_\infty^r$ provided 
$|z_1 a_1 + \cdots + z_r a_r| = 0$ implies that $a = (a_1, \ldots, a_r) = 0$. 
This is the case exactly for
\[
z \in \mathbb{C}_\infty^r - \{\text{all $F_\infty$-rational hyperplanes}\} \,.
\]

Further, for $c \in \mathbb{C}_\infty^*$ and $z \in \mathbb{C}_\infty^r$ the relation 
$\alpha_{cz} = |c|\alpha_z$, which is dilation of norms, holds and this 
leads to the following definition.

\begin{definition}
The building map $\lambda$ defined on the $p$-adic symmetric space 
$\Omega^r(\mathbb{C}_\infty)$ to the building $N(F_\infty^r) = I(F_\infty^r)_\mathbb{R}$ 
of the group $\PGL(r, F_\infty)$ is given by
\[
\lambda(z) = \text{dilation class of } \alpha_z \quad\text{for } z = (z_i) \in \Omega^r \,.
\]
For a representative $r$-tuple $(z_1, \ldots, z_r) = z$ of $z \in \Omega^r$, 
we represent $\lambda(z)$ as the norm $\alpha_z$, i.e.
\[
\lambda(z)(a_1, \ldots, a_r) = |z_1 a_1 + \cdots + z_r a_r| \,.
\]
Note, since $|\mathbb{C}_\infty^*| = q^\mathbb{Q}$, in fact 
$\lambda\colon \Omega^r(\mathbb{C}_\infty) \to I(F_\infty^r)_\mathbb{Q}$.
\end{definition}

\begin{proposition}
The building map $\lambda\colon \Omega^r(\mathbb{C}_\infty) \to I(F_\infty^r)_\mathbb{Q}$ 
is $\GL(r, F_\infty)$-equivariant for right actions. In particular, it is 
also $\GL(r,F)$-equivariant.
\end{proposition}

\begin{proof}
For $s \in \GL(r, F_\infty)$ we view the matrix as acting on the left and 
$F_\infty^r$ the action on the right. The norm $\lambda(z)$ acted on by $s$ 
on the right is $\lambda(z)s\colon a \mapsto \lambda(z)(sa)$ for $a \in F_\infty^r$, 
it is $|\langle z|sa\rangle| = |\langle z^t s|a\rangle| = \lambda(z^t s)(a)$. Thus 
$\lambda(z)s = \lambda(z^t s)$ which proves the proposition.
\end{proof}

For a subgroup $\Gamma \subset \GL(r, F_\infty)$ we have a quotient building 
mapping
\[
\lambda_\Gamma\colon \Omega^r(\mathbb{C}_\infty)/\Gamma \longrightarrow I(F_\infty^r)_\mathbb{Q}/\Gamma \subset I(F_\infty^r)_\mathbb{R}/\Gamma \,.
\]
In the case $\Gamma$ is a certain subgroup of $\GL(r,A)$ this is a mapping 
of the corresponding moduli space associated with $\Gamma$ to a quotient of 
the building by the discrete group $\Gamma$.

\subsection*{4.3 Remarks}

The building map $\lambda$ is useful for several purposes.

First, the sets $\lambda^{-1}$ (ball around a vertex or simplex) can be used 
to describe the rigid analytic structure on the $p$-adic symmetric space. In 
the special case $r = 2$ so that $I(F_\infty^2)$ is a tree $\mathcal{T}$, we 
will describe a topological model for $\Omega^2(\mathbb{C}_\infty) = \mathbb{C}_\infty - F_\infty$ 
and the map $\lambda\colon \mathbb{C}_\infty - F_\infty \to \mathcal{T}$. This is used 
to calculate the cohomology of $\Omega^2(\mathbb{C}_\infty)$ and the map 
$\lambda_\Gamma\colon \Omega^2(\mathbb{C}_\infty)/\Gamma \to \mathcal{T}/\Gamma$.

There are some coverings of $\mathcal{T}/\Gamma$ which induce back to 
admissible covers of $\Omega^2(\mathbb{C}_\infty)/\Gamma$, and these give rise to 
a Leray spectral sequence of the map 
$\lambda_\Gamma\colon \Omega^2(\mathbb{C}_\infty)/\Gamma \to \mathcal{T}/\Gamma$. In 
this case the spectral sequence which could be thought of as the defining 
the rigid analytic structure, is just the cohomology $H^*(\mathcal{T}/\Gamma)$ 
of the group $\Gamma$.

In order to illustrate further what is involved in defining the rigid 
analytic structure, we must calculate $\rho(\lambda(z), \alpha_\Lambda)$ where 
$\Lambda$ is the standard lattice $\mathcal{O}_\infty^r \subset F_\infty^r$. 
Since any other lattice is $\GL(r, F_\infty)$-equivalent to $\Lambda$ and 
since both $\lambda$ and $\rho$ are $\GL(r, F_\infty)$-invariant, this 
calculation leads to $\rho(\lambda(z), \alpha_L)$ for any vertex 
$\{L\}$.

By (3.3) we have
\[
\rho(\lambda(z), \alpha_\Lambda) = \log_q\Bigl(\sup_{(a_1,\ldots,a_r) \in S(\Lambda)} |a_1 z_1 + \cdots + a_r z_r|\Bigr) - \log_q\Bigl(\inf_{(a_1,\ldots,a_r) \in S(\Lambda)} |a_1 z_1 + \cdots + a_r z_r|\Bigr)
\]
where $S(\Lambda) = \Lambda - \mathfrak{p}\Lambda$, the set of all 
$(a_1, \ldots, a_r) \in F_\infty^r$ with all $|a_i| \leqslant 1$ and at least 
one $|a_i| = 1$.

Thus we see that $\rho(\lambda(z), \alpha_\Lambda) < s$ if and only if for all 
$(a_1, \ldots, a_r) \in F_\infty^r$ with all $|a_i| \leqslant 1$ and at least 
one $|a_i| = 1$, we have
\[
q^{-s} < |a_1 z_1 + \cdots + a_r z_r| < q^s \,.
\]
In the case $S(\Lambda) = \Lambda - \mathfrak{p}\Lambda$, the set of all 
$(a_1, \ldots, a_r) \in F_\infty^r$ with all ratios $|z_i|/|z_j| \notin q^\mathbb{Z}$ 
for $i \neq j$, we have
\[
\rho(\lambda(z), \alpha_\Lambda) = \log_q(|z_1|/|z_r|)
\]
by an easy straightforward calculation.

When some of the ratios $|z_i|/|z_j| \in q^\mathbb{Z}$, the calculation of 
$\rho(\lambda(z), \alpha_\Lambda)$ is more complicated, and in fact, it is 
at the basis for the structure of $\Omega^r(\mathbb{C}_\infty)$ as a rigid 
analytic space.


\section{Filtration of the $1$-dimensional $p$-adic symmetric space}

In this section we study the case $r = 2$ in detail where
\[
\Omega^2(\mathbb{C}_\infty) = \Omega^1(\mathbb{C}_\infty) = \mathbb{C}_\infty - F_\infty \,.
\]
We use the notation $u \in \Omega$ for $u \in \mathbb{C}_\infty$ and 
$\mathcal{T} = I(F_\infty^2)$ is the building which in this case is a tree (a 
contractible $1$-dimensional simplicial complex).

The building map $\lambda\colon \Omega \to \mathcal{T}$ is given by
\[
\lambda(u)(a,b) = |a + bu|
\]
and we wish to study
\[
\rho(\lambda(u), \alpha_\Lambda) = \log_q\Bigl(\sup_{(a,b) \in S(\Lambda)} |a + bu|\Bigr) - \log_q\Bigl(\inf_{(a,b) \in S(\Lambda)} |a + bu|\Bigr)
\]
relative to elements of $u$ in terms of congruence properties of $F_\infty$. 
This is formulated by using the following notion.

\begin{definition}
For $u \in \mathbb{C}_\infty$, we define the irrational absolute value
\[
|u|_{\text{irr}} = \inf_{a \in F_\infty} |u - a|
\]
to be the distance from $u$ to $F_\infty \subset \mathbb{C}_\infty$.
\end{definition}

Observe that this is just the distance from $u$ to $F_\infty \subset \mathbb{C}_\infty$. 
The following properties are easily deduced from the definition.

\begin{proposition}
The irrational norm satisfies the following:
\begin{enumerate}[label=(\arabic*)]
\item For $u \in \mathbb{C}_\infty$, $|u|_{\text{irr}} = 0$ if and only if 
$u \in F_\infty$.
\item For $u \in \mathbb{C}_\infty$ and $c \in F_\infty$, we have 
$|cu|_{\text{irr}} = |c| \cdot |u|_{\text{irr}}$.
\item For $|u| \leqslant q$ and $u \in c\mathcal{O}_\infty$ it follows that 
$|u|_{\text{irr}} = |u|$.
\item For $|u| = 1$ with residue class $\bar{u} \in \bar{\mathbb{F}}_q$ the 
irrational norm $|u|_{\text{irr}} = |u| = 1$ if and only if 
$\bar{u} \notin \mathbb{F}_q = \mathbb{F}_q$.
\end{enumerate}
\end{proposition}

From the previous section we see for $|u| \leqslant 1$ that
\[
\rho(\lambda(u), \alpha_\Lambda) = \log_q(\max(|u|, 1/|u|))
\]
which for $|u| < 1$ becomes simply $\rho(\lambda(u), \alpha_\Lambda) = -\log_q|u|$.

\begin{proposition}
The distance from $\lambda(u)$ to the standard vertex $\alpha_\Lambda$ in the 
tree $\mathcal{T}$ is given by
\[
\rho(\lambda(u), \alpha_\Lambda) =
\begin{cases}
-\log_q|u|_{\text{irr}} & \text{for } |u| \leqslant 1 \,,\\[2mm]
-\log_q|1/u|_{\text{irr}} & \text{for } |u| \geqslant 1 \,.
\end{cases}
\]
\end{proposition}

\begin{proof}
For the case $|u| < 1$, we have $\sup_{(a,b) \in S(\Lambda)} |a + bu| = 1$ 
since $|a + bu| \leqslant \max(|a|, |b| \cdot |u|) \leqslant 1$ and 
$|1 + 0 \cdot u| = 1$. Further,
\[
\inf_{(a,b) \in S(\Lambda)} |a + bu| = \inf_{|a| \leqslant 1, |b| = 1} |a + bu| = \inf_{|c| < 1} |c + u| = |u|_{\text{irr}}
\]
since $|a + bu| \leqslant \max(|a|, |b| \cdot |u|) \leqslant 1$ for $|u| \leqslant 1$.
Hence, we have
\[
\rho(\lambda(u), \alpha_\Lambda) = \log_q(1/|u|_{\text{irr}}) = -\log_q|u|_{\text{irr}} \,.
\]
This proves the proposition.

For the case $|u| > 1$, we observe that $\rho(\lambda(u), \alpha_\Lambda) = \rho(\lambda(1/u), \alpha_\Lambda) = -\log_q|1/u|_{\text{irr}}$ by the first part.
\end{proof}

Now we are in a position to describe the inverse image in $\Omega$ of closed 
balls around the standard lattice vertex $\alpha_\Lambda$. First, we choose some 
notation. Let $r\colon \mathbb{P}_1(\mathcal{O}_\infty) \to \mathbb{P}_1(\bar{\mathbb{F}}_q)$ 
be the residue class reduction and $s\colon \mathbb{P}_1(\bar{\mathbb{F}}_q) \to \mathbb{P}_1(\mathcal{O}_\infty)$ 
a cross section such that $r \circ s = \text{id}$ on $\mathbb{P}_1(\bar{\mathbb{F}}_q)$. 
We choose $s$ so that $s(0) = 0$, $s(\infty) = \infty$.

\begin{enumerate}[label=(\arabic*)]
\item Clearly $u \in \lambda^{-1}(\alpha_\Lambda)$ from (4.2) if and only if 
$|u|_{\text{irr}} = 1$. This gives the following relations for 
$\lambda^{-1}(\alpha_\Lambda) \subset \Omega$:
\begin{align*}
\lambda^{-1}(\alpha_\Lambda) &= \{u \in \mathbb{C}_\infty : |u|_{\text{irr}} = 1\} \\
&= \mathbb{P}_1(\mathbb{C}_\infty) - \bigcup_{\xi \in \mathbb{P}_1(\mathbb{F}_q)} B(s(\xi), 1) \\
&= \Omega^2(\mathbb{C}_\infty) - \bigcup_{\eta \in \mathbb{P}_1(\mathbb{F}_q)} B(s(\eta), 1)
\end{align*}
where $B(v,\epsilon) = \{u \in \mathbb{C}_\infty : |u - v| < \epsilon\}$ is the 
open ball of radius $\epsilon$ around $v$ in $\mathbb{C}_\infty$ for 
$v \neq \infty$ and $B(\infty, \epsilon) = \{0\} \cup \{u \in \mathbb{C}_\infty : |u| > 1/\epsilon\} \subset \mathbb{P}_1(\mathbb{C}_\infty)$.

For the closed ball $B^*(\star, c)$ with $0 \leqslant c < 1$ observe that 
$\lambda^{-1}(B^*(\star, c)) = \Omega^2(\mathbb{C}_\infty) - \bigcup_{\eta \in \mathbb{P}_1(\mathbb{F}_q)} B(s(\eta), q^{-c})$. In this case $q^{-c} < 1$.

\item In this case $\lambda^{-1}(B^*(\star, c))$ is $\Omega^2(\mathbb{C}_\infty)$ minus 
$(q+1)$ balls, but this time of slightly smaller radius. Now we see the 
relation between these balls and the edges of the tree coming out from the 
vertex $v_\Lambda$. All the points of $B(s(\eta), 1) - B(s(\eta), q^{-c})$ 
project to the edge corresponding to $\eta \in \mathbb{P}_1(\mathbb{F}_q)$.

\item As the radius $c$ increases in the tree, we come to new vertices each 
with $q$ new edges coming out as $c$ approaches $1$ from $0 < c < 1$. The 
inverse image of the closed ball $B^*(v_\Lambda, c)$ as $c$ increases is 
$\Omega^2(\mathbb{C}_\infty)$ minus $(q+1)$ balls of radius $q^{-c}$ centered 
at the points $s(\eta)$ for $\eta \in \mathbb{P}_1(\mathbb{F}_q)$.

As $c$ increases past integer values, the inverse image changes by removing 
more balls. This describes the rigid analytic structure on 
$\Omega^2(\mathbb{C}_\infty)$ in terms of the tree $\mathcal{T}$.
\end{enumerate}
