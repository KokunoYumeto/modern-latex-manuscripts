{\Large \textbf{UNIRATIONAL VARIETIES THAT ARE NOT RATIONAL}}
\end{center}

\begin{flushright}
\textit{[after M.~Artin and D.~Mumford]}\\[4pt]
by Pierre Deligne
\end{flushright}

\begin{flushright}
Bourbaki Seminar\\
24th year, 1971/72, no.~402\\

\end{flushright}

\bigskip

This report contains a description of the variety constructed by Artin and
Mumford, together with the proof that it is unirational but not rational.
It also states the theorems proved by Clemens and Griffiths~\cite{4}, and by
Manin and Iskovskih~\cite{6}, which yield other examples. I attended talks by
Artin and Mumford on the same subject, and I have drawn heavily on them.

Throughout, the expression ``algebraic variety'' means ``a separated scheme of
finite type over $\mathbf{C}$, reduced and irreducible.'' We write $k(X)$ for the
field of rational functions on an algebraic variety $X$.

\section{Rationality and unirationality}

Let $K$ be a finitely generated extension of $\mathbf{C}$. Then there exists an
algebraic variety $X$ such that $K$ is isomorphic to $k(X)$. One calls $X$ a
\emph{model} of $K$. By~\cite{5}, $K$ always admits a smooth projective model.

Let $K$ and $L$ be finitely generated extensions of $\mathbf{C}$, with models
$X$ and $Y$. Homomorphisms $a^*\colon K\to L$ correspond bijectively to
dominant rational maps $a\colon Y\dashrightarrow X$, by the rule
$a^*(f)=f\circ a$. By~\cite{5}, if $X$ and $Y$ are smooth and projective, every
rational map $a\colon Y\dashrightarrow X$ admits a factorization
\begin{equation}\label{eq:1.1}
a=b\circ c_n^{-1}\circ\cdots\circ c_1^{-1}
\end{equation}
where $b$ and the $c_i$ are everywhere defined morphisms and where $Y_{i+1}$ is
obtained from $Y_i$ by blowing up a smooth subvariety $Z_i$ of codimension
$\geqslant 2$.

If $a^*\colon K\to L$ is an isomorphism, one can apply the preceding result to
$(a^{*-1},Y_{n+1},X)$. Iterating, one obtains a commutative diagram of scheme
morphisms
\begin{equation}\label{eq:1.2}
\xymatrix{
& Y_n' \ar[r]^{y_n} \ar[d]^{a_n} & \cdots \ar[r]^{y_2} & Y_1' \ar[r]^{y_1} \ar[d]^{a_1} & Y \ar[d]^a \\
X & X_n' \ar[l]_{x_n} & \cdots \ar[l]_{x_2} & X_1' \ar[l]_{x_1} & X \ar[l]
}
\end{equation}
In this diagram, the horizontal arrows $x_i,y_i$ are composites of blow-ups with
smooth centers, as in~\eqref{eq:1.1}; the $a_i$ and $b_i$ are birational.

An invariant of a smooth projective variety $X$ is said to be
\emph{birational} if it depends only on $k(X)$. One often verifies birational
invariance using~\eqref{eq:1.2} (for a typical example, see
Proposition~\ref{prop:2.1}). Invariants of a field $K$ are usually defined as the
value of some birational invariant of any smooth projective model of $K$.

\begin{definition}\label{def:1.3}
Let $X$ be an algebraic variety of dimension $n$. One says that $X$ is
\emph{unirational} if the following equivalent conditions hold:
\begin{enumerate}
\item[(i)] $k(X)$ is a subfield of a purely transcendental extension
$\mathbf{C}(T_1,\ldots,T_r)$ of $\mathbf{C}$;
\item[(ii)] a finite extension of $k(X)$ is isomorphic to
$\mathbf{C}(T_1,\ldots,T_n)$;
\item[(iii)] if $X$ is smooth and projective, there exists a diagram
\eqref{eq:1.1} with $Y=\mathbf{P}^r(\mathbf{C})$;
\item[(iv)] if $X$ is smooth and projective, there exists a diagram
\eqref{eq:1.1} with $Y=\mathbf{P}^n(\mathbf{C})$.
\end{enumerate}
One may suppose $X$ smooth and projective, and one has
\[
\text{(ii)}\Rightarrow\text{(i)}\Rightarrow\text{(iii)}\Rightarrow\text{(iv)}\Rightarrow\text{(ii)}.
\]
\end{definition}

\begin{definition}\label{def:1.4}
One says that $X$ is \emph{rational} if, equivalently,
\begin{enumerate}
\item[(i)] $k(X)$ is a purely transcendental extension of $\mathbf{C}$;
\item[(ii)] if $X$ is smooth and projective, there exists a diagram
\eqref{eq:1.2} with $Y=\mathbf{P}^n(\mathbf{C})$.
\end{enumerate}
\end{definition}

The most obvious birational invariants do not distinguish unirational varieties
from rational varieties.

\begin{proposition}\label{prop:1.5}
Let $X$ be a smooth projective unirational algebraic variety of dimension
$n>0$.
\begin{enumerate}
\item[(i)] The plurigenera
$P_k=\dim H^0(X,(\Omega_X^n)^{\otimes k})$ $(k>0)$ vanish. More generally,
$H^0(X,(\Omega_X^1)^{\otimes k})=0$ for $k>0$ (see~\cite[Remark, p.~102]{9}).
\item[(ii)] $X$ is simply connected~\cite{10}.
\end{enumerate}
\end{proposition}

For $n=1$ or $2$, property~(i) already characterizes rational varieties. Thus a
unirational variety of dimension at most $2$ is rational, and a subextension of
$\mathbf{C}(T_1,\ldots,T_k)$ of transcendence degree at most $2$ is automatically
purely transcendental. More precisely:

\smallskip
\noindent a) If $n=1$ and $H^0(X,\Omega_X^1)=0$, then
$X\simeq \mathbf{P}^1(\mathbf{C})$. Indeed, by Riemann--Roch, for every
$x\in X$, the line bundle $\mathcal O(x)$ defines an embedding of $X$ into
$\mathbf{P}^1(\mathbf{C})$.

\smallskip
\noindent b) For $n=2$, the Castelnuovo--Enriques criterion says that $X$ is
rational if $P_a=P_2=0$, i.e.
if $H^0(X,\Omega_X^2)=H^0(X,(\Omega_X^2)^{\otimes 2})=0$; see~\cite{9}.

For $n>2$, it has been very difficult to define or compute birational invariants
capable of distinguishing rational from unirational varieties.

\section{A few birational invariants}

\subsection*{A) Artin and Mumford}

\begin{proposition}\label{prop:2.1}
The torsion subgroup $H^3(X,\mathbf{Z})_{\mathrm{tors}}$ of $H^3(X,\mathbf{Z})$
is a birational invariant of the smooth projective variety $X$.
\end{proposition}

If $X_2$ is obtained from a smooth variety $X_1$ by blowing up a smooth
subvariety $Z\subset X_1$ of codimension $r\geqslant 2$, then
(SGA~5, VII)
\[
H^n(X_2,\mathbf{Z})\simeq H^n(X_1,\mathbf{Z})\oplus\bigoplus_{i=1}^{r-1}H^{n-2i}(Z,\mathbf{Z}).
\]
In particular,
$H^3(X_2,\mathbf{Z})\simeq H^3(X_1,\mathbf{Z})\oplus H^1(Z,\mathbf{Z})$.
Since $H^1(Z,\mathbf{Z})$ is torsion-free, one gets
\[
H^3(X_1,\mathbf{Z})_{\mathrm{tors}}\xrightarrow{\sim}H^3(X_2,\mathbf{Z})_{\mathrm{tors}}.
\]

Assume $k(X)\simeq k(Y)$, and choose a diagram~\eqref{eq:1.2}. One obtains
\[
\xymatrix{
H^3(X,\mathbf Z)_{\mathrm{tors}} \ar[r]^{\sim}
  & H^3(X_1',\mathbf Z)_{\mathrm{tors}}
  & H^3(X_n',\mathbf Z)_{\mathrm{tors}} \ar[l] \ar[r]^{\sim}
  & H^3(X,\mathbf Z)_{\mathrm{tors}} \\
H^3(Y,\mathbf Z)_{\mathrm{tors}} \ar[r]^{\sim} \ar[u]^{a_1^*}
  & H^3(Y_1',\mathbf Z)_{\mathrm{tors}} \ar[u]_{b_1^*}
  & H^3(Y_n',\mathbf Z)_{\mathrm{tors}} \ar[l] \ar[r]^{\sim} \ar[u]_{b_n^*}
  & H^3(Y,\mathbf Z)_{\mathrm{tors}} \ar[u]_{a_2^*}
}
\]
Since $a_2^*\circ b_n^*$ is bijective, $b_n^*$ is injective. Since
$b_n^*\circ a_n^*$ is bijective and $b_n^*$ is injective, $a_n^*$ is bijective, and
the assertion follows.

\begin{construction}\label{cons:2.2}
We shall construct a smooth projective unirational variety $X$ of dimension $3$
with $H^3(X,\mathbf Z)_{\mathrm{tors}}\neq 0$ (in fact,
$H^3(X,\mathbf Z)\simeq \mathbf Z/(2)$). Multiplying this variety by
$\mathbf P^r(\mathbf C)$ yields an analogous example in every dimension
$\geqslant 3$.
\end{construction}

\subsection*{B) Clemens and Griffiths}

Let $X$ be a smooth projective variety of dimension $3$ such that
$H^0(X,\Omega_X^3)=0$. Then the Hodge decomposition of
$H^3(X,\mathbf C)$ reduces to
\[
H^3(X,\mathbf C)=H^{2,1}\oplus H^{1,2},
\]
and, by Weil~\cite{11}, the complex torus
\[
J(X)=H^3(X,\mathbf Z)\backslash H^3(X,\mathbf C)/H^{2,1}
\]
is an abelian variety: the \emph{intermediate Jacobian} of $X$. Let
\[
\Phi:H^3(X,\mathbf Z)\otimes H^3(X,\mathbf Z)\to H^6(X,\mathbf Z)\simeq \mathbf Z
\]
be the cup product. By Poincar\'e duality, the induced alternating form on
$H^3(X,\mathbf Z)/\mathrm{torsion}$ has discriminant one. If
$H^1(X,\mathbf Z)=0$, it follows from Weil~\cite{11} that this defines a principal
polarization on $J(X)$.

\begin{proposition}[\cite[3.26]{4}]\label{prop:2.3}
If $X$ is rational, the principally polarized abelian variety $J(X)$ is a product
of Jacobians.
\end{proposition}

In the sketch of proof below, the sign $\simeq$ denotes an isomorphism of
principally polarized abelian varieties.

\smallskip
\noindent a) $J(\mathbf P^3(\mathbf C))=0$.

\smallskip
\noindent b) If $Y''$ is obtained from a smooth projective variety $Y'$ by blowing
up a smooth curve $Z$ with Jacobian $J(Z)$ (resp. a point), one has
\cite[3.11]{4}
\[
J(Y'')\simeq J(Y')\times J(Z)
\]
(resp. $J(Y'')\simeq J(Y')$).

\smallskip
\noindent c) If a morphism $b\colon Y'\to X$ is birational, then $J(X)$ is a
direct factor of $J(Y')$: there exists a principally polarized abelian variety $B$
from the main series such that
\[
J(Y')\simeq J(X)\times B.
\]

\smallskip
\noindent d) Let $(A_i)_{1\leqslant i\leqslant n}$ be nonzero principally polarized
abelian varieties. Assume that none of the $A_i$ admits a decomposition
$A_i\simeq A_i'\times A_i''$ (for example, assume $A_i$ is a Jacobian). Then any
decomposition of $A=\prod A_i$ is of the form
\[
A\simeq \left(\prod_{i\in I}A_i\right)\times \left(\prod_{i\notin I}A_i\right)
\quad\text{for } I\subset [1,n]
\]
(\cite[3.23]{4}).

If $X$ is rational, there exists a diagram~\eqref{eq:1.2} with
$Y=\mathbf P^3(\mathbf C)$ and $b$ birational. By a) and b),
$J(Y'_{n+1})$ is a product of Jacobians. By c) and d), so is $J(X)$.

Now take for $X$ a smooth cubic hypersurface in $\mathbf P^4(\mathbf C)$. One has
\[
H^1(X,\mathbf Z)=H^3(X,\mathbf Z)_{\mathrm{tors}}=H^0(X,\Omega_X^1)=0,
\]
\[
\dim J(X)=\dim H^1(X,\Omega_X^2)=5,
\]
and $X$ is unirational by~\cite{7}. Let $S$ be the surface parametrizing the
lines contained in $X$, and let $\alpha\colon S\to\Alb(S)$ be the Albanese map.
One of the essential results of~\cite{4} is the following.

\begin{theorem}[\cite[11.19 and 13.4]{4}]\label{thm:2.4}
There exists an isomorphism $J(X)\xrightarrow{\sim}\Alb(S)$ carrying the
divisor $\Theta$ of $J(X)$ to the image of $S\times S$ under the map
$(x,y)\mapsto \alpha(x)-\alpha(y)$.
\end{theorem}

Clemens and Griffiths then use results from~\cite{1} to prove that
$(\Alb(S),\theta(S\times S))$ cannot be the polarized Jacobian of a curve.
Another proof of the irrationality of $X$ is suggested in the appendix to~\cite{4};
it amounts to proving the following.

\begin{theorem}\label{thm:2.5}
The singular locus of the divisor $\Theta$ of $J(X)$ has dimension $0$, hence
codimension $>4$. In fact it is reduced to the image under $\theta$ of the diagonal
of $S\times S$.
\end{theorem}

This is known never to happen for the theta divisor of a Jacobian.

\subsection*{C) Manin and Iskovskih}

The group of birational automorphisms of a variety $X$ (equivalently, the group of
$\mathbf C$-automorphisms of $k(X)$) is a birational invariant of $X$. It is this
invariant that Manin and Iskovskih use to show that a smooth quartic hypersurface
in $\mathbf P^4(\mathbf C)$ is never rational.

\begin{theorem}[\cite{6}]\label{thm:2.6}
Let $X$ and $Y$ be smooth quartic hypersurfaces in $\mathbf P^4(\mathbf C)$. Every
birational isomorphism $a\colon X\dashrightarrow Y$ is automatically biregular.
\end{theorem}

Since some quartic hypersurfaces are unirational~\cite{7}, this theorem provides a
new example of unirational but nonrational varieties.

\section{The Artin--Mumford example}

Consider in $\mathbf P^2(\mathbf C)$ a configuration consisting of

\smallskip
\noindent a) two smooth cubic curves $C_1$ and $C_2$, with equations
$f_1=0$ and $f_2=0$, meeting transversally;

\smallskip
\noindent b) a smooth conic $Q$, with equation $q=0$, meeting each $C_i$ at
three distinct tangency points.

\medskip
We shall construct
\begin{enumerate}
\item[a)] a rank-$3$ vector bundle $V$ on $\mathbf P^2(\mathbf C)$;
\item[b)] a quadratic form $\Phi\colon V\to \mathcal O$ ($\mathcal O$ an invertible
sheaf) with discriminant $f_1f_2$ (if $\Delta$, a section of
$\mathcal L^{\otimes 3}\otimes(\Lambda^3V)^{\otimes(-2)}$, denotes the discriminant, then
the subscheme of $\mathbf P^2(\mathbf C)$ cut out by $\Delta=0$ is $C_1\cup C_2$).
\end{enumerate}

Let $X\subset \mathbf P(V^*)$ be the conic bundle over $\mathbf P^2(\mathbf C)$,
degenerating along $C_1\cup C_2$, defined by the homogeneous equation
$\Phi=0$. The following conditions will hold.

\medskip
\noindent c) $\Phi$ has rank $\geqslant 2$ at every point: for $s\in C_1\cup C_2$,
the fiber $X_s=p^{-1}(s)$ is the union of two concurrent lines; the only singular
points of $X$ are the singular points of the fibers $X_s$ for $s\in C_1\cap C_2$.

\medskip
\noindent d) Let $S$ be the double cover of $\mathbf P^2(\mathbf C)$ branched along
$Q$, and write
$X_S=X\times_{\mathbf P^2(\mathbf C)}S$. Then $X_S/S$ admits a section.

\medskip
\noindent e) For $s\in (C_1\cup C_2)-Q$, the two points $\varphi(t)$ with
$\varphi(t)=s$ lie on the two irreducible components of $X_s$.

\medskip
Locally on $C_1\cup C_2$ (for the usual topology), the restriction of the bundle $X$
to $C_1\cup C_2$ is obtained by gluing two $\mathbf P^1$-bundles along a section.
However, when one goes around a loop in $C_1\cup C_2$, these two bundles may be
interchanged: there is an unramified double covering $(C_1\cup C_2)^*$ of
$C_1\cup C_2$ whose two local sheets correspond locally to the two
$\mathbf P^1$-bundles whose union is $X|_{C_1\cup C_2}$. Let
$\pi_i\colon C_i^*\to C_i$ be the double covering of $C_i$ induced by
$(C_1\cup C_2)^*$.

\begin{lemma}\label{lem:3.1}
$C_i^*$ is irreducible.
\end{lemma}

By e), $C_i^*$ is the normalization of the double cover of $C_i$ induced by $S$.
The lemma follows from the fact that the section $q$ of $\mathcal O(2)|_{C_i}$ is
not the square of a section of $\mathcal O(1)|_{C_i}$; indeed, if $q$ were a square,
the three intersection points of $Q$ with $C_i$ would be collinear.

\medskip
The double covering $C_i^*\to C_i$ corresponds to an order-$2$ character of the
fundamental group of $C_i$ and to a local system $A_i$, locally isomorphic to
$\mathbf Z$, defined by the exact sequence of sheaves on $C_i$
\begin{equation}\label{eq:3.2}
0\longrightarrow A_i\longrightarrow \pi_{i*}\mathbf Z\xrightarrow{\;\Tr\;}\mathbf Z\longrightarrow 0,
\end{equation}
where $\Tr$ is ``the sum of the values on the two sheets of the covering.''

One has
\begin{equation}\label{eq:3.2coho}
H^0(C_i,A_i)=0,\qquad H^1(C_i,A_i)\simeq \mathbf Z/2,\qquad H^2(C_i,A_i)=\mathbf Z/2.
\end{equation}

\begin{lemma}\label{lem:3.3}
$X$ is unirational.
\end{lemma}

The surface $S$ is a quadric in $\mathbf P^3(\mathbf C)$, hence is rational. Since
$X_S/S$ admits a section, there is a birational isomorphism
$X_S\dashrightarrow S\times \mathbf P^1(\mathbf C)$, and $X$ is an image of the rational
variety $X_S$.

\medskip
The nine singular points of $X$ are ordinary quadratic points. One resolves each
of them as follows (cf.~\cite{3}).

\smallskip
\noindent a) Blow it up. This resolves $X$ and replaces the singular point by a
quadric $P$, isomorphic to $\mathbf P^1(\mathbf C)\times \mathbf P^1(\mathbf C)$.

\smallskip
\noindent b) Choose a projection $P\to \mathbf P^1(\mathbf C)$, and contract inside
the blow-up $\widetilde X$ of $X$ the fibers of this projection. This is possible
because $\widetilde X$ is smooth, its fibers are isomorphic to
$\mathbf P^1(\mathbf C)$, and the normal bundle to $P$ in $\widetilde X$ induces on each
fiber a line bundle $\mathcal O(-1)$~\cite{8}.

\medskip
Let $\bar X$ be the resulting variety. There is a morphism from $\bar X$ to $X$, and
$\bar X$ is obtained from $X$ by replacing each singular point by a projective line.

The fibers of the projection $p\colon \bar X\to \mathbf P^2(\mathbf C)$ are of the
following three kinds:
\begin{itemize}
\item[$s\notin C_1\cup C_2$:] a conic;
\item[$s\in C_1\cup C_2$, $s\notin C_1\cap C_2$:] two concurrent lines;
\item[$s\in C_1\cap C_2$:] three lines arranged as indicated.
\end{itemize}

When one specializes from the generic point $\xi$ of $\mathbf P^2(\mathbf C)$ to the
generic point $\eta_i$ of $C_i$ ($i=1,2$), and then to a point
$s\in C_1\cap C_2$, the irreducible components specialize as indicated in the
following diagram:
\begin{equation}\label{eq:3.4}
\xymatrix@R=8pt@C=12pt{
& \eta_1 \ar@{-}[rd] & & \xi \ar@{-}[ld] \ar@{-}[rd] & & \eta_2 \ar@{-}[ld] \\
(3.4) & & \times & & \times & \\
& \eta_2 \ar@{-}[ru] & & \eta_1 \ar@{-}[lu] \ar@{-}[ru] & & \eta_2 \ar@{-}[lu]
}
\end{equation}
(the crosses represent irreducible components; all multiplicities are one). The
rather tedious verification of this local result will be omitted. One deduces that
$R^0p_*\mathbf Z=\mathbf Z$, $R^1p_*\mathbf Z=0$, and the following exact
sequence of sheaves.

\begin{lemma}\label{lem:3.5}
One has
\[
0\longrightarrow A_1\oplus A_2\longrightarrow R^2p_*\mathbf Z\xrightarrow{\;\Tr\;}\mathbf Z\longrightarrow 0.
\]
\end{lemma}

One can now use the Leray spectral sequence of $p$ to compute
$H^*(\bar X,\mathbf Z)$. The essential fact is the following, which implies the
irrationality of $X$.

\begin{lemma}\label{lem:3.6}
One has $H^3(\bar X,\mathbf Z)=\mathbf Z/(2)$.
\end{lemma}

Let us compute the $E_2$-terms of the Leray spectral sequence of $p$. Since
$R^0p_*\mathbf Z=\mathbf Z$, we have
$E_2^{p,0}=\mathbf Z,0,\mathbf Z,0,\mathbf Z$ for $p=0$ to $4$; and since
$R^1p_*\mathbf Z=0$, we have $E_2^{p,1}=0$.

To compute $E_2^{p,2}=H^p(R^2p_*\mathbf Z)$, one uses the long exact sequence
attached to the short exact sequence of sheaves in Lemma~\ref{lem:3.5}. One finds
\[
0\longrightarrow E_2^{0,2}\xrightarrow{\;\alpha\;}\mathbf Z\longrightarrow E_2^{1,2}\longrightarrow 0,
\]
\[
0\longrightarrow \mathbf Z/2\oplus \mathbf Z/2\longrightarrow E_2^{2,2}\longrightarrow 0,
\]
\[
E_2^{3,2}=0 \qquad\text{and}\qquad E_2^{4,2}=\mathbf Z.
\]
One checks easily that $\alpha(E_2^{0,2})=2\mathbf Z$ (this is not in fact necessary
for the proof that $H^3(\bar X,\mathbf Z)_{\mathrm{tors}}\neq 0$). Thus the
$E_2^{p,q}$ are as follows:

\begin{center}
\begin{tabular}{c|ccccc}
$q$ & & & & & \\ \hline
$2$ & $\mathbf{Z}$ & $\mathbf{Z}/2$ & $E_2^{2,2}\supset 0$ & $0$ & $\mathbf{Z}$ \\
$1$ & $0$ & $0$ & $0$ & $0$ & $0$ \\
$0$ & $\mathbf{Z}$ & $0$ & $\mathbf{Z}$ & $0$ & $\mathbf{Z}$ \\ \hline
& $0$ & $1$ & $2$ & $3$ & $4$ \quad $p$
\end{tabular}
\end{center}

The differentials $d_r$ can only vanish, so $E_2^{p,q}=E_\infty^{p,q}$, and the
lemma follows.

One sees that it was essential to have two curves $C_i\subset \mathbf P^2(\mathbf C)$
along which $X$ degenerates, because one kills a $\mathbf Z/2$ in passing from
\[
H^1\bigl(\Ker(R^2p_*\mathbf Z\to \mathbf Z)\bigr)=H^1(R^2p_*\mathbf Z)=E_2^{1,2}
\]
to $H^1(R^2p_*\mathbf Z)=E_2^{1,2}$.

It remains to carry out the promised constructions. Artin and Mumford are able to
prove \emph{a priori} the existence of a conic bundle of the required type. I shall
content myself with giving equations that define one.

\medskip
\noindent 1) The image of $f_1f_2$ in $H^0(Q,\mathcal O(6))$ is the square of an
element $g\in H^0(Q,\mathcal O(3))$, because $Q$ is rational and the zeros of
$f_1f_2$ on $Q$ are double. This $g$ lifts to an element still denoted
$g\in H^0(\mathbf P^2(\mathbf C),\mathcal O(3))$ because
$H^1(\mathbf P^2(\mathbf C),\mathcal O(1))=0$. Since $f_1f_2-g^2$ vanishes on $Q$,
it is a multiple of $q$:
\[
f_1f_2=g^2+qd.
\]

\noindent 2) Take
$V=\mathcal O(-1)\oplus \mathcal O(-2)\oplus \mathcal O$, and the quadratic form
\[
\Phi\colon V\to \mathcal O\colon (x,y,z)\longmapsto qx^2+2gxy-dy^2-z^2.
\]
Then $\Delta=g^2+qd=f_1f_2$. At a point where $\Phi$ had rank $\leqslant 1$, one
would have $\Delta=q=d=0$, hence $g=0$, and $\Delta=g^2+qd$ would have a double
zero. This is impossible because $C_1\cap C_2\cap Q=\varnothing$. Finally, $S$ is
identified with the subscheme of $X$ cut out by the equation $y=0$, and d) and e)
follow.
