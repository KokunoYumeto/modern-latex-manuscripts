\begin{center}
\textbf{A CONGRUENCE BETWEEN MULTINOMIAL COEFFICIENTS}

\smallskip
\small by P.~DELIGNE

\smallskip
\footnotesize\textit{Institut des Hautes Études Scientifiques, Bures-sur-Yvette}
\end{center}

Let $p$ be a prime number. For every nonzero integer $m$, let $\vord(m)$ denote the exponent of the highest power of $p$ dividing $m$. We also put $\vord(0)=\infty$.

For a $k$-tuple $\mathfrak n=(n_1,\ldots,n_k)$ of integers $\geq 0$, set
\[
|\mathfrak n|=\sum_1^k n_i,\qquad
 a\mathfrak n=(an_1,\ldots,an_k),\qquad
 X^{\mathfrak n}=\prod_1^k X_i^{n_i},\qquad
 \mathfrak n!=\prod_1^k n_i!.
\]
Let $C(\mathfrak n)$ be the coefficient of the monomial $X^{\mathfrak n}$ in
\[
\left(\sum_1^k X_i\right)^{|\mathfrak n|};
\]
these are the multinomial coefficients. Thus
\begin{equation}
\left(\sum_1^k X_i\right)^n=\sum_{|\mathfrak n|=n} C(\mathfrak n)X^{\mathfrak n}
\end{equation}
\begin{equation}
C(\mathfrak n)=\frac{|\mathfrak n|!}{\mathfrak n!}.\tag{2}
\end{equation}

For $k=2$, these are the binomial coefficients:
\begin{equation}
C(n_1,n_2)=\binom{n_1+n_2}{n_1}.\tag{3}
\end{equation}

\begin{prop}
Let $\mathfrak n$ be a $k$-tuple of integers $\geq 0$, and put $n=|\mathfrak n|$. If $p$ is a prime number $\geq 5$, then
\begin{equation}
C(p^l\mathfrak n)\equiv C(\mathfrak n)\pmod {p^{\vord(n)+3}}.\tag{4}
\end{equation}
\end{prop}

\noindent\textsc{Example.} --- For $p=5$, $k=2$, $l=1$, and $\mathfrak n=(1,1)$, one has
\[
\binom{10}{5}=252=\binom{2}{1}+2\cdot 5^3.
\]

\medskip
\noindent\textsc{Proof.} --- It is known, or follows from (2), that
\begin{equation}
\left(\sum_1^k X_i\right)^{p^l}=\left(\sum_1^k X_i^{p^l}\right)+pQ,\tag{5}
\end{equation}
where $Q$ is a polynomial with integer coefficients, of degree $<p^l$ in each variable. The binomial formula then gives
\begin{equation}
\left(\sum_1^k X_i\right)^{p^l n}=\left(\sum_1^k X_i^{p^l}\right)^n+
\sum_1^n p^i\binom{n}{i}R_i,\tag{6}
\end{equation}
\begin{equation}
R_i=\left(\sum_1^k X_i^{p^l}\right)^{n-i}Q^i.\tag{7}
\end{equation}

For two polynomials $A$ and $B$ with integer coefficients, write $A\equiv B\pmod r$ if the coefficients of $X^{p^l\mathfrak n}$ in $A$ and $B$ are congruent modulo $r$. With this notation, (4) is rewritten as
\begin{equation}
\left(\sum_1^k X_i\right)^{p^l n}\equiv \left(\sum_1^k X_i^{p^l}\right)^n\pmod {p^{\vord(n)+3}}.\tag{8}
\end{equation}

\begin{lem}
\begin{equation}
\vord\!\left(\binom{n}{i}\right)\geq \vord(n)-\vord(i).\tag{9}
\end{equation}
\end{lem}

Let the additive group $\mathbb Z/(n)$ act on itself by translation. Thus each $x\in\mathbb Z/(n)$ determines the permutation $y\mapsto x+y$ of $\mathbb Z/(n)$. If a subset $P$ with $i$ elements in $\mathbb Z/(n)$ has stabilizer $H$, that is, if $H\cdot P=P$, then $P$ is a union of cosets of $H$, and $\#H$ divides $i$. The cardinal
\[
r_P=n/\#H
\]
of the orbit of $P$ under $\mathbb Z/(n)$, acting on the power set $\mathscr P(\mathbb Z/(n))$, therefore satisfies
\[
p^{\vord(n)-\vord(i)}\mid r_P.
\]
Since the set of subsets of $\mathbb Z/(n)$ with $i$ elements is a disjoint union of orbits, it follows that
\[
p^{\vord(n)-\vord(i)}\mid \binom{n}{i}.
\]

\begin{lem}
For $p\neq 2,3$ and $i\geq 3$, one has
\begin{equation}
p^i\binom{n}{i}\equiv 0\pmod {p^{\vord(n)+3}}.\tag{10}
\end{equation}
\end{lem}

Indeed,
\[
\vord\!\left(p^i\binom{n}{i}\right)=i+\vord\!\left(\binom{n}{i}\right)
\geq \vord(n)+(i-\vord(i))
\]
by Lemma~1. If $3\leq i<p$, then $\vord(i)=0$ and $i-\vord(i)\geq 3$. If $i\geq p$, then
\[
i-\vord(i)\geq i-\log i/\log p.
\]
The last function of $i$ is increasing for $i\geq p$, and its value at $i=p$ is $p-1\geq 3$.

\begin{lem}
The coefficient of $X^{p^l\mathfrak n}$ in $R_1$ is zero.
\end{lem}

No monomial $X^{\mathfrak r}$ appearing in $Q$ has an exponent $\mathfrak r$ divisible by $p^l$. The polynomial
\[
R_1=\left(\sum_1^k X_i^{p^l}\right)^{n-1}Q
\]
inherits this property.

By Lemmas~2 and~3,
\begin{equation}
\left(\sum_1^k X_i\right)^{p^l n}\equiv \left(\sum_1^k X_i^{p^l}\right)^n+
\binom{n}{2}p^2R_2\pmod {p^{\vord(n)+3}}.\tag{11}
\end{equation}
This congruence already implies, by Lemma~1, that
\begin{equation}
C(p^l\mathfrak n)\equiv C(\mathfrak n)\pmod {p^{\vord(n)+2}}.\tag{12}
\end{equation}

The only monomials $X^{\mathfrak r}$ with exponent divisible by $p^l$ which occur in $(pQ)^2$ are the monomials
\[
X_i^{p^l}X_j^{p^l}\qquad (i<j).
\]
The coefficient of each of them is
\begin{equation}
\sum_{i\neq 0,p^l}\binom{p^l}{i}\binom{p^l}{p^l-i}=
\sum \binom{p^l}{i}^2-2=\binom{2p^l}{p^l}-2.\tag{13}
\end{equation}
By (12),
\[
\binom{2p^l}{p^l}-2\equiv \binom{2p}{p}-2\pmod {p^3}.
\]
Thus
\[
\left(\sum_1^k X_i\right)^{p^l n}\equiv
\left(\sum_1^k X_i^{p^l}\right)^n+
\binom{n}{2}\left[\binom{2p}{p}-2\right]R_2
\pmod {p^{\vord(n)+3}},
\]
with $R_2$ a polynomial with integer coefficients.

Since
\[
\vord\!\left(\binom{n}{2}\right)\geq \vord(n)
\]
by Lemma~1, the proposition follows from the next lemma.

\begin{lem}
For every prime $p\geq 5$,
\[
\binom{2p}{p}\equiv 2\pmod {p^3}.
\]
\end{lem}

For $1\leq i\leq p-1$, one has
\[
p^{-1}\binom{p}{i}\equiv (-1)^{i-1}i^{-1}\pmod p.
\]
Hence
\begin{equation}
p^{-2}\left[\binom{2p}{p}-2\right]
=\sum_1^{p-1} p^{-2}\binom{p}{i}^2
\equiv \sum_1^{p-1}(1/i)^2\pmod p.\tag{14}
\end{equation}
Define
\[
z\in\mathbb Z/(p),\qquad z=\sum_{i\in\mathbb Z/(p)^*}1/i^2.
\]
For $u\in\mathbb Z/(p)^*$, one has
\[
z=\sum_{i\in\mathbb Z/(p)^*}1/(u^{-1}i)^2=u^2z,
\]
and therefore
\[
(u^2-1)z=0.
\]
Since $p\neq 2,3$, there exists $u$ such that $u^2\neq 1$, and hence $z=0$.

Formula (14) consequently becomes
\[
p^{-2}\left[\binom{2p}{p}-2\right]\equiv 0\pmod p,
\]
which completes the proof.
