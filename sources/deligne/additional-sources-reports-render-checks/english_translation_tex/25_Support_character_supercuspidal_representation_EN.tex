\title{The Support of the Character of a Supercuspidal Representation}
\author{Pierre Deligne}
\date{C. R. Acad. Sci. Paris, t. 283, Serie A, 24 May 1976}
\maketitle

The character of a supercuspidal representation of a semisimple group over a local field is supported on the union of the compact open subgroups.

\section*{1. Review of monogenic groups}
Let $V$ be a finite-dimensional vector space over a field $k$, and let $g\in\GL(V)$. If $k$ is perfect, the Zariski closure $\langle g\rangle$ of the group generated by $g$ is the product of a multiplicative group $\langle g\rangle_m$ and a unipotent group $\langle g\rangle_{\mathrm{un}}$. If $g=su=us$ is the Jordan decomposition of $g$ into its semisimple and unipotent parts, then
\[
\langle g\rangle_m=\langle s\rangle,\qquad \langle g\rangle_{\mathrm{un}}=\langle u\rangle.
\]
Without any hypothesis on $k$, define the subgroup $\langle g\rangle_m$ of $\langle g\rangle$ as the Zariski closure of the union of the subgroups of $\langle g\rangle$ that are kernels of multiplication by $n$, where $n$ is prime to the characteristic exponent of $k$. This definition is compatible with extension of scalars, and for $k$ perfect it is equivalent to the preceding one. Over an algebraic closure $\bar k$ of $k$, the character group of $\langle g\rangle_m$ is the subgroup $X$ of $\bar k^*$ generated by the eigenvalues of $g$, and
\[
s\in \langle g\rangle_m(\bar k)=\Hom(X,\bar k^*)
\]
corresponds to the identity inclusion of $X$.

Assume that $k$ is a complete valued field, with uniformizer $\pi$. Let $v$ be the valuation of $k$, with values in $\Q$, normalized by $v(\pi)=1$, and let $n>0$ be an integer such that $nv(X)\subset\Z$. The homomorphism $nv(\chi):X\to\Z$ transposes to a homomorphism
\[
m_g:\Gm\to \langle g\rangle_m.
\]

For a linear algebraic group $G$ over $k$ and an element $g\in G(k)$, the preceding constructions may be repeated after choosing an embedding $G\subset\GL(V)$. The homomorphism $m_g:\Gm\to\langle g\rangle_m\subset G$ does not depend on the embedding, once $n$ is fixed. For every linear representation $W$ of $G$, it defines an action of $\Gm$ on $W$, equivalently a grading of $W$ (SGA 3, I, 4.7.3). The part of degree $d$ is the sum of the generalized eigenspaces of $g$ corresponding to eigenvalues $\alpha\in\bar k$ such that $nv(\alpha)=d$.

Assume that $G$ is reductive. Every homomorphism $m:\Gm\to G$ defines two opposite parabolic subgroups of $G$, namely the parabolics contracted and expanded by $m$. Let $P_g^+$ be the parabolic contracted by $m_g$, or by $g$. It has the following equivalent characterizations:
\begin{enumerate}
\item[(a)] An element $x\in G(\bar k)$ lies in $P_g^+$, respectively in its unipotent radical $U_g^+$, if and only if the morphism
\[
\lambda\mapsto m_g(\lambda)x m_g(\lambda)^{-1}:\Gm\to G
\]
extends to $\Ga$, respectively if this extension has limit $e$ as $\lambda\to0$. Equivalently, $\lim_{n\to\infty}g^n x g^{-n}$ exists, respectively equals $e$.
\item[(b)] $\Lie P_g^+$, respectively $\Lie U_g^+$, is the sum in $\Lie G$ of the generalized eigenspaces of $\ad g$ whose eigenvalues have valuation $\ge0$, respectively $>0$.
\end{enumerate}
Put $P_g^-=P_{g^{-1}}^+$, the parabolic expanded by $g$ and contracted by $m_{g^{-1}}=m_g^{-1}$, and let $U_g^-$ be its unipotent radical. The parabolics $P_g^+$ and $P_g^-$ are opposite, and put $M_g=P_g^+\cap P_g^-$.

\section*{2. Statement and proof of the theorem}
Assume that $k$ is locally compact and that $G$ is reductive. Let $\tau$ be the projection from $G$ to its adjoint group $G^{\mathrm{ad}}$. An admissible representation $(\mathscr H,\rho)$ of $G(k)$ is \emph{supercuspidal} if its matrix coefficients are supported on inverse images of compact subsets of $G^{\mathrm{ad}}(k)$.

If $K$ is a compact subgroup of $G(k)$, let $dk$ be the normalized Haar measure on $K$ and let $[K]$ be its image in the algebra of compactly supported distributions on $G(k)$. The operator $\rho(K)=\int_K\rho([K])$ is the projector from $\mathscr H$ onto the subspace $\mathscr H^K$ of $K$-invariants. Saying that $\rho$ is supercuspidal is equivalently saying that, for every compact open subgroup $K$, one has $\rho(K)\rho(g)\rho(K)=0$ for $\tau(g)\in G^{\mathrm{ad}}$ outside a suitable compact set depending on $K$. Let $G_c^{\mathrm{ad}}$ denote the union of the compact subgroups, or equivalently of the compact open subgroups, of $G^{\mathrm{ad}}(k)$.

\begin{theorem*}
If $(\mathscr H,\rho)$ is an admissible supercuspidal representation of $G(k)$, then its character is supported on $\tau^{-1}(G_c^{\mathrm{ad}})$.
\end{theorem*}

Choose local coordinates near $e$: a lattice $L\subset\Lie(G)$ and an analytic isomorphism $\alpha$ from $L$ to a neighborhood of $e$, such that
\begin{enumerate}
\item[(1)] $\alpha(0)=e$, and the differential of $\alpha$ at $0$ is the identity on $\Lie(G)$.
\end{enumerate}
Let $g\in G(k)$. We choose $L$ so that
\begin{enumerate}
\item[(2)] $L=(L\cap\Lie U_g^+)\oplus(L\cap\Lie M_g)\oplus(L\cap\Lie U_g^-)$. Write this as $L^+\oplus L^0\oplus L^-$.
\item[(3)] $\ad g(L^+)\subset L^+$, $\ad g(L^0)=L^0$, and $\ad g^{-1}(L^-)\subset L^-$.
\end{enumerate}
Since $P_g^+$ and $P_g^-$ are stable under $\ad g$, condition (3) may also be written as
\begin{enumerate}
\item[(3$'$)] $\ad g(L\cap\Lie P_g^+)\subset L$ and $\ad g^{-1}(L\cap\Lie P_g^-)\subset L$.
\end{enumerate}
Denote by an index $i$ the intersection with $G_i=\alpha(\pi^iL)$. Under hypotheses (2) and (3), there is $i_0$ such that, for $i\ge i_0$,
\begin{enumerate}
\item[(4)] $G_i$ is a subgroup of $G(k)$, and $G_i=U_{gi}^+M_{gi}U_{gi}^-$;
\item[(5)] $\Int g(U_{gi}^+)\subset U_{gi}^+$, $\Int g(M_{gi})=M_{gi}$, and $\Int g^{-1}(U_{gi}^-)\subset U_{gi}^-$.
\end{enumerate}
This follows from the decomposition
\[
U_g^+\times M_g\times U_g^-\hookrightarrow G,
\quad (u^+,m,u^-)\mapsto u^+mu^-,
\]
of a neighborhood of $e$.

Moreover, after increasing $i_0$ if necessary, there is a neighborhood $V$ of $g$ such that (2)--(5) remain true after replacing $g$ by any $g'\in V$. For (2) and (3$'$), one uses the continuous dependence of $P_g^+$ and $\ad g$ on $g$; for (4) and (5), one uses the analytic dependence of $m_g$, and hence of $P_g^\pm$, on $g$, so that in the decomposition $x=u^+mu^-$ the components $u^+$, $m$, and $u^-$ are analytic functions of the pair $(x,g)$.

\begin{lemma*}
Assume (1)--(5), let $i\ge i_0$, and suppose $\tau(g)\notin G_c^{\mathrm{ad}}$. Then $\rho(G_i)\rho(g)\rho(G_i)$ is nilpotent.
\end{lemma*}

For the big-cell decomposition $U_g^+\times M_g\times U_g^-\to G$, Haar measures satisfy $dg=\lambda(m)\,du^+\,dm\,du^-$, where $\lambda$ is a character of $M_g$. On every compact subgroup of $M_g$, $\lambda=1$, so
\[
[G_i]=[U_{gi}^+]\star[M_{gi}]\star[U_{gi}^-].
\]
We prove by induction on $N$ that
\[
([G_i]g[G_i])^N=[G_i]g^N[G_i]
\]
in the algebra of compactly supported distributions on $G$, writing $g$ for $\delta_g$. Indeed,
\begin{align*}
([G_i]g^N[G_i])([G_i]g[G_i])
&=[G_i]g^N[G_i]g[G_i]\\
&=[G_i]g^N[U_{gi}^+][M_{gi}][U_{gi}^-]g[G_i]\\
&=[G_i][g^NU_{gi}^+g^{-N}][g^NM_{gi}g^{-N}]g^Ng[g^{-1}U_{gi}^-g][G_i]\\
&=[G_i]g^{N+1}[G_i].
\end{align*}

If $\tau(g)\notin G_c^{\mathrm{ad}}$, then the set of $\tau(g^N)$, $N\ge1$, is not relatively compact. Since $\rho$ is supercuspidal, there is an $N$ such that
\[
(\rho(G_i)\rho(g)\rho(G_i))^N=\rho(G_i)\rho(g^N)\rho(G_i)=0.
\]

\begin{proof}[Proof of the theorem]
Let $g\in G(k)$ be such that $\tau(g)\notin G_c^{\mathrm{ad}}$. Choose $L$, $\alpha$, $i_0$, and $V$ as above, with $gG_{i_0}\subset V$. Let $\Theta$ be the character of $\rho$. For every $g'\in gG_i$ and every $i\ge i_0$, we have
\[
\Theta(g'[G_i])=\Theta([G_i]g'[G_i])=\Tr(\rho(G_i)\rho(g')\rho(G_i))=0
\]
by the lemma. Since the subgroups $G_i$ form a fundamental system of neighborhoods, $\Theta$ is zero on $gG_i$; hence its support does not contain $g$.
\end{proof}

\begin{remarks*}
(1) The same theorem, with the same proof, holds for an analytic group that is a finite etale cover of an open subgroup of $G$. The hypothesis becomes: the image in $G^{\mathrm{ad}}$ of the support of the matrix coefficients is relatively compact. The conclusion becomes: the image in $G^{\mathrm{ad}}$ of the support of $\Theta$ lies in $G_c^{\mathrm{ad}}$.

(2) Let $p$ be the residue characteristic of $k$. The proof uses Haar measures $[K]$ in an essential way only for small $K$, hence in particular for a $p$-group. Therefore the theorem and its proof also hold for representations with coefficients in any field of characteristic $\ne p$.
\end{remarks*}

Once a Haar measure $dg$ on $G$ is chosen, the character $\Theta$ of $\rho$ is the distribution on $G$ whose value on a test function $\varphi$ is
\[
\Tr\left(\int\varphi(g)\rho(g)\,dg\right).
\]
More intrinsically, it is the generalized function, i.e. the linear form on the space of measures $\varphi(g)\,dg$ with $\varphi$ locally constant of compact support and $dg$ a Haar measure, given by the same formula.
