\noindent\textbf{Abstract.}
We give an analogue, in super mathematics, to the theorem that over an algebraically closed field of characteristic zero, categories of representations of affine group schemes, with their associative, commutative and unital tensor product, are characterized by the property that for any object, large enough exterior powers vanish.

\smallskip
\noindent\textbf{Keywords and phrases.} tensor category, Tannaka duality, fiber functor, supergroup.

\section{Introduction}

\subsection{}
\label{0.1}

Fixons a field algebriquement clos $k$ of caracteristique $0$. We appellerons \emph{category $k$-tensor} a systeme $(A,\otimes,\text{donn\'ees auxiliaires})$ of the type suivant.

\begin{itemize}
\item[(0.1.1)] $A$ is a category essentiellement petite (equivalent a petite category);
\item[(0.1.2)] it is abelienne, $k$-lineaire;
\item[(0.1.3)] the tensor product $\otimes$ is a functor of $A\times A$ in $A$ exact and $k$-lineaire en each variable;
\item[(0.1.4)] it is equipped with contraintes of associativity, of commutativity and of unite (the unite will be notee $1$);
\item[(0.1.5)] $(A,\otimes)$ is rigide: all object $X$ of $A$ is dualisable, to the sens that there exists $X^\vee$, the dual of $X$, equipped with $\delta:1\to X\otimes X^\vee$ and $\ev:X^\vee\otimes X\to 1$, such that the morphisms composes deduits of $\delta$ and $\ev$:
\[
X\to X\otimes X^\vee\otimes X\quad\text{et}\quad X^\vee\to X^\vee\otimes X\otimes X^\vee
\]
let the identity;
\item[(0.1.6)] one has $k\xrightarrow{\sim}\End(1)$.
\end{itemize}

A category $k$-tensor will be dite \emph{of $\otimes$-generation finite} si it admet a $\otimes$-generateur: a object $X$ of which all object se deduces by map iteree of operations of direct sum, tensor product, dual, passage a under-object or a object quotient.

\section{Preliminaires}\label{1}

\begin{proposition}\label{1.1}
In a category $k$-tensor of which all the objects are of longueur finite, the $\Hom(X,Y)$ are of dimension finite.
\end{proposition}

\begin{proof}
The identites (0.1.4) between $\delta$ and $\ev$ equivalent a this that the functor $Y\mapsto Y\otimes X^\vee$ let adjoint a line to the functor $Y\mapsto Y\otimes X$, for the morphisms of adjonction $Y\to Y\otimes X\otimes X^\vee$ and $Y\otimes X^\vee\otimes X\to Y$ deduits of $\delta$ and $\ev$. En of other termes, $\Hom(X,Y):=Y\otimes X^\vee$ is a $\Hom$ interne: one dispose of a isomorphism fonctoriel
\[
\Hom(Z,\Hom(X,Y))=\Hom(Z\otimes X,Y).
\]
For $Z=1$, one obtains
\begin{equation}\label{1.1.1}
\Hom(1,\Hom(X,Y))=\Hom(X,Y).
\end{equation}
the object $1$ being simple (Deligne-Milne 1982, 1.17), it follows of (0.1.5) and (\ref{1.1.1}) that $n$ morphisms lineairement independants $f_i:X\to Y$ definissent a monomorphisme of $1^n$ in $\Hom(X,Y)$. Si $\Hom(X,Y)$ is of longueur finite, sa longueur borne $n$ and $\Hom(X,Y)$ is of dimension finite.
\end{proof}

\begin{align}

\begin{equation}\label{1.3.3}
\Hom_S(V,X):=\Hom(V,X)^S.
\end{equation}

Si one choisit a representant $V_\lambda$ of each class of isomorphie of representations lineaires irreductibles of $S$, it follows formally of this that $k[S]\xrightarrow{\sim}\prod_\lambda\End_k(V_\lambda)$ that the map
\begin{equation}\label{1.3.4}

\end{equation}
is a isomorphism.

Let $n_1,\ldots,n_r$ of the integers of sum $n$ and plongeons the product of the $S_{n_i}$ in $S_n$ en identifiant $\{1,\ldots,n\}$ a the sum disjointe of the $\{1,\ldots,n_i\}$. Si $\lambda$ is a partition of $n$, one denotes $[\lambda:\mu_1,\ldots,\mu_r]$ the multiplicite of this representation in the restriction of $V_\lambda$ a $\prod S_{n_i}$. By reciprocite of Frobenius, c'is also the multiplicite of $V_\lambda$ in the induite of $\bigotimes V_{\mu_i}$ a $S_n$:
\[

These multiplicites are data by the regle of Littlewood-Richardson of which we shall use the consequences suivantes.

\begin{itemize}
\item[(1.5.1)] Si $|\lambda|=|\mu|+1$, one has $[\lambda:\mu,(1)]=1$ si $[\mu]\subset[\lambda]$, $0$ otherwise. Utilisant the formule
\[
[\lambda:\mu_1,\mu_2,\mu_3]=\sum_{\lambda'}[\lambda:\lambda',\mu_3][\lambda':\mu_1,\mu_2]

\item[(1.5.2)] Fixons $r$ and a partition $\lambda$ of $n$. For that there exists $n_1,\ldots,n_r$ of sum $n$ such that $[\lambda:(n_1),\ldots,(n_r)]\neq0$, it is necessary and sufficient that $[\lambda]$ ait to the more $r$ lignes.
\item[(1.5.3)] Fixons $r$, $s$ and a partition $\lambda$ of $n$. For that there exists $n_1,\ldots,n_r$, $m_1,\ldots,m_s$ of sum $n$ such that $[\lambda:(n_1),\ldots,(n_r),(m_1)^t,\ldots,(m_s)^t]\neq0$, it is necessary and sufficient that
\[
[\lambda]\subset\{(i,j)\mid i\leq r\text{ ou }j\leq s\},

\begin{equation}\label{4.8.1}
\longueur(X^{\otimes n})\leq N^n.
\end{equation}
\end{lemma}

\begin{proof}
Let $K(A)$ the group of Grothendieck of the abelian category $A$. Si $(S_i)_{i\in I}$ is a systeme of representants of the classes of isomorphie of objects simples, $K(A)$ is the $\mathbb{Z}$-module free of base the $[S_i]$. Let $\ell:K(A)\to\mathbb{Z}$ the map ``sum of the coordonnees''. For $X$ in $A$, one has
\[
\longueur(X)=\ell([X]).
\]
The tensor product of $A$ fait of $K(A)$ a ring commutative, and (\ref{4.8.1}) se recrit
\[
\ell([X]^n)\leq N^n.
\]
The matrice of the multiplication by $[X]$ is a coefficients integers $\geq0$. Si $|I|$ is leur maximum, one has $\ell([X][Y])\leq|I|\cdot a\cdot\ell(Y)$ and $\ell([X]^n)\leq(|I|\cdot a)^n$.
\end{proof}

\begin{proof}[Proof of \ref{0.8}]
Supposons en outre $A$ semisimple, and let $\omega$ a functor fiber of $A$ on $k$.

Let $X$ the sum of the $S_i$. All object of $A$ being a under-object of a sum of copies of $X$, the map canonical of $\omega(X)\otimes\omega(X)^\vee$ in $\Lambda(\omega,\omega)$ is surjective, and $\Lambda(\omega,\omega)$ is therefore of dimension finite.

The category $k$-tensor $A$ is of the form $\Rep(G,\varepsilon)$ with $O(G)=\Lambda(\omega,\omega)$, of dimension finite. Divisons the algebra affine $O(G)$ of $G$ by the ideal generated by sa part impaire. One obtains the algebra affine of the under-algebraic group $G_{\text{red}}$ of $G$, necessarily finite.

Let $G^0$ the composante connected of $G$. Puisque $G_{\text{red}}$ is trivial, the algebra of Lie of $G^0$ is purement impaire. It determine $G^0$: $O(G^0)$ is the dual of the algebra enveloppante of the superalgebre of Lie commutative $\operatorname{Lie}(G^0)$.

Puisque $G^0$ is invariant in $G$, the restriction a $G^0$ of a super representation semisimple of $G$ is semisimple, therefore trivial: $\omega$ induces a equivalence of $A$ with $\Rep(G/G^0,\varepsilon)$.
\end{proof}
