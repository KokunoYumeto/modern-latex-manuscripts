Cet exposé contient une partie des résultats fondamentaux de Griffiths sur les

\section{Structures de Hodge}

Soit $H_\bR$ un espace vectoriel réel de dimension finie.

\begin{definition}\label{def:1.1}
Une \emph{structure de Hodge réelle} sur $H_\bR$ est une bigraduation de
\[
H_\bC = H_\bR \otimes_\bR \bC = \bigoplus_{p,q} H^{pq}
\]
telle que $H^{pq}$ et $H^{qp}$ soient complexes conjugués.
\end{definition}

Les nombres de Hodge d'une structure de Hodge réelle $H$ sont les entiers
$h^{pq} = \dim_\bC(H^{pq})$. On dit que $H$ est \emph{de poids} $n$ si $h^{pq} = 0$ pour $p+q \neq n$.

Une variante de \ref{def:1.1} est :

\begin{definition}\label{def:1.2}
Une structure de Hodge réelle de poids $n$ sur $H_\bR$ est une filtration finie décroissante $F$ de $H_\bC$ (la filtration de Hodge) telle que, pour $p+q = n+1$, on ait
\begin{equation}\label{eq:1.2.1}
F^p(H_\bC) \oplus F^q(H_\bC) \xrightarrow{\sim} H_\bC.
\end{equation}
\end{definition}

On passe de \ref{def:1.1} à \ref{def:1.2} en posant
\begin{equation}\label{eq:1.2.2}
F^p(H_\bC) = \bigoplus_{p' \geq p} H^{p',q'},
\end{equation}
et on passe de \ref{def:1.2} à \ref{def:1.1} en posant, pour $p+q = n$,
\begin{equation}\label{eq:1.2.3}
H^{pq} = F^p(H_\bC) \cap F^q(H_\bC).
\end{equation}

\begin{definition}\label{def:1.3}
Désignons par $\mathbb{S}$ le groupe algébrique réel obtenu par restriction des scalaires à la Weil, de $\bC$ à $\bR$, du groupe multiplicatif $\mathbb{G}_m$. On a $\mathbb{S}(\bR) = \bC^*$ ; on dispose d'un morphisme naturel $w$ de $\mathbb{G}_m$ dans $\mathbb{S}$ qui, sur les points réels, est l'inclusion de $\bR^*$ dans $\bC^*$, et on dispose d'un morphisme naturel $t$ de $\mathbb{S}$ dans $\mathbb{G}_m$ qui, sur les points réels, est la norme. Le composé $tw$ est l'élévation au carré. Une nouvelle variante de \ref{def:1.1} est :
\end{definition}

\begin{definition}\label{def:1.4}
Une structure de Hodge réelle sur $H_\bR$ est une action (définie sur $\bR$) du groupe algébrique $\mathbb{S}$ sur $H_\bR$.
\end{definition}

Si $z$ désigne l'isomorphisme de définition $z : \mathbb{S}(\bR) \xrightarrow{\sim} \bC^*$, on passe de \ref{def:1.1} à \ref{def:1.4} en définissant $H^{pq}$ comme le sous-espace de $H_\bC$ où $\mathbb{S}(\bR)$ agit par multiplication par $z^p\bar{z}^q$.

Si $H_\bR$ est muni d'une structure de Hodge réelle, on définit l'automorphisme $C$ de Weil comme étant la multiplication par $i^{p-q}$ sur $H^{pq}$. Cet automorphisme est réel, et n'est autre que l'action de l'élément $i$ de $\mathbb{S}(\bR) \cong \bC^*$.

\begin{definition}\label{def:1.5}
Une \emph{polarisation} d'une structure de Hodge réelle, de poids $n$, $H$ est une forme bilinéaire $\Psi$ sur $H_\bR$, invariante par le sous-groupe compact $\Ker(t)$ de $\mathbb{S}$, et telle que la forme $\Psi(x,Cy)$ soit symétrique et définie positive.
\end{definition}

De l'identité $C^2 = (-1)^n$, on tire que
\[
\Psi(x,y) = (-1)^n\,\Psi(x,C^2y) = (-1)^n\,\Psi(Cy,Cx) = (-1)^n\,\Psi(y,x)
\]
de sorte que $\Psi$ est alternée ou symétrique selon la parité de $n$. Les propriétés de variance sous $\mathbb{S}$ et de positivité s'écrivent :

\begin{enumerate}
\item[(1.5.1)] L'orthogonal pour $\Psi$ de $F^p(H_\bC)$ est $F^{n-p}(H_\bC)$ ;
\item[(1.5.2)] Pour $x$ non nul dans $H^{pq}$, on a $i^{p-q}\,\Psi(x,\bar{x}) > 0$ (relations bilinéaires de Riemann).
\end{enumerate}

\begin{definition}\label{def:1.6}
\begin{enumerate}
\item[(i)] Une structure de Hodge, de poids $n$, $H$ consiste en un $\bZ$-module libre de type fini $H_\bZ$ (le ``réseau entier''), et en une structure de Hodge réelle de poids $n$ sur $H_\bR = H_\bZ \otimes_\bZ \bR$.
\item[(ii)] Une polarisation d'une structure de Hodge, de poids $n$, $H$ est une polarisation $\Psi$ de la structure de Hodge réelle sous-jacente, qui est à valeurs entières sur le réseau entier $H_\bZ$.
\end{enumerate}
\end{definition}

\section{Théorie de Hodge}

\subsection{}

Soient $X \subset \mathbb{P}^r(\bC)$ une variété algébrique complexe projective non singulière, purement de dimension $d$, et $\xi \in H^2(X,\bZ)$ la classe de cohomologie d'une section hyperplane de $X$, i.e. la première classe de Chern de $\mathscr{O}(1)$.

Le complexe de De Rham holomorphe $\Omega_X^*$ est une résolution du faisceau constant $\bC$ (lemme de Poincaré holomorphe), d'où une suite spectrale
\begin{equation}\label{eq:2.1.1}
E_1^{p,q} = H^q(X, \Omega_X^p) \Rightarrow H^{p+q}(X,\bC).
\end{equation}

\subsection{}

La théorie de Hodge affirme que

\begin{enumerate}
\item[(A)] La suite spectrale (\ref{eq:2.1.1}) dégénère ($E_1 = E_\infty$).
\item[(B)] La filtration de $H^n(X,\bC)$ à laquelle elle aboutit est une structure de Hodge de poids $n$ (\ref{def:1.2}) sur $H^n(X,\bR)$.
\end{enumerate}

D'après (A), on a $H^{pq} = H^q(X, \Omega_X^p)$.

\begin{enumerate}
\item[(C)] Pour $n \leq d$, désignons par $P_\bC^n$ le noyau du cup-produit itéré
\[
\xi^{d-n+1} : H^n(X,\bC) \longrightarrow H^{2d-n+2}(X,\bC)
\]
(partie primitive de la cohomologie), et par $P_\bZ^n$ l'intersection de $P_\bC^n$ avec l'image de $H^n(X,\bZ)$ dans $H^n(X,\bC)$. Le classe $\xi$ étant de type $(1,1)$, $P_\bR^n = P_\bZ^n \otimes_\bZ \bR$ est une sous-structure de Hodge de $H^n(X,\bR)$. À un signe ne dépendant que de $d$ et $n$ près, la forme
\[
\Psi(x,y) = \int_X \xi^{d-n} \wedge x \wedge y
\]
est une polarisation de la structure de Hodge $P^n$.

\item[(D)] L'application naturelle
\[
\bigoplus_k \xi^k \wedge : \bigoplus_k P_\bC^{n-2k} \longrightarrow H^n(X,\bC)
\]
est un isomorphisme (décomposition de Hodge-Lepage).
\end{enumerate}

\section{Familles de structures de Hodge}

\subsection{}

Soit $f : X \to S$ un morphisme projectif et lisse d'espaces analytiques, purement de dimension relative $d$. Pour simplifier, on supposera que $S$ est non singulier et que $f$ est muni d'une factorisation par $\mathbb{P}^r_S = S \times \mathbb{P}^r(\bC)$. Les fibres $X_s = f^{-1}(s)$ forment donc une famille analytique, paramétrée par $S$, de sous-variétés non singulières de $\mathbb{P}^r(\bC)$.

\subsection{}

Du point de vue $C^\infty$, $f$ est une fibration localement triviale. L'algèbre de cohomologie des fibres de $f$ forme donc un système local sur $S$, qu'on notera
\[
R^n f_* \bZ \qquad (\text{resp. } R^n f_* \bQ,\; R^n f_* \bC,\; R^n f_* \bR).
\]
Pour tout faisceau abélien $\mathscr{F}$ sur $S$ (par exemple $\bZ$, $\bQ$, $\bC$, $\bR$), on posera
\begin{equation}\label{eq:3.2.1}
R^n f_*(\mathscr{F}) = \bC \otimes_\bZ R^n f_* \bZ.
\end{equation}

Puisque $f$ est propre, on a encore $R^n f_* \bC = R^n f_* (f^*\bC_S)$.

Les classes de cohomologie des sections hyperplanes des $X_s$ définissent une section localement constante
\begin{equation}\label{eq:3.2.2}
\xi \in H^0(S, R^2 f_* \bZ).
\end{equation}

Pour $n \leq d$, les $P_\bC^n(X_s)$ forment donc un sous-système local $P_\bC^n(f)$ de $R^n f_*(\bC)$. Posant $P_\bR^n(f) = \bR \otimes_\bZ P_\bZ^n(f)$, on a par définition
\begin{equation}\label{eq:3.2.3}
P_\bR^n(f) = \Ker(\xi^{d-n+1} \wedge : R^n f_*(\bR) \longrightarrow R^{2d-n+2} f_*(\bR)).
\end{equation}

La forme de polarisation 2.2 (C)
\begin{equation}\label{eq:3.2.5}
\Psi(x,y) = \pm \int_{X_s} \xi^{d-n} \wedge x \wedge y
\end{equation}
est localement constante sur $S$, i.e. définit
\begin{equation*}
\Psi : R^n f_*(\bR) \otimes R^n f_*(\bR) \longrightarrow \bR.
\end{equation*}

\subsection{}

Pour $\mathscr{F}$ un faisceau analytique cohérent sur $S$, on désignera par $f^*\mathscr{F}$ son image réciproque faisceautique, et par $f^*\mathscr{F} = \mathscr{O}_X \otimes_{f^{-1}\mathscr{O}_S} f^{-1}\mathscr{F}$ son image réciproque en tant que faisceau de modules. Pour $s \in S$, on désignera par $\mathscr{F}_{(s)}$ le $\mathscr{O}_{S,s}$-module des germes de sections de $\mathscr{F}$ en $s$ ; on appellera \emph{fibre} de $\mathscr{F}$ en $s$ le vectoriel $\mathscr{F}_s = \mathscr{F}_{(s)} \otimes_{\mathscr{O}_{S,s}} \bC$. Rappelons enfin que le complexe de De Rham relatif $\Omega_{X/S}^*$ est le quotient de $\Omega_X^*$ de composantes $\Omega_{X/S}^n = \Omega_X^n / f^*\Omega_S^1 \wedge \Omega_X^{n-1}$. Si $T_{X/S}$, dual de $\Omega_{X/S}^1$, est le fibré tangent relatif, on dispose d'accouplements ``produits contractés''
\begin{equation*}
\llcorner \, : \Omega_{X/S}^p \otimes T_{X/S} \longrightarrow \Omega_{X/S}^{p-1}.
\end{equation*}

\subsection{}

D'après le lemme de Poincaré holomorphe relatif, $\Omega_{X/S}^*$ est une résolution de $f^*\bC_S$. On dispose donc d'une suite spectrale
\begin{equation}\label{eq:3.4.1}
E_1^{p,q} = R^q f_* \Omega_{X/S}^p \Rightarrow R^{p+q} f_* (\bC).
\end{equation}

On déduit de 2.2 (A) que les $E_1^{p,q}$ sont localement libres, que la suite spectrale (\ref{eq:3.4.1}) dégénère ($E_1 = E_\infty$) et que sa formation commute à tout changement de base. La filtration de $R^n f_*(\bC)$ à laquelle elle aboutit (la filtration de Hodge) induit donc sur $R^n f_*(\bC)_s \cong H^n(X_s, \bC)$ la filtration de Hodge 2.2 : cette dernière varie de façon holomorphe avec $s \in S$.

\begin{definition}\label{def:3.5}
La \emph{connexion de Gauss-Manin} sur le fibré vectoriel holomorphe $R^n f_*(\bC)$ (identifié ici à son faisceau de sections holomorphes) est la connexion holomorphe et intégrable
\begin{equation*}
\nabla : R^n f_*(\bC) \longrightarrow \Omega_S^1 \otimes R^n f_*(\bC)
\end{equation*}
ayant pour sections locales horizontales ($\nabla v = 0$) les sections locales de $R^n f_*(\bC)$.
\end{definition}

\begin{theorem}[Théorème de transversalité \ref{def:3.5}--\ref{thm:3.6}]\label{thm:3.6}
La filtration de Hodge $F$ sur $R^n f_*(\bC)$ vérifie
\begin{equation*}
\nabla F^p(R^n f_*(\bC)) \subset \Omega_S^1 \otimes F^{p-1}(R^n f_*(\bC)).
\end{equation*}
\end{theorem}

\begin{proof}[Preuve] (d'après Katz-Oda \cite{10} et une communication personnelle de Katz). La question est locale sur $S$. Soit $\mathfrak{U} = (U_i)_{0 \leq i \leq N}$ un recouvrement fini de $X$ par des ouverts de Stein. Pour $Q \subset [0,N]$, soient $U_Q = \bigcap_{i \in Q} U_i$, $j_Q$ l'inclusion de $U_Q$ dans $X$, et
\begin{equation*}
f_*(\mathfrak{U}, \Omega_{X/S}^*)^q = \bigoplus_{\#Q = q+1} f_*(j_Q)_* \Omega_{U_Q/S}^*.
\end{equation*}

On désignera par $f_*(\mathfrak{U}, \Omega_{X/S}^*)$ le complexe double de faisceaux sur $S$ de composantes
\begin{equation}\label{eq:3.6.1}
f_*(\mathfrak{U}, \Omega_{X/S}^*)^{p,q} = \bigoplus_{\#Q = q+1} f_*(U_Q, \Omega_{X/S}^p)
\end{equation}
de première différentielle induite par la différentielle extérieure, et de seconde différentielle ``Čechiste''. La suite spectrale \ref{eq:3.4.1} est la suite spectrale de ce double complexe déduite de la filtration par le premier degré.

Soit $v$ un champ de vecteurs (holomorphe) sur $S$, et supposons donné sur chaque $U_i$ un champ de vecteurs $v_i$ relevant $v$. On désignera par $\theta(v_i)$ l'endomorphisme de $f^*(\Omega_{X/S}^n)$, vérifiant, pour $Q = (i_1, \ldots, i_q)$, avec $i_1 < \cdots < i_q$, et de coordonnées la dérivée de Lie $L_{v_i} : f_*(U_Q, \Omega_{X/S}^p) \to f_*(U_Q, \Omega_{X/S}^p)$ et les produits contractés
\begin{equation*}
\sum_{i_0} (-1)^p (v_{i_0} - v_{i_0}) \,\llcorner\, : f_*(U_Q, \Omega_{X/S}^p) \longrightarrow \bigoplus_{i_0} f_*(U_{\{i_0\} \cup Q}, \Omega_{X/S}^{p-1}).
\end{equation*}

Un calcul facile montre que $\theta(v_i)$ commute à $d$.

\begin{lemma}\label{lem:3.7}
Sur les faisceaux de cohomologie $R^n f_*(\bC)$, $\theta(v_i)$ induit $\nabla_v$.
\end{lemma}

Soit $\mathscr{C}_S^\infty$ le faisceau des fonctions $C^\infty$ sur $S$. Si $f_*(\mathfrak{U}, \Omega_{X/S}^{*,\infty})$ est l'analogue $C^\infty$ de (\ref{eq:3.6.1}), construit en termes du complexe des formes différentielles relatives $C^\infty$, à valeurs complexes, sur $X$, on a
\begin{equation*}
R^n f_*(\bC) = H^n(f_*(\mathfrak{U}, \Omega_{X/S}^{*,\infty})).
\end{equation*}

Les faisceaux $\Omega_{X/S}^{p,\infty}$ étant mous, cette formule reste même valable pour tout recouvrement $\mathfrak{U}'$.

Si les $(v_i')_{0 \leq i \leq N}$ sont des relèvements $C^\infty$ de $v$ sur les $U_i$, la construction 3.6 définit encore des endomorphismes $\theta(v_i')$ de $R^n f_*(\bC)$. Si $(v_i')$ et $(v_i'')$ sont deux systèmes de relèvements, un calcul facile montre que, au niveau des complexes,
\begin{equation}\label{eq:3.7.1}
\theta(v_i') - \theta(v_i'') = dH + Hd,
\end{equation}
$H$ ayant pour coordonnées non nulles les produits contractés
\begin{equation*}
\sum_{i_0,i_1} (v_{i_1}' - v_{i_1}'') \,\llcorner\, : f_*(U_Q, \Omega_{X/S}^p) \longrightarrow \bigoplus_{i_0,i_1} f_*(U_{\{i_0,i_1\} \cup Q}, \Omega_{X/S}^{p-1})
\end{equation*}
($Q = [i_2, \ldots, i_q]$, $i_2 < \cdots < i_q$).

L'endomorphisme $\theta(v_i')$ de $R^n f_*(\bC)$ est donc indépendant du choix des $v_i'$ ; il est compatible, via l'injection de $R^n f_*(\bZ)$ dans $R^n f_*(\bC)$, à l'endomorphisme $\theta(v_i)$ de \ref{lem:3.7}. Il suffit dès lors de vérifier que $\theta(v_i') = \nabla_v$ sur $R^n f_*(\bC)$. On vérifie que $\theta(v_i')$, indépendant des $v_i'$, est aussi indépendant de $\mathfrak{U}$. Prenons $\mathfrak{U} = \{X\}$, et pour $v'$ un relèvement de $v$. On a alors
\begin{equation*}
f^*(\Omega_{X/S}^*) = f^*(\Omega_{X/S}^*)
\end{equation*}
et $\theta(v')$ est la dérivée de Lie selon $v'$, visiblement égale à $\nabla_v$.
\end{proof}

\subsection{}

Achevons la démonstration de \ref{thm:3.6}. Par construction, on a
\begin{equation}\label{eq:3.8.1}
\theta(v_i)(F^p) \subset F^p \oplus F^{p-1}.
\end{equation}
Puisque la suite spectrale (\ref{eq:3.4.1}) se déduit du complexe double $f_*(\mathfrak{U}, \Omega_{X/S}^*)$, on tire de \ref{lem:3.7} et (\ref{eq:3.8.1}) que
\begin{equation}\label{eq:3.8.2}
\theta(v_i) F^p(R^n f_*(\bC)) = \nabla_v F^p(R^n f_*(\bC)) \subset F^{p-1}(R^n f_*(\bC)).
\end{equation}
Ceci étant vrai localement sur $S$, et pour tout $v$, implique \ref{thm:3.6}.

\subsection{}

Il résulte de la formule de Leibniz
\begin{equation*}
\nabla(\omega h) = d\omega \cdot h + \omega \cdot \nabla h
\end{equation*}
que l'application $\nabla$ induite par
\begin{equation*}
\mathrm{Gr}_F^p(\nabla) : \mathrm{Gr}_F^p(R^n f_*(\bC)) \longrightarrow \Omega_S^1 \otimes \mathrm{Gr}_F^{p-1}(R^n f_*(\bC))
\end{equation*}
est $\mathscr{O}_S$-linéaire. Elle s'identifie d'après \ref{eq:3.4.1}
\begin{equation}\label{eq:3.9.1}
\mathrm{Gr}_F^p(\nabla) : R^q f_* \Omega_{X/S}^p \longrightarrow \Omega_S^1 \otimes R^{q+1} f_* \Omega_{X/S}^{p-1}.
\end{equation}

On déduit de la formule \ref{eq:3.9.1} pour $\nabla_v$ que

\begin{proposition}\label{prop:3.10}
L'homomorphisme (\ref{eq:3.9.1}) est le cup-produit, via l'accouplement (3.3.1), avec la classe de Kodaira-Spencer
\begin{equation*}
c \in H^1(S, \Omega_S^1 \otimes R^1 f_* T_{X/S})
\end{equation*}
qui exprime comment $X$ se déforme avec $s$.
\end{proposition}

\begin{definition}\label{def:3.11}
\begin{enumerate}
\item[(i)] Une \emph{famille de structures de Hodge réelles}, de poids $n$, sur $S$ consiste en
\begin{enumerate}
\item[a)] un système local de vectoriels réels $H_\bR$ sur $S$ ;
\item[b)] une filtration holomorphe finie $F$ du faisceau analytique localement libre $H_\bC = H_\bR \otimes_\bR \mathscr{O}_S$,
\end{enumerate}
ces données vérifiant les conditions suivantes :
\begin{enumerate}
\item[(H.1)] La connexion naturelle $\nabla$ de $H_\bC$ est telle que
\[
\nabla F^p(H_\bC) \subset \Omega_S^1 \otimes F^{p-1}(H_\bC).
\]
\item[(H.2)] En tout point $s \in S$, $F$ définit sur $(H_\bC)_s$ une structure de Hodge de poids $n$.
\end{enumerate}
\item[(ii)] Une \emph{polarisation} de $H$ est une forme bilinéaire localement constante sur $H_\bR$ qui induise en tout point $s \in S$ une polarisation de $(H_\bR)_s$.
\end{enumerate}
\end{definition}

\subsection{}

Une \emph{famille de structures de Hodge} $H$ de poids $n$ sur $S$ est un système local de $\bZ$-modules libres de type fini $H_\bZ$ sur $S$, plus une famille de structures de Hodge réelles sur $H_\bR = \bR \otimes_\bZ H_\bZ$. Une polarisation de $H$ est une polarisation $\Psi$ de $H_\bR$ qui soit à valeurs entières sur $H_\bZ$.

Avec ces définitions, on peut reformuler 2.2 (C) et \ref{thm:3.6} en disant que, sous les hypothèses de 3.1, $P_\bZ^n(f)$ est une famille polarisée de structures de Hodge sur $S$.

\section{Le théorème de régularité}

\subsection{}

Lorsqu'on part d'un morphisme projectif et lisse $f : X \to S$ de schémas de type fini sur $\bC$, tel que $f^{\mathrm{an}} : X^{\mathrm{an}} \to S^{\mathrm{an}}$ vérifie les hypothèses de 3.1, certains des objets construits au $n^\circ$ 3 se déduisent d'objets de nature purement algébrique par application du foncteur de ``passage à l'analytique''. Ce sont :
\begin{enumerate}
\item[a)] les faisceaux de modules $R^n f_* \Omega_{X/S}^*$, leur filtration de Hodge, la suite spectrale \ref{eq:3.4.1}, et la connexion de Gauss-Manin ;
\item[b)] la structure d'algèbre (cup-produit) sur $R^n f_* \Omega_{X/S}^*$, et la ``partie primitive'' $P_\bR^n(f)$ ;
\item[c)] le produit, par une puissance convenable de $2\pi i$, de la forme de polarisation.
\end{enumerate}

Par contre, le ``réseau entier'' $R^n f_* (\bZ)$, et la structure réelle qui s'en déduit sur la cohomologie complexe des fibres de $f$, sont de nature transcendante.

\subsection{}

Soient en effet $S$ un schéma lisse de type fini sur $\bC$ et $X$ un sous-schéma fermé de $\mathbb{P}^r(\bC) \times S$, tel que la projection $f : X \to S$ soit un morphisme lisse purement de dimension relative $d$.

D'après la variante relative de GAGA, pour tout faisceau algébrique cohérent $\mathscr{F}$ sur $X$, on a
\begin{equation*}
(R^n f_* \mathscr{F})^{\mathrm{an}} \xrightarrow{\sim} R^n f_*^{\mathrm{an}} (\mathscr{F}^{\mathrm{an}}).
\end{equation*}
Plus généralement, si $K^*$ est un complexe de faisceaux algébriques cohérents dont les différentielles $d_i$ sont des opérateurs différentiels (algébriques) $f^*\mathscr{O}_S$-linéaires, alors les images directes en hypercohomologie vérifient
\begin{equation*}
R^n f_* (K^*)^{\mathrm{an}} \xrightarrow{\sim} R^n f_*^{\mathrm{an}} (K^{*\,\mathrm{an}})
\end{equation*}
(ce cas se ramène au précédent via l'une des suites spectrales d'hypercohomologie).

Prenons pour $K$ le complexe de De Rham algébrique relatif $\Omega_{X/S}^*$. On trouve que (notation de \ref{eq:3.2.1})
\begin{equation*}
(R^n f_* \Omega_{X/S}^*)^{\mathrm{an}} \xrightarrow{\sim} R^n f_*^{\mathrm{an}} \Omega_{X^{\mathrm{an}}/S^{\mathrm{an}}}^* \xrightarrow{\sim} R^n f_*^{\mathrm{an}} (f^* \bC_{S^{\mathrm{an}}}) = R^n f_*(\bC).
\end{equation*}

De plus, la suite spectrale (\ref{eq:3.4.1}) provient d'une suite spectrale purement algébrique
\begin{equation}\label{eq:4.2.1}
E_1^{p,q} = R^q f_* \Omega_{X/S}^p \Rightarrow R^{p+q} f_* \Omega_{X/S}^*.
\end{equation}

D'après \ref{def:3.5}, les $E_1^{p,q}$ sont localement libres (puisque les $E_1^{p,q\,\mathrm{an}}$ le sont), la suite spectrale (\ref{eq:4.2.1}) dégénère et sa formation est compatible à tout changement de base. La filtration de Hodge, à laquelle elle aboutit, est ipso facto algébrique.

Le cup-produit se définit en termes du produit extérieur dans $\Omega_{X/S}^*$. La classe de cohomologie de De Rham d'une section hyperplane pouvant se définir algébriquement (à multiplication par une puissance de $2\pi i$ près), à partir de la différentielle logarithmique $df/f : \mathscr{O}^* \to \Omega_{X/S}^1$, la partie primitive $P_\bR^n(f)$ et $\Psi$ sont aussi de nature algébrique.

Enfin, la construction 3.6--\ref{lem:3.7} de la connexion de Gauss-Manin se transpose aisément au cas algébrique (Katz-Oda \cite{10}).

\subsection{}

Soient $D$ une surface de Riemann isomorphe au disque unité, $0$ un point de $D$, $D^* = D - \{0\}$, $V$ un fibré vectoriel holomorphe sur $D^*$, $\nabla$ une connexion holomorphe sur $V$ et $V^0$ le système local des sections horizontales de $\nabla$. On a $V \cong V^0 \otimes_\bC \mathscr{O}_{D^*}$. Le groupe fondamental (commutatif) $\pi_1(D^*) \cong \bZ$ agit sur $V^0$. On désigne par $T$ la transformation de monodromie, i.e. l'action sur $V^0$ ou $V$ du générateur positif de $\pi_1(D^*)$. Il existe $U$ tel que
\[
T = \exp(2\pi i U).
\]

Soient $z : D \xrightarrow{\sim} \bC$ une coordonnée, vérifiant $z(0) = 0$, et $\widetilde{D}^*$ le revêtement (universel) de $D^*$ sur lequel $\log z$ est défini. Pour $v$ une section de $V^0$ sur $\widetilde{D}^*$,
\begin{equation*}
\exp(-\log z \cdot U)(v)
\end{equation*}
est l'image réciproque d'une section holomorphe de $V$ sur $\widetilde{D}^*$.

Cette construction définit un isomorphisme
\begin{equation*}
\exp(\log z \cdot U) : \mathscr{O} \otimes_\bC H^0(\widetilde{D}^*, V^0) \xrightarrow{\sim} V,
\end{equation*}
et par là un prolongement naturel $V^U$ de $V$ sur $D$, qui ne dépend que de $U$. Une section $v$ de $V$ sur $D^*$ sera dite \emph{$\nabla$-modérée} si, en tant que section de $V^U$, elle est méromorphe en $0$. Cette notion ne dépend pas du choix de $U$.

\subsection{}

Soient $\bar{S}$ le complément dans une courbe projective et lisse $\bar{S}$ d'un ensemble fini $T$, et $V$ un fibré vectoriel algébrique sur $\bar{S}$. Une connexion $\nabla$ (algébrique) sur $V$ est dite \emph{régulière} si, pour tout $t \in T$, les sections méromorphes de $V$ au voisinage de $t$ sont $\nabla$-modérées.

\begin{theorem}\label{thm:4.5}
Pour $f : X \to S$ un morphisme de schémas projectif et lisse, la connexion de Gauss-Manin $\nabla$ sur $R^n f_* \Omega_{X/S}^*$ est régulière.
\end{theorem}

Ce théorème est prouvé dans \cite{2}. Une démonstration complètement différente, procédant par voie arithmétique, est due à Katz \cite{9}.

\subsection{}

Soient $D^*$ comme en 4.3, $H$ une famille de structures de Hodge réelles sur $D^*$, $T$ la transformation de monodromie de $H_\bC$, $U$ une transformation de $H_\bC$ telle que $T = \exp(2\pi i U)$ et $H_\bC^U$ le prolongement localement libre correspondant de $H_\bC$ sur $D$.

On dit que $H$ est \emph{régulière} en $0$ si la filtration de Hodge de $H_\bC$ se prolonge à $H_\bC^U$. Cette condition ne dépend pas de $U$. Le théorème \ref{thm:4.5} a le corollaire suivant, de conclusion purement analytique.

\begin{corollary}\label{cor:4.7}
Sous les hypothèses de \ref{thm:4.5}, les familles de structures de Hodge $R^n f_*(\bZ)$ sur $S^{\mathrm{an}}$ sont régulières au voisinage de tout point $t \in T$.
\end{corollary}

On peut espérer que, en fait, toute famille polarisée de structures de Hodge sur $D^*$ soit automatiquement régulière.

\section{Espaces modulaires}

\subsection{}

Soient $G$ un groupe algébrique réel, $C \in G(\bR)$, et $V$ une représentation réelle de $G$. Une \emph{$C$-polarisation} de $V$ est une forme bilinéaire $\Psi$ sur $V$, $G$-invariante, et telle que $\Psi(x,Cy)$ soit symétrique et définie $> 0$.

\begin{lemma}\label{lem:5.2}
Si $G$ est connexe (par quoi on entend que $G(\bC)$ l'est), les conditions suivantes sont équivalentes :
\begin{enumerate}
\item[(i)] $G$ admet une représentation $\rho : G \to \GL(V)$ de noyau $\Ker(\rho)$ fini qui soit $C$-polarisable.
\item[(ii)] Toute représentation de $G$ est $C$-polarisable.
\item[(iii)] $G$ est réductif et $\ad \circ C$ est une involution de Cartan (de $G$).
\end{enumerate}
\end{lemma}

La démonstration est laissée au lecteur.

\begin{remark}\label{rem:5.3}
Il existe dans $G(\bR)$ des éléments $C$ vérifiant les conditions de \ref{lem:5.2} si et seulement si $G$ est réductif et admet un tore maximal compact. Dans ce cas :
\begin{enumerate}
\item[a)] Un tel élément $C$ est contenu dans un seul sous-groupe compact maximal, son centralisateur.
\item[b)] Pour $G$ adjoint, la correspondance $C \leftrightarrow$ (centralisateur de $C$) est une bijection entre l'ensemble de ces éléments et l'ensemble des sous-groupes compacts maximaux de $G(\bR)$.
\end{enumerate}
\end{remark}

\subsection{}

Soit $G$ un groupe algébrique réel, muni d'homomorphismes
\begin{equation}\label{eq:5.4.1}
\mathbb{G}_m \xrightarrow{w} G \xrightarrow{t} \mathbb{G}_m
\end{equation}
tels que $w$ soit central et que $tw(x) = x^2$. Une représentation $\rho : G \to \GL(V)$ sera dite \emph{homogène de poids} $n$ si $\rho \circ w(x) = x^n$.

Par \emph{morphisme} de $\mathbb{S}$ dans $G$ on entendra toujours un homomorphisme $h : \mathbb{S} \to G$ rendant commutatif le diagramme
\[
\begin{CD}
\mathbb{G}_m @>w>> \mathbb{S} @>t>> \mathbb{G}_m \\
@| @VV{h}V @| \\
\mathbb{G}_m @>w>> G @>t>> \mathbb{G}_m
\end{CD}
\]

Si $h : \mathbb{S} \to G$ est un morphisme et $\rho : G \to \GL(V)$ une représentation réelle de $G$, on notera $\rho_h$ la structure de Hodge réelle (\ref{def:1.4}) $\rho \circ h$ de $V$. Pour $\rho$ homogène de poids $n$, une polarisation de $V$ est une polarisation $\Psi$ de $(V, \rho_h)$ qui vérifie
\begin{equation}\label{eq:5.4.2}
\Psi(gx, gy) = t(g)^n \,\Psi(x,y).
\end{equation}

On dira que $h$ est \emph{positif} si toute représentation de $G$ est polarisable.

\subsection{}

Le groupe $G$, muni de (\ref{eq:5.4.1}), est uniquement déterminé par $G^0 = \Ker(t)$, muni de l'élément central, d'ordre divisant 2, $\sigma = w(-1)$. Le groupe $G$ s'identifie en effet au quotient de $\mathbb{G}_m \times G^0$ par le sous-groupe engendré par $(-1, \sigma)$.

De ce point de vue, un morphisme $h : \mathbb{S} \to G$ s'identifie à $h^0 : \mathbb{S} \to G^0$ tel que $h^0(-1) = \sigma$. Une représentation homogène de poids $n$ de $G$ s'identifie à une représentation $\rho$ de $G^0$ telle que $\rho(\sigma) = (-1)^n$, et une polarisation de cette représentation s'identifie à une $h^0(i)$-polarisation (5.1). D'après \ref{lem:5.2}, si $G^0$ est connexe, $h$ est un morphisme positif si et seulement si $G^0$ est réductif et $\ad \circ h^0(i)$ est une involution de Cartan de $G^0$.

\subsection{}

On suppose désormais que $G^0$ est réductif connexe.

Soit $h$ un morphisme de $\mathbb{S}$ dans $G$. Pour $\rho : G \to \GL(V)$ une représentation de $G$, on désigne par $F_h$ la filtration de Hodge de $V_\bC$ déduite de $\rho_h$. Pour la représentation adjointe $\Lie(G)$,
\begin{equation}\label{eq:5.6.1}
(\Lie(G)_\bC)^{-p,-p} \Lie(G_\bC) \subset F^{-p}
\end{equation}
est l'algèbre de Lie d'un sous-groupe parabolique $P(h)$ de $G_\bC$. Pour toute représentation $\rho : G \to \GL(V)$ de $G$, $P(h)$ respecte la filtration de Hodge $F_h$ de $V_\bC$ ; si $\Ker(\rho) \cap G^0$ est central, alors $P(h)$ est exactement le sous-groupe de $G$ qui respecte cette filtration.

Enfin, on désigne par $H(h)$ le centralisateur de $h$ dans $G(\bR)$.

\subsection{}

Soit $X$ une classe de conjugaison de morphismes de $\mathbb{S}$ dans $G$. On désignera par $\check{X}$ l'ensemble des conjugués dans $G(\bC)$ de $P(h)$, pour $h \in X$.

\begin{lemma}\label{lem:5.8}
L'application $G(\bC)$-équivariante $P : h \mapsto P(h)$ identifie $X$ à un ouvert de l'espace de drapeaux $\check{X}$.
\end{lemma}

Soient $h \in X$ et $\rho : G \to \GL(V)$ une représentation homogène de $G$ telle que $\Ker(\rho) \cap G^0$ soit fini. L'ensemble $X$ s'identifie à l'ensemble des structures de Hodge sur $V$, conjuguées sous $G(\bR)$ à $\rho_h$, tandis que $\check{X}$ s'identifie à l'ensemble des filtrations sur $V_\bC$, conjuguées sous $G(\bC)$ à $F_h$. Une structure de Hodge homogène étant déterminée par sa filtration de Hodge, l'application $P$ est injective. Elle s'identifie à l'application
\[
G(\bR)/H(h) \longrightarrow G(\bC)/P(h)
\]
de différentielle à l'origine
\begin{equation*}
\Lie(G)/\Lie(H) \hookrightarrow \Lie(G_\bC)/\Lie(P(h)),
\end{equation*}
i.e. $\Lie(G)/(F^0 \cap \bar{F}^0)(\Lie(G)) \hookrightarrow F^{-1}/F^0(\Lie(G_\bC))$. Cette différentielle est bijective, d'où \ref{lem:5.8}.

On a vu en passant que
\begin{equation}\label{eq:5.8.1}
H(h) = G(\bR) \cap P(h).
\end{equation}

\subsection{}

Soient $V$ une représentation réelle de $G$, $V(\check{X}, \bR)$ le faisceau constant sur $\check{X}$ de valeur $V$, $V(\check{X}, \bC) = \bC \otimes_\bR V(\check{X}, \bR)$, $V(\check{X}, \mathscr{O}) = \mathscr{O} \otimes_\bC V(\check{X}, \bC)$ et $\nabla$ la connexion intégrable naturelle de $V(\check{X}, \mathscr{O})$. L'application $G(\bR)$-équivariante $h \mapsto \rho_h$ de $X$ dans les structures de Hodge sur $V$ se prolonge de façon unique en une application $G(\bC)$-équivariante et donc holomorphe de $\check{X}$ dans les filtrations de $V_\bC$. Celle-ci correspond à une filtration $F$ de $V(\check{X}, \mathscr{O})$, la filtration de Hodge. Le système $(V(\check{X}, \mathscr{O}), \nabla, F)$ est $G(\bC)$-équivariant. La structure réelle $V(\check{X}, \bR)$ de $V(\check{X}, \bC)$ est $G(\bR)$-équivariante. Enfin, en $h \in X$, $F$ induit sur $V_\bC$ la filtration de Hodge $F_h$.

\subsection{}

L'action de $G(\bC)$ sur $\check{X}$ définit un morphisme
\begin{equation}\label{eq:5.10.1}
\Lie(G) \otimes_\bC \mathscr{O}_{\check{X}} \longrightarrow T_{\check{X}},
\end{equation}
où $T_{\check{X}}$ est le fibré tangent. En $h \in X$, le noyau de (\ref{eq:5.10.1}) est $\Lie(P(h)) = F^0(\Lie(G_\bC))$ (5.6), de sorte que (\ref{eq:5.10.1}) se factorise par un isomorphisme $G(\bC)$-équivariant
\begin{equation}\label{eq:5.10.2}
F^{-1}/F^0(\Lie(G) \otimes \mathscr{O}_{\check{X}}) \xrightarrow{\sim} T_X.
\end{equation}

On désigne par $T_X^I$ le sous-fibré holomorphe de $T_X$ image de $F^{-1}(\Lie(G) \otimes \mathscr{O}_X)$.

On vérifie facilement que si $v$ est une section locale de $T_X^I$, alors
\begin{equation*}
\nabla_v(F^p(V(\check{X}, \mathscr{O}))) \subset F^{p-1}(V(\check{X}, \mathscr{O})),
\end{equation*}
quelle que soit la représentation $\rho : G \to \GL(V)$. La réciproque est vraie pour $\Ker(\rho) \cap G^0$ central.

\subsection{}

Supposons maintenant que $X$ soit une classe de conjugaison de morphismes positifs de $\mathbb{S}$ dans $G$. Pour $h \in X$, on posera $C_h = h(i)$ et on désignera par $K(h)$ le centralisateur de $C_h$ dans $G(\bR)$. Le groupe $K^0(h) = K(h) \cap G^0(\bR)$ est compact.

Soient $V$ une représentation de $G$ et $\Psi$ une forme bilinéaire $G$-invariante sur $V$. Si $\Psi$ est une polarisation de $(V, \rho_h)$ pour un $h \in X$, alors $\Psi$ est une polarisation de $(V, \rho_h)$ pour tout $h \in X$. En effet, posant $\Psi_h(x,y) = \Psi(x, C_h y)$, on a
\begin{equation}\label{eq:5.11.1}
\Psi_{gh} = g(\Psi_h).
\end{equation}

On dira que $\Psi$ est alors une polarisation de $V$. Par hypothèse, toute représentation de $G$ est polarisable.

\subsection{}

Soit $\Psi$ une polarisation de la représentation adjointe de $G$ sur $\Lie(G)$. D'après 5.10.2, $\Psi$ induit en chaque point $h \in X$ une forme hermitienne définie positive sur
\begin{equation}\label{eq:5.12.1}
(T_X)_h = F^{-1}/F^0(\Lie(G_\bC)_h) = \bigoplus_{p < 0} \Lie(G_\bC)^{p,-p}.
\end{equation}

La polarisation $\Psi$ définit donc une structure hermitienne, invariante sous $G(\bR)$, sur $X$.

Si $Z$ est le centre de $G$, on a déjà
\begin{equation}\label{eq:5.12.2}
F^{-1}/F^0(\Lie(G/Z) \otimes \mathscr{O}_X) \xrightarrow{\sim} T_X
\end{equation}
et une polarisation $\Psi$ de $\Lie(G/Z)$ définit déjà une structure de variété hermitienne sur $X$. On peut prendre pour $\Psi$ l'opposé de la forme de Killing de $G/Z$.

Posons
\begin{align*}
(T_X^h)_h &= \bigoplus_{p < 0} (\Lie(G/Z))^{2p,-2p}, \\
i\,v^h &= \bigoplus_{p < 0} (\Lie(G/Z))^{2p-1,-2p+1}
\end{align*}
du fibré tangent en deux sous-fibrés complexes. On a
\begin{equation}\label{eq:5.12.4}
[T_X^h, T_X^h] \subset T_X^h \qquad (h \in X).
\end{equation}

\subsection{}

Le sous-groupe $K(h)_\bC \cap P(h)$ de $K(h)_\bC$ est parabolique, de sorte que $K(h)_\bC \subset K(h) \cdot P(h)$ et que $K(h) \cdot h = K(h)_\bC \cdot h$ est une sous-variété analytique complexe compacte de $X$. Ces variétés sont les fibres de la fibration $C^\infty$ $G(\bR)$-équivariante
\begin{equation*}
X = G/H(h) \longrightarrow G/K(h).
\end{equation*}

On vérifie facilement que $T_X^v$ est le fibré tangent à cette fibration.

\subsection{}

Rappelons que sur un fibré hermitien holomorphe $F$, il existe une et une seule connexion hermitienne $\nabla$ telle que $\nabla^{0,1} = \bar{\partial}$. La courbure $R$ du fibré est par définition la courbure de cette connexion. C'est une forme de type $(1,1)$ à valeur dans l'algèbre de Lie du groupe unitaire du fibré.

Si $F$ est le fibré constant défini par un vectoriel $F_0$, et si la forme hermitienne s'écrit
\begin{equation}\label{eq:5.14.1}
\Psi(x,y) = \Psi_0(hx,y),
\end{equation}
alors, pour $x : S \to F_0$ une section $C^\infty$ de $F$, on a
\begin{equation}\label{eq:5.14.2}
\nabla_{T_X} = \partial x + h^{-1}\partial' h \cdot x.
\end{equation}
On en tire que, pour $u$ et $v$ deux champs tangents,
\begin{equation}\label{eq:5.14.3}
R(u,v) = \partial_u(h^{-1}\bar{\partial}_v h) - \partial_v(h^{-1}\bar{\partial}_u h) - h^{-1}\bar{\partial}_{[u,v]}h + [h^{-1}\bar{\partial}_u h, h^{-1}\bar{\partial}_v h].
\end{equation}

\subsection{}

Soit $F$ le fibré tangent d'une variété analytique complexe hermitienne. Pour $u$ un vecteur tangent à $S$, on désignera par $u_{1,0}$ et $u_{0,1} = \overline{u_{1,0}}$ ses composantes de type $(1,0)$ et $(0,1)$ dans la complexifiée du fibré tangent. On appelle \emph{courbure holomorphe} de $S$ dans la direction $u$ le nombre réel
\begin{equation}\label{eq:5.15.1}
R(u) = \frac{\langle R(u_{1,0}, u_{0,1})(u), u \rangle}{\langle u, u \rangle^2}.
\end{equation}

Si on munit $X$ de la structure hermitienne 5.12, on a le résultat fondamental (\cite{8}, § 9) :

\begin{theorem}\label{thm:5.16}
Il existe $k < 0$, telle que $R(u) \leq k$ dans toute direction appartenant à $T_X^h$.
\end{theorem}

\begin{proof}[Preuve (une esquisse)]
Par homogénéité, il suffit de prouver \ref{thm:5.16} en un point $e \in X$. Posons $H = S(e)$, $P = P(e)$, soit $N^+$ le radical unipotent de $P$ et soit $N^-$ le sous-groupe complexe conjugué d'algèbre de Lie
\[
n^- = \bigoplus_{p < 0} \Lie(G)^{p,-p}.
\]

Dans les calculs qui suivent, on identifiera, près de $e$, $X$ à $N^-$ par les applications
\begin{equation*}
X = G/H \hookrightarrow G(\bC)/P \xleftarrow{\sim} N^-.
\end{equation*}

Les translations à gauche sur $N^-$ permettent d'identifier le fibré tangent à $n^-$.

On désignera par $t_+$, $t^0$ et $t_-$ les projections orthogonales de $\Lie(G)_\bC$ sur $\Lie(H_\bC)$ et $n^+ = \Lie(N^+)$. Pour $g = np$ ($g \in G(\bC)$, $n \in N^-$, $p \in P$), la différentielle, en $e$, de l'action de $g$ sur $X$ s'identifie à
\begin{equation*}
t_- \circ \ad p : n^- \longrightarrow n^-.
\end{equation*}

La structure hermitienne s'écrit donc
\begin{equation*}
\Psi_n(x,y) = \Psi_e(t_- \circ \ad p^{-1} \cdot x,\, t_- \circ \ad p^{-1} \cdot y).
\end{equation*}

Désignant par $\Psi$ la forme de polarisation et par $C$ l'involution de Cartan associée à $e$, on a
\begin{align*}
\Psi_n(x,y) &= \Psi(t_- \circ \ad p^{-1} x,\, C \cdot t_- \circ \ad p^{-1} y) \\
&= \Psi(t_- \circ \ad p^{-1} x,\, \ad C(p)^{-1} \cdot C y) \\
&= \Psi(\ad C(p) \cdot t_- \circ \ad p^{-1} \cdot x,\, y).
\end{align*}

On vérifie que pour $n = \exp(u)$ avec $u \neq 0$, on a au $3^{\text{e}}$ ordre près
\[
p = \exp(u - \tfrac{1}{2} t^0[u,u] - t_+[u,u])
\]
et $h = \Psi_e \circ t_- \circ \ad p^{-1} = \exp(H)$ pour
\[
H = t_- \circ \ad(Cu - u + [u,u]) - \tfrac{1}{2}[t_- \circ \ad Cu,\, t_- \circ \ad u].
\]

Sous les hypothèses de 5.14, si $h = \exp(H)$ et si $H = 0$ en $s_0 \in S$, on déduit de (\ref{eq:5.14.3}) que, en $s_0$,
\begin{multline*}
R(u,v) = (\partial_u \bar{\partial}_v - \partial_v \bar{\partial}_u)H - \partial_{[u,v]}H - [\partial_u H, \bar{\partial}_v H] \\
+ \tfrac{1}{2}[\bar{\partial}_v H, \partial_u H] + (\partial_v \bar{\partial}_u - \partial_u \bar{\partial}_v) \cdots
\end{multline*}

Appliquant cette formule, on voit que la courbure est donnée, en $e$, par
\begin{equation*}
R(u,v)e : x \longmapsto [S(u,v), x]
\end{equation*}
avec $S(u,v) = -t_- \circ \ad[u,v] + \tfrac{1}{2}[t_- \circ \ad Cu,\, t_- \circ \ad v] + \tfrac{1}{2}[t_- \circ \ad u,\, t_- \circ \ad Cv] = [t_- \circ \ad Cu,\, t_- \circ \ad v] - t_- \circ \ad[u,v]$.

Quels que soient $u$, $v \in n^-$, si $u_{1,0}$ et $u_{0,1}$ sont les composantes de type $(1,0)$ et $(0,1)$ de $u$ comme en 5.15, on a grâce à l'invariance de $\Psi$,
\begin{align*}
\Psi(R(u_{1,0}, u_{0,1})(v), v) &= \Psi([Cu, t_-[\bar{u}, v]] - t_-[Cu, [\bar{u}, v]] - t_-[[u, \bar{u}], v], v) \\
&= \Psi(t_-[\bar{u}, v], [\bar{u}, Cv]) + \Psi([Cu, v], [\bar{u}, Cv]) - \Psi([u, \bar{u}], [v, Cv]).
\end{align*}

Pour $u = v \in T_X^h$, on a $Cu = -u$ et
\[
\Psi(R(u_{1,0}, u_{0,1})(u), u) = \Psi([u, \bar{u}], [u, \bar{u}]) + \Psi([u, \bar{u}], [v, \bar{v}]) = -\Psi(t_-[u, \bar{u}], t_-[u, \bar{u}]) - \Psi([u, \bar{u}], [u, \bar{u}]) < 0,
\]
ce qui prouve \ref{thm:5.16}.
\end{proof}

Une généralisation du lemme de Schwarz permet de déduire de \ref{thm:5.16} (voir \cite{1}, III, $n^\circ$ 10) :

\begin{corollary}\label{cor:5.17}
Soit $f : D \to X$ une application holomorphe du disque unité dans $X$. Munissons $D$ de la métrique hyperbolique $d$ pour laquelle il est de courbure $-1$, et $X$ de la métrique 5.12. Alors, si $f(D)$ est tangent aux $T_X^h$, on a
\begin{equation*}
d(f(x), f(y)) \leq |x|\, d(x,y).
\end{equation*}
\end{corollary}

\section{Le théorème de monodromie, d'après A. Borel}

\subsection{}

Soient $S$ une variété analytique complexe connexe, munie d'un point base $s_0 \in S$, et $p : \widetilde{S} \to S$ son revêtement universel. Soit $H$ une famille polarisée de structures de Hodge de poids $n$ sur $S$. On désignera par $H_0$ la structure de Hodge polarisée fibre de $H$ en $s_0$.

La polarisation $\Psi$ est une forme bilinéaire sur $H_0 \otimes \bQ$. On désignera par $G^0$ le groupe algébrique, défini sur $\bQ$, composante neutre du groupe des automorphismes de $(H_0 \otimes \bQ, \Psi)$, par $\sigma \in G^0$ l'homothétie de rapport $(-1)^n$, par $G$ le groupe déduit, par le procédé 5.5 de $(G^0, \sigma)$ et par $\Gamma$ le groupe discret des automorphismes de $(H_0 \otimes \bQ, \Psi)$. On dispose de
\begin{equation}\label{eq:6.1.1}
T : \pi_1(S, s_0) \longrightarrow \Gamma.
\end{equation}

\subsection{}

Pour chaque point $s \in \widetilde{S}$, on dispose d'un isomorphisme naturel $H_0 \otimes_\bZ \bQ \xrightarrow{\sim} H_{p(s)} \otimes \bQ$ et la structure de Hodge $H_{p(s)}$ s'identifie à une structure de Hodge sur $H_0 \otimes \bZ \bQ$, définie par $h(s) : \mathbb{S} \to G_\bR$. Les morphismes $h(s)$ sont positifs, et conjugués entre eux, car ils forment une famille continue connexe et que $G$ est réductif. Si $X$ est l'espace modulaire correspondant, défini au § 5, on dispose donc d'une application
\begin{equation*}
h : \widetilde{S} \longrightarrow X.
\end{equation*}

La famille $H$ est l'image réciproque par $h$ du système de structures de Hodge sur $X$ défini (5.9) par la représentation de poids $n$ naturelle de $G$ dans $H_0 \otimes \bQ$.

Le morphisme $h$ est équivariant, via (5.5.1), pour les actions de $\pi_1(S, s_0)$ et $\Gamma$ sur $\widetilde{S}$ et $X$.

D'après \ref{eq:3.4.1} et \ref{thm:3.6}, l'application $h$ est holomorphe et $\im(dh) \subset T_X^I$ (cf. 5.10).

\subsection{}

Supposons que $S$ soit le disque unité épointé
\[
D^* = \{z \mid 0 < |z| < 1\}
\]
et prenons pour revêtement universel de $D^*$ le demi-plan de Poincaré $Y$, muni de
\begin{equation*}
\exp(2\pi i y) : Y \longrightarrow D^*.
\end{equation*}

Si $T \in \Gamma$ est la transformation de monodromie (4.3) de $H_\bZ$, l'application $h$ vérifie
\begin{equation}\label{eq:6.3.1}
h(y + 1) = Th(y).
\end{equation}

Choisissons $h_0 \in X$. Si $h(y) = g_y \cdot h_0$, la distance dans $X$, pour une métrique (5.12), vérifie
\begin{equation}\label{eq:6.3.2}
d(h(y+1), h(y)) = d(Tg_y h_0, g_y h_0) = d(g_y^{-1} T g_y h_0, h_0).
\end{equation}

D'après \ref{cor:5.17}, si $Y$ est muni de sa distance hyperbolique naturelle, l'application $h$ décroit les distances (convenablement normalisées). Dans $Y$, on a
\begin{equation*}
d(y, y+1) \sim (\mathrm{Im}(y))^{-1}
\end{equation*}
et donc
\begin{equation}\label{eq:6.3.3}
d(g_y^{-1} T g_y h_0, h_0) \to 0 \quad \text{pour } \mathrm{Im}(y) \to \infty.
\end{equation}

Le même argument s'applique à $T^2$ ; de plus, $T^2 \in G(\bR)$, et (\ref{eq:6.3.2}) implique qu'une limite de conjugués de $T^2$ se trouve dans le sous-groupe compact $H(h_0)$ de $G(\bR)$. Les valeurs propres de $T^2$ (et de $T$) sont donc toutes de valeur absolue 1.

Ces valeurs propres forment un ensemble d'entiers algébriques stable par conjugaison.

Or,

\begin{lemma}\label{lem:6.4}
Si tous les conjugués complexes d'un entier algébrique sont de valeur absolue 1, alors c'est une racine de l'unité.
\end{lemma}

Les valeurs propres de $T$ sont donc des racines de l'unité, ce qui prouve

\begin{theorem}[Théorème de monodromie 6.5 (A. Borel)]\label{thm:6.5}
Soit $H$ une famille de structures de Hodge polarisée sur le disque épointé $D^*$. Il existe des entiers $N$ et $M$ tels que la transformation de monodromie $T$ vérifie
\[
(T^N - I)^M = 0.
\]
\end{theorem}
