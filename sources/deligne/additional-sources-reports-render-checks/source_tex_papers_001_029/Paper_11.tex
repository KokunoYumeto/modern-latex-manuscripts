\begin{center}
\textbf{UNE CONGRUENCE ENTRE COEFFICIENTS MULTINOMIAUX}

\smallskip
\small par P.~DELIGNE

\smallskip
\footnotesize\textit{Institut des Hautes \'Etudes Scientifiques, Bures sur Yvette}
\end{center}

Soit $p$ un nombre premier. Pour tout entier $m \neq 0$, on d\'esigne par $\vord(m)$
l'exposant de la plus haute puissance de $p$ qui divise $m$. On pose de plus
$\vord(0) = \infty$.

Pour $\mathfrak{n} = (n_1, \dots, n_k)$ un $k$-uple d'entiers $\geqslant 0$, on pose
\[
|\mathfrak{n}| = \sum_1^k n_i, \qquad
a\mathfrak{n} = (an_1, \dots, an_k), \qquad
X^{\mathfrak{n}} = \prod_1^k X_i^{n_i}, \qquad
\mathfrak{n}! = \prod_1^k n_i!
\]
et on d\'esigne par $C(\mathfrak{n})$ le coefficient du mon\^ome $X^{\mathfrak{n}}$ dans
\[
\Bigl(\sum_1^k X_i\Bigr)^{|\mathfrak{n}|}
\]
(coefficients multin\^omiaux).

On a donc
\begin{equation}
\Bigl(\sum_1^k X_i\Bigr)^n = \sum_{|\mathfrak{n}|=n} C(\mathfrak{n})\, X^{\mathfrak{n}}
\end{equation}
\begin{equation}
C(\mathfrak{n}) = \frac{|\mathfrak{n}|!}{\mathfrak{n}!}.\tag{2}
\end{equation}

Pour $k = 2$, ce sont l\`a les coefficients bin\^omiaux.
\begin{equation}
C(n_1, n_2) = \binom{n_1 + n_2}{n_1}.\tag{3}
\end{equation}

\begin{prop}
Soient $\mathfrak{n}$ un $k$-uple d'entiers $\geqslant 0$, et $n = |\mathfrak{n}|$. Si $p$ est
un nombre premier $\geqslant 5$, on a
\begin{equation}
C(p^l \mathfrak{n}) \equiv C(\mathfrak{n}) \pmod{p^{\vord(n)+3}}.\tag{4}
\end{equation}
\end{prop}

\noindent\textsc{Exemple.} --- Pour $p = 5$, $k = 2$, $l = 1$ et $\mathfrak{n} = (1,1)$, on a
\[
\binom{10}{5} = 252 = \binom{2}{1} + 2 \cdot 5^3.
\]

\medskip
\noindent\textsc{Preuve.} --- On sait, ou on d\'eduit de (2) que
\begin{equation}
\Bigl(\sum_1^k X_i\Bigr)^{p^l} = \Bigl(\sum_1^k X_i^{p^l}\Bigr) + pQ\tag{5}
\end{equation}
o\`u $Q$ est un polyn\^ome \`a coefficients entiers, de degr\'e $< p^l$ en chaque
variable. La formule du bin\^ome fournit alors
\begin{equation}
\Bigl(\sum_1^k X_i\Bigr)^{p^l n} = \Bigl(\sum_1^k X_i^{p^l}\Bigr)^{\!n} + \sum_1^n p^i \binom{n}{i} R_i\tag{6}
\end{equation}
\begin{equation}
R_i = \Bigl(\sum_1^k X_i^{p^l}\Bigr)^{\!n-i} \cdot Q^i.\tag{7}
\end{equation}

Pour $A$ et $B$ deux polyn\^omes \`a coefficients entiers, on \'ecrira $A \equiv B \pmod{r}$
si les coefficients de $X^{p^l \mathfrak{n}}$ dans $A$ et $B$ sont congrus modulo $r$.
Avec cette notation, (4) se r\'e\'ecrit
\begin{equation}
\Bigl(\sum_1^k X_i\Bigr)^{p^l n} \equiv \Bigl(\sum_1^k X_i^{p^l}\Bigr)^{\!n} \pmod{p^{\vord(n)+3}}.\tag{8}
\end{equation}

\begin{lem}
\begin{equation}
\vord\!\left(\binom{n}{i}\right) \geqslant \vord(n) - \vord(i).\tag{9}
\end{equation}
\end{lem}

Faisons agir le groupe additif $\mathbb{Z}/(n)$ sur lui-m\^eme par translation\,\footnote{A tout
$x \in \mathbb{Z}/(n)$, est donc associ\'ee la permutation de $\mathbb{Z}/(n)$ qui applique $y$ sur
$x + y$.};
si une partie \`a $i$ \'el\'ements $P$ de $\mathbb{Z}/(n)$ a pour stabilisateur\,\footnote{Si un
groupe $G$ op\`ere dans un ensemble $E$, le stabilisateur d'une partie $P$ de $E$ est
l'ensemble des $g \in G$ pour lesquels $P$ est stable.} un sous-groupe
$H$ de $\mathbb{Z}/(n)$ ($H \cdot P = P$), alors $P$ est r\'eunion de classes lat\'erales de $H$, et
$\#H \mid i$. Le cardinal $r_P = n/\#H$ de l'orbite de $P$ sous $\mathbb{Z}/(n)$, dans
$\mathscr{P}(\mathbb{Z}/(n))$, v\'erifie donc
\[
p^{\vord(n)-\vord(i)} \mid r_P,
\]
d'o\`u, puisque l'ensemble des parties \`a $i$ \'el\'ements de $\mathbb{Z}/(n)$ est somme disjointe
d'orbites,
\[
p^{\vord(n)-\vord(i)} \mid \binom{n}{i}.
\]

\begin{lem}
Pour $p \neq 2, 3$ et $i \geqslant 3$, on a
\begin{equation}
p^i \binom{n}{i} \equiv 0 \pmod{p^{\vord(n)+3}}.\tag{10}
\end{equation}
\end{lem}

On a
\[
\vord\!\left(p^i \binom{n}{i}\right) = i + \vord\!\left(\binom{n}{i}\right) \geqslant \vord(n) + (i - \vord(i)) \quad \text{(lemme 1)}.
\]

Pour $3 \leqslant i < p$, on a $\vord(i) = 0$ et $i - \vord(i) \geqslant 3$. Pour $i \geqslant p$, on a
$i - \vord(i) \geqslant i - \log i/\log p$. Cette derni\`ere fonction de $i$ est croissante pour
$i \geqslant p$, et vaut $p - 1 \geqslant 3$ pour $i = p$.

\begin{lem}
Le coefficient de $X^{p^l \mathfrak{n}}$ dans $R_1$ est nul.
\end{lem}

Aucun des mon\^omes $X^{\mathfrak{r}}$ apparaissant dans $Q$ n'a un exposant $\mathfrak{r}$ divisible
par $p^l$. Le polyn\^ome
\[
R_1 = \Bigl(\sum_1^k X_i^{p^l}\Bigr)^{\!n-1} \cdot Q
\]
h\'erite de cette propri\'et\'e.

Il r\'esulte des lemmes 2 et 3 que
\begin{equation}
\Bigl(\sum_1^k X_i\Bigr)^{p^l n} \equiv \Bigl(\sum_1^k X_i^{p^l}\Bigr)^{\!n} + \binom{n}{2}\, p^2 R_2 \pmod{p^{\vord(n)+3}}.\tag{11}
\end{equation}

De cette congruence r\'esulte d\'ej\`a (d'apr\`es le lemme 1) que
\begin{equation}
C(p^l \mathfrak{n}) \equiv C(\mathfrak{n}) \pmod{p^{\vord(n)+2}}.\tag{12}
\end{equation}

Les seuls mon\^omes $X^{\mathfrak{r}}$ d'exposant divisible par $p^l$ qui figurent dans $(pQ)^2$
sont les mon\^omes
\[
X_i^{p^l} \cdot X_j^{p^l} \quad (i < j).
\]
Le coefficient de chacun d'eux est
\begin{equation}
\sum_{i \neq 0, p^l} \binom{p^l}{i} \binom{p^l}{p^l-i} = \sum \binom{p^l}{i}^{\!2} - 2 = \binom{2p^l}{p^l} - 2\tag{13}
\end{equation}
et
\[
\binom{2p^l}{p^l} - 2 \equiv \binom{2p}{p} - 2 \pmod{p^3}
\]
d'apr\`es (12).

On a donc
\[
\Bigl(\sum_1^k X_i\Bigr)^{p^l n} \equiv \Bigl(\sum_1^k X_i^{p^l}\Bigr)^{\!n} + \binom{n}{2} \biggl[\binom{2p}{p} - 2\biggr] \cdot R_2 \pmod{p^{\vord(n)+3}}
\]
avec $R_2$ polyn\^ome \`a coefficients entiers.

Puisque
\[
\vord\!\left(\binom{n}{2}\right) \geqslant \vord(n) \quad \text{(lemme 1)},
\]
la proposition r\'esulte du

\begin{lem}
On a, pour $p$ premier $\geqslant 5$:
\[
\binom{2p}{p} \equiv 2 \pmod{p^3}.
\]
\end{lem}

On a
\[
p^{-1} \binom{p}{i} \equiv (-1)^{i-1} \cdot i^{-1} \pmod{p}
\]
pour $1 \leqslant i \leqslant p-1$ et donc
\begin{equation}
p^{-2}\biggl[\binom{2p}{p} - 2\biggr] = \sum_1^{p-1} p^{-2} \binom{p}{i}^{\!2} \equiv \sum_1^{p-1} (1/i)^2 \pmod{p}.\tag{14}
\end{equation}

D\'efinissons
\[
z \in \mathbb{Z}/(p) \quad \text{par} \quad z = \sum_{i \in \mathbb{Z}/(p)^*} 1/i^2.
\]
Pour
\[
u \in \mathbb{Z}/(p)^*,
\]
on a
\[
z = \sum_{i \in \mathbb{Z}/(p)^*} 1/(u^{-1}i)^2 = u^2 z
\]
et
\[
(u^2 - 1)z = 0.
\]

Puisque $p \neq 2, 3$, il existe $u$ tel que $u^2 \neq 1$, et $z = 0$.

La formule (14) se simplifie donc en
\[
p^{-2}\biggl[\binom{2p}{p} - 2\biggr] \equiv 0 \pmod{p},
\]
ce qui ach\`eve la d\'emonstration.
