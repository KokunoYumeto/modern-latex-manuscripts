\section*{16. $P^1$ moins trois points: un quotient du $\pi_1$ motivique (suite)}

\subsection*{16.11}

Pour tout point base $b$ de $X$, l'inclusion de $X$ dans $\Gm$ induit
$\pi_1(X,b)^\mot \to \pi_1(\Gm,b)^\mot = \ZZ(1)$.
Notons $U(b)$ le quotient suivant de $\pi_1(X,b)^\mot$ (muni de la structure
enti\`ere quotient):
\begin{equation}\label{eq:16.11.0}
U(b) := \pi_1(X,b)^\mot / \Ker(\pi_1(X,b)^\mot \to \ZZ(1))',
\end{equation}
o\`u $'$ indique le passage au groupe d\'eriv\'e. Soit $U_1(b)$ le groupe
$\Ker(\pi_1(X,b)^\mot \to \ZZ(1))^{\mathrm{ab}}$. On a
\begin{equation}\label{eq:16.11.1}
1 \to U_1(b) \to U(b) \to \ZZ(1) \to 1.
\end{equation}
Par abus de notation, on notera $U_1(b)(N)$ l'image de $U_1(b)$ dans
$U(b)(N)$: on a
\begin{equation}\label{eq:16.11.2}
1 \to U_1(b)(N) \to U(b)(N) \to \ZZ(1) \to 1.
\end{equation}
L'action par automorphismes int\'erieurs de $U(b)$ sur $U_1(b)$ se factorise
par $\ZZ(1)$. Si $b'$ est un autre point base, $U(b')$ se d\'eduit de $U(b)$
en tordant par le $U(b)$-torseur des $U(b)$-chemins de $b$ \`a $b'$, et
$U_1(b')$ se d\'eduit donc de $U_1(b)$ en tordant par le $\ZZ(1)$-torseur
image: le torseur des chemins, dans $\Gm$, de $b$ \`a $b'$:
\begin{equation}\label{eq:16.11.3}
U_1(b') = U_1(b) \text{ tordu par } P_{b',b} \text{ calcul\'e dans } \Gm.
\end{equation}

Lorsque le point base $b$ est un vecteur tangent en $0$, on dispose de la
monodromie en $0$: $\mu_0 : \ZZ(1) \to U(b)$ qui scinde
\eqref{eq:16.11.1}. Lorsque $b$ est un vecteur tangent en $1$, on dispose de
$\mu_1 : \ZZ(1) \to U_1(b)$.

Prenons pour point base le vecteur tangent $1$ en $0$ et \'ecrivons
simplement $U$ et $U_1$ pour $U(b)$ et $U_1(b)$. On a
\begin{equation}\label{eq:16.11.4}
\mu_0 : \ZZ(1) \to U : U = \ZZ(1) \ltimes U_1.
\end{equation}

Pour $b'$ un vecteur tangent en $1$, le torseur $P_{b',(0,1)}$, calcul\'e
dans $\Gm$, est trivial. Par \eqref{eq:16.11.3}, on a $U_1(b') = U_1$,
d'o\`u
\begin{equation}\label{eq:16.11.5}
\mu_1 : \ZZ(1) \to U_1.
\end{equation}

Une reformulation de cette d\'efinition de $\mu_1$ nous sera utile.
Appliquant 13.15 \`a \eqref{eq:16.11.1}, on trouve que pour $t$ un vecteur
tangent en $1$, le torseur des $U_1$-chemins de $(0,1)$ \`a $t$ est d\'efini.
Notons le $Q$. Le morphisme
\begin{equation}\label{eq:16.11.6}
Q \times \ZZ(1) \to U_1 : (q,\alpha) \mapsto q^{-1}\mu_t(\alpha)q
\end{equation}
envoie $Q_\ZZ \times \ZZ(1)_\Betti$ dans $U_{1\ZZ}$. Parce que $U_1$ est
commutatif, $q^{-1}\mu_t(\alpha)q$ est ind\'ependant de $q$ et
\eqref{eq:16.11.6} se factorise par un morphisme de $\ZZ(1)$ dans $U_1$:
c'est le morphisme $\mu_t$ de \eqref{eq:16.11.5}.

\subsection*{16.12}

Sur le produit $\prod_1^N \QQ(n)$, $\QQ(1)$ agit (action d'alg\`ebre de Lie
ab\'elienne) par les $\QQ(1) \otimes \QQ(n) \to \QQ(n+1)$:
\[
(a,b) \longmapsto ab
\]
avec $a \in \QQ(1)$, $b \in \QQ(n)$, $ab \in \QQ(n+1)$. Exponentiant cette
action nilpotente, on obtient une action du groupe motivique $\QQ(1)$ sur
le produit des $\QQ(n)$ ($1 \leq n \leq N$). On peut passer \`a la limite
projective sur $N$. Nous aurons \`a consid\'erer le groupe produit
semi-direct d\'efini par cette action
\begin{equation}\label{eq:16.12.1}
\QQ(1) \ltimes \prod_{n \geq 1} \QQ(n)
\end{equation}
et les morphismes $\mu_0$ et $\mu_1$ suivant de $\QQ(1)$ dans ce produit
semi-direct: $\mu_0$ est l'isomorphisme identique avec le premier facteur,
et $\mu_1$ l'inclusion de $\QQ(1)$ dans le produit des $\QQ(n)$.

L'alg\`ebre de Lie de \eqref{eq:16.12.1} est le produit semi-direct
analogue, pour l'action de Lie de $\QQ(1)$ sur le produit des $\QQ(n)$.

\begin{proposition}[16.13]\label{prop:16.13}
Avec les notations de \ref{eq:16.11.1}, le groupe $U \otimes \QQ$ muni de
$\mu_0$ et $\mu_1$ est isomorphe \`a \eqref{eq:16.12.1}, muni de $\mu_0$ et
$\mu_1$. L'isomorphisme est unique.
\end{proposition}

\noindent \textit{Mise en garde.} 16.13 d\'ecrit le groupe motivique
$U \otimes \QQ$, mais non sa structure enti\`ere. Pour celle-ci, voir 16.16.

\noindent \textsc{Preuve.} Il suffit de prouver l'analogue de 16.13 pour
les alg\`ebres de Lie de $U$ et du produit semi-direct
\eqref{eq:16.12.1}. Soient encore $\mu_0$ et $\mu_1$ les morphismes
d\'efinis par $\mu_0$ et $\mu_1$ de $\QQ(1)$ dans $\Lie U$. Soit
$\nu_n := (\ad\mu_0)^{n-1}(\mu_1) : \QQ(n) \to \Lie U$ ($n \geq 1$). Le
morphisme $\mu_0$ et les $\nu_n$ d\'efinissent pour chaque $N$ un morphisme
d'alg\`ebres de Lie
\begin{equation}\label{eq:16.13.1}
\QQ(1) \ltimes \prod_1^N \QQ(n) \to \Lie U^{(N)}.
\end{equation}

En r\'ealisation de de Rham, $\Lie U_{\dR}$ est engendr\'e par
$e_0 := \mu_0(1)$, $e_1 := \mu_1(1)$ et est le plus grand quotient de
$\mathrm{Lib}(e_0,e_1)^{(N)} = \Lie \pi_1(X,(0,1))_{\dR}^{(N)}$ dans lequel
l'id\'eal engendr\'e par $e_1$ soit commutatif. Ce quotient a pour base
$e_0$ et les $(\ad e_0)^{n-1}(e_1)$ ($1 \leq n \leq N$), et les
$(\ad e_0)^{n-1}(e_1)$ commutent entre eux. Cette description montre que
\eqref{eq:16.13.1} est un isomorphisme en r\'ealisation de de Rham.
C'est donc un isomorphisme. C'est d'ailleurs l'unique isomorphisme qui
respecte $\mu_0$ et $\mu_1$. Passant \`a la limite sur $N$, on obtient
16.13. \hfill $\bullet$

\subsection*{16.14}

Pour le groupe fondamental classique $\pi_1(X(\CC),b)$, les construction de
16.11 ont l'analogue suivant. L'inclusion de $X(\CC)$ dans
$\Gm(\CC) = \CC^*$ induit
$\pi_1(X(\CC),b) \to \pi_1(\Gm(\CC),b) = \ZZ(1)_\Betti$ et on d\'efinit
\begin{align*}
\Gamma(b) &:= \pi_1(X(\CC),b)/\Ker(\pi_1(X(\CC),b) \to \ZZ(1)_\Betti)', \\
\Gamma_1(b) &:= \Ker(\pi_1(X(\CC),b) \to \ZZ(1)_\Betti)^{\mathrm{ab}}.
\end{align*}
Prenons $b = (0,1)$ et \'ecrivons $\Gamma$ et $\Gamma_1$ pour $\Gamma(b)$ et
$\Gamma_1(b)$. On dispose encore de $\mu_0, \mu_1 : \ZZ(1)_\Betti \to \Gamma$.
Le g\'en\'erateur $2\pi i$ de $\ZZ(1)_\Betti$ a pour images $a_0$ et $a_1$ par
$\mu_0$ et $\mu_1$ respectivement la monodromie autour de $0$, et le
conjugu\'e de la monodromie autour de $1$ par le chemin r\'eel (16.10) de
$(0,1)$ \`a $(1,-1)$. Le groupe $\Gamma$ est engendr\'e par $a_0$ et $a_1$
avec pour relation que les conjugu\'es de $a_1$ commutent. Ces conjugu\'es
engendrent $\Gamma_1$.

Faisons agir $t \in \ZZ(1)_\Betti$ sur $\Gamma_1$ par
$\gamma \mapsto \mu_0(t)\gamma\mu_0(t)^{-1}$. Cette action fait de $\Gamma_1$
une repr\'esentation de $\ZZ(1)_\Betti$, i.e.\ un module sous l'alg\`ebre de
groupe $\ZZ[\ZZ(1)_\Betti]$. La propri\'et\'e universelle de $\Gamma$ assure
que $\Gamma_1$ est le $\ZZ[\ZZ(1)_\Betti]$-module libre de base $a_1$ et que
$\Gamma$ est le produit semi-direct $\ZZ(1)_\Betti \ltimes \Gamma_1$.

Se rappelant que $\mu_1$ identifie $\ZZ(1)_\Betti$ \`a $a_1^\ZZ$, on a plus
intrins\`equement
\begin{align}
\Gamma_1 &= \ZZ(1)_\Betti \otimes_\ZZ \ZZ[\ZZ(1)_\Betti] \label{eq:16.14.1} \\
\Gamma &= \ZZ(1)_\Betti \ltimes \Gamma_1 \qquad
\text{(action de $\ZZ(1)_\Betti$: par multiplication dans
$\ZZ[\ZZ(1)_\Betti]$).} \label{eq:16.14.2}
\end{align}

\subsection*{16.15}

L'alg\`ebre de groupe $\ZZ[\ZZ(1)_\Betti]$ est encore l'alg\`ebre affine du
tore sur $\Spec(\ZZ)$ de groupe de caract\`eres $\ZZ(1)_\Betti$, not\'e
$\Hom(\ZZ(1)_\Betti, \Gm)$. Le dual de l'alg\`ebre de Lie de ce tore
($1$-formes invariantes par translations) s'identifie au groupe des
caract\`eres $\ZZ(1)_\Betti$ par
$\chi \mapsto d\log\chi := \chi^{-1}d\chi$. Ceci permet de r\'e\'ecrire
\eqref{eq:16.14.1} comme
\begin{equation}\label{eq:16.15.1}
\Gamma_1 = \ZZ(1)_\Betti \otimes_\ZZ \text{(alg\`ebre enveloppante du
tore).}
\end{equation}
L'action d'un \'el\'ement de $\ZZ(1)_\Betti$, identifi\'e \`a un caract\`ere
$\chi$, est la multiplication par $\chi$.

Choisissons un g\'en\'erateur $u$ de $\ZZ(1)_\Betti$ (par exemple
$u = 2\pi i$). Le choix de $u$ identifie $\ZZ[\ZZ(1)_\Betti]$ \`a
$\ZZ[u,u^{-1}]$ et $\Gamma_1$ \`a
$\ZZ[u,u^{-1}] \cdot \frac{du}{u}$. L'action de $u \in \ZZ(1)_\Betti$ est la
multiplication par $u$. Dans ce mod\`ele, la s\'erie centrale descendante
de $\Gamma$ induit sur $\Gamma_1$
\begin{equation}\label{eq:16.15.2}
\Gamma_1^{(N)} = (u-1)^{N-1} \cdot \ZZ[u,u^{-1}] \cdot \frac{du}{u}
\qquad (N \geq 1)
\end{equation}
et le compl\'et\'e correspondant de $\Gamma_1$ est le compl\'et\'e formel le
long de $1 \in \Gm$.

\subsection*{16.16}

Les constructions 16.14, 16.15 vont nous permettre de d\'eterminer la
structure enti\`ere des groupes motiviques $U$ et $U_1$.

L'isomorphisme 16.13 induit un isomorphisme
\begin{equation}\label{eq:16.16.1}
U_1 \otimes \QQ \xrightarrow{\sim} \prod_{n \geq 1} \QQ(n),
\end{equation}
limite projective des
\begin{equation}\label{eq:16.16.2}
U_1^{(N)} \otimes \QQ \xrightarrow{\sim} \prod_1^N \QQ(n).
\end{equation}

Par abus de notation, notons $\Gamma_1^{(N)}$ l'image de $\Gamma_1$ dans
$\Gamma^{(N)}$. Par \eqref{eq:16.15.2}, c'est un groupe ab\'elien sans
torsion. C'est la structure enti\`ere de $U_1^{(N)}$. Par
\eqref{eq:16.15.2}, une fois choisi un g\'en\'erateur $u$ de
$\ZZ(1)_\Betti$, $\Gamma_1^{(N)}$ s'identifie \`a
$\ZZ[u,u^{-1}]/((u-1)^N) \cdot \frac{du}{u}$. Calculons
\begin{equation}\label{eq:16.16.3}
\text{\eqref{eq:16.16.2} : } \Gamma_1^{(N)} \otimes \QQ \xrightarrow{\sim}
\prod_1^N \QQ(n)_\Betti
\end{equation}
d\'efini par \eqref{eq:16.16.2}. Par d\'efinition, $\frac{du}{u}$ s'envoie dans
$u \in \QQ(1)_\Betti$ et \`a la multiplication par $\log u$ correspond
l'action de Lie 16.12 de $u \in \QQ(1)_\Betti$: on a
\[
\frac{du}{u} \otimes (\log u)^{n-1} \longmapsto u \otimes n \in \QQ(n)_\Betti.
\]
Posons $u = e^v$. Si $f(u) = \sum a_n v^{n-1}$, l'image de
$f(u)\frac{du}{u}$ est donc $\sum a_n u \otimes n$. Pour $v$ un
 g\'en\'erateur de $\QQ(1)_\Betti$, la structure enti\`ere de
$\prod_1^N \QQ(n)$, d\'eduite de celle de $U_1^{(N)}$ par
\eqref{eq:16.16.2}, est donc la suivante: on identifie
$\prod_1^N \QQ(n)_\Betti$ \`a $\QQ[v]/(v^{N-1}) \cdot dv$ par
$v^i dv \mapsto v \otimes (i+1)$, et on prend les combinaisons lin\'eaires
\`a coefficients entiers des $\exp(kv)dv$ ($k \in \ZZ$).

Cette description fournit la

\begin{proposition}[16.17]\label{prop:16.17}
Le morphisme
\[
(k-1)! \pr_k \circ \text{\eqref{eq:16.16.1}} : U_1^{(N)} \otimes \QQ \to \QQ(k)
\]
envoie $U_{1\ZZ}^{(N)} = \Gamma_{1\ZZ}^{(N)}$ dans $\ZZ(k)_\Betti$, donc
d\'efinit un morphisme
\begin{equation}\label{eq:16.17.1}
U_1 \to \ZZ(k).
\end{equation}
\end{proposition}

La proposition 16.8 permet d'appliquer 13.15 \`a
\[
1 \to U_1 \to U \to \ZZ(1) \to 1
\]
et de d\'efinir:

\begin{definition}[16.18]\label{def:16.18}
\begin{enumerate}
\item[(i)] On note $P_1$ le $U_1$-torseur des $U_1$-chemins de
$(0 \text{ vers } 1)$ \`a $(1 \text{ vers } 0)$.
\item[(ii)] On note $P_{1,k}$ le $\ZZ(k)$-torseur d\'eduit de $P_1$ en
poussant par \eqref{eq:16.17.1}.
\end{enumerate}
\end{definition}

\begin{lemma}[16.19]\label{lem:16.19}
Le $\ZZ(1)$-torseur $P_{1,1}$ est trivial.
\end{lemma}

\noindent \textsc{Preuve.} Par 16.8, le $U$-torseur des chemins de
$(0 \text{ vers } 1)$ \`a $(1 \text{ vers } 0)$ admet une r\'eduction du groupe
structural non seulement \`a $U_1$, mais m\'eme \`a
$\Ker(U \to H^1(X))$. Ce groupe est le noyau dans $U_1$ de
\eqref{eq:16.17.1} pour $k=1$ et 16.19 en r\'esulte.

\begin{theorem}[16.20]\label{thm:16.20}
Pour chaque $k \geq 2$, le $\ZZ(k)$-torseur $P_{1,k}$ est isomorphe au
$\ZZ(k)$-torseur d\'ecrit en 3.1, 3.2.
\end{theorem}

La preuve sera donn\'ee au paragraphe 19.

\subsection*{16.21}

Soient $d \geq 2$ et $\mu_d^*$ le sch\'ema des racines $d$-i\`emes de $1$
autres que $1$:
\[
\mu_d^* = \Spec\bigl(\ZZ[\zeta]/((\zeta^d-1)/(\zeta-1))\bigr) \subset X.
\]
Le torseur des $U$-chemins de $(0 \text{ vers } 1)$ \`a
$\zeta \in \mu_d^*$, pouss\'e par $U \to \ZZ(1) = \pi_1(\Gm)$, est le
$\ZZ(1)$-torseur des chemins, sur $\Gm$, de $(0 \text{ vers } 1)$ \`a
$\zeta \in \mu_d^*$. Par (15.51.1), c'est le torseur de Kummer $K(q)$ sur
$\Gm$, restreint \`a $\mu_d^*$. D'apr\`es 14.3, il est de $d$-torsion,
trivialis\'e par le changement de structure enti\`ere
$\ZZ(1) \to \frac{1}{d}\ZZ(1)$.

D\'efinissons de nouvelles structures enti\`eres:
\begin{enumerate}
\item[(a)] Pour tout point base $b$, $U_1(b)(d)$ est $U_1(b) \otimes \QQ$,
muni de la plus petite structure enti\`ere contenant celle de $U_1(b)$
et stable sous $\frac{1}{d}\ZZ(1)_\Betti \supset \ZZ(1)_\Betti$. L'action
est comme en 16.11. Pour $b = (0,1)$, on \'ecrit $U_1(d)$.
\item[(b)] Pour $b = (0,1)$, $U(d)$ est $U \otimes \QQ$, muni de la plus
petite structure enti\`ere contenant celle de $U$ et l'image par
$\mu_0$ de $\frac{1}{d}\ZZ(1)$: on a
\begin{equation}\label{eq:16.21.1}
U(d) = \frac{1}{d}\ZZ(1) \ltimes U_1(d).
\end{equation}
\end{enumerate}

Comme en 16.16, choisissons un g\'en\'erateur $v$ de $\ZZ(1)_\Betti$ et
identifions $U_1 \otimes \QQ^{(N)} = \prod_1^N \QQ(n)_\Betti$ \`a
$\QQ[v]/(v^{N-1}) \cdot dv$. La structure enti\`ere
$U_1(d)_\ZZ^{(N)}$ est le $\ZZ$-module engendr\'e par les
$\exp(kv/d)dv$ ($k \in \ZZ$). On a donc

\begin{lemma}[16.22]\label{lem:16.22}
Le morphisme
\begin{equation}\label{eq:16.22.1}
d^{k-1}(k-1)! \pr_k \circ \text{\eqref{eq:16.16.1}} : U_1 \otimes \QQ \to \QQ(k)
\end{equation}
envoie $U_1(d)_\ZZ$ dans $\ZZ(k)_\Betti$: c'est un morphisme
\begin{equation}\label{eq:16.22.2}
U_1(d) \to \ZZ(k).
\end{equation}
Pour $d=1$, on retrouve \eqref{eq:16.17.1}.
\end{lemma}

\subsection*{16.23}

D'apr\`es 16.21, on peut appliquer 13.15 \`a
\[
1 \to U_1(d) \to U(d) \to \frac{1}{d}\ZZ(1) \to 1
\]
et d\'efinir le $U_1(d)$-torseur des $U_1(d)$-chemin de
$(0 \text{ vers } 1)$ \`a $\zeta \in \mu_d^*$. C'est un $U_1(d)$-torseur
sur $\mu_d^*[1/d]$. On le prolonge \`a
$\mu_d[1/d] = \mu_d^*[1/d] \amalg \{1\}$ en prenant sur $1$ le
$U_1(d)$-torseur des $U_1(d)$-chemins de $(0 \text{ vers } 1)$ \`a
$(1 \text{ vers } 0)$.

\noindent \textsc{Notation.}
\begin{enumerate}
\item[(i)] On note $P_d$ ce torseur.
\item[(ii)] On note $P_{d,k}$ le $\ZZ(k)$-torseur sur $\mu_d[1/d]$
d\'eduit de $P_d$ en poussant par \eqref{eq:16.22.2}.
\end{enumerate}
En $1$, $P_{d,k}$ est donc $d^{k-1}$ fois le torseur $P_{1,k}$.

\begin{theorem}[16.24]\label{thm:16.24}
Les torseurs $P_{d,k}$ ci-dessus sont isomorphes \`a ceux de 3.11.
\end{theorem}

La preuve sera donn\'ee au paragraphe 19.

\begin{proposition}[16.25]\label{prop:16.25}
La restriction de $P_{dd',k}$ \`a $\mu_d^*[1/dd']$ (resp.
$\mu_d[1/dd']$ si $k \geq 2$) est $(d')^{k-1}$ fois le torseur
$P_{d,k}$.
\end{proposition}

\noindent \textsc{Preuve.} Ceci refl\`ete le facteur $d^{k-1}$ dans
\eqref{eq:16.22.1}.

\begin{proposition}[16.26]\label{prop:16.26}
Pour $k=1$, le $\ZZ(1)$-torseur $P_{d,k}$ est, sur $\mu_d^*$, le
torseur de Kummer $K(1-\zeta)$.
\end{proposition}

\noindent \textsc{Preuve.} L'espace $P^1 \setminus \{1,\infty\}$ est
isomorphe \`a $\Gm = P^1 \setminus \{0,\infty\}$ par $x \mapsto 1-x$.
Son $\pi_1$ est donc $\ZZ(1)$. Pour un point base vecteur tangent en
$1$, l'isomorphisme du $\pi_1$ avec $\ZZ(1)$ est donn\'e par la
monodromie autour de $1$ (16.1.2). L'inclusion de $X$ dans
$P^1 - \{1,\infty\}$ induit donc un morphisme de $\pi_1(X,(0,1))$
dans $\ZZ(1)$. Ce morphisme se factorise par
$\pi_1^{\mathrm{ab}} = \ZZ(1) \times \ZZ(1)$ (par $p_0, p_1$), et
m\^eme par $\frac{1}{d}\ZZ(1) \times \ZZ(1)$. Il se factorise donc par
$U(d)$, et le diagramme
\[
\begin{CD}
\pi_1(X,(0,1)) @>>> U(d) @>>> U_1(d) \\
@VVV @. @VV{(\ref{eq:16.22.2})_{d,1}}V \\
\ZZ(1) @= \ZZ(1) @= \ZZ(1)
\end{CD}
\]
est commutatif. Le torseur $P_{d,1}$ est donc d\'eduit du torseur des
chemins de $(0,1)$ \`a $\zeta \in \mu_d^*$ en poussant par
$\pi_1(x \mapsto 1-x) : \pi_1(X,(0,1)) \to \pi_1(\Gm,1) = \ZZ(1)$. Par
fonctorialit\'e du torseur fondamental et 14.2(ii), le torseur obtenu
est $K(1-\zeta)$.

\subsection*{16.27}

Explicitons ce que 8.17 affirme sur $\pi_1(X,(0,1))^\mot$. Posons
$\pi := \pi_1(X,(0,1))^\mot$. On n\'eglige la structure enti\`ere. Soit
$G_\pi$ le groupe fondamental de la cat\'egorie tannakienne $\mathcal{T}$
engendr\'ee par les syst\`emes de r\'ealisations $\Lie \pi^{(N)}$.
Rappelons que $G_\pi$ est un $\mathcal{T}$-sch\'ema en groupes, que, pour
chaque foncteur r\'ealisation $\omega$, $\omega(G_\pi)$ est le sch\'ema en
groupes des automorphismes du foncteur fibre $\omega$ de $\mathcal{T}$ et que
les actions de $\omega(G_\pi)$ sur les $\omega(\pi)$ sont induites par une
action de $G_\pi$ sur $\pi$. Cette action est fid\`ele et nous
identifierons $G_\pi$ \`a un sch\'ema en groupes d'automorphismes de
$\pi$.

Par 6.17(iv), $G_\pi$ est une extension de $\Gm$ par un groupe unipotent
$U_\pi$. Comme en 8.9, on dispose en r\'ealisation de de Rham de
$\mathcal{T}$: $\Gm \to G_{\pi\dR}$ qui fait de $G_{\pi\dR}$ le produit
semi-direct de $\Gm$ et de $U_{\pi\dR}$.

La pro-alg\`ebre de Lie $\Lie \pi_\dR$ est librement engendr\'ee par
$e_0$, $e_1$, $e_\infty$ soumis \`a $e_0+e_1+e_\infty = 0$. La graduation
pour laquelle les $e_i$ sont de degr\'e $1$ donne l'action de $\Gm$.
Puisque $U_{\pi\dR}$ est normalis\'e par $r(\Gm)$, la pro-alg\`ebre de
Lie de $U_{\pi\dR}$ est gradu\'ee: c'est le produit de ses composantes
homog\`enes $[\Lie U_{\pi\dR}]_n$ ($n > 0$), et
$[\Lie U_{\pi\dR}]_n$ est form\'e de d\'erivations de degr\'e $n$. Comme
en 8.10, il existe $a_0 \in U_{\pi\dR}(\RR)$ tel que
$a_0 \cdot r(2\pi i)$ transforme la structure rationnelle $\pi_\dR$ de
$\pi_\dR \otimes \CC \cong \pi_\Betti \otimes \CC$ en la structure
rationnelle $\pi_\Betti$.

Soit $V_\dR$ le sch\'ema en groupes pro-unipotent des automorphismes de
$\Lie \pi_\dR$ qui respectent la filtration $W$ et qui induisent
l'identit\'e sur le gradu\'e. Le groupe $U_{\pi\dR}$ est le plus petit
sous-groupe $X$ normalis\'e par $r(\Gm)$ de $V_\dR$, d\'efini sur $\QQ$,
ayant les propri\'et\'es suivantes.
\begin{enumerate}
\item[(i)] Il existe $b_0 \in X(\RR)$ tel que $b := b_0 \cdot r(2\pi i)$
transforme $\pi_\dR$ en $\pi_\Betti$.
\item[(ii)] Soient $H_\dR$ le produit semi-direct $\Gm \ltimes X$, et
$H_\Betti$ la $\QQ$-structure de $H_\dR \otimes \CC$ de points rationnels
$\mathrm{int}(b)(H_\dR(\QQ))$. Il agit sur $\pi_\Betti$. On veut que
l'action de $\Gal(\overline{\QQ}/\QQ)$ sur $\pi_\ell$ se factorise par
$H_\Betti(\QQ_\ell)$.
\item[(iii)] Pour tout $p$, on veut que le Frobenius cristallin
$\phi_p$ soit dans $H_\dR(\QQ_p)$. \hfill $\bullet$
\end{enumerate}

Le Frobenius r\'eel $F_\infty$ (un automorphisme de $\pi_\Betti$) est,
avec les notations de (i)
\[
F_\infty = b_0 \cdot r(-1) \cdot b_0^{-1} \cdot r(-1)
\]
et si $X$ v\'erifie (i), il contient $F_\infty \cdot r(-1)$. La conjecture
8.17 est que $X$ est, sur $\RR$, le plus petit sous-groupe $Y$ de
$V_\dR \otimes \RR$, d\'efini sur $\RR$, normalis\'e par $r(\Gm)$ et
contenant $F_\infty \cdot r(-1)$. Avec les notations de 8.10,
$F_\infty \cdot r(-1) = \exp(2N)$ et l'alg\`ebre de Lie de $Y$ est
engendr\'ee par les composantes homog\`enes de $N$.

\begin{remark}[16.28]
Il existe un plus petit sous-groupe $X \subset V_\dR$ v\'erifiant la
condition (i) de 16.27. On peut le d\'eduire du lemme suivant,
appliqu\'e aux $\Lie \pi_\dR^{(N)}$.

\begin{lemma}[16.28.1]\label{lem:16.28.1}
Soient $V_1$ et $V_2$ deux espaces vectoriels sur $\QQ$, munis d'une
filtration $W$ et $a : V_1 \otimes \CC \xrightarrow{\sim} V_2 \otimes \CC$
un isomorphisme filtr\'e, avec $\Gr^W(a)$ d\'efini sur $\QQ$. Soit
$H \subset \GL(V_1)$ le sous-groupe des $g$ respectant $W$ et tels que
$\Gr^W(g) = 1$. Il existe un plus petit sous-groupe $X$ d\'efini sur
$\QQ$ de $H$ pour lequel il existe $b \in X(\CC)$ qui transporte $V_1$
en $a^{-1}(V_2) \subset V_1 \otimes \CC$.
\end{lemma}

\noindent \textsc{Preuve.} Soit $P$ le $H$-torseur des isomorphismes
filtr\'es $V_1 \to V_2$ de m\^eme gradu\'e que $a$. La condition sur $X$
peut encore s'\'ecrire: l'image de $a$ dans $P/X$ est d\'efinie sur
$\QQ$. Puisque $P/(X_1 \cap X_2) \hookrightarrow P/X_1 \times P/X_2$,
l'intersection de deux groupes la v\'erifiant la v\'erifie encore, et
16.28.1 en r\'esulte.
\end{remark}

\subsection*{16.29}

Le groupe $G_\pi$ respecte toutes les structures motiviques de $\pi$.
Voici des exemples.

Le morphisme $\mu_0 : \QQ(1) \to \pi$ est respect\'e par $G_\pi$. Ceci
revient \`a dire que $U_{\pi\dR}$ fixe $e_0$.

Soit $P$ le $\pi$-torseur des chemins de $(0 \text{ vers } 1)$ \`a
$(1 \text{ vers } 0)$ (resp.\ $(\infty \text{ vers } 0)$). Transportant la
monodromie autour de $1$ (resp.\ $\infty$) par un tel chemin, on
d\'efinit un morphisme de sch\'emas motiviques
\[
\mu : \QQ(1) \to \pi.
\]
La classe de conjugaison de $\mu$ est donc stable par $G_\pi$. Ceci
revient \`a dire que tout $u \in \Lie U_{\pi\dR}$ transforme $e_1$
(resp.\ $e_\infty$) en un crochet $[X_1, e_1]$ (resp.\ $[X_\infty,
e_\infty]$). Si $u$ est de degr\'e $n$, $X_1$ et $X_\infty$ sont de
degr\'e $n$, donc uniquement d\'etermin\'es par $u$ si $n \geq 2$, puisque
le centralisateur de $e_1$ (resp.\ $e_\infty$) est r\'eduit \`a
$\QQ \cdot e_1$ (resp.\ $\QQ \cdot e_\infty$). La relation
$e_0+e_1+e_\infty=0$ impose par ailleurs
\[
X_\infty = -X_1.
\]

Par 16.8 (compl\'et\'e dans le cas de $\infty$ par 14.3, qui fournit un
chemin de $(0 \text{ vers } 1)$ \`a $(0 \text{ vers } \infty)$), le
$\pi$-torseur $P$ admet une r\'eduction $P'$ de son groupe structural au
groupe d\'eriv\'e $\pi'$. Le torseur $P'$ s'injecte dans la classe de
conjugaison de $\mu_1$ (resp.\ $\mu_\infty$) et
\[
P' \to \Hom(\QQ(1), \Lie \pi)
\]
est motivique. Il en r\'esulte que $\Lie U_{\pi\dR}$ est de degr\'e
$\geq 2$. Une d\'erivation int\'erieure de degr\'e $\geq 2$ de
$\Lie \pi_\dR$ qui fixe $e_0$ est nulle car le centralisateur de $e_0$
est r\'eduit \`a $\QQ \cdot e_0$. Le groupe $G_\pi$ s'injecte donc dans
celui des automorphismes ext\'erieurs de $\pi_\dR$ (automorphismes modulo
automorphismes int\'erieurs).

L'action de $\mathfrak{S}(3)$ sur $P^1$ qui permute $0,1,\infty$ induit
une action ext\'erieure de $\mathfrak{S}(3)$ sur $\pi$. Elle est
motivique, donc $G_\pi$ la centralise. Cela signifie que pour $u$ de
degr\'e $n \geq 2$ dans $\Lie U_{\pi\dR}$, si on pose
\begin{equation}\label{eq:16.29.1}
\nu(e_i) = [X_i, e_i] \qquad (i=0,1,\infty,\ X_i \text{ polyn\^ome de
Lie de degr\'e } n,\ X_0 = 0),
\end{equation}
alors pour $\sigma$ une permutation de $\{0,1,\infty\}$, on a
\[
\nu(\sigma e_i) = [X_{\sigma(i)}, \sigma e_i]
\]
independant de $i$. Ceci signifie
\begin{align*}
X_\infty(e_0,e_1,e_\infty) &= X_1(e_0,e_\infty,e_1) \\
X_1(e_0,e_1,e_\infty) &= -X_1(e_1,e_0,e_\infty) \\
X_1(e_0,e_1,e_\infty) &+ X_1(e_1,e_\infty,e_0) +
X_1(e_\infty,e_0,e_1) = 0.
\end{align*}

On dispose encore d'autres structures motiviques sur $\pi$, mais je ne
sais pas montrer qu'elles imposent \`a $G_\pi$ d'\^etre aussi petit que
conjectur\'e par 8.17.

\subsection*{16.30}

Notons $1$ l'image dans $\pi_\dR(\RR)$ du chemin r\'eel de $(0,1)$ \`a
$(1,-1)$ (16.10). L'image du $\pi_1$ transcendant dans $\pi_\dR(\CC)$
est engendr\'ee par $l^{-1}\exp(2\pi i e_0)l$ et
$l^{-1}\exp(2\pi i e_1)l$. Soit $b_0'$ l'automorphisme de $\Lie \pi_\dR$
qui envoie $e_0$ sur $e_0$ et $e_1$ sur $\ad l^{-1}(e_1)$.
L'automorphisme $b' := b_0' \cdot r(2\pi i)$ conjugue $\pi_\dR$ en
$\pi_\Betti$. La conjecture 16.27 affirme que si on remplace $b'$ par
$b' \cdot c$, avec $c$ un automorphisme convenable de $\Lie \pi_\dR$,
d\'efini sur $\QQ$, et centralisant $r(-1)$, alors le $b_0$ correspondant
a un logarithme dont les composantes homog\`enes sont dans l'alg\`ebre
de Lie de d\'erivations engendr\'ee par les composantes homog\`enes du
logarithme de $F_\infty \cdot r(-1) = b_0 \cdot r(-1) \cdot b_0^{-1} \cdot
r(-1)$.

C'est l\`a une propri\'et\'e de rationalit\'e de $1$. On notera que les
coordonn\'ees de $1$ sont les int\'egrales it\'er\'ees de Chen, de $0$
\`a $1$, convenablement r\'egularis\'ees, de suites de $\frac{dx}{x}$ et
$\frac{dx}{x-1}$.

\begin{proposition}[16.31]\label{prop:16.31}
L'action de $U_\pi$ sur $\pi/\pi''$ est ab\'elienne.
\end{proposition}

Il suffit de le v\'erifier en r\'ealisation de de Rham, o\`u on applique
le

\begin{lemma}[16.32]\label{lem:16.32}
Soit $\mathcal{C}$ l'alg\`ebre de Lie libre engendr\'ee par $e_0$ et $e_1$.
L'alg\`ebre de Lie $A$ des d\'erivations $D$ de $\mathcal{C}/\mathcal{C}''$
telles que $De_0 = 0$ et $De_1 \in [\mathcal{C}',e_1]$, est une alg\`ebre
de Lie ab\'elienne.
\end{lemma}

\noindent \textsc{Preuve.} L'alg\`ebre de Lie ab\'elienne
$\mathcal{C}/\mathcal{C}'$ agit sur $\mathcal{C}'/\mathcal{C}''$, d'o\`u
une application
\[
\alpha : \QQ[x,y] \to \mathcal{C}'/\mathcal{C}'' : P \mapsto P(\ad
e_0, \ad e_1)[e_0,e_1].
\]
La propri\'et\'e universelle de $\mathcal{C}$ assure que cette application
est un isomorphisme. Elle munit $\mathcal{C}'/\mathcal{C}''$ d'une structure
de $\QQ[x,y]$-module, libre de rang un, traduisant l'action de
$\mathcal{C}/\mathcal{C}'$. Pour $D$ une d\'erivation de
$\mathcal{C}/\mathcal{C}''$, triviale sur $\mathcal{C}/\mathcal{C}'$, la
d\'erivation induite de $\mathcal{C}'/\mathcal{C}''$ est
$\QQ[x,y]$-lin\'eaire.

Soit $D \in A$ la d\'erivation pour laquelle $De_0 = 0$ et
$De_1 = [\alpha(P), e_1] = \alpha(-yP)$. On a
\[
D[e_0,e_1] = [e_0, [\alpha(P),e_1]] = [\alpha(xP), e_1] =
\alpha(-xyP).
\]
Puisque $P \mapsto xyP$ est injectif, $A$ s'injecte dans l'alg\`ebre de
Lie ab\'elienne des endomorphismes $\QQ[x,y]$-lin\'eaires de
$\mathcal{C}'/\mathcal{C}''$ et 16.32 en r\'esulte.

\subsection*{16.33}

La proposition 16.31 rend $\pi/\pi''$ justiciable des conjectures plus
pr\'ecises 8.18. Les m\'ethodes de [A2] devraient permettre de
v\'erifier 8.18 pour $\pi/\pi''$. \hfill $\bullet$


\section{Relations de distribution: voie geometrique}\label{sec:17}

Dans ce paragraphe, nous montrons par voie geometrique que les
torseurs $P_{d,k}$ de 16.23 verifient les relations de distribution 3.13.

\subsection{17.1}

L'assertion (3.13.1)(ii) a deja ete prouvee en 16.25. Il reste a
prouver (3.13.1)(i). Si $k=1$, $P_{d,1}$ est le torseur de Kummer
$K(1-\zeta)$ (16.26) et l'assertion resulte de (3.1.3). Nous
supposerons $k \geq 2$; cela simplifie les notations, mais la meme
preuve marcherait encore pour $k=1$, en remplacant $\mu_d^*$ par
$\mu_d$ et $P_{dd'}^*$ par $\mu_{dd'}^* - \mu_{d'}^*$.

Notons $\Tr$ la trace, rel.\ le morphisme fini etale
$z \mapsto z^{d'}$ de $\mu_{dd'}^*[1/dd']$ sur $\mu_d^*[1/dd']$. Il
s'agit de definir un isomorphisme de $\ZZ(k)$-torseurs sur
$\mu_d^*[1/dd']$
\begin{equation}\label{eq:17.1.1}
\Tr P_{dd',k} \xrightarrow{\sim} (d')^{k-1} P_{d,k}.
\end{equation}
Par descente galoisienne sur les motifs, il suffit le faire localement
pour la topologie etale: on definira \eqref{eq:17.1.1} sur un corps
contenant les racines $(dd')$-iemes de l'unite, mais sans en choisir
une.

Il s'agit, pour $\zeta$ une racine $d$-ieme de $1$, de definir
\begin{equation}\label{eq:17.1.2}
\bigotimes_{\zeta'^{d'}=\zeta} (P_{dd',k})_{\zeta'}
\xrightarrow{\sim} (d')^{k-1}(P_{d,k})_\zeta.
\end{equation}

\subsection{17.2}

Soit $R'$ un revetement de $\Gm$, isomorphe au revetement
$z \mapsto z^{d'} : \Gm \to \Gm$. On pourrait prendre $R'$ egal a ce
revetement, mais les arguments seront plus intrinseques si on ne le
suppose pas. Soit $S$ l'image inverse de $1$ et $R = R' - S$ le
revetement de $X = \Gm - \{1\}$ induit par $R'$.

Le revetement $R'$ de $\Gm$ se prolonge en un revetement ramifie en
$0$ de $\GA \supset \Gm$, isomorphe a $z \mapsto z^{d'} : \GA \to \GA$.
On note $\tilde{0}$ le point au-dessus de $0 \in \GA$. Parce que
$R' \cong \Gm$, son espace tangent epointe en $\tilde{0}$ lui est
canoniquement isomorphe. Pour $s \in R'$, on note $t(s)$ le vecteur
tangent en $\tilde{0}$ correspondant. On dispose, sur $R'$, d'un chemin
motivique de $t(s)$ a $s$ (cf.\ 15.51).

Fixons dans $R$ un point base $b$. Posons les notations suivantes:
\begin{enumerate}
\item[(a)] L'inclusion $R \subset R'$ definit
$\pi_1(R',b)^\mot \to \pi_1(R',b)^\mot = \ZZ(1)$, et
\begin{align}
V(b) &:= \pi_1(R',b)^\mot/\Ker(\pi_1(R',b)^\mot \to \ZZ(1))', \notag \\
V_1(b) &:= \Ker(\pi_1(R,b)^\mot \to \ZZ(1))^{\mathrm{ab}}: \notag \\
1 &\to V_1(b) \to V(b) \to \ZZ(1) \to 1. \label{eq:17.2.1}
\end{align}
\item[(b)] Pour chaque $s \in S$, l'inclusion $R' - \{s\} \subset R'$
definit de meme $\pi_1(R'-\{s\},b) \to \ZZ(1)$ et
\begin{align}
U_s(b) &:= \pi_1(R'-\{s\},b)^\mot/\Ker(\pi_1(R'-\{s\},b)^\mot \to \ZZ(1))', \notag \\
U_s^1(b) &:= \Ker(\pi_1(R'-\{s\},b)^\mot \to \ZZ(1))^{\mathrm{ab}}: \notag \\
0 &\to U_s^1(b) \to U_s(b) \to \ZZ(1) \to 0. \label{eq:17.2.2}
\end{align}
\end{enumerate}
Les inclusions $R \subset R' - \{s\} \subset R'$ definissent des
diagrammes commutatifs
\begin{equation}\label{eq:17.2.3}
\begin{CD}
0 @>>> V_1(b) @>>> V(b) @>>> \ZZ(1) @>>> 0 \\
@. @VVV @VVV @| \\
0 @>>> U_s^1(b) @>>> U_s(b) @>>> \ZZ(1) @>>> 0.
\end{CD}
\end{equation}
De \eqref{eq:17.2.3}, on deduit
\begin{align}
V_1(b) &\to \prod_S U_s^1(b) \label{eq:17.2.4} \\
V(b) &\to \text{produit fibre sur } \ZZ(1) \text{ des } U_s(b). \label{eq:17.2.5}
\end{align}

\begin{proposition}[17.3]\label{prop:17.3}
Les morphismes \eqref{eq:17.2.4}, \eqref{eq:17.2.5} sont des
isomorphismes.
\end{proposition}

\noindent \textsc{Preuve.} Il suffit de verifier que 17.2.5 induit
pour chaque $N$ un isomorphisme
\begin{equation}\label{eq:17.3.1}
V(b)^{(N)} \xrightarrow{\sim} \text{produit fibre des } U_s^{(N)}(b)
\text{ sur } \ZZ(1).
\end{equation}
Il suffit de le verifier en realisation de Betti. Il suffit meme de le
verifier pour les groupes discrets qui definissent les structures
entieres. Soient
$\Gamma(b) := \Ker(\pi_1(R(\CC),b) \to \ZZ(1)_\Betti)^{\mathrm{ab}}$ et
$\Gamma_s(b) := \Ker(\pi_1(R'(\CC)-s,b) \to
\ZZ(1)_\Betti)^{\mathrm{ab}}$. Ces groupes sont munis d'une action de
$\ZZ(1)_\Betti$, induite par l'action par conjugaison de $\pi_1$ sur
lui-meme.

A gauche de \eqref{eq:17.3.1}, $V(b)_\Betti^{(N)}$ est une extension
de $\ZZ(1)_\Betti$ par le plus grand quotient sans torsion de
$\Gamma(b)$ sur lequel l'action de $\ZZ(1)_\Betti$ soit nilpotente
d'echelon $N$. A droite, on a une extension de $\ZZ(1)_\Betti$ par le
produit des plus grands quotients sans torsion des $\Gamma_s(b)$ sur
lesquels l'action de $\ZZ(1)_\Betti$ soit nilpotente d'echelon $N$.

Il reste a verifier que
\begin{equation}\label{eq:17.3.2}
\Gamma(b) \xrightarrow{\sim} \prod_S \Gamma_s(b).
\end{equation}
Le groupe fondamental $\pi_1(R',b)$ est un groupe libre, et on peut
choisir un systeme generateur libre $(e_0, (e_s)_{s \in S})$ ou $e_0$
(resp.\ $e_s$) est la monodromie autour de $0$ (resp.\ $s$), ramenee
dans $\pi_1(R',b)$ par un chemin de $b$ a un vecteur tangent en $0$
(resp.\ $s$). Pour un tel choix, la projection
$\pi_1(R',b) \to \pi_1(R-\{s\},b)$ annule les $e_t$ ($t \in S$,
$t \neq s$) et envoie $e_0$ et $e_s$ sur un systeme generateur libre.
Le groupe commutatif $\Gamma(b)$ (resp.\ $\Gamma_s(b)$) est un module
sous l'algebre du groupe $\ZZ(1)_\Betti$. C'est un module libre, de
base les images de $e_s$ ($s \in S$) (resp.\ de base l'image de $e_s$)
et \eqref{eq:17.3.2} en resulte.

\subsection{17.4}

Prenons pour point base $b$ un vecteur tangent en $\tilde{0}$. La
monodromie en $\tilde{0}$: $\mu_{\tilde{0}} : \ZZ(1) \to V(b)$ scinde
les extensions 17.2.3:
\begin{align*}
V(b) &= \ZZ(1) \ltimes V_1(b) \\
U_s(b) &= \ZZ(1) \ltimes U_s^1(b).
\end{align*}

Soit $a$ un entier et definissons $V(b)(a)$ comme etant
$V(b) \otimes \QQ$, muni de la plus petite structure entiere contenant
celle de $V(b)$ et l'image de
$\mu_{\tilde{0}}(\frac{1}{a}\ZZ(1)_\Betti)$. Soit
$V_1(b)(a) := V_1(b) \otimes \QQ$, muni de la structure entiere
induite: la plus petite contenant celle de $V_1(b)$ et stable sous
l'action de $\frac{1}{a}\ZZ(1)_\Betti$. Definissons de meme $U_s(b)(a)$
et $U_s^1(b)(a)$. Le diagramme \eqref{eq:17.2.3} devient
\begin{equation}\label{eq:17.4.1}
\begin{CD}
1 @>>> V_1(a) @>>> V(a) @>>> \frac{1}{a}\ZZ(1) @>>> 1 \\
@. @VVV @VVV @| \\
1 @>>> U_s^1(a) @>>> U_s(a) @>>> \frac{1}{a}\ZZ(1) @>>> 1
\end{CD}
\end{equation}
et la monodromie en $\tilde{0}$ est un morphisme
\begin{equation}\label{eq:17.4.2}
\mu_{\tilde{0}} : \tfrac{1}{a}\ZZ(1) \to V(a)
\end{equation}
qui scinde les extensions \eqref{eq:17.4.1}.

Si $b_0$ et $b_1$ sont deux vecteurs tangents en $\tilde{0}$, les
groupes \eqref{eq:17.4.1} en $b_1$ se deduisent de ceux en $b_0$ en
tordant par le $\ZZ(1)$-torseur de Kummer $K(b_1/b_0)$ des chemins,
dans l'espace tangent epointe, de $b_0$ a $b_1$.

Prenons pour $a$ un multiple de $d'$ et prenons des vecteurs tangents
de la forme $t(s)$ ($s \in S$). Le rapport $t(s_1)/t(s_0)$ est une
racine $d'$-ieme de $1$ et $K(t(s_1)/t(s_0))$ est donc trivialise
quand on le pousse par $\ZZ(1) \to \frac{1}{a}\ZZ(1)$. L'action de
$\ZZ(1)$ sur \eqref{eq:17.4.1} se prolongeant a
$\frac{1}{a}\ZZ(1)$, les diagrammes 17.4.1 pour $b = t(s)$ sont
canoniquement isomorphes entre eux. Pour $d' | a$ et $b = t(s)$
($s \in S$), on notera
\begin{equation}\label{eq:17.4.3}
\begin{CD}
1 @>>> V^1(a) @>>> V(a) @>>> \frac{1}{a}\ZZ(1) @>>> 1 \\
@. @VVV @VVV @| \\
1 @>>> U_s^1(a) @>>> U_s(a) @>>> \frac{1}{a}\ZZ(1) @>>> 1
\end{CD}
\end{equation}
leur valeur commune. On dispose de
\begin{equation}\label{eq:17.4.4}
\mu_{\tilde{0}} : \tfrac{1}{a}\ZZ(1) \to V(a)
\end{equation}
scindant les extensions \eqref{eq:17.4.3}. On deduit encore de 17.3
que 17.4.3 fait de $V^1(a)$ le produit des $U_s^1(a)$
\begin{equation}\label{eq:17.4.5}
V^1(a) \xrightarrow{\sim} \prod_S U_s^1(a).
\end{equation}

Pour chaque $s \in S$, il existe un unique isomorphisme $\alpha_s$ de
$\Gm$ avec $R'$ tel que $R' \to \Gm$ soit $z \mapsto z^{d'}$ et que
$\alpha_s(1) = s$. L'isomorphisme $\alpha_s$ envoie $(0,1)$ sur
$t(s)$. Il identifie $U_s(t(s))$ avec $U$ (16.11), et
$U_s(t(s))(a)$ avec $U(a)$ (16.21). Puisque
$U_s(a) = U_s(t(s_0))(a)$ est independant de $s_0 \in S$, on a pour
tout $s$
\[
U(a) \xrightarrow{\sim} U_s(a)
\]
et \eqref{eq:17.4.5} devient un isomorphisme
\begin{equation}\label{eq:17.4.6}
V^1(a) \xrightarrow{\sim} (U^1(a))^S.
\end{equation}
On a donc
\begin{equation}\label{eq:17.4.7}
V(a) \xrightarrow{\sim} \tfrac{1}{a}\ZZ(1) \ltimes (U^1(a))^S.
\end{equation}

On dispose de $\mu_1 : \ZZ(1) \to U^1$ (16.11.5). Notons $\mu_s$
($s \in S$) les morphismes
\begin{equation}\label{eq:17.4.8}
\ZZ(1) \to V^1(a)
\end{equation}
qui s'en deduisent par \eqref{eq:17.4.6}.

\subsection{17.5}

Soient $s_0 \in S$ et $b = t(s_0)$. La projection $\pi : R \to X$
envoie $b$ sur le vecteur tangent $1$ en $0$: dans des coordonnees ou
$R'$ est $\Gm$ et ou $\pi$ est $z \mapsto z^{d'}$, les $t(s)$ sont les
racines $d'$-iemes de l'unite et la partie principale en $0$,
$\mathrm{pr}_0(\pi)$ est $z \mapsto z^{d'}$. Le diagramme commutatif
d'espaces pointes
\[
\begin{CD}
R @>>> R' \\
@VVV @VVV \\
X @>>> \Gm
\end{CD}
\]
induit des diagrammes commutatifs
\[
\begin{CD}
\pi_1(R,t(s)) @>>> \pi_1(R',t(s)) \\
@VVV @VVV \\
\pi_1(X,0 \text{ vers } 1) @>>> \pi_1(\Gm, 0 \text{ vers } 1) = \ZZ(1)
\end{CD}
\]
et
\[
\begin{CD}
1 @>>> V_1(t(s)) @>>> V(t(s)) @>>> \ZZ(1) @>>> 1 \\
@. @VVV @VVV @VV{d'}V \\
1 @>>> U^1 @>>> U @>>> \ZZ(1) @>>> 1.
\end{CD}
\]

Par definition des structures entieres $V(dd')$ de
$V(t(s)) \otimes \QQ$, et $U(d)$ de $U \otimes \QQ$, ce diagramme
persiste pour les nouvelles structures entieres
\begin{equation}\label{eq:17.5.1}
\begin{CD}
1 @>>> V^1(dd') @>>> V(dd') @>>> \frac{1}{dd'}\ZZ(1) @>>> 1 \\
@. @VVV @VVV @VV{d'}V \\
1 @>>> U^1(d) @>>> U(d) @>>> \frac{1}{d}\ZZ(1) @>>> 1.
\end{CD}
\end{equation}

\begin{lemma}[17.6]\label{lem:17.6}
Pour $s \in S$, le diagramme
\[
\begin{CD}
\ZZ(1) @>{\mu_s}>> V^1(dd') \\
@| @VVV \\
\ZZ(1) @>{\mu_1}>> U^1(d)
\end{CD}
\]
est commutatif.
\end{lemma}

\noindent \textsc{Preuve.} Prenons pour point base $t(s)$. Pour $t$ un
vecteur tangent en $s$, on dispose de la monodromie autour de $s$:
$\mu' : \ZZ(1) \to V^1(s)$. Parce que, dans $R'$, on dispose d'un
chemin motivique de $t(s)$ a $s$, le $V^1(t(s))$-torseur $Q$ des
chemins de $t(s)$ a $t$ est defini (13.15). La composition des
chemins definit
\begin{equation}\label{eq:17.6.1}
Q \times \QQ(1) \to V^1(t(s)) : (q,t) \mapsto q^{-1}\mu'(t)q.
\end{equation}
Parce que $V^1(s)$ est commutatif, $q^{-1}\mu'(t)q$ ne depend pas de
$q$ et \eqref{eq:17.6.1} se factorise par un morphisme de $\QQ(1)$
dans $V^1(t(s))$, respectant les structures entieres:
\[
\mu' : \ZZ(1) \to V^1(t(s)).
\]
Si on le compose avec l'inclusion de $V^1(t(s))$ dans $V^1(dd')$, on
obtient $\mu_s$.

Il s'agit donc de verifier la commutativite
\[
\begin{CD}
\ZZ(1) @>{\mu'}>> V^1(t(s)) \\
@| @VVV \\
\ZZ(1) @>{\mu_1}>> U^1
\end{CD}
\]
Le morphisme $\mu_1$ admet en effet une construction parallele a celle
de $\mu'$ (16.11.6), et le torseur des $V^1(t(s))$-chemins de $t(s)$ a
$t$ se projette par $\pi : R \to X$, sur celui des $U^1(0 \text{ vers
} 1)$-chemin de $(0 \text{ vers } 1)$ au vecteur tangent $d\pi(t)$ en
$1$.

Negligeons les structures entieres. Par \eqref{eq:16.16.1}
$U^1 \otimes \QQ = \prod_n \QQ(n)$. Par \eqref{eq:17.4.6},
$V^1 \otimes \QQ = (\prod_n \QQ(n))^S$.

\begin{lemma}[17.7]\label{lem:17.7}
Le morphisme 17.5.1 de
$V^1 \otimes \QQ = (\prod_n \QQ(n))^S$ dans
$U^1 \otimes \QQ = \prod_n \QQ(n)$ est
\[
(q_{n,s})_{n \geq 1,\, s \in S} \longmapsto
\bigl(\sum_s d'^{\,n-1} q_{n,s}\bigr)_{n \geq 1}.
\]
\end{lemma}

\noindent \textsc{Preuve.} Par 17.6, la formule proposee est correcte
sur les facteurs $\QQ(1)$. Par \eqref{eq:17.5.1}, le morphisme cherche
n'est pas $\QQ(1)$-equivariant, mais transforme l'action de $t$ en
celle de $d't$. La formule proposee fait de meme, et est caracterisee
par cette propriete. \hfill $\bullet$

De ce lemme resulte le

\begin{corollary}[17.8]\label{cor:17.8}
Le diagramme
\[
\begin{CD}
V^1(dd') = U^1(dd')^S @>{(16.22.2)_{dd',k}^S}>> \ZZ(k)^S \\
@VVV @VV{\sum}V \\
U^1(d) @>{(16.22.2)_{d,k}}>> \ZZ(k)
\end{CD}
\]
est commutatif.
\end{corollary}

\subsection{17.9}

Construisons l'isomorphisme \eqref{eq:17.1.2} pour $\zeta \neq 1$.
Choisissons $a \in R$ au-dessus de $\zeta$. Le choix de $a$ n'importe
pas. Les revetements pointes $(R,a)$ de $(X,\zeta)$ sont d'ailleurs
tous isomorphes. On prend un point base $b$ de la forme $t(s)$
($s \in S$). Puisque, dans $\Gm$, le torseur des
$\frac{1}{dd'}\ZZ(1)$-chemins de $b$ a $a$ est trivial, le
$V^1(dd')$-torseur $Q_a$ des $V^1(dd')$-chemins de $b$ a $a$ est
defini.

Si on pousse ce torseur $Q_a$ par $V^1(dd') \to U_s^1(dd')$ on
obtient le $U_s^1(dd')$-torseur des chemins, sur $R'$, de $b$ a $a$.
Le choix de $b$ n'importe pas. Prenons $b = t(s)$. L'isomorphisme
$\alpha_s$ de $X$ avec $R'$ (17.4) identifie alors le torseur obtenu
au $U^1(dd')$-torseur $(T_{dd'})_{\alpha_s^{-1}(a)}$ des
$U^1(dd')$-chemins de $(0,1)$ a $\alpha_s^{-1}(a)$. Lorsque $s$
parcourt $S$, $\alpha_s^{-1}(a)$ parcourt toutes les racines
$(d')$-iemes de $\zeta$.

Si on pousse $Q_a$ par $V^1(dd') \to U^1(d)$, on obtient le
$U^1(d)$-torseur des chemins de $(0,1)$ a $\zeta$. Appliquant 17.8, on
obtient l'isomorphisme \eqref{eq:17.1.2}.

\subsection{17.10}

Construisons 17.1.2 pour $\zeta = 1$. On choisit encore $a$
au-dessus de $\zeta$. Cette fois, $a \in S$ ne peut pas servir de point
base. On le remplace par le vecteur tangent $(a,-a)$ en $a$. Son
image dans $X$ est le vecteur tangent $(1,-d')$ en $1$.

Soit encore $Q_a$ le $V^1(dd')$-torseur des chemins de $b$ a
$(a,-a)$. Le poussant par $V^1(dd') \to U_s^1(dd')$, on obtient: (a)
si $s \neq a$, le torseur $(T_{dd'})_{\alpha_s^{-1}(a)}$ comme en 17.9;
(b) si $s = a$, le $U^1(dd')$-torseur des chemins de $(0,1)$ a
$(1,-1)$. A nouveau, lorsque $s$ parcourt $S$, $\alpha_s^{-1}(a)$
parcourt toutes les racines $d'$-iemes de $\zeta = 1$.

Si on pousse $Q_a$ par $V^1(dd') \to U^1(d)$, on obtient le
$U^1(d)$-torseur des chemins de $(0,1)$ a $(1,-d')$. Ce torseur, et
celui des chemins de $(0,1)$ a $(1,-1)$, ne different que par
l'image par $\mu_1$ du torseur de Kummer $K(d')$. Ils ont donc meme
image par $(16.22.2)_{d,n}$ pour $n \geq 2$, et on conclut comme en
17.9.


\section{Le torseur $P_{d,k} + (-1)^k \varepsilon P_{d,k}$ est d'ordre fini:
voie geometrique}\label{sec:18}

Le but du paragraphe est de montrer, par voie geometrique, que les
torseurs $P_{d,k}$ de 16.23 verifient 3.14.

\subsection{18.1}

L'automorphisme $z \mapsto 1/z$ de $X = \Gm - \{1\}$ envoie le point
base a l'infini $(0 \text{ vers } 1)$, i.e.\ le vecteur tangent $1$ en
$0$ (16.6) sur $(\infty \text{ vers } 1)$. Il definit
\begin{equation}\label{eq:18.1.1}
\eta : U(0 \text{ vers } 1) \to U(\infty \text{ vers } 1).
\end{equation}
Sur le $\pi_1$ de $\Gm$, $z \mapsto 1/z$ induit $t \mapsto -t$. Le
morphisme \eqref{eq:18.1.1} donne donc lieu a un diagramme commutatif
\begin{equation}\label{eq:18.1.2}
\begin{CD}
1 @>>> U^1(0 \text{ vers } 1) @>>> U(0 \text{ vers } 1) @>>> \ZZ(1) @>>> 1 \\
@. @VV{\eta^1}V @VV{\eta}V @VV{-1}V \\
1 @>>> U^1(\infty \text{ vers } 1) @>>> U(\infty \text{ vers } 1) @>>> \ZZ(1) @>>> 1.
\end{CD}
\end{equation}

Sur $\Gm$, il existe un chemin motivique de $(0 \text{ vers } 1)$ a
$(\infty \text{ vers } 1)$: passer par $1$ et appliquer deux fois 15.51.
La formule (16.11.3) fournit
\begin{equation}\label{eq:18.1.3}
U^1(0 \text{ vers } 1) \xrightarrow{\sim} U^1(\infty \text{ vers } 1).
\end{equation}
Par \eqref{eq:18.1.3}, le morphisme $\eta^1$ de \eqref{eq:18.1.2}
s'identifie a un automorphisme encore note $\eta^1$ de
$U^1 := U(0 \text{ vers } 1)$. Calculons-le.

\begin{proposition}[18.2]\label{prop:18.2}
Via l'isomorphisme \eqref{eq:16.16.1} de $U^1 \otimes \QQ$ avec le
produit des $\QQ(n)$ ($n \geq 1$), $\eta^1$ est
$(x_n) \mapsto ((-1)^{n-1} x_n)$.
\end{proposition}

\begin{lemma}[18.3]\label{lem:18.3}
L'automorphisme $\eta^1$ de $U^1$ preserve $\mu_1 : \ZZ(1) \to U^1$.
\end{lemma}

\noindent \textsc{Preuve.} Le morphisme $\mu_1$ de $\ZZ(1)$ dans
$U^1(0 \text{ vers } 1)$ est la monodromie autour de $1$, ramenee dans
$U^1(0 \text{ vers } 1)$ par le chemin motivique, dans $\Gm$, de
$(0 \text{ vers } 1)$ a $1$. Par $z \mapsto 1/z$, le morphisme
$\mu_1$ est donc transforme en la monodromie autour de $1$, ramenee
dans $U^1(\infty \text{ vers } 1)$ par le chemin motivique, dans $\Gm$,
de $(\infty \text{ vers } 1)$ a $1$. Le compose de ces chemins, de
$(\infty \text{ vers } 1)$ a $(0 \text{ vers } 1)$ ramene donc $\mu_1$
(dans $U^1(0 \text{ vers } 1)$) en le transforme de $\mu_1$ par
$z \mapsto z^{-1}$. Il definit aussi l'isomorphisme de
$U^1(\infty \text{ vers } 1)$ avec $U^1(0 \text{ vers } 1)$, et ceci
verifie 18.3.

\noindent \textsc{Preuve de 18.2.} Par \eqref{eq:18.1.2}, $\eta^1$
transforme l'action de $\ZZ(1)$ sur $U^1$ en son opposee. Par 18.3,
elle respecte $\mu_1$ et ceci force 18.2. \hfill $\bullet$

\subsection{18.4}

Le torseur $P_{d,k}$ a ete defini comme deduit d'un $U^1(d)$-torseur
$T_1$ sur $\mu_d^*$ en poussant par un morphisme
$(16.22.2)_{d,k}$ de $U^1(d)$ dans $\ZZ(k)$. L'automorphisme
$\eta^1$ transforme $T_1$ en $\eta^1 T_1$: un nouveau torseur sur
$\mu_d^*$. Prenant son image inverse par
$f : z \mapsto z^{-1} : \mu_d^* \to \mu_d^*$, on obtient
$f^*\eta^1 T_1$. D'apres 18.2, on a
\[
f^*P_{d,k} = (-1)^{k-1} \varepsilon P_{d,k}.
\]

\begin{lemma}[18.5]\label{lem:18.5}
Le $\ZZ(k)$-torseur $P_{d,k} + (-1)^k \varepsilon P_{d,k}$ est l'image
par $(16.22.2)_{d,k}$ du $U^1(d)$-torseur $T_1 - f^*\eta^1 T_1$.
\end{lemma}

\subsection{18.6}

Dans la suite de ce paragraphe, nous negligeons les structures
entieres. Pour alleger les notations, nous omettrons le $\otimes \QQ$
qui indique l'omission d'une structure entiere. Avec cette
convention, il s'agit de prouver que les
$P_{d,k} + (-1)^k \varepsilon P_{d,k}$ sont triviaux ou, ce qui revient
au meme, que $T_1 - f^*\eta^1 T_1$ est trivial (16.16.1, 18.5).

\subsection{18.7}

Considerons les points base suivant sur $X$: les points base a
l'infini $(a \text{ vers } b)$ pour $a, b \in \{0, 1, \infty\}$ et
$\zeta \in \mu_d^*$, $\zeta \neq 1$. Pour $b_1, b_2$ deux de ces points
base, le $\QQ(1)$-torseur des chemins, dans $\Gm$, de $b_1$ a $b_2$
est trivial. Appelons ``chemin canonique'' sa trivialisation. Ceci
permet
\begin{enumerate}
\item[(a)] Comme en 18.1 d'obtenir un systeme transitif
d'isomorphismes entre ces $U^1(b)$ --- tous identifies a $U^1$.
\item[(b)] De definir le torseur $Q_{b_2,b_1}$ des
$U^1(b_1)$-chemins de $b_1$ a $b_2$ (13.15).
\end{enumerate}

Sur ces torseurs, l'action droite de $U^1 = U(b_1)$ coincide avec
l'action gauche de $U^1 = U(b_2)$. La composition des chemins fournit
un isomorphisme de $U^1$-torseurs
\begin{equation}\label{eq:18.7.1}
Q_{b_3,b_2} \times Q_{b_2,b_1} \xrightarrow{\sim} Q_{b_3,b_1}.
\end{equation}

Pour $a, b, c = 0, 1, \infty$, dans un ordre quelconque, le
$\QQ(1)$-torseur des chemins, dans $T_a - \{0\}$, de
$(a \text{ vers } b)$ a $(a \text{ vers } c)$ est trivial. Ceci et
\eqref{eq:18.7.1} permet d'identifier
$Q_{a \text{ vers } b,\, b_1}$ et $Q_{a \text{ vers } c,\, b_1}$. On
notera ce torseur simplement $Q_{a,b_1}$. De meme pour
$Q_{b_2, a \text{ vers } b}$, note $Q_{b_2,a}$.

\begin{lemma}[18.8]\label{lem:18.8}
Le torseur $Q_{\zeta,\infty}$ est $f^*\eta^1 T_1$.
\end{lemma}

En effet, $z \mapsto 1/z$ envoie $\infty$ vers $0$, $\zeta$ sur
$\zeta^{-1}$, agit par $\eta^1$ sur $U^1$ et definit un isomorphisme de
$\eta^1 Q_{\zeta,\infty}$ avec $(T_1)_{\zeta^{-1}}$.

De \eqref{eq:18.7.1} on deduit donc

\begin{lemma}[18.9]\label{lem:18.9}
Le $U^1$-torseur $T_1 - f^*\eta^1 T_1$ sur $\mu_d^*$ est l'image
inverse sur $\mu_d^*$ du $U^1$-torseur sur $\Spec(\ZZ)$ $Q_{\infty,0}$.
\end{lemma}

Ceci nous ramene a prouver le theoreme, ou $\mu_d^*$ n'apparait plus:

\begin{theorem}[18.10]\label{thm:18.10}
Le torseur $Q_{\infty,0}$ est trivial.
\end{theorem}

\noindent \textit{Mise en garde:} le theoreme affirme seulement
l'existence d'un $U \otimes \QQ$-chemin motivique de $(0 \text{ vers
} 1)$ a $(\infty \text{ vers } 1)$: cf.\ 18.6.

\noindent \textbf{Construction 18.11.}\label{cons:18.11}
On construit un morphisme de schemas motiviques
$h : \QQ(1) - \{0\} \to Q_{\infty,0}$.

Dans 18.11, $\QQ(1) - \{0\}$ est le schema motivique tel que pour
tout foncteur fibre $\omega$ sur un schema $S$, son image par $\omega$
est la droite $\omega\QQ(1)$ sur $S$, moins la section $0$.

Pour construire 18.11, nous utiliserons le langage 5.11: on
choisira $x \in \QQ(1) - \{0\}$, et on construira
$h(x) \in Q_{\infty,0}$. Cela signifie: pour tout foncteur fibre
$\omega$ sur un schema $S$, et pour tout point $x$ de
$\omega(\QQ(1)-\{0\}) = \omega(\QQ(1)) - \{0\}$ sur $S$, la
construction fournit un point $h(x)$ sur $S$ de $\omega(Q_{\infty,0})$,
et on applique 5.11.

\subsection{18.12}

Soit donc $x \neq 0$ dans $\QQ(1)$. Pour $a = 0, 1$ ou $\infty$ et $t$
un vecteur tangent $\neq 0$ en $a$, le $\QQ(1)$-torseur des chemin,
dans l'espace tangent epointe, de $t$ a $-t$ est trivial. On posera
$\alpha_a(x) :=$ (la trivialisation canonique) $+ x$.

\noindent \textsc{Exemple.} En realisation de Betti, et pour
$x = \pm \pi i \in \QQ(1)_\Betti$, $\alpha_a(x)$ est le chemin entier
suivant:
\[
\begin{picture}(100,40)
\put(10,20){$t$}
\put(40,35){$\cdot$}
\put(40,5){$\cdot$}
\put(85,20){$-t$}
\put(15,22){$\line(3,2){20}$}
\put(15,18){$\line(3,-2){20}$}
\put(50,35){$\line(3,-2){30}$}
\put(50,5){$\line(3,2){30}$}
\end{picture}
\]

L'automorphisme $z \mapsto 1/z$ de $X$ induit un isomorphisme
$\eta^1$-equivariant de $Q_{0,1}$ avec $Q_{\infty,1}$. On le note
encore $\eta^1$. On definit $h(x)$ par la

\begin{proposition}[18.13]\label{prop:18.13}
Soit $x \neq 0$ dans $\QQ(1)$. Localement fppf, il existe un chemin
$\gamma \in Q_{1,0}$ de $(0 \text{ vers } 1)$ a $(1 \text{ vers } 0)$
pour lequel les deux $U$-chemins suivant de $(0 \text{ vers }
\infty)$ a $(\infty \text{ vers } 0)$ coincident:
\begin{multline}\label{eq:18.13.1}
\alpha_\infty(x) \cdot \eta^1 \gamma^{-1} \cdot \alpha_1(x)
\cdot \gamma \cdot \alpha_0(x) \\
\qquad \text{et} \\quad
\eta^1 \alpha_0(x) \cdot \eta^1 \gamma^{-1} \cdot \eta^1
\alpha_1(-x) \cdot \gamma \cdot \alpha_\infty(-x)^{-1}.
\end{multline}
Le chemin $h(x) := \eta^1 \gamma^{-1} \cdot \gamma \in
Q_{\infty,0}$ est independant du choix de $\gamma$ verifiant
18.13.1.
\end{proposition}

\begin{remark}[18.14]\label{rem:18.14}
En realisation de Betti et pour $x = \pi i \in \QQ(1)_\Betti$, on
peut prendre pour $\gamma$ le chemin reel $\gamma_\RR$ de $(0
\text{ vers } 1)$ a $(1 \text{ vers } 0)$. La relation
\eqref{eq:18.13.1} exprime l'homotopie des chemins suivants dans
$P^1(\CC) - \{0,1,\infty\}$.
\end{remark}

\begin{remark}[18.15]\label{rem:18.15}
Il est clair sur \eqref{eq:18.13.1} qu'on aura $h(x) = -h(-x)$.
\end{remark}

\noindent \textsc{Preuve de 18.13.} Deux foncteurs fibres etant
localement isomorphes, il suffit de prouver 18.13 en realisation de
de Rham. En realisation de de Rham, $\pi_1(X,b)_\dR$, $U(b)_\dR$,
$U^1(b)_\dR$ sont independants du point base et le
$U(b_2)_\dR$-torseur des chemins de $b_1$ a $b_2$ est trivialise. De
meme pour le $U^1(b_2)_\dR$-torseur $(Q_{b_2,b_1})_\dR$, lorsque ce
dernier est defini.

Dans $\Lie U_\dR$, notons $e_0$, $e_1$ et $e_\infty$ les images par
$\mu_0$, $\mu_1$ et $\mu_\infty$ de $1 \in \QQ(1)_\dR$. On a
\[
e_0 + e_1 + e_\infty = 0.
\]
Posons $e_i' = (\ad e_0)^{n-1}(e_1)$ ($n \geq 1$). La pro-algebre de
Lie $\Lie U_\dR$ a pour base $e_0$ et les $e_i'$, et
$\Lie U^1_\dR$ a pour base les $e_i'$. Un element $\gamma$ de
$U^1_\dR$ s'ecrit
\[
\gamma = \exp\bigl(\sum_{n \geq 1} \gamma_n e_n'\bigr).
\]
On lui attache la serie generatrice
\[
\gamma(t) = \sum_{n \geq 1} \gamma_n t^n.
\]
La serie generatrice de $\exp(xe_0) \gamma \exp(-xe_0)$ est
$\exp(xt) \cdot \gamma(t)$.

Soit $\gamma \in (Q_{1,0})_\dR = U^1_\dR$,
$\gamma = \exp(\sum \gamma_n t^n)$, et explicitons
\eqref{eq:18.13.1}.

On a
\begin{align*}
\alpha_a(x) &= \exp(xe_a), \\
\eta^1 \gamma^{-1} &= \exp\bigl(-\sum (-1)^{n-1} \gamma_n
e_n'\bigr), \\
\eta^1 \gamma^{-1} \cdot \alpha_1(x) \cdot \gamma &=
\exp\bigl(xe_1 + 2\sum \gamma_{2n} e_{2n}'\bigr), \\
\eta^1 \gamma^{-1} \cdot \alpha_1(-x) \cdot \gamma &=
\exp\bigl(-xe_1 + 2\sum \gamma_{2n} e_{2n}'\bigr),
\end{align*}
et \eqref{eq:18.13.1} se recrit
\begin{multline}\label{eq:18.16.1}
\exp(xe_\infty) \exp\bigl(xe_1 + 2\sum \gamma_{2n}
e_{2n}'\bigr) \exp(xe_0) = \\(= 
\exp(-xe_\infty) \exp\bigl(-xe_1 + 2\sum \gamma_{2n}
e_{2n}'\bigr) \exp(-xe_0).
\end{multline}
On voit que l'equation obtenue ne porte que sur les coordonnees
paires $\gamma_{2n}$ de $\gamma$.

On a $-xe_\infty = xe_0 + xe_1$. Appliquant Bourbaki Lie II \S 6
prop.\ 5, et le fait que tout polynome de Lie en $e_0$ et $e_1$,
homogene de degre $\geq 2$ en $e_1$ est nul, on trouve
\[
\exp(xe_\infty) = \exp(xe_0) \exp(xe_1).
\]
Le second membre de \eqref{eq:18.16.1} est donc
\[
\exp\bigl(\frac{\exp(\ad xe_0) - 1}{\ad xe_0}(xe_1)\bigr)
\cdot \exp(xe_0) \exp(-xe_1 + 2\sum \gamma_{2n} e_{2n}')
\exp(-xe_0),
\]
de serie generatrice
\[
e^{-xt} \cdot \frac{e^{xt} - 1}{xt} \cdot (-xt + S),
\]
ou $S = 2\sum \gamma_{2n} t^{2n}$. Remplacant $x$ par $-x$, on
obtient de meme la serie generatrice du premier membre.

Posons $S = 2\sum \gamma_{2n} t^{2n}$. L'equation
\eqref{eq:18.16.1} donne,
\begin{align*}
e^{-xt} \cdot \frac{e^{xt}-1}{xt} \cdot (xt + S) &= e^{xt}
\cdot \frac{e^{-xt}-1}{-xt} \cdot (-xt + S), \\
S(e^{xt} - e^{-xt}) &= -(e^{xt} - e^{-xt}) + xt(e^{xt} + e^{-xt}),
\\
S &= -1 + xt \cdot \frac{e^{xt} + e^{-xt}}{e^{xt} - e^{-xt}}.
\end{align*}

Le membre de droite est une fonction paire de $t$, nulle en $t = 0$
et \eqref{eq:18.16.1} equivaut a
\begin{equation}\label{eq:18.16.2}
2\gamma_{2n} = \text{coefficient de } t^{2n} \text{ dans } xt
\coth(xt) \quad (n \geq 1).
\end{equation}
Ceci prouve l'existence de $\gamma$. Enfin,
\[
h(x) = \eta^1 \gamma^{-1} \cdot \gamma =
\exp\bigl(-2\sum (-1)^{n-1} \gamma_n e_n'\bigr) \cdot
\exp\bigl(\sum \gamma_n e_n'\bigr)
\]
est independant du choix de $\gamma$. \hfill $\bullet$

\begin{remark}[18.17]\label{rem:18.17}
Le chemin reel $\gamma_\RR$ de $(0 \text{ vers } 1)$ a $(1 \text{
vers } 0)$ verifie 18.13.1 pour $x = \pm \pi i$ (18.14). Son image
$\comp_\dR^\Betti(\gamma_\RR)$ dans $(Q_{1,0})_\dR \otimes \CC
\cong U^1_\dR \otimes \CC$ a donc des coordonnees
$\gamma_{\RR,n}$ verifiant
\begin{equation}\label{eq:18.17.1}
2\gamma_{\RR,2n} = \text{ coefficient de } t^{2n} \text{ dans }
\pi i t \coth(\pi i t) \quad (n \geq 1).
\end{equation}
En 19.4, pour verifier l'egalite des torseurs 3.1 et 16.18, nous
prouverons que
\[
\gamma_{\RR,n} = \zeta(n).
\]
Le calcul 18.16 fournit donc une nouvelle preuve de la formule
d'Euler pour $\zeta(2n)$, dans laquelle les nombres de Bernoulli
apparaissent comme lies aux coefficients de la loi de
Campbell-Hausdorff.
\end{remark}

\end{proposition}

\noindent \textsc{Preuve.} Il suffit de le verifier apres application

\section{Comparaison de $\ZZ(k)$-torseurs des paragraphes 3 et 16}\label{sec:19}

\subsection{19.1}

Dans ce paragraphe, nous noterons $P^3_{d,k}$ les torseurs sous
$\ZZ(k)$ construits au paragraphe 3, et $P_{d,k}$ ceux construits au
paragraphe 16 par voie geometrique.

Si $k=1$, le torseur $P^3_{d,k}$ n'a ete defini que sur
$\mu_d^{**} := \mu_d[1/d] \setminus \{1\}$, tandis que $P_{d,k}$ a
ete defini sur $\mu_d$ tout entier, mais est ininteressant en le point
$1$, ou il se reduit au torseur trivial. Sur $\mu_d^{**}$, $P^3_{d,1}$
et $P_{d,1}$ sont l'un et l'autre isomorphes au torseur de Kummer
$K(1-\zeta)$ ($\zeta \in \mu_d^*$) (3.12 et 16.26). Pour simplifier les
notations, nous supposerons dans la suite de ce paragraphe que
$k \geq 2$. Notre but est de prouver que $P^3_{d,k}$ et $P_{d,k}$
sont isomorphes. C'est l'assertion (16.24) et le cas particulier $d=1$
est (16.20).

La meme preuve s'appliquerait pour $k=1$, quitte a remplacer partout
$\mu_d$ par $\mu_d^{**}$.

Au paragraphe 3, en meme temps que $P^3_{d,k}$, nous avons defini une
trivialisation $u_\Betti$ du $\ZZ(k)_\Betti$-torseur qui en est la
realisation de Betti. Dans ce paragraphe, nous la noterons $u^3_\Betti$.

Les torseurs $P_{d,k}$ ont ete definis comme image par
$(16.22.2)_{d,k}$ d'un $U^1(d)$-torseur $P_d$. En 19.2 nous
definirons une trivialisation $V_\Betti$ du $U^1(d)_\ZZ$-torseur qui
en est la realisation de Betti. Nous prouverons ensuite l'existence
d'un isomorphisme
\begin{equation}\label{eq:19.1.1}
P^3_{d,k} \xrightarrow{\sim} P_{d,k}
\end{equation}
qui envoie $u^3_\Betti$ sur l'image dans $(P_{d,k})_\Betti$ de
$V_\Betti$.

Voici ce qu'il s'agit de prouver.
\begin{enumerate}
\item[(A)] En realisation de de Rham, un $\ZZ(k)$ ou un
$U^1(d)$-torseur est automatiquement trivialise. Via
$\comp^\dR_\Betti$, $u^3_\Betti$ devient la donnee, pour chaque
$\zeta \in \mu_d^*$, d'un nombre $u^3_{\Betti,\zeta} \in
\ZZ(k)_\dR \otimes \CC = \CC$. De meme, $V_\Betti$ devient la
donnee, pour chaque $\zeta \in \mu_d^*$, de $V_{\Betti,\zeta} \in
U^1_\dR \otimes \CC$. Il faut verifier que l'image par
$(16.22.2)_{d,k}$ de $V_{\Betti,\zeta}$ est $u^3_{\Betti,\zeta}$.
\item[(B)] Soit $U_\Betti$ l'image de $V_\Betti$ dans
$P_{d,k\,\Betti}$. Il faut verifier que l'isomorphisme de
$\ZZ_\ell(k)(\CC)$-torseur sur $\mu_d(\CC)$,
$(P^3_{d,k})_\ell \to (P_{d,k})_\ell$, qui envoie
$\comp_{\ell,\Betti}(u^3_\Betti)$ sur
$\comp_{\ell,\Betti}(U_\Betti)$, est $\Gal(\overline{\QQ}/\QQ)$-equivariant.
\item[(C)] Identifions la realisation de de Rham de $P^3_{d,k}$
(resp.\ $P_d$) au torseur trivial --- sur $\mu_d^*$ --- sous
$\ZZ(k)_\dR$ (resp.\ $U^1_\dR$). L'inverse $\phi_p^{-1}$ du
Frobenius cristallin de $P^3_{d,k}$ (resp.\ $P_d$) est determine par
sa valeur en $0$. Il s'agit de montrer que $(16.22.2)_{d,k}$ envoie
$\phi_p^{-1}(0)$ pour $P_d$ dans $\phi_p^{-1}(0)$ pour $P^3_{d,k}$.
\end{enumerate}

Les relations de distribution 3.13.1 ont ete prouvees tant pour
$P^3_{d,k}$ que pour $P_{d,k}$ (\S\ref{sec:17}). Elles impliquent qu'il
suffit de verifier l'isomorphie de $P^3_{d,k}$ et $P_{d,k}$ sur
$\mu_d^{**}$, pour l'avoir sur $\mu_d$ tout entier. Nous utiliserons
cette remarque en ne prouvant (C) que sur $\mu_d^{**}$. A cela pres,
les calculs du present paragraphe sont independants de ceux du
paragraphe 17. Le lecteur pourra noter que la definition (3.2.6,
3.2.7) du Frobenius cristallin de $P^3_{d,k}$ en $1 \in \mu_d$ est
artificielle, concoctee pour rendre vraie la relation de distribution
(3.13.1). Pour $p \neq 2$, elle revient aussi a definir $P^3_{d,k}$
en $1$ pour avoir un isomorphisme de $\ZZ(k)$-torseurs sur
$\Spec(\ZZ[1/2])$:
\[
\bigotimes_{\zeta^2=1} (P^3_{d,k})_\zeta \xrightarrow{\sim} P^3_{d,k}\big|_{\zeta=1}.
\]

\subsection{19.2}

Soit $z \in \CC^*$, suppose ne pas etre reel $\geq 1$. Dans
$X(\CC) = \CC^* \setminus \{0\}$ appelons \textit{droit chemin} de
$(0,z)$ (vecteur tangent en $0$) a $z$ le chemin suivant. Dans le
langage 15.9: aller de $(0,z)$ a $(0,\varepsilon z)$, $\varepsilon$
reel $> 0$, par valeurs $(0,tz)$ avec $t$ reel ($\varepsilon \leq t
\leq 1$), sauter de $(0,\varepsilon z)$ a $\varepsilon z$, et aller de
$\varepsilon z$ a $z$ par valeurs $tz$ avec $t$ reel ($\varepsilon \leq t
\leq 1$).

Supposons que $z$ soit une racine $d$-ieme de $1$, $z \neq 1$. On
dispose alors d'un $\frac{1}{d}\ZZ(1)$-chemin motivique (dans l'espace
tangent epointe) de $(0,1)$ a $(0,z)$. Le composant avec le droit
chemin de $(0,z)$ a $z$, on en deduit, en realisation de Betti, un
$U(d)_\ZZ$-chemin $\gamma_z$ de $(0,1)$ a $z$. C'est en fait un
$U^1(d)_\ZZ$-chemin: il s'envoie sur le $\frac{1}{d}\ZZ(1)$-chemin
canonique, sur $\Gm$, de $(0,1)$ a $z$. Pour $z$ le vecteur tangent
$-1$ en $1$, on dispose de meme du chemin reel (16.10) de $(0,1)$ a
$(1,-1)$. On le note $\gamma_1$. Ces chemins constituent la
trivialization promise en 19.1, $V_\Betti$, de $P_d$.

\subsection{19.3}

Verifions 19.1(A). Nous prendrons sur $U^1_\dR$ les coordonnees $u$
et $v_i$ ($1 \leq i \leq N$) suivantes:
\[
(u, (v_i)) \longmapsto \exp\bigl(\sum v_i (\ad e_0)^{i-1}(e_1)\bigr) \exp(u e_0)
\]
ou $e_0$ et $e_1$ sont les images par $\mu_0$ et $\mu_1$ de $1 \in
\QQ(1)_\dR$.

\begin{lemma}[19.3.1]\label{lem:19.3.1}
Le droit chemin de $(0,z)$ a $z$ a pour image dans $U^1_\dR(\CC)$ le
point de coordonnees $u = 0$ et $v_i = -\ell_i(z)$.
\end{lemma}

\noindent \textsc{Preuve.} L'image $I_z^{z'}$ d'un chemin de $z_1$ a
$z_2$ s'obtient en resolvant le long du chemin
\[
dI(t) \cdot I(t)^{-1} = \frac{dt}{t} e_0 + \frac{dt}{t-1} e_1.
\]
on prend la condition initiale $I(z_1) = \text{identite}$, et on
evalue en $z_2$.

L'equation a resoudre est
\begin{align*}
du &= \frac{dt}{t} \\
dv_1 &= \frac{dt}{t-1} \\
dv_i &= \frac{dt}{t} v_{i-1}.
\end{align*}
Pour $z_1$ un vecteur tangent $T$ en $0$, la condition initiale est a
remplacer par
\[
I(t) \exp(-\log(t/T)e_0) \to \text{identite} \quad \text{pour } t \to 0.
\]
Pour le droit chemin de $(0,z)$ a $z$, on obtient
\begin{align*}
u &= \log(t/z) \\
v_i &= -\ell_i(t).
\end{align*}

\begin{lemma}[19.4]\label{lem:19.4}
Le chemin reel de $(0,1)$ a $(1,-1)$ a pour image dans
$U^1_\dR(\CC)$ le point de coordonnees $u = 0$, $v_1 = 0$,
$v_i = \ell_i(1) = \zeta(i)$ pour $i \geq 2$.
\end{lemma}

\noindent \textsc{Preuve.} Il s'agit de considerer $I(t)$ defini par
\[
u = \log t, \quad v_i = -\ell_i(t)
\]
et de prendre la limite pour $\varepsilon \to 0$ de
\[
\exp(-\log \varepsilon \cdot e_1) I(1-\varepsilon).
\]
La multiplication par $\exp(-\log \varepsilon \cdot e_1)$ a pour effet
de remplacer $v_1$ par $0$ et 19.4 en resulte.

\subsection{19.5}

Soit $\zeta$ une racine $d$-ieme de $1$ autre que $1$. L'image par
$(16.22.2)_{d,k}$ du point de $U^1(\CC)$ de coordonnees
$(-\ell_i(\zeta))$ est (3.11.1). De meme, l'image par
$(16.17.1)_k$ du point de coordonnees $(0, -\ell_i(1) \text{ pour }
i \geq 2)$ est (3.1.1). Ceci verifie 19.1(A).

\subsection{19.6}

Verifions 19.1(C). Placons-nous sur $\ZZ_p$. L'image directe par
Frobenius
\[
F_* : \pi_1(X,(0,1))_\dR \to \pi_1(X,(0,1))_\dR
\]
induit sur l'algebre de Lie du quotient $U_\dR$ le morphisme $F_*$
tel que
\[
F_*(e_0) = p e_0, \quad F_*(e_1) = p e_1.
\]
C'est l'inverse du Frobenius cristallin.

Calculons la structure cristalline du $U_\dR$-torseur $P_\dR$ des
$U_\dR$-chemin de $(0,1)$ a $x \in X$. Sur le complete formel de
$P^1$ le long de $P^1 \setminus \{1\}$ en caracteristique $p$, le
Frobenius admet le relevement $F : x \mapsto x^p$. La structure
cristalline est determinee par
\begin{equation}\label{eq:19.6.1}
F_* : F^* P_\dR \to P_\dR.
\end{equation}
En termes de l'isomorphisme $P_\dR = \text{torseur trivial } U_\dR$,
le morphisme \eqref{eq:19.6.1} est de la forme $g \mapsto v \cdot F_*(g)$.

Quand on pousse par $U_\dR \to \pi_1(\Gm,(0,1))_\dR$, on obtient le
torseur de Kummer, avec $F_*(e) = e$. On a donc ici $F_*(e) \in
U^1_\dR \subset P_\dR \cdot U_\dR$.

Soit $v = F_*(e)$. Que $F_*$ est horizontal s'ecrit
\begin{equation}\label{eq:19.6.2}
v^{-1} \nabla v = F_*(-\alpha) - v^{-1} d(v).
\end{equation}
A gauche, $e^{-1} \nabla e$ est $-\alpha = -(\frac{dt}{t} e_0 +
\frac{dt}{t-1} e_1)$ et $F_*$ est l'image par $e_0 \mapsto p e_0$,
$e_1 \mapsto p e_1$. A droite, la connexion est celle de
$F^* P_\dR \cong U_\dR$, avec $\nabla e = -F^*\alpha$. En coordonnees:
$v = \exp(\sum v_i (\ad e_0)^{i-1}(e_1))$, \eqref{eq:19.6.2} s'ecrit
\[
-\frac{dt}{t} \cdot p e_0 - \frac{dt}{t-1} \cdot p e_1 = F_*(-\alpha) -
[\sum v_i (\ad e_0)^{i-1}(e_1), F_*(-\alpha)] + dv.
\]
Developpant, cela donne
\begin{align*}
dv_1 &= F_*\bigl(\frac{dt}{t-1}\bigr) - \frac{p\,dt}{t-1} \\
dv_i &= \frac{p\,dt}{t} v_{i-1} \quad (i > 1).
\end{align*}
Puisque le point base est $(0,1)$, on cherche la solution nulle en
$t=0$. Le developpement en serie formelle de $v_i(t)$ est donc
\begin{equation}\label{eq:19.6.3}
v_i(t) = \sum_{n \geq 1,\, p \nmid n} \frac{t^n}{n^i}.
\end{equation}

\subsection{19.7}

Puisque $F$ respecte $\mu_d^*$, on obtient la structure cristalline
du torseur des $U_\dR$-chemins de $(0,1)$ a $\zeta \in \mu_d^*$
simplement en se restreignant a $\mu_d^*$. Appliquant
$(16.22.2)_{d,k}$ a \eqref{eq:19.6.3}, on obtient $(3.11.4)_{d,k}$
comme requis.

\subsection{19.8}

Soient $N$ et $M$ deux entiers. Soit $R_{N,M}$ le revetement de $X$
dont la fibre en un point geometrique $z$ est l'ensemble des systemes
$(u, (v_\zeta))$ avec
\begin{equation}\label{eq:19.8.1}
u^N = z, \quad v_\zeta^M = 1 - \zeta u \quad (\zeta^N = 1).
\end{equation}
Le revetement $R_{N,M}(\CC)$ de $X(\CC)$ est donc celui sur lequel
$z^{1/N}$ et les $(1 - \zeta z^{1/N})^{1/M}$ sont definis.

Il sera plus commode d'indexer les $v_\zeta$ par la racine $N$-ieme
$\zeta u$ de $z$. Avec cette notation, \eqref{eq:19.8.1} devient
\begin{equation}\label{eq:19.8.2}
v(x)^M = 1 - x \qquad (x^N = z).
\end{equation}
Les revetements $R_{N,M}$ forment un systeme projectif, pour $N$ et
$M$ ordonnes par divisibilite. Les morphismes de transition envoient
une solution $(u^{(NN')}, v^{(NN',MM')}(x))$ de \eqref{eq:19.8.2} sur
\begin{equation}\label{eq:19.8.3}
u^{(N)} := u^{(NN')N'}, \quad v^{(N,M)}(x) =
\prod_{y^{N'}=x} v^{(NN',MM')}(y)^{M'}.
\end{equation}
On pose $R_N := R_{N,N}$.

\begin{proposition}[19.9]\label{prop:19.9}
Soit $R$ un revetement fini connexe de $X(\CC)$. Supposons qu'il
existe un revetement $X'$ de $X$, induit par un revetement connexe de
$\CC^* \supset X(\CC)$, tel que le revetement $R'$ de $X'$ image
inverse de $R$ soit somme de revetements abeliens. Alors, $R$ est
domine par un $R_N$.
\end{proposition}

\noindent \textsc{Preuve.} Le revetement $X'$ est $z \mapsto z^A :
\CC^* - \mu_A \to \CC^* - \{1\}$. Si on remplace $A$ par un
multiple, $R'$ sera non ramifie en $0$, et ses composantes sont
dominees par des revetements kummeriens de $\CC - \mu_A$, obtenus en
extrayant les racines $B$-iemes des $1 - \zeta u$ ($\zeta^A = 1$).
Le revetement $R$ est alors domine par $R_{A,B}$, et par $R_N$ pour
$N$ multiple de $A$ de $B$.

\subsection{19.10}

Les revetements $R_{N,M}$ sont munis d'un point marque au-dessus de
$(0,1)$: $R_{N,M} \to X$ est le compose de (a) $z \mapsto z^N : \Gm -
\mu_N \to \Gm - \{1\} = X$, et (b) un revetement induit par un
revetement de $\GA - \mu_N$, de fibre en $0$ les systemes de racines
$M$-iemes des $1 - \zeta \cdot 0$. On releve $(0,1)$ en le vecteur
tangent $1$ en $0$ de $\GA - \mu_N \supset \Gm - \mu_N$, et
$0 \in \GA - \mu_N$ en le systeme des racines $M$-iemes $1$ des
$1 - \zeta \cdot 0$. Les morphismes de transition du systeme projectif
des $R_{N,M}$ envoient point marque sur point marque.

\subsection{19.11}

Pour $G$ un groupe, faisons agir $G$ par multiplication a gauche sur
l'algebre de groupe $\ZZ/M[G]$ et definissons
\[
\Gamma_{G,M} := G \ltimes \ZZ/M[G](1).
\]
Pour $M$ parcourant les entiers ordonnes par divisibilite les
$\Gamma_{G,M}$ forment un systeme projectif. Il est fonctoriel en
$G$. Pour $N$ un entier, posons
\[
\Gamma_{N,M} := \Gamma_{\ZZ/N(1),M}.
\]
Ces groupes forment un systeme projectif, pour $N$ et $M$ ordonnes
par divisibilite. Notation: on note $\sum a_g \delta_g$ un element de
l'algebre de groupe $\ZZ/M[G]$. De meme pour $\ZZ/M[G](1)$, avec
cette fois $a_g \in \ZZ/M(1)$. On definit $\mu_0 : \ZZ/N(1) \to
\Gamma_{N,M}$ comme inclusion du premier facteur $\ZZ/N(1)$ et
$\mu_1 : \ZZ/M(1) \to \Gamma_{N,M}$ par $\alpha \mapsto \alpha \delta_0$.

Il existe une unique action a droite de $\Gamma_{N,M}$ sur $R_{N,M}$
pour laquelle $\mu_0(\alpha)$ ($\alpha \in \ZZ/N(1) = \mu_N$) agit sur
$(u,v(x))$ en multipliant $u$ par $\alpha$, et $\mu_1(\beta)$
($\beta \in \ZZ/M(1) = \mu_M$) agit en multipliant $v(u)$ par
$\beta$; les autres coordonnees sont inchangees. L'element
$\sum a_\zeta \delta_\zeta$ de $\ZZ/M[\ZZ/N(1)](1)$ agit en
multipliant $v(x)$ par $a_x/u \in \ZZ/M(1) = \mu_M$. Pour $N$ et $M$
ordonnes par divisibilite, et les $R_{N,M}$ forment un systeme
projectif de torseurs sous le systeme projectif des $\Gamma_{N,M}$.

Posons $\Gamma_N := \Gamma_{N,N}$:
\[
\Gamma_N = \ZZ/N(1) \ltimes \Gamma_N^1 \quad \text{avec} \quad
\Gamma_N^1 := \ZZ/N[\ZZ/N(1)](1).
\]
La limite projective $R^\wedge$ des $R_N$ est un torseur profini sous
la limite projective $\Gamma^\wedge$ des $\Gamma_N$.

\subsection{19.12}

Placons-nous sur $\CC$. D'apres 19.9, $\Gamma^\wedge$ est le complete
profini du quotient $\Gamma$ de $\pi_1(X(\CC),(0,1))$ defini en 16.14:
\[
\Gamma^\wedge = \hat{\ZZ}(1) \ltimes \hat{\ZZ}[\hat{\ZZ}(1)](1).
\]

Soit
\begin{equation}\label{eq:19.12.1}
\varphi_{N,M} : \Gamma \to \Gamma_{N,M}
\end{equation}
le systeme projectif des morphismes tels que $R_{N,M}(\CC)$ se
deduise du $\Gamma$-torseur fondamental en poussant $\varphi_{N,M}$.

\begin{lemma}[19.13]\label{lem:19.13}
Les $\varphi_{N,M}$ envoient les $\mu_i$ de (16.14) sur ceux de 19.11
($i=0,1$).
\end{lemma}

\noindent \textsc{Preuve.} Pour $\alpha = 2\pi i \in \ZZ(1)_\Betti$,
$\mu_0(\alpha)$ donne la monodromie autour de $0$. Elle envoie
$(u,v(x))$ sur $(\exp(2\pi i/N)u, v(x))$. De meme, $\mu_1(\alpha)$
donne la monodromie au tour de $1$, joint a $0$ par le chemin reel de
$(0,1)$ a $(1,-1)$. Par ce chemin, le point marque de $R_{N,M}$
au-dessus de $(0,1)$ est transforme en un vecteur tangent en un point
ou la racine $N$-ieme $u$ de $z=1$ est $1$. Cette monodromie
multiplie $v(u)$ par $\exp(2\pi i\alpha/M)$ et respecte les autres
coordonnees.

\begin{corollary}[19.14]\label{cor:19.14}
Les $\varphi_{N,M}$ sont les morphismes evidents de
$\ZZ(1)_\Betti \ltimes \ZZ[\ZZ(1)_\Betti](1)$ dans
$\ZZ/N(1) \ltimes \ZZ/M[\ZZ/N(1)](1)$.
\end{corollary}

Par passage a la limite projective, 19.14 identifie le complete
profini de $\Gamma = \ZZ(1)_\Betti \ltimes \ZZ[\ZZ(1)_\Betti](1)$ avec
la limite projective $\Gamma^\wedge$ des $\Gamma_N$:
\[
\Gamma^\wedge = \hat{\ZZ}(1) \ltimes \hat{\ZZ}[\hat{\ZZ}(1)](1),
\]
ou l'algebre de groupe $\hat{\ZZ}[\hat{\ZZ}/N(1)]$ est par definition la
limite projective des $\ZZ/N[\ZZ/N(1)]$.

L'algebre de groupe $\ZZ[\ZZ(1)_\Betti]$ est l'algebre affine du tore
$T$ sur $\Spec(\ZZ)$ de groupe de caracteres $\ZZ(1)_\Betti$. Le
complete $\hat{\ZZ}[\hat{\ZZ}(1)_\Betti]$ est le produit des completes de
cette algebre affine en tous les points fermes de $T$.

\subsection{19.15}

Les $\Gamma_\ell^A := \Gamma_{N,N}$ pour $N$ une puissance de $\ell$
forment un systeme cofinal de $\ell$-groupes quotients de $\Gamma$, et
\[
\Gamma_\ell^\wedge = \ZZ_\ell(1) \ltimes \ZZ_\ell[\ZZ_\ell(1)](1).
\]

L'algebre $\ZZ_\ell[\ZZ_\ell(1)]$ est encore le complete de l'algebre
affine du tore $T$ de 19.14 en le point ferme suivant: element neutre
en caracteristique $\ell$. Pour obtenir $\Gamma_\ell^\wedge(A)$, il
faut la remplacer par son quotient par $(U-1)^A$, $U$ un generateur de
$\ZZ_\ell(1)$:
\begin{align*}
\Gamma_\ell^\wedge(A) &= \ZZ_\ell(1) \ltimes \Gamma_\ell^{1\wedge}(A) \\
\Gamma_\ell^{1\wedge}(A) &:= \ZZ_\ell[u]/((u-1)^A)(1)
\end{align*}
pour $u$ un generateur de $\ZZ_\ell(1)$.

\begin{proposition}[19.15]\label{prop:19.15}
Le morphisme $(16.17.1)_k$:
\[
\hat{\ZZ}[\hat{\ZZ}(1)](1) \to \hat{\ZZ}(k) \qquad (1 \leq k \leq A)
\]
est la limite projective pour $N$ une puissance de $\ell$ des
\begin{equation}\label{eq:19.15.1}
\ZZ/N[\ZZ/N(1)](1) \to \ZZ/N(k) : \quad \sum a_\zeta \delta_\zeta
\longmapsto \sum a_\zeta \cdot \zeta^{\otimes(k-1)}.
\end{equation}
\end{proposition}

\noindent \textsc{Preuve.} Regardons $\ZZ_\ell[\ZZ_\ell(1)]$ comme
l'espace des mesures sur $\ZZ_\ell(1)$ a valeurs dans $\ZZ_\ell$. La
limite projective des \eqref{eq:19.15.1} est alors le produit tensoriel
par $\ZZ_\ell(1)$ de
\begin{equation}\label{eq:19.15.2}
\ZZ_\ell[\ZZ_\ell(1)] \to \ZZ_\ell(k) : \quad \alpha \longmapsto
\int \zeta^{\otimes k-1} d\alpha.
\end{equation}

Le morphisme $(16.17.1)_k$ est le produit tensoriel avec
$\ZZ_\ell(1)$ du morphisme de $\ZZ_\ell[\ZZ_\ell(1)]$ dans
$\ZZ_\ell(k-1)$ suivant: on choisit un generateur $u$ de
$\ZZ_\ell(1)$, on envoie $\ZZ_\ell[\ZZ_\ell(1)]$ sur
$\ZZ_\ell[u]/((u-1)^A)$, on fait $u = e^v$ et on prend le coefficient
de $v^{k-1}/(k-1)!$, multiplie par $u^{\otimes k-1}$.

L'image de $\delta_n \cdot u \in \ZZ_\ell[\ZZ_\ell(1)]$ dans
$\ZZ_\ell[u]/((u-1)^A)$ est $u^n$. Image par
\eqref{eq:19.15.2}: $(nu)^{\otimes k-1}$. Image comme ci-dessus:
$u^{\otimes k-1}$ fois le coefficient de $v^{k-1}/(k-1)!$ dans
$e^{nv}$. Ces images coincident pour tout $n$, et ceci verifie
19.15.

\subsection{19.16}

Verifions 19.1(B) pour $d=1$. D'apres 19.15, le torseur
$P_{1,k}\, \ZZ/N$ sous $\ZZ/N(k)$ s'obtient en prenant la fibre de
$R_N$ en $(1,-1)$, en restreignant le groupe structural a
$\Gamma_N^1$ et en poussant par \eqref{eq:19.15.1}. Le
$\Gamma_N^1$-torseur obtenu est celui des systemes $v_\zeta$
($\zeta \in \mu_N$) ou $v_\zeta^N = 1-\zeta$ pour $\zeta \neq 1$ et
ou $v_1^N = 1/N$ pour $\zeta = 1$. On est retombe sur (3.1.2)$_k$ et
on verifie qu'au chemin reel de $(0,1)$ a $(1,-1)$ correspond le choix
des determinations principales des racines $N$-iemes.

\subsection{19.17}

Posons
\[
\Gamma_N(d) := \Gamma_{G,N} \quad \text{pour } G =
\tfrac{1}{d}\ZZ/\tfrac{1}{d}\ZZ(1).
\]
C'est le produit semi-direct de $G$ par $\Gamma_N^1(d) :=
\ZZ/N[G](1)$. L'inclusion de $\ZZ$ dans $\frac{1}{d}\ZZ$ induit un
morphisme de $\ZZ/N\ZZ(1)$ dans
$\frac{1}{d}\ZZ/\frac{1}{d}\ZZ(1)$, d'ou
\begin{equation}\label{eq:19.17.1}
\Gamma_N \to \Gamma_N(d).
\end{equation}
L'isomorphisme ``multiplication par $d$'' de
$\frac{1}{d}\ZZ/\frac{1}{d}\ZZ(1)$ avec $\ZZ/N\ZZ(1)$ definit un
isomorphisme
\begin{equation}\label{eq:19.17.2}
\Gamma_N^1(d) \xrightarrow{\sim} \Gamma_N^1.
\end{equation}
La limite projective des $\Gamma_N(d)$ est le complete profini de
$\Gamma(d)$. Si on restreint $N$ a etre une puissance de $\ell$, on
obtient le $\ell$-complete
\[
\Gamma_\ell^\wedge(d) = \tfrac{1}{d}\ZZ_\ell(1) \ltimes
\Gamma_\ell^{1\wedge}(d).
\]

Pour obtenir $\Gamma_\ell^\wedge(A)$, il faut remplacer l'algebre de
groupe par son quotient par $(u-1)^A$, pour $u$ un generateur de
$\frac{1}{d}\ZZ_\ell(1)$:
\begin{align*}
\Gamma_\ell^\wedge(A) &= \tfrac{1}{d}\ZZ_\ell(1) \ltimes
\Gamma_\ell^{1\wedge}(d)(A) \\
\Gamma_\ell^{1\wedge}(d)(A) &= \ZZ_\ell[u]/((u-1)^A)(1).
\end{align*}

Parallelement a 19.15, on verifie que

\begin{proposition}[19.18]\label{prop:19.18}
Le morphisme $(16.22.2)_{d,k}$
\[
\hat{\ZZ}[\tfrac{1}{d}\hat{\ZZ}(1)](1) \to \hat{\ZZ}(k) \qquad (1 \leq k
\leq A)
\]
est la limite projective pour $N$ une puissance de $\ell$ des
\begin{equation}\label{eq:19.18.1}
\ZZ/N[\tfrac{1}{d}\ZZ/\tfrac{1}{d}\ZZ(1)](1) \to \ZZ/N(k) : \quad
\sum a_\zeta \delta_\zeta \longmapsto \sum a_\zeta \cdot (d\zeta)^{\otimes k-1}.
\end{equation}
\end{proposition}

\subsection{19.19}

Verifions 19.1(B) sur $\mu_d^{**}$. D'apres 19.18, le torseur
$P_{d,k}\, \ZZ/N$ sur $\mu_d^{**}$ sous $\ZZ/N(k)$ s'obtient en
restreignant $R_N$ a $\mu_d^{**}$, en poussant ce
$\Gamma_N$-torseur a $\Gamma_N(1)$ et en poussant par
\eqref{eq:19.18.1}$_{d,k}$. Detailons ces operations.

La fibre de $R_N$ en $\zeta \in \mu_d^{**}$ est l'ensemble des
systemes $(u, v(x))$ avec
\[
u^N = \zeta, \quad v(x)^N = 1-x \quad (x^N = \zeta).
\]
Le morphisme \eqref{eq:19.18.1}$_{d,k}$ est le compose de
\eqref{eq:19.15.1}$_k$ et de l'isomorphisme induit par
\eqref{eq:19.17.2} de $\Gamma_N^1(d)$ avec $\Gamma_N^1$. Poussons le
$\Gamma_N$-torseur $R_N$, par le compose ``$d$'' de
\eqref{eq:19.17.2} et \eqref{eq:19.17.1}: $\Gamma_N \to \Gamma_N$,
induit par la multiplication par $d$ dans $\ZZ/N(1)$. Le torseur
obtenu est celui des systemes $(u', v'(y))$ (ou $y$ parcourt
$\mu_N$) avec $u'^N = 1$ et $v'(y)^N = \prod(1-x)$ le produit etant
etendu aux racines $d$-iemes de $y$ telles que $x^N = \zeta$. On
definit en effet un ``$d$''-morphisme de $R_N$, dans ce torseur en
envoyant $(u, v(x)_{x^N=\zeta})$ sur $(u', v'(y)_{y^N=1})$ avec
$u' = u^d$ et $v'(y) = \prod_{z^d=y} v(x)$.

On restreint le groupe structural a $\Gamma_N^1$ en imposant
$u' = 1$. Le $\Gamma_N^1$-torseur obtenu est donc celui des systemes
$v(y)_{y^N=1}$ comme plus haut. Appliquant
\eqref{eq:19.15.1}$_k$, on obtient (3.11.2)$_{d,k}$. Comme
precedemment, $V_\Betti$ donne par l'isomorphisme de comparaison la
determination principale des racines $N$-ieme.
