\begin{center}
{\large Excerpt from \og Mathematica \& Paedagogia\fg\ no. 46}\\[2.4em]
{\Large\bfseries A CONGRUENCE BETWEEN MULTINOMIAL COEFFICIENTS}\\[1.0em]
by P. DELIGNE\\
{\itshape Institut des Hautes Études Scientifiques, Bures-sur-Yvette}
\end{center}

\medskip

Let $p$ be a prime number. For every integer $m\ne0$, let $v(m)$ denote the exponent of the highest power of $p$ which divides $m$. Put, moreover, $v(0)=\infty$.

For $\nn=(n_1,\ldots,n_k)$ a $k$-tuple of integers $\ge0$, put
\[
|\nn|=\sum_{i=1}^k n_i,\qquad
a\nn=(an_1,\ldots,an_k),\qquad
X^\nn=\prod_{i=1}^k X_i^{n_i},\qquad
\nn! = \prod_{i=1}^k n_i!,
\]
and denote by $C(\nn)$ the coefficient of the monomial $X^\nn$ in
\[
\left(\sum_{i=1}^k X_i\right)^{|\nn|}
\]
(the multinomial coefficients).

Thus, for every integer $n\ge0$,
\[
\left(\sum_{i=1}^k X_i\right)^n
=\sum_{|\nn|=n} C(\nn)X^\nn, \tag{1}
\]
\[
C(\nn)=\frac{|\nn|!}{\nn!}. \tag{2}
\]
For $k=2$ these are the binomial coefficients:
\[
C(n_1,n_2)=\binom{n_1+n_2}{n_1}. \tag{3}
\]

\paragraph{Proposition.} Let $\nn$ be a $k$-tuple of integers $\ge0$, let $n=|\nn|$, and let $l\ge1$. If $p$ is a prime number $\ge5$, then
\[
C(p^l\nn)\equiv C(\nn)\pmod {p^{v(n)+3}}. \tag{4}
\]

\paragraph{Example.} For $p=5$, $k=2$, $l=1$, and $\nn=(1,1)$, one has
\[
\binom{10}{5}=252=\binom21+2\cdot5^3.
\]

\paragraph{Proof.} It is known, or follows from (2), that
\[
\left(\sum_{i=1}^k X_i\right)^{p^l}
=
\left(\sum_{i=1}^k X_i^{p^l}\right)+pQ, \tag{5}
\]
where $Q$ is a polynomial with integral coefficients, of degree $<p^l$ in each variable. The binomial formula then gives
\[
\left(\sum_{i=1}^k X_i\right)^{p^l n}
=
\left(\sum_{i=1}^k X_i^{p^l}\right)^n
+\sum_{i=1}^n p^i\binom ni R_i, \tag{6}
\]
with
\[
R_i=
\left(\sum_{j=1}^k X_j^{p^l}\right)^{n-i}Q^i. \tag{7}
\]

For two polynomials $A$ and $B$ with integral coefficients, we shall write
\[
A\dd B \pmod r
\]
if the coefficients of $X^{p^l\nn}$ in $A$ and $B$ are congruent modulo $r$. With this notation, (4) is rewritten as
\[
\left(\sum_{i=1}^k X_i\right)^{p^l n}
\dd
\left(\sum_{i=1}^k X_i^{p^l}\right)^n
\pmod {p^{v(n)+3}}. \tag{8}
\]

\paragraph{Lemma 1.}
\[
v\binom ni \ge v(n)-v(i). \tag{9}
\]

Let the additive group $\Z/(n)$ act on itself by translation.\footnote{Thus to every $x\in\Z/(n)$ is associated the permutation of $\Z/(n)$ which sends $y$ to $x+y$.} If a subset $P$ with $i$ elements of $\Z/(n)$ has as stabilizer\footnote{If a group $G$ acts on a set $E$, the stabilizer of a subset $P$ of $E$ is the set of $g\in G$ for which $P$ is stable.} a subgroup $H$ of $\Z/(n)$, $H\cdot P=P$, then $P$ is a union of cosets of $H$, and $\#H$ divides $i$. The cardinal
\[
r_P=\frac{n}{\#H}
\]
of the orbit of $P$ under $\Z/(n)$, in $\mathcal P(\Z/(n))$, therefore satisfies
\[
p^{v(n)-v(i)}\mid r_P.
\]
Hence, since the set of subsets with $i$ elements of $\Z/(n)$ is a disjoint union of orbits,
\[
p^{v(n)-v(i)}\mid\binom ni.
\]

\paragraph{Lemma 2.} For $p\ne2,3$ and $i\ge3$, one has
\[
p^i\binom ni\equiv0\pmod {p^{v(n)+3}}. \tag{10}
\]
Indeed,
\[
v\left(p^i\binom ni\right)
=i+v\binom ni
\ge v(n)+(i-v(i))
\]
by Lemma 1. For $3\le i<p$ one has $v(i)=0$ and $i-v(i)\ge3$. For $i\ge p$, one has
\[
i-v(i)\ge i-\frac{\log i}{\log p}.
\]
This last function of $i$ is increasing for $i\ge p$, and is equal to $p-1\ge3$ for $i=p$.

\paragraph{Lemma 3.} The coefficient of $X^{p^l\nn}$ in $R_1$ is zero.

None of the monomials $X^\rr$ appearing in $Q$ has an exponent $\rr$ divisible by $p^l$. The polynomial
\[
R_1=\left(\sum_{j=1}^k X_j^{p^l}\right)^{n-1}Q
\]
inherits this property.

It follows from Lemmas 2 and 3 that
\[
\left(\sum_{i=1}^k X_i\right)^{p^l n}
\dd
\left(\sum_{i=1}^k X_i^{p^l}\right)^n
+\binom n2 p^2R_2
\pmod {p^{v(n)+3}}. \tag{11}
\]
This congruence already implies, by Lemma 1, that
\[
C(p^l\nn)\equiv C(\nn)\pmod {p^{v(n)+2}}. \tag{12}
\]

The only monomials $X^\rr$ with exponent divisible by $p^l$ which occur in $(pQ)^2$ are the monomials
\[
X_i^{p^l}X_j^{p^l}\qquad(i<j).
\]
The coefficient of each of them is
\[
\sum_{a\ne0,p^l}\binom{p^l}{a}\binom{p^l}{p^l-a}
=
\sum_a \binom{p^l}{a}^2-2
=
\binom{2p^l}{p^l}-2, \tag{13}
\]
and
\[
\binom{2p^l}{p^l}-2
\equiv
\binom{2p}{p}-2
\pmod {p^3}
\]
by (12).

Thus
\[
\left(\sum_{i=1}^k X_i\right)^{p^l n}
\dd
\left(\sum_{i=1}^k X_i^{p^l}\right)^n
+\binom n2\left[\binom{2p}{p}-2\right]R_2
\pmod {p^{v(n)+3}},
\]
with $R_2$ a polynomial with integral coefficients. Since
\[
v\binom n2\ge v(n)
\]
by Lemma 1, the proposition follows from the next lemma.

\paragraph{Lemma 4.} For $p$ prime $\ge5$, one has
\[
\binom{2p}{p}\equiv2\pmod {p^3}.
\]

One has
\[
p^{-1}\binom pi\equiv(-1)^{i-1}i^{-1}\pmod p
\]
for $1\le i\le p-1$, and hence
\[
p^{-2}\left[\binom{2p}{p}-2\right]
=
\sum_{i=1}^{p-1}p^{-2}\binom pi^2
\equiv
\sum_{i=1}^{p-1}\frac{1}{i^2}
\pmod p. \tag{14}
\]
Define $z\in\Z/(p)$ by
\[
z=\sum_{i\in(\Z/(p))^*}\frac1{i^2}.
\]
For $u\in(\Z/(p))^*$ one has
\[
z=\sum_{i\in(\Z/(p))^*}\frac{1}{(u^{-1}i)^2}=u^2z
\]
and
\[
(u^2-1)z=0.
\]
Since $p\ne2,3$, there exists $u$ such that $u^2\ne1$, and $z=0$. Formula (14) therefore simplifies to
\[
p^{-2}\left[\binom{2p}{p}-2\right]\equiv0\pmod p,
\]
which completes the proof.
