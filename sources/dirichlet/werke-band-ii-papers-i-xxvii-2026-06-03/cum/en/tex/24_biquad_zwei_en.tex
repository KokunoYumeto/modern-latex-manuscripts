{\LARGE \textbf{On the biquadratic character of the number ``two''}}\par\vspace{1em}

\end{center}

\bigskip
\begin{center}
{\large \textbf{On the biquadratic character of the number ``two''.}}
\end{center}

Let $p$ be a prime number of the form $4n+1$ and
\[
p=a^2+b^2,
\]
where $a$ is odd. From this equation and from the equation following from it,
\[
2p=(a+b)^2+(a-b)^2,
\]
one obtains at once, using Legendre's generalized symbol,
\[
\left(\frac{p}{a}\right)=1, \qquad \left(\frac{2p}{a+b}\right)=1
\]
or
\[
\left(\frac{p}{a+b}\right)=\left(\frac{2}{a+b}\right),
\]
and then, by known theorems,
\[
\left(\frac{a}{p}\right)=1, \qquad \left(\frac{a+b}{p}\right)=(-1)^{\frac18[(a+b)^2-1]},
\]
or, what is the same thing,
\[
a^{\frac12(p-1)}\equiv 1, \qquad (a+b)^{\frac12(p-1)}\equiv (-1)^{\frac18[(a+b)^2-1]} \quad (\mathrm{mod}.\,p).
\]
On the other hand, from
\[
(a+b)^2 \equiv a^2+b^2+2ab \equiv 2ab \quad (\mathrm{mod}.\,p)
\]
one obtains, by raising to the power $\frac14(p-1)$,
\[
(a+b)^{\frac12(p-1)}\equiv 2^{\frac14(p-1)}a^{\frac14(p-1)}b^{\frac14(p-1)} \quad (\mathrm{mod}.\,p),
\]
or, if one sets
\[
b\equiv af
\]
and takes account of the two last congruences,
\[
(-1)^{\frac18[(a+b)^2-1]}=(-1)^{\frac18(p-1+2ab)}\equiv 2^{\frac14(p-1)}f^{\frac14(p-1)} \quad (\mathrm{mod}.\,p).
\]
Now, since
\[
a^2+b^2=p, \qquad b^2\equiv a^2f^2,
\]
therefore
\[
f^2\equiv -1 \quad (\mathrm{mod}.\,p),
\]
and hence
\[
(f^2)^{\frac18(p-1+2ab)}=f^{\frac14(p-1)}\cdot f^{\frac12 ab}\equiv 2^{\frac14(p-1)}\cdot f^{\frac14(p-1)} \quad (\mathrm{mod}.\,p).
\]
Dividing by $f^{\frac14(p-1)}$ gives
\[
2^{\frac14(p-1)}\equiv f^{\frac12 ab} \quad (\mathrm{mod}.\,p),
\]
which is transformed into the result given by Gauss in § 24 of his \emph{theoria residuorum biquadraticorum} if one multiplies by $a^{\frac12ab}$ and again writes $b$ for $af$.

\begin{center}
Göttingen, 21 January 1857.
\end{center}
