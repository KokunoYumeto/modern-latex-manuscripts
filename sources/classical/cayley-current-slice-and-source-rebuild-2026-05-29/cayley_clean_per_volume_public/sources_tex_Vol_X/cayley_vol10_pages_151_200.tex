\section{[655] A Memoir on Differential Equations (conclusion)}
\label{sec:655}

\noindent\textit{[From the \emph{Philosophical Transactions}, vol.~167 (1877), pp.~243--273; continuation from preceding pages.]}

\medskip

\noindent\textbf{77.} More symmetrically, we have the solution
\[
V = \lambda - \tfrac{1}{2}(x^2 + y^2 + z^2) + \frac{(ax + by + cz)^2}{a^2 + b^2 + c^2},
\]
as can be at once verified.

\smallskip

\noindent\textbf{78.} In the same particular case $H = 0$, introducing the corresponding values $p_0$, $q_0$, $r_0$, $x_0$, $y_0$, $z_0$, we find a very simple expression for $V - V_0$, as a function of $x$, $y$, $z$, $x_0$, $y_0$, $z_0$.

We have, writing $T_0 = e^{t_0 - \tau}$, $T_0' = e^{-t_0 + \tau}$, and therefore $T_0 T_0' = 1$,
\begin{align*}
x_0 &= aT_0 + a' T_0', & p_0 &= aT_0 - a' T_0', \\
y_0 &= bT_0 + b' T_0', & q_0 &= bT_0 - b' T_0', \\
z_0 &= cT_0 + c' T_0', & r_0 &= cT_0 - c' T_0',
\end{align*}
and thence
\begin{align*}
x - x_0 &= a(T - T_0) + a'\Bigl(\frac{1}{T} - \frac{1}{T_0}\Bigr) = (T - T_0)\Bigl(a - \frac{a'}{TT_0}\Bigr), \\
x + x_0 &= a(T + T_0) + a'\Bigl(\frac{1}{T} + \frac{1}{T_0}\Bigr) = (T - T_0)\Bigl(a + \frac{a'}{TT_0}\Bigr).
\end{align*}
Forming the analogous quantities $y - y_0$, \&c., we deduce
\begin{align*}
(x - x_0)^2 + (y - y_0)^2 + (z - z_0)^2 &= (T - T_0)^2 \Bigl\{a^2 + b^2 + c^2 + (a'^2 + b'^2 + c'^2)\frac{1}{T^2 T_0^2}\Bigr\}, \\
(x + x_0)^2 + (y + y_0)^2 + (z + z_0)^2 &= (T + T_0)^2 \Bigl\{a^2 + b^2 + c^2 + (a'^2 + b'^2 + c'^2)\frac{1}{T^2 T_0^2}\Bigr\}.
\end{align*}
But we have
\begin{align*}
V - V_0 &= \tfrac{1}{2}\Bigl\{(a^2 + b^2 + c^2)(T^2 - T_0^2) - (a'^2 + b'^2 + c'^2)\Bigl(\frac{1}{T^2} - \frac{1}{T_0^2}\Bigr)\Bigr\} \\
&= \tfrac{1}{2}(T^2 - T_0^2)\Bigl\{a^2 + b^2 + c^2 + (a'^2 + b'^2 + c'^2)\frac{1}{T^2 T_0^2}\Bigr\},
\end{align*}
and hence the required formula
\[
V - V_0 = \tfrac{1}{2}\sqrt{\{(x - x_0)^2 + (y - y_0)^2 + (z - z_0)^2\}}\,\sqrt{\{(x + x_0)^2 + (y + y_0)^2 + (z + z_0)^2\}},
\]
or, say, for shortness,
\[
= \tfrac{1}{2}\sqrt{(R)}\sqrt{(S)}.
\]

\smallskip

\noindent\textbf{79.} We ought, therefore, to have
\[
\tfrac{1}{2}d\sqrt{(R)}\sqrt{(S)} = p\,dx + q\,dy + r\,dz - p_0\,dx_0 - q_0\,dy_0 - r_0\,dz_0,
\]
where $p$, $q$, $r$, $p_0$, $q_0$, $r_0$ denote as above, and consequently
\[
p^2 + q^2 + r^2 = x^2 + y^2 + z^2, \qquad p_0^{\,2} + q_0^{\,2} + r_0^{\,2} = x_0^{\,2} + y_0^{\,2} + z_0^{\,2}.
\]

We have in fact
\begin{align*}
p^2 + q^2 + r^2 &= \tfrac{1}{2}\{R + S + 2(x^2 + y^2 + z^2 - x_0^{\,2} - y_0^{\,2} - z_0^{\,2})\} = x^2 + y^2 + z^2, \\
p_0^{\,2} + q_0^{\,2} + r_0^{\,2} &= \tfrac{1}{2}\{R + S - 2(x^2 + y^2 + z^2 - x_0^{\,2} - y_0^{\,2} - z_0^{\,2})\} = x_0^{\,2} + y_0^{\,2} + z_0^{\,2},
\end{align*}
or the last-mentioned results are thus verified.

\medskip

\subsection*{Partial Differential Equation containing the Dependent Variable: Reduction to Standard Form}
\textit{Art.~Nos.~80, 81.}

\smallskip

\noindent\textbf{80.} The equation $H = \text{const.}$, is the most general form of a partial differential equation not containing the dependent variable $V$; but if a partial differential equation does contain the independent variable, we can, by regarding this as one of the dependent variables, and in place of it introducing a new independent variable, exhibit the equation in the standard form $H = \text{const.}$, $H$ being here a homogeneous function of the order zero in the differential coefficients.

Thus, if the independent variables are $x$, $y$, the dependent variable $z$, and its differential coefficients $p$, $q$, then the given partial differential equation may be
\[
H_1 = H(p, q, x, y, z) = \text{const.}
\]
But we may determine $V$ as a function of $x$, $y$ by an equation $V = \text{const.}$, $V$ being a desired function of $x$, $y$, $z$; and then writing $p$, $q$, $r$ for the differential coefficients $\frac{dV}{dx}$, $\frac{dV}{dy}$, $\frac{dV}{dz}$, we have $p = -\frac{r}{P}$, $q = -\frac{r}{Q}$, and the proposed partial differential equation becomes
\[
H\Bigl(-\frac{r}{p}, -\frac{r}{q}, x, y, z\Bigr) = \text{const.},
\]
viz. this is an equation containing only the differential coefficients $p$, $q$, $r$ of the dependent variable $V$, a function of $x$, $y$, $z$. And, moreover, $H$ is homogeneous of the order zero in $p$, $q$, $r$; consequently
\[
p\frac{dH}{dp} + q\frac{dH}{dq} + r\frac{dH}{dr} = 0,
\]
or, in the augmented Hamiltonian system, the last equality is $\frac{dV}{dt} = 0$, so that an integral is $V = \text{const.}$; as already stated, this is the equation by which $z$ is determined as a function of $x$, $y$.

\smallskip

\noindent\textbf{81.} Thus, if the given partial differential equation be $pq - z = H$, we here consider the equation $\frac{PQ}{R} - z = H$. The Hamiltonian system is
\[
\frac{dx}{\frac{P}{R}} = \frac{dy}{\frac{Q}{R}} = \frac{-dz}{\frac{PQ}{R^2}} = \frac{dp}{q - \frac{PQ}{R^2}} = \frac{dq}{0} = \frac{dr}{0} = \frac{dV}{0},
\]
having the integrals
\[
a = p, \quad b = q, \quad c = px - qy, \quad r = \frac{q}{pq}, \quad \frac{r}{p} = \frac{1}{pq}
\]
(whence $H = -abc$). We have $H$, $a$, $b$, a system of conjugate integrals and, in terms of these,
\[
p = a, \quad q = b, \quad r = \surd\{ab(z + H)\};
\]
hence, writing $\lambda$ for the constant value of $V$, we have
\[
\lambda = \int\Bigl(a\,dx + b\,dy + \surd\{ab(z + H)\}\,dz\Bigr),
\]
that is,
\[
\lambda = ax + by + 2\surd\{ab(z + H)\},
\]
or say,
\[
4ab(z + H) = (ax + by - \lambda)^2,
\]
a solution containing really the two constants $\lambda$ and $H$, and thus a complete solution of the given equation $pq - z = H$. We have, in fact,
\begin{align*}
2ab\,p &= a(ax + by - \lambda), \\
2ab\,q &= b(ax + by - \lambda),
\end{align*}
that is,
\[
4a^2b^2\,pq = ab(ax + by - \lambda)^2 = 4a^2b^2(z + H),
\]
or $pq = z + H$, as it should be.

\section{[656] On the Theory of Partial Differential Equations}
\label{sec:656}

\noindent\textit{[From the \emph{Mathematische Annalen}, t.~\textsc{xi}. (1877), pp.~194--198.]}

\medskip

In what follows, any letter not otherwise explained denotes a function of certain variables $(x, y, p, q)$, or $(x, y, z, p, q, r)$, \&c., as will be stated in each particular case.

An equation $a = \text{const.}$, denotes that the function $a$ of the variables is, in fact, a constant (viz.~by such equation we establish a relation between the variables): and when this is so, we use the same letter $a$ to denote the constant value of the function in question; I find this a very convenient notation.

Thus if the variables are $x$, $y$, $z$, $p$, $q$, $r$ and if $p$, $q$, $r$ are the differential coefficients in regard to $x$, $y$, $z$ respectively of a function $V$ of $x$, $y$, $z$, then $H$ (as a letter not otherwise explained) denotes a function of $x$, $y$, $z$, $p$, $q$, $r$ and considering it as a given function, $H = \text{const.}$ will be a partial differential equation containing the constant $H$.

For instance, if $H$ denote the function $pqr - xyz$, $H = \text{const.}$, is the partial differential equation, $pqr - xyz = H$ (a given constant).

The integration of the partial differential equation, $H = \text{const.}$, depends upon that of the linear partial differential equation
\[
(H, \Theta) = 0,
\]
where as usual $(H, \Theta)$ signifies
\[
\frac{\partial(H, \Theta)}{\partial(p, x)} + \frac{\partial(H, \Theta)}{\partial(q, y)} + \frac{\partial(H, \Theta)}{\partial(r, z)}.
\]
It can be effected if we know two conjugate solutions $a$, $b$ of the equation $(H, \Theta) = 0$, viz.~$a$, $b$ as solutions are such that $(H, a) = 0$, $(H, b) = 0$, and (as conjugate solutions) are also such that $(a, b) = 0$; in this case if from the equations $H = \text{const.}$, $a = \text{const.}$, $b = \text{const.}$, we determine $p$, $q$, $r$ as functions of $x$, $y$, $z$, the resulting value of $p\,dx + q\,dy + r\,dz$ is an exact differential, and we have
\[
V = \lambda + \int(p\,dx + q\,dy + r\,dz),
\]
a solution containing three arbitrary constants, $\lambda$, $a$, $b$, and therefore a complete solution of the proposed partial differential equation $H = \text{const.}$

But (as is known) there is a different process of integration, for which the conjugate solutions are not required, and which has reference to a system of initial values $x_0$, $y_0$, $z_0$, $p_0$, $q_0$, $r_0$: viz.~if the independent solutions of $(H, \Theta) = 0$, are $a$, $b$, $c$, $d$, $e$, and if $a_0$, $b_0$, $c_0$, $d_0$, $e_0$ denote respectively the same functions of the initial variables that $a$, $b$, $c$, $d$, $e$ are of $x$, $y$, $z$, $p$, $q$, $r$, then if from the equations
\[
a = a_0, \quad b = b_0, \quad c = c_0, \quad d = d_0, \quad e = e_0, \quad H = \text{const.}
\]
we express $p$, $q$, $r$ as functions of $x$, $y$, $z$ and of $x_0$, $y_0$, $z_0$, $H$, these last being regarded as constants, we have $p\,dx + q\,dy + r\,dz$ an exact differential, and
\[
V = \lambda + \int(p\,dx + q\,dy + r\,dz),
\]
a solution containing the constants $\lambda$, $x_0$, $y_0$, $z_0$ (that is, one supernumerary constant), and as such a complete solution.

It is interesting to prove directly that $p\,dx + q\,dy + r\,dz$ is an exact differential.

I consider first the more simple case where the variables are $p$, $q$, $x$, $y$. Here $p$, $q$ are to be found from the equations
\[
a = a_0, \quad b = b_0, \quad c = c_0, \quad H = \text{const.}
\]
and it is to be shown that $p\,dx + q\,dy$ is an exact differential.

Considering $p$, $q$, $p_0$, $q_0$ as functions of the independent variables $x$, $y$, then differentiating in regard to $x$, and eliminating $\frac{dp}{dx}$, $\frac{dp_0}{dx}$, $\frac{dq_0}{dx}$, we have
\[
\begin{vmatrix}
\dfrac{da}{dx} + \dfrac{da}{dq}\dfrac{dq}{dx}, & \dfrac{da}{dp}, & \dfrac{da_0}{dp_0}, & \dfrac{da_0}{dq_0} \\[6pt]
\dfrac{db}{dx} + \dfrac{db}{dq}\dfrac{dq}{dx}, & \dfrac{db}{dp}, & \dfrac{db_0}{dp_0}, & \dfrac{db_0}{dq_0} \\[6pt]
\dfrac{dc}{dx} + \dfrac{dc}{dq}\dfrac{dq}{dx}, & \dfrac{dc}{dp}, & \dfrac{dc_0}{dp_0}, & \dfrac{dc_0}{dq_0} \\[6pt]
\dfrac{dH}{dx} + \dfrac{dH}{dq}\dfrac{dq}{dx}, & \dfrac{dH}{dp}, & 0, & 0
\end{vmatrix} = 0,
\]
or introducing a well-known notation for functional determinants, and expanding the determinant, this is
\[
\frac{\partial(a_0, b_0)}{\partial(p_0, q_0)} \cdot \frac{\partial(H, c)}{\partial(p, q)} \cdot \frac{dq}{dx} + \cdots = 0.
\]
But in the same way
\[
\frac{\partial(p_0, q_0)}{\partial(p_0, q_0)} \cdot \frac{\partial(H, c)}{\partial(q, y)} \cdot \frac{dq}{dy} + \cdots = 0,
\]
or adding these, attending to the value of $(H, c)$, and observing that
\[
\frac{\partial(a_0, b_0)}{\partial(p_0, q_0)} \cdot \frac{\partial(H, c)}{\partial(p, q)} \Bigl(\frac{dq}{dx} - \frac{dp}{dy}\Bigr) + \cdots = 0.
\]
We have $(H, a) = 0$, $(H, b) = 0$, $(H, c) = 0$, and the equation in fact is
\[
\frac{\partial(a_0, b_0)}{\partial(p, q)} \cdot \frac{\partial(H, c)}{\partial(p, q)} \Bigl(\frac{dq}{dx} - \frac{dp}{dy}\Bigr) = 0,
\]
viz.~we have $\frac{dq}{dx} - \frac{dp}{dy} = 0$, the condition for the exact differential.

Coming now to the case where the variables are $x$, $y$, $z$, $p$, $q$, $r$, and in the six equations treating $p$, $q$, $r$, $p_0$, $q_0$, $r_0$ as functions of the independent variables $x$, $y$, $z$,---then differentiating with regard to $x$ and proceeding as before, we find for $\frac{dr}{dx}$ the equation
\[
\frac{\partial(c_0, d_0, e_0)}{\partial(p_0, q_0, r_0)} \Bigl(\frac{dr}{dx} - \frac{dp}{dz}\Bigr) \frac{\partial(a, b, H)}{\partial(r, p, q)} \frac{\partial(a, b, H)}{\partial(x, p, q)} + \cdots = 0.
\]
We have, in the same way, for $\frac{dp}{dy}$ the equation
\[
\frac{\partial(c_0, d_0, e_0)}{\partial(p_0, q_0, r_0)} \Bigl(\frac{dp}{dy} - \frac{dq}{dx}\Bigr) \frac{\partial(a, b, H)}{\partial(p, r, q)} \frac{\partial(a, b, H)}{\partial(z, r, q)} + \cdots = 0,
\]
or, adding the two equations,
\[
\frac{\partial(c_0, d_0, e_0)}{\partial(p_0, q_0, r_0)} \Bigl\{\Bigl(\frac{dr}{dx} - \frac{dp}{dz}\Bigr) \frac{\partial(a, b, H)}{\partial(r, p, q)} \frac{\partial(a, b, H)}{\partial(x, p, q)} + \Bigl(\frac{dp}{dy} - \frac{dq}{dx}\Bigr) \frac{\partial(a, b, H)}{\partial(z, r, q)}\Bigr\} + \cdots = 0,
\]
where the terms denoted by the \&c.~indicate the like terms corresponding to the different partitions of the letters $a$, $b$, $c$, $d$, $e$.

The equation may be simplified; we have identically
\[
\frac{\partial(a_0, b_0, H)}{\partial(q_0, \cdot, \cdot)} \frac{\partial(a, b, H)}{\partial(x, p, q)} + \frac{\partial(a, b, H)}{\partial(z, r, q)} + \cdots = 0,
\]
or since $(H, a) = 0$, $(b, H) = 0$, the left-hand side is simply $-\frac{\partial(a, b)}{\partial(\cdot, \cdot)}$, and the equation becomes
\[
\frac{\partial(c_0, d_0, e_0)}{\partial(p_0, q_0, r_0)} \Bigl\{\Bigl(\frac{dr}{dx} - \frac{dp}{dz}\Bigr) \frac{\partial(a, b, H)}{\partial(r, p, q)} + \cdots\Bigr\} = 0.
\]
This ought to give $\frac{dr}{dx} - \frac{dp}{dz} = 0$, \&c.~for each $(a, b, c, d, e)$; this is then the equation which has to be proved.

By the Poisson-Jacobi theorem, $(a, b)$ is a function of $a$, $b$, $c$, $d$, $e$: if we write
\[
f(\cdot) = \frac{\partial(a_0, b_0)}{\partial(\cdot)} \frac{\partial(a_0, b_0)}{\partial(\cdot)},
\]
then $(a_0, b_0)$ is the same function of $a_0$, $b_0$, $c_0$, $d_0$, $e_0$; but these are $= a$, $b$, $c$, $d$, $e$ respectively, and we thence have $(a, b) = (a_0, b_0)$, and the theorem to be proved is
\[
\frac{\partial(c_0, d_0, e_0)}{\partial(p_0, q_0, r_0)} \cdot \frac{\partial(a, b)}{\partial(\cdot)} = 0.
\]
But substituting for $(a_0, b_0)$ its value, the function on the left-hand is (it is easy to see) the sum of the three functional determinants
\[
\frac{\partial(a_0, b_0, c_0, d_0, e_0)}{\partial(p_0, q_0, r_0, \cdot, \cdot)}, \quad
\frac{\partial(a_0, b_0, c_0, d_0, e_0)}{\partial(p_0, q_0, \cdot, r_0, \cdot)}, \quad
\frac{\partial(a_0, b_0, c_0, d_0, e_0)}{\partial(p_0, q_0, \cdot, \cdot, r_0)},
\]
each of which vanishes as containing the same letter twice in the denominator, that is, as having two identical columns; and the theorem in question is thus proved.

And in the same way $\frac{dr}{dy} - \frac{dq}{dz}$, $\frac{dp}{dz} - \frac{dr}{dx}$ are each $= 0$: or we have $p\,dx + q\,dy + r\,dz$ an exact differential.

The proof would fail if the factors multiplying $\frac{dq}{dx} - \frac{dp}{dy}$ \&c., or if any one of these factors, were $= 0$; I have not particularly examined this, but the meaning would be, that here the equations in question $a = a_0$, \&c., $H = \text{const.}$, are such as not to give rise to expressions for $p$, $q$, $r$ as functions of $x$, $y$, $z$, $x_0$, $y_0$, $z_0$, $H$, as assumed in the theorem; whenever such expressions are obtainable, then we have $p\,dx + q\,dy + r\,dz$ an exact differential.

The proof in the case of a greater number of variables, say in the next case where the variables are $x$, $y$, $z$, $w$, $p$, $q$, $r$, $s$, would present more difficulty---but I have not proceeded further in the question.

It is worth while to put the two processes into connexion with each other: taking in each case the variables to be $x$, $y$, $z$, $p$, $q$, $r$, and the partial differential equation to be $H = \text{const.}$

In the one case, $a$, $b$ being conjugate solutions of $(H, 0) = 0$, from the equations $H = \text{const.}$, $a = \text{const.}$, $b = \text{const.}$, we find $p$, $q$, $r$ functions of $x$, $y$, $z$, $H$, $a$, $b$ and then $p\,dx + q\,dy + r\,dz$ is an exact differential.

In the other case, $a$, $b$, $c$, $d$, $e$ being the solutions of $(H, 0) = 0$, from the equations $H = \text{const.}$, $a = a_0$, $b = b_0$, $c = c_0$, $d = d_0$, $e = e_0$, we find $p$, $q$, $r$ functions of $x_0$, $y_0$, $z_0$, $H$ and then $p\,dx + q\,dy + r\,dz$ is an exact differential.

It may be added that, if from the last mentioned equations we determine also $p_0$, $q_0$, $r_0$ as functions of $x$, $y$, $z$, $x_0$, $y_0$, $z_0$, then considering only $H$ as a constant, we ought to have $p\,dx + q\,dy + r\,dz - p_0\,dx_0 - q_0\,dy_0 - r_0\,dz_0$ an exact differential; I have not examined the direct proof.

\section{[657] Note on the Theory of Elliptic Integrals}
\label{sec:657}

\noindent\textit{[From the \emph{Mathematische Annalen}, t.~\textsc{xii}. (1877), pp.~143--146.]}

\medskip

The equation
\[
\frac{M\,dy}{\sqrt{1 - y^2 \cdot 1 - k^2 y^2}} = \frac{dx}{\sqrt{1 - x^2 \cdot 1 - k^2 x^2}}
\]
is integrable algebraically when $M$ is rational: and so long as the modulus is arbitrary, then conversely, in order that the equation may be integrable algebraically, $M$ must be rational. For particular values however of the modulus, the equation is integrable algebraically for values of the form $M$, or (what is the same thing) $\frac{1}{M}$, $=$ a rational quantity $\pm$ square root of a negative rational quantity, say $= \frac{1}{p}(l + m\surd -n)$, where $l$, $m$, $n$, $p$ are integral and $n$ is positive; we may for shortness call this a \emph{half-rational numerical value}.

The theory is considered by Abel in two Memoirs in the \emph{Astr.~Nach.} Nos.~138 \& 147 (1828), being the Memoirs* XIII \& XIV in the \emph{OEuvres Compl\`etes} (Christiania 1839). I here reproduce the investigation in a somewhat altered (and, as it appears to me, improved) form.

Putting the two differentials each $= du$, we have $x = \sn(u + \alpha)$, $y = \sn\bigl(\frac{u}{M} + \beta\bigr)$; and the question is whether there exists an algebraical relation between these functions, or, what is the same thing, an algebraical relation between the functions $x = \sn u$ and $y = \sn\frac{u}{M}$.

Suppose that $A$ and $B$ are independent periods of $\sn u$; so that $\sn(u + A) = \sn u$, $\sn(u + B) = \sn u$, and that every other period is $= mA + nB$, where $m$ and $n$ are integers. Then if $u$ has successively the values $u$, $u + A$, $u + 2A$, etc., the value of $x$ remains always the same, and if $x$ and $y$ are algebraically connected, $y$ can have only a finite number of values: there are consequently integer values $p'$, $p''$ for which
\[
\sn\tfrac{1}{M}(u + p'A) = \sn\tfrac{1}{M}(u + p''A):
\]
or writing $v$, $-p'A$ for $u$ and putting $p'' - p' = p$, there is an integer value $p$ for which $\sn\frac{1}{M}(u + pA) = \sn\frac{u}{M}$.

Similarly there is an integer value $q$ for which $\sn\frac{1}{M}(u + qB) = \sn\frac{u}{M}$; and we are at liberty to assume $q = p$; for if the original values are unequal, we have only in the place of each of them to substitute their least common multiple.

We have thus an integer $p$, for which $\sn\frac{1}{M}(u + pA) = \sn\frac{u}{M}$, $\sn\frac{1}{M}(u + pB) = \sn\frac{u}{M}$. There are consequently integers $m$, $n$, $r$, $s$ such that
\[
\frac{pA}{M} = mA + nB, \qquad \frac{pB}{M} = rA + sB,
\]
equations which will constitute a single relation $\frac{p}{M} = m$, if $m = s$, $r = n = 0$; but in every other case will be two independent relations.

In the case first referred to, the modulus is arbitrary and $M$ is rational. But excluding this case, the equations give
\[
B(mA + nB) = A(rA + sB),
\]
or, what is the same thing,
\[
rA^2 - (m - s)AB - nB^2 = 0,
\]
an equation which implies that the modulus has some one value out of a set of given values.

The ratio $A : B$ of the two periods is of necessity imaginary, and hence the integers $m$, $n$, $r$, $s$ must be such that $(m - s)^2 + 4nr$ is negative.

The foregoing equations may be written
\[
\frac{A}{B} = \frac{n}{\frac{1}{2}(m - s) - \sqrt{\frac{1}{4}(m - s)^2 + nr}},
\]
whence the ratio $A : B$ is given as a function of $m$, $n$, $r$, $s$; and if these have any values such that $(m - s)^2 + 4nr$ is negative, then there exists a corresponding modulus for which the original differential equation is integrable algebraically for a half-rational value of $M$.

The question arises, whether for any such modulus the equation is integrable algebraically for any value of $M$ other than the half-rational values; or say, whether the modulus being given, the differential equation is integrable algebraically for values of $M$ other than the half-rational value or values which belong to the given modulus.

We have $x = \sn(u + \alpha)$; hence if as before $A$, $B$ are the periods, the general value of $u$ is $= \{\am(x)\} + mA + nB$; and we thence have
\[
\frac{u}{M} = \frac{\am(x)}{M} + \frac{mA + nB}{M}.
\]
But
\[
\frac{mA + nB}{M} = m'A + n'B,
\]
where $m'$, $n'$ are proper integer values depending on $m$, $n$; in fact this equation gives $mA + nB = M(m'A + n'B)$; and if we write herein $A$, $B = pA + qB$, $rA + sB$ for $MA$, $MB$, the resulting equations must be satisfied identically; that is, we must have
\[
\begin{vmatrix} m - p & -q \\ -r & n - s \end{vmatrix} = 0,
\]
or $ps - (m - s)p - qr = 0$; and the values of $p$, $q$, $r$, $s$ which satisfy this equation are
\[
p = m, \quad q = \frac{n}{r}(m - s), \quad r = \text{anything}, \quad s = \text{anything}.
\]
And then we have $y = \sn\bigl(\frac{\am(x)}{M} + \beta\bigr)$. But $m'$ and $n'$ are to be integers for all integer values of $m$ and $n$; and we thence have $M = m$, a rational value. But as already mentioned, this is the case where the modulus is arbitrary.

The conclusion thus is, that if the modulus is not arbitrary but has a value such that the differential equation is integrable algebraically for a half-rational value of $M$, then for no other value of $M$ is the equation integrable algebraically.

\section{[658] On Some Formul\ae\ in Elliptic Integrals}
\label{sec:658}

\noindent\textit{[From the \emph{Mathematische Annalen}, t.~\textsc{xii}. (1877), pp.~369--374.]}

\medskip

I reproduce in a modified form an investigation contained in the memoir, Zolotareff, ``Sur la m\'ethode d'int\'egration de M.~Tch\'ebychef,'' \emph{Mathematische Annalen}, t.~\textsc{v}. (1872), pp.~560--580.

Starting from the quartic
\[
(a, b, c, d, e)(x, 1)^4 = a \cdot x - \alpha \cdot x - \beta \cdot x - \gamma \cdot x - \delta,
\]
we derive from it the quartic
\[
(a_1, b_1, c_1, d_1, e_1)(x_1, 1)^4 = a_1 \cdot x_1 - \alpha_1 \cdot x_1 - \beta_1 \cdot x_1 - \gamma_1 \cdot x_1 - \delta_1,
\]
where, writing for shortness
\begin{align*}
\lambda &= -\alpha + \beta + \gamma - \delta, \\
\mu &= \phantom{-}\alpha - \beta + \gamma - \delta, \\
\nu &= \phantom{-}\alpha + \beta - \gamma - \delta,
\end{align*}
the roots of the new quartic are
\begin{align*}
\alpha_1 &= \theta + \frac{\mu\nu}{2\lambda}, \\
\beta_1 &= \theta + \frac{\nu\lambda}{2\mu}, \\
\gamma_1 &= \theta + \frac{\lambda\mu}{2\nu}, \\
\delta_1 &= \theta,
\end{align*}
$\theta$ being arbitrary: the differences of the roots $\alpha_1$, $\beta_1$, $\gamma_1$, $\delta_1$ are, it will be observed, functions of the differences of the roots $\alpha$, $\beta$, $\gamma$, $\delta$.

We assume
\[
\frac{dx}{\sqrt{(a, b, c, d, e)(x, 1)^4}} = \frac{dx_1}{\sqrt{(a_1, b_1, c_1, d_1, e_1)(x_1, 1)^4}}.
\]
And consequently
\[
(a_1, b_1, c_1, d_1, e_1)(x_1, 1)^4 = (1, -B, -2C, -D, 0)(x_1, -1)^4.
\]
Hence also
\[
x_1 = \frac{A(x_1 - 1)}{x_1 - 1}, \quad \text{or} \quad x_1 = \frac{A + x}{1 + x}.
\]

We have
\[
\frac{dx}{\surd X} = \frac{dx_1}{\surd X_1},
\]
where
\begin{align*}
X &= (a, b, c, d, e)(x, 1)^4, \\
X_1 &= (a_1, b_1, c_1, d_1, e_1)(x_1, 1)^4.
\end{align*}

But the formulae may be presented in a more convenient form. Writing
\[
\alpha = \alpha, \quad \beta = \alpha + \beta, \quad \gamma = \alpha + \beta + \gamma, \quad \delta = \alpha + \beta + \gamma + \delta,
\]
we have
\[
\lambda = \gamma - \alpha, \quad \mu = \delta - \beta, \quad \nu = \alpha + \delta - \beta - \gamma.
\]

In particular, if we write $\alpha$, $\beta$, $\gamma$, $\delta = a$, $b$, $c$, $d$, the formulae become
\begin{align*}
\lambda &= c - a, \\
\mu &= d - b, \\
\nu &= a + d - b - c.
\end{align*}

The differences of $\alpha_1$, $\beta_1$, $\gamma_1$, $\delta_1$ are proportional to those of $\alpha$, $\beta$, $\gamma$, $\delta$, and we thus obtain the transformation of the elliptic differential in a very convenient form.

\section{[659] A Theorem on Groups}
\label{sec:659}

\noindent\textit{[From the \emph{Mathematische Annalen}, t.~\textsc{xiii}. (1878), pp.~561--565.]}

\medskip

The following theorem is very simple; but it seems to belong to a class of theorems, the investigation of which is desirable.

I consider a substitution-group of a given order upon a given number of letters; and I seek to \emph{double} the group, that is to derive from it a group of twice the order upon twice the number of letters. This can be effected for \emph{any} group, in a manner which is self-evident and in nowise interesting: but in a different manner for a \emph{commutative} group (or group such that any two of its substitutions satisfy the condition $AB = BA$): it is to be observed that the double group is not in general commutative.

Let the letters of the original group be $abcde\ldots$, we may for shortness write $U = abcde\ldots$; and take $U$ as the primitive arrangement: and let the group then be $1$, $A$, $B$,\ldots\ where $A$, $B$,\ldots\ represent substitutions: the corresponding arrangements are $U$, $AU$, $BU$,\ldots\ and these may for shortness be represented by $1$, $A$, $B$,\ldots; viz.~$1$, $A$, $B$,\ldots\ represent, properly and in the first instance, substitutions; but when it is explained that they represent arrangements, then they represent the arrangements $U$, $AU$, $BU$,\ldots

For the double group the letters are taken to be $a_1 b_1 c_1 d_1 e_1 \ldots$ and $a_2 b_2 c_2 d_2 e_2 \ldots$, $= U_1$ and $U_2$ suppose, and $U_1 U_2$ is regarded as the primitive arrangement; $A_1$ and $A_2$ denote the same substitutions in regard to $U_1$ and $U_2$ respectively, that $A$ denotes in regard to $U$: and so for $B_1$, $B_2$, etc.; moreover $12$ denotes the substitution $(a_1 a_2)(b_1 b_2)(c_1 c_2)(d_1 d_2)(e_1 e_2)\ldots$, or interchange of the suffixes $1$ and $2$. The substitutions $A_1$, $A_2$, or any powers of these $A_1^\alpha$, $A_2^\beta$, are obviously commutative; applying them to the primitive arrangement $U_1 U_2$, we have $A_1^\alpha A_2^\beta U_1 U_2$ and $A_2^\beta A_1^\alpha U_1 U_2$ each $= A_1^\alpha U_1 A_2^\beta U_2$. But $A_1^\alpha$, $A_2^\beta$ are not commutative with $12$: we have for instance $12 A_1^\alpha \cdot U_1 U_2 = 12 A_1^\alpha U_1 \cdot U_2 = A_1^\alpha U_2 \cdot U_1$, but $A_1^\alpha 12 U_1 U_2 = A_1^\alpha \cdot U_2 U_1 = U_2 \cdot A_1^\alpha U_1$. If instead of the substitutions we consider the arrangements obtained by operating upon $U_1 U_2$, then we have $A_1^\alpha A_2^\beta U_1 U_2 = A_1^\alpha U_1 \cdot A_2^\beta U_2$, and $12 \cdot A_1^\alpha U_1 \cdot A_2^\beta U_2 = A_1^\alpha U_2 \cdot A_2^\beta U_1 = A_2^\beta U_1 \cdot A_1^\alpha U_2$.

The substitutions of the double group are $A_1^\alpha A_2^\beta$, $A_1^\alpha A_2^\beta \cdot 12$; the arrangements are $A_1^\alpha U_1 \cdot A_2^\beta U_2$, $A_1^\alpha U_2 \cdot A_2^\beta U_1$, viz.~these last may be represented by $A_1^\alpha A_2^\beta$ and $A_1^\alpha A_2^\beta \cdot 12$.

We may write
\begin{align*}
(A_1^\alpha B_2^\beta) \cdot (A_1^{\alpha'} B_2^{\beta'}) &= A_1^{\alpha + \alpha'} B_2^{\beta + \beta'}, \\
(A_1^\alpha B_2^\beta) \cdot (A_1^{\alpha'} B_2^{\beta'} \cdot 12) &= A_1^{\alpha + \alpha'} B_2^{\beta + \beta'} \cdot 12, \\
(A_1^\alpha B_2^\beta \cdot 12) \cdot (A_1^{\alpha'} B_2^{\beta'}) &= A_1^{\alpha + \alpha'} B_2^{\beta + \beta'} \cdot 12, \\
(A_1^\alpha B_2^\beta \cdot 12) \cdot (A_1^{\alpha'} B_2^{\beta'} \cdot 12) &= A_1^{\alpha + \alpha'} B_2^{\beta + \beta'},
\end{align*}
where in each case the order of the factors is the right order. But the same formula may be written in the opposite order; and we then have, for instance, the last equation
\[
A_1^{\alpha'} B_2^{\beta'} \cdot 12 \cdot A_1^\alpha B_2^\beta \cdot 12 = A_2^{\beta'} B_1^{\alpha'} \cdot A_2^\beta B_1^\alpha = A_1^\alpha B_2^\beta \cdot A_1^{\alpha'} B_2^{\beta'},
\]
which is right if $AB = BA$, but not otherwise.

The theorem is as follows: \emph{if $A$, $B$,\ldots\ are the substitutions of a commutative group, then the foregoing expressions $A_1^\alpha B_2^\beta$, $A_1^\alpha B_2^\beta \cdot 12$ (viz.~for the double group, these are the arrangements, and, substituting for them the corresponding substitutions, the substitutions of the double group) form a group.}

This is proved by showing that the product of any two substitutions of the double group is always a substitution of the double group. The proof is immediate; for instance, from the foregoing equations, the product of any two of the double-group substitutions is equal to one or other of the forms $A_1^\alpha B_2^\beta$, $A_1^\alpha B_2^\beta \cdot 12$, that is, to a substitution of the double group. And moreover, if the original group is $1$, $A$, $B$,\ldots\ ($\mu$ substitutions in all), the double group is $A_1^\alpha B_2^\beta$, $A_1^\alpha B_2^\beta \cdot 12$ ($2\mu^2$ substitutions in all).

It is to be observed, in regard to the double group, that if the original substitutions are $1$, $A$, $B$,\ldots, then writing down the corresponding arrangements $1$, $A$, $B$,\ldots\ for the first set, and the same arrangements $1$, $A$, $B$,\ldots\ for the second set, we have in all the $2\mu^2$ arrangements; and that the substitutions of the double group are those which change any one of these arrangements into any other of them.

\medskip

\noindent\textit{Example.}---For the group $1$, $A$, $A^2$ ($A^3 = 1$), the double group is $A_1^\alpha A_2^\beta$ ($\alpha, \beta = 0, 1, 2$), and $A_1^\alpha A_2^\beta \cdot 12$ ($\alpha, \beta = 0, 1, 2$): $18$ substitutions in all. And similarly, for the group $1$, $A$ ($A^2 = 1$), the double group is $1$, $A_1$, $A_2$, $A_1 A_2$, $12$, $A_1 \cdot 12$, $A_2 \cdot 12$, $A_1 A_2 \cdot 12$: $8$ substitutions in all.

\section{[660] On the Correspondence of Homographies and Rotations}
\label{sec:660}

\noindent\textit{[From the \emph{Mathematische Annalen}, t.~\textsc{xv}. (1879), pp.~238--240.]}

\medskip

It is a fundamental notion in Prof.~Klein's theory of the ``Icosahedron'' that homographies correspond to rotations (of a solid body about a fixed point): in such wise that, considering the homographies which correspond to two given rotations, the homography compounded of these corresponds to the rotation compounded of the two rotations.

Say the two homographies are $A + Bp + Cq + Dpq = 0$, $A_1 + B_1 q + C_1 r + D_1 qr = 0$, then, eliminating $q$, the compound homography is $A_2 + B_2 p + C_2 r + D_2 pr = 0$, where
\[
A_2, B_2, C_2, D_2 = B_1 A - A_1 C, \; B_1 B - A_1 D, \; D_1 A - C_1 C, \; D_1 B - C_1 D;
\]
and the theorem is that, corresponding to these, we have rotations depending on the parameters $(\lambda, \mu, \nu)$, $(\lambda_1, \mu_1, \nu_1)$, $(\lambda_2, \mu_2, \nu_2)$ respectively, such that the third rotation is that compounded of the first and second rotations. The question arises to find the expression for the parameters of the homography in terms of the parameters of the corresponding rotation.

The rotation $(\lambda, \mu, \nu)$ is taken to denote a rotation through an angle $\vartheta$ about an axis the inclinations of which to the axes of coordinates are $f$, $g$, $h$, the values of $\lambda$, $\mu$, $\nu$ then being $= \tan\frac{1}{2}\vartheta \cos f$, $\tan\frac{1}{2}\vartheta \cos g$, $\tan\frac{1}{2}\vartheta \cos h$ respectively: $(\lambda_1, \mu_1, \nu_1)$ and $(\lambda_2, \mu_2, \nu_2)$ have of course the like significations; and then, if $(\lambda, \mu, \nu)$ refer to the first rotation, and $(\lambda_1, \mu_1, \nu_1)$ to the second rotation, the values of $(\lambda_2, \mu_2, \nu_2)$ for the rotation compounded of these are taken to be*:

\mu_2 &= \mu + \mu_1 + \nu\lambda_1 - \nu_1\lambda, \\
\nu_2 &= \nu + \nu_1 + \lambda\mu_1 - \lambda_1\mu,
\end{align*}
each divided by $1 - \lambda\lambda_1 - \mu\mu_1 - \nu\nu_1$.

\section{[661] On the Double $\vartheta$-Functions}
\label{sec:661}

\noindent\textit{[From the \emph{Proceedings of the London Mathematical Society}, vol.~\textsc{ix}. (1878), pp.~197--199.]}

\medskip

I consider the double $\te$-functions in a form slightly different from that of G\"{o}pel and Rosenhain: the sixteen functions are represented by
\[
\te\begin{pmatrix} a \\ b \end{pmatrix}(x, y \mid \omega_1, \omega_2, \omega_3),
\]
where $a$, $b$ each denote $0$ or $\frac{1}{2}$; and where $\omega_1$, $\omega_2$, $\omega_3$ are parameters connected by a relation $\Omega = 0$ (so that only two of them are arbitrary): I write also $a'$, $b'$ to denote each of them $0$ or $\frac{1}{2}$, independently of $a$, $b$.

The functions satisfy equations
\[
\te\begin{pmatrix} a \\ b \end{pmatrix}(x + m\omega_1 + n\omega_3, y + n\omega_2 + m\omega_3) = e^{2\pi i(\cdots)} \te\begin{pmatrix} a \\ b \end{pmatrix}(x, y),
\]
where $m$, $n$ are integers.

The theory may be developed very elegantly by introducing in place of $x$, $y$ the variables $u$, $v$, where
\[
x = u + v, \qquad y = u - v.
\]
We have then formulae of the form
\[
\te\begin{pmatrix} 0 \\ 0 \end{pmatrix}(u + v, u - v) = \te(u + v) \te(u - v) + \te'(u + v) \te'(u - v),
\]
and the quarter-period additions give rise to a system of linear equations connecting the sixteen functions.

The formulae may be presented in a more convenient form by the introduction of the variables $p$, $q$, $r$, $s$, connected by the equation $pq = rs$; and we then obtain the theory of the double $\te$-functions in connexion with that of Kummer's $16$-nodal quartic surface.

\begin{align*}
A &= [bc] = (b - c)[x\alpha' \cdot abc], \\
B &= [ca] = (c - a)[x\beta' \cdot bcd], \\
C &= [ab] = (a - b)[x\gamma' \cdot cde], \\
D &= [ad] = (a - d)[x\delta' \cdot def], \\
E &= [ae] = (a - e)[x\epsilon' \cdot efa], \\
F &= [af] = (a - f)[x\zeta' \cdot fab],
\end{align*}
and similarly
\begin{align*}
G &= [de] = (d - e)[y\alpha' \cdot abc], \\

\noindent\textbf{4.} We have moreover
\begin{align*}
[ab] - [cd] &= \frac{1}{(x - x')^2} \begin{cases}
\phantom{-}abf \cdot c'd'e' + a'b'f' \cdot cde \\
- cdf \cdot a'b'e' - c'd'f' \cdot abe
\end{cases} \\
&= \frac{1}{(x - x')^2}(abc'd' - a'b'cd)(e'f - ef'),
\end{align*}
where for the moment $a$, $b$, $a'$, etc., are written to denote $a - x$, $b - x$, $a - x'$, etc.; we have then
\begin{align*}
e'f - ef' &= (e - x')(f - x) - (e - x)(f - x') \\
&= -(e - f)(x - x') = -ef(x - x'),
\end{align*}
and
\begin{align*}
abc'd' - a'b'cd &= (a - x)(b - x)(c - x')(d - x') - (a - x')(b - x')(c - x)(d - x) \\
&= -(x - x') \begin{vmatrix} 1, & x + x', & xx' \\ 1, & a + b, & ab \\ 1, & c + d, & cd \end{vmatrix} \\
&= -(x - x')[xx' \cdot abcd].
\end{align*}
Hence $[ab] - [cd] = ef[xx' \cdot abcd]$, and the equation to be verified becomes
\[
(ef \cdot cd - cd \cdot ef)[xx' \cdot abcd] = 0,
\]
viz.~this is, in fact, an identity.

\smallskip

It is to be noticed that
\begin{align*}
\delta^2 + \alpha^2 - \beta^2 - \gamma^2, &\qquad 2(\alpha\beta - \gamma\delta), &\qquad 2(\gamma\alpha + \beta\delta), \\
2(\alpha\beta + \gamma\delta), &\qquad \delta^2 + \beta^2 - \gamma^2 - \alpha^2, &\qquad 2(\beta\gamma - \alpha\delta), \\
2(\gamma\alpha - \beta\delta), &\qquad 2(\beta\gamma + \alpha\delta), &\qquad \delta^2 + \gamma^2 - \alpha^2 - \beta^2,
\end{align*}
each divided by $\delta^2 + \alpha^2 + \beta^2 + \gamma^2$, form a system of coefficients in the transformation between two sets of rectangular coordinates. We have therefore
\begin{align*}
\surd(ab, \quad \surd(ad, \quad \surd(ce, \\
\surd(be, \quad \surd(de, \quad \surd(ac, \\
\surd(bc, \quad \surd(cd, \quad \surd(ae,
\end{align*}
each divided by $\surd(bd)$ and the several terms taken with proper signs, as a system of coefficients in the transformation between two sets of rectangular axes: a result which seems to be the same as that obtained by Hesse in the Memoir, ``Transformations-Formeln f\"ur rechtwinklige Raum-Co\"ordinaten''; \emph{Crelle}, t.~\textsc{lxiii}. (1864), pp.~247--251.

The composition of the last mentioned system of functions is better seen by writing them under the fuller form $\surd(abf \cdot cde)$, etc.; viz.~omitting the radical signs, the terms are
\begin{align*}
& abf \cdot cde, \quad adf \cdot bce, \quad abd \cdot cef, \\
& bcf \cdot acd, \quad def \cdot abc, \quad acf \cdot bde, \\
& bce \cdot ade, \quad cdf \cdot abe, \quad aef \cdot bcd,
\end{align*}
each divided by $bdf \cdot ace$; or, in an easily understood algorithm, the terms are
\[
\begin{array}{c|ccc}
& bf.d & df.b & bd.f \\ \hline
a.ce & bf.d & df.b & bd.f \\
e.ac & bf.d & df.b & bd.f \\
c.ae & bf.d & df.b & bd.f,
\end{array}
\]
each divided by $bdf \cdot ace$.
