\noindent\textbf{[p.~381]}\quad
\textbf{96.} The positive root $\theta$ of the equation
\[
\frac{a^{2}}{f^{2}+\theta} + \frac{c^{2}}{h^{2}+\theta} - 1 + \frac{e^{2}}{\theta} = 0
\]
has certain properties which connect themselves with the function
\[
\Theta_{0} = \theta^{q-1}\bigl\{(\theta+f^{2})\ldots(\theta+h^{2})\bigr\}^{-\tfrac{1}{2}}.
\]

We have, the accents denoting differentiations in regard to $\theta$,
\[
J'\,\frac{d\theta}{da} = \frac{2a}{\theta+f^{2}}\,,\qquad J'\,\frac{d\theta}{de} = -\frac{2e}{\theta}\,J'\,;
\quad\text{and}\quad
\frac{d\theta}{da} = \frac{1}{J'}\cdot\frac{2a}{\theta+f^{2}},
\]
where
\[
J' = \frac{a^{2}}{(f^{2}+\theta)^{2}} + \cdots + \frac{c^{2}}{(h^{2}+\theta)^{2}} + \frac{e^{2}}{\theta^{2}}\,;
\]
and we have the like formulae for $\dfrac{d\theta}{dc}$, $\dfrac{d\theta}{de}$.

We deduce
\[
\frac{a}{\theta+f^{2}}\,\frac{d\theta}{da} + \cdots + \frac{c}{\theta+h^{2}}\,\frac{d\theta}{dc} + \frac{e}{\theta}\,\frac{d\theta}{de} = \frac{2}{J'},
\]
and to this we may join, $\eta$ being arbitrary,
\[
\frac{a}{\theta+\eta+f^{2}}\,\frac{d\theta}{da} + \cdots + \frac{c}{\theta+\eta+h^{2}}\,\frac{d\theta}{dc} + \frac{e}{\theta+\eta}\,\frac{d\theta}{de}
= \frac{2}{J'}\Bigl[1 - \Bigl\{ \frac{a^{2}\,\eta}{(\theta+f^{2})(\theta+\eta+f^{2})} + \cdots + \frac{e^{2}\,\eta}{\theta(\theta+\eta)} \Bigr\}\Bigr].
\]

Again, defining $\nabla_{a}\theta$ and $\square\theta$ as immediately appears, we have
\[
\nabla_{a}\theta = \left(\frac{d\theta}{da}\right)^{2} + \cdots + \left(\frac{d\theta}{dc}\right)^{2}, \qquad = \frac{4}{J'},\quad \square\theta = \frac{8a^{2}}{J'(\theta+f^{2})^{2}}\,;
\]
and passing to the second differential coefficients, we have
\[
\frac{d^{2}\theta}{da^{2}} = \frac{2}{J'(\theta+f^{2})} - \frac{8a^{2}}{J'^{2}(\theta+f^{2})^{2}} - \frac{4a^{2}J''}{J'^{3}(\theta+f^{2})^{2}}\,,
\]
where
\[
J'' = -\frac{2a^{2}}{(f^{2}+\theta)^{3}} - \cdots - \frac{2c^{2}}{(h^{2}+\theta)^{3}} - \frac{2e^{2}}{\theta^{3}}\,,
\]
and the like formulae for $\dfrac{d^{2}\theta}{dc^{2}}$, $\dfrac{d^{2}\theta}{de^{2}}$. Joining to these $\dfrac{1}{e}\,\dfrac{d\theta}{de} = \dfrac{2}{J'}\cdot\dfrac{2q+1}{\theta}\,\dfrac{d\theta}{de}$, we obtain
\[
\square\theta = \left( \frac{d^{2}\theta}{da^{2}} + \cdots + \frac{d^{2}\theta}{dc^{2}} + \frac{d^{2}\theta}{de^{2}} + \frac{2q+1}{e}\,\frac{d\theta}{de} \right)
\]
\[
= \frac{2}{J'}\left[ \frac{1}{\theta+f^{2}} + \cdots + \frac{1 + (2q+1)}{\theta} \right]
- \frac{2}{J'^{2}}\left[ \frac{2a^{2}}{(\theta+f^{2})^{2}} + \cdots + \frac{2e^{2}}{\theta^{2}} \right]
- \frac{J''}{J'^{3}}\,(4a^{2}\cdots).
\]

\noindent\textbf{[p.~382]}\quad
where the last two terms destroy each other; observing that we have
\[
\frac{1}{e^{2}}\cdot\frac{d}{de}\bigl(e^{2q+1}\bigr) = \frac{2q+1}{e^{2}}\cdot e^{2q}\cdot\frac{1}{e}\cdot e = \frac{2q+1}{e^{2}}\cdot e^{2q-2}\cdot e^{2}\,;
\]
the result is
\[
\square\theta = \frac{2}{J'}\left( \frac{2q+2}{\theta} - \frac{2\theta'}{\theta} \right) + \frac{4\theta'}{\theta\,J'^{2}}\,.
\]

\textbf{97.} First example, $x^{2} = a^{2} + \cdots + c^{2}$, and $\theta$ the positive root of $\dfrac{x^{2}}{f^{2}+\theta} + \dfrac{e^{2}}{\theta} = 1$.

\smallskip
$V$ is assumed $= J\int_{\theta}^{\infty} t^{q-1}(t+f^{2})^{-\tfrac{1}{2}s}\,dt$, where $q+1$ is positive.

I do not work the example out; it corresponds step by step with, and is hardly more simple than, the next example, which relates to the ellipsoid. The result is
\[
\rho = (\sqrt{\pi})^{s}\,\Gamma(q+1)^{-1}\left[\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \tfrac{e^{2}}{f^{2}}\Bigr)^{q}\right],
\quad\text{if}\quad \tfrac{x^{2}}{f^{2}} + \tfrac{e^{2}}{f^{2}} < 1;
\]
hence the integral
\[
\int \frac{\bigl\{ 1 - \tfrac{x^{2}}{f^{2}} - \tfrac{e^{2}}{f^{2}} \bigr\}^{q}\, dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s+q}}\,,
\]
taken over the sphere $a^{2} + \cdots + z^{2} = f^{2}$,
\[
= \frac{(\sqrt{\pi})^{s}\,\Gamma(q+1)}{\Gamma(\tfrac{1}{2}s + q + 1)}\, f^{s}\int_{\theta}^{\infty} t^{q-1}(t+f^{2})^{-\tfrac{1}{2}s}\,dt.
\]

\textbf{98.} Second example, $\theta$ the positive root of $\dfrac{a^{2}}{f^{2}+\theta} + \cdots + \dfrac{c^{2}}{h^{2}+\theta} + \dfrac{e^{2}}{\theta} = 1$, $q+1$ positive.

Consider here the function
\[
V = \int_{\theta}^{\infty} t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}\,dt;
\]
this satisfies the prepotential equation. We have in fact
\[
\frac{dV}{da} = -\Theta_{0}\,\frac{d\theta}{da}\,,\qquad
\frac{d^{2}V}{da^{2}} = -\frac{d\Theta_{0}}{d\theta}\left(\frac{d\theta}{da}\right)^{2} - \Theta_{0}\,\frac{d^{2}\theta}{da^{2}}\,,
\]
with the like expressions for $\dfrac{d^{2}V}{dc^{2}}$, $\dfrac{d^{2}V}{de^{2}}$; also
\[
\frac{2q+1}{e}\,\frac{dV}{de} = -\Theta_{0}\,\frac{2q+1}{e}\,\frac{d\theta}{de}\,.
\]
Hence
\[
\square V = -\Theta_{0}\,\square\theta - \theta'\,\nabla_{a}\theta\,,
\]

\noindent\textbf{[p.~383]}\quad
or, substituting for $\square\theta$ and $\nabla_{a}\theta$ their values, this is
\[
= \Theta_{0}\left\{ -\frac{2}{J'}\left( \frac{2q+2}{\theta} - \frac{2\theta'}{\theta} \right) - \frac{4\theta'}{\theta\,J'^{2}} \right\} - \frac{4\theta'}{J'} = 0.
\]

Moreover $V$ does not become infinite for any values of $(a,\ldots, c, e)$, $e$ not $= 0$; and it vanishes for points at $\infty$. And not only so, but for indefinitely large values of any of the coordinates $(a, \ldots, c, e)$ it reduces itself to a numerical multiple of $(a^{2} + \cdots + c^{2} + e^{2})^{-\tfrac{1}{2}s+q}$; in fact, in this case $\theta$ is indefinitely large, $= a^{2} + \cdots + c^{2} + e^{2}$. Consequently throughout the integral, $t$ is indefinitely large, and we may therefore write
\[
V = \int_{\theta}^{\infty} t^{q-1} t^{-\tfrac{1}{2}s}\,dt
= \frac{-1}{\tfrac{1}{2}s - q}\bigl[t^{-\tfrac{1}{2}s + q}\bigr]_{\theta}^{\infty}
= \frac{1}{\tfrac{1}{2}s - q}\,\theta^{-\tfrac{1}{2}s + q};
\]
that is,
\[
V = \frac{1}{\tfrac{1}{2}s - q}\,(a^{2} + \cdots + c^{2} + e^{2})^{-\tfrac{1}{2}s + q}.
\]
The conditions of the theorem are thus satisfied, and we have for $\rho$ either of the formulae
\[
\rho = \frac{\Gamma(\tfrac{1}{2}s + q)}{(\sqrt{\pi})^{s}\,\Gamma(q)}\, e^{2q}\,W,\qquad
\rho = \frac{\Gamma(\tfrac{1}{2}s + q)}{2(\sqrt{\pi})^{s}\,\Gamma(q+1)}\,\frac{1}{e^{2q+1}}\,\frac{dW}{de}\,;
\]
in the former of them $q$ must be positive; in the latter it is sufficient if $q + 1$ be positive.

\textbf{99.} We have $W$ the same function of $(x, \ldots, z, e)$ that $V$ is of $(a, \ldots, c, e)$; viz.\ writing $\lambda$ for the positive root of
\[
\frac{x^{2}}{f^{2}+\lambda} + \cdots + \frac{z^{2}}{h^{2}+\lambda} + \frac{e^{2}}{\lambda} = 1,
\]
the value of $W$ is
\[
W = \int_{\lambda}^{\infty} t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}\,dt.
\]

Considering the formula which involves $e^{2q}W$, --- first, if $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} > 1$, then, when $e$ is $= 0$ the value of $\lambda$ is not $= 0$; the integral $W$ is therefore finite (not indefinitely large), and we have $e^{2q}W = 0$, consequently $\rho = 0$.

But if $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} < 1$, then, when $e$ is indefinitely small $\lambda$ is also indefinitely small; viz.\ we then have $\dfrac{e^{2}}{\lambda} = 1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}$, and the value of $W$ is
\[
W = \frac{1}{q}\,\lambda^{q}(f\ldots h)^{-1},
\]
and hence
\[
\rho = \frac{\Gamma(\tfrac{1}{2}s + q)}{(\sqrt{\pi})^{s}\,\Gamma(q+1)}\,(f\ldots h)^{-1}\left(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\right)^{q}.
\]

\noindent\textbf{[p.~384]}\quad
\textbf{100.} Again, using the formula which involves $\bigl(e^{2q+1}\dfrac{dW}{de}\bigr)$, we have first $\dfrac{dV}{da} = -\Theta\,\dfrac{d\theta}{da}$, or substituting for $\Theta$ and $\dfrac{d\theta}{da}$ their values and multiplying by $e^{2q+1}$, we find
\[
e^{2q+1}\,\frac{dV}{da} = 2 e^{2q}\theta^{q-1}\,\frac{a}{(f^{2}+\theta)^{2}}\bigl\{(\theta + f^{2})\ldots(\theta + h^{2})\bigr\}^{-\tfrac{1}{2}}\cdot \frac{1}{(\theta + f^{2})^{2}} + \cdots + \frac{1}{(h^{2}+\theta)^{2}} + \tfrac{1}{\theta^{2}}\,,
\]
and therefore
\[
e^{2q+1}\,\frac{dW}{de} = -2 e^{2q}\lambda^{q-1}\bigl\{(\lambda+f^{2})\ldots(\lambda + h^{2})\bigr\}^{-\tfrac{1}{2}}\,\frac{1}{\cdots}.
\]

Hence, writing $e = 0$: first, for an exterior point or $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} > 1$, $\lambda$ is not $= 0$, and the expression vanishes in virtue of the factor $e^{2q+1}$, whence also $\rho = 0$; next, for an interior point or $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} < 1$, $\lambda$ is $= 0$, hence also $\dfrac{e^{2}}{\lambda} = 1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}$ is infinite; neglecting in comparison with it the other terms $\dfrac{a^{2}}{(\lambda + f^{2})^{2}} + \cdots$, the value is
\[
= \frac{2}{e^{2}}\left( 1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}} \right)^{-1};
\]
and we have, as before,
\[
\rho = \frac{\Gamma(\tfrac{1}{2}s + q)}{(\sqrt{\pi})^{s}\,\Gamma(q+1)}\,(f\ldots h)^{-1}\left( 1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}} \right)^{q}.
\]

\textbf{101.} Hence in the formula
\[
V = \int \frac{\rho\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s + q}}
\]
$\rho$ has the value just found, or, what is the same thing, we have
\[
\int \frac{\bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\bigr)^{q}\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s + q}}
\]
taken over ellipsoid $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} = 1$,
\[
= \frac{(\sqrt{\pi})^{s}\,\Gamma(q+1)}{\Gamma(\tfrac{1}{2}s + q)}\,(f\ldots h)\int_{\theta}^{\infty} t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}\,dt.
\]

\noindent\textbf{[p.~385]}\quad
\textbf{102.} We may in this result write $e = 0$. There are two cases, according as the attracted point is exterior or interior: if it is exterior, $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} > 1$, $\theta$ will denote the positive root of the equation $\tfrac{a^{2}}{f^{2}+\theta} + \cdots + \tfrac{c^{2}}{h^{2}+\theta} = 1$; if it be interior, $\tfrac{a^{2}}{f^{2}} + \cdots + \tfrac{c^{2}}{h^{2}} < 1$, $\theta$ will be $= 0$; and we thus have
\[
\int \frac{\bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\bigr)^{q}\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2}\bigr\}^{\tfrac{1}{2}s + q}}
\]
\[
= \frac{(\sqrt{\pi})^{s}\,\Gamma(q+1)}{\Gamma(\tfrac{1}{2}s + q)}\,(f\ldots h)\int_{\theta}^{\infty} t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}\,dt,
\quad\text{for exterior point } \tfrac{a^{2}}{f^{2}} + \cdots + \tfrac{c^{2}}{h^{2}} > 1,
\]
\[
= \frac{(\sqrt{\pi})^{s}\,\Gamma(q+1)}{\Gamma(\tfrac{1}{2}s + q)}\,(f\ldots h)\int_{0}^{\infty} t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}\,dt,
\quad\text{for interior point } \tfrac{a^{2}}{f^{2}} + \cdots + \tfrac{c^{2}}{h^{2}} < 1;
\]
but as regards the value for an interior point it is to be observed that, unless $q$ be negative (between $0$ and $-1$, since $1 + q$ is positive by hypothesis), the two sides of the equation will be each of them infinite.

\textbf{103.} Third example. We assume here
\[
V = \int_{\theta}^{\infty} dt\, I^{m}\,T,
\]
where
\[
I = 1 - \frac{a^{2}}{f^{2}+t} - \cdots - \frac{c^{2}}{h^{2}+t} - \frac{e^{2}}{t},
\]
\[
T = t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}};
\]
as before, $\theta$ is the positive root of the equation
\[
1 - \frac{a^{2}}{f^{2}+\theta} - \cdots - \frac{c^{2}}{h^{2}+\theta} - \frac{e^{2}}{\theta} = 0,
\]
and $\tfrac{1}{2}s + q$ is positive in order that the integral may be finite; also $m$ is positive.

\textbf{104.} In order to show that $V$ satisfies the prepotential equation $\square V = 0$, I shall, in the first place, consider the more general expression
\[
V = \int_{\theta + \eta}^{\infty} dt\, I_{0}^{m}\,T,
\]
where $\eta$ is a constant positive quantity which will be ultimately put $= 0$. The functions previously called $J$ and $\Theta$ will be written $J_{0}$ and $\Theta_{0}$, and $J_{\eta}$ will now denote
\[
J_{\eta} = 1 - \frac{a^{2}}{\theta + \eta + f^{2}} - \cdots - \frac{c^{2}}{\theta + \eta + h^{2}} - \frac{e^{2}}{\theta + \eta},
\]
\[
\Theta_{\eta} = (\theta + \eta)^{q-1}\bigl\{(\theta + \eta + f^{2})\ldots(\theta + \eta + h^{2})\bigr\}^{-\tfrac{1}{2}};
\]

\noindent\textbf{[p.~386]}\quad
whence also, subtracting from $J$ the evanescent function $J_{0}$, we have
\[
J - J_{0} = \frac{a^{2}\,\eta}{(\theta + f^{2})(\theta + \eta + f^{2})} + \cdots + \frac{e^{2}\,\eta}{\theta(\theta + \eta)};
\]
say this is $= \eta P$;
and we have thence, by former equations and in the present notation,
\[
\frac{a}{\theta + \eta + f^{2}}\,\frac{d\theta}{da} + \cdots + \frac{c}{\theta + \eta + h^{2}}\,\frac{d\theta}{dc} + \frac{e}{\theta + \eta}\,\frac{d\theta}{de} = \frac{2}{J_{0}'}\bigl(1 - \eta P\bigr),
\]
\[
\nabla_{a}\theta = \frac{4}{J_{0}'^{2}}\,,\qquad
\square \theta = \frac{4 \Theta_{0}'}{\Theta_{0}\,J_{0}}\,.
\]

In virtue of the equation which determines $\theta$, we have
\[
\frac{dV}{da} = \int_{\theta + \eta}^{\infty} dt\, m I^{m-1}\frac{dI}{da}\,T,\qquad
\frac{d^{2}V}{da^{2}} = \int_{\theta + \eta}^{\infty} dt\Bigl\{ m(m-1) I^{m-2}\Bigl(\frac{dI}{da}\Bigr)^{2}T + m I^{m-1}\,\frac{d^{2}I}{da^{2}}\,T \Bigr\}
\]
\[
- J_{\eta}^{m}\Theta_{\eta}\,\frac{d\theta}{da}\,;
\]
and thence
\[
- m J_{\eta}^{m-1}\Theta_{\eta}\,\frac{d\theta}{da}
- J_{\eta}^{m}\Theta_{\eta}\,\frac{d^{2}\theta}{da^{2}}
- J_{\eta}^{m}\Theta_{\eta}\,\left(\frac{d\theta}{da}\right)^{2}
\]
\[
- J_{\eta}^{m}\Theta_{\eta}'\,\frac{d\theta}{da}\,,
\]
with like expressions for $\dfrac{d^{2}V}{dc^{2}}$, $\dfrac{d^{2}V}{de^{2}}$. Also
\[
\frac{2q+1}{e}\,\frac{dV}{de} = \int_{\theta + \eta}^{\infty} dt\, m I^{m-1}\,\frac{2q+1}{e}\,\frac{dI}{de}\,T - J_{\eta}^{m}\,\Theta_{\eta}\cdot\frac{2q+1}{e}\,\frac{d\theta}{de}\,;
\]
and hence
\[
\square V = \int_{\theta + \eta}^{\infty} dt\,\Bigl\{ m(m-1) I^{m-2}\,\Bigl[\,\Bigl(\tfrac{dI}{da}\Bigr)^{2} + \cdots\,\Bigr]\,T + m I^{m-1}\,\square I\,T \Bigr\}
\]
\[
+ 4 m J_{\eta}^{m-1}\Theta_{\eta}\left( \frac{a}{\theta + \eta + f^{2}}\,\frac{d\theta}{da} + \cdots + \frac{e}{\theta + \eta}\,\frac{d\theta}{de} \right)
- J_{\eta}^{m}\Theta_{\eta}\,\square\theta - J_{\eta}^{m}\Theta_{\eta}'\,\nabla_{a}\theta\,.
\]

\noindent\textbf{[p.~387]}\quad
\textbf{105.} Writing $I'$, $T'$ for the first derived coefficients of $I$, $T$ in regard to $t$, we have
\[
I' = \frac{a^{2}}{(f^{2}+t)^{2}} + \cdots + \frac{c^{2}}{(h^{2}+t)^{2}} + \frac{e^{2}}{t^{2}}\,,\qquad
\frac{T'}{T} = \frac{q-1}{t} - \tfrac{1}{2}\left( \frac{1}{f^{2}+t} + \cdots + \frac{1}{h^{2}+t} \right) = \frac{2q+1}{2t} \cdot \cdots.
\]

The integral is therefore
\[
\int_{\theta + \eta}^{\infty} dt\,\Bigl\{ 2m I^{m-1}\,I'\,T + m(m-1) I^{m-2}\cdot 4 I' \cdot T \Bigr\}
\]
\[
= \int_{\theta + \eta}^{\infty} dt\,\Bigl\{ 4m I^{m-1}\,T' + 4 m(m-1) I^{m-2}\,I'\,T \Bigr\},
\]
viz.\ $I^{m-1}T$ vanishing for $t = \infty$, this is
\[
= -4 m J_{\eta}^{m-1}\Theta\,.
\]
Hence, writing $(J^{m}\Theta)'$ instead of $\dfrac{d}{dt}(J^{m}\Theta)$, we have
\[
\square V = -4 m J_{\eta}^{m-1}\Theta_{\eta}
+ 4 m J_{\eta}^{m-1}\,\Theta_{\eta}\,\Bigl( \frac{a}{\theta + \eta + f^{2}}\,\frac{d\theta}{da} + \cdots + \frac{e}{\theta + \eta}\,\frac{d\theta}{de} \Bigr)
\]
\[
- (J^{m}\Theta)'\,\nabla_{a}\theta\,,
\]
viz.\ this is
\[
\square V = -4 m J_{\eta}^{m-1}\Theta_{\eta}
+ 4 m J_{\eta}^{m-1}\,\frac{1}{J_{0}'}\bigl(1 - \eta P\bigr)\cdot 2
- (J_{\eta}^{m}\Theta_{\eta})'\,\frac{4}{J_{0}'^{2}}\,,
\]
or, writing $m J_{\eta}^{m-1}J_{\eta}'\Theta + J_{\eta}^{m}\Theta'$ instead of $(J^{m}\Theta)'$, this is
\[
\square V = -\frac{4 m J_{\eta}^{m-1}\Theta_{\eta}}{J_{0}'}\bigl(J_{\eta}' - 2 J P + J_{0}'\bigr) - \frac{4 J_{\eta}^{m}}{J_{0}'^{2}}\bigl(\Theta_{\eta}'\,J_{0} - \Theta(J_{0}')'\bigr).
\]
We have here
\[
J_{\eta}' - 2 J P + J_{0}' = \eta^{2}\,Q,\quad\text{suppose}.
\]
Also $\Theta_{\eta}'\Theta_{0} - \Theta\Theta_{0}'$ contains the factor $\eta$, is $= \eta M$ suppose.

\noindent\textbf{[p.~388]}\quad
\textbf{106.} Substituting for $J$, $J' - 2JP + J$, and $\Theta'\Theta_{0} - \Theta\Theta_{0}'$ their values $\eta P$, $\eta Q$, and $\eta M$, the whole result contains the factor $\eta^{m+1}$, viz.\ we have
\[
\square V = -\frac{4 \eta^{m+1}\,\Theta_{\eta}}{J_{0}^{m+1}}\,\bigl(m P^{m}Q + P^{m}M\bigr).
\]
If here, except in the term $\eta^{m+1}$, we write $\eta = 0$, we have
\[
P = \frac{a^{2}}{(\theta + f^{2})^{2}} + \cdots + \frac{c^{2}}{(\theta + h^{2})^{2}} + \frac{e^{2}}{\theta^{2}}\,,
\]
the formula becomes
\[
\square V = -\frac{4 \eta^{m+1}\,\Theta_{0}}{J_{0}^{m+1}}\,\Bigl( m P^{m}\,J_{0}'^{-} + P^{m}M_{0} \Bigr),
\]
or (instead of $J_{0}$, $\Theta_{0}$) using now $J$, $\Theta$ in their original significations
\[
J = 1 - \frac{a^{2}}{f^{2}+\theta} - \cdots - \frac{c^{2}}{h^{2}+\theta} - \frac{e^{2}}{\theta}, \quad\text{and}\quad
\Theta = \theta^{q-1}\bigl\{(\theta + f^{2})\ldots(\theta + h^{2})\bigr\}^{-\tfrac{1}{2}},
\]
this is
\[
\square V = -\frac{4 \eta^{m+1}\,\Theta}{J^{m+1}}\,\Bigl( m P^{m}\,J^{-1}_{0} + P^{m}M_{0} \Bigr),
\]
or, what is the same thing,
\[
\square V = -\frac{4 \eta^{m+1}\,\Theta}{J^{m+1}}\,\bigl( m P + (\,\ldots\,) \bigr),
\]
viz.\ the expression in $\{\;\}$ is
\[
-\frac{1}{(\theta + f^{2})^{2}} + \cdots + \frac{1}{(\theta + h^{2})^{2}} + \frac{1}{\theta^{2}} - \frac{2q+2}{\theta}.
\]
We thus see that, $\eta$ being infinitesimal, $\square V$ is infinitesimal of the order $\eta^{m+1}$; and hence, $\eta$ being $= 0$, we have
\[
\square V = 0;
\]
viz.\ the prepotential equation is satisfied by the value
\[
V = \int_{\theta}^{\infty} dt\, I^{m}\,T,
\]
where $m + 1$ is positive.

\textbf{107.} We have consequently a value of $\rho$ corresponding to the foregoing value of $V$; and this value is
\[
\rho = \frac{\Gamma(\tfrac{1}{2}s + q)}{2(\sqrt{\pi})^{s}\,\Gamma(q+1)}\,\frac{1}{e^{2q+1}}\,\frac{dW}{de}\,,
\]

\noindent\textbf{[p.~389]}\quad
where, writing $\lambda$ for the positive root of
\[
1 - \frac{x^{2}}{f^{2}+\lambda} - \cdots - \frac{z^{2}}{h^{2}+\lambda} - \frac{e^{2}}{\lambda} = 0,
\]
we have
\[
W = \int_{\lambda}^{\infty} dt\,\Bigl(1 - \frac{x^{2}}{f^{2}+t} - \cdots - \frac{z^{2}}{h^{2}+t} - \frac{e^{2}}{t}\Bigr)^{m}\, t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}};
\]
we thence obtain
\[
\frac{dW}{de} = \int_{\lambda}^{\infty} dt\,\frac{-2me}{t}\Bigl(1 - \frac{x^{2}}{f^{2}+t} - \cdots - \frac{z^{2}}{h^{2}+t} - \frac{e^{2}}{t}\Bigr)^{m-1}\, t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}
- \bigl(\ldots\bigr)\frac{d\lambda}{de},
\]
or, multiplying by $e^{2q+1}$ and substituting for $\dfrac{d\lambda}{de}$ its value $\dfrac{2e}{\lambda}\cdot\dfrac{1}{\cdots}$, we have
\[
e^{2q+1}\,\frac{dW}{de} = -2 m e^{2q+2}\int_{\lambda}^{\infty} dt\,\frac{1}{t}\Bigl(1 - \frac{x^{2}}{f^{2}+t} - \cdots - \frac{z^{2}}{h^{2}+t} - \frac{e^{2}}{t}\Bigr)^{m-1}\, t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}
+ \cdots,
\]
where the second term, although containing the evanescent factor
\[
\Bigl(1 - \frac{x^{2}}{f^{2}+\lambda} - \cdots - \frac{z^{2}}{h^{2}+\lambda} - \frac{e^{2}}{\lambda}\Bigr)^{m},
\]
is for the present retained.

\textbf{108.} I attend to the second term.

\smallskip
$1^{\circ}$. Suppose $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} > 1$; then, as $e$ diminishes and becomes $= 0$, $\lambda$ does not become zero, but it becomes the positive root of the equation
\[
1 - \frac{x^{2}}{f^{2}+\lambda} - \cdots - \frac{z^{2}}{h^{2}+\lambda} = 0;
\]
hence the term, containing as well the evanescent factor $e^{2q+2}$ as the other evanescent factor $\Bigl(1 - \frac{x^{2}}{f^{2}+\lambda} - \cdots - \frac{z^{2}}{h^{2}+\lambda} - \frac{e^{2}}{\lambda}\Bigr)^{m}$, is $= 0$.

\noindent\textbf{[p.~390]}\quad
$2^{\circ}$. Suppose $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} < 1$; then, as $e$ diminishes to zero, $\lambda$ tends to become $= 0$, but $\dfrac{e^{2}}{\lambda}$ is finite and $= 1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}$, whence $\dfrac{1}{\lambda}$ is indefinitely large; and since $\dfrac{x^{2}}{(\lambda + f^{2})^{2}} + \cdots + \dfrac{z^{2}}{(\lambda + h^{2})^{2}}$ becomes $= + \dfrac{x^{2}}{f^{4}} + \cdots + \dfrac{z^{2}}{h^{4}}$, which is finite, the denominator may be reduced to $\dfrac{e^{2}}{\lambda}$, and the term therefore is
\[
= -2\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{m-1}\,e^{2q+2}(f\ldots h)^{-1}\lambda^{-q-1},
\]
which, the other factor being finite, vanishes in virtue of the evanescent factor
\[
\Bigl(1 - \frac{x^{2}}{f^{2}+\lambda} - \cdots - \frac{z^{2}}{h^{2}+\lambda} - \frac{e^{2}}{\lambda}\Bigr)^{m}.
\]

Hence the second term always vanishes, and we have ($e$ being $= 0$)
\[
e^{2q+1}\,\frac{dW}{de} = -2 m\int_{\lambda}^{\infty} dt\,\frac{e^{2q+2}}{t}\Bigl(1 - \frac{x^{2}}{f^{2}+t} - \cdots - \frac{z^{2}}{h^{2}+t} - \frac{e^{2}}{t}\Bigr)^{m-1}\, t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}.
\]

\textbf{109.} Considering first the case $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} > 1$: then, as $e$ diminishes to zero, $\lambda$ does not become $= 0$; the integral contains no infinite element, and it consequently vanishes in virtue of the factor $e^{2q+2}$.

But if $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} < 1$, then, introducing instead of $t$ the new variable $\xi$, $= t \cdot \dfrac{1}{\lambda}$, that is, $t = \dfrac{\lambda}{\xi}$, $dt = -\dfrac{\lambda}{\xi^{2}}\,d\xi$, and writing for shortness
\[
R = 1 - \frac{x^{2}}{f^{2}+\xi} - \cdots - \frac{z^{2}}{h^{2}+\xi}\,,
\]
the term becomes
\[
= \int_{0}^{R} d\xi \cdot 2 m (R - \xi)^{m-1}\xi^{q}\bigl\{(f^{2}+\xi)\ldots(h^{2}+\xi)\bigr\}^{-\tfrac{1}{2}},
\]
where, as regards the limits, corresponding to $t = \infty$ we have $\xi = 0$, and corresponding to $t = \lambda$ we have $\xi$ the positive root of $R - \xi = 0$. But $e$ is indefinitely small; except for indefinitely small values of $\xi$, we have
\[
R = 1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}},\quad\text{and}\quad
\bigl\{(f^{2}+\xi)\ldots(h^{2}+\xi)\bigr\}^{-\tfrac{1}{2}} = (f\ldots h)^{-1};
\]

\noindent\textbf{[p.~391]}\quad
and if $\xi$ be indefinitely small, then, whether we take the accurate or the reduced expressions, the elements are finite, and the corresponding portion of the integral is indefinitely small. We may consequently reduce as above; viz.\ writing now
\[
R = 1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}},
\]
the formula is
\[
e^{2q+1}\,\frac{dW}{de} = \int_{0}^{R} d\xi\, 2 m (R - \xi)^{m-1}\,\xi^{q}\,(f\ldots h)^{-1}
\]
\[
= -2 m (f\ldots h)^{-1}\int_{0}^{R} d\xi\,\xi^{q}(R - \xi)^{m-1},
\]
or writing $\xi = R u$, the integral becomes $= R^{q+m}\int_{0}^{1} du\, u^{q}(1 - u)^{m-1}$, which is
\[
= R^{q+m}\,\frac{\Gamma(1 + q)\Gamma(m)}{\Gamma(1 + q + m)};
\]
that is, we have
\[
e^{2q+1}\,\frac{dW}{de} = -\frac{2 m (f\ldots h)^{-1}\Gamma(1+q)\Gamma(m)}{\Gamma(1+q+m)}\,R^{q+m},
\]
and consequently
\[
\rho = \frac{\Gamma(\tfrac{1}{2}s + q)\Gamma(m)}{(\sqrt{\pi})^{s}\,\Gamma(1+q+m)}\,(f\ldots h)^{-1}\,R^{q+m},
\]
that is,
\[
\rho = \frac{\Gamma(\tfrac{1}{2}s + q)\Gamma(m)}{(\sqrt{\pi})^{s}\,\Gamma(1 + q + m)}\,(f\ldots h)^{-1}\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{q+m},
\]
viz.\ $\rho$ has this value for values of $(x, \ldots, z)$ such that $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} < 1$, but is $= 0$ if $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} > 1$.

\textbf{110.} Multiplying by a constant factor so as to reduce $\rho$ to the value $R^{q+m}$, the final result is that the integral
\[
V = \int \frac{\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{q+m}\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s + q}}\,,
\]
the limits being given by the equation
\[
\frac{x^{2}}{f^{2}} + \cdots + \frac{z^{2}}{h^{2}} = 1,
\]
is equal to
\[
\frac{\Gamma(\tfrac{1}{2}s)\Gamma(1+q+m)}{\Gamma(1+q)\Gamma(\tfrac{1}{2}s + q + m)}\,(f\ldots h)\int_{\theta}^{\infty} dt\, t^{m}\Bigl(1 - \frac{a^{2}}{f^{2}+t} - \cdots - \frac{c^{2}}{h^{2}+t} - \frac{e^{2}}{t}\Bigr)^{m}\, t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}},
\]
where $\theta$ is the positive root of
\[
1 - \frac{a^{2}}{\theta + f^{2}} - \cdots - \frac{c^{2}}{\theta + h^{2}} - \frac{e^{2}}{\theta} = 0.
\]

\noindent\textbf{[p.~392]}\quad
In particular, if $e = 0$, or
\[
V = \int \frac{\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{q+m}\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2}\bigr\}^{\tfrac{1}{2}s + q}}\,,
\]
there are two cases:
\[
\text{exterior,}\quad \tfrac{a^{2}}{f^{2}} + \cdots + \tfrac{c^{2}}{h^{2}} > 1,\ \theta\text{ is positive root of }1 - \tfrac{a^{2}}{f^{2}+\theta} - \cdots - \tfrac{c^{2}}{h^{2}+\theta} = 0,
\]
\[
\text{interior,}\quad \tfrac{a^{2}}{f^{2}} + \cdots + \tfrac{c^{2}}{h^{2}} < 1,\ \theta\text{ vanishes, viz.\ the limits in the integral are }\infty, 0;
\]
$q$ must be negative, $1 + q$ positive as before, in order that the $t$-integral may not be infinite in regard to the element $t = 0$.

It is assumed in the proof that $m$ and $1 + q$ are each of them positive; but, as appears by the second example, the theorem is true for the extreme value $m = 0$; it does not, however, appear that the proof can be extended to include the extreme value $q = -1$. The formula seems, however, to hold good for values of $m$, $q$ beyond the foregoing limits; and it would seem that the only necessary conditions are $\tfrac{1}{2}s + q$, $1 + m$, and $1 + q + m$, each of them positive. The theorem is, in fact, a particular case of the following one, proved Annex X. No.~162, viz.
\[
V = \int \frac{\phi\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s + q}}\,,
\]
taken over the ellipsoid $\tfrac{x^{2}}{f^{2}} + \cdots + \tfrac{z^{2}}{h^{2}} = 1$, is equal to
\[
\frac{(\sqrt{\pi})^{s}\,\Gamma(1+q)}{\Gamma(1+\tfrac{1}{2}s + q)}\,(f\ldots h)\int_{\theta}^{\infty} dt\,\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}(1 - \sigma)^{q + \tfrac{1}{2}s}\int_{0}^{1}(1 - x)^{q}\phi\{\sigma + (1-\sigma)x\}\,dx,
\]
where $\sigma$ denotes $\dfrac{a^{2}}{f^{2}+t} + \cdots + \dfrac{c^{2}}{h^{2}+t} + \dfrac{e^{2}}{t}$; assuming $\phi u = (1 - u)^{q+m}$, we have
\[
\phi\{\sigma + (1 - \sigma)x\} = (1 - \sigma)^{q+m}(1 - x)^{q+m},
\]
and the theorem is thus proved.

\textbf{111.} Particular cases: $m = 0$;
\[
\int \frac{\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{q}\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s + q}}
= \frac{(\sqrt{\pi})^{s}\,\Gamma(1+q)}{\Gamma(1+\tfrac{1}{2}s + q)}\,(f\ldots h)\int_{\theta}^{\infty} dt\,t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}.
\]
Cor. In a somewhat similar manner it may be shown that
\[
\int \frac{\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{q}\, x\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s + q}}
= \frac{a(\sqrt{\pi})^{s}\,\Gamma(1+q)}{\Gamma(1+\tfrac{1}{2}s + q)}\,(f\ldots h)\int_{\theta}^{\infty} dt\,t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}}.
\]

\noindent\textbf{[p.~393]}\quad
Multiplying the first by $a$ and subtracting it from the second, we have
\[
\int \frac{\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{q}\,(a-x)\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s + q}}
= -\frac{(\sqrt{\pi})^{s}\,\Gamma(1+q)}{\Gamma(1+\tfrac{1}{2}s + q)}\,(f\ldots h)\int_{\theta}^{\infty} dt\,\frac{a}{f^{2}+t}\,t^{q-1}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}},
\]
or, writing $q+1$ for $q$, this is
\[
\int \frac{\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{q}\,(a-x)\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s + q + 1}}
= \frac{(\sqrt{\pi})^{s}\,\Gamma(1+q)}{\Gamma(1+\tfrac{1}{2}s + q)}\,(f\ldots h)\int_{\theta}^{\infty} dt\,\frac{a}{f^{2}+t}\,t^{q}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}},
\]
and we have similar formulae with $\ldots$, $(c - z)$, $e$, instead of $(a - x)$, in the numerator.

\textbf{112.} If $m = 1$, we have
\[
\int \frac{\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{q+1}\,dx\ldots dz}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + e^{2}\bigr\}^{\tfrac{1}{2}s + q}}
= \frac{(\sqrt{\pi})^{s}\,\Gamma(1+q)}{\Gamma(2+\tfrac{1}{2}s + q)}\,(f\ldots h)\int_{\theta}^{\infty} dt\,t^{q}\bigl\{(t+f^{2})\ldots(t+h^{2})\bigr\}^{-\tfrac{1}{2}},
\]
which, differentiated in respect to $a$, gives the $(a - x)$-formula; hence conversely, assuming the $a - x, \ldots, c - z, e$-formulae, we obtain by integration the last preceding formula to a constant pr\`es, viz.\ we thereby obtain the multiple integral $= C +$ right-hand function, where $C$ is independent of $(a, \ldots, c, e)$; by taking these all infinite, and observing that then $\theta = \infty$, the two integrals each vanish, and we obtain $C = 0$.

In particular, when $s = 3$, $q = -\tfrac{1}{2}$, then
\[
\int \frac{\Bigl(1 - \tfrac{x^{2}}{f^{2}} - \cdots - \tfrac{z^{2}}{h^{2}}\Bigr)^{\tfrac{1}{2}}\,dx\,dy\,dz}{\bigl\{(a-x)^{2} + (b-y)^{2} + (c-z)^{2} + e^{2}\bigr\}^{1}}
= \frac{\pi^{2}}{2}(f g h)\int_{\theta}^{\infty}\frac{dt}{\sqrt{t(t+f^{2})(t+g^{2})(t+h^{2})}},
\]
which, putting therein $e = 0$, gives the potential of an ellipsoid for the cases of an exterior point and an interior point respectively.

\begin{center}
\textbf{Annex V.}\quad Green's Integration of the Prepotential Equation. Art.\ Nos.~113 to 128.
\end{center}

\noindent\textbf{[p.~393, cont.]}\quad
\textbf{113.} In the present Annex, I in part reproduce Green's process for the integration of this equation by means of a series of functions, which are analogous to Laplace's Functions and may be termed ``Greenians'' (see his Memoir on the Attraction of Ellipsoids, referred to above, p.~320); each such function gives rise to a Prepotential Integral.

Green shows, by a complicated and difficult piece of general reasoning, that there exist solutions of the form $V = \Theta\phi$ (see post.\ No.~116), where $\phi$ is a function of the

\noindent\textbf{[p.~394]}\quad
$s$ new variables $\alpha, \beta, \ldots, \gamma$ without $\theta$, such that $\nabla^{2}\phi = \kappa\phi$, $\kappa$ being a function of $\theta$ only; these functions $\phi$ of the variables $\alpha, \beta, \ldots, \gamma$ are in fact the Greenian Functions in question. The function of the order $0$ is $\phi = 1$; those of the order $1$ are $\phi = \alpha$, $\phi = \beta, \ldots, \phi = \gamma$; those of the order $2$ are $\phi = \alpha\beta$, etc., and $s$ functions each of the form
\[
\phi = \tfrac{1}{2}(A\alpha^{2} + B\beta^{2} + \cdots + C\gamma^{2}) + D.
\]
The existence of the functions just referred to other than the $s$ functions involving the squares of the variables is obvious enough; the difficulty first arises in regard to these $s$ functions; and the actual development of them appears to me important by reason of the light which is thereby thrown upon the general theory. This I accomplish in the present Annex; and I determine by Green's process the corresponding prepotential integrals. I do not go into the question of the Greenian Functions of orders superior to the second.

\textbf{114.} I write for greater clearness $(a, b, \ldots, c, e)$ instead of $(a, \ldots, c, e)$ to denote the series of $(s + 1)$ variables; viz.\ $(a, b, \ldots, c)$ will denote a series of $s$ variables; corresponding to these we have the semiaxes $(f, g, \ldots, h)$, and the new variables $(\alpha, \beta, \ldots, \gamma)$; these last, with the before-mentioned function $\theta$, are the $s + 1$ new variables of the problem; and, for convenience, there is introduced also a quantity $\varepsilon$; viz.\ we have
\[
a = \sqrt{f^{2}+\theta}\,\alpha,\quad b = \sqrt{g^{2}+\theta}\,\beta,\ \ldots,\ c = \sqrt{h^{2}+\theta}\,\gamma,\quad e = \sqrt{\theta}\,\varepsilon,
\]
where $\lambda = \alpha^{2} + \beta^{2} + \cdots + \gamma^{2} + \varepsilon^{2}$.

That is, we have $\theta$ a function of $a, b, \ldots, c, e$, determined by
\[
\frac{a^{2}}{f^{2}+\theta} + \frac{b^{2}}{g^{2}+\theta} + \cdots + \frac{c^{2}}{h^{2}+\theta} + \frac{e^{2}}{\theta} = 1,
\]
and then $\alpha, \beta, \ldots, \gamma$ are given as functions of the same quantities $a, b, \ldots, c, e$ by the equations
\[
\alpha^{2} = \frac{a^{2}}{f^{2}+\theta},\quad \beta^{2} = \frac{b^{2}}{g^{2}+\theta},\ \ldots,\ \gamma^{2} = \frac{c^{2}}{h^{2}+\theta};
\]
also $\varepsilon$, considered as a function of the same quantities, is
\[
\varepsilon^{2} = 1 - \frac{a^{2}}{f^{2}+\theta} - \frac{b^{2}}{g^{2}+\theta} - \cdots - \frac{c^{2}}{h^{2}+\theta} = \frac{e^{2}}{\theta}.
\]

\textbf{115.} Introducing instead of $a, b, \ldots, c, e$ the new variables $\alpha, \beta, \ldots, \gamma, \theta$, the transformed differential equation is
\[
\nabla^{2}V + \frac{d^{2}V}{d\theta^{2}}\Bigl(s + 2q + 2 - \frac{\alpha^{2}}{f^{2}+\theta} - \cdots - \frac{\gamma^{2}}{h^{2}+\theta}\Bigr)\cdot 4 + \nabla^{2}V = 0,
\]

\noindent\textbf{[p.~395]}\quad
where for shortness
\[
\nabla^{2}V = \theta\left[\frac{f^{2} + \theta}{a^{2}} - \frac{h^{2}+\theta}{c^{2}} + 1\right]\frac{d^{2}V}{d\alpha^{2}} + \cdots
\]
\[
+ \theta\left[\frac{f^{2} + \theta}{a^{2}} + \cdots\right]\frac{d^{2}V}{d\gamma^{2}} - \frac{\theta\cdot 2q}{e^{2}}\,\frac{d^{2}V}{d\varepsilon^{2}}\,\cdots
\]
\[
+ \frac{2\alpha}{f^{2}+\theta}\,\frac{d^{2}V}{d\alpha\,d\beta} - \&c.
\]
\[
+ \frac{1}{f^{2}+\theta}\Bigl(-\frac{\alpha}{f^{2}+\theta} - \&c.\Bigr)\frac{dV}{d\alpha}
\]
\[
+ \frac{1}{\theta}\Bigl(-\frac{\varepsilon}{\theta} - \&c.\Bigr)\frac{dV}{d\beta}
\]
\[
+ \frac{1}{\theta}\Bigl(-\frac{\varepsilon}{\theta}\Bigr)\frac{dV}{d\gamma}\,.
\]
Also
\[
\frac{d^{2}V}{d\theta^{2}} = \frac{1}{4(f^{2}+\theta)}\Bigl[\frac{f^{2}+\theta}{a^{2}} + \cdots + \frac{h^{2}+\theta}{c^{2}} + \frac{\theta}{e^{2}}\Bigr]\Bigl[(f^{2}+\theta)\frac{d^{2}V}{d\alpha^{2}} + \cdots + (h^{2}+\theta)\frac{d^{2}V}{d\gamma^{2}} + \theta\,\frac{d^{2}V}{d\varepsilon^{2}}\Bigr]\cdots\,.
\]

\textbf{116.} To integrate the equation for $V$, we assume
\[
V = \Theta\phi,
\]
where $\Theta$ is a function of $\theta$ only, and $\phi$ a function of $\alpha, \beta, \ldots, \gamma$ (without $\theta$), such that
\[
\nabla^{2}\phi = \kappa\phi,
\]
$\kappa$ being a function of $\theta$ only. Assuming that this is possible, the remaining equation to be satisfied is obviously
\[
\Theta'' + \kappa\Theta = 0.
\]
Solutions of the form in question are
\[
\phi = 1,\quad \kappa = 0,
\]
\[
\phi = \alpha,\quad \kappa = \frac{1}{f^{2}+\theta},\quad \text{or} \quad \phi = \beta, \ldots, \kappa = \frac{1}{g^{2}+\theta},\ \ldots,\ \frac{1}{h^{2}+\theta},
\]
\[
\phi = \alpha\beta,\quad \kappa = \frac{1}{f^{2}+\theta} + \frac{1}{g^{2}+\theta},\ \&c.,
\]
and the corresponding values of $\Theta$ involve $\theta$.

\noindent\textbf{[p.~396]}\quad
and it can be shown next that there is a solution of the form
\[
\phi = \tfrac{1}{2}(A\alpha^{2} + \cdots + C\gamma^{2}) + D.
\]

\textbf{117.} In fact, assuming that this satisfies $\nabla^{2}\phi - \kappa\phi = 0$, we must have identically
\[
\frac{A}{f^{2}+\theta}\Bigl\{-\alpha^{2}\frac{\theta}{f^{2}+\theta}(\beta^{2} + \cdots + \gamma^{2}) - \alpha^{2}\frac{\theta}{f^{2}+\theta}\,\varepsilon^{2} + 1\Bigr\}
\]
\[
+ \frac{B}{g^{2}+\theta}\Bigl\{\cdots\Bigr\} + \cdots
\]
\[
+ \tfrac{2q}{\varepsilon^{2}}\Bigl\{-\alpha^{2}\frac{\theta}{f^{2}+\theta} - \beta^{2}\frac{\theta}{g^{2}+\theta} - \cdots\Bigr\}
\]
\[
+ \frac{A}{f^{2}+\theta}\Bigl\{-\alpha^{2}\tfrac{\theta}{f^{2}+\theta} - \cdots + \tfrac{2}{f^{2}+\theta}\Bigr\} + \cdots
\]
\[
+ \frac{2q}{\varepsilon^{2}}\Bigl\{-\alpha^{2}\tfrac{\theta}{f^{2}+\theta} - \cdots\Bigr\}
\]
\[
+ \kappa\bigl\{\tfrac{1}{2}(A\alpha^{2} + B\beta^{2} + \cdots + C\gamma^{2}) + D\bigr\} = 0\,;
\]
so that, from the term in $\alpha^{2}$, we have
\[
\frac{A}{f^{2}+\theta}\Bigl\{-\frac{\theta}{f^{2}+\theta}(s + 2q + 2)\Bigr\} - \frac{B\theta}{(g^{2}+\theta)(f^{2}+\theta)} - \cdots - \frac{C\theta}{(h^{2}+\theta)(f^{2}+\theta)} + \tfrac{1}{2}\kappa A = 0,
\]
or, what is the same thing,
\[
A\bigl\{2q + 2 + \Omega + \tfrac{1}{2}\kappa(f^{2} + \theta)\bigr\} + \kappa f^{2} D = 0,
\]
with the like equations from $\beta^{2}, \ldots, \gamma^{2}$; and from the constant term we have
\[
\frac{A}{f^{2}+\theta} + \frac{B}{g^{2}+\theta} + \cdots + \frac{C}{h^{2}+\theta} + \kappa D = 0.
\]

\textbf{118.} Multiplying this last by $f^{2}$, and adding it to the first, we obtain
\[
A\Bigl\{1 + \frac{f^{2}}{f^{2}+\theta}\Bigr\}\cdot 1 + \frac{B f^{2}}{g^{2}+\theta} + \cdots + \frac{C f^{2}}{h^{2}+\theta} + \tfrac{1}{2}\kappa A(f^{2}+\theta) + \kappa f^{2} D = 0;
\]
viz.\ putting for shortness $\Omega = \dfrac{a^{2}}{f^{2}+\theta} + \cdots = \dfrac{\theta}{f^{2}+\theta}\alpha^{2} + \cdots + \dfrac{\theta}{h^{2}+\theta}\gamma^{2}$, this is
\[
A\bigl\{2q + 2 + \Omega + \tfrac{1}{2}\kappa(f^{2}+\theta)\bigr\} + \kappa f^{2} D = 0;
\]
and similarly
\[
B\bigl\{2q + 2 + \Omega + \tfrac{1}{2}\kappa(g^{2}+\theta)\bigr\} + \kappa g^{2} D = 0,
\]
\[
C\bigl\{2q + 2 + \Omega + \tfrac{1}{2}\kappa(h^{2}+\theta)\bigr\} + \kappa h^{2} D = 0.
\]

\noindent\textbf{[p.~397]}\quad
To these we join the foregoing equation
\[
\frac{A}{f^{2}+\theta} + \frac{B}{g^{2}+\theta} + \cdots + \frac{C}{h^{2}+\theta} + \kappa D = 0.
\]
Eliminating $A, B, \ldots, C, D$, we have an equation which determines $\kappa$ as a function of $\theta$; and the equations then determine the ratios of $A, B, \ldots, C, D$, so that these quantities will be given as determinate multiples of an arbitrary quantity $M$. The equation for $\kappa$ is in fact
\[
\frac{f^{2}(f^{2}+\theta)}{2q + 2 + \Omega + \tfrac{1}{2}\kappa(f^{2}+\theta)} + \cdots + \frac{h^{2}(h^{2}+\theta)}{2q + 2 + \Omega + \tfrac{1}{2}\kappa(h^{2}+\theta)} + 1 = 0;
\]
and the values of $A, B, \ldots, C, D$ are then
\[
A = \frac{M f^{2}}{2q + 2 + \Omega + \tfrac{1}{2}\kappa(f^{2}+\theta)},\quad
B = \frac{M g^{2}}{2q + 2 + \Omega + \tfrac{1}{2}\kappa(g^{2}+\theta)},\ \ldots,\
C = \frac{M h^{2}}{2q + 2 + \Omega + \tfrac{1}{2}\kappa(h^{2}+\theta)},\quad
D = -\frac{M}{\kappa},
\]
values which seem to be dependent on $\theta$: if they were so, it would be fatal to the success of the process; but they are really independent of $\theta$.

\textbf{119.} That they are independent of $\theta$ depends on the theorems; that we have
\[
\kappa = \frac{(2q + 2 + \Omega)\kappa_{0}}{2q + 2 - \tfrac{1}{2}\kappa_{0}\theta},
\]
where $\kappa_{0}$ is a quantity independent of $\theta$ determined by the equation
\[
\frac{f^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0} f^{2}} + \frac{g^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0} g^{2}} + \cdots + \frac{h^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0} h^{2}} + 1 = 0
\]
($\kappa_{0}$ is in fact the value of $\kappa$ on writing $\theta = 0$): and that, omitting the arbitrary multiplier, the values of $A, B, \ldots, C, D$ then are
\[
\frac{f^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0} f^{2}},\quad \frac{g^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0} g^{2}},\ \ldots,\ \frac{h^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0} h^{2}},\quad -\frac{1}{\kappa_{0}},
\]
or, what is the same thing, the value of $\phi$ is
\[
\phi = \frac{f^{2}\alpha^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0} f^{2}} + \frac{g^{2}\beta^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0} g^{2}} + \cdots + \frac{h^{2}\gamma^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0} h^{2}} - \frac{1}{\kappa_{0}}.
\]

\textbf{120.} To explain the ground of the assumption
\[
\kappa = \frac{(2q + 2 + \Omega)\kappa_{0}}{2q + 2 - \tfrac{1}{2}\kappa_{0}\theta},
\]
observe that, assuming
\[
\frac{2q + 2 + \Omega + \tfrac{1}{2}\kappa(f^{2}+\theta)}{2q + 2 + \tfrac{1}{2}\kappa_{0} f^{2}} = \frac{2q + 2 + \Omega + \tfrac{1}{2}\kappa(g^{2}+\theta)}{2q + 2 + \tfrac{1}{2}\kappa_{0} g^{2}},
\]

\noindent\textbf{[p.~398]}\quad
then multiplying out and reducing, we obtain
\[
\bigl(g^{2} - f^{2}\bigr)\bigl\{\kappa_{0}(2q + 2 + \Omega) - (2q + 2)\kappa + \tfrac{1}{2}\kappa\kappa_{0}\theta\bigr\} = 0;
\]
viz.\ the equation divides out by the factor $g^{2} - f^{2}$, thereby becoming
\[
\kappa_{0}(2q + 2 + \Omega) - (2q + 2)\kappa + \tfrac{1}{2}\kappa\kappa_{0}\theta = 0,
\]
that is, it gives for $\kappa$ the foregoing value: hence clearly, $\kappa$ having this value, we obtain by symmetry
\[
2q + 2 + \Omega + \tfrac{1}{2}\kappa(f^{2}+\theta),\ 2q + 2 + \Omega + \tfrac{1}{2}\kappa(g^{2}+\theta),\ \ldots,\ 2q + 2 + \Omega + \tfrac{1}{2}\kappa(h^{2}+\theta),
\]
proportional to
\[
2q + 2 + \tfrac{1}{2}\kappa_{0} f^{2},\ 2q + 2 + \tfrac{1}{2}\kappa_{0} g^{2},\ \ldots,\ 2q + 2 + \tfrac{1}{2}\kappa_{0} h^{2};
\]
viz.\ the ratios, not only of $A : B$, but of $A : B : \cdots : C$ will be independent of $\theta$.

\textbf{121.} To complete the transformation, starting with the foregoing value of $\kappa$, we have
\[
2q + 2 + \Omega + \tfrac{1}{2}\kappa(f^{2}+\theta) = (2q + 2 + \Omega)\,\frac{2q + 2 + \tfrac{1}{2}\kappa_{0}(f^{2}+\theta)}{2q + 2 - \tfrac{1}{2}\kappa_{0}\theta},\ \&c.;
\]
so that we have
\[
A\bigl\{2q + 2 + \tfrac{1}{2}\kappa_{0} f^{2}\bigr\} + \kappa_{0} f^{2} D = 0,
\]
\[
B\bigl\{2q + 2 + \tfrac{1}{2}\kappa_{0} g^{2}\bigr\} + \kappa_{0} g^{2} D = 0,
\]
\[
C\bigl\{2q + 2 + \tfrac{1}{2}\kappa_{0} h^{2}\bigr\} + \kappa_{0} h^{2} D = 0,
\]
and
\[
\frac{A}{f^{2}+\theta} + \frac{B}{g^{2}+\theta} + \cdots + \frac{C}{h^{2}+\theta} = \frac{(2q + 2 + \Omega)\kappa_{0} D}{2q + 2 - \tfrac{1}{2}\kappa_{0}\theta}.
\]
Substituting for $A, B, \ldots, C$ their values, this last becomes
\[
\frac{\kappa_{0} f^{2} D}{2q + 2 + \tfrac{1}{2}\kappa_{0}\theta}\Bigl\{\frac{2q + 2}{2q + 2 + \tfrac{1}{2}\kappa_{0} f^{2}} - \cdots - \frac{\kappa_{0} h^{2} D}{h^{2}+\theta}\Bigr\}
+ \frac{2q + 2}{2q + 2 - \tfrac{1}{2}\kappa_{0}\theta}\Bigl\{\frac{2q + 2}{2q + 2 + \tfrac{1}{2}\kappa_{0} h^{2}} - \frac{\theta}{h^{2}+\theta}\Bigr\}\,;
\]
viz.\ this is
\[
\Bigl\{\frac{2q + 2}{2q + 2 + \tfrac{1}{2}\kappa_{0} f^{2}} - \cdots\Bigr\}\bigl(2q + 2\bigr)\kappa_{0} D,
\]
or, substituting for $\kappa$ its value, and dividing out by $2q + 2$, we have
\[
\frac{1}{2q + 2 + \tfrac{1}{2}\kappa_{0} f^{2}} + \frac{1}{2q + 2 + \tfrac{1}{2}\kappa_{0} g^{2}} + \cdots + \frac{1}{2q + 2 + \tfrac{1}{2}\kappa_{0} h^{2}} = -1,
\]
the equation for the determination of $\kappa_{0}$.

\noindent\textbf{[p.~399]}\quad
\textbf{122.} The equation for $\kappa_{0}$ is of the order $s$; there are consequently $s$ functions of the form in question, and each of the terms $\alpha^{2}, \beta^{2}, \ldots, \gamma^{2}$ can be expressed as a linear function of these. It thus appears that any quadric function of $\alpha, \beta, \ldots, \gamma$ can be expressed as a sum of Greenian functions; viz.\ the form is
\[
\begin{aligned}
& A \alpha^{2} + B\beta^{2} + \&c.\\
& + C'\alpha\beta + \&c.\\
& + D'\Bigl(\frac{f^{2}\alpha^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0}' f^{2}} + \frac{g^{2}\beta^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0}' g^{2}} + \cdots + \frac{h^{2}\gamma^{2}}{2q + 2 + \tfrac{1}{2}\kappa_{0}' h^{2}} - \frac{1}{\kappa_{0}'}\Bigr)\\
& + D''\bigl(\;,\quad\ldots,\quad\ldots,\quad\ldots\bigr)\\
& (s\text{ lines}),
\end{aligned}
\]
viz.\ the terms multiplied by $D', D'',\&c.$ respectively are those answering to the roots $\kappa_{0}', \kappa_{0}'', \ldots$ of the equation in $\kappa_{0}$.

The general conclusion is that any rational and integral function of $\alpha, \beta, \ldots, \gamma$ can be expressed as a sum of Greenian functions.

\textbf{129.} First example: relating to the $(s + 1)$-coordinal sphere $a^{2} + \cdots + z^{2} + w^{2} = f^{2}$.
Assume
\[
V' = \frac{M}{\bigl(a^{2} + \cdots + c^{2} + e^{2}\bigr)^{\tfrac{1}{2}(s-1)}},\qquad
V'' = \frac{M}{f^{s-1}}\quad(\text{a constant});
\]
these values each satisfy the potential equation.

$V'$ is not infinite for any point outside the surface, and for indefinitely large distances it is of the proper form.

$V''$ is not infinite for any point inside the surface; and at the surface $V' = V''$.
The conditions of the theorem are therefore satisfied. Writing
\[
V = \int \frac{\rho\,dS}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + (e-w)^{2}\bigr\}^{\tfrac{1}{2}(s-1)}}\,,
\]
we have
\[
\rho = \frac{1}{4(\sqrt{\pi})^{s+1}}\Bigl(\frac{dV'}{dn'} - \frac{dV''}{dn''}\Bigr),
\]
where
\[
\frac{dV''}{dn''} = -\Bigl(\tfrac{x}{f}\,\tfrac{d}{dx} + \cdots + \tfrac{z}{f}\,\tfrac{d}{dz} + \tfrac{w}{f}\,\tfrac{d}{dw}\Bigr)\,\frac{M}{(x^{2} + \cdots + z^{2} + w^{2})^{\tfrac{1}{2}(s-1)}}
= \frac{(s-1)M(x^{2} + \cdots + z^{2} + w^{2})^{\tfrac{1}{2}(s-1)}}{f\cdot(x^{2} + \cdots + z^{2} + w^{2})^{\tfrac{1}{2}(s+1)}},
\]

\noindent\textbf{[p.~405]}\quad
which at the surface is
\[
= \frac{(s-1)M}{f^{s}}.
\]
Hence
\[
\rho = \frac{(s-1)\Gamma(\tfrac{1}{2}(s+1))M}{4(\sqrt{\pi})^{s+1}f^{s}} = \frac{\Gamma(\tfrac{1}{2}s + 1)M}{2(\sqrt{\pi})^{s+1}f^{s}}\quad(\text{viz.\ }\rho\text{ is constant}).
\]

\textbf{130.} Writing for convenience $M = \dfrac{2(\sqrt{\pi})^{s+1}}{\Gamma(\tfrac{1}{2}(s+1))}\,\delta\!f^{s}\,Sf^{s}$ ($\delta\!f^{s}$ a constant which may be put $= 1$), also $a^{2} + \cdots + c^{2} + e^{2} = s^{2}$, we have $\rho = Sf^{s}$, and consequently
\[
\int \frac{\delta\!f^{s}\,dS}{\bigl\{(a-x)^{2} + \cdots + (c-z)^{2} + (e-w)^{2}\bigr\}^{\tfrac{1}{2}(s-1)}}
= \frac{2(\sqrt{\pi})^{s+1}}{\Gamma(\tfrac{1}{2}(s+1))}\cdot\frac{f^{s}}{s^{s-1}},\quad\text{for exterior point }s > f,
\]
\[
= \frac{2(\sqrt{\pi})^{s+1}}{\Gamma(\tfrac{1}{2}(s+1))}\cdot f^{s-1},\quad\text{for interior point }s < f.
\]
By making $a, \ldots, c, e$ all indefinitely large, we find
\[
\int \delta\!f^{s}\,dS = \frac{2(\sqrt{\pi})^{s+1}}{\Gamma(\tfrac{1}{2}(s+1))}\,f^{s},
\]
viz.\ the expression on the right-hand side is here the mass of the shell thickness $\delta\!f$.

Taking $s = 3$, we have the ordinary formulae for the Potential of a uniform spherical shell.
