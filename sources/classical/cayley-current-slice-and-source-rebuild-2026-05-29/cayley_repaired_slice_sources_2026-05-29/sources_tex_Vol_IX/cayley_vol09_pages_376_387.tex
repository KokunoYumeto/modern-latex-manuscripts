\noindent\textbf{60.}\quad It is now easy to see in general how the foregoing transformed value
$\tfrac{1}{2}e^{-q}\displaystyle\int_{X}^{\infty} x^{q-1}(1-x)^{\frac{1}{2}s-1}\,dx$, where $q$ is negative and fractional, gives at once the value of

\bigskip

\noindent\textbf{[p.~357]}\quad
the term in $e^{-q}$. Observe that in the integral $x$ is always between $1$ and $X$ $\left(=\dfrac{e^{2}}{e^{2}+R^{2}}\right)$, a positive quantity less than $1$; the function to be integrated never becomes infinite. Imagine for a moment an integral $\displaystyle\int_{X}^{\infty} x^{a}\,dx$, where $a$ is positive or negative. We may conventionally write this $= \displaystyle\int_{0}^{\infty} x^{a}\,dx - \int_{0}^{X} x^{a}\,dx$, understanding the first symbol to mean $\dfrac{1^{1+a}}{1+a}$, and the second to mean $\dfrac{X^{1+a}}{1+a}$; they of course properly mean $\dfrac{1^{1+a} - 0^{1+a}}{1+a}$ and $\dfrac{X^{1+a} - 0^{1+a}}{1+a}$; but the terms in $0^{1+a}$, whether zero or infinite, destroy each other, the original form $\displaystyle\int_{X}^{\infty} x^{a}\,dx$, in fact, showing that no such terms can appear in the result.

In accordance with the convention, we write
\[
\int_{X}^{\infty} x^{q-1}(1-x)^{\frac{1}{2}s-1}\,dx
= \int_{0}^{\infty} x^{q-1}(1-x)^{\frac{1}{2}s-1}\,dx - \int_{0}^{X} x^{q-1}(1-x)^{\frac{1}{2}s-1}\,dx;
\]
and it follows that the term in $e^{-q}$ is
\[
\tfrac{1}{2}e^{-q}\int_{0}^{\infty} x^{q-1}(1-x)^{\frac{1}{2}s-1}\,dx,
\]
this last expression (wherein $q$, it will be remembered, is a negative fraction) being understood according to the convention; and so understanding it, the value of the term is
\[
= \tfrac{1}{2}e^{-q}\,\frac{\Gamma q\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}s+q)},
\]
where the $\Gamma$ of the negative $q$ is to be interpreted in accordance with the equation
$\Gamma(q+1)=q\Gamma q$; viz.\ we have $\Gamma q = \dfrac{1}{q}\,\Gamma(q+1)$, $= \dfrac{1}{q(q+1)}\,\Gamma(q+2)$, \&c., so as to make the argument of the $\Gamma$ positive. Observe that under this convention we have
\[
\Gamma q\,\Gamma(1-q) = \frac{\Gamma(\tfrac{1}{2})^{2}}{\sin q\pi} = \frac{\pi}{\sin q\pi};
\quad\text{or the term is}\quad
\tfrac{1}{2}e^{-q}\,\frac{\pi}{\sin q\pi}\,\frac{\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}s+q)\,\Gamma(1-q)}.
\]

\bigskip

\noindent\textbf{61.}\quad An example in which $-\tfrac{1}{2}s-q-1$ is integral will make the process clearer, and will serve instead of a general proof. Suppose $q=-\tfrac{7}{2}$, $\tfrac{1}{2}s=\tfrac{1}{4}$, the expression
\[
\int_{0}^{1} x^{-\frac{9}{2}}(1-x)^{\frac{1}{4}}\,dx
= \int_{0}^{1}\bigl(x^{-\frac{9}{2}} - \tfrac{9}{2}\,x^{-\frac{7}{2}} + \tfrac{63}{8}\,x^{-\frac{5}{2}} - \tfrac{63}{16}\,x^{-\frac{3}{2}} + \cdots\bigr)dx
\]
is used, in accordance with the convention, to denote the value
\[
= -\tfrac{2}{7} + \tfrac{2}{5}\cdot\tfrac{9}{2} - \tfrac{2}{3}\cdot\tfrac{63}{8} + \tfrac{2}{1}\cdot\tfrac{63}{16}
= 7\bigl(-\tfrac{2}{7\cdot 1} + \tfrac{2}{5\cdot 3} - \tfrac{2}{3\cdot 5} + \tfrac{2}{1\cdot 7}\bigr)
= \frac{-7\cdot 2401}{5\cdot 13\cdot 27} = \frac{-7^{4}}{5\cdot 13\cdot 27}.
\]

\bigskip

\noindent\textbf{[p.~358]}\quad But we have
\[
\frac{\Gamma\tfrac{1}{2}s\,\Gamma q}{\Gamma(\tfrac{1}{2}s+q)}
= \frac{\Gamma\tfrac{5}{4}\,\Gamma(-\tfrac{7}{2})}{\Gamma(\tfrac{5}{4}-\tfrac{7}{2})}
= \frac{2^{4}\,\Gamma(-\tfrac{7}{2})}{\tfrac{1}{4}\cdot\tfrac{5}{4}\cdot\tfrac{9}{4}\cdot\tfrac{13}{4}\cdot\Gamma(-\tfrac{7}{2})}
= \frac{2^{4}\cdot 4^{4}}{5\cdot 13\cdot 27} = \frac{-7^{4}}{5\cdot 13\cdot 27},
\]
agreeing with the former value.

\bigskip

\noindent\textbf{62.}\quad The case of a negative integer is more simple. To find the logarithmic term of
\[
\tfrac{1}{2}e^{-q}\int_{X}^{\infty} x^{q-1}(1-x)^{\frac{1}{2}s-1}\,dx,
\]
we have only to expand the factor $(1-x)^{\frac{1}{2}s-1}$ so as to obtain the term involving $x^{-q}$.

We have thus the term
\[
\tfrac{1}{2}e^{-q}\,(-)^{-q}\,\frac{\Gamma\tfrac{1}{2}s}{\Gamma(1-q)\,\Gamma(\tfrac{1}{2}s+q)}\,e^{-q}\log\tfrac{1}{X},
\]
where $\log\tfrac{1}{X} = \log\left(1+\dfrac{R^{2}}{e^{2}}\right) = 2\log\dfrac{R}{e} + 2\log\sqrt{1+\dfrac{e^{2}}{R^{2}}}$; so that, neglecting the terms in $\dfrac{e^{2}}{R^{2}}$, \&c., this is $2\log\dfrac{R}{e}$, and the term in question is
\[
= (-)^{q}e^{-q}\,\frac{\Gamma\tfrac{1}{2}s}{\Gamma(1-q)\,\Gamma(\tfrac{1}{2}s+q)}\,\log\frac{R}{e}.
\]

The general conclusion is that $q$ being negative, the $r$-integral
\[
\int_{0}^{R}\frac{r^{s-1}\,dr}{(r^{2}+e^{2})^{\frac{1}{2}s+q}}
\]
has for its value a series proceeding in powers of $e^{2}$, which series up to a certain point is equal to the series obtained by expanding in ascending powers of $e^{2}$ and integrating each term separately; viz.\ the series to the point in question is
\[
\frac{R^{-2q}}{-2q} - \frac{\tfrac{1}{2}s+q}{1}\cdot\frac{R^{-2q-2}}{-2q-2}\,e^{2}
+ \frac{\tfrac{1}{2}s+q\cdot\tfrac{1}{2}s+q+1}{1\cdot 2}\cdot\frac{R^{-2q-4}}{-2q-4}\,e^{4} - \cdots,
\]
continued so long as the exponent of $e$ is less than $-2q$: together with a term $Ke^{-q}$ when $q$ is fractional, and $Ke^{-q}\log\dfrac{R}{e}$ when $q$ is integral; viz.\ $q$ fractional, this term is
\[
= \tfrac{1}{2}e^{-q}\,\frac{\Gamma q\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}s+q)}
= \tfrac{1}{2}e^{-q}\,\frac{\pi}{\sin q\pi}\,\frac{\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}s+q)\,\Gamma(1-q)},
\]
and $q$ integral, it is
\[
= (-)^{q}e^{-q}\,\frac{\Gamma\tfrac{1}{2}s}{\Gamma(1-q)\,\Gamma(\tfrac{1}{2}s+q)}\,\log\frac{R}{e}.
\]

\noindent\textbf{[p.~359]}\quad
\textbf{63.}\quad It has been tacitly assumed that $\tfrac{1}{2}s+q$ is positive; but the formulae hold good if $\tfrac{1}{2}s+q$ is $=0$ or negative. Suppose $\tfrac{1}{2}s+q$ is $=0$ or a negative integer, then $\Gamma(\tfrac{1}{2}s+q)=\infty$, and the special term involving $e^{-q}$ or $e^{-q}\log e$ vanishes; in fact, in this case the $r$-integral is
\[
= \int_{0}^{R} r^{s-1}\,(r^{2}+e^{2})^{-(\frac{1}{2}s+q)}\,dr,
\]
where $(r^{2}+e^{2})^{-(\frac{1}{2}s+q)}$ has for its value a finite series, and the integral is therefore equal to a finite series $A + Be^{2} + Ce^{4} + \&c$. If $\tfrac{1}{2}s+q$ be fractional, then the $\Gamma$ of the negative quantity $\tfrac{1}{2}s+q$ must be understood as above, or, what is the same thing, we may, instead of $\Gamma(\tfrac{1}{2}s+q)$, write
\[
\frac{\pi}{\sin(\tfrac{1}{2}s+q)\pi\cdot\Gamma(1-q-\tfrac{1}{2}s)};
\]
thus, $q$ being integral, the exceptional term is
\[
= \frac{(-)^{q}\sin(\tfrac{1}{2}s+q)\pi\cdot\Gamma(1-q-\tfrac{1}{2}s)}{\Gamma(\tfrac{1}{2}s)\,\Gamma(1-q)}\,\Gamma\tfrac{1}{2}s \cdot R^{-2q}e^{-q}.
\]

For instance, $s=1$, $q=-2$, the term is
\[
\tfrac{1}{2}\,\frac{\Gamma\tfrac{1}{2}\sin(-\tfrac{3}{2})\pi\cdot\Gamma\tfrac{5}{2}}{\Gamma\tfrac{1}{2}\cdot\Gamma 3}\,e^{2}\log\frac{R}{e},
\]
or, since $\Gamma\tfrac{5}{2} = \tfrac{3}{2}\cdot\tfrac{1}{2}\,\Gamma\tfrac{1}{2}$, and $\Gamma 3 = 2$, the term is $+\tfrac{3}{8}e^{2}\log\dfrac{R}{e}$, agreeing with a preceding result.

\noindent\textbf{64.}\quad The prepotentials in question depend ultimately upon two integrals, which also arise, as will presently appear, from prepotential problems in two-dimensional space, and which are for convenience termed the \emph{ring-integral} and the \emph{disk-integral} respectively. The analytical investigation in regard to these, depending as it does on a transformation of a function allied with the hypergeometric series, is I think interesting.

\bigskip

\noindent\textbf{65.}\quad Consider first the prepotential of a uniform $(s+1)$-dimensional spherical shell. This is
\[
V = \int\frac{dS}{\{(a-x)^{2}+\ldots+(c-z)^{2}+(e-w)^{2}\}^{\frac{1}{2}s+q}},
\]
the equation of the surface being $x^{2}+\ldots+z^{2}+w^{2}=f^{2}$; and there are the two cases of an internal point, $a^{2}+\ldots+c^{2}+e^{2}<f^{2}$, and an external point, $a^{2}+\ldots+c^{2}+e^{2}>f^{2}$.

The value is a function of $a^{2}+\ldots+c^{2}+e^{2}$, say this is $=\kappa^{2}$. Taking the axes so that the coordinates of the attracted point are $(0,\ldots,0,\kappa)$, the integral is
\[
= \int\frac{dS}{\{x^{2}+\ldots+z^{2}+(\kappa-w)^{2}\}^{\frac{1}{2}s+q}},
\]

\bigskip

\noindent\textbf{[p.~360]}\quad where the equation of the surface is still $x^{2}+\ldots+z^{2}+w^{2}=f^{2}$. Writing $x=f\xi,\ldots,z=f\zeta$, $w=f\omega$, where $\xi^{2}+\ldots+\zeta^{2}+\omega^{2}=1$, we have $dS=f^{s}\,\dfrac{d\xi\ldots d\zeta}{\omega}$, or the integral is
\[
= f^{s}\int\frac{d\xi\ldots d\zeta}{\omega\,(f^{2}-2\kappa f\omega+\kappa^{2})^{\frac{1}{2}s+q}}.
\]

Assume $\xi=\rho x,\ldots,\zeta=\rho z$, where $x^{2}+\ldots+z^{2}=1$; then $\rho^{2}+\omega^{2}=1$. Moreover, $d\xi\ldots d\zeta = \rho^{s-1}\,d\rho\,d\Sigma$, where $d\Sigma$ is the element of surface of the $s$-dimensional unit-sphere $x^{2}+\ldots+z^{2}=1$; or for $\rho$, substituting its value $\sqrt{1-\omega^{2}}$, we have $d\rho = -\dfrac{\omega\,d\omega}{\sqrt{1-\omega^{2}}}$; and thence $d\xi\ldots d\zeta = -(1-\omega^{2})^{\frac{1}{2}s-\frac{3}{2}}\,\omega\,d\omega\,d\Sigma$. The integral as regards $\rho$ is from $\rho=-1$ to $+1$, or as regards $\omega$ from $1$ to $-1$; whence reversing the sign, the integral will be from $\omega=-1$ to $+1$; and the required integral is thus
\[
= f^{s}\int d\Sigma\int_{-1}^{+1}\frac{(1-\omega^{2})^{\frac{1}{2}s-\frac{3}{2}}\,d\omega\,d\Sigma}{(f^{2}-2\kappa f\omega+\kappa^{2})^{\frac{1}{2}s+q}}
= f^{s}\int d\Sigma\int_{-1}^{+1}\frac{(1-\omega^{2})^{\frac{1}{2}s-\frac{3}{2}}\,d\omega}{(f^{2}-2\kappa f\omega+\kappa^{2})^{\frac{1}{2}s+q}},
\]
where $\displaystyle\int d\Sigma$ is the surface of the $s$-dimensional unit-sphere (see Annex I.), $=\dfrac{2\,(\Gamma\tfrac{1}{2})^{s}}{\Gamma(\tfrac{1}{2}s)}$; and for greater convenience transforming the second factor by writing therein $\omega=\cos\theta$, the required integral is $=\dfrac{2\,(\Gamma\tfrac{1}{2})^{s}}{\Gamma(\tfrac{1}{2}s)}\,f^{s}$ multiplied by
\[
\int_{0}^{\pi}\frac{\sin^{s-1}\theta\,d\theta}{(f^{2}-2\kappa f\cos\theta+\kappa^{2})^{\frac{1}{2}s+q}},
\]
which last expression including the factor $2f^{s}$, but without the factor $\dfrac{(\Gamma\tfrac{1}{2})^{s}}{\Gamma(\tfrac{1}{2}s)}$, is the ring-integral discussed in the present Annex. It may be remarked that the value can be at once obtained in the particular case $s=2$, which belongs to tridimensional space; viz.\ we then have
\[
V = \int_{0}^{\pi}\frac{2\pi f^{2}\sin\theta\,d\theta}{(f^{2}-2f\kappa\cos\theta+\kappa^{2})^{q+1}}
= \frac{2\pi f}{\kappa(1-q)}\bigl|\,(f^{2}-2f\kappa\cos\theta+\kappa^{2})^{-q+1}\bigr|,
\]
which agrees with a result given, \textit{M\'ecanique C\'eleste}, Book \textsc{xii.} Chap.\ \textsc{ii.}

\bigskip

\noindent\textbf{66.}\quad Consider next the prepotential of the uniform solid $(s+1)$-dimensional sphere,
\[
V = \int\frac{dx\ldots dz\,dw}{\{(a-x)^{2}+\ldots+(c-z)^{2}+(e-w)^{2}\}^{\frac{1}{2}s+q}},
\]
the equation of the surface being $x^{2}+\ldots+z^{2}+w^{2}=f^{2}$; there are the two cases of an internal point $\kappa<f$, and an external point $\kappa>f$ ($a^{2}+\ldots+c^{2}+e^{2}=\kappa^{2}$ as before).

\bigskip

\noindent\textbf{[p.~361]}\quad Transforming so that the coordinates of the attracted point are $0,\ldots,0,\kappa$, the integral is
\[
= \int\frac{dx\ldots dz\,dw}{\{x^{2}+\ldots+z^{2}+(\kappa-w)^{2}\}^{\frac{1}{2}s+q}},
\]
where the equation is still $x^{2}+\ldots+z^{2}+w^{2}=f^{2}$. Writing here $x=r\xi,\ldots,z=r\zeta$, where $\xi^{2}+\ldots+\zeta^{2}=1$, we have $dx\ldots dz = r^{s-1}\,dr\,dS$, where $dS$ is an element of surface of the $s$-dimensional unit-sphere $\xi^{2}+\ldots+\zeta^{2}=1$; the integral is therefore
\[
= \int\int\frac{r^{s-1}\,dr\,dS\,dw}{\{r^{2}+(\kappa-w)^{2}\}^{\frac{1}{2}s+q}},
\]
where, as regards $r$ and $w$, the integration extends over the circle $r^{2}+w^{2}=f^{2}$. The value of the first factor (see Annex I.) is $=\dfrac{2\,(\Gamma\tfrac{1}{2})^{s}}{\Gamma(\tfrac{1}{2}s)}$; writing $y$ and $x$ in place of $r$ and $w$ respectively, the integral is $=\dfrac{2\,(\Gamma\tfrac{1}{2})^{s}}{\Gamma(\tfrac{1}{2}s)}$ multiplied by
\[
\int\frac{y^{s-1}\,dx\,dy}{\{(x-\kappa)^{2}+y^{2}\}^{\frac{1}{2}s+q}}
\]
over the circle $x^{2}+y^{2}=f^{2}$; viz.\ this last expression (without the factor $\dfrac{(\Gamma\tfrac{1}{2})^{s}}{\Gamma(\tfrac{1}{2}s)}$) is the disk-integral discussed in the present Annex.

\bigskip

\noindent\textbf{67.}\quad We find, for the value in regard to an internal point $\kappa<f$,
\[
V = \frac{(\Gamma\tfrac{1}{2})^{s+1}}{\Gamma(\tfrac{1}{2}s+q)\,\Gamma(\tfrac{1}{2}-q)}\,f^{s+1}\int_{0}^{\infty}(t+f^{2}-\kappa^{2})^{q-\frac{1}{2}}\,t^{-q-1}\,(t+f^{2})^{\frac{1}{2}s+q-\frac{1}{2}}\,dt,
\]
which, in the particular case $q=-\tfrac{1}{2}$, is
\[
= \frac{(\Gamma\tfrac{1}{2})^{s+1}}{\Gamma(\tfrac{1}{2}s-\tfrac{1}{2})}\,f^{s+1}\int_{0}^{\infty}(t+f^{2}-\kappa^{2})^{-1}\,(t+f^{2})^{\frac{1}{2}s-1}\,dt;
\]
viz.\ the integral in $t$ is here
\[
= \int_{f^{2}-\kappa^{2}}^{\infty}(t+f^{2}-\kappa^{2})^{-1}\,t^{-\frac{1}{2}}\,(t+f^{2})^{\frac{1}{2}s-1}\,dt
= \frac{1}{f^{s+1}}\left(\frac{f^{s+1}}{\tfrac{1}{2}s-\tfrac{1}{2}} - \frac{\kappa^{s+1}}{\tfrac{1}{2}s+\tfrac{1}{2}}\right),
\]
or we have
\[
V = \frac{(\Gamma\tfrac{1}{2})^{s+1}}{\Gamma(\tfrac{1}{2}s-\tfrac{1}{2})}\left(\frac{f^{s+1}}{\tfrac{1}{2}s-\tfrac{1}{2}} - \frac{\kappa^{s+1}}{\tfrac{1}{2}s+\tfrac{1}{2}}\right).
\]
It may be added that, in regard to an external point $\kappa>f$, the value is
\[
V = \frac{(\Gamma\tfrac{1}{2})^{s+1}}{\Gamma(\tfrac{1}{2}s+q)\,\Gamma(\tfrac{1}{2}-q)}\,f^{s+1}\int_{\kappa^{2}-f^{2}}^{\infty}(t+f^{2}-\kappa^{2})^{q-\frac{1}{2}}\,t^{-q-1}\,(t+f^{2})^{\frac{1}{2}s+q-\frac{1}{2}}\,dt,
\]

\bigskip

\noindent\textbf{[p.~362]}\quad which, in the same case $q=-\tfrac{1}{2}$, is
\[
= \frac{(\Gamma\tfrac{1}{2})^{s+1}}{\Gamma(\tfrac{1}{2}s-\tfrac{1}{2})}\,f^{s+1}\int_{\kappa^{2}-f^{2}}^{\infty}(t+f^{2}-\kappa^{2})^{-1}\,(t+f^{2})^{\frac{1}{2}s-1}\,dt,
\]
where the $t$-integral is
\[
= \int_{\kappa^{2}-f^{2}}^{\infty}(t+f^{2}-\kappa^{2})^{-1}\,t^{-\frac{1}{2}}\,(t+f^{2})^{\frac{1}{2}s-1}\,dt
= \frac{\kappa^{s+1}}{\tfrac{1}{2}s-\tfrac{1}{2}}\cdot\frac{\kappa^{s-2}}{\tfrac{1}{2}s+\tfrac{1}{2}}\cdot\frac{1}{\kappa^{s+1}} = \frac{\kappa^{s-1}}{\tfrac{1}{2}s-\tfrac{1}{2}\cdot\tfrac{1}{2}s+\tfrac{1}{2}};
\]
and the value of $V$ is therefore
\[
= \frac{(\Gamma\tfrac{1}{2})^{s+1}}{\Gamma(\tfrac{1}{2}s+\tfrac{1}{2})}\,\frac{f^{s+1}}{\kappa^{s-1}}.
\]

Recurring to the case of the internal point; then, writing $\nabla = \dfrac{d^{2}}{da^{2}}+\ldots+\dfrac{d^{2}}{dc^{2}}+\dfrac{d^{2}}{de^{2}}$, and observing that $\nabla\,(\kappa^{s}) = s\,(\tfrac{1}{2}s+\tfrac{1}{2})$, we have
\[
\nabla V = -\frac{4\,(\Gamma\tfrac{1}{2})^{s+1}}{\Gamma(\tfrac{1}{2}s-\tfrac{1}{2})};
\]
(in particular, for ordinary space $s+1=3$, or the value is $\dfrac{-4\pi^{2}}{\sqrt{\pi}}=-4\pi$, which is right).

\bigskip

\noindent\textbf{68.}\quad The integrals referred to as the ring-integral and the disk-integral arise also from the following integrals in two-dimensional space, viz.\ these are
\[
\int\frac{y^{s-1}\,dS}{\{(x-\kappa)^{2}+y^{2}\}^{\frac{1}{2}s+q}}\,,\qquad
\int\frac{y^{s-1}\,dx\,dy}{\{(x-\kappa)^{2}+y^{2}\}^{\frac{1}{2}s+q}}\,,
\]
in the first of which $dS$ denotes an element of arc of the circle $x^{2}+y^{2}=f^{2}$, the integration being extended over the whole circumference, and in the second the integration extends over the circle $x^{2}+y^{2}=f^{2}$; $y^{s-1}$ is written for shortness instead of $(y^{2})^{\frac{1}{2}(s-1)}$, viz.\ this is considered as always positive, whether $y$ is positive or negative; it is moreover assumed that $s-1$ is zero or positive.

Writing in the first integral $x=f\cos\theta$, $y=f\sin\theta$, the value is
\[
= f^{s}\int\frac{(\sin\theta)^{s-1}\,d\theta}{(f^{2}-2\kappa f\cos\theta+\kappa^{2})^{\frac{1}{2}s+q}},
\]
viz.\ this represents the prepotential of the circumference of the circle, density varying as $(\sin\theta)^{s-1}$, in regard to a point $x=\kappa$, $y=0$ in the plane of the circle; and similarly the second integral represents the prepotential of the circular disk, density of the element at the point $(x,y)=y^{s-1}$, in regard to the same point $x=\kappa$, $y=0$; it being in each case assumed that the prepotential of an element of mass $\rho\,d\varpi$ at a point at distance $d$ is $=\dfrac{\rho\,d\varpi}{d^{2q+s}}$.

\noindent\textbf{[p.~363]}\quad
\textbf{69.}\quad In the case of the circumference, it is assumed that the attracted point is not on the circumference, $\kappa$ not $=f$; and the function under the integral sign, and therefore the integral itself, is in every case finite. In the case of the circle, if $\kappa$ be an interior point, then if $2q-1$ be $=0$ or positive, the element at the attracted point becomes infinite; but to avoid this we consider, not the potential of the whole circle, but the potential of the circle less an indefinitely small circle radius $\epsilon$ having the attracted point for its centre; which being so, the element under the integral sign, and consequently the integral itself, remains finite.

It is to be remarked that the two integrals are connected with each other; viz.\ the circle of the second integral being divided into rings by means of a system of circles concentric with the bounding circle $x^{2}+y^{2}=f^{2}$, then the prepotential of each ring or annulus is determined by an integral such as the first integral; or, analytically, writing in the second integral $x=r\cos\theta$, $y=r\sin\theta$, and therefore $dx\,dy = r\,dr\,d\theta$, the second integral is
\[
= \int r^{s}\,dr\int\frac{(\sin\theta)^{s-1}\,d\theta}{(r^{2}+\kappa^{2}-2f r\cos\theta)^{\frac{1}{2}s+q}},
\]
viz.\ the integral in regard to $\theta$ is here the same function of $r$, $\kappa$ that the first integral is of $f$, $\kappa$; and the integration in regard to $r$ is of course to be taken from $r=0$ to $r=f$. But the $\theta$-integral is not, in its original form, such a function of $r$ as to render possible the integration in regard to $r$; and I, in fact, obtain the second integral by a different and in some respects a better process.

\bigskip

\noindent\textbf{70.}\quad Consider first the ring-integral which, writing therein as above $x=f\cos\theta$, $y=f\sin\theta$, and multiplying by $2$ in order that the integral, instead of being taken from $0$ to $2\pi$, may be taken from $0$ to $\pi$, becomes
\[
= 2f^{s}\int_{0}^{\pi}\frac{(\sin\theta)^{s-1}\,d\theta}{(f^{2}+\kappa^{2}-2\kappa f\cos\theta)^{\frac{1}{2}s+q}}.
\]

Write $\cos\tfrac{1}{2}\theta = \sqrt{x}$; then $\sin\tfrac{1}{2}\theta = \sqrt{1-x}$, $\sin\theta = 2\sqrt{x(1-x)}$, $d\theta = -x^{-\frac{1}{2}}(1-x)^{-\frac{1}{2}}\,dx$, $\cos\theta = -1+2x$; $\theta = 0$ gives $x=1$, $\theta = \pi$ gives $x=0$, and the integral is
\[
= 2^{s-1}\int_{0}^{1}\frac{x^{\frac{1}{2}s-1}(1-x)^{\frac{1}{2}s-1}\,dx}{\{(f+\kappa)^{2} - 4f\kappa x\}^{\frac{1}{2}s+q}}
= \frac{2^{s-1}}{\{(f+\kappa)^{2}\}^{\frac{1}{2}s+q}}\int_{0}^{1}\frac{x^{\frac{1}{2}s-1}(1-x)^{\frac{1}{2}s-1}\,dx}{(1-ux)^{\frac{1}{2}s+q}},
\]
if for shortness $u = \dfrac{4\kappa f}{(\kappa+f)^{2}}$, so that obviously $u<1$.

The integral in $x$ is here an integral belonging to the general form
\[
\Pi = \Pi(\alpha,\beta,\gamma,u) = \int_{0}^{1} x^{\alpha-1}(1-x)^{\beta-1}(1-ux)^{-\gamma}\,dx,
\]
viz.\ we have
\[
\text{Ring-integral} = \frac{2^{s-1}f^{s}}{(f+\kappa)^{s+2q}}\,\Pi(\tfrac{1}{2}s,\tfrac{1}{2}s,\tfrac{1}{2}s+q,u).
\]

\bigskip

\noindent\textbf{[p.~364]}\quad
\textbf{71.}\quad The general function $\Pi(\alpha,\beta,\gamma,u)$ is
\[
\Pi(\alpha,\beta,\gamma,u) = \frac{\Gamma\alpha\,\Gamma\beta}{\Gamma(\alpha+\beta)}\,F(\alpha,\gamma,\alpha+\beta,u),
\]
or, what is the same thing,
\[
\Pi(\alpha,\beta,\gamma,u) = \frac{\Gamma\gamma}{\Gamma\alpha\,\Gamma(\gamma-\alpha)}\,F(\alpha,\beta,\gamma,u),
\]
and consequently transformable by means of various theorems for the transformation of the hypergeometric series, in particular, by the theorems
\[
F(\alpha,\beta,\gamma,u) = F(\beta,\alpha,\gamma,u),
\]
\[
F(\alpha,\beta,\gamma,u) = (1-u)^{\gamma-\alpha-\beta}\,F(\gamma-\alpha,\gamma-\beta,\gamma,u);
\]
and if $v = \left(\dfrac{1-\sqrt{1-u}}{1+\sqrt{1-u}}\right)^{2}$, or, what is the same thing, $u = \dfrac{4\sqrt{v}}{(1+\sqrt{v})^{2}}$, then
\[
F(\alpha,\beta,2\beta,u) = (1-\sqrt{v})^{2\alpha}\,F\bigl(\alpha,\,\alpha-\beta+\tfrac{1}{2},\,\beta+\tfrac{1}{2},\,v\bigr).
\]
In verification, observe that if $u=1$ then also $v=1$, and that with these values, calculating each side by means of the formulae
\[
F(\alpha,\beta,\gamma,1) = \frac{\Gamma\gamma\,\Gamma(\gamma-\alpha-\beta)}{\Gamma(\gamma-\alpha)\,\Gamma(\gamma-\beta)},
\]
the resulting equation $F(\alpha,\beta,2\beta,1) = 2^{2\alpha}\,F(\alpha,\alpha-\beta+\tfrac{1}{2},\beta+\tfrac{1}{2},1)$, becomes
\[
\frac{\Gamma 2\beta\,\Gamma(\beta-\alpha)}{\Gamma(2\beta-\alpha)\,\Gamma\beta}
= 2^{2\alpha}\,\frac{\Gamma(\beta+\tfrac{1}{2})\,\Gamma(2\beta-2\alpha)}{\Gamma(2\beta-\alpha)\,\Gamma(\beta-\alpha+\tfrac{1}{2})},
\]
that is,
\[
\frac{\Gamma 2\beta}{\Gamma\beta\,\Gamma(\beta+\tfrac{1}{2})}
= 2^{2\alpha}\,\frac{\Gamma(2\beta-2\alpha)}{\Gamma(\beta-\alpha)\,\Gamma(\beta-\alpha+\tfrac{1}{2})},
\]
which is true, in virtue of the relation $\dfrac{\Gamma 2x}{\Gamma x\,\Gamma(x+\tfrac{1}{2})} = \dfrac{2^{2x-1}}{\Gamma\tfrac{1}{2}}$.

\bigskip

\noindent\textbf{72.}\quad The foregoing formulae, and in particular the formula which I have written
\[
F(\alpha,\beta,2\beta,u) = (1-\sqrt{v})^{2\alpha}\,F(\alpha,\,\alpha-\beta+\tfrac{1}{2},\,\beta+\tfrac{1}{2},\,v),
\]
are taken from Kummer's Memoir, ``Ueber die hypergeometrische Reihe,'' \textit{Crelle}, t.\ \textsc{xv}.\ (1836), viz.\ the formula in question is, under a slightly different form, his formula (41), p.\ 76; the formula (43), p.\ 77, is intended to be equivalent thereto; but there is an error of transcription, $2\alpha-2\beta+1$, in place of $\alpha+\beta-1$, which makes the formula (43) erroneous.

It may be remarked as to the formulae generally that, although very probably $\Pi(\alpha,\beta,\gamma,u)$ may denote a proper function of $u$, whatever be the values of the indices $(\alpha,\beta,\gamma)$, and the various transformation-theorems hold good accordingly (the $\Gamma$-function of a negative argument being interpreted in the usual manner by means of the equation $\Gamma x = \dfrac{1}{x}\,\Gamma(1+x) = \dfrac{1}{x(x+1)}\,\Gamma(2+x)$, \&c.), yet that the function $\Pi(\alpha,\beta,\gamma,u)$,

\bigskip

\noindent\textbf{[p.~365]}\quad used as denoting the definite integral $\displaystyle\int_{0}^{1} x^{\alpha-1}(1-x)^{\beta-1}(1-ux)^{-\gamma}\,dx$, has no meaning except in the case where $\alpha$ and $\beta$ are each of them positive.

In what follows we obtain for the ring-integral and the disk-integral various expressions in terms of $\Pi$-functions, which are afterwards transformed into $t$-integrals with a superior limit $\infty$ and inferior limit $0$, or $=f^{2}-\kappa^{2}$; but for values of the variable index, $q$ lying beyond certain limits, the indices $\alpha$ and $\beta$, or one of them, of the $\Pi$-function will become negative, viz.\ the integral represented by the $\Pi$-function, or, what is the same thing, the $t$-integral, will cease to have a determinate value, and at the same time, or usually so, the argument or arguments of one or more of the $\Gamma$-functions will become negative. It is quite possible that in such cases the results are not without meaning, and that an interpretation for them might be found; but they have not any obvious interpretation, and we must in the first instance consider them as inapplicable.

\bigskip

\noindent\textbf{73.}\quad We require further properties of the $\Pi$-functions. Starting with the foregoing equation
\[
F(\alpha,\beta,2\beta,u) = (1+\sqrt{v})^{2\alpha}\,F\bigl(\alpha,\,\alpha-\beta+\tfrac{1}{2},\,\beta+\tfrac{1}{2},\,v\bigr),
\]
each side may be expressed in a fourfold form:
\begin{align*}
F(\alpha,\beta,2\beta,u) &\;=\; (1+\sqrt{v})^{2\alpha}\,F(\alpha,\,\alpha-\beta+\tfrac{1}{2},\,\beta+\tfrac{1}{2},\,v) \\
&\;=\; F(\beta,\alpha,2\beta,u) \;=\; \tfrac{1}{2}(1+\sqrt{v})^{2\alpha-1}\,F(\alpha-\tfrac{1}{2}+1,\,\beta+\tfrac{1}{2},\,v) \\
&\;=\; (1-u)^{\beta-\alpha}\,F(2\beta-\alpha,\,\beta,\,2\beta,\,u) \\
&\;=\; (1-u)^{\beta-\alpha}\,F(\alpha,\,2\beta-\alpha,\,2\beta,\,u) \\
&\;=\; (1+\sqrt{v})^{2\alpha}(1-v)^{2\beta-2\alpha-1}\,F(\beta-\alpha+\tfrac{1}{2},\,2\beta-\alpha,\,\beta+\tfrac{1}{2},\,v) \\
&\;=\; (1+\sqrt{v})^{2\alpha}(1-v)^{-\alpha-\beta-\frac{1}{2}}\,F(2\beta-\alpha,\,\beta-\alpha+\tfrac{1}{2},\,\beta+\tfrac{1}{2},\,v),
\end{align*}
where, instead of $(1+\sqrt{v})^{2\alpha}(1-v)^{2\beta-2\alpha-1}$, it is proper to write $(1+\sqrt{v})^{2\alpha}(1-\sqrt{v})^{2\beta-2\alpha-1}$;

And then to each form applying the transformation
\[
F(\alpha,\beta,\gamma,u) = \frac{\Gamma\gamma}{\Gamma\alpha\,\Gamma(\gamma-\alpha)}\,\Pi(\alpha,\gamma-\alpha,\beta,u),
\]
we have
\[
\frac{\Gamma 2\beta}{\Gamma\alpha\,\Gamma(2\beta-\alpha)}\,\Pi(\alpha,\,2\beta-\alpha,\,\beta,\,u)
\]
\[
= \frac{\Gamma 2\beta}{\Gamma\beta\,\Gamma\beta}\,\Pi(\beta,\beta,\alpha,u)
\]
\[
= (1-u)^{\beta-\alpha}\,\frac{\Gamma 2\beta}{\Gamma(2\beta-\alpha)\,\Gamma\alpha}\,\Pi(2\beta-\alpha,\,\alpha,\,\beta,\,u)
\]

\bigskip

\noindent\textbf{[p.~366]}\quad
\[
= (1-u)^{\beta-\alpha}\,2^{2\alpha-1}\,\frac{\Gamma(\beta+\tfrac{1}{2})}{\Gamma\tfrac{1}{2}\,\Gamma(2\beta-\alpha)}\,\Pi(\alpha,\,\alpha-\beta+\tfrac{1}{2},\,2\beta-\alpha,\,v)
\]
\[
= (1+\sqrt{v})^{2\alpha}(1-\sqrt{v})^{2\beta-2\alpha-1}\,\frac{\Gamma(\beta+\tfrac{1}{2})}{\Gamma(\beta-\alpha+\tfrac{1}{2})\,\Gamma\alpha}\,\Pi(\beta-\alpha+\tfrac{1}{2},\,\alpha,\,2\beta-\alpha,\,v).
\]

I select the second of the first four forms; equating it successively to each of the second four forms, and attending to the relation $\dfrac{\Gamma 2\beta}{\Gamma\beta\,\Gamma\beta} = 2^{2\beta-1}\,\dfrac{1}{\Gamma\tfrac{1}{2}}$, we find
\[
\Pi(\beta,\beta,\alpha,u) = (1-\sqrt{v})^{2\alpha}\,2^{1-2\alpha}\,\frac{\Gamma\alpha\,\Gamma\beta}{\Gamma(\beta-\alpha+\tfrac{1}{2})\,\Gamma\beta}\,\Pi(\alpha,\,\beta-\alpha+\tfrac{1}{2},\,\alpha-\beta+\tfrac{1}{2},\,v)
\]
\[
= (1+\sqrt{v})^{2\alpha}(1-\sqrt{v})^{2\beta-2\alpha-1}\,2^{1-2\alpha}\,\frac{\Gamma\alpha\,\Gamma\beta}{\Gamma(\beta-\alpha+\tfrac{1}{2})\,\Gamma\alpha}\,\Pi(\beta-\alpha+\tfrac{1}{2},\,\alpha,\,2\beta-\alpha,\,v),
\]
\[
= (1+\sqrt{v})^{2\alpha}(1-\sqrt{v})^{-\alpha-\beta+\frac{1}{2}}\,2^{-1}\,\frac{\Gamma\alpha\,\Gamma\beta}{\Gamma(2\beta-\alpha)\,\Gamma\alpha}\,\Pi(2\beta-\alpha,\,\alpha-\beta+\tfrac{1}{2},\,\beta-\alpha+\tfrac{1}{2},\,v).
\]

Putting herein $\beta=\tfrac{1}{2}s$, $\alpha=\tfrac{1}{2}s+q$, the formulae become
\begin{align*}
\Pi(\tfrac{1}{2}s,\tfrac{1}{2}s,\tfrac{1}{2}s+q,u)
&= (1+\sqrt{v})^{-s-2q}\,2^{-q}\,\frac{\Gamma\tfrac{1}{2}s\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}s+q)\,\Gamma(\tfrac{1}{2}-q)}\,\Pi(\tfrac{1}{2}s+q,\,\tfrac{1}{2}-q,\,\tfrac{1}{2}+q,\,v) \tag{I.} \\[4pt]
&= (1+\sqrt{v})^{-s-2q}\,2^{-q}\,\frac{\Gamma\tfrac{1}{2}s\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}+q)\,\Gamma(\tfrac{1}{2}s-q)}\,\Pi(\tfrac{1}{2}+q,\,\tfrac{1}{2}s-q,\,\tfrac{1}{2}s+q,\,v) \tag{II.} \\[4pt]
&= (1+\sqrt{v})^{-s}(1-\sqrt{v})^{-2q}\,2^{-q}\,\frac{\Gamma\tfrac{1}{2}s\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}-q)\,\Gamma(\tfrac{1}{2}s+q)}\,\Pi(\tfrac{1}{2}-q,\,\tfrac{1}{2}s+q,\,\tfrac{1}{2}s-q,\,v) \tag{III.} \\[4pt]
&= (1+\sqrt{v})^{-s}(1-\sqrt{v})^{-2q}\,2^{-q}\,\frac{\Gamma\tfrac{1}{2}s\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}s-q)\,\Gamma(\tfrac{1}{2}+q)}\,\Pi(\tfrac{1}{2}s-q,\,\tfrac{1}{2}+q,\,\tfrac{1}{2}-q,\,v) \tag{IV.}
\end{align*}
where observe that on the right-hand side the $\Pi$-functions in I.\ and IV.\ only differ by the sign of $q$, and so also the $\Pi$-functions in II.\ and III.\ only differ by the sign of $q$. We hence have
\[
\Pi(\tfrac{1}{2}s,\tfrac{1}{2}s,\tfrac{1}{2}s-q,u)
= (1+\sqrt{v})^{-s+2q}\,2^{q}\,\frac{\Gamma\tfrac{1}{2}s\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}s-q)\,\Gamma(\tfrac{1}{2}+q)}\,\Pi(\tfrac{1}{2}-q,\,\tfrac{1}{2}s+q,\,\tfrac{1}{2}s-q,\,v);
\]

\bigskip

\noindent\textbf{[p.~367]}\quad and comparing with (IV.),
\[
\Pi(\tfrac{1}{2}s,\tfrac{1}{2}s,\tfrac{1}{2}s+q,u)
= \left(\frac{1+\sqrt{v}}{1-\sqrt{v}}\right)^{2q}\,\Pi(\tfrac{1}{2}s,\tfrac{1}{2}s,\tfrac{1}{2}s-q,u).
\]

\bigskip

\noindent\textbf{74.}\quad The foregoing formula,
\[
\text{Ring-integral} = \frac{2^{s-1}f^{s}}{(f+\kappa)^{s+2q}}\,\Pi(\tfrac{1}{2}s,\tfrac{1}{2}s,\tfrac{1}{2}s+q,u),
\]
where $u = \dfrac{4\kappa f}{(f+\kappa)^{2}}$, gives, as well in the case of an exterior as an interior point, a convergent series for the integral; but this series proceeds according to the powers of $\dfrac{4\kappa f}{(f+\kappa)^{2}}$. We may obtain more convenient formulae applying to the cases of an internal and an external point respectively.

\bigskip

\noindent\textbf{75.}\quad For an internal point $\kappa<f$, $\sqrt{1-u} = \dfrac{f-\kappa}{f+\kappa}$, and therefore $v = \dfrac{\kappa^{2}}{f^{2}}$.
\begin{align*}
\Pi(\tfrac{1}{2}s,\tfrac{1}{2}s,\tfrac{1}{2}s+q,u)
&= \left(\frac{f+\kappa}{f}\right)^{s+2q} 2^{-q}\,\frac{\Gamma\tfrac{1}{2}s\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}s+q)\,\Gamma(\tfrac{1}{2}-q)}\,\Pi\!\left(\tfrac{1}{2}s+q,\,\tfrac{1}{2}-q,\,\tfrac{1}{2}+q,\,\tfrac{\kappa^{2}}{f^{2}}\right) \\[4pt]
&= \left(\frac{f+\kappa}{f}\right)^{s+2q} 2^{-q}\,\frac{\Gamma\tfrac{1}{2}s\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}+q)\,\Gamma(\tfrac{1}{2}s-q)}\,\Pi\!\left(\tfrac{1}{2}+q,\,\tfrac{1}{2}s-q,\,\tfrac{1}{2}s+q,\,\tfrac{\kappa^{2}}{f^{2}}\right) \\[4pt]
&= \left(\frac{f+\kappa}{f}\right)^{s}\left(\frac{f-\kappa}{f}\right)^{-2q} 2^{-q}\,\frac{\Gamma\tfrac{1}{2}s\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}-q)\,\Gamma(\tfrac{1}{2}s+q)}\,\Pi\!\left(\tfrac{1}{2}-q,\,\tfrac{1}{2}s+q,\,\tfrac{1}{2}s-q,\,\tfrac{\kappa^{2}}{f^{2}}\right) \\[4pt]
&= \left(\frac{f+\kappa}{f}\right)^{s}\left(\frac{f-\kappa}{f}\right)^{-2q} 2^{-q}\,\frac{\Gamma\tfrac{1}{2}s\,\Gamma\tfrac{1}{2}s}{\Gamma(\tfrac{1}{2}s-q)\,\Gamma(\tfrac{1}{2}+q)}\,\Pi\!\left(\tfrac{1}{2}s-q,\,\tfrac{1}{2}+q,\,\tfrac{1}{2}-q,\,\tfrac{\kappa^{2}}{f^{2}}\right);
\end{align*}
where the $\Pi$-functions on the right-hand side are respectively
\[
\int_{0}^{1}\frac{x^{\frac{1}{2}s+q-1}(1-x)^{-q-\frac{1}{2}}\,dx}{(1-\tfrac{\kappa^{2}}{f^{2}}\,x)^{\frac{1}{2}+q}}
= \int_{0}^{\infty}\frac{t^{\frac{1}{2}s+q-1}\,dt}{(t+f^{2}-\kappa^{2})^{q+\frac{1}{2}}(t+f^{2})^{\frac{1}{2}s+q-\frac{1}{2}}}\cdot f^{s-1},
\]
\[
\int_{0}^{1}\frac{x^{q-\frac{1}{2}}(1-x)^{\frac{1}{2}s-q-1}\,dx}{(1-\tfrac{\kappa^{2}}{f^{2}}\,x)^{\frac{1}{2}s+q}}
= \int_{0}^{\infty}\frac{t^{q-\frac{1}{2}}(t+f^{2}-\kappa^{2})^{\frac{1}{2}s-q-1}\,dt}{(t+f^{2})^{\frac{1}{2}s+q}}\cdot f^{-1},
\]
\[
\int_{0}^{1}\frac{x^{-q-\frac{1}{2}}(1-x)^{\frac{1}{2}s+q-1}\,dx}{(1-\tfrac{\kappa^{2}}{f^{2}}\,x)^{\frac{1}{2}s-q}}
= \int_{0}^{\infty}\frac{t^{-q-\frac{1}{2}}(t+f^{2}-\kappa^{2})^{\frac{1}{2}s+q-1}\,dt}{(t+f^{2})^{\frac{1}{2}s-q}}\cdot f^{-1},
\]
\[
\int_{0}^{1}\frac{x^{\frac{1}{2}s-q-1}(1-x)^{q-\frac{1}{2}}\,dx}{(1-\tfrac{\kappa^{2}}{f^{2}}\,x)^{\frac{1}{2}-q}}
= \int_{0}^{\infty}\frac{t^{\frac{1}{2}s-q-1}(t+f^{2}-\kappa^{2})^{q-\frac{1}{2}}\,dt}{(t+f^{2})^{\frac{1}{2}+q}}\cdot f^{-1},
\]
the $t$-forms being obtained by means of the transformation $x = \dfrac{t}{t+f^{2}-\kappa^{2}}$; viz.\ this gives
\[
1 - x = \dfrac{f^{2}-\kappa^{2}}{t+f^{2}-\kappa^{2}},\quad
1 - \tfrac{\kappa^{2}}{f^{2}}\,x = \dfrac{t+f^{2}}{t+f^{2}-\kappa^{2}} \cdot \dfrac{1}{f^{2}},\quad
dx = \dfrac{(f^{2}-\kappa^{2})\,dt}{(t+f^{2}-\kappa^{2})^{2}},
\]
whence the results just written down.
