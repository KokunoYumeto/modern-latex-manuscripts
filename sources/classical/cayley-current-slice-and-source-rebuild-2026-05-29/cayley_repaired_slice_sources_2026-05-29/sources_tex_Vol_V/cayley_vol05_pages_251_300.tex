\section*{341. On the Sextactic Points of a Plane Curve \textit{(continued)}}

\noindent\textit{[Continuation from p.~226, in the calculation of the condition for a sextactic point, near the end of Art.~9 and into Art.~10.]}

\medskip

\noindent\ldots [the terms of the foregoing expansion combine, and using the various intermediate substitutions of the foregoing articles] we have
\[
\partial_x U\,\partial_x H + 2\,\partial_y U\,\partial_y H + 2\,\partial_z U\,\partial_z H
\]
expressible as a linear combination of $H\,d^2\Omega$, $H^2\partial\Phi$, and $\Phi\,\partial H$ (with rational coefficients in $m$).

\textbf{10.} And by means of these the condition becomes
\[
0 \;=\; \tfrac{1}{(m-1)^7}\bigl\{(153m^2 - 594m + 549)H\partial^2\Omega + (-102m^2 + 396m + 366)\Phi\,\partial H\bigr\}
\]
\[
\;+\; \tfrac{1}{(m-1)^6}\bigl\{(-96m + 168)H(\partial.\nabla)H + (-90m + 162)H\partial\nabla\!H + (120m - 240)\partial H\nabla H\bigr\}
\]
\[
\;+\; \tfrac{1}{(m-1)^5}\bigl(9H^2\partial\Omega - 45H\Omega\partial H + 40\Phi\partial H\bigr),
\]
this being, as already remarked, of the degree~5 in the arbitrary coefficients $(\lambda,\mu,\nu)$, and of the order $12m-22$ in the coordinates $(x,y,z)$.

\medskip

\textbf{11.} But throwing out the factor $\tfrac{1}{(m-1)^5}$ and observing that in the first line the quadratic functions of $m$ are each a numerical multiple of $51m^2 - 198m + 183$, the condition becomes
\[
0 \;=\; (51m^2 - 198m + 183)\,H\,(3H\partial^2\Omega - 2\Phi\,\partial H)
\]
\[
\;+\; \tfrac{1}{m-1}\bigl\{(-96m+168)H(\partial.\nabla)H + (-90m+162)H\partial\nabla H + (120m-240)\partial H\nabla H\bigr\}
\]
\[
\;+\; \tfrac{1}{(m-1)^2}\bigl(9H^2\partial\Omega - 45H\Omega\partial H + 40\Phi\partial H\bigr).
\]

\bigskip

\begin{center}
\textit{Article Nos.\ 12 and 13. -- Second transformation.}
\end{center}

\textbf{12.} We effect this by means of the formula
\[
(m-2)(3H\partial^2\Omega - 2\Phi\,\partial H) \;=\; \tfrac{1}{H}\,\Jac(U,\Phi,H), \qquad (J)\,\,(^{1})
\]
for, substituting this value of $(3H\partial^2\Omega - 2\Phi\,\partial H)$, the equation becomes divisible by $H$; and dividing out accordingly, the condition becomes
\[
\frac{51m^2-198m+183}{m-2}\,H\,\Jac(U,\Phi,H)
\]
\[
\;+\;(-96m+168)H^2(\partial.\nabla)H + (-90m+162)H^2\partial\nabla H + (120m-240)H\partial H\nabla H
\]
\[
\;+\;\tfrac{1}{m-1}\bigl(9H^2\partial\Omega - 45H\Omega\partial H + 40\Phi\partial H\bigr) \;=\; 0.
\]

\footnote{$(^1)$ Here and elsewhere $(J)$ refers to the Jacobian Formula, see post.\ Article Nos.~34 and~35.}

\medskip

\textbf{13.} We have (see post.\ Article Nos.~36 to~40)
\[
\Jac(U,\Phi,H) \;=\; -(\partial.\nabla)H,
\]
and introducing also $\partial\nabla H$ in place of $\nabla\partial H$ by means of the formula
[as in former memoir], the condition becomes
\[
\bigl(51m^2 - 198m + 183 - (6m-6)\bigr)H^2(\partial.\nabla)H
\]
\[
\;+\; (-90m+162)H^2\partial\nabla H + 120(m-2)H\partial H\nabla H
\]
\[
\;+\; \tfrac{1}{m-1}\bigl(9H^2\partial\Omega - 45H\Omega\partial H + 40\Phi\partial H\bigr) \;=\; 0,
\]
or, as this may be written,
\[
(45m^2-180m+171)H^2(\partial.\nabla)H + (-90m+162)(m-2)H^2(\partial\nabla H) + 120(m-2)^2H\partial H\nabla H
\]
\[
\;+\; (m-2)\bigl(9H^2\partial\Omega - 45H\Omega\partial H + 40\Phi\partial H\bigr) \;=\; 0.
\]

\bigskip

\begin{center}
\textit{Article Nos.\ 14 to 17. -- Third transformation.}
\end{center}

\textbf{14.} We have the following formul{\ae},
\begin{align*}
\tfrac{1}{H}\,\Jac(U,\nabla H,H) &= -(5m-11)\partial H\,\nabla H + (3m-6)H\partial(\nabla H) = 0,\quad (J) \\
\tfrac{1}{H}\,\Jac(U,\nabla,H)H &= -(2m-4)\partial H\,\nabla H + (3m-6)H(\partial.\nabla)H = 0,\quad (J)
\end{align*}
in the latter of which, treating $\nabla$ as a function of the coordinates, we first form the symbol $\Jac(U,\nabla,H)$, and then operating therewith on $H$, we have $\Jac(U,\nabla,H)H$; these give
\[
H\partial(\nabla H) \;=\; \frac{5m-11}{3m-6}\,\partial H\,\nabla H - \tfrac{1}{3(m-2)}\,\Jac(U,\nabla H,H),
\]
\[
H(\partial.\nabla)H \;=\; \frac{2m-4}{3m-6}\,\partial H\,\nabla H - \tfrac{1}{3(m-2)}\,\Jac(U,\nabla,H)H;
\]
and substituting these values, the resulting coefficient of $H\partial H\nabla H$ is
\[
(45m^2-180m+171) + 120 + \tfrac{(-90m+162)(5m-11)}{3} \;=\; 0.
\]

\medskip

\textbf{15.} Hence the condition will contain the factor $H$, and throwing this out, and also the constant factor $\frac{2}{m-2}$, it becomes
\[
(-15m^2+60m-57)\,H\,\Jac(U,\nabla,H)H \;+\; (30m-54)(m-2)\,H\,\Jac(U,\nabla H,H)
\]
\[
+ (m-2)^2\bigl(9H^2\partial\Omega - 45H\Omega\partial H + 40\Phi\partial H\bigr) \;=\; 0.
\]

\textbf{16.} We have
\[
\partial_x(\nabla H) \;=\; \partial_x.\nabla\!H + \partial_x\nabla\!H,
\]
viz.\ in $(\partial_x.\nabla)H$, treating $\nabla$ as a function of $(x,y,z)$ we operate upon it with $\partial_x$ to obtain the new symbol $\partial_x.\nabla$, and with this we operate on $H$; in $\partial_x\nabla H$ we simply multiply together the symbols $\partial_x$ and $\nabla$, giving a new symbol of the form $(\partial^2,\partial_x\partial_y,\partial_x\partial_z)$ which then operates on $H$. We have the like values of $\partial_y(\nabla H)$ and $\partial_z(\nabla H)$; and thence also
\[
\Jac(U,\nabla H,H) \;=\; \Jac(U,\nabla,H)H + \Jac(U,\nabla\!H,H),
\]
viz.\ in the determinant $\Jac(U,\nabla,H)$ the second line corresponding to $\nabla$ is $\partial_x.\nabla$, $\partial_y.\nabla$, $\partial_z.\nabla$ ($\nabla$ being the operand); and the Jacobian thus obtained is a symbol which operates on $H$ giving $\Jac(U,\nabla,H)H$; and in the determinant $\Jac(U,\nabla\!H,H)$ the second line is $\partial_x\nabla H$, $\partial_y\nabla H$, $\partial_z\nabla H$ ($\nabla$ being simply multiplied by $\partial_x,\partial_y,\partial_z$ respectively).

\medskip

\textbf{17.} Substituting, the condition becomes
\[
(-15m^2+60m-57)H\,\Jac(U,\nabla,H)H
\]
\[
\;+\; (30m-54)(m-2)\bigl\{H\,\Jac(U,\nabla,H)H + \Jac(U,\nabla\!H,H)\bigr\}
\]
\[
\;+\; (m-2)^2\bigl(9H^2\partial\Omega - 45H\Omega\partial H + 40\Phi\partial H\bigr) \;=\; 0,
\]
or, what is the same thing,
\[
(15m^2-54m+51)\,H\,\Jac(U,\nabla,H)H \;+\; (30m-54)(m-2)\,H\,\Jac(U,\nabla H,H)
\]
\[
+ (m-2)^2\bigl(9H^2\partial\Omega - 45H\Omega\partial H + 40\Phi\partial H\bigr) \;=\; 0.
\]

\bigskip

\begin{center}
\textit{Article Nos.\ 18 to 27. -- Fourth transformation, and final form of the condition for a Sextactic Point.}
\end{center}

\textbf{18.} I write
\[
(5m-12)\,\Omega\,\partial H - (3m-6)\,H\,\partial\Omega \;=\; \tfrac{1}{H}\,\Jac(U,\Omega,H),\quad (J)
\]
\[
\Omega\partial H + H\partial\Omega \;=\; \partial(\Omega H);
\]
and, introducing for convenience the new symbol $\Psi$,
\[
-5\Psi H + H\partial\Omega \;=\; W, \qquad = 0,
\]
so that we have
\[
\begin{vmatrix} 5m-12 & -(3m-6) & \tfrac{1}{H}\Jac(U,\Omega,H) \\[2pt] 1 & 1 & \partial(\Omega H) \\[2pt] -5 & 1 & W \end{vmatrix}
\]
or, what is the same thing,
\[
(8m-18)W + 6\partial\Jac(U,\Omega,H) + (10m-18)\partial(\Omega H) \;=\; 0,
\]
\[
W \;=\; \Omega\partial H - 5\Omega\partial H \;=\; -\tfrac{3}{4m-9}\,\Phi\,\Jac(U,\Omega,H) - \tfrac{5m-9}{4m-9}\,\partial(\Omega H).
\]

\medskip

\textbf{19.} We have also \hfill $(J)$
\[
(8m-18)\Psi\partial H - (3m-6)H\partial\Psi - \tfrac{1}{H}\Jac(U,\Psi,H) \;=\; 0,
\]
that is
\[
\Psi H^2 \;=\; \tfrac{3m-6}{8m-18}\,H\partial\Psi + \tfrac{1}{(8m-18)H}\,\Jac(U,\Psi,H),
\]
and thence
\[
\Phi H W + W\partial H \;=\; 9H^2\partial\Omega - 45H\Omega\partial H + 40\Phi\partial H,
\]
\[
= -\tfrac{9(m-2)}{4m-9}\,H\partial(\Omega H) + \tfrac{60(m-2)}{4m-9}\,\Phi\partial H
\]
\[
\;+\; \tfrac{1}{4m-9}\bigl\{-27H\Jac(U,\Omega,H) + 40\Jac(U,\Psi,H)\bigr\}.
\]

\medskip

\textbf{20.} The condition thus becomes
\[
(15m^2-54m+51)(4m-9)\,H\,\Jac(U,\nabla,H)H
\]
\[
\;+\; 6(5m-9)(m-2)(4m-9)\,H\,\Jac(U,\nabla H,H)
\]
\[
\;+\; 3(m-2)\bigl\{-3(5m-9)(m-2)\partial(\Omega H) + 20(m-2)^2\Phi\partial H\bigr\}
\]
\[
\;+\; (m-2)^2\bigl\{-27H\Jac(U,\Omega,H) + 40\Jac(U,\Psi,H)\bigr\} \;=\; 0,
\]
which for shortness I represent by
\[
3H\Pi + (m-2)^2\bigl\{-27H\Jac(U,\Omega,H) + 40\Jac(U,\Psi,H)\bigr\} \;=\; 0,
\]
so that we have
\[
\Pi \;=\; (5m^2-18m+17)(4m-9)\,\Jac(U,\nabla,H)H
\]
\[
\;+\; 2(5m-9)(m-2)(4m-9)\,\Jac(U,\nabla H,H)
\]
\[
\;+\; (m-2)\bigl\{-3(5m-9)(m-2)\partial(\Omega H) + 20(m-2)^2\Phi\partial H\bigr\}.
\]

\medskip

\textbf{21.} Write
\[
\Psi_1 \;=\; (\mathfrak{A}',\mathfrak{B}',\mathfrak{C}',\mathfrak{F}',\mathfrak{G}',\mathfrak{H}'\mid A,B,C)^2,
\]
where $(A,B,C)$ are as before the differential coefficients of $U$, and $(a',b',c',f',g',h')$ being the second differential coefficients of $H$, $(\mathfrak{A}',\mathfrak{B}',\mathfrak{C}',\mathfrak{F}',\mathfrak{G}',\mathfrak{H}')$ are the inverse coefficients, viz., $\mathfrak{A}' = b'c' - f'^2$, etc. We have
\[
-(m-1)^2\,\partial\Psi_1 \;=\; (3m-6)(3m-7)\partial(\Omega H) - (3m-7)^2\,\partial\Psi
\]
(see post, Nos.~41 to~46), that is
\[
(3m-6)\,\partial(\Omega H) \;=\; (3m-7)\,\partial\Psi - \tfrac{(m-1)^2}{3m-7}\,\partial\Psi_1,
\]
and thence
\[
\Pi \;=\; (5m^2-18m+17)(4m-9)\,\Jac(U,\nabla,H)H
\]
\[
\;+\; 2(5m-9)(m-2)(4m-9)\,\Jac(U,\nabla H,H)
\]
\[
\;+\; (m-2)^2\Bigl\{(5m^2-18m+17)\partial\Psi + \tfrac{(m-1)^2(5m-9)}{3m-7}\,\partial\Psi_1\Bigr\}.
\]

\medskip

\textbf{22.} Now
\[
\Psi \;=\; (\mathfrak{A},\mathfrak{B},\mathfrak{C},\mathfrak{F},\mathfrak{G},\mathfrak{H}\mid A',B',C')^2 \;=\; (\mathfrak{A}',\mathfrak{B}',\mathfrak{C}',\mathfrak{F}',\mathfrak{G}',\mathfrak{H}'\mid A,B,C)^2,
\]
and writing for shortness
\[
E_\Psi \;=\; (\partial\mathfrak{A},\ldots\mid A',B',C')^2, \qquad F_\Psi \;=\; (\mathfrak{A},\ldots\mid A',B',C'\mid \partial A',\partial B',\partial C'),
\]
\[
E_{\Psi_1} \;=\; (\partial\mathfrak{A}',\ldots\mid A,B,C)^2, \qquad F_{\Psi_1} \;=\; (\mathfrak{A}',\ldots\mid A,B,C\mid \partial A,\partial B,\partial C),
\]
(we might, in a notation above explained, write $E_\Psi = \partial_{N}\Psi$, $F_\Psi = \partial_{N'}\Psi$, and in like manner $E_{\Psi_1} = \partial_{N',\Psi}$, $F_{\Psi_1} = \partial_{N,\Psi}$), then we have
\[
\partial\Psi \;=\; E_\Psi + 2F_\Psi, \qquad \partial\Psi_1 \;=\; E_{\Psi_1} + 2F_{\Psi_1}.
\]
We have moreover
\[
\Jac(U,\nabla H,H) \;=\; -\tfrac{m-1}{3m-7}\,F_{\Psi_1}, \qquad \text{post, Nos.~47 to~50.}
\]
\[
\Jac(U,\nabla,H)H \;=\; -E_{\Psi_1}, \qquad \text{post, Nos.~51 to~53.}
\]

\medskip

\textbf{23.} The just-mentioned formul{\ae} give
\[
\Pi \;=\; -(5m^2-18m+17)(4m-9)\,E_{\Psi_1}
\]
\[
\;-\; 2(5m-9)(m-2)(4m-9)\tfrac{m-1}{3m-7}\,F_{\Psi_1}
\]
\[
\;+\; (m-2)(5m^2-18m+17)(E_\Psi + 2F_\Psi)
\]
\[
\;+\; \tfrac{(5m-9)(m-1)^2(m-2)}{3m-7}\,(E_{\Psi_1} + 2F_{\Psi_1}),
\]
which simplifies to
\[
\Pi \;=\; -(3m-7)(5m^2-18m+17)\,E_{\Psi_1}
\]
\[
\;+\; 2(m-2)(5m^2-18m+17)\,F_\Psi
\]
\[
\;+\; (5m-9)(m-1)^2(m-2)/(3m-7)\,F_{\Psi_1}
\]
\[
\;-\; 2(m-1)(m-2)(3m-8)(5m-9)/(3m-7)\,F_{\Psi_1},
\]
or, as this may also be written,
\[
(3m-7)\Pi \;=\; -(5m^2-18m+17)\bigl\{-2(m-1)(m-2)F_{\Psi_1} + (3m-7)^2 E_{\Psi_1}\bigr\}
\]
\[
\;-\; (5m-9)(m-2)\bigl\{(m-1)(3m-8)F_{\Psi_1} + (3m-7)(3m-8)F_\Psi - (m-1)^2 E_{\Psi_1}\bigr\}
\]
\[
\;+\; (25m^2-103m+106)(m-2)\bigl\{-(m-1)F_{\Psi_1} + (3m-7)F_\Psi\bigr\}.
\]

\medskip

\textbf{24.} But recollecting that
\[
\Omega \;=\; (\mathfrak{A},\mathfrak{B},\mathfrak{C},\mathfrak{F},\mathfrak{G},\mathfrak{H}\mid \partial_x,\partial_y,\partial_z)^2 H
\]
and putting
\[
= (\mathfrak{A},\mathfrak{B},\mathfrak{C},\mathfrak{F},\mathfrak{G},\mathfrak{H}\mid a',b',c',2f',2g',2h'),
\]
\[
E_\Omega \;=\; (\partial\mathfrak{A},\ldots\mid a',\ldots) \;\;(=\partial_N\Omega),
\]
\[
F_\Omega \;=\; (\mathfrak{A},\ldots\mid \partial a',\ldots) \;\;(=\partial_{N'}\Omega),
\]
we have, post, Nos.~41 to~46,
\[
-2(m-1)(m-2)F_{\Psi_1} + (3m-7)^2 E_{\Psi_1} \;=\; (3m-6)(3m-7)HE\Omega,
\]
\[
(m-1)(3m-8)F_{\Psi_1} + (3m-7)(3m-8)F_\Psi - (m-1)^2 H E_{\Psi_1} \;=\; (3m-6)(3m-7)HF\Omega,
\]
\[
-(m-1)F_{\Psi_1} + (3m-7)F_\Psi \;=\; (3m-7)\,\Omega\,\partial H,
\]
and the foregoing equation becomes
\[
(3m-7)\Pi \;=\; -(5m^2-18m+17)(3m-6)(3m-7)\,HE\Omega
\]
\[
\;-\; (5m-9)(m-2)(3m-6)(3m-7)\,HF\Omega
\]
\[
\;+\; (m-2)(25m^2-103m-106)(3m-7)\,\Omega\,\partial H.
\]

\medskip

\textbf{25.} But we have
\[
\tfrac{1}{H}\Jac(U,H,\Omega_U) - (3m-6)HE\Omega + (2m-4)\Omega\partial H \;=\; 0,\quad (J)
\]
that is
\[
\tfrac{1}{H}\Jac(U,H,\Omega_H) - (3m-6)HF\Omega + (3m-6)\Omega\partial H \;=\; 0,\quad (J)
\]
\[
3(m-2)HE\Omega \;=\; 2(m-2)\Omega\partial H + \tfrac{1}{H}\Jac(U,H,\Omega_U),
\]
\[
3(m-2)HF\Omega \;=\; (3m-8)\Omega\partial H + \tfrac{1}{H}\Jac(U,H,\Omega_H),
\]
and we thus obtain
\[
\Pi \;=\; -(5m^2-18m+17)\bigl(2(m-2)\Omega\partial H + \tfrac{1}{H}\Jac(U,H,\Omega_U)\bigr)
\]
\[
\;-\; (5m-9)(m-2)\bigl((3m-8)\Omega\partial H + \tfrac{1}{H}\Jac(U,H,\Omega_H)\bigr)
\]
\[
\;+\; (25m^2-103m+106)(m-2)\,\Omega\,\partial H,
\]
where the coefficient of $(m-2)\Omega\partial H$ is
\[
-(10m^2-36m+34) - (5m-9)(3m-8) + (25m^2-103m+106) \;=\; 0.
\]
Hence
\[
\Pi \;=\; -(5m^2-18m+17)\tfrac{1}{H}\Jac(U,H,\Omega_U) - (5m-9)(m-2)\tfrac{1}{H}\Jac(U,H,\Omega_H).
\]

\medskip

\textbf{26.} Substituting this in the equation
\[
3H\Pi + (m-2)^2\bigl\{-27H\Jac(U,H,\Omega) + 40\Jac(U,H,\Psi)\bigr\} \;=\; 0,
\]
the result contains the factor $H$, and, throwing this out, the condition is
\[
3H\bigl\{-(5m^2-18m+17)\Jac(U,H,\Omega_U) - (5m-9)(m-2)\Jac(U,H,\Omega_H)\bigr\}
\]
\[
\;+\; (m-2)^2\bigl\{27H\Jac(U,H,\Omega) - 40\Jac(U,H,\Psi)\bigr\} \;=\; 0,
\]
or, as this may also be written,
\[
-(15m^2-54m+51)H\Jac(U,H,\Omega_U) - 3(5m-9)(m-2)H\Jac(U,H,\Omega_H)
\]
\[
+ 27(m-2)^2\bigl\{H\Jac(U,H,\Omega_U) + H\Jac(U,H,\Omega_H)\bigr\}
\]
\[
- 40(m-2)^2\Jac(U,H,\Psi) \;=\; 0.
\]

\medskip

\textbf{27.} Hence the condition finally is
\[
(12m^2-54m+57)\,H\,\Jac(U,H,\Omega_U) + (m-2)(12m-27)\,H\,\Jac(U,H,\Omega_H)
\]
\[
- 40(m-2)^2\,\Jac(U,H,\Psi) \;=\; 0,
\]
or, as this may also be written,
\[
-3(m-1)\,H\,\Jac(U,H,\Omega_U) + (m-2)(12m-27)\,H\,\Jac(U,H,\Omega)
\]
\[
- 40(m-2)^2\,\Jac(U,H,\Psi) \;=\; 0,
\]
viz.\ the sextactic points are the intersections of the curve $m$ with the curve represented by this equation; and observing that $U,\,H,\,H\Omega$ and $\Psi$ are of the orders $m,\,3m-6,\,8m-18$ respectively, the order of the curve is as above mentioned $=12m-27$.

\bigskip

\begin{center}
\textit{Article Nos.\ 28 to 30. -- Application to a Cubic.}
\end{center}

\textbf{28.} I have in my former memoir, No.~30, shown that for a cubic curve
\[
\Omega \;=\; (\mathfrak{A},\mathfrak{B},\mathfrak{C},\mathfrak{F},\mathfrak{G},\mathfrak{H}\mid \partial_x,\partial_y,\partial_z)^2 H \;=\; -2S\cdot U \;=\; 0;
\]
this implies $\Jac(U,H,\Omega) = 0$, and hence if one of the two Jacobians, $\Jac(U,H,\Omega_U)$, $\Jac(U,H,\Omega_H)$ vanish, the other will also vanish. Now, using the canonical form, we have
\[
\Omega \;=\; (\mathfrak{A},\ldots\mid a',\ldots)
\]
\[
= \bigl(yz - l^2 x^2,\; zx - l^2 y^2,\; xy - l^2 z^2,\; l^2 yz - lx^2,\; l^2 zx - ly^2,\; l^2 xy - lz^2\bigr)
\]
acting on $(-3l^2, 0, 0, (1+2l^3), 0, 0,\ldots)$, the development of which in fact gives the last-mentioned result. But applying this formula to the calculation of $\Jac(U,H,\Omega_U)$, then disregarding numerical factors, we have
\[
\partial_x \Omega_U \;=\; (yz - l^2 x^2,\ldots\mid l^2 yz - lx^2,\ldots\mid -3l^2, 0, 0, (1+2l^3), 0, 0)
\]
\[
= -3l^2(yz - l^2 x^2) + (1 + 2l^3)(l^2 yz - lx^2)
\]
\[
= (-l + l^4)(x^3 + 2l\,yz) \;=\; S\,\partial_x U;
\]
and in like manner $\partial_y \Omega_U = S\,\partial_y U$, $\partial_z \Omega_U = S\,\partial_z U$, and therefore
\[
\Jac(U,H,\Omega_U) \;=\; S\,\Jac(U,H,U) \;=\; 0,
\]
whence also $\Jac(U,H,\Omega_H) = 0$. And the condition for a sextactic point assumes the more simple form,
\[
\Jac(U,H,\Psi) \;=\; 0.
\]

\medskip

\textbf{29.} Now (former memoir, No.~32) we have
\[
\Psi \;=\; (\mathfrak{A},\mathfrak{B},\mathfrak{C},\mathfrak{F},\mathfrak{G},\mathfrak{H}\mid \partial_x H,\partial_y H,\partial_z H)^2
\]
\[
= (1 + 8l^3)^2 (y^2 z^2 + z^2 x^2 + x^2 y^2)
\]
\[
\;+\; (-9 l^2)^2 (x^6 + y^6 + z^6)
\]
\[
\;+\; (-2l - 5l^4 - 20l^7)(x^3 + y^3 + z^3)\,xyz
\]
\[
\;+\; (-15l^2 - 78l^5 + 12l^8)\,x^2 y^2 z^2,
\]

\textbf{34.} Identically, if $P, Q, R, S$ be functions of the degrees $p, q, r, s$ respectively, we have
\[
\begin{vmatrix} pP & qQ & rR & sS \\ \partial_x P & \partial_x Q & \partial_x R & \partial_x S \\ \partial_y P & \partial_y Q & \partial_y R & \partial_y S \\ \partial_z P & \partial_z Q & \partial_z R & \partial_z S \end{vmatrix} \;=\; 0,
\]
or, what is the same thing,
\[
pP\,\Jac(Q,R,S) - qQ\,\Jac(R,S,P) + rR\,\Jac(S,P,Q) - sS\,\Jac(P,Q,R) \;=\; 0.
\]
Hence in particular if $P = U$, and assuming $U = 0$, we have
\[
-qQ\,\Jac(R,S,U) + rR\,\Jac(S,U,Q) - sS\,\Jac(U,Q,R) \;=\; 0.
\]
If moreover $Q = \tfrac{1}{H}$, and therefore $S = 1$, we have
\[
-\tfrac{1}{H}\Jac(R,S,U) + rR\,\Jac(S,U,\tfrac{1}{H}) - sS\,\Jac(U,\tfrac{1}{H},R) \;=\; 0;
\]
or, as this may also be written,
\[
-\tfrac{1}{H}\Jac(U,R,S) + rR\,\partial S - sS\,\partial R \;=\; 0,
\]
that is
\[
-\tfrac{1}{H}\Jac(U,R,S) + rR\,\partial S - sS\,\partial R \;=\; 0.
\]

\medskip

\textbf{35.} Particular cases are
\begin{align*}
(2m-4)\Phi\partial H - (3m-6)H\partial\Phi &\;=\; \tfrac{1}{H}\Jac(U,\Phi,H), \quad \text{ante, No.~12},\\
(5m-11)\nabla H\partial H - (3m-6)H\partial(\nabla H) &\;=\; \tfrac{1}{H}\Jac(U,\nabla H,H),\\
(2m-4)\nabla:\partial H - (3m-6)H\partial.\nabla &\;=\; \tfrac{1}{H}\Jac(U,\nabla,H),\\
(5m-12)\Omega\partial H - (3m-6)H\partial\Omega &\;=\; \tfrac{1}{H}\Jac(U,\Omega,H),\quad \text{Art.~18},\\
(8m-18)\Psi\partial H - (3m-6)H\partial\Psi &\;=\; \tfrac{1}{H}\Jac(U,\Psi,H), \quad \text{Art.~25},\\
(2m-4)\Omega\partial H - (3m-6)HE\Omega &\;=\; \tfrac{1}{H}\Jac(U,\Omega_U,H),\\
(3m-8)\Omega\partial H - (3m-6)HF\Omega &\;=\; \tfrac{1}{H}\Jac(U,\Omega_H,H),
\end{align*}
where it is to be observed that in the third of these formul{\ae} I have, in accordance with the notation before employed, written $\partial.\nabla$ to denote the result of the operation $\partial$ performed on $\nabla$ as operand. I have also written $\nabla:\partial H$ to show that the operation $\nabla$ is not to be performed on the following $\partial H$ as an operand, but that it remains as an unperformed operation. As regards the last two equations, it is to be remarked that the demonstration in the last preceding number depends merely on the homogeneity of the functions, and the orders of these functions: in the former of the two formul{\ae}, the differentiation of $\Omega$ is performed upon $\Omega$ in regard to the coordinates $(x,y,z)$ in so far only as they enter through $U$, and $\Omega$ is therefore to be regarded as a function of the order $2m - 4$; in the latter of the two formul{\ae}, the differentiation is to be performed in regard to the coordinates in so far only as they enter through $H$, and $\Omega$ is therefore to be regarded as a function of the order $3m - 8$. The two formul{\ae} might also be written
\begin{align*}
(2m-4)\Omega\partial H - (3m-6)H\partial\Omega_U &\;=\; \tfrac{1}{H}\Jac(U,\Omega_U,H),\\
(3m-8)\Omega\partial H - (3m-6)H\partial\Omega_H &\;=\; \tfrac{1}{H}\Jac(U,\Omega_H,H);
\end{align*}
and it may be noticed that, adding these together, we obtain the foregoing formula,
\[
(5m-12)\Omega\partial H - (3m-6)H\partial\Omega \;=\; \tfrac{1}{H}\Jac(U,\Omega,H).
\]

\textbf{36.} We have

\partial.\nabla \;=\; (P\lambda + Q\mu + R\nu)\cdot(\mathfrak{A}\partial_x + \mathfrak{H}\partial_y + \mathfrak{G}\partial_z,\; \mathfrak{H}\partial_x + \mathfrak{B}\partial_y + \mathfrak{F}\partial_z,\; \mathfrak{G}\partial_x + \mathfrak{F}\partial_y + \mathfrak{C}\partial_z\mid \lambda,\mu,\nu),

\textbf{37.} Again, in $\Jac(U,H,\Phi)$, where $\Phi = (\mathfrak{A},\mathfrak{B},\mathfrak{C},\mathfrak{F},\mathfrak{G},\mathfrak{H}\mid\lambda,\mu,\nu)^2$, the coefficients of $\lambda^2$, etc.\ are $\Jac(U,H,\mathfrak{A})$, etc.; and hence the assumed equation
\[

in regard to the term in $\lambda^2$, is
\[
(P\mathfrak{A}\partial_x + P\mathfrak{H}\partial_y + P\mathfrak{G}\partial_z)H \;=\; \Jac(U,H,\mathfrak{A}),

\Jac(U,H,\mathfrak{A}) \;=\; \begin{vmatrix} A & B & C \\ \partial_x H & \partial_y H & \partial_z H \\ \partial_x \mathfrak{A} & \partial_y \mathfrak{A} & \partial_z \mathfrak{A} \end{vmatrix},

\[
\;+\; \bigl[(B\partial_z - C\partial_y)\mathfrak{G} - (A\partial_y - B\partial_x)\mathfrak{A}\bigr]\partial_z H \;=\; 0.

\medskip

\]
\[
E_\Omega = (\partial\mathfrak{A},\ldots\mid a',\ldots),\quad F_\Omega = (\mathfrak{A},\ldots\mid \partial a',\ldots),
\]
then the three equations are
\[
-2(m-1)(m-2)F_{\Psi_1} \;=\; (3m-6)(3m-7)HE\Omega - (3m-7)^2 E_\Psi,
\]
\[
-(m-1)^2 E_{\Psi_1} \;=\; (3m-7)(3m-8)\Omega\partial H + (3m-6)(3m-7)HF\Omega - 2(3m-7)(3m-8)F_\Psi,
\]
\[
-2(m-1)F_{\Psi_1} \;=\; 2(3m-7)\Omega\partial H - 2(3m-7)F_\Psi,
\]
whence, adding, we have
\[
-(m-1)^2(E_{\Psi_1} + 2F_{\Psi_1}) \;=\; -(3m-7)^2(E_\Psi + 2F_\Psi)
\]
\[
\;+\; (3m-6)(3m-7)\bigl\{\Omega\partial H + H(E\Omega + F\Omega)\bigr\},
\]
(that is
\[
-(m-1)^2\,\partial\Psi_1 \;=\; -(3m-7)^2\,\partial\Psi + (3m-6)(3m-7)\partial.\Omega H,
\]
which is right). And by linearly combining the three equations, we deduce
\[
(3m-6)(3m-7)HE\Omega \;=\; -2(m-1)(m-2)F_{\Psi_1} + (3m-7)^2 E_\Psi,
\]
\[
(3m-7)\Omega\partial H \;=\; -(m-1)F_{\Psi_1} + (3m-7)F_\Psi,
\]
\[
(3m-6)(3m-7)HF\Omega \;=\; (m-1)(3m-8)F_{\Psi_1} + (3m-7)(3m-8)F_\Psi - (m-1)^2 E_{\Psi_1},
\]
which are the formul{\ae}, ante No.~24.

\bigskip

\begin{center}
\textit{Article Nos.\ 47 to 50. -- Proof of an identity used in the fourth transformation, viz.,}
\end{center}
\[
\Jac(U,\nabla H,H) \;=\; -\tfrac{m-1}{3m-7}\,F_{\Psi_1},
\]
or say
\[
\Jac(U,H,\nabla H) \;=\; \tfrac{m-1}{3m-7}\,(\mathfrak{A}',\ldots\mid A,B,C\mid \partial A,\partial B,\partial C).
\]

\textbf{47.} We have
\[
\nabla \;=\; (\mathfrak{A},\ldots\mid \lambda,\mu,\nu\mid \partial_x,\partial_y,\partial_z)
\]
\[
= \bigl((\mathfrak{A},\mathfrak{H},\mathfrak{G}\mid \lambda,\mu,\nu),\; (\mathfrak{H},\mathfrak{B},\mathfrak{F}\mid \lambda,\mu,\nu),\; (\mathfrak{G},\mathfrak{F},\mathfrak{C}\mid \lambda,\mu,\nu)\mid \partial_x,\partial_y,\partial_z\bigr);
\]
or, attending to the effect of the bar as denoting the exemption of the $(\mathfrak{A},\ldots)$ from differentiation,
\[
\Jac(U,H,\nabla H) \;=\; (\mathfrak{A},\mathfrak{H},\mathfrak{G}\mid \lambda,\mu,\nu)\,\Jac(U,H,\partial_x H)
\]
\[
\;+\; (\mathfrak{H},\mathfrak{B},\mathfrak{F}\mid \lambda,\mu,\nu)\,\Jac(U,H,\partial_y H)
\]
\[
\;+\; (\mathfrak{G},\mathfrak{F},\mathfrak{C}\mid \lambda,\mu,\nu)\,\Jac(U,H,\partial_z H).
\]

\medskip

\textbf{48.} Now
\[
\Jac(U,H,\partial_x H) \;=\; -\tfrac{1}{m-2}\,\Jac(U,\,x\partial_x H + y\partial_y H + z\partial_z H,\,\partial_x H),
\]
and the last-mentioned Jacobian is
\[
= \partial_x H\,\Jac(U,x,\partial_x H) + \partial_y H\,\Jac(U,y,\partial_x H) + \partial_z H\,\Jac(U,z,\partial_x H)
\]
\[
\;+\; y\,\Jac(U,\partial_y H,\partial_x H) + z\,\Jac(U,\partial_z H,\partial_x H),
\]
where the second line is
\[
= -y\,\Jac(U,\partial_x H,\partial_y H) + z\,\Jac(U,\partial_x H,\partial_z H),
\]
or, writing $(A',B',C') = $ first differential coefficients of $H$, this is the second differential coefficients of $H$ and $(a',b',c',f',g',h')$ for those, in
\[
\begin{vmatrix} A & B & C \\ a' & h' & g' \\ h' & b' & f' \end{vmatrix} - y\,\ldots + z\,\ldots \;=\; -y(\mathfrak{G}',\mathfrak{F}',\mathfrak{C}'\mid A,B,C) + z(\mathfrak{H}',\mathfrak{B}',\mathfrak{F}'\mid A,B,C).
\]
The first line, reduced by the formul{\ae}
\[
(3m-7)(A',B',C') \;=\; (a'x + h'y + g'z,\; h'x + b'y + f'z,\; g'x + f'y + c'z),
\]
is
\[
= \tfrac{1}{3m-7}\bigl\{-y(\mathfrak{G}',\mathfrak{F}',\mathfrak{C}'\mid A,B,C) + z(\mathfrak{H}',\mathfrak{B}',\mathfrak{F}'\mid A,B,C)\bigr\}.
\]
Hence we have
\[
\Jac(U,H,\partial_x H) \;=\; \tfrac{1}{3m-7}\bigl(1 + \tfrac{3m-7}{m-2}\bigr)\bigl\{-y(\mathfrak{G}',\mathfrak{F}',\mathfrak{C}'\mid A,B,C) + z(\mathfrak{H}',\mathfrak{B}',\mathfrak{F}'\mid A,B,C)\bigr\}
\]
\[
= \tfrac{1}{m-2}\bigl\{-y(\mathfrak{G}',\mathfrak{F}',\mathfrak{C}'\mid A,B,C) + z(\mathfrak{H}',\mathfrak{B}',\mathfrak{F}'\mid A,B,C)\bigr\};
\]
and in like manner
\[
\Jac(U,H,\partial_y H) \;=\; \tfrac{1}{m-2}\bigl\{-z(\mathfrak{A}',\mathfrak{H}',\mathfrak{G}'\mid A,B,C) + x(\mathfrak{G}',\mathfrak{F}',\mathfrak{C}'\mid A,B,C)\bigr\},
\]
\[
\Jac(U,H,\partial_z H) \;=\; \tfrac{1}{m-2}\bigl\{-x(\mathfrak{H}',\mathfrak{B}',\mathfrak{F}'\mid A,B,C) + y(\mathfrak{A}',\mathfrak{H}',\mathfrak{G}'\mid A,B,C)\bigr\}.
\]

\medskip

\textbf{49.} We thence have
\[
\Jac(U,H,\nabla H) \;=\; \tfrac{1}{3m-7}\,\begin{vmatrix} (\mathfrak{A}',\mathfrak{H}',\mathfrak{G}'\mid A,B,C) & (\mathfrak{H}',\mathfrak{B}',\mathfrak{F}'\mid A,B,C) & (\mathfrak{G}',\mathfrak{F}',\mathfrak{C}'\mid A,B,C) \\[2pt] \lambda & \mu & \nu \\[2pt] x & y & z \end{vmatrix},
\]
or multiplying the two sides by
\[
H \;=\; \begin{vmatrix} a & h & g \\ h & b & f \\ g & f & c \end{vmatrix},
\]
the right-hand side is
\[
\tfrac{1}{3m-7}\,\begin{vmatrix} H\lambda & H\mu & H\nu \\ X & Y & Z \\ (m-1)A & (m-1)B & (m-1)C \end{vmatrix}
\]
which is
\[
= H\cdot\tfrac{m-1}{3m-7}\,\begin{vmatrix} \lambda & \mu & \nu \\ X & Y & Z \\ A & B & C \end{vmatrix},
\]
if for a moment
\[
X = (\mathfrak{A}',\ldots\mid A,B,C\mid a,h,g),
\]
\[
Y = (\mathfrak{A}',\ldots\mid A,B,C\mid h,b,f),
\]
\[
Z = (\mathfrak{A}',\ldots\mid A,B,C\mid g,f,c).
\]

\medskip

\textbf{50.} Hence observing that these equations may be written
\[
X = (\mathfrak{A}',\ldots\mid A,B,C\mid \partial_x A,\partial_x B,\partial_x C),
\]
\[
Y = (\mathfrak{A}',\ldots\mid A,B,C\mid \partial_y A,\partial_y B,\partial_y C),
\]
\[
Z = (\mathfrak{A}',\ldots\mid A,B,C\mid \partial_z A,\partial_z B,\partial_z C),
\]
and that we have
\[
\partial \;=\; \begin{vmatrix} \lambda & \mu & \nu \\ \partial_x & \partial_y & \partial_z \\ A & B & C \end{vmatrix},
\]
we obtain for $H\,\Jac(U,H,\nabla H)$ the value
\[
= H\cdot\tfrac{m-1}{3m-7}\,(\mathfrak{A}',\ldots\mid A,B,C\mid \partial A,\partial B,\partial C),
\]
or throwing out the factor $H$, we have the required result.

\begin{center}
\textit{Article Nos.\ 51 to 53. -- Proof of identity used in the fourth transformation, viz.}
\end{center}
\[
\Jac(U,\nabla,H)H \;=\; -E_{\Psi_1},\quad \text{or say } \Jac(U,H,\nabla)H \;=\; (\partial\mathfrak{A}',\ldots\mid A,B,C)^2.
\]

\textbf{51.} We have
\[
\nabla \;=\; (\mathfrak{A},\ldots\mid \lambda,\mu,\nu\mid \partial_x,\partial_y,\partial_z),
\]
and thence
\[
(\partial_x.\nabla)H \;=\; (\partial_x\mathfrak{A},\partial_x\mathfrak{H},\partial_x\mathfrak{G}\mid \lambda,\mu,\nu\mid x,y,z),
\]
with the like values for $(\partial_y.\nabla)H$ and $(\partial_z.\nabla)H$. And then
\[
\Jac(U,H,\nabla)H \;=\; \begin{vmatrix} A & B & C \\ A' & B' & C' \\ (\partial_x.\nabla)H & (\partial_y.\nabla)H & (\partial_z.\nabla)H \end{vmatrix},
\]
in which the coefficient of $\lambda^2$ is
\[
= (C\partial_y - B\partial_z)(\mathfrak{A},\mathfrak{H},\mathfrak{G}\mid \lambda,\mu,\nu);
\]
or putting for shortness
\[
(C\partial_y - B\partial_z, A\partial_z - C\partial_x, B\partial_x - A\partial_y) \;=\; (P,Q,R),
\]
the coefficient is $\partial = (P\lambda + Q\mu + R\nu)$,
and thence
\[
\text{coefficient } \lambda^2 - \partial\mathfrak{A} \;=\; (P\mathfrak{A},P\mathfrak{H},P\mathfrak{G}\mid \lambda,\mu,\nu) - (P\mathfrak{A},Q\mathfrak{A},R\mathfrak{A}\mid \lambda,\mu,\nu)
\]
which is
\[
= \mu\bigl\{(C\partial_y - B\partial_z)\mathfrak{H} - (A\partial_z - C\partial_x)\mathfrak{A}\bigr\}
\]
\[
\;+\; \nu\bigl\{(C\partial_y - B\partial_z)\mathfrak{G} - (B\partial_x - A\partial_y)\mathfrak{A}\bigr\},
\]
where coefficient of $\mu$ is
\[
= -A\partial_z\mathfrak{A} - B\partial_y\mathfrak{H} + C(\partial_x\mathfrak{A} + \partial_y\mathfrak{H}) \;=\; -(A\partial_z\mathfrak{A} + B\partial_z\mathfrak{H} + C\partial_z\mathfrak{G}) \;=\; -\tfrac{1}{m-1}\,x\partial_z H,
\]
and coefficient of $\nu$ is
\[
= +(A\partial_y\mathfrak{A} + B\partial_y\mathfrak{H} + C\partial_y\mathfrak{G}) \;=\; \tfrac{1}{m-1}\,x\partial_y H,
\]
so that
\[
\text{coefficient } \lambda^2 - \partial\mathfrak{A} \;=\; -\tfrac{1}{m-1}\,x(\mu\partial_z H - \nu\partial_y H).
\]

\medskip

\textbf{53.} By forming in a similar manner the coefficients of the other terms, it appears that
\[
\Jac(U,H,\nabla)H - (\partial\mathfrak{A},\ldots\mid A',B',C')^2
\]
\[
\;\propto\; \begin{vmatrix} \lambda & \mu & \nu \\ \partial_x H & \partial_y H & \partial_z H \\ A' & B' & C' \end{vmatrix}\,(A'x + B'y + C'z),
\]
or since the determinant
\[
\begin{vmatrix} \lambda & \mu & \nu \\ \partial_x H & \partial_y H & \partial_z H \\ A' & B' & C' \end{vmatrix} \;=\; 0,
\]
we have the required equation,
\[
\Jac(U,H,\nabla)H \;=\; (\partial\mathfrak{A},\ldots\mid A',B',C')^2.
\]
This completes the series of formul{\ae} used in the transformations of the condition for the sextactic point.

\begin{center}
\textit{Appendix, Nos.\ 54 to 74.}
\end{center}

For the sake of exhibiting in their proper connexion some of the formul{\ae} employed in the foregoing first transformation of the condition for a sextactic point, I have investigated them in the present Appendix, which however is numbered continuously with the memoir.

\medskip

\textbf{54.} The investigations of my former memoir and the present memoir have reference to the operations
\[
\partial_1 \;=\; dx\,\partial_x + dy\,\partial_y + dz\,\partial_z,
\]
\[
\partial_2 \;=\; d^2 x\,\partial_x + d^2 y\,\partial_y + d^2 z\,\partial_z,
\]
\[
\partial_3 \;=\; d^3 x\,\partial_x + d^3 y\,\partial_y + d^3 z\,\partial_z,
\]
where, if $(A,B,C)$ are arbitrary constants, then we have $\partial_1 = \partial$, the first differential coefficients of a function $U' = (*\mid x,y,z)^{m'}$, so that putting $dx = B\nu - C\mu$, $dy = C\lambda - A\nu$, $dz = A\mu - B\lambda$;
\[
\partial \;=\; (B\nu - C\mu)\partial_x + (C\lambda - A\nu)\partial_y + (A\mu - B\lambda)\partial_z \;=\; \begin{vmatrix} A & B & C \\ \lambda & \mu & \nu \\ \partial_x & \partial_y & \partial_z \end{vmatrix},
\]
we have $\partial_1 = \partial$. The foregoing expressions of $(dx,dy,dz)$ determine of course the values of $(d^2 x, d^2 y, d^2 z)$, $(d^3 x, d^3 y, d^3 z)$, etc., and it is throughout assumed that these values are substituted in the symbols $\partial_2,\partial_3$, etc., so that $\partial_1 = \partial$, and $\partial_2,\partial_3,$ etc.\ denote each of them an operator such as $X\partial_x + Y\partial_y + Z\partial_z$, where $(X,Y,Z)$ are functions of the coordinates; such operator, in so far as it is a function of the coordinates, may therefore be made an operand, and be operated upon by itself or any other like operator.

\medskip

\textbf{55.} Taking $(a,b,c,f,g,h)$ for the second differential coefficients of $U$, $(\mathfrak{A},\mathfrak{B},\mathfrak{C},\mathfrak{F},\mathfrak{G},\mathfrak{H})$ for the inverse coefficients, and $H$ for the Hessian, I write also
\[
\Phi \;=\; (\mathfrak{A},\ldots\mid \lambda,\mu,\nu)^2.
\]
\[
\Pi \;=\; \lambda x + \mu y + \nu z,
\]
\[
\Psi \;=\; (\mathfrak{A},\ldots\mid \partial_x H,\partial_y H,\partial_z H)^2,
\]
\[
\nabla \;=\; (\mathfrak{A},\ldots\mid \mu\partial_z - \nu\partial_y,\;\nu\partial_x - \lambda\partial_z,\;\lambda\partial_y - \mu\partial_x),
\]
and I notice that we have
\[
\nabla\Pi^2 \;=\; 2\Phi,\quad \nabla\Pi U \;=\; \tfrac{1}{m-1}H,\quad \Omega U \;=\; SH,
\]
the last of which is proved, post No.~65; the others are found without any difficulty.

\medskip

\textbf{56.} I form the Table [of substitution rules and coefficients to follow; consult facsimile p.~248 for the explicit Table; symbolic entries involve powers of $\tfrac{1}{m-1}$ multiplied by combinations of $\Phi$, $\Psi$, and $H$].

\medskip

And assuming $U = 0$, we have $\partial U = 0$, and
\[
(\partial_2 H)^2 \;=\; (SH)^2 \;=\; \tfrac{1}{(m-1)^2}\bigl(\Phi^2\partial^2 H - 2\Phi\Psi\partial H + \Psi^2\bigr),
\]
which are for the most part given in my former memoir; the expressions for $\partial_3 U$, $\partial_4 U$, which are not explicitly given, follow at once from the equations
\[
(\partial_1^3 + \partial_2)U \;=\; 0,\qquad (\partial_1^4 + 2\partial_2\partial_3 + \partial_4)U \;=\; 0;
\]
those for $\partial_2\partial_3 U$, $\partial_3^2 U$, and $\partial_4 U$ are new, but when the expressions for $\partial_3\partial_2 U$ and $\partial_3 U$ are known, that for $\partial_4 U$ is at once found from the equation
\[
(\partial_1^4 + 6\partial_1^2\partial_2 + 4\partial_1\partial_3 + 3\partial_2^2 + \partial_4)U \;=\; 0.
\]

\medskip

\textbf{57.} Before going further, I remark that we have identically
\[
(a,\ldots\mid x,y,z)^2(a,\ldots\mid \mu y - \nu\beta,\;\nu\alpha - \lambda y,\;\lambda\beta - \mu\alpha)^2
\]
\[
\;\propto\; \begin{vmatrix} ax + hy + gz & hx + by + fz & gx + fy + cz \\ \alpha & \beta & \gamma \\ \lambda & \mu & \nu \end{vmatrix}
\]
\[
\;=\; (\mathfrak{A},\ldots\mid \lambda p - \alpha\eta,\;\mu p - \beta\eta,\;\nu p - \gamma\eta)^2,
\]
(if for shortness $p = \alpha x + \beta y + \gamma z$, $\eta = \lambda x + \mu y + \nu z$)
\[
= p^2(\mathfrak{A},\ldots\mid \lambda,\mu,\nu)^2 - 2 p\eta(\mathfrak{A},\ldots\mid \lambda,\mu,\nu\mid \alpha,\beta,\gamma) + \eta^2(\mathfrak{A},\ldots\mid \alpha,\beta,\gamma)^2.
\]

\medskip

\textbf{58.} If in this equation we take $(a,b,c,f,g,h)$ to be the second differential coefficients of $U$, and write also $(\alpha,\beta,\gamma) = (\partial_x,\partial_y,\partial_z)$, the equation becomes
\[
m(m-1)U\Gamma - (m-1)^2 \Omega \;=\; \Phi(x\partial_x + y\partial_y + z\partial_z)^2 - 2\Pi(x\partial_x + y\partial_y + z\partial_z)\nabla
\]
which is a general equation for the transformation of $\partial^2_1 (= \partial^2)$.

\medskip

\textbf{59.} If with the two sides of this equation we operate on $U$, we obtain
\[
m(m-1)U\Gamma U - (m-1)^2\,\partial^2 U \;=\; m(m-1)\Phi U - 2(m-1)\Pi\nabla U,
\]
and substituting the values we find the before-mentioned expression of $\partial_1{}^2 U$.

\medskip

\textbf{60.} Operating with the two sides of the same equation on a function $H$ of the order $m'$, we find
\[
m(m-1)U\Gamma H - (m-1)^2\,\partial^2 H \;=\; m'(m'-1)\Phi H - 2(m'-1)\Pi\nabla H + \Phi\Omega H;
\]
and in particular if $H$ is the Hessian, then writing $m' = 3m - 6$, and putting $U = 0$, we find the before-mentioned expression for $\partial^2 H$.

\medskip

\textbf{61.} But we may also from the general identical equation deduce the expression for $(\partial H)^2$. In fact taking $H$ a function of the degree $m'$ and writing
\[
(\alpha,\beta,\gamma) \;=\; (\partial_x H,\partial_y H,\partial_z H),
\]
we have
\[
m(m-1)U(\mathfrak{A},\ldots\mid \mu\partial_z H - \nu\partial_y H,\;\nu\partial_x H - \lambda\partial_z H,\;\lambda\partial_y H - \mu\partial_x H)^2 - (m-1)^2(\partial H)^2
\]
\[
\;=\; m'^2 \Phi H^2 - 2m'\Pi H\nabla H + \Pi^2(\mathfrak{A},\ldots\mid \partial_x H,\partial_y H,\partial_z H)^2;
\]
and if $H$ be the Hessian, then writing $m' = 3m - 6$ and putting also $U = 0$, we find the before-mentioned expression for $(\partial H)^2$.

\medskip

\textbf{62.} Proof of equation $\partial \;=\; \tfrac{1}{m-1}\bigl(x\partial_x + y\partial_y + z\partial_z\bigr) + \tfrac{1}{m-1}\Pi\nabla.$

We have
\[
\partial \;=\; \tfrac{1}{m-1}\bigl(\lambda(C\partial_y - B\partial_z) + \mu(A\partial_z - C\partial_x) + \nu(B\partial_x - A\partial_y)\bigr),
\]
where
\[
= \lambda(C'\partial_y - B'\partial_z) + \mu(A'\partial_z - C'\partial_x) + \nu(B'\partial_x - A'\partial_y),
\]
\[
A' \;=\; \partial A \;=\; a(B\nu - C\mu) + h(C\lambda - A\nu) + g(A\mu - B\lambda)
\]
\[
= \lambda(hC - gB) + \mu(gA - aC) + \nu(aB - hA),
\]
with the like values for $B'$ and $C'$. Substituting the values
\[
(m-1)(A,B,C) \;=\; (ax + hy + gz,\;hx + by + fz,\;gx + fy + cz),
\]
we have
\[
(m-1)A' \;=\; \lambda(\mathfrak{H}y - \mathfrak{G}z) + \mu(\mathfrak{G}y - \mathfrak{H}z) + \nu(\mathfrak{H}y - \mathfrak{G}z);
\]
and similarly
\[
(m-1)B' \;=\; \lambda(\mathfrak{H}z - \mathfrak{G}x) + \mu(\mathfrak{F}z - \mathfrak{G}x) + \nu(\mathfrak{C}z - \mathfrak{G}x),
\]
\[
(m-1)C' \;=\; \lambda(\mathfrak{H}x - \mathfrak{B}y) + \mu(\mathfrak{F}x - \mathfrak{B}y) + \nu(\mathfrak{C}x - \mathfrak{F}y),
\]
and then
\[
(m-1)(C'\partial_y - B'\partial_z) \;=\; \lambda\bigl\{(\mathfrak{B}x - \mathfrak{A}y)\partial_y - (\mathfrak{H}z - \mathfrak{G}x)\partial_z\bigr\}
\]
\[
\;+\; \mu\bigl\{(\mathfrak{F}x - \mathfrak{H}y)\partial_y - (\mathfrak{B}z - \mathfrak{F}x)\partial_z\bigr\}
\]
\[
\;+\; \nu\bigl\{\ldots\bigr\}
\]
\[
= x(\mathfrak{A},\mathfrak{H},\mathfrak{G}\mid \partial_x,\partial_y,\partial_z\mid \lambda,\mu,\nu) - \mathfrak{A}(x\partial_x + y\partial_y + z\partial_z);
\]
and so
\[
(m-1)(A'\partial_z - C'\partial_x) \;=\; y(\mathfrak{H},\mathfrak{B},\mathfrak{F}\mid \lambda,\mu,\nu\mid \partial_x,\partial_y,\partial_z) - \mathfrak{H}(x\partial_x + y\partial_y + z\partial_z),
\]
\[
(m-1)(B'\partial_x - A'\partial_y) \;=\; z(\mathfrak{G},\mathfrak{F},\mathfrak{C}\mid \lambda,\mu,\nu\mid \partial_x,\partial_y,\partial_z) - \mathfrak{G}(x\partial_x + y\partial_y + z\partial_z);
\]
\[
(m-1)\partial \;=\; (\lambda x + \mu y + \nu z)\nabla - (\mathfrak{A},\ldots\mid x,\mu,\nu)(x\partial_x + y\partial_y + z\partial_z),
\]
\[
= \Pi\nabla - \Phi(x\partial_x + y\partial_y + z\partial_z);
\]
or finally,
\[
\partial \;=\; -\tfrac{\Phi}{m-1}(x\partial_x + y\partial_y + z\partial_z) + \tfrac{1}{m-1}\Pi\nabla.
\]

\medskip

\textbf{63.} This leads to the expression for $\partial^2 U$; we have
\[
\partial^2 \;=\; \tfrac{1}{(m-1)^2}\bigl[\Phi^2(x\partial_x + y\partial_y + z\partial_z)^2 - 2\Phi\Pi\nabla(x\partial_x + y\partial_y + z\partial_z) + \Pi^2\nabla^2\bigr],
\]
and operating herewith on $U$, we find
\[
\partial^2 U \;=\; \tfrac{1}{(m-1)^2}\bigl[m(m-1)\Phi^2 U - 2(m-1)\Phi\Pi\nabla U + \Pi^2\nabla^2 U\bigr];
\]
or since $\nabla U = \tfrac{1}{m-1}H$, this is
\[
\partial^2 U \;=\; \tfrac{1}{(m-1)^2}\bigl[m(m-1)\Phi^2 U - 2\Phi\Pi H + \Pi^2\nabla^2 U\bigr].
\]

\textbf{64.} We have $\partial_4 U = 0$, and thence
\[
(\partial_1^4 + \partial_4)U \;=\; 0,
\]
that is
\[
\partial_4 U \;=\; -\partial_1^4 U - \partial_1\partial_3 U;
\]
or substituting the values of $\partial_1\partial_3 U$ and $\partial_1^4 U$, we find the value of $\partial_4 U$ as given in the Table. And then from the equation
\[
(\partial_1^4 + 6\partial_1^2\partial_2 + 4\partial_1\partial_3 + 3\partial_2^2 + \partial_4)U \;=\; 0,
\]
we find the value of $\partial_4 U$, and the proof of the expressions in the Table is thus completed.

\medskip

\textbf{65.} Proof of equation $\nabla.\partial = 0$.

We have
\[
\nabla.\partial \;=\; \nabla.\bigl((B\nu - C\mu)\partial_x + (C\lambda - A\nu)\partial_y + (A\mu - B\lambda)\partial_z\bigr)
\]
\[
= \nabla.\bigl(\lambda(\mu\partial_z - \nu\partial_y) + \mu(\nu\partial_x - \lambda\partial_z) + \nu(\lambda\partial_y - \mu\partial_x)\bigr)
\]
and then
\[
\nabla A \;=\; (\mathfrak{A},\ldots\mid \lambda,\mu,\nu\mid a,h,g) \;=\; H\lambda,
\]
\[
\nabla B \;=\; (\mathfrak{A},\ldots\mid \lambda,\mu,\nu\mid h,b,f) \;=\; H\mu,
\]
\[
\nabla C \;=\; (\mathfrak{A},\ldots\mid \lambda,\mu,\nu\mid g,f,c) \;=\; H\nu;
\]
or substituting these values, we have the equation in question.

\medskip

\textbf{66.} Proof of the expression for $\partial_2$.

We have, operating on the two sides respectively with $\partial_1 = \partial$, we have
\[
\partial_2 \;=\; \tfrac{1}{m-1}\bigl\{\partial(\Phi(x\partial_x + y\partial_y + z\partial_z)) + \Phi\partial.(x\partial_x + y\partial_y + z\partial_z)\bigr\}
\]
\[
= \tfrac{1}{m-1}\bigl\{\partial\Phi\cdot\nabla + \Phi\partial\nabla\bigr\}.
\]

\medskip

\textbf{67.} Proof of expression for $\partial_2 H$.

Operating with $\partial_2$ upon $H$, we have at once
\[
\partial_2 H \;=\; -\tfrac{m'-1}{m-1}H\partial^2 + \tfrac{1}{m-1}\Phi\partial H + \tfrac{1}{m-1}(\partial.\nabla)H.
\]

\medskip

The remainder of the present Appendix is preliminary, or relating to the investigation of the expressions for $\partial_2\partial_3 U$ and $\partial_3\partial_2 U$, used ante, No.~31.

\textbf{68.} Proof of equation $\nabla^2.\partial U \;=\; \Psi\,\partial H - H\,\partial\Phi.$

We have identically
\[
(\mathfrak{A},\ldots\mid \lambda,\mu,\nu)^2(\mathfrak{A},\ldots\mid \partial_x,\partial_y,\partial_z)^2 - (\mathfrak{A},\ldots\mid \lambda,\mu,\nu\mid \partial_x,\partial_y,\partial_z)^2
\]
that is
\[
= (ax - \&c.)(\mathfrak{A},\ldots\mid \mu\partial_z - \nu\partial_y,\;\nu\partial_x - \lambda\partial_z,\;\lambda\partial_y - \mu\partial_x)^2;
\]
\[
\Phi\Omega - \nabla^2 \;=\; H\Gamma;
\]
and then multiplying by $\partial$ and with the result operating on $U$, we find
\[
\Phi\Omega U - \nabla^2.\partial U \;=\; H\partial\Gamma U.
\]
Now
\[
\Omega U \;=\; (\mathfrak{A},\ldots\mid \partial_x,\partial_y,\partial_z)^2 U \;=\; (\mathfrak{A},\ldots\mid a,b,c,2f,2g,2h),
\]
and thence
\[
\Omega.\partial U \;=\; (\mathfrak{A},\ldots\mid \partial a,\partial b,\partial c,2\partial f,2\partial g,2\partial h);
\]
and observing that
\[
H \;=\; \begin{vmatrix} a & h & g \\ h & b & f \\ g & f & c \end{vmatrix},\qquad \partial H \;=\; \begin{vmatrix} \partial a & h & g \\ \partial h & b & f \\ \partial g & f & c \end{vmatrix} + \begin{vmatrix} a & \partial h & g \\ h & \partial b & f \\ g & \partial f & c \end{vmatrix} + \begin{vmatrix} a & h & \partial g \\ h & b & \partial f \\ g & f & \partial c \end{vmatrix}
\]
\[
= (\mathfrak{A},\mathfrak{H},\mathfrak{G}\mid \partial a,\partial h,\partial g) + (\mathfrak{H},\mathfrak{B},\mathfrak{F}\mid \partial h,\partial b,\partial f) + (\mathfrak{G},\mathfrak{F},\mathfrak{C}\mid \partial g,\partial f,\partial c),
\]
\[
= (\mathfrak{A},\ldots\mid \partial a,\partial b,\partial c,2\partial f,2\partial g,2\partial h),
\]
we see that
\[
\Omega\,\partial U \;=\; \partial H.
\]
Moreover
\[
(\mathfrak{A},\ldots\mid \lambda,\mu,\nu\mid \partial_x,\partial_y,\partial_z)^2.\partial U
\]
\[
= a(b\rho^2 + c\mu^2 - 2f\mu\nu) + b(c\lambda^2 + a\nu^2 - 2g\nu\lambda) + c(a\mu^2 + b\lambda^2 - 2h\lambda\mu)
\]
\[
\;+\; 2f(-f\lambda^2 + g\nu^2 + h\lambda\nu - c\mu\nu) + 2g(\ldots) + 2h(\ldots)
\]
\[
= \nabla^2 H = (\mathfrak{A},\ldots\mid \lambda,\mu,\nu\mid \partial_x H,\partial_y H,\partial_z H)
\]
\[
= (\mathfrak{A},\mathfrak{B},\mathfrak{C},\mathfrak{F},\mathfrak{G},\mathfrak{H}\mid \lambda,\mu,\nu)\cdot\partial H.
\]
Hence the equation
\[
\Phi U\partial U - \nabla^2\partial U \;=\; H\Gamma\partial U
\]
becomes
\[
\nabla^2\partial U \;=\; \Psi\,\partial H - H\,\partial\Phi.
\]

\medskip

\textbf{69.} Proof of equation $\partial\partial_1^2 U \;=\; \tfrac{1}{(m-1)^2}(\Psi\partial H - H\partial\Phi).$

We have
\[
\partial^2 \;=\; \tfrac{1}{(m-1)^2}\bigl(\Phi^2(x\partial_x + y\partial_y + z\partial_z)^2 - 2\Phi\Pi(x\partial_x + y\partial_y + z\partial_z)\nabla + \Pi^2\nabla^2\bigr),
\]
and thence multiplying by $\partial_1 = \partial$ and with the result operating upon $U$, we find
\[
((m-1)^2\,\partial_1^2 U) \;=\; \tfrac{1}{(m-1)^4}\bigl(\Phi^2 H U - 2\Phi\Pi H + \Pi^2\nabla^2 U\bigr).
\]
But $\partial U = 0$, and thence also $\nabla(\partial U) = 0$, that is $(m-1)(\nabla.\partial)U + \nabla\partial U = 0$; moreover $\nabla.\partial = 0$, and therefore $(\nabla.\partial)U = 0$, whence also $\nabla\partial U = 0$. Therefore
\[
\partial\partial_1^2 U \;=\; \tfrac{1}{(m-1)^2}\,\Pi^2\nabla^2.\partial U;
\]
or substituting for $\nabla^2.\partial U$ its value $= \Psi\partial H - H\partial\Phi$, we have the required expression for $\partial\partial_1^2 U$.

\medskip

\textbf{70.} Proof of equation [for $\partial_1\partial_2 U$].

We have
\[
\partial_2 \;=\; \tfrac{1}{m-1}\Phi(x\partial_x + y\partial_y + z\partial_z) + \tfrac{1}{m-1}\nabla,
\]
and thence multiplying by $\partial_1 = \partial$, and operating on $U$,
[reduction by similar manipulations as above, using $\nabla\partial U = 0$ and the explicit expressions in Article No.~31].

To reduce $(\partial.\nabla)\partial^2 U$, we have
\[
\partial(\nabla\partial^2 U) \;=\; \nabla\partial^3 U + (\partial.\nabla)\partial^2 U,
\]
\[
= \nabla\partial^3 U + \bigl((\partial.\nabla)\partial^2 + \nabla(\partial.\partial^2)\bigr)U,
\]
\[
= \nabla\partial^3 U + (\partial.\nabla)\partial^2 U + 2\nabla\partial\partial_2 U,
\]
and since
\[
\partial \;=\; -\tfrac{\Phi}{m-1}(x\partial_x + y\partial_y + z\partial_z) + \tfrac{1}{m-1}\Pi\nabla,
\]
multiplying by $\nabla\partial$, and with the result operating on $U$, we obtain
\[
\nabla\partial.\partial U \;=\; \tfrac{m-1}{m-1}\,\nabla.\partial U\cdot(\ldots),
\]
or since $\nabla\partial U = 0$, this is
\[
(\partial.\nabla)\partial^2 U \;=\; \partial(\nabla\partial^2 U) - \nabla\partial^3 U - \tfrac{2}{m-1}\nabla^2\partial U.
\]
Hence
\[
\partial.\nabla\partial^2 U \;=\; \partial(\nabla\partial^2 U) - \nabla\partial^3 U - \tfrac{2}{m-1}\nabla^2\partial U.
\]
Substituting this value of $(\partial.\nabla)\partial^2 U$, we find
\[
\partial_1\partial_2 U \;=\; -\tfrac{1}{m-1}\Phi\partial^2 U - \tfrac{1}{m-1}\Phi\partial^2 U + \bigl(-2\nabla^2\partial U\bigr),

the three lines whereof are to be separately further reduced.

\textbf{71.} For the first line we have
\[
\partial^2 \;=\; \tfrac{1}{(m-1)^2}\bigl(\Phi^2 - \Phi\Pi\nabla + (m-1)\Pi^2\Omega\bigr),
\]
and hence
\[
\text{first line of } \partial_1\partial_2 U \;=\; \tfrac{1}{(m-1)^3}\bigl((m-2)m\Phi^2 + \Phi\partial H\bigr).
\]

\textbf{72.} For the second line, we have
\[
\nabla(\partial^2 U) \;=\; \nabla\partial^2 U + 2(\nabla.\partial)\partial U
\]
or writing
\[
= \nabla\partial^2 U,\quad \text{since } \nabla.\partial = 0,\text{ and therefore }(\nabla.\partial)\partial U = 0;
\]
\[
\nabla\partial^2 U \;=\; \nabla(\partial^2 U) \;=\; \nabla\bigl(\tfrac{\Phi}{m-1}\Phi - \tfrac{1}{m-1}\Phi\nabla\bigr).
\]
By computation,
\[
\nabla\partial^2 U \;=\; \tfrac{m'}{(m-1)^2}(\Phi\nabla\Phi + \Phi\nabla\Psi) - \tfrac{1}{(m-1)^2}(\Psi\nabla H + 2\Phi\nabla\Psi);
\]
\[
U = 0,\ \nabla U = \tfrac{1}{m-1}H,\ \nabla\partial U = \Phi,
\]
this is
\[
\nabla\partial^2 U \;=\; \tfrac{1}{(m-1)^2}\bigl(m\Phi\nabla\Phi - \Psi\nabla H\bigr),
\]
whence also
\[
\nabla\partial^3 U \;=\; \tfrac{1}{(m-1)^2}(m\Phi\nabla\Phi - 2\Phi\partial H) - \tfrac{2}{(m-1)^2}\Pi\nabla\nabla\partial^2 U,
\]
\[
\nabla\partial^3 U \;=\; \nabla(\partial^3 U) \;=\; \tfrac{1}{(m-1)^3}\bigl[(\text{combinations of }\Phi,\Psi,H,\partial H,\nabla H,\&\text{c.})\bigr].
\]
Similarly
\[
\nabla\partial^4 U \;=\; \nabla(\partial^4 U) \;=\; \tfrac{1}{(m-1)^4}\bigl[\ldots\bigr],
\]
\[
= \tfrac{m'}{(m-1)^3}(\Phi\nabla H + \Phi\nabla\Psi) - \tfrac{1}{(m-1)^4}(\Phi^2\nabla H + 2\Phi\Psi\partial H + 2\Phi^2\nabla\Psi);
\]
or putting
\[
U = 0,\ \nabla U = \tfrac{1}{m-1}H,\ \nabla\partial U = \Phi,
\]
and observing also that $\nabla(\partial H) = \nabla\partial H + (\nabla.\partial)H$ is equal to $\nabla\partial H$, that is to $\partial\nabla H$, we obtain
\[
\nabla\partial^2 U \;=\; -\tfrac{1}{(m-1)^2}(m\Phi\partial\Phi - 2\Phi\partial H) - \tfrac{1}{(m-1)^2}\partial\nabla H;

and then from the above value of $\partial(\nabla\partial^2 U)$, we find
\[
\partial(\nabla\partial^2 U) - \nabla\partial^3 U \;=\; \tfrac{1}{(m-1)^2}\bigl(-2\Psi\partial\Phi + m\Phi\partial H\bigr) + \tfrac{1}{(m-1)^3}\bigl(-(\partial.\nabla)H\bigr);

or observing that the term multiplied by $\tfrac{1}{(m-1)^3}$ is $= -(\partial.\nabla)H$, we find
\[
\text{second line of } \partial_1\partial_2 U \;=\; \tfrac{1}{(m-1)^3}\bigl(-2\Psi\partial\Phi + m\Phi\partial H\bigr) + \tfrac{1}{(m-1)^4}\bigl(-(\partial.\nabla)H\bigr).

\medskip

\textbf{73.} For the third line, substituting for $\nabla^2\partial U$ its value $= \Psi\partial H - H\partial\Phi$, we have

\text{third line of } \partial_1\partial_2 U \;=\; -\tfrac{2}{(m-1)^2}(\Psi\partial H - H\partial\Phi).

\textbf{74.} Hence, uniting the three lines, we have

\tfrac{1}{(m-1)^3}\bigl((m-2)m\Phi^2 + \Phi\partial H\bigr)\;+\;\tfrac{1}{(m-1)^3}\bigl(-2\Psi\partial\Phi + m\Phi\partial H\bigr) \;+\; \tfrac{1}{(m-1)^4}\bigl(-(\partial.\nabla)H\bigr)

\[
\;+\; \tfrac{1}{(m-1)^2}\bigl((2m-2)H\partial\Phi + (-2m+2)\Psi\partial H\bigr),

and, reducing, we have the above-mentioned value of $\partial_1\partial_2 U$.

\section*{342. On the Conics which Pass through Three Given Points and Touch a Given Line.}

\noindent\textit{[From the Quarterly Journal of Pure and Applied Mathematics, vol.~vi.\ (1864), pp.~24--30.]}

\medskip

Consider the system of conics which pass through three given points and touch a given line; if among these we select the conics which touch an assumed line, it is easy to show analytically that there are four such conics, all real or else all imaginary; viz.\ the three points form a triangle, and if the two lines cut the three sides produced or cut the same two sides and the third side produced, then the conics are all real; but in every other case they are all imaginary. The latter part of the theorem may also be seen geometrically; in fact, if a triangle is inscribed in a conic, say first in an ellipse, or in a parabola, or in one branch of a hyperbola, then all the tangents of the conic (and therefore any two tangents whatever) cut the three sides produced, but if the triangle is inscribed in the two branches of a hyperbola (that is, two vertices on one branch and the remaining vertex on the other branch), then all the tangents of the conic (and therefore any two tangents whatever) cut the same two sides and the third side produced: and thus the only real conics are those which cut the three sides produced, or else the same two sides and the third side produced.

\medskip

The analytical proof referred to is as follows: taking $(x = 0,\,y = 0,\,z = 0)$ for the equations of the sides of the triangle, the equation of a conic through the three points is
\[
\tfrac{f}{x} + \tfrac{g}{y} + \tfrac{h}{z} \;=\; 0,
\]
or, what is the same thing,
\[
2fyz + 2gzx + 2hxy \;=\; 0,
\]
that is
\[
(0,\,0,\,0,\,f,\,g,\,h\mid x,\,y,\,z)^2 \;=\; 0.
\]
The inverse coefficients are
\[
(-f^2,\,-g^2,\,-h^2,\,gh,\,hf,\,fg),
\]
and hence the condition in order that the conic may touch the line $\alpha x + \beta y + \gamma z = 0$ is
\[
(f^2,\,g^2,\,h^2,\,-gh,\,-hf,\,-fg\mid \alpha,\,\beta,\,\gamma)^2 \;=\; 0,
\]
or, what is the same thing,
\[
\sqrt{\alpha f} + \sqrt{\beta g} + \sqrt{\gamma h} \;=\; 0.
\]
Similarly the condition in order that the conic may touch the line $lx + my + nz = 0$ is
\[
\sqrt{lf} + \sqrt{mg} + \sqrt{nh} \;=\; 0.
\]
Hence if the conic touch the two lines, we have
\[
\sqrt{lf} : \sqrt{mg} : \sqrt{nh} \;=\; \sqrt{\beta n} - \sqrt{\gamma m} : \sqrt{\gamma l} - \sqrt{\alpha n} : \sqrt{\alpha m} - \sqrt{\beta l},
\]
or, what is the same thing,
\[
f : g : h \;=\; \beta n + \gamma m - 2\sqrt{\beta\gamma m n} : \gamma l + \alpha n - 2\sqrt{\gamma\alpha n l} : \alpha m + \beta l - 2\sqrt{\alpha\beta l m},
\]
which, since the radicals must be so taken that the product may be $= \alpha\beta\gamma lmn$, gives in all four conics: and these will be all real if the signs of $(l,m,n)$ are the same with, or opposite to those of $(\alpha,\beta,\gamma)$ respectively; which proves the theorem.

\medskip

In particular since infinity is a line meeting the three sides produced; if the given line meet the three sides produced, the system will contain four real parabolas; but, if the given line meets two sides and a side produced, there is not any real parabola. In the latter case, as is obvious geometrically, the conics of the system are all hyperbolas.

\medskip

Any side of the triangle, and the line joining the opposite vertex with the point of intersection of the side and given line, form a pair of lines passing through the three points and meeting on the given line; such pair of lines is a conic of the system; and we have thus three pairs of lines, each pair a conic of the system.

\medskip

We may by what precedes form some idea of the nature of the system of conics which pass through the three given points and touch the given line. In fact writing at the point of contact the letters $H, P, E, L$ according as the conic is a hyperbola, parabola, ellipse, or pair of lines, then if the given line cut the three sides produced, we have as in fig.~1.

\begin{center}\textit{[Fig.~1: schematic showing labels $H,\,H,\,L$ along the given line, indicating sequence of conic types.]}\end{center}

\medskip

Whereas, if the given line cuts two sides and a side produced, we have more simply as in fig.~2.

\begin{center}\textit{[Fig.~2: simpler diagram with labels along the given line.]}\end{center}

\medskip

But to gain a more precise knowledge, it is proper to consider the curve which is the locus of the centres of the conics of the system.

\medskip

Such locus which, as will presently be seen, is a curve of the fourth order, must, it is clear, pass through the points of intersection ($L_1,\,L_2,\,L_3$ in figs.~1 and~2 respectively and $p,\,q,\,r$ in fig.~3 presently referred to) of the sides with the given line; and it is not difficult to show geometrically that it touches, at these points, the sides of the triangle. It may be shown also that the curve has three nodes (double points), viz.\ the middle point of each side of the triangle is a node of the curve. In fact if upon any side as base we apply an equal and opposite triangle so as to form with the given triangle a parallelogram, then any conic through the four vertices of the parallelogram will have for its centre the central point of the parallelogram; that is, the middle point of the side in question. But we may through the four vertices describe two conics, each of them touching the given line; that is the middle point of the side is the centre of two different conics of the system, and it is therefore a node upon the curve of centres. And moreover the node will be a crunode or an acnode (i.e.\ a double point with two real branches, or else a conjugate or isolated point) according as the conics are real or imaginary: and it is easy to see that if the given line does not cut the parallelogram, or if it cuts two opposite sides, the conics will be both real; but if it cuts two adjacent sides the conics will be both imaginary; that is, in the former case we have a crunode, and in the latter an acnode. Through each node may be drawn two tangents to the curve; and it is a known property of curves of the fourth order that the six points of contact lie on a conic; one of the tangents through the node is however the side whereon the node lies, and the points of contact of the three sides lie on a line, viz.\ the given line: hence the last mentioned conic is composed of the given line, and another line; that is, the three points of contact of the other tangents through the three nodes lie on this other line.

\medskip

It is proper to add that the points at infinity of the curve of centres are the centres of the four parabolas; that is, there will be four infinite branches, if the parabolas are real, viz.\ if the given line cuts the three sides produced; but no infinite branch if the parabolas are imaginary, viz.\ if the given line cut two sides and a side produced.

\medskip

The triangle and the three triangles applied to the three sides form together a triangle similar to the original triangle but of double the linear magnitude, and the form of the curve of centres depends as has been shown on the position of the given line in regard to the triangle and the double triangle. The cases to be considered are tolerably numerous, but it is easy from the foregoing considerations, to determine in any particular case what is the form of the curve of centres; for facility of selection I select a form without infinite branches, see fig.~3, in which the given line cuts the two sides $CA,\,CB$, and the third side $AB$ produced; it is moreover to be observed

\begin{center}\textit{[Fig.~3: triangle $ABC$ with the given line cutting $CA,\,CB$ and producing across $AB$; middle points $Q,P,R$ marked; the curve of centres is drawn through $p,\,q,\,r$ with two crunodes at $Q,P$ and an acnode at $R$.]}\end{center}

that as the figure is drawn the given line cuts the two sides $CA,\,CB$ below their middle points $Q$ and $P$ respectively. By what precedes it appears that the middle points $Q,\,P$ of these two sides $CA,\,CB$ are each of them crunodes, but that the middle point $R$ of the remaining side $AB$ is an acnode. And this being so the general form of the curve is at once perceived to be that shown by fig.~3.

\medskip

It is very interesting to trace the corresponding positions of the point of contact on the given line, and of the centre on the curve of centres. When the point of contact is at $\infty$, the centre is at $I$; as the point of contact moves from $\infty$ to $q$, the centre moves from $I$ to $q$, and at $q$ the two coincide; as the point of contact moves from $q$ to a point $Q_2$, the centre moves from $q$ to $Q$ (along the branch $Q_2$); as the point of contact moves from $Q_2$ to a point $P_1$, the centre moves from $Q$ to $P$ (along the branch $Q_2\,1\,P$); as the point of contact moves from $P_1$ to $p$, the centre moves from $P$ to $p$ (along the branch $P_1$) and at $p$ the point of contact and the centre again coincide; as the point of contact moves from $p$ to $r$, the centre moves from $p$ to $r$ and at $r$ they again coincide; as the point of contact moves from $r$ to a point $P_2$ the centre moves from $r$ to $P$ (along the branch $2P$); as the point of contact moves from $P_2$ to a point $Q_1$, the centre moves from $P$ to $Q$ (along the branch $P_2\,1\,Q$) and finally as the point of contact moves from $Q_1$ to $\infty$, the centre moves from $Q$ (along the branch $Q_1$) to $I$, thus completing the circuit.

\medskip

The equation of the curve of centres was given in the late Mr Hearn's ``\textit{Researches on Curves of the Second Order}, \&c.\ London, 1846,'' viz.\ if $x = 0,\;y = 0,\;z = 0$ be the equations of the sides of the triangle formed by the given points; $x + y + z = 0$ the equation of the line infinity, and $\alpha x + \beta y + \gamma z = 0$ the equation of the given line, then the equation of the curve of centres is
\[
\sqrt{\alpha x(-x + y + z)} + \sqrt{\beta y(x - y + z)} + \sqrt{\gamma z(x + y - z)} \;=\; 0,
\]
or more generally if $x + y + z = 0$ be the equation of an assumed line, then this equation is that of the locus of the pole of the assumed line in regard to the conics passing through the given points and touching the given line, see my paper ``Note on a Family of Curves of the Fourth Order,'' \textit{Cambridge and Dublin Mathematical Journal}, t.~V.\ (1850), pp.~148--152, [85], where I have noticed the above mentioned property, that the conic through the points of contact of the tangents through the nodes breaks up into a pair of lines. It is I think worth while to show how the equation is obtained. The equation of a conic through the given points and touching the given line is
\[
(0,\,0,\,0,\,f,\,g,\,h\mid x,\,y,\,z)^2 \;=\; 0
\]
with the condition $\sqrt{\alpha f} + \sqrt{\beta g} + \sqrt{\gamma h} = 0$, and this being so, the coordinates of the pole in relation thereto, of the assumed line $x + y + z = 0$, are
\[
x : y : z \;=\; (-f + g + h)f : (f - g + h)g : (f + g - h)h.
\]
We have thence
\[
-x + y + z \;\;\text{proportional to}\;\; -(-f + g + h)f + (f - g + h)g + (f + g - h)h,
\]
that is, to $f^2 - (g - h)^2$, which is $= (f - g + h)(f + g - h)$, and combining with this the equation
\[
\alpha x \;\;\text{proportional to}\;\; (-f + g + h)f,
\]
we obtain
\[
\alpha x(-x + y + z) \;\;\text{proportional to}\;\; \alpha f,
\]
that is
\[
\alpha x(-x + y + z) : \beta y(x - y + z) : \gamma z(x + y - z) \;=\; \alpha f : \beta g : \gamma h,
\]
so that from the equation $\sqrt{\alpha f} + \sqrt{\beta g} + \sqrt{\gamma h} = 0$, we have at once the foregoing equation
\[
\sqrt{\alpha x(-x + y + z)} + \sqrt{\beta y(x - y + z)} + \sqrt{\gamma z(x + y - z)} \;=\; 0.
\]
The rationalised form is
\[
(1,\,1,\,1,\,-1,\,-1,\,-1\mid \alpha x(-x + y + z),\;\beta y(x - y + z),\;\gamma z(x + y - z))^2 \;=\; 0,
\]
which shows what has been all along assumed, that the curve is of the fourth order. This equation may be transformed into
\[
x^2(\alpha^2 x^2 + \beta^2 y^2 + \gamma^2 z^2 - 2\beta\gamma yz + 2\gamma\alpha zx + 2\alpha\beta xy)
\]
\[
\;+\; y^2(\alpha^2 x^2 + \beta^2 y^2 + \gamma^2 z^2 + 2\beta\gamma yz - 2\gamma\alpha zx + 2\alpha\beta xy)
\]
\[
\;+\; z^2(\alpha^2 x^2 + \beta^2 y^2 + \gamma^2 z^2 + 2\beta\gamma yz + 2\gamma\alpha zx - 2\alpha\beta xy)
\]
\[
\;-\; 2yz(\alpha x + \beta y + \gamma z)(-\alpha x + \beta y + \gamma z)
\]
\[
\;-\; 2zx(\alpha x + \beta y + \gamma z)(\alpha x - \beta y + \gamma z)
\]
\[
\;-\; 2xy(\alpha x + \beta y + \gamma z)(\alpha x + \beta y - \gamma z) \;=\; 0;
\]
if with this equation we combine the equation $\alpha x + \beta y + \gamma z = 0$, we find at the points of intersection with the given line
\[
x^2\cdot 2\beta\gamma yz + y^2\cdot 2\gamma\alpha zx + z^2\cdot 2\alpha\beta xy \;=\; 0,
\]
that is
\[
xyz(\beta\gamma x + \gamma\alpha y + \alpha\beta z) \;=\; 0,
\]
so that the points in question are the intersections of the given line $\alpha x + \beta y + \gamma z = 0$, with the lines $x = 0,\;y = 0,\;z = 0,\;\tfrac{\alpha}{x} + \tfrac{\beta}{y} + \tfrac{\gamma}{z} = 0$. The point $\alpha x + \beta y + \gamma z = 0$, $\tfrac{\alpha}{x} + \tfrac{\beta}{y} + \tfrac{\gamma}{z} = 0$ corresponds to the conic which touches the given line at its intersection with the assumed line $x + y + z = 0$, the pole in relation to this conic is obviously a point on the given line. The point in question, if $x + y + z = 0$ denote the line infinity, is the point $I$ of fig.~3.

\medskip

It may be proper to mention a far less symmetrical form of the equation of the conic, but which has the advantage of putting in evidence the point of contact; viz.\ the equation is expressed in terms of the parameter $\alpha$ denoting the distance of the point of contact from a given point in the base line, and which is therefore very convenient for tracing the changes of form of the conic. Assuming as before that the base line cuts the sides produced, then (see fig.~4) if of the three points $1$ denote

\begin{center}\textit{[Fig.~4: schematic with axes $x$ and the base line $y = 0$, points labelled $1$ farthest, $2$ closest to base line.]}\end{center}

that which is furthest from, and $2$ that which is nearest to the base line, and if the base line be taken as the axis of $x$, and $23$ as the axis of $y$; the equation of the base line is $y = 0$, and the equations of the sides $23,\,31,\,12$ are $x = 0$, $\tfrac{x}{a} + \tfrac{y}{b} = 1$, $\tfrac{x}{a'} + \tfrac{y}{b'} = 1$, where $a,\,b,\,a',\,b',\,a' - a,\,b' - b,\,a'b - ab'$ are all positive, so that, by choosing the axes as above, we avoid the consideration of the several cases corresponding to different signs of these quantities. And this being so, if $x = \alpha$ is the coordinate of the point of contact, the equation of the conic is
\[
(A,\,B,\,C,\,F,\,G,\,H\mid x,\,y,\,1)^2 \;=\; 0,
\]
where
\[
A \;=\; 2bb'(a' - a),
\]
\[
B \;=\; 2\alpha^2(\alpha' - c),
\]
and these give
\[
C \;=\; -2\alpha aa'b(b' - b)\alpha,
\]
\[
F \;=\; -2\alpha bb'(a' - a),
\]
\[
G \;=\; -2abb'(a' - a),
\]
\[
H \;=\; \bigl\{(\alpha^2 + a a')(b' - b) - 2\alpha(ab' - a'b)\bigr\},
\]
\[
AB - H^2 \;=\; -\bigl[\bigl((a - \alpha)\sqrt{b'} + (a - a')\sqrt{b}\bigr)^2 - (a' - a)(a'b - ab')\bigr]\times\bigl[\bigl((a - \alpha)\sqrt{b'} - (a - a')\sqrt{b}\bigr)^2 - (a' - a)(a'b - ab')\bigr],
\]
\[
BC - F^2 \;=\; -a^4(a' - a)^2(b' - b)^2,
\]
\[
CA - G^2 \;=\; 0,
\]
\[
GH - AF \;=\; -2bb'(a' - a)(b' - b)\alpha(a - \alpha)(\alpha - a'),
\]
\[
HF - BG \;=\; -(a' - a)(b' - b)\alpha^2\bigl\{(b' + b)(\alpha^2 + aa') - 2\alpha(ab' + a'b)\bigr\},
\]
\[
FG - CH \;=\; -2bb'(a' - a)(b' - b)\alpha^2(a - \alpha)(\alpha - a').
\]

\medskip

The condition that the conic may be a parabola is $AB - H^2 = 0$, which gives, as it should do, four real values of $\alpha$.

\section*{343. On the Cusp of the Second Kind or Nodecusp.}

\noindent\textit{[From the Quarterly Journal of Pure and Applied Mathematics, vol.~vi.\ (1864), pp.~74--75.]}

\medskip

The so-called cusp of the second kind or ramphoid cusp, is not an ordinary singularity of plane curves, but it is a singularity of a higher order. It is however particularly considered in Pl\"ucker's \textit{Theorie der Analytischen Curven}, 1839; and it is there, not only in the analytical discussion of the singularities of plane curves, but in the author's theory of the generation of a curve, considered as described and enveloped by a point moving along a line which at the same time rotates round the point; when the motion along the line vanishes, we have a cusp; when the motion round the point vanishes, we have an inflexion; when the two motions vanish together, we have a cusp of the second kind, which thus presents itself as a singularity uniting the characters of a cusp or stationary point, and an inflexion or stationary tangent: (I remark in passing that in this explanation it is not clear what is the independent variable wherewith the motions are compared). But there is another point of view from which the singularity in question may be considered, viz.\ it may be regarded as a singularity arising from the union and amalgamation of a cusp, and a double point or node; in fact, in the figure, which represents a curve having a cusp and also a node, we have only to imagine the node approaching nearer and nearer to and ultimately coinciding with the cusp, and it will be at once seen that the point will become a cusp of the second kind; or as it might properly, with reference to this generation of it, be termed, a ``nodecusp.'' It is to be noticed that in the point-theory of curves, there is between the cusp and the nodecusp the intermediate singularity of the tacnode, which arises from the union and amalgamation of two nodes, and possesses the character of a cusp.

\medskip

I return to the nodecusp; taking the point in question as the origin, and the tangent for the axis of $x$, the equation will be a specialised form of the equation
\[
\tfrac{1}{2}y^2 + \tfrac{1}{6}(a,\,b,\,c,\,d\mid x,\,y)^3 + \tfrac{1}{24}(a',\,b',\,c',\,d',\,e'\mid x,\,y)^4 + \&c. \;=\; 0,
\]
which belongs to the case of a cusp, viz.\ (see Pl\"ucker, p.~165) the conditions satisfied by the special form are $a = 0,\;a' = 3b^2$, or the equation is
\[
\tfrac{1}{2}(y + \tfrac{1}{2}bx^2)^2 + \tfrac{1}{6}y^2(3cx + dy) + \tfrac{1}{24}y(4b'x^3 + 6c'x^2 y + 4d'xy^2 + e'y^3) + \&c. \;=\; 0,
\]
which is most easily verified, by observing that (this being so) the expansion of $y$ in terms of $x$ will be of the form
\[
y \;=\; -\tfrac{1}{2}bx^2 + Ax^3 + \&c.
\]

\medskip

It is now to be shown how the foregoing conditions $a = 0,\;a' = 3b^2$, are obtained by assuming that the curve has, besides the cusp, a node which ultimately coincides with the cusp. Let $(\alpha,\,\beta)$ be the coordinates of the node; we must have
\[
\tfrac{1}{2}\beta^2 + \tfrac{1}{6}(a,\,b,\,c,\,d\mid \alpha,\,\beta)^3 + \tfrac{1}{24}(a',\,b',\,c',\,d',\,e'\mid \alpha,\,\beta)^4 + \&c. \;=\; 0,
\]
\[
\tfrac{1}{2}(a,\,b,\,c\mid \alpha,\,\beta)^2 + \tfrac{1}{6}(a',\,b',\,c',\,d'\mid \alpha,\,\beta)^3 + \&c. \;=\; 0,
\]
\[
\beta + \tfrac{1}{2}(b,\,c,\,d\mid \alpha,\,\beta)^2 + \tfrac{1}{6}(b',\,c',\,d',\,e'\mid \alpha,\,\beta)^3 + \&c. \;=\; 0.
\]
Assume $\beta = m\alpha^2$, and then let $\alpha$ vanish; the equations become in the first instance
\[
\tfrac{1}{2}\alpha\alpha^2 + (\tfrac{1}{6}m^2 + \tfrac{1}{2}bm + \tfrac{1}{6}a)\alpha^3 + \&c. \;=\; 0,
\]
\[
\tfrac{1}{2}a\alpha^2 + \&c. \;=\; 0,
\]
\[
(m + \tfrac{1}{2}b)\alpha^2 + \&c. \;=\; 0,
\]
the second and third equation give $a = 0,\;m + \tfrac{1}{2}b = 0$, and the first equation then gives
\[
\tfrac{1}{2}m^2 + \tfrac{1}{2}bm + \tfrac{1}{6}a' \;=\; 0;
\]
or substituting for $m$ its value $= -\tfrac{1}{2}b$, this is $a' = 3b^2$, or the required conditions are $a = 0,\;a' = 3b^2$, ut sup. The single condition $a = 0$ corresponds to the case of the cusp at a node.

\section*{344. On Certain Developable Surfaces.}

\noindent\textit{[From the Quarterly Journal of Pure and Applied Mathematics, vol.~vi.\ (1864), pp.~108--126.\,\footnote{Presented to the Royal Society and read 27 Nov., 1862, but withdrawn by permission of the Council.}]}

\medskip

If $U = 0$ be the equation of a developable surface, or say a developable, then the Hessian $HU$ vanishes, not identically, but only by virtue of the equation $U = 0$ of the surface; that is, $HU$ contains $U$ as a factor, or we may write $HU = U\cdot PU$; the function $PU$, which for the developable replaces as it were the Hessian $HU$, is termed the \textit{Prohessian}; and (since if $r$ be the order of $U$ the order of $HU$ is $4r - 8$) we have $3r - 8$ for the order of the Prohessian. If $r = 4$, the order of the Prohessian is also $4$, and in fact, as is known, the Prohessian is in this case $= U$. The Prohessian is considered, but not in much detail, in Dr Salmon's \textit{Geometry of Three Dimensions}, (1862), pp.~338 and 426 [Ed.\ 4 (1882), p.~408]: the theorem given in the latter place is almost all that is known on the subject. I call to mind that the tangent plane along a generating line of the developable meets the developable in this line taken $2$ times, and in a curve of the order $r - 2$; the line touches the curve at the point of contact, or say the \textit{ineunt}, on the edge of regression, and besides meets it in $r - 4$ points. The ineunt taken $3$ times, and the $r - 4$ points form a linear system of the order $r - 1$, and the Hessian of this system (considered as a curve of one dimension, or binary quantic) is a linear system of $2r - 6$ points; viz.\ it is composed of the ineunt taken $4$ times, and of $2r - 10$ other points. This being so, the theorem is that the generating line meets the Prohessian in the ineunt taken $6$ times, in the $r - 4$ points, and in the $2r - 10$ points ($6 + r - 4 + 2r - 10 = 3r - 8$); it is assumed that $r$ is at least $= 5$.

\medskip

The developables which first present themselves are those which are the envelopes of a plane
\[
(a,\,b,\,\ldots\mid t,\,1)^n \;=\; 0,
\]
where $t$ is an arbitrary parameter, and the coefficients $(a,\,b,\,\ldots)$ are linear functions of the coordinates; the equation of the developable is
\[
\Disct(a,\,b,\,\ldots\mid t,\,1)^n \;=\; 0,
\]
the discriminant being taken in regard to the parameter $t$. Such developable is in general of the order $2n - 2$, but if the second coefficient $b$ is $= 0$, or, more generally, if it is a mere numerical multiple of $a$, then $a$ will divide out from the equation, and we have a developable of the order $2n - 3$: the like property of course exists in regard to the last but one, and the last, of the coefficients of the function. We thus obtain developables of the orders $4,\,5,\,$ and $6$, sufficiently simple to allow of the actual calculation of their Prohessians, and the chief object of the present Memoir is to exhibit these Prohessians; but the Memoir contains some other researches in relation to the developables in question.

\bigskip

\begin{center}
\textit{Quartic Developable, Nos.\ 1 to 6.}
\end{center}

\textbf{1.} I consider first the developable of the fourth order
\[
U \;=\; a^2 d^2 - 6abcd + 4ac^3 + 4b^3 d - 3b^2 c^2,
\]
derived from the cubic function $(a,\,b,\,c,\,d\mid t,\,1)^3$, and which is in fact the general quartic developable.

\medskip

\textbf{2.} Taking $(a,\,b,\,c,\,d)$ as coordinates and omitting common numerical factors, the first derived functions are
\[
ad^2 - 3bcd + 2c^3,
\]
\[
-3acd + 6b^2 d - 3bc^2,
\]
\[
-3abd + 6ac^2 - 3b^2 c,
\]
\[
a^2 d - 3abc + 2b^3,
\]
(quantities which, if $(X,Y,Z,W\mid t,1)^3$ denote the cubicovariant of $(a,b,c,d\mid t,1)^3$, are equal to $(-W,\,3Z,\,-3Y,\,X)$ respectively). And the second derived functions are
\[
\begin{array}{cccc}
d^2 & -3cd & -3bd + 6c^2 & 2ad - 2bc \\
-3cd & 12bd - 3c^2 & -3ad - 6bc & -3ac + 6b^2 \\
-3bd + 6c^2 & -3ad - 6bc & 12ac - 3b^2 & -3ab \\
2ad - 3bc & -3ac + 6b^2 & -3ab & a^2
\end{array}
\]

\medskip

\textbf{3.} Representing these by
\[
\begin{pmatrix} A & H & G & L \\ H & B & F & M \\ G & F & C & N \\ L & M & N & P \end{pmatrix},
\]
and expressing the determinant in the partially developed form
\[
= (AM - LH)(FN - GM) + (AN - LG)(FM - BN) + (AP - L^2)(BC - F^2)
\]
\[
\;+\; (HN - GM)^2 + (HP - LM)(FG - CH) + (GP - LN)(HF - BG),
\]
and proceeding to the calculation, we find a tabular expansion of seven component products $AM - LH$, $FN - GM$, $AN - LG$, $FM - BN$, $AP - L^2$, $BC - F^2$, etc., each producing several terms in the monomials $a^p b^q c^r d^s$ with explicit integer coefficients. \textit{[See facsimile pp.~268--269 for the full numerical tableau.]}

\medskip

\textbf{4.} Hence, forming the six parts and collecting, we find the Hessian (the first column of the developed tableau). This is in fact $= U^2$, and hence the Prohessian is
\[
PU \;=\; U \;=\; a^2 d^2 - 6abcd + 4ac^3 + 4b^3 d - 3b^2 c^2,
\]
viz.\ for the quartic developable the Prohessian coincides with $U$, as expected.

\medskip

\textbf{5.} To complete the theory it is proper to calculate the inverse coefficients
\[
\mathfrak{A},\,\mathfrak{H},\,\mathfrak{G},\,\mathfrak{L},
\]
\[
\mathfrak{H},\,\mathfrak{B},\,\mathfrak{F},\,\mathfrak{M},
\]
\[
\mathfrak{G},\,\mathfrak{F},\,\mathfrak{C},\,\mathfrak{N},
\]
\[
\mathfrak{L},\,\mathfrak{M},\,\mathfrak{N},\,\mathfrak{P}.
\]
We have for example
\[
\mathfrak{P} \;=\; ABC - AF^2 - BG^2 - CH^2 + 2FGH,
\]
which is found to be
\[
\;=\; 3(ad^2 - 3bcd + 2c^3)^2 - 4d^2(a^2 d^2 - 6abcd + 4ac^3 + 4b^3 d - 3b^2 c^2),
\]
that is $= 3W^2 - 4d^2 U$; and calculating in like manner the other coefficients, the system is found to be
\[
\begin{array}{cccc}
3X^2 - 4a^2 U, & 3XY - 4ab U, & 3XZ + (2ac - 6b^2)U, & 3XW + (5ad - 9bc)U, \\
3YX - 4ab U, & 3Y^2 - 4ac U, & 3YZ - (7ad + 3bc)U, & 3YW + (2bd - 6c^2)U, \\
3ZX + (2ac - 6b^2)U, & 3ZY - (7ad + 3bc)U, & 3Z^2 - 4bd U, & 3ZW - 4cd U, \\
3WX + (5ad - 9bc)U, & 3WY + (2bd - 6c^2)U, & 3WZ - 4cd U, & 3W^2 - 4d^2 U
\end{array}
\]

\medskip

\textbf{6.} Let $(\lambda,\,\mu,\,\nu,\,\rho)$ be any arbitrary multipliers, and write
\[
(\mathfrak{X},\,\mathfrak{Y},\,\mathfrak{Z},\,\mathfrak{W}) \;=\; (\mathfrak{A},\,\mathfrak{H},\,\mathfrak{G},\,\mathfrak{L}\mid \lambda,\,\mu,\,\nu,\,\rho),
\]
\[
\phantom{(\mathfrak{X},\,\mathfrak{Y},\,\mathfrak{Z},\,\mathfrak{W}) = }(\mathfrak{H},\,\mathfrak{B},\,\mathfrak{F},\,\mathfrak{M}),
\]
\[
\phantom{(\mathfrak{X},\,\mathfrak{Y},\,\mathfrak{Z},\,\mathfrak{W}) = }(\mathfrak{G},\,\mathfrak{F},\,\mathfrak{C},\,\mathfrak{N}),
\]
\[
\phantom{(\mathfrak{X},\,\mathfrak{Y},\,\mathfrak{Z},\,\mathfrak{W}) = }(\mathfrak{L},\,\mathfrak{M},\,\mathfrak{N},\,\mathfrak{P}),
\]
then if $\theta = 3(\lambda X + \mu Y + \nu Z + \rho W)$, we have
\[
(\mathfrak{X},\,\mathfrak{Y},\,\mathfrak{Z},\,\mathfrak{W}) \;=\; (\theta X + \alpha U,\;\theta Y + \beta U,\;\theta Z + \gamma U,\;\theta W + \delta U).
\]
The functions $(X,\,Y,\,Z,\,W\mid t,\,1)^3$ are the cubicovariant of $(a,\,b,\,c,\,d\mid t,\,1)^3$, respectively, and if we also write for a moment the functions $u + \theta v$, $U = a^2 d^2 - \&c. = \mathit{ff}(a,b,c,d)$, then
\[
\mathit{ff}(a + \theta X,\,b + \theta Y,\,c + \theta Z,\,d + \theta W) \;=\; \Disct(u + \theta v) \;=\; (1 - \theta^2 U)\,U,
\]
by a formula given in my ``Fifth Memoir on Quantics,'' \textit{Phil.\ Trans.}, t.~CXLVIII.\ (1858), see p.~442 [156]; the function on the left-hand side thus contains $U$ as a factor, and it at once follows that the function
\[
U(a + \mathfrak{X},\,b + \mathfrak{Y},\,c + \mathfrak{Z},\,d + \mathfrak{W});
\]
viz., the function obtained from $U$ by writing therein $(a + \mathfrak{X},\,b + \mathfrak{Y},\,c + \mathfrak{Z},\,d + \mathfrak{W})$ in the place of $(a,\,b,\,c,\,d)$ respectively, contains $U$ as a factor, and therefore vanishes if $U = 0$; that is $a + \mathfrak{X},\,b + \mathfrak{Y},\,c + \mathfrak{Z},\,d + \mathfrak{W}$, are the coordinates of a point on the surface $U = 0$; they are in fact the coordinates of a point on the generating line through $(a,\,b,\,c,\,d)$; this is a theorem which applies to any developable whatever, as appears by the following considerations.

\bigskip

\begin{center}
\textit{Remarks on the General Theory of Developables, Nos.\ 7 to 9.}
\end{center}

\textbf{7.} In general for any surface whatever, taking a point on the surface, the successive polars of this point (the last of them being the tangent plane) all touch at this point; and not only so, but the tangents to the two branches of the curve in which the surface itself (or any of its polars down to the quadric polar) is intersected by the ultimate polar or tangent plane, are respectively coincident. Suppose that for any point on the surface, the quadric polar becomes a cone: the vertex of this cone is not the point itself; hence the tangent plane at the point touches the cone along a generating line; that is the tangents to the curve of intersection with the surface, or with any of its polars, coincide with the generating line of the cone---and the curve of intersection of the tangent plane with the surface, or any of its polars, at the point of contact (instead of, as in general, a node) has a cusp. In particular the curve of intersection with the surface has at the point of contact a cusp. The condition that the quadric polar may be a cone is $HU = 0$, and when this differential equation is satisfied in virtue of the equation $U = 0$ (that is, when we have identically $HU = U\cdot PU$), the surface is a developable. Now all that is proved in the first instance by the equation $HU = 0$ is that every point of the surface has the above mentioned property; viz., that the tangent plane at the point cuts the surface in a curve having a cusp at the point in question.

\medskip

\textbf{8.} What really happens in the case of a developable is more than this; viz.\ the curve of intersection is made up of the generating line taken twice, and of a curve of an order less by $2$ than the order of the surface. Let $(x,\,y,\,z,\,w)$ be the coordinates of the point on the developable, $U = 0$ the equation of the developable, $(A,\,B,\,C,\,P,\,F,\,Q,\,H,\,L,\,M,\,N)$ the second derived functions of $U$, $(\mathfrak{A},\,\mathfrak{B},\,\mathfrak{C},\,\mathfrak{F},\,\mathfrak{G},\,\mathfrak{H},\,\mathfrak{L},\,\mathfrak{M},\,\mathfrak{N},\,\mathfrak{P})$ the inverse system, $K$ the determinant formed with the second derived functions, so that we have $K = HU = 0$. The coordinates of the vertex of the cone are given by the equations
\[
\dot x : \dot y : \dot z : \dot w \;=\; \mathfrak{A} : \mathfrak{H} : \mathfrak{G} : \mathfrak{L}
\]
\[
= \mathfrak{H} : \mathfrak{B} : \mathfrak{F} : \mathfrak{M}
\]
\[
= \mathfrak{G} : \mathfrak{F} : \mathfrak{C} : \mathfrak{N}
\]
\[
= \mathfrak{L} : \mathfrak{M} : \mathfrak{N} : \mathfrak{P};
\]
these several sets of ratios being equivalent to each other in virtue of the equation $K = 0$. Hence $(\lambda,\,\mu,\,\nu,\,\rho)$ being arbitrary multipliers, if we write
\[
(\mathfrak{X},\,\mathfrak{Y},\,\mathfrak{Z},\,\mathfrak{W}) \;=\; (\mathfrak{A},\,\mathfrak{H},\,\mathfrak{G},\,\mathfrak{L}\mid \lambda,\,\mu,\,\nu,\,\rho),
\]
\[
\phantom{(\mathfrak{X},\,\mathfrak{Y},\,\mathfrak{Z},\,\mathfrak{W})\;=\;}(\mathfrak{H},\,\mathfrak{B},\,\mathfrak{F},\,\mathfrak{M}),
\]
\[
\phantom{(\mathfrak{X},\,\mathfrak{Y},\,\mathfrak{Z},\,\mathfrak{W})\;=\;}(\mathfrak{G},\,\mathfrak{F},\,\mathfrak{C},\,\mathfrak{N}),
\]
\[
\phantom{(\mathfrak{X},\,\mathfrak{Y},\,\mathfrak{Z},\,\mathfrak{W})\;=\;}(\mathfrak{L},\,\mathfrak{M},\,\mathfrak{N},\,\mathfrak{P}),
\]
the coordinates of the vertex of the cone will be as $\mathfrak{X} : \mathfrak{Y} : \mathfrak{Z} : \mathfrak{W}$, and hence observing that the absolute magnitudes of these quantities are arbitrary, $x + \mathfrak{X} : y + \mathfrak{Y} : z + \mathfrak{Z} : w + \mathfrak{W}$ will represent the coordinates of any point on the line joining the point $(x,\,y,\,z,\,w)$ with the vertex of the cone, that is, the generating line through the point $(x,\,y,\,z,\,w)$; which is the theorem in question, the coordinates being in the present investigation denoted by $(x,\,y,\,z,\,w)$ instead of the $(a,\,b,\,c,\,d)$ of the example.

\medskip

\textbf{9.} Reverting to the developable $U = a^2 d^2 - \&c. = 0$, the results previously obtained show that the coordinates of the vertex of the cone which is the quadric polar of the point $(a,\,b,\,c,\,d)$ are as $X : Y : Z : W$ (these quantities denoting as above the coefficients of the cubicovariant), and thence also that the coordinates of any point on the generating line will be as $a + \theta X : b + \theta Y : c + \theta Z : d + \theta W$, where $\theta$ is arbitrary.

\bigskip

\begin{center}
\textit{Special Quintic Developable, Nos.\ 10 to 25.}
\end{center}

\textbf{10.} We have, secondly, the developable of the fifth order
\[
U \;=\; a^3 e^2 + 6a^2 c^2 e - 24 ab^2 ce + 9ac^4 + 16b^4 e - 8b^2 c^3 \;=\; 0,
\]
derived from the quartic function $(a,\,2b,\,3c,\,0,\,-27e\mid t,\,1)^4$, or, what is the same thing, $at^4 + 8bt^3 + 18ct^2 - 27e = 0$, where it will be observed that, as well in the quartic function as in the equation of the developable, the sum of the numerical coefficients is $=$~zero; it was on this account that the foregoing form of the quartic function was selected in preference to the form $(a,\,b,\,c,\,0,\,e\mid t,\,1)^4$. The last mentioned form has for its discriminant
\[
(ae + 3c^2)^3 - 27(ace - b^2 e - c^3)^2 \;=\; e(a^3 e^2 - 18a^2 c^2 e + 54 ab^2 ce + 81 ac^4 - 27 b^4 e - 54 b^2 c^3),
\]
and writing therein in place of $(a,\,b,\,c,\,e)$, the values $(a,\,2b,\,3c,\,-27e)$, the second factor divided by $729$ gives the foregoing expression for $U$, belonging to the form
\[
(a,\,2b,\,3c,\,0,\,-27e\mid t,\,1)^4.
\]

\medskip

\textbf{11.} Taking $(a,\,b,\,c,\,e)$ as coordinates, and omitting common numerical factors, the first derived functions of $U$ are
\[
3a^2 e^2 + 12 ac^2 e - 24b^2 ce + 9c^4,
\]
\[
-48 abce + 64 b^3 e - 16 bc^3,
\]
\[
12 a^2 ce - 24 ab^2 c + 36 ac^3 - 24 b^2 c^2,
\]
\[
2a^3 e + 6a^2 c^2 - 24 ab^2 c + 16 b^4,
\]
and the second derived functions are
\[
\begin{array}{cccc}
3(ae^2 + 2c^2 e) & -24 bce & 6(2ace - 2b^2 e + 3c^3) & 3(a^2 e + 2a c^2 - 4b^2 c) \\
-24 bce & 8(-3ace + 12 b^2 e - c^3) & 24(-abe - bc^2) & 8(-3abc + 4b^3) \\
6(2ace - 2b^2 e + 3c^3) & 24(-abe - bc^2) & 6(a^2 e + 9 ac^2 - 4b^2 c) & 6(a^2 c - 2ab^2) \\
3(a^2 e + 2 ac^2 - 4b^2 c) & 8(-3abc + 4b^3) & 6(a^2 c - 2 ab^2) & a^3
\end{array}
\]

\medskip

\textbf{12.} Representing these by
\[
\begin{pmatrix} A & H & G & L \\ H & B & F & M \\ G & F & C & N \\ L & M & N & P \end{pmatrix},
\]
and expressing the determinant in the partially developed form
\[
P\,(ABC - AF^2 - BG^2 - CH^2 + 2FGH)
\]
\[
\;-\; L^2(BC - F^2) - M^2(CA - G^2) - N^2(AB - H^2)
\]
\[
\;-\; 2MN(GH - AF) - 2NL(HF - BG) - 2LM(FG - CH),
\]
then, proceeding to the calculation, we have a tabular expansion of the component products, each producing several terms in the monomials with explicit integer coefficients. \textit{[See facsimile pp.~272--274 for the full numerical tableau, with detailed expansion of $A^2$, $AH$, $AG$, $AL$, $H^2$, $HG$, $HL$, $M^2$, $JK$, $JL$, $2MJ$, $2LM$, etc., and the resulting expression of $H U$ as a tableau of monomials and integer coefficients.]}

\medskip

\textbf{13.} It is now easy to form the seven component terms of the determinant, and thence the determinant itself; each of the component terms divides by $576$, and omitting this factor, the sum of the seven terms divides by $2$; the result is the explicit polynomial expansion of $HU$. \textit{[See facsimile p.~273 for the full tableau.]}

\medskip

\textbf{14.} This divides as it should do by
\[
U \;=\; a^3 e^2 + 6a^2 c^2 e - 24 ab^2 ce + 9 ac^4 + 16 b^4 e - 8 b^2 c^3,
\]
and the quotient, which is the Prohessian, is found to be
\[
PU \;=\; -24 a^2 e^3 + 3 a^2 c^2 e + 10 a^2 b^2 c^2 e + 32 a c^4 + 8 a b^2 c e - 24 a c^3 + 32 b^2 e + 32 b^4 c^3, \;\;\text{etc.}
\]
\textit{[Full polynomial coefficients appear in the facsimile expansion p.~274.]}

\medskip

\textbf{15.} But before discussing the Prohessian, I will further consider the developable itself. Regarding it as derived from the equation $(a,\,2b,\,3c,\,0,\,-27e\mid t,\,1)^4 = 0$, we have, observing that
\[
eU \;=\; (27)^2\bigl[(ae - c^2)^3 + (-3ace + 4b^2 e - c^3)^2\bigr] \;=\; 0,
\]
\[
-3ace + 4b^2 e - c^3 \;=\; c(ae - c^2) - 4e(ac - b^2),
\]
it appears that the equations of the cuspidal curve or edge of regression of the developable are $ae - c^2 = 0,\;ac - b^2 = 0$ (so that the cuspidal curve is a curve of the fourth order, the intersection of two quadric surfaces, or say a quadri-quadric curve). This is perhaps better seen by writing the equation of the developable in the form
\[
U \;=\; a(ae - c^2)^2 - 8c(ae - c^2)(ac - b^2) + 16 e(ac - b^2)^2 \;=\; 0,
\]
or what is the same thing
\[
U \;=\; (a,\,c,\,e\mid ae - c^2,\;4ac - 4b^2)^2 \;=\; 0,
\]
where the discriminant of the quadric function is $= ae - c^2$, which vanishes for the curve $(ae - c^2 = 0,\;ac - b^2 = 0)$.

\medskip

\textbf{16.} Another form of the equation is
\[
U \;=\; a(ae + 3c^2)^2 - 8b^2(3ace - 2b^2 e + c^3) \;=\; 0,
\]
which shows that the conic $ae + 3c^2 = 0,\;b = 0$, is a nodal curve on the developable.

And, again, another form is
\[
U \;=\; c^3(9ac - 8b^2) + e(a^3 e + 6a^2 c^2 - 24 ab^2 c + 16 b^4) \;=\; 0,
\]
which shows that the conic $9ac - 8b^2 = 0,\;e = 0$, is a simple line on the developable.

\medskip

\textbf{17.} In my paper ``On the Developable Surfaces which arise from Two Surfaces of the Second Order,'' \textit{Camb.\ and Dub.\ Math.\ Jour.}, t.~v.\ (1850), pp.~46--57, [84], I considered first the developable having for its edge of regression the intersection of two quadric surfaces; in the general case the developable is of the order $8$; but if the two surfaces have an ordinary contact it is of the order $6$; and if they have a singular contact (as there explained) it is of the order $5$. And in the last mentioned case, if the equations of the two quadric surfaces are taken to be $x^2 - 2wz = 0$, $y^2 - 2zx = 0$, then the equation of the developable was found to be
\[
4z^2 w^2 + 12 z^2 x^2 + 9zx^4 - 24 zxy^2 w - 4x^3 y^2 + 8y^4 w \;=\; 0,
\]
which putting therein $z = a,\;y = 2b,\;x = 2c,\;w = 2e$, becomes
\[
a^3 e^2 + 6a^2 c^2 e - 24 ab^2 ce + 9 ac^4 + 16 b^4 e - 8 b^2 c^3 \;=\; 0,
\]
which is the before mentioned developable $U = 0$; the two equations $x^2 - 2wz = 0$, $y^2 - 2zx = 0$ become by the same substitution $ae - c^2 = 0,\;ac - b^2 = 0$. We have in fact already seen that the developable $U = 0$ has this curve for its edge of regression.

\medskip

\textbf{18.} But in the paper just referred to, it is also shown that considering the developable which is the envelope of the common tangent planes of two quadric surfaces; in the general case the developable is of the order $8$, but if the two surfaces have an ordinary contact it is of the order $6$, and if they have a singular contact it is of the order $5$.

\medskip

In the last mentioned case the surfaces may without loss of generality be reduced to conics, and their equations may be taken to be $(y^2 - 2zx = 0,\;w = 0)$ and $(x^2 - 2yw = 0,\;y = 0)$\,\footnote{These are in fact the conics made use of, p.~37 in the paper above referred to [and p.~495 in the reprint], but the equations are by mistake given as $(x^2 - 2yz = 0,\;w = 0)$, $(y^2 - 2zw = 0,\;x = 0)$, that is, $x$ and $y$ are interchanged.}, and this being so the equation of the developable is
\[
32 z^2 w^2 - 27 z^2 x^2 w + 72 zxy^2 w + 8 zx^4 - 27 y^4 w - 4 c^4 y^2 \;=\; 0.
\]
This is really a developable of the same kind with the first mentioned developable of the order $5$; for writing $x = 12c,\;y = 8b,\;z = 8a,\;w = -8e$, the equation becomes
\[
a^3 e^2 + 6 a^2 c^2 e - 24 ab^2 ce + 9 ac^4 + 16 b^4 e - 8 b^2 c^3 \;=\; 0,
\]
which is the before mentioned developable $U = 0$. The equations of the two conics become $(9 ac - 8b^2 = 0,\;e = 0)$ and $(ae + 3c^2 = 0,\;b = 0)$, and the developable is thus the envelope of the common tangent planes of these two conics. It has been seen that the first conic is a simple line, but the second conic a nodal line, on the developable.

\medskip

\textbf{19.} Recapitulating, the developable of the fifth order $U = 0$, which is the envelope of the plane $(a,\,2b,\,3c,\,0,\,-27e)(t,\,1)^4 = 0$ is the locus of the tangents of the quadri-quadric curve $(ae - c^2 = 0,\;ac - b^2 = 0)$, and it is also the envelope of the common tangent planes of the conics $(9ac - 8b^2 = 0,\;e = 0)$ and $(ae + 3c^2 = 0,\;b = 0)$.

\medskip

\textbf{20.} Returning now to the Prohessian, its equation may be written in the form
\[
PU \;=\; (3a^2 c,\;-8 b^2 c,\;ace + 4b^2 e\mid ae - c^2,\;4ac - 4b^2)^2 \;=\; 0,
\]
and the discriminant of the quadric function is
\[
3 a^2 ce(ac + 2b^2) - 9 b^2 c^3,
\]
which is
\[
\;=\; 3c\bigl\{(a^2 e + 3b^2 c)(ac - b^2) + 3 ab^2(ae - c^2)\bigr\},
\]
and recollecting that the equations of the cuspidal curve or edge of regression of the developable are $ae - c^2 = 0,\;ac - b^2 = 0$, it thus appears that the curve in question is also a cuspidal curve on the Prohessian.

\medskip

\textbf{21.} Consider for a moment the surface
\[
(3 a^2 c,\;-8 b^2 c,\;ace + 2b^2 e\mid ae - c^2,\;4ac - 4b^2)^2 \;=\; 0,
\]
and suppressing the discriminant is, taking $c$ as a coefficient, of the form
\[
(B,\,C)(ae - c^2)^2 + 2(\ldots)(ae - c^2)(ac - b^2) + (B,\,D)(ac - b^2)^2,
\]
where the coefficients $A,\,B,\,C$ are as before the differential coefficients of $U$ \textit{[see facsimile p.~275 for the complete development]}.
