\section{Deuxi\`eme Note sur une Formule pour la R\'eversion des S\'eries}

\textsc{[From the Journal f\"ur die reine und angewandte Mathematik (Crelle), t.~LIV. (1857), pp.~156--161: Sequel to Note t.~LIII. (1856), 229.]}
\label{paper:234}

I propose to show, in this second note, in what manner Jacobi's theorem leads to a formula given without demonstration by Mr.~Sylvester in his memoir entitled ``On the Change of Systems of Independent Variables,'' Quarterly Math.\ Journal, t.~I.\ [1857], p.~42 to 56 and 126 to 134. To fix the ideas I take the case of three variables, and I suppose that $f(x,y,z)$ is a rational and entire function of $x,y,z$, and that these variables are given as functions of $u,v,w$ by means of the equations $u=X$, $v=Y$, $w=Z$, where $X,Y,Z$ are rational and entire functions of $x,y,z$. But these functions are no longer subject to the condition (admitted in my first note) that $X-x$, $Y-y$, $Z-z$ contain only the terms of the second and higher orders. I write, to abridge,
\begin{align*}
\delta &= \frac{d(X,Y,Z)}{d(x,y,z)}, \\
D &= \frac{d}{dx}\,X'\frac{d}{dy}\,Y'\frac{d}{dz}\,Z',
\end{align*}
where $X'$, $Y'$, $Z'$ are the same functions of $x',y',z'$ that $X,Y,Z$ are of $x,y,z$; and I seek the coefficient of $x'^\alpha y'^\beta z'^\gamma$ in $f(x,y,z)$. If in the equation $f(x,y,z)=F(u,v,w)$, instead of $u,v,w$, we substitute their values $X,Y,Z$, we shall have an identical equation, and by differentiating with respect to $x,y,z$, we obtain
\[
\frac{df}{dx} = \frac{dF}{du}\frac{dX}{dx} + \frac{dF}{dv}\frac{dY}{dx} + \frac{dF}{dw}\frac{dZ}{dx},
\]
etc.; that is
\[
\frac{df}{dx} = DF \cdot \frac{dX}{dx}, \quad
\frac{df}{dy} = DF \cdot \frac{dX}{dy}, \quad
\frac{df}{dz} = DF \cdot \frac{dX}{dz},
\]
where $DF \cdot dX/dx$ denotes
\[
\frac{dF}{du}\frac{dX}{dx} + \frac{dF}{dv}\frac{dY}{dx} + \frac{dF}{dw}\frac{dZ}{dx}.
\]
But we have
\[
\frac{d(F\cdot\delta)}{dx}
= \frac{dF}{du}\frac{d(X,Y,Z)}{d(x,y,z)} + F\frac{d\delta}{dx}
= \delta \cdot DF\cdot\frac{dX}{dx} + F\frac{d\delta}{dx},
\]
and consequently
\[
\frac{df}{dx}\delta - f\frac{d\delta}{dx} = \frac{d(F\delta)}{dx};
\]
whence, integrating,
\[
f\delta = \int\left(\frac{df}{dx}\delta - f\frac{d\delta}{dx}\right)dx
= \int \frac{d(f\delta)}{dx}\,dx,
\]
that is $f\delta = F\delta$, an equation which may also be written
\[
f(x,y,z)\,\frac{d(X,Y,Z)}{d(x,y,z)} = F(X',Y',Z')\,\frac{d(X',Y',Z')}{d(x',y',z')}.
\]

\medskip
I write
\[
x' = A_{100}\,x + A_{010}\,y + A_{001}\,z,\qquad
y' = B_{100}\,x + B_{010}\,y + B_{001}\,z,\qquad
z' = C_{100}\,x + C_{010}\,y + C_{001}\,z,
\]
and representing by
\[
\begin{vmatrix}
a_{100} & b_{100} & c_{100} \\
a_{010} & b_{010} & c_{010} \\
a_{001} & b_{001} & c_{001}
\end{vmatrix}
\quad\text{the inverse matrix of}\quad
\begin{vmatrix}
A_{100} & A_{010} & A_{001} \\
B_{100} & B_{010} & B_{001} \\
C_{100} & C_{010} & C_{001}
\end{vmatrix},
\]
one obtains the equations
\begin{align*}
x &= a_{100}\,x' + b_{100}\,y' + c_{100}\,z', \\
y &= a_{010}\,x' + b_{010}\,y' + c_{010}\,z', \\
z &= a_{001}\,x' + b_{001}\,y' + c_{001}\,z'.
\end{align*}

It is almost superfluous to observe that if one denotes by $\nabla$ the determinant
\[
\nabla =
\begin{vmatrix}
A_{100} & A_{010} & A_{001} \\
B_{100} & B_{010} & B_{001} \\
C_{100} & C_{010} & C_{001}
\end{vmatrix},
\]
one has
\[
a_{100} = \frac{1}{\nabla}\,\frac{d\nabla}{dA_{100}},\qquad
a_{010} = \frac{1}{\nabla}\,\frac{d\nabla}{dA_{010}},\quad\text{etc.}
\]

We have in general
\[
\text{Coeff. } x'^\alpha y'^\beta z'^\gamma \text{ in } f(x,y,z)
= \frac{1}{\alpha!\,\beta!\,\gamma!}
\left(\frac{d}{dx'}\right)^\alpha
\left(\frac{d}{dy'}\right)^\beta
\left(\frac{d}{dz'}\right)^\gamma
f(x,y,z);
\]
and if in the function $f(x,y,z)$ we substitute for $x,y,z$ their values in $x',y',z'$, we obtain
\begin{multline*}
\left(\frac{d}{dx'}\right)^\alpha
\left(\frac{d}{dy'}\right)^\beta
\left(\frac{d}{dz'}\right)^\gamma f(x,y,z) \\
= \left(a_{100}\frac{d}{dx} + b_{100}\frac{d}{dy} + c_{100}\frac{d}{dz}\right)^\alpha
\left(a_{010}\frac{d}{dx} + b_{010}\frac{d}{dy} + c_{010}\frac{d}{dz}\right)^\beta \\
\times \left(a_{001}\frac{d}{dx} + b_{001}\frac{d}{dy} + c_{001}\frac{d}{dz}\right)^\gamma f(x,y,z).
\end{multline*}

But this formula may be replaced by a more convenient one. We have
\begin{align*}
\frac{d}{dx'} &= a_{100}\frac{d}{dx} + a_{010}\frac{d}{dy} + a_{001}\frac{d}{dz}, \\
\frac{d}{dy'} &= b_{100}\frac{d}{dx} + b_{010}\frac{d}{dy} + b_{001}\frac{d}{dz}, \\
\frac{d}{dz'} &= c_{100}\frac{d}{dx} + c_{010}\frac{d}{dy} + c_{001}\frac{d}{dz};
\end{align*}
and by means of these equations we may express
\[
\left(\frac{d}{dx'}\right)^\alpha
\left(\frac{d}{dy'}\right)^\beta
\left(\frac{d}{dz'}\right)^\gamma
\]
in the form of a sum of terms
\[
C \left(\frac{d}{dx}\right)^{\alpha'} \left(\frac{d}{dy}\right)^{\beta'} \left(\frac{d}{dz}\right)^{\gamma'}
\]
where $\alpha'+\beta'+\gamma' = \alpha+\beta+\gamma$. But the calculation of the coefficients $C$ would be very complicated. We may avoid this calculation by observing that the system of equations is equivalent to a single operation. In fact, let $\xi$, $\eta$, $\zeta$ be arbitrary quantities; we have identically
\begin{multline*}
(\xi x' + \eta y' + \zeta z')^n
= \bigl\{\xi(A_{100}x + A_{010}y + A_{001}z) \\
\mbox{}+ \eta(B_{100}x + B_{010}y + B_{001}z)
+ \zeta(C_{100}x + C_{010}y + C_{001}z) \bigr\}^n
= (Xx + Yy + Zz)^n,
\end{multline*}
where $X,Y,Z$ are linear functions of $\xi,\eta,\zeta$; hence
\[
\left(\xi\frac{d}{dx'} + \eta\frac{d}{dy'} + \zeta\frac{d}{dz'}\right)^n
= \left(X\frac{d}{dx} + Y\frac{d}{dy} + Z\frac{d}{dz}\right)^n.
\]

By developing the two sides and comparing the coefficients of $\xi^\alpha\eta^\beta\zeta^\gamma$, we obtain the value of
\[
\left(\frac{d}{dx'}\right)^\alpha
\left(\frac{d}{dy'}\right)^\beta
\left(\frac{d}{dz'}\right)^\gamma
\]
expressed as a sum of terms of the form
\[
C \left(\frac{d}{dx}\right)^{\alpha'} \left(\frac{d}{dy}\right)^{\beta'} \left(\frac{d}{dz}\right)^{\gamma'}.
\]

Now if in the equation $f(x,y,z)=F(u,v,w)$ we substitute $X,Y,Z$ for $u,v,w$, we obtain an identity; and if we differentiate with respect to $x,y,z$, we have as before
\[
\frac{df}{dx} = \delta \cdot DF \cdot \frac{dX}{dx}, \quad \text{etc.}
\]
But we have also
\[
f(x,y,z)\,\delta = F(X',Y',Z')\,\delta',
\]
and by differentiating this equation with respect to $x',y',z'$, we obtain the value of
\[
\left(\frac{d}{dx'}\right)^\alpha
\left(\frac{d}{dy'}\right)^\beta
\left(\frac{d}{dz'}\right)^\gamma f(x,y,z)
\]
expressed in terms of the derivatives of $F(X',Y',Z')\,\delta'$ with respect to $x',y',z'$. But these last derivatives may be calculated directly; for $X',Y',Z'$ being functions of $x',y',z'$, we have
\[
\frac{d(F\delta')}{dx'} = \frac{d(F\delta')}{dX'}\frac{dX'}{dx'} + \frac{d(F\delta')}{dY'}\frac{dY'}{dx'} + \frac{d(F\delta')}{dZ'}\frac{dZ'}{dx'},
\]
and similarly for the derivatives with respect to $y'$ and $z'$.

By means of these formulas we obtain the general term of the reverted series in the form given by Mr.~Sylvester. In fact, if we write
\[
X = x + A_{200}x^2 + 2A_{110}xy + A_{020}y^2 + \ldots,
\]
\[
Y = y + B_{200}x^2 + 2B_{110}xy + B_{020}y^2 + \ldots,
\]
\[
Z = z + C_{200}x^2 + 2C_{110}xy + C_{020}y^2 + \ldots,
\]
where the coefficients are such that
\[
A_{\alpha\beta\gamma} = \frac{1}{\alpha!\,\beta!\,\gamma!}
\left(\frac{d}{dx}\right)^\alpha
\left(\frac{d}{dy}\right)^\beta
\left(\frac{d}{dz}\right)^\gamma X,
\]
etc., and if we suppose that $X,Y,Z$ vanish for $x=y=z=0$, then the formula of Mr.~Sylvester gives the coefficient of $x'^\alpha y'^\beta z'^\gamma$ in $f(x,y,z)$ as the coefficient of $x^\alpha y^\beta z^\gamma$ in the development of
\[
F(X',Y',Z')\,\frac{d(X',Y',Z')}{d(x',y',z')}.
\]

The formula may be written in the form
\[
\text{Coeff. } x'^\alpha y'^\beta z'^\gamma \text{ in } f(x,y,z)
= \text{Coeff. } x^\alpha y^\beta z^\gamma \text{ in }
\left[ f(x,y,z)\,\frac{d(X,Y,Z)}{d(x,y,z)} \right]_{x=x',\,y=y',\,z=z'},
\]
or what is the same thing
\[
\text{Coeff. } x'^\alpha y'^\beta z'^\gamma \text{ in } f(x,y,z)
= \left[ f(x,y,z)\,\frac{d(X,Y,Z)}{d(x,y,z)} \right]_{x=x',\,y=y',\,z=z'}.
\]

Let now $X,Y,Z$ be
\[
X = x\ldots + A_{\alpha\beta\gamma}\,x^\alpha y^\beta z^\gamma + \ldots,
\]
\[
Y = y\ldots + B_{\alpha\beta\gamma}\,x^\alpha y^\beta z^\gamma + \ldots,
\]
\[
Z = z\ldots + C_{\alpha\beta\gamma}\,x^\alpha y^\beta z^\gamma + \ldots;
\]
in these expressions and everywhere in what follows, the etc.'s relate to terms which one obtains by affixing accents in any number to the indeterminate symbols. By employing the notation of Gauss $\Pi a = 1.2.3\ldots a$, one obtains for the general term of $-X^{-1}$,
\[
\frac{\Pi(-\alpha-\beta-\gamma)}{\Pi\alpha\,\Pi\beta\,\Pi\gamma\ldots}
\frac{A_{\alpha\beta\gamma\ldots}^{\alpha'}\,A_{\alpha'\beta'\gamma'\ldots}^{\alpha''}\ldots}
{x^{\alpha+\alpha'+\ldots}\,y^{\beta+\beta'+\ldots}\,z^{\gamma+\gamma'+\ldots}},
\]
where
\[
\alpha + \alpha' + \ldots = -1, \quad
\beta + \beta' + \ldots = 0, \quad
\gamma + \gamma' + \ldots = 0, \quad \text{etc.}
\]

The general term of $-Y^{-1}$ and $-Z^{-1}$ are of analogous forms; and the general term of the reverted series is obtained by multiplying together the three expressions $-X^{-1}$, $-Y^{-1}$, $-Z^{-1}$ and taking the coefficient of $x^{\alpha-1}y^{\beta-1}z^{\gamma-1}$. We thus fall upon the formula of Mr.~Sylvester, which may be written
\begin{multline*}
\text{Coeff. } x'^\alpha y'^\beta z'^\gamma \text{ in } f(x,y,z) \\
= \frac{1}{\Pi\alpha\,\Pi\beta\,\Pi\gamma}
\left(
\frac{d}{dA_{100}} + \frac{d}{dB_{100}} + \frac{d}{dC_{100}}
\right)^\alpha
\left(
\frac{d}{dA_{010}} + \frac{d}{dB_{010}} + \frac{d}{dC_{010}}
\right)^\beta \\
\times\left(
\frac{d}{dA_{001}} + \frac{d}{dB_{001}} + \frac{d}{dC_{001}}
\right)^\gamma
F(A_{100},B_{010},C_{001})\,\frac{d(X,Y,Z)}{d(x,y,z)},
\end{multline*}
where after the differentiations we put $A_{100}=B_{010}=C_{001}=1$, and all the other coefficients equal to zero.

\bigskip
\noindent\textit{Londres, le 30 Septembre, 1856.}
