\subsubsection*{3.3}
Es evidente que, cualesquiera que sean $S$ y $E$, $c_0(E)=1$, y que, si el rango de $E$ está acotado por $d$, entonces $c_i(E)=0$ para $i>d$.

\begin{statement}{Proposición 3.4}
Las clases de Chern definidas anteriormente satisfacen las propiedades siguientes.
\begin{enumerate}[label=(\roman*)]
\item \emph{Normalización.} Sea $L$ un $\mathcal O_S$-módulo invertible. Denotando por $[L]$ la imagen de la clase de $L$ por el morfismo de borde
\[
H^1(S,\mathbb G_m)\longrightarrow H^2(S,\mu),
\]
se tiene
\[
c(L)=1+[L].
\]
\item \emph{Estabilidad por cambio de base.} Sean $f:S'\to S$ un morfismo de esquemas y $E$ un $\mathcal O_S$-módulo localmente libre. Para todo entero $i$, se tiene
\[
f^*(c_i(E))=c_i(f^*E).
\]
\item \emph{Aditividad.} Sea
\[
0\to E'\to E\to E''\to0
\]
una sucesión exacta de $\mathcal O_S$-módulos localmente libres. Para todo entero $n$, se tiene la relación
\[
c_n(E)=\sum_{i+j=n}c_i(E')c_j(E'').
\]
\end{enumerate}
\end{statement}

Antes de abordar la demostración, extraigamos el corolario siguiente.

\begin{statement}{Corolario 3.5}
Para toda sucesión exacta $0\to E'\to E\to E''\to0$ de $\mathcal O_S$-módulos localmente libres, se tiene
\[
c(E)=c(E')c(E'')
\]
en $\widehat A^*(S)$.
\end{statement}
