\begin{statement}{Proposición 8.6.3}
\begin{enumerate}[label=\alph*)]
\item Para todo par $(M,N)$ de $A_X$-módulos localmente constantes constructibles en un entorno de $Y$, el diagrama
\[
\begin{tikzcd}
\nabla^*(X',M)\otimes\nabla^*(X',N)\arrow[r,"\mu\otimes\mu"]\arrow[d,"{(8.6.1)}"']&
H^*(X',M)\otimes H^*(X',N)\arrow[d,"\cup"]\\
\nabla^*(X',M\otimes N)\arrow[r,"\mu"]&
H^*(X',M\otimes N)
\end{tikzcd}
\]
es conmutativo. En particular, la aplicación sobreyectiva
\[
\mu_A=\mu:\nabla^*(X',A)\to H^*(X',A)
\]
es un morfismo de anillos, y la aplicación
\[
\mu_M=\mu:\nabla^*(X',M)\to H^*(X',M)
\]
es un morfismo de módulos sobre anillos variables.
\item Si se hace actuar $\nabla^*(X',A)$ sobre $H^*(Y,M(-d))(-2d)$ mediante la aplicación
\[
\nabla^p(X',A)\otimes H^{q-2d}(Y,M(-d))
\to H^{p+q-2d}(Y,M(-d)),
\qquad
(y',x)\otimes y\mapsto i^*(x)y,
\]
entonces la aplicación $\lambda$ de 8.5 b) se convierte en un morfismo de $\nabla^*(X',A)$-módulos.
\end{enumerate}
\end{statement}
