\begin{statement}{Proposición 8.5}
Sea $M$ un $A_X$-módulo localmente constante constructible en un entorno de $Y$.
\begin{enumerate}[label=\alph*)]
\item Denotando por
\[
\gamma:H_{Y'}^*(X',M\otimes\mu^{\otimes d})\to
H^*(Y',M\otimes\mu^{\otimes(d-1)})(-2)
\]
la única aplicación definida por la relación
\[
f^*\circ f_* = j_*\circ\gamma+\operatorname{id},
\tag{8.5.1}
\]
se tiene
\[
g_*\circ\gamma=0,
\tag{8.5.2}
\]
y la matriz inversa de (8.3.2) es
\[
\begin{pmatrix}
f_*&-i_*\\
-\gamma&g^*(\cdot)c_{d-1}(\check F)
\end{pmatrix}.
\tag{8.5.3}
\]
\item Para todo entero $p\ge0$, la sucesión
\[
0\to H^{p-2d}(Y,M(-d))
\xrightarrow{\lambda}
H^{p-2}(Y',M(-1))\oplus H^p(X,M)
\xrightarrow{\mu}
H^p(X',M)\to0,
\]
en la cual $\lambda$ y $\mu$ designan las aplicaciones
\[
\lambda(x)=\bigl(g^*(x)c_{d-1}(\check F),-i_*(x)\bigr),
\qquad
\mu(x,y)=j_*(x)+f^*(y),
\]
es exacta. Además, la aplicación $\lambda$ admite como inversa por la izquierda
\[
\lambda'(x,y)=g_*(x).
\]
\end{enumerate}
\end{statement}
