\begin{aligned}
\clK{\Delta}(\RG_C(F))

&=\EP(C)\clK{\Delta}F_{\bar\eta}
 -\sum_{y\in Y}\varepsilon_y^\Delta(F).
\end{aligned}
\]
Teniendo en cuenta que $\varepsilon_x^\Delta(F)=0$ para $x\in\mathcal U$, esta no es sino la fórmula (7.2). Para un complejo de haces $F$ que satisface las condiciones de 7.1, introduciremos la notación
\[
\EP_\Delta(C,F)=\clK{\Delta}(\RG_C(F)).
\]

\begin{statement}{Corolario 7.9}
Si $F$, $F'$ son complejos de haces que satisfacen las condiciones de 7.1 y si son localmente isomorfos (para la topología étale) como complejos de $A$-módulos, se tiene:
\[
\EP_\Delta(C,F)=\EP_\Delta(C,F').
\tag{7.10}
\]
En particular, si $F$ es localmente constante de valor $M^\bullet$, se tiene:
\[
\EP_\Delta(C,F)=\EP(C)\clK{\Delta}(M^\bullet).
\tag{7.11}
\]
\end{statement}
