\subsection*{6.11}
Sean $f:X\to Y$ un $G$-$S$-morfismo propio y
\begin{equation}
\begin{tikzcd}[column sep=large,row sep=large]
C \arrow[r,"c"] \arrow[d] & X\times_SX \arrow[d,"f\times_Sf"]\\
D \arrow[r,"d"'] & Y\times_SY
\end{tikzcd}
\tag{6.11.1}
\end{equation}
un cuadrado conmutativo $G$-equivariante de $G$-$S$-esquemas, con $c$ y $d$ propios. Se supone que $G$ actúa trivialmente sobre $Y$ y $D$. Denotaremos por
\begin{equation}
f^{c/d}\quad(\text{o } f^c):X^c\longrightarrow Y^d
\tag{6.11.2}
\end{equation}
el morfismo inducido por 6.11.1 sobre los esquemas de puntos fijos de $c$ y $d$, con las notaciones de 6.5.1. Sea $L\in\Ob D_{\mathrm{ctf}}(G,{}_AX)$, y sea
\[
u\in\Hom(c_1^*L,c_2^!L)
\]
una correspondencia $G$-equivariante con soporte en $c$, de donde se obtiene, por imagen directa, una correspondencia $G$-equivariante
\[
f_*u\in\Hom_{\Lambda[G]}(d_1^*f_*L,d_2^!f_*L).
\]
Para $g\in G$, denotaremos por $Z_g$ el centralizador de $g$, y sea
\begin{equation}
gu\in\Hom((gc)_1^*L,c_2^!L)
\tag{6.11.3}
\end{equation}
la correspondencia $Z_g$-equivariante con soporte en
\begin{equation}
gc\stackrel{\mathrm{dfn}}{=}(gc_1=c_1g,c_2):C\to X\times_SX,
\tag{6.11.4}
\end{equation}
definida como la compuesta de las flechas
\[
g^*(c_1^*L)\xrightarrow{\alpha_g}c_1^*L\xrightarrow{u}c_2^!L,
\]
donde $\alpha_g$ es la flecha que define la acción de $G$ sobre $c_1^*L$. Es claro que se tiene
\begin{equation}
f_*(gu)=g(f_*u).
\tag{6.11.5}
\end{equation}

\begin{statement}{Proposición 6.12}
Bajo las hipótesis de 6.11, supongamos que se tiene
\[
f_*L\in\Ob D_{\mathrm{ctf}}({}_{\Lambda[G]}Y).
\]
Entonces, para todo $g\in G$, se tiene
\begin{equation}
f_*^{gc}\Tr_{\Lambda}(gu)
=
|Z_g|\,\Tr_{\Lambda[G]}(f_*u)(g^{-1}),
\tag{6.12.1}
\end{equation}
donde
\[
f_*^{gc}:H^0(X^{gc},K_{X^{gc}})\longrightarrow H^0(Y^d,K_{Y^d})
\]
es el morfismo ``de Gysin'' definido por la flecha de adjunción
\[
f_*^{gc}K_{X^{gc}}=f_*^{gc}f^{gc!}K_{Y^d}\longrightarrow K_{Y^d}.
\]
\end{statement}

En efecto, por 6.10.5 aplicado a $f_*u$ y por 6.11.5, se tiene
\[
\Tr_{\Lambda}(f_*(gu))=|Z_g|\,\Tr_{\Lambda[G]}(f_*u)(g^{-1}),
\]
y, por consiguiente, 6.12.1 se deduce de la fórmula de Lefschetz (III 4.5.1).
