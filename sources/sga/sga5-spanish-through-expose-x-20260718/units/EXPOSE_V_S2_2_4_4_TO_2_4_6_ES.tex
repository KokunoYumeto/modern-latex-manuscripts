\begin{statement}{Proposición 2.4.4}
\begin{enumerate}[label=(\roman*)]
\item La categoría $P_{AR}$ es abeliana.
\item El funtor $p_{AR}:P\to P_{AR}$ es exacto.
\item Para que un objeto $X$ de $P$ sea AR-nulo, es necesario y suficiente que $p_{AR}(X)=0$.
\item El par $(p_{AR},P_{AR})$ es un cociente de la categoría abeliana $P$ por la subcategoría abeliana gruesa de los objetos AR-nulos.
\end{enumerate}
\end{statement}

\begin{prooftext}[Demostración]
Si se admite (iii), las demás afirmaciones son consecuencia de (2.3.5). De (2.3.4) se deduce que, para que un objeto $X$ de $P$ sea AR-nulo, es necesario y suficiente que $p_{AR}(\id_X)=0$, de donde se sigue (iii).
\end{prooftext}

\begin{statement}{Corolario 2.4.5}
Sea $u:X\to Y$ de la categoría $P$. Las afirmaciones siguientes son equivalentes:
\begin{enumerate}[label=(\roman*)]
\item $p_{AR}(u)$ es un isomorfismo.
\item El núcleo y el conúcleo de $u$ son AR-nulos.
\end{enumerate}
Un morfismo $u:X\to Y$ de la categoría $P$ de los sistemas proyectivos que verifica las condiciones equivalentes (i) y (ii) se llamará AR-isomorfismo.
\end{statement}

\begin{statement}{Definición 2.4.6}
Dos objetos $X$ e $Y$ de la categoría $P$ se dicen AR-isomorfos si $p_{AR}(X)$ y $p_{AR}(Y)$ son isomorfos en la categoría $P_{AR}$. Si $X$ e $Y$ son dos objetos AR-isomorfos de la categoría $P$, no existe en general un AR-isomorfismo $u$ de $X$ en $Y$.
\end{statement}
