\begin{statement}{Lema 8.4.3}
Se denota por $F$ el $\mathcal O_{Y'}$-módulo localmente libre definido por la sucesión exacta canónica
\[
0\to F\to g^*(N)\to\mathcal O_{Y'}(1)\to0.
\]
Sean $M$ un $A_X$-módulo localmente constante constructible en un entorno de $Y$, y $p$ un entero $\ge0$. Para todo $y\in H^p(Y,M)$, se tiene la relación
\[
f^*(i_*(y))=j_*(g^*(y)c_{d-1}(\check F))
\]
en $H_{Y'}^{p+2d}(X',M\otimes\mu^{\otimes d})$.
\end{statement}

\emph{Demostración.} Pongamos
\[
z=f^*(i_*(y))-j_*(g^*(y)c_{d-1}(\check F)).
\]
Se tiene
\[
j^*(z)=g^*i^*i_*(y)-j^*j_*(g^*(y)c_{d-1}(\check F)),
\]
de donde, gracias a (4.1) y al lema de linealidad de (SGA 4 XVIII),
\[
j^*(z)=g^*(y)\bigl(c_d(g^*\check N)-c_{d-1}(\check F)c_1(\mathcal O_{Y'}(-1))\bigr)=0.
\tag{8.4.4}
\]
Poniendo $U=X-Y$ y $U'=X'-Y'$, se tiene el diagrama conmutativo exacto evidente
\[
\begin{array}{ccccc}
H^{p+2d-1}(U')&\xrightarrow{\partial}&H^{p+2d}_{Y'}(X')&\xrightarrow{j^*}&H^{p+2d}(Y')\\
\uparrow\wr&&\uparrow f^*&&\uparrow g^*\\
H^{p+2d-1}(U)&\xrightarrow{\partial}&H_Y^{p+2d}(X)&\xrightarrow{i^*}&H^{p+2d}(Y),
\end{array}
\tag{8.4.5}
\]
con las simplificaciones de notación evidentes. La consideración de (8.4.5) y (8.4.4) muestra entonces que $z$ es de la forma
\[
z=f^*(t),
\]
con $t\in H_Y^{p+2d}(X,M\otimes\mu^{\otimes d})$. Por (4.2), se tiene $t=f_*(z)$, de modo que basta ver que $f_*(z)=0$. Ahora bien,
\[
f_*(z)=f_*f^*i_*(y)-f_*j_*(g^*(y)c_{d-1}(\check F)),
\]
de donde, utilizando (4.2),
\[
f_*(z)=i_*(y)-i_*g_*(g^*(y)c_{d-1}(\check F)).
\]
Se concluye utilizando la fórmula de proyección para el morfismo $g$ y (8.4.2).
