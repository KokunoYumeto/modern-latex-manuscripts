\section*{Expose VIII -- The Riemann--Roch theorem, continuation}

\subsection*{3.5. The ``Grothendieckian'' Riemann--Roch theorem}

\paragraph{Theorem 3.5.}
Let $f:X\to Y$ be a projective complete-intersection morphism, of virtual relative dimension $d$. Assume that $Y$ is quasi-compact and admits an ample sheaf. Then, for every $x\in K'(X)$, one has
\[
 f_*\bigl(c(x)\,\tau(f)\bigr)=c\bigl(f_*(x)\bigr).
\]
Equivalently, the following square is commutative:
\[
\begin{array}{ccc}
K'(X) & \xto{\,c(\cdot)\tau(f)\,} & \Gr K'(X)_\QQ \\
\scriptstyle f_*\downarrow && \downarrow\scriptstyle f_{\mathrm{gr}} \\
K'(Y) & \xto{\,c\,} & \Gr K'(Y)_\QQ .
\end{array}
\]

\emph{Sketch of proof.}
The proof is by reduction to the case of a projective bundle. Choose a factorization
\[
X\xto{i}\bP(E)\xto{p}Y
\]
with $i$ a regular immersion and $p$ the projection. The formula is first checked for $p$ by the explicit computation of $K'(\bP(E))$ as a free $K'(Y)$-module with the usual powers of the tautological class as basis. For the immersion $i$ the formula is the self-intersection formula, expressed by the inverse Todd character of the conormal bundle. The general assertion follows by the projection formula and by the multiplicativity
\[
\tau(p\circ i)=\tau(i)\,i^*\tau(p).
\]

\paragraph{Corollary 3.6.}
If $f:X\to Y$ is projective and smooth of relative dimension $d$, then
\[
 f_*\bigl(c(x)\,\Todd(T_f)\bigr)=c\bigl(f_*(x)\bigr),
\]
where $T_f$ is the relative tangent bundle. Indeed, for a smooth morphism the virtual conormal bundle is $-T_f$, hence $\tau(f)=\Todd(T_f)$.

\subsection*{3.7. Compatibility formulae}

\paragraph{Proposition 3.7.}
Consider a cartesian square
\[
\begin{array}{ccc}
X'&\longrightarrow&X\\
\downarrow f'&&\downarrow f\\
Y'&\longrightarrow&Y,
\end{array}
\]
with $f$ projective and a complete-intersection morphism, and with $Y'\to Y$ flat. Then, for every $x\in K'(X)$,
\[
 g^*f_*(x)=f'_*g'^*(x).
\]
This is the usual flat base-change formula for proper direct image in $K'$-theory.

\paragraph{Proposition 3.8.}
If $f:X\to Y$ and $g:Y\to Z$ are projective complete-intersection morphisms, then
\[
 (g\circ f)_*(x)=g_*f_*(x)\qquad (x\in K'(X)).
\]
This is the transitivity of the Gysin morphisms.

\section*{4. The Todd class}

\paragraph{Definition 4.1.}
For a regular scheme $X$ admitting an ample sheaf, define
\[
\Todd(X)=\Todd(T_X)\in \Gr K'(X)_\QQ,
\]
where $T_X$ is the tangent bundle of $X$. In the general formulation one treats this class through the preceding functorial Todd character.

\paragraph{Proposition 4.2.}
If $f:X\to Y$ is smooth, then
\[
\Todd(X)=\Todd(T_f)\,f^*\Todd(Y).
\]
Indeed $T_X=T_f\oplus f^*T_Y$.

\paragraph{Proposition 4.3.}
If $i:Y\to X$ is a regular closed immersion, then
\[
\Todd(Y)=\Todd(i)\,i^*\Todd(X).
\]
This is the same multiplicativity, now applied to the virtual tangent bundle of the immersion.

\paragraph{Theorem 4.4.}
Let $X$ and $Y$ be regular schemes admitting ample sheaves, and let $f:X\to Y$ be a projective complete-intersection morphism. Then, for every $x\in K'(X)$,
\[
 f_*\bigl(c(x)\,\Todd(X)\bigr)=c\bigl(f_*(x)\bigr)\,\Todd(Y).
\]

\emph{Proof.}
Apply Theorem 3.5 and use
\[
 \tau(f)=\frac{\Todd(X)}{f^*\Todd(Y)}.
\]
This relation is precisely the expression, in $K'(X)$, of the equality of virtual tangent classes
\[
T_X=T_f+f^*T_Y.
\]
The projection formula then transports the factor $f^*\Todd(Y)$ from the source to $\Todd(Y)$ on the target.

\paragraph{Corollary 4.5.}
For a projective smooth morphism $f:X\to Y$ between regular schemes with ample sheaves,
\[
 f_*\bigl(c(x)\,\Todd(T_f)\bigr)=c\bigl(f_*(x)\bigr).
\]
This is Corollary 3.6 written in terms of the Todd class of the relative tangent bundle.

\section*{5. The Riemann--Roch theorem}

\subsection*{5.1. Statement}

\paragraph{Theorem 5.1.}
Let $f:X\to Y$ be a projective complete-intersection morphism of virtual relative dimension $d$. Assume that $X$ and $Y$ are quasi-compact and admit ample sheaves. Let $\cE$ be a vector bundle on $X$, and let $e$ be its class in $K'(X)$. Then
\[
 f_*\bigl(c(\cE)\,\Todd(T_f)\bigr)=c\bigl(f_!(\cE)\bigr)
 \quad\hbox{in }\Gr K'(Y)_\QQ .
\]
Here $f_!$ denotes the alternating class of the higher direct images when this notation is available.

\emph{Proof.}
Factor $f$ as $X\xto{i}\bP(E)\xto{p}Y$, where $i$ is a regular immersion. Then
\[
f_!(\cE)=p_!i_!(\cE).
\]
For the immersion, Riemann--Roch gives
\[
 c(i_!\cE)=i_*\bigl(c(\cE)\,\Todd(N_i)^{-1}\bigr),
\]
and for the projective bundle projection
\[
 c(p_!\cF)=p_*\bigl(c(\cF)\,\Todd(T_p)\bigr).
\]
Combining these two identities and using the projection formula gives
\[
 c(f_!\cE)=f_*\bigl(c(\cE)\,\Todd(N_i)^{-1}\,i^*\Todd(T_p)\bigr).
\]
The virtual tangent bundle of $f$ is the sum of the virtual tangent bundle of $i$ and the pullback of $T_p$. Since $N_i=-T_i$ in $K$-theory, the displayed product is $\Todd(T_f)$. This proves the formula.

\paragraph{Corollary 5.1.}
If $Y=\Spec(k)$ for a field $k$, then
\[
 \sum_i(-1)^i\dim_k H^i(X,\cE)=\int_X c(\cE)\,\Todd(T_X).
\]
Thus the theorem recovers the usual Euler characteristic formula.

\subsection*{5.2. Projective bundles and the Hirzebruch formula}

Let $E$ be a vector bundle of rank $r+1$ on a scheme $Y$, let $P=\bP(E)$, and let $f:P\to Y$ be the projection. Denote by $\cO_P(1)$ the tautological invertible sheaf and put $\xi=c_1(\cO_P(1))$.

\paragraph{Proposition 5.2.}
There is an exact sequence on $P$
\[
0\longto \Omega^1_{P/Y}(1)\longto f^*E\longto \cO_P(1)\longto0.
\]
Equivalently, after twisting,
\[
0\longto \Omega^1_{P/Y}\longto f^*E(-1)\longto \cO_P\longto0.
\]

\emph{Proof.}
Let $V$ be the vector bundle associated with $E$, and let $Z$ be the complement of the zero section. The morphism $Z\to P$ is the principal $\GG_m$-bundle associated with $\cO_P(1)$. If
\[
B=\bigoplus_{n\in\ZZ}\cO_P(n),
\]
then $Z$ is affine over $P$ with coordinate algebra $B$. The exact sequence of relative differentials for $Z\to P\to Y$ is a graded exact sequence of $B$-modules
\[
0\longto \Omega^1_{P/Y}\otimes B\longto \Omega^1_{Z/Y}\longto \Omega^1_{Z/P}\longto0.
\]
The module $\Omega^1_{Z/Y}$ is obtained from $f^*E$, placed in degree $1$, and $\Omega^1_{Z/P}$ is generated by the differential in the $\GG_m$-direction. Taking homogeneous degree $0$ gives the asserted Euler sequence.

\paragraph{Proposition 5.3.}
For every integer $n\geq0$,
\[
 f_*\bigl(c(\cO_P(1)^n)\,\Todd(T_f)\bigr)=1.
\]
The Todd class of $T_f$ is computed from the Euler sequence above.

\emph{Proof.}
In $K(P)$ one has
\[
[T_f]=[f^*E^\vee\otimes \cO_P(1)]-[\cO_P].
\]
The projective bundle formula writes $K'(P)$ as a free $K'(Y)$-module on the powers of $\cO_P(1)$, and the pushforward of these powers is the alternating class of the symmetric powers of $E^\vee$. Substitution into the characteristic series defining the Todd class gives the claimed identity; it is the standard Hirzebruch computation for projective space, performed relative to $Y$.

\paragraph{Theorem 5.4.}
For $f:\bP(E)\to Y$ and every $x\in K'(\bP(E))$,
\[
 f_*\bigl(c(x)\,\Todd(T_f)\bigr)=c\bigl(f_*(x)\bigr).
\]
By additivity it is enough to check this on the standard basis $1,\cO(1),\ldots,\cO(r)$ of $K'(\bP(E))$ over $K'(Y)$; Proposition 5.3 supplies exactly this check.

\subsection*{5.5. The general Riemann--Roch theorem}

\paragraph{Theorem 5.5.}
Let $f:X\to Y$ be a projective complete-intersection morphism between regular schemes admitting ample sheaves. Then, for every $x\in K'(X)$,
\[
 f_*\bigl(c(x)\,\Todd(X)\bigr)=c\bigl(f_*(x)\bigr)\,\Todd(Y)
 \quad\hbox{in }\Gr K'(Y)_\QQ .
\]

\emph{Proof.}
Choose a factorization $X\xto{i}P\xto{p}Y$ with $P=\bP(E)$ and $i$ a regular immersion. Apply the projective-bundle case to $p$ and the self-intersection Riemann--Roch formula to $i$:
\[
 c(i_*x)=i_*\bigl(c(x)\,\Todd(N_i)^{-1}\bigr).
\]
Then
\[
 c(f_*x)=p_*\bigl(i_*\bigl(c(x)\Todd(N_i)^{-1}\bigr)\Todd(T_p)\bigr)
       =f_*\bigl(c(x)\Todd(N_i)^{-1}i^*\Todd(T_p)\bigr).
\]
The last Todd factor equals $\Todd(T_f)$, and the passage from $\Todd(T_f)$ to $\Todd(X)/f^*\Todd(Y)$ gives the displayed formula.

\paragraph{Corollary 5.6.}
If $f:X\to Y$ is proper and smooth between regular schemes with ample sheaves, then, for every vector bundle $\cE$ on $X$,
\[
 f_*\bigl(c(\cE)\,\Todd(T_f)\bigr)=c\bigl(f_!\cE\bigr).
\]

\subsection*{5.7. Examples and applications}

For an invertible sheaf $\cL$ one has
\[
 c(\cL)=1+c_1(\cL),\qquad
 \Todd(\cL)=\frac{c_1(\cL)}{1-e^{-c_1(\cL)}}.
\]
If $D$ is an effective Cartier divisor and $\cL=\cO(D)$, then
\[
 c(\cO(D))=1+[D].
\]
For a regular proper surface $X$ over a field and an invertible sheaf $\cL$, Riemann--Roch gives
\[
\chi(\cL)=\chi(\cO_X)+\frac12\,\cL\cdot(\cL-K_X),
\]
which is the classical surface formula. For a regular proper curve of genus $g$, and an invertible sheaf of degree $d$, the formula becomes
\[
 \chi(\cL)=d+1-g.
\]

\paragraph{Remark 5.8.}
The form above is broader than the classical theorem: it applies through $K'$-theory to singular schemes and to complete-intersection morphisms which need not be smooth.

\subsection*{5.9. Compatibilities and variants}

\paragraph{Proposition 5.9.}
The Riemann--Roch theorem is compatible with flat base change. For a cartesian square
\[
\begin{array}{ccc}
X'&\longrightarrow&X\\
\downarrow f'&&\downarrow f\\
Y'&\xto{g}&Y
\end{array}
\]
with $g$ flat, and for every $x\in K'(X)$,
\[
 g^* f_*\bigl(c(x)\Todd(f)\bigr)=
 f'_*\bigl(c(g'^*x)\Todd(f')\bigr).
\]

\paragraph{Proposition 5.10.}
The theorem is compatible with open immersions: if $j:U\to X$ is open, then
\[
 j^*\bigl(c(x)\Todd(X)\bigr)=c(j^*x)\Todd(U).
\]

\paragraph{Theorem 5.11.}
For a proper complete-intersection morphism $f:X\to Y$ between regular schemes admitting ample sheaves, the square
\[
\begin{array}{ccc}
K'(X) & \xto{\,c(\cdot)\Todd(X)\,} & \Gr K'(X)_\QQ \\
\scriptstyle f_*\downarrow && \downarrow\scriptstyle f_{\mathrm{gr}} \\
K'(Y) & \xto{\,c(\cdot)\Todd(Y)\,} & \Gr K'(Y)_\QQ
\end{array}
\]
is commutative. This is Theorem 5.5 rewritten using
\[
\Todd(f)=\Todd(X)/f^*\Todd(Y).
\]

\subsection*{5.12. Conclusion of Expose VIII}

The Riemann--Roch theorem in the form just stated is the culmination of the construction developed in the preceding exposes. It connects $K'$-theory with the graded rings attached to the Chern character, and it inserts precisely the Todd correction accounting for the geometry of the morphism. The proof rests on the Gysin morphisms for proper complete-intersection maps, on the self-intersection formula for regular immersions, on the construction of the Chern and Todd characters in the lambda-ring formalism, and on the explicit verification for projective bundles.

\section*{Expose IX -- Some computations of $K$-groups}

\begin{center}
{\large\textbf{Some computations of $K$-groups}}\\[0.25em]
{\normalsize P. Berthelot}
\end{center}

This expose corresponds to no oral lecture and will not be used later in the seminar. It contains a number of computations of the groups $K_\bullet$ for various fibrations: vector bundles, projective bundles, torsors under a linear group, and related objects. Throughout, the base scheme is noetherian, and $K_\bullet(X)$ denotes the Grothendieck group of coherent sheaves on a noetherian scheme $X$. Recall that, for $X$ noetherian, the natural homomorphism from the $K$-group of coherent $\cO_X$-modules to the $K$-group of pseudo-coherent complexes on $X$ is an isomorphism.

The results are analogues of the corresponding computations for the Chow ring, as in Grothendieck's lectures in the Chevalley seminar. One may also ask to what extent statements of the same type hold for the graded group associated with $K_\bullet$ for the filtration by dimension of support. Even for the homotopy exact sequence below, which is one of the most frequently used tools in the computations, this is not known in full generality.

\subsection*{1. Vector bundles}

\paragraph{Proposition 1.1 (homotopy exact sequence).}
Let $X$ be a noetherian scheme, let $X'$ be a closed subscheme of $X$, let $U=X-X'$ be the open complement, and let $i:X'\to X$ and $j:U\to X$ be the immersions. Then
\[
K_\bullet(X')\xto{\,i_*\,}K_\bullet(X)\xto{\,j^*\,}K_\bullet(U)\longto0
\]
is exact.

\emph{Proof.}
For a coherent $\cO_{X'}$-module $F'$, the sheaf $i_*F'$ is supported on $X'$, hence its restriction to $U$ is zero. Thus $j^*i_*=0$. We obtain a homomorphism
\[
\rho:K_\bullet(X)/\Imop K_\bullet(X')\longto K_\bullet(U).
\]
We construct its inverse. Let $G$ be a coherent $\cO_U$-module. Since $X$ is noetherian, there exists a coherent $\cO_X$-module $F$ restricting to $G$. If $F'$ is another such extension, then
\[
\cl(F)-\cl(F')\in\Imop K_\bullet(X').
\]
Indeed, embed $G$ diagonally in $j^*(F\oplus F')$ and extend it to a coherent subsheaf $F''\subset F\oplus F'$. Comparing $F''$ with $F$ and $F'$ produces kernels and cokernels supported on $X'$. Since a coherent sheaf supported on $X'$ is killed by a sufficiently high power of the ideal of $X'$, it may be devissed into coherent $\cO_{X'}$-modules. Hence its class lies in the image of $K_\bullet(X')$. The class of any extension of $G$ is therefore well defined modulo this image, and this construction is additive in short exact sequences. It gives the inverse of $\rho$.

\paragraph{Proposition 1.2.}
Let $f:X\to S$ be a flat morphism of noetherian schemes. Let $(\xi_i)_{i\in I}$ be a family of elements of $K^*(X)$ and consider
\[
\varphi:K_\bullet(S)^{(I)}\longto K_\bullet(X),\qquad
(x_i)\longmapsto \sum_i \xi_i\,f^*(x_i).
\]
Assume that for every point $s\in S$ the images of the $\xi_i$ in $K_\bullet(X_s)$ generate that group. Then $\varphi$ is surjective.

\emph{Proof.}
Let $S'$ be a closed subscheme of $S$ and $U$ its complement. Write $X_{S'}$ and $X_U$ for the restrictions of $X$. The homomorphisms $\varphi_{S'}$ and $\varphi_U$ fit into a commutative diagram with exact rows
\[
\begin{array}{ccccccccc}
K_\bullet(S')^{(I)}&\to&K_\bullet(S)^{(I)}&\to&K_\bullet(U)^{(I)}&\to&0\\
\downarrow&&\downarrow&&\downarrow&&\\
K_\bullet(X_{S'})&\to&K_\bullet(X)&\to&K_\bullet(X_U)&\to&0.
\end{array}
\]
The commutativity of the left square is the compatibility of proper direct image with flat inverse image; the right square follows from functoriality of inverse image. If $\varphi_{S'}$ and $\varphi_U$ are surjective, so is $\varphi$.

Noetherian induction on closed subschemes of $S$ reduces the proof to the case where $S=\Spec(A)$ is integral and affine. Let $K$ be the fraction field and let $s$ be the generic point. For $y\in K_\bullet(X)$, the class $y_s$ in $K_\bullet(X_s)$ is, by hypothesis, a finite sum $\sum_i \xi_i n_i$. Since $K=\varinjlim A_f$ and $X_s=\varprojlim X\times_S\Spec(A_f)$, the same equality holds over some non-empty open $U=\Spec(A_f)$. By induction on the closed complement, the remaining difference lies in the image of $\varphi$. Thus $\varphi$ is surjective.

\paragraph{Corollary 1.3.}
If $f:X\to S$ is flat and, for every $s\in S$, the map $K^*(X)\to K_\bullet(X_s)$ is surjective, then
\[
K_\bullet(S)\otimes_{K^*(S)}K^*(X)\longto K_\bullet(X)
\]
is surjective.

\paragraph{Corollary 1.4.}
If $f:X\to S$ is flat and the canonical map $\ZZ\to K_\bullet(X_s)$ is an isomorphism for every $s\in S$, then
\[
f^*:K_\bullet(S)\longto K_\bullet(X)
\]
is surjective. If, moreover, $f$ has a section $g$ of finite Tor-dimension, then $f^*$ is an isomorphism, with inverse $g^*$.

\paragraph{Remark 1.5.}
For such a section $g$ to have finite Tor-dimension it is necessary and sufficient that the fibres of $X$ be regular at the points of $g(S)$; equivalently, $f$ is regular at those points, and smooth there if $f$ is flat of finite presentation.

\paragraph{Corollary 1.6.}
Let $E$ be a locally free $\cO_S$-module of rank $n$, let $X=\VV(E)$ be the corresponding vector bundle, and let $f:X\to S$ be its structural morphism. Then
\[
f^*:K_\bullet(S)\longto K_\bullet(X)
\]
is an isomorphism.

\paragraph{Corollary 1.7.}
If $g:S\to X$ is the zero section of the vector bundle above, then
\[
g^*:K_\bullet(X)\longto K_\bullet(S)
\]
is the inverse of $f^*$. The zero section is a regular immersion, hence has finite Tor-dimension.

\subsection*{2. Projective bundles and torsors}

Let $S$ be noetherian, let $E$ be a locally free $\cO_S$-module of rank $r+1$, and let $X=\VV(E)$. Suppose that a torus $T$ with fibres isomorphic to $\GG_m^r$ acts freely on $X$, that the quotient $P=X/T$ exists, and that $X$ is a $T$-torsor over $P$. The aim is to compute $K_\bullet(P)$ using Proposition 1.2.

When $T=\GG_m$ and $X$ is the complement of the zero section in a line bundle $L$, the homotopy exact sequence gives
\[
K_\bullet(S)\xto{\,1-L\,}K_\bullet(S)\longto K_\bullet(X)\longto0.
\]
Thus $K_\bullet(X)$ is the quotient of $K_\bullet(S)$ by the subgroup generated by $(L-1)K_\bullet(S)$.

More generally, let $M$ be the character group of the torus $T$, and let $\eta(m)$ denote the invertible sheaf on $S$ associated with a character $m\in M$. Then
\[
K_\bullet(P)\simeq K_\bullet(S)\Big/\sum_{m\in M}(\eta(m)-1)K_\bullet(S).
\]
The proof is by induction on the rank of the torus. Choose a subtorus $T'\subset T$ of rank $r-1$ with quotient $T/T'\simeq\GG_m$. Put $P'=X/T'$. Applying the induction hypothesis to $X\to P'$ and then the rank-one case to $P'\to P$ gives the formula.

\paragraph{Remark 2.3.}
If $T=\GG_m^r$, the preceding result may be expressed as
\[
K_\bullet(P)\simeq K_\bullet(S)\otimes_{\ZZ[M]}\ZZ,
\]
where $\ZZ[M]$ acts on $K_\bullet(S)$ through the classes $\eta(m)$ and on $\ZZ$ through the augmentation.

\subsection*{3. Projective line bundles}

\paragraph{Proposition 3.1.}
Let $S$ be noetherian, let $E$ be locally free of rank $2$, put $P=\bP(E)$, and let $f:P\to S$ be the structural morphism. Let $L$ be the canonical invertible sheaf on $P$. Then:
\begin{enumerate}[label=\roman*)]
\item The homomorphism
\[
K_\bullet(S)\oplus K_\bullet(S)\longto K_\bullet(P),\qquad
(x,y)\longmapsto f^*x+L\,f^*y,
\]
is an isomorphism.
\item If $E'$ is a locally free rank-one submodule of $E$, and $i:S\to P$ is the corresponding section, then
\[
i^*L=(E')^{-1}\otimes(E/E').
\]
\end{enumerate}

For a locally free $E$ of rank $r+1$, set $P=\bP(E)$ and let $\xi=\cl(\cO_P(1))$. One obtains a homomorphism
\[
\ZZ[T]/(T^{r+1})\longto K^*(P),\qquad T\longmapsto \xi^{-1}.
\]
When $E$ is trivial, this gives the familiar formula
\[
K^*(\bP^r_S)=K^*(S)[\xi]/(\xi-1)^{r+1}.
\]

\paragraph{Proposition 3.3.}
Let $f:X\to S$ be flat, and let $E$ be locally free of rank $r+1$ on $X$. Suppose that for every $s\in S$
\[
K^*(\bP(E_s))\simeq K^*(X_s)[\xi_s]/(\xi_s^{r+1}).
\]
Then
\[
K^*(\bP(E))\simeq K^*(X)[\xi]/(\xi^{r+1}).
\]
The proof applies Proposition 1.2 to the elements $1,\xi,\ldots,\xi^r$ and then uses noetherian induction to reduce injectivity to the case of a field.

\paragraph{Corollary 3.4.}
Under the hypotheses of Proposition 3.3, if $f$ has a section, then
\[
K^*(S)[T]/(T^{r+1})\longto K^*(P),\qquad T\longmapsto\xi^{-1},
\]
is an isomorphism.

\paragraph{Corollary 3.5.}
For a locally free $\cO_S$-module $E$ of rank $r+1$,
\[
K^*(\bP(E))=
K^*(S)[\xi]\Big/\left(\sum_{i=0}^{r+1}(-1)^i\lambda^i(E)\xi^{r+1-i}\right).
\]

If $0\to E'\to E\to E''\to0$ is exact with $E'$ of rank $1$, and if $s:S\to\bP(E)$ is the corresponding section, then
\[
s^*(\xi)=\lambda^1((E')^*)=(E')^{-1}.
\]

\paragraph{Corollary 3.7.}
For $P=\bP(E)$ with $E$ of rank $r+1$,
\[
K_\bullet(S)^{r+1}\longto K_\bullet(P),\qquad
(x_0,\ldots,x_r)\longmapsto \sum_{i=0}^r \xi^i f^*(x_i),
\]
is an isomorphism. The proof is by induction on $r$, using the section associated with a rank-one subbundle and the open complement, which is a vector bundle over the smaller projective bundle.

\subsection*{4. Flag schemes}

Let $E$ be locally free of rank $n$ on a noetherian scheme $S$. The flag scheme $\Drap(E)$ represents the functor sending $S'\to S$ to complete flags
\[
0=E_0\subset E_1\subset\cdots\subset E_n=E_{S'},\qquad \rank E_i=i.
\]

\paragraph{Proposition 4.2.}
Let $D=\Drap(E)$ and $f:D\to S$. Then:
\begin{enumerate}[label=\roman*)]
\item $f$ is proper and flat;
\item $f^*:K^*(S)\to K^*(D)$ is injective;
\item $K^*(D)$ is a free $K^*(S)$-module of rank $n!$, with a basis given by monomials in the Chern classes of the successive universal quotient line bundles.
\end{enumerate}
The proof is the usual induction on $n$: the passage from a flag of rank $n-1$ to a flag of rank $n$ is governed by a projective bundle, and the complement of the closed stratum is a vector bundle.

\paragraph{Proposition 4.3.}
Let $L_i=E_i/E_{i-1}$ on $D$, and put $\xi_i=c^1(L_i)$. Then
\[
K^*(D)=K^*(S)[\xi_1,\ldots,\xi_n]/I,
\]
where $I$ is generated by
\[
\sigma_k(\xi_1,\ldots,\xi_n)=c^k(E),\qquad 1\leq k\leq n,
\]
with $\sigma_k$ the elementary symmetric function. These relations come from the exact sequences
\[
0\to E_{i-1}\to E_i\to L_i\to0
\]
and the resulting factorization of the total Chern class of $E$.

\subsection*{5. Torsors under a linear group}

Let $G$ be a smooth group scheme over $S$ with connected fibres, and let $P$ be a $G$-torsor over $S$. The objective is again to compute $K^*(P)$. If $G=\Gl(n)$, then $P$ is the scheme of isomorphisms $\Isom(\cO_S^n,E)$ for a locally free module $E$ of rank $n$. Let $B\subset\Gl(n)$ be the Borel subgroup of upper triangular matrices. Then $P/B$ is the flag scheme of $E$, and $P\to P/B$ is a $B$-torsor. Since $B$ is obtained by successive extensions by copies of $\GG_m$ and affine spaces, the preceding computations give
\[
K^*(P)=K^*(\Drap(E)).
\]
More explicitly,
\[
K^*(P)=K^*(S)[\xi_1^{\pm1},\ldots,\xi_n^{\pm1}]/\bigl(\sigma_k(\xi_1,\ldots,\xi_n)=c^k(E)\bigr).
\]

\paragraph{Proposition 5.2.}
Let $G$ be a split reductive group scheme over $S$, let $T$ be a split maximal torus, let $B$ be a Borel subgroup containing $T$, and let $W$ be the Weyl group. For a $G$-torsor $P$, set $D=P/B$. Then:
\begin{enumerate}[label=\roman*)]
\item $K^*(S)\to K^*(D)$ is injective;
\item $K^*(D)$ is a free $K^*(S)$-module of rank $|W|$, with basis represented by the Bruhat cells.
\end{enumerate}
The assertion follows fibrewise from the cellular decomposition and then globally from Proposition 1.2.

\paragraph{Corollary 5.3.}
Under the same hypotheses, if $R$ is the weight lattice of $T$, $R_+$ a system of positive roots defined by $B$, and $\rho=\frac12\sum_{\alpha\in R_+}\alpha$, then
\[
K^*(P)=K^*(D)^W=(K^*(S)\otimes\ZZ[R])^W.
\]

\subsection*{6. Appendix: complements on lambda-rings}

A lambda-ring is a commutative ring $K$ with unit, equipped with operations $\lambda^n:K\to K$ satisfying the standard identities. One defines
\[
\gamma^n(x)=\lambda^n(x+n-1)
\]
and the gamma-filtration by declaring $F^n_\gamma K$ to be generated by products
\[
\gamma^{n_1}(x_1)\cdots\gamma^{n_k}(x_k),\qquad \sum n_i\geq n.
\]

\paragraph{Proposition 6.2.}
If $K$ is a lambda-ring and $x\in F^1_\gamma K$, then
\[
\gamma_t(x)=\sum_{n\geq0}\gamma^n(x)t^n
=\exp\left(\sum_{m\geq1}(-1)^{m-1}\psi^m(x)\frac{t^m}{m}\right),
\]
where $\psi^m$ are the Adams operations. The identity is checked on elements of the form $\xi-1$, with $\xi$ invertible, and then follows for special lambda-rings by the splitting principle.

\paragraph{Corollary 6.3.}
For $x,y\in F^1_\gamma K$,
\[
\gamma^n(x+y)=\sum_{i=0}^n\gamma^i(x)\gamma^{n-i}(y),
\]
and, modulo $F^{m+1}_\gamma K$,
\[
\psi^m(x)=\sum_{n\geq m}(-1)^{n-m}m{n\choose m}\gamma^n(x).
\]

One also considers the codimension filtration
\[
F^n_cK=\{x\in K\mid c^i(x)=0\hbox{ for }i<n\},
\]
where $c^i(x)=\gamma^i(x-\eps(x))$ and $\eps$ is the augmentation.

{\normalsize P. Berthelot}

\subsection*{1. Filtrations on $K$-groups}

Let $X$ be a noetherian scheme. The group $K_\bullet(X)$ is the Grothendieck group of coherent $\cO_X$-modules, and $K^*(X)$ is the Grothendieck group of locally free $\cO_X$-modules of finite type. There is a canonical map
\[
K^*(X)\longto K_\bullet(X),
\]
which is an isomorphism when $X$ is regular.

On $K_\bullet(X)$ one has the filtration by dimension of support:
\[
\Filt_nK_\bullet(X)=
\left\langle \cl(F)\mid F\hbox{ coherent and }\dim \Supp(F)\leq n\right\rangle.
\]
On $K^*(X)$ one has the filtration by codimension:
\[
\Filt^nK^*(X)=\{x\in K^*(X)\mid c^i(x)=0\hbox{ for }i<n\}.
\]

The gamma-filtration satisfies
\[
F^n_\gamma K^*(X)\subset \Filt^nK^*(X),
\]
though equality is not known in general.

\paragraph{Proposition 1.3.}
Let $X$ be noetherian.
\begin{enumerate}[label=\roman*)]
\item The codimension filtration on $K^*(X)$ is compatible with products.
\item If $X$ is regular of dimension at most $d$, then $\Filt^{d+1}K^*(X)=0$.
\item If $f:X\to Y$ is proper, then
\[
f_*(\Filt_nK_\bullet(X))\subset \Filt_nK_\bullet(Y).
\]
\end{enumerate}
The first assertion follows from the formula for Chern classes of tensor products, the second from dimension, and the third from the behaviour of supports under proper direct image.

\subsection*{1.4. Dimension functions}

Let $S$ be noetherian. We want, for each proper $S$-scheme $Y$ and each point $y\in Y$, an integer $D_Y(y)$ satisfying:
\begin{enumerate}[label=\roman*)]
\item $D_Y(y)\geq0$;
\item $D_Y(y)=0$ precisely at the generic points of irreducible components;
\item for an $S$-morphism $Y_1\to Y_2$, one has
\[
D_{Y_1}(y_1)\geq D_{Y_2}(f(y_1));
\]
\item $D_{Y_{\mathrm{red}}}(y)=D_Y(y)$.
\end{enumerate}

If $S$ is locally noetherian and universally catenary, choose an integer $n$ and put
\[
D_S(x)=n-\codim(\overline{\{x\}},S)=n-\dim \cO_{S,x}.
\]
For $s_Y:Y\to S$ proper and $y\in Y$ set
\[
D_Y(y)=D_S(s_Y(y))+\trdeg_{k(s_Y(y))}k(y).
\]
For an $S$-morphism $Y_1\to Y_2$ this gives
\[
D_{Y_1}(y)=D_{Y_2}(f(y))+\trdeg_{k(f(y))}k(y).
\]
The only substantive point is the characterization of generic points. By replacing $Y$ by its reduction and then by an irreducible closed subscheme, and by factoring its structural morphism through its scheme-theoretic image, the assertion follows from the catenarity dimension formula.

\subsection*{2. Polynomial maps}

Let $G$ be an abelian group and $A$ a commutative ring. A map $f:G\to A$ is called polynomial if its restriction to every finitely generated subgroup of $G$ is polynomial in the usual sense. Thus on each finitely generated subgroup one may write it as a polynomial in finitely many additive characters.

\paragraph{Lemma 2.1.1.}
For a map $f:G\to A$, the following are equivalent.
\begin{enumerate}[label=\roman*)]
\item $f$ is polynomial.
\item Locally on every finitely generated subgroup it is a finite linear combination of functions
\[
x\longmapsto {\chi(x)\choose m},
\]
where $\chi:G\to A$ is a homomorphism and $m\in\NN$.
\end{enumerate}
The reason is that the polynomial functions on a finitely generated free abelian group are generated by the binomial polynomials in additive coordinates.

Let $X$ be noetherian and let $L$ be invertible. The map
\[
\varphi_L:\Pic(X)\longto K_\bullet(X),\qquad
M\longmapsto \cl(L\otimes M)-\cl(L),
\]
is polynomial. Indeed, in $K^*(X)$ one has
\[
\cl(L\otimes M)-\cl(L)=L(M-1),
\]
and $M-1$ lies in the first step of the gamma-filtration.

\paragraph{Proposition 2.3.}
Let $E$ be locally free of rank $r+1$ on a noetherian $S$, and put $P=\bP(E)$. The map
\[
\Pic(P)\longto K_\bullet(P),\qquad L\longmapsto \cl(L),
\]
is polynomial on each coset coming from $\Pic(S)$. Since
\[
K_\bullet(P)=\bigoplus_{i=0}^r K_\bullet(S)\xi^i,
\]
every line bundle on $P$ is of the form $f^*M\otimes\cO_P(n)$, and its class is polynomial in the integer $n$ and in the class of $M$.

\subsection*{2.4. Snapper polynomials}

Let $X$ be proper and smooth over a field $k$, let $L$ be an ample invertible sheaf, and let $F$ be coherent. Snapper's theorem says that
\[
n\longmapsto \chi(X,L^{\otimes n}\otimes F)
\]
is a polynomial in $n$. More generally, for invertible sheaves $L_1,\ldots,L_d$, the function
\[
(n_1,\ldots,n_d)\longmapsto
\chi(X,L_1^{\otimes n_1}\otimes\cdots\otimes L_d^{\otimes n_d}\otimes F)
\]
is a polynomial, the Snapper polynomial. Intersection numbers are extracted as its top multi-degree coefficients.

\subsection*{5. Intersection numbers}

Let $X$ be proper over a field $k$, let $L_1,\ldots,L_d$ be invertible sheaves, and let $F$ be a coherent sheaf with $\dim\Supp(F)\leq d$. The intersection number
\[
(L_1\cdots L_d;F)
\]
is the coefficient of $n_1\cdots n_d$ in the Snapper polynomial
\[
\chi(L_1^{\otimes n_1}\otimes\cdots\otimes L_d^{\otimes n_d}\otimes F).
\]
Equivalently it may be described by iterated first Chern classes acting on $\cl(F)$ and then applying the Euler characteristic.

This integer is symmetric and $d$-linear in the variables $L_i\in\Pic(X)$, is zero if $\dim\Supp(F)<d$, and is additive in the class of $F$ in $K_\bullet(X)$. If $Y$ is a closed subscheme of $X$, $i:Y\to X$ is the immersion, and $F=i_*G$, then
\[
(i^*L_1,\ldots,i^*L_d;G)=(L_1,\ldots,L_d;F).
\]
More generally, the projection formula gives
\[
\chi(f^*x\cdot y)=\chi(x\cdot f_*y)
\]
for every morphism $f:Y\to X$ of proper $k$-schemes.

If $f:Y\to X$ is a morphism of irreducible proper $k$-schemes of dimension at most $d$ and of degree $n$, then
\[
(f^*L_1,\ldots,f^*L_d;\cO_Y)=n(L_1,\ldots,L_d;\cO_X).
\]
The definition of the degree is the usual one: it is zero unless the generic point of $Y$ maps to the generic point of $X$ and the dimensions agree; in that case it is the degree of the corresponding extension of function fields, expressed in terms of the lengths of the local rings at the generic points.

\subsection*{4.5. Numerical equivalence}

Two elements of $\Gr^i(X)$ are numerically equivalent if they define the same homomorphism on $\Gr_i(X)$ by the intersection pairing. The analogous definition is used for $\Gr_i(X)$. Two invertible sheaves are numerically equivalent when their first Chern classes in $\Gr^1(X)$ are numerically equivalent.

An invertible sheaf $L$ is numerically equivalent to zero if and only if
\[
\chi(L^{\otimes m}\otimes\cO_Z)=\chi(\cO_Z)
\]
for all integers $m$ and all integral closed curves $Z\subset X$. This is an immediate consequence of the expression of the intersection product in terms of Euler characteristics. Write $\Pic^{\mathrm{num}}(X)$ for the subgroup of $\Pic(X)$ of classes numerically equivalent to zero. One has
\[
\Pic^c(X)\subset \Pic^{\mathrm{num}}(X),
\]
and a later expose proves the stronger equality $\Pic^\tau(X)=\Pic^{\mathrm{num}}(X)$.

\section*{5. The isomorphism $\Pic(X)\simeq\Gr^1(X)$}

Let $X$ be any ringed topos. The ring $K^\bullet(X)$ and its filtration are defined as for schemes. The group $\Pic(X)$ is the group of isomorphism classes of invertible $\cO_X$-modules. The map
\[
i:\Pic(X)\longto K^\bullet(X)
\]
sends an invertible module $L$ to its class. It is a homomorphism into the group of units of $K^\bullet(X)$.

The determinant gives a left inverse. If $E$ is locally free of rank $n$, set
\[
\detm(E)=\bigwedge^nE.
\]
For a short exact sequence of locally free modules
\[
0\to E'\to E\to E''\to0
\]
there is a canonical isomorphism
\[
\detm(E)\simeq \detm(E')\otimes\detm(E'').
\]
Thus one obtains a group homomorphism
\[
\detm:K^\bullet(X)\longto\Pic(X)
\]
which is a left inverse to $i$. Hence $\Pic(X)$ is identified with a subgroup of $K^\bullet(X)^*$.

Define
\[
\theta:\Pic(X)\longto\Filt^1(X),\qquad \theta(L)=L-1.
\]
The composition with $\Filt^1(X)\to\Gr^1(X)$ is the first Chern class
\[
c_1:\Pic(X)\longto\Gr^1(X).
\]

\paragraph{Theorem 5.3.2.}
The map $c_1$ is injective. Moreover, $\detm:\Filt^1(X)\to\Pic(X)$ induces a left inverse
\[
\alpha:\Gr^1(X)\longto\Pic(X).
\]
If every section of the constant sheaf $\ZZ_X$ is bounded, for example if $X$ is quasi-compact or if the final object is a finite sum of connected objects, then $c_1$ is an isomorphism and $\alpha$ is its inverse.

\emph{Proof.}
The identity
\[
c_1(L\otimes L')=c_1(L)+c_1(L')+c_1(L)c_1(L')
\]
shows that $c_1$ is a homomorphism after passage to $\Gr^1$. The determinant kills $\Filt^2(X)$, so it passes to $\Gr^1$ and defines $\alpha$. Then
\[
\alpha c_1(L)=L.
\]
If the rank function is bounded, the group $\Filt^1(X)$ is generated by the elements $1-[E]$ for locally free rank-one modules $E$, and on these generators one checks directly that $c_1\alpha$ is the identity.

\paragraph{Lemma 5.3.4.}
For every ringed topos $X$ and every $x\in\Filt^2(X)$,
\[
\detm(x)=1.
\]
The determinant formula
\[
\detm(xy)=\detm(x)^{\eps(y)}\detm(y)^{\eps(x)}
\]
with the augmentation $\eps:K^\bullet(X)\to\ZZ$ implies this after applying it to $x\in\Filt^2(X)$, for which $\eps(x)=0$.

\subsection*{5.3.5. Complement}

The boundedness assumption on the rank function is not needed to show that $\alpha$ is well defined, namely that $\detm(\Filt^2(X))=1$. It enters only in the proof that $c_1$ is surjective. If $E$ has constant rank $n_0$, then
\[
\detm(E)=\bigwedge^{n_0}E
      =\gamma^{n_0}((E-n_0)+1)
      =\sum_{0\leq i\leq n_0}\gamma^i(E-n_0),
\]
and modulo $\Filt^2(X)$ this is congruent to
\[
1+n_0(E-n_0),
\]
which gives the required congruence.

\section*{6. Devissage of $\Gl(n)$}

Let $S$ be a scheme. Denote by $\Gl(n)$ the group scheme of automorphisms of $\cO_S^n$, by $\Sl(n)$ the subgroup of determinant $1$, and by $\GG_m$ the multiplicative group scheme. There is an exact sequence
\[
1\longto\Sl(n)\longto\Gl(n)\xto{\detm}\GG_m\longto1.
\]

\paragraph{Proposition 6.2.1.}
For $n\geq2$, the group $\Sl(n)$ is equal to its commutator subgroup; this commutator group is simply connected.

\paragraph{Proposition 6.2.2.}
For every positive section $n$ of $\ZZ$, the etale sheaf of commutators $\Der(\Gl(n))$ is represented by
\[
\Sl(n)=\Ker(\detm:\Gl(n)\to\GG_m).
\]

\paragraph{Corollary 6.2.3.}
Every representation
\[
\sigma:G=\Gl(n_1)\times\cdots\times\Gl(n_r)\longto\GG_m
\]
is of the form
\[
\sigma(s_1,\ldots,s_r)=\detm(s_1)^{m_1}\cdots\detm(s_r)^{m_r},
\]
with uniquely determined sections $m_1,\ldots,m_r$ of $\ZZ$.

\paragraph{Corollary 6.2.4.}
With the same notation, for locally free modules $E_1,\ldots,E_r$ one has
\[
\detm\bigl(\pi_!(E_1,\ldots,E_r)\bigr)=
\detm(E_1)^{m_1}\otimes\cdots\otimes\detm(E_r)^{m_r}
\]
for uniquely determined exponents $m_i$.

The exponents are computed by substituting $s_i=\lambda\,\operatorname{id}$ in the $i$-th factor and the identity in the others. In particular
\[
\detm(E_1\otimes E_2)=\detm(E_1)^{n_2}\otimes\detm(E_2)^{n_1},
\]
\[
\detm(E_1\oplus E_2)=\detm(E_1)\otimes\detm(E_2),
\]
and
\[
\detm(\bigwedge^m E)=\detm(E)^{{n-1\choose m-1}}.
\]

The proof of Corollary 6.2.3 reduces to the case of a single factor. By Proposition 6.2.2 a character of $\Gl(n)$ factors through $\Gl(n)/\Sl(n)$; the determinant identifies this quotient with $\GG_m$, and the characters of $\GG_m$ are the powers $\lambda\mapsto\lambda^m$.

For Proposition 6.2.2 one reduces locally on the base to the usual elementary statement for matrices: the commutator subgroup of $\Gl(n)$ is $\Sl(n)$, generated by elementary matrices. Over an algebraically closed field this follows from Jordan normal form and the elementary row and column operations; descent then gives the sheaf statement.

\section*{7. Appendix: specialization in intersection theory}

\begin{flushright}
\emph{A. Grothendieck}
\end{flushright}

\paragraph{Lemma 7.1.}
Let $i:Y\to X$ be a regular closed immersion of bounded codimension, hence a perfect morphism. Let $N$ be its conormal sheaf, locally free of finite type on $Y$. Then for the direct image
\[
i_*:K_\bullet(Y)\longto K_\bullet(X)
\]
and the extraordinary inverse image
\[
i^!:K_\bullet(X)\longto K_\bullet(Y)
\]
one has
\[
i^!i_*(x)=x\cdot \lambda_{-1}(N),\qquad x\in K_\bullet(Y),
\]
where multiplication is through the natural $K^\bullet(Y)$-module structure on $K_\bullet(Y)$. The proof is the same as the corresponding self-intersection formula in $K^\bullet$.

\paragraph{Corollary 7.2.}
Let $f:X\to S$ be flat, with $S=\Spec(A)$ the spectrum of a discrete valuation ring. Let $s$ and $t$ be the closed and generic points, and let $i:X_s\to X$ be the inclusion. Then $i$ has finite Tor-dimension and therefore gives a pullback
\[
i^*:K_\bullet(X)\longto K_\bullet(X_s).
\]
Moreover the composite $i^*i_*$ on $K_\bullet(X_s)$ is zero, because the conormal bundle is trivial of rank one and $\lambda_{-1}(\cO)=1-1=0$.

\paragraph{Proposition 7.3.}
Let $j:X_t\to X$ denote the inclusion of the generic fibre. There is a unique homomorphism
\[
\sigma:K_\bullet(X_t)\longto K_\bullet(X_s),
\]
called the specialization homomorphism, making the diagram
\[
\begin{array}{ccc}
K_\bullet(X)&\xto{\,j^*\,}&K_\bullet(X_t)\\
\scriptstyle i^*\downarrow&&\downarrow\scriptstyle \sigma\\
K_\bullet(X_s)&=&K_\bullet(X_s)
\end{array}
\]
commute. It exists because $j^*$ is the quotient map given by the homotopy exact sequence, and because $i^*i_*=0$.

When $K_\bullet(X_t)$ and $K_\bullet(X_s)$ are regarded as $K^\bullet(X)$-modules through $j^*$ and $i^*$, the map $\sigma$ is $K^\bullet(X)$-linear.

\subsection*{7.4. Compatibility with filtrations}

Put on $K_\bullet$ the filtration by dimension of support. If $f$ is of finite type, or more generally if $X$ is catenary, the specialization homomorphism is compatible with these filtrations and induces
\[
\sigma:\Gr_\bullet(X_t)\longto\Gr_\bullet(X_s).
\]
It is enough to check this on the class of an integral closed subscheme $Y_t\subset X_t$. Let $Y$ be its schematic closure in $X$. Then $Y$ is flat over $S$, and
\[
\sigma(\cl(Y_t))=\cl(Y_s).
\]
Catenarity gives $\dim Y_s\leq\dim Y_t$, so the filtration bound follows.

\subsection*{7.5. Conservation of number}

Assume $f$ is proper. Then the fibres $X_t$ and $X_s$ are proper over their residue fields, and one has Euler characteristic homomorphisms
\[
\chi_t:K_\bullet(X_t)\longto\ZZ,
\qquad
\chi_s:K_\bullet(X_s)\longto\ZZ.
\]
They satisfy
\[
\chi_s(\sigma(x))=\chi_t(x),\qquad x\in K_\bullet(X_t).
\]
This is the conservation-of-number theorem. Reducing to $x=\cl(Y_t)$ and taking the schematic closure $Y$ of $Y_t$ in $X$, it becomes
\[
\chi(Y_s)=\chi(Y_t),
\]
which is the standard constancy of Euler characteristic in a proper flat family.



\subsection*{7.6. Compatibility with the unit class}

It is immediate from the definition that
\[
\sigma(\cl(\cO_{X_t}))=\cl(\cO_{X_s}).
\]
Since $\sigma$ is $K^\bullet(X)$-linear, the resulting diagram is commutative:
\[
\begin{array}{ccccc}
&&K^\bullet(X)&&\\
&\swarrow\scriptstyle j^*&&\searrow\scriptstyle i^* &\\
K^\bullet(X_t)&&&&K^\bullet(X_s)\\
\downarrow&&&&\downarrow\\
K_\bullet(X_t)&\xto{\,\sigma\,}&&&K_\bullet(X_s),
\end{array}
\]
where the vertical arrows are the canonical maps from $K^\bullet$ to $K_\bullet$.

\subsection*{7.7. Specialization for $K^\bullet$}

Assume now that $X$ is separated and that the fibres $X_t$ and $X_s$ are regular. The canonical maps
\[
K^\bullet(X_t)\longto K_\bullet(X_t),
\qquad
K^\bullet(X_s)\longto K_\bullet(X_s)
\]
are then isomorphisms. We define the specialization homomorphism for $K^\bullet$ as the unique homomorphism making the square
\[
\begin{array}{ccc}
K^\bullet(X_t)&\xto{\,\sigma\,}&K^\bullet(X_s)\\
\downarrow\scriptstyle\sim&&\downarrow\scriptstyle\sim\\
K_\bullet(X_t)&\xto{\,\sigma\,}&K_\bullet(X_s)
\end{array}
\]
commutative.

Since $X$ is regular, hence normal, its connected components coincide with its irreducible components, and they dominate $S$. Therefore the map
\[
H^0(X,\ZZ)\longto H^0(X_t,\ZZ)
\]
induced by the inclusion of the generic fibre is bijective. The analogous map to $H^0(X_s,\ZZ)$ may therefore be interpreted as a specialization map
\[
H^0(X_t,\ZZ)\longto H^0(X_s,\ZZ).
\]
With respect to this map, the specialization on $K^\bullet$ is compatible with the augmented algebra structures. In particular, it induces a homomorphism of graded rings
\[
\sigma^*:\Gr^\bullet(X_t)\longto\Gr^\bullet(X_s).
\]
The Chern character is compatible with this specialization: the square
\[
\begin{array}{ccc}
K^\bullet(X_t)&\xto{\,c_{X_t}\,}&\Ch\Gr^\bullet(X_t)\\
\downarrow\scriptstyle\sigma&&\downarrow\\
K^\bullet(X_s)&\xto{\,c_{X_s}\,}&\Ch\Gr^\bullet(X_s)
\end{array}
\]
commutes.

\subsection*{7.8. Specialization for the Picard group}

Keep the hypotheses of 7.7. By the isomorphism
\[
\Gr^1\simeq\Pic
\]
constructed in 5.3.2, the specialization map on $\Gr^1$ gives a canonical homomorphism
\[
\Pic(X_t)\longto\Pic(X_s).
\]
It may be described more concretely as follows. Since $X$ is regular, its local rings are factorial. Hence every invertible sheaf $L_t$ on $X_t$ extends to an invertible sheaf $L$ on $X$. The image of $L_t$ is the class of the restriction
\[
L_s=L|_{X_s}.
\]

This description is independent of the chosen extension. It is enough to check the case $L_t\simeq\cO_{X_t}$. Then $L\simeq\cO_X(D)$ for a divisor $D$ supported on the special fibre. If $X$ is integral, $D=nX_s$, and this divisor is principal, defined by a power of a uniformizer of the discrete valuation ring. Thus $L_s\simeq\cO_{X_s}$.

\subsection*{7.9. Specialization and numerical equivalence}

Assume $f$ is proper and that $X_t$ and $X_s$ are regular. We have specialization maps
\[
\sigma:\Gr_\bullet(X_t)\longto\Gr_\bullet(X_s),
\qquad
\sigma^*:\Gr^\bullet(X_t)\longto\Gr^\bullet(X_s).
\]
They are compatible with the intersection pairings. If $x_t\in\Gr_\bullet(X_t)$ and $y_t\in\Gr^\bullet(X_t)$, then after specialization one has the equality of intersection numbers
\[
\langle \sigma(x_t),\sigma^*(y_t)\rangle
=
\langle x_t,y_t\rangle.
\]
Equivalently, the specialization homomorphism on $K_\bullet$ respects multiplication by the specialized classes in $K^\bullet$ and respects Euler characteristic by the conservation-of-number theorem.

It follows formally that if the specialization of a class is numerically equivalent to zero, then the original class is numerically equivalent to zero. The converse would follow from surjectivity, modulo torsion, of the specialization map on the appropriate graded group, but this occurs only in special situations.

\paragraph{7.9.4.}
It is nevertheless plausible, at least when $f$ is proper and smooth, that the converse should hold: numerical triviality should be preserved by specialization. If true, the specialization maps would give monomorphisms on the quotients by numerical equivalence. At least the preceding argument shows that the numerical group of the generic fibre is a quotient of a subgroup of the numerical group of the special fibre; in particular its rank is no larger.

\subsection*{7.10. Compatibility with direct and inverse images}

Let
\[
g:X\longto Y
\]
be a morphism of noetherian schemes flat over $S$. Write $\sigma_X$ and $\sigma_Y$ for the specialization maps of $X$ and $Y$, and write
\[
g_t:X_t\to Y_t,
\qquad
 g_s:X_s\to Y_s
\]
for the morphisms on fibres.

If $g$ is proper, then
\[
\begin{array}{ccc}
K_\bullet(X_t)&\xto{\,g_{t*}\,}&K_\bullet(Y_t)\\
\downarrow\scriptstyle\sigma_X&&\downarrow\scriptstyle\sigma_Y\\
K_\bullet(X_s)&\xto{\,g_{s*}\,}&K_\bullet(Y_s)
\end{array}
\]
commutes. Indeed, using the defining diagram of specialization, this is reduced to the standard compatibility of proper direct image with restriction to the open generic fibre and with pullback to the special fibre.

If $g$ has finite Tor-dimension, so that inverse images in $K$-theory are defined, then the analogous diagram
\[
\begin{array}{ccc}
K_\bullet(Y_t)&\xto{\,g_t^*\,}&K_\bullet(X_t)\\
\downarrow\scriptstyle\sigma_Y&&\downarrow\scriptstyle\sigma_X\\
K_\bullet(Y_s)&\xto{\,g_s^*\,}&K_\bullet(X_s)
\end{array}
\]
commutes as well. This follows from the functoriality of finite-Tor pullback and from the same quotient description used to define $\sigma$.

Consequently, whenever the hypotheses allow passage to associated graded objects, the specialization maps commute with the corresponding Gysin morphisms, proper direct images, and pullbacks.

\subsection*{7.11. Algebraic equivalence of invertible sheaves}

Let $T$ be a proper scheme over a field $k$. An invertible sheaf $L\in\Pic(T)$ is algebraically equivalent to zero if there exist a connected nonsingular curve $C$ over $k$, two rational points $t_0,t_1\in C(k)$, and an invertible sheaf $M$ on $T\times C$ such that
\[
L=M_{t_1}\otimes M_{t_0}^{-1}.
\]
The algebraically trivial classes form a subgroup $\Pic^{\mathrm a}(T)$, and one has
\[
\Pic^c(T)\subset \Pic^{\mathrm a}(T)\subset\Pic^{\mathrm{num}}(T).
\]
If $k$ has characteristic zero, or if $T$ is proper over $k$, one can characterize $\Pic^{\mathrm a}(T)$ as the subgroup of classes some power of which belongs to the connected component subgroup $\Pic^c(T)$.

This recovers the classical geometric notion. A divisor $t_1-t_0$ on $C$ pulls back to a divisor on $T\times C$, and specialization from one point of the curve to another gives the displayed relation in the Picard group of $T$.

\subsection*{7.12. Curves and the Picard group}

Let $C$ be a proper smooth connected curve over $k$, and let $t_0,t_1\in C(k)$. Then there is a homomorphism
\[
\Pic(T\times_k C)\xto{\,i_1^*-i_0^*\,}\Pic(T),
\]
where $i_\nu:T\to T\times_k C$ is defined by $t_\nu$ for $\nu=0,1$. Its image is contained in $\Pic^{\mathrm a}(T)$. Conversely, if $k$ is algebraically closed, every algebraically trivial class is obtained in this way for a suitable curve $C$.

If $k$ is finite and $\bar k$ is an algebraic closure, then the Galois group acts on $\Pic^{\mathrm a}(T_{\bar k})$, and one has the usual exact sequence beginning
\[
0\longto\Pic^{\mathrm a}(T)
\longto\Pic^{\mathrm a}(T_{\bar k})^{\operatorname{Gal}(\bar k/k)}
\longto H^1(k,\Pic^c(T_{\bar k}))\longto\cdots .
\]

\subsection*{7.13. Relation with $\ell$-adic cohomology}

For a locally free sheaf $E$ of finite type on a scheme $T$, and for a prime $\ell$ invertible on $T$, one has $\ell$-adic Chern classes
\[
c^i(E)\in H^{2i}(T,\ZZ_\ell(i)).
\]
They satisfy the usual multiplicativity for short exact sequences. Hence the total Chern class defines
\[
c:K^\bullet(T)\longto
1+\prod_{i\geq1}H^{2i}(T,\ZZ_\ell(i)).
\]

Now let $X$ be proper and flat over the trait $S$, with regular fibres, and assume $S$ henselian and $\ell$ different from the residue characteristic. Proper base change gives a specialization isomorphism in cohomology. The diagram
\[
\begin{array}{ccc}
K^\bullet(X_t)&\xto{\,c^i\,}&H^{2i}(X_t,\ZZ_\ell(i))\\
\downarrow\scriptstyle\sigma&&\downarrow\\
K^\bullet(X_s)&\xto{\,c^i\,}&H^{2i}(X_s,\ZZ_\ell(i))
\end{array}
\]
commutes. This is proved by the same argument as for compatibility with pullback: Chern classes commute with inverse image for the immersions of the two fibres.

Passing to finite extensions of the fraction field and then to the limit gives the corresponding statement for geometric fibres
\[
\begin{array}{ccc}
K^\bullet(X_{\bar t})&\longto&H^{2i}(X_{\bar t},\QQ_\ell(i))\\
\downarrow&&\downarrow\\
K^\bullet(X_{\bar s})&\longto&H^{2i}(X_{\bar s},\QQ_\ell(i)).
\end{array}
\]
Equivalently, if a class in $K^\bullet(X_t)$ specializes to a class in $K^\bullet(X_s)$, then their $i$-th Chern classes are the values at the two geometric points of a single section of $R^{2i}f_*\ZZ_\ell(i)$.

After passage to associated graded groups, one obtains maps
\[
c^i:\Gr^i(T)\longto H^{2i}(T,\QQ_\ell(i)),
\]
with the convention that if $\tilde x\in\Filt^iK^\bullet(T)$ lifts $x\in\Gr^i(T)$, then
\[
c^i(x)=c^i(\tilde x)/i!.
\]
The specialization diagrams remain commutative for these graded Chern class maps.

\subsection*{7.14. Passage to limits on the base}

Let the base trait be replaced by a filtered inverse limit of traits, or by the normalization in finite extensions of the generic point. The preceding constructions are compatible with such base change. The key inputs are the compatibility of $K$-theory with filtered limits under affine transition morphisms, the proper base-change theorem in cohomology, and the preservation of regularity under the smooth base changes considered here.

Thus the specialization maps on $K$-groups, on associated graded groups, on Picard groups, and on $\ell$-adic Chern classes may all be transported to geometric fibres without ambiguity.

\subsection*{7.15. Specialization and geometric fibres}

For a smooth proper morphism $X\to S$, the cohomological specialization map on geometric fibres is an isomorphism. The $K$-theoretic specialization need not be an isomorphism, but it is compatible with all constructions used above. In particular Chern classes of specialized $K$-classes correspond under the cohomological specialization isomorphism.

This observation is often used in the following form: a cycle or vector bundle class on the generic fibre whose specialization is numerically trivial has cohomologically trivial Chern classes after specialization, and hence the same Chern classes on the generic fibre.

\subsection*{7.16. Geometric specialization diagrams}

Choose an algebraic closure $\bar K$ of the fraction field of the trait and an induced algebraic closure $\bar k$ of the residue field. For the geometric fibres $X_{\bar t}$ and $X_{\bar s}$, one has compatible specialization maps
\[
K^\bullet(X_{\bar t})\longto K^\bullet(X_{\bar s}),
\]
\[
\Gr^i(X_{\bar t})\longto \Gr^i(X_{\bar s}),
\]
and comparison maps to $\ell$-adic cohomology. The squares
\[
\begin{array}{ccc}
\Gr^i(X_{\bar t})&\xto{\,c^i\,}&H^{2i}(X_{\bar t},\QQ_\ell(i))\\
\downarrow&&\downarrow\scriptstyle\sim\\
\Gr^i(X_{\bar s})&\xto{\,c^i\,}&H^{2i}(X_{\bar s},\QQ_\ell(i))
\end{array}
\]
commute.

\subsection*{7.17. Variants and complements}

The same formalism applies to Chow groups when they are identified with the associated graded pieces of $K$-theory in the cases covered by Riemann--Roch. It also applies to Chern classes with torsion coefficients before passage to $\QQ_\ell$, by working modulo $\ell^n$ and taking limits.

If the morphism is not smooth, one must distinguish the specialization of coherent sheaf classes from that of locally free sheaf classes. The construction of $\sigma$ for $K_\bullet$ remains available under flatness, while the specialization for $K^\bullet$ requires regularity of the fibres or another hypothesis giving the comparison $K^\bullet\simeq K_\bullet$.

\subsection*{7.18. Questions}

These considerations are tied to the standard conjectures. For a smooth projective variety over a field, one may form categories of correspondences modulo rational, algebraic, numerical, or homological equivalence. There are functors
\[
\mathcal C_{\mathrm{rat}}\longto\mathcal C_{\mathrm{num}}\longto\mathcal C_{\mathrm{hom}}.
\]
Grothendieck's standard conjectures assert, in particular, that the comparison with homological equivalence has the expected equivalence properties. Over finite fields some of these statements are known for special classes such as abelian varieties and certain surfaces, but the numerical and homological comparison remains open in general.

One may also use algebraic equivalence and obtain functors
\[
\mathcal C_{\mathrm{rat}}\longto\mathcal C_{\mathrm{alg}}\longto\mathcal C_{\mathrm{num}}.
\]
The assertion that the first of these is an equivalence is a form of Bloch's conjecture; it remains open in general, although many special cases are known.

\section*{Expose XII -- A relative representability theorem for the Picard functor}

\begin{center}
{\large\textbf{A relative representability theorem for the Picard functor}}\\[0.25em]
{\normalsize M. Raynaud, written up by S. Kleiman}
\end{center}

\subsection*{Warning}

The purpose of Exposes XII and XIII is to prove the nine finiteness theorems for the Picard functor stated by Grothendieck in the comments to the \emph{Fondements de la geometrie algebrique}. In this first expose one proves the first finiteness theorem in the following more precise form. Let $X$ and $Y$ be proper schemes over an integral noetherian base $S$, and let
\[
f:X\to Y
\]
be a surjective $S$-morphism. Then there exists a non-empty open subset $V\subset S$ over which the induced morphism
\[
f^*:\Pic_{Y/S}|_V\longto\Pic_{X/S}|_V
\]
is representable by a quasi-affine morphism of finite presentation. The second expose proves the remaining finiteness results.

The two exposes use rather different methods. In the first, the proof uses devissage of $\Gl_n$, non-flat descent, and representability of unramified functors. In the second, one first reduces to cases where the schemes over the base are projective and flat with geometrically integral fibres, sometimes normal or smooth; one then uses the theory of bounded families of sheaves developed by Castelnuovo, Matsusaka, Mumford, Nakai, and Kleiman.

\subsection*{1. Statement of the main theorem and applications}

\paragraph{Theorem 1.1.}
Let $S$ be a noetherian scheme, let $X$ and $Y$ be proper $S$-schemes, and let $f:X\to Y$ be a morphism. Assume $S$ integral and $f$ surjective. Then there is a non-empty open subset $V\subset S$ such that
\[
f^*:\underline{\Pic}_{Y/S}|_V\longto \underline{\Pic}_{X/S}|_V
\]
between the fppf Picard functors is representable by a quasi-affine morphism of finite presentation.

\paragraph{Corollary 1.2.}
If $X$ is proper over an integral noetherian scheme $S$, then after replacing $S$ by a non-empty open subset $V$, the Picard functor $\underline{\Pic}_{X/S}|_V$ is representable by a group scheme over $V$ which is a union of quasi-projective open subschemes of finite type.

\paragraph{Corollary 1.3.}
For a surjective morphism $f:X\to Y$ of proper schemes over a field $k$, the kernel of
\[
f^*:\Pic_{Y/k}\longto\Pic_{X/k}
\]
is an algebraic group scheme of finite type over $k$.

\subsection*{1.4. Reductions of the main theorem}

The proof of Theorem 1.1 is reduced step by step to more special cases.

First, after shrinking the integral noetherian base $S$, one may assume suitable normality or regularity properties. Secondly, by flat descent one reduces to the case where $Y$ is integral and $f$ is generically finite. Thirdly, using flattening, one may suppose that $f$ is flat. Finally, devissage reduces the problem to the case where $X$ and $Y$ are projective and flat over $S$ with geometrically integral fibres.

\subsection*{1.5. Plan of proof}

The proof is organized as follows. One first treats the theorem when $f$ is a regular immersion of codimension one, and more generally when $f$ is finite and flat. Next one proves stability under composition: if the result is known for $X\to Y$ and for $Y\to Z$, it is known for $X\to Z$. Then one reduces to the generically finite flat case. The final step uses a devissage argument in which the structure of $\Gl(n)$, established above, is essential.

\paragraph{Corollary 1.5.}
Let $k$ be a field.
\begin{enumerate}[label=\alph*)]
\item For a proper $k$-scheme $X$, the functor $\underline{\Pic}_{X/k}$ is representable.
\item If $f:X\to Y$ is a surjective morphism of proper $k$-schemes, then
\[
f^*:\underline{\Pic}_{Y/k}\longto\underline{\Pic}_{X/k}
\]
is affine.
\end{enumerate}

\paragraph{Corollary 1.6.}
Let $S$ be integral and noetherian, and let $f:X\to Y$ be a surjective $S$-morphism between proper $S$-schemes.
\begin{enumerate}[label=\alph*)]
\item If $Y$ is normal over $S$, meaning flat with geometrically normal fibres, then after shrinking $S$ the map $f^*$ on Picard functors is quasi-finite.
\item If $\underline{\Pic}_{Y/S}$ is essentially proper over $S$--for instance if $Y$ is smooth over $S$--then, after shrinking $S$, the morphism $f^*$ is finite, hence affine.
\end{enumerate}
Indeed, fibrewise the kernel of $f_s^*$ is an affine algebraic group whose identity component is proper; hence it is finite. In the second case $f^*$ is both proper and quasi-affine, therefore finite.

\paragraph{Remark 1.7.}
It is not known whether Theorem 1.1 remains true with ``affine'' in place of ``quasi-affine'', although the preceding corollaries suggest that this might be the case.

\section*{2. First reductions}

Applying the preliminary reductions to $X/S$ and $Y/S$, one finds, after replacing $S$ by a finite flat cover of a non-empty open subset, that each irreducible component of $X$ and $Y$, endowed with its reduced structure, is integral over $S$. By fppf descent, this replacement is harmless for the representability and quasi-affineness statements.

The diagram obtained by replacing schemes by their reductions shows that it is enough to prove two assertions:

\begin{enumerate}[label=\Roman*.]
\item Theorem 1.1 when $f$ is an immersion, necessarily nilpotent in the reduction step.
\item Theorem 1.1 when $X$ and $Y$ are reduced and every irreducible component of each, with the induced reduced structure, is integral over $S$.
\end{enumerate}

For the second assertion, group the irreducible components $X_i$ according to the component $Y_j$ that contains their image. One obtains diagrams comparing the Picard functor of a scheme with the product of the Picard functors of its components. After shrinking $S$, it is enough to know that the relevant maps to the products are quasi-affine and to treat the case where both $X$ and $Y$ are integral over $S$.

For the maps from a scheme to the disjoint union of its irreducible components, one reduces to the case of two closed subschemes $Y_1,Y_2\subset Y$ whose union is $Y$.

\paragraph{Proposition 2.3.}
Let $g:Y\to S$ be proper, and let $Y_1,Y_2$ be closed subschemes defined by ideals $I_1,I_2$. Assume:
\begin{enumerate}[label=\alph*)]
\item $Y$, $Y_1$, $Y_2$, and $Y_1\cap Y_2$ are faithfully flat and of finite presentation over $S$;
\item $g_*\cO_Y=\cO_S$ and $(g_i)_*\cO_{Y_i}=\cO_S$ universally;
\item $I_1\cap I_2=0$.
\end{enumerate}
Then
\[
\underline{\Pic}_{Y/S}\longto
\underline{\Pic}_{(Y_1\amalg Y_2)/S}
\]
is representable by an affine morphism of finite presentation.

\emph{Proof.}
The assertion is fppf-local on the base. After a base change one may suppose that $Y_1\cap Y_2$ has a section over $S$, giving compatible rigidifications. A point of the Picard functor of $Y_1\amalg Y_2$ is a pair of rigidified invertible sheaves $L_1,L_2$ on $Y_1,Y_2$. To glue them to an invertible sheaf on $Y$, it is necessary and sufficient to give an isomorphism
\[
L_1|_{Y_1\cap Y_2}\iso L_2|_{Y_1\cap Y_2}
\]
compatible with the rigidifications. Conversely, such an isomorphism defines the glued line bundle as the kernel of
\[
(\varepsilon_1)_*L_1\oplus(\varepsilon_2)_*L_2
\longto
(\varepsilon_{12})_*(L_1|_{Y_1\cap Y_2}).
\]
Thus the fibre of the Picard map over $(L_1,L_2)$ is an isomorphism functor.

\paragraph{Lemma 2.4.}
Let $X$ be flat, proper, and of finite presentation over $S$, and let $L,M$ be invertible sheaves on $X$. The functor
\[
T\longmapsto \Isom(L_T,M_T)
\]
is representable by an affine $S$-scheme of finite presentation. If $L$ and $M$ are rigidified along a section, the subfunctor of isomorphisms compatible with the rigidifications is represented by a closed subscheme.

The proof uses the representability of the Hom functors by vector bundles and cuts out the invertible homomorphisms by the equations $uv=1$ and $vu=1$.

\subsection*{2.5. Reduction to an effective-epimorphism condition}

It remains to reduce the integral case to the following assertion. Assume $X$ and $Y$ are integral over $S$ and that the sequence
\[
\cO_Y\longto f_*\cO_X\rightrightarrows g_*\cO_{X\times_YX}
\]
is exact, where $g:X\times_YX\to Y$ is the structural morphism. Then Theorem 1.1 holds.

It is enough to factor $f$ as a finite sequence
\[
X=X_r\to X_{r-1}\to\cdots\to X_1\to X_0=Y
\]
where each map satisfies this exactness condition and, after shrinking the base, each $X_i$ is integral over $S$.

\paragraph{Lemma 2.6.}
Let $Y$ be noetherian and let $f:X\to Y$ be a finite epimorphism, meaning that $\cO_Y\to f_*\cO_X$ is injective. Then $f$ factors into a finite sequence of finite morphisms
\[
X=X_r\to X_{r-1}\to\cdots\to X_1\to X_0=Y
\]
which are effective epimorphisms, i.e. for each step the sequence
\[
0\to\cO_{X_i}\to (f_i)_*\cO_{X_{i+1}}
\rightrightarrows
(f_i\times f_i)_*\cO_{X_{i+1}\times_{X_i}X_{i+1}}
\]
is exact.



\section*{3. Proof of I: Oort's devissage}

\paragraph{Proposition 3.1 (Oort).}
Let $S$ be locally noetherian, let $f:X\to S$ be proper, and let $X'\subset X$ be a closed subscheme. Assume:
\begin{enumerate}[label=\roman*)]
\item $X'$ is defined by an ideal $I$ contained in the nilradical $N$ of $X$, and $NI=0$;
\item $X'$ is flat over $S$ and cohomologically flat over $S$ in dimension $0$;
\item $I$ is flat over $S$ and cohomologically flat in dimension $0$, in such a way that there is a coherent $\cO_{X'}$-module $Q$ with
\[
R^qf'_*(I\otimes M)=\Hom_{\cO_{X'}}(Q,M)
\]
functorially in coherent $M$;
\item the cokernel $E$ of $f'_*\cO_{X'}\to R^qf'_*(I)$ is locally free;
\item for every point $s\in S$, the map
\[
(f'_*\cO_{X'})\otimes k(s)\longto (f_*\cO_X)\otimes k(s)
\]
is surjective.
\end{enumerate}
Then the canonical map
\[
\underline{\Pic}_{X/S}\longto \underline{\Pic}_{X'/S}
\]
is representable by an affine morphism of finite presentation. More precisely, there is an exact sequence of fppf sheaves
\[
0\to\mathbb V(E')\to\underline{\Pic}_{X/S}\xto{u}\underline{\Pic}_{X'/S}\to\mathbb V(Q),
\]
and, if $P=\ker(u)$, then $P\to\underline{\Pic}_{X/S}$ is represented by a closed immersion and $\underline{\Pic}_{X/S}$ is a torsor over $P$ under the vector group $\mathbb V(E')$.

\emph{Proof.}
For an $S$-scheme $T$, apply cohomology to the exact sequence
\[
0\to I_T\to\cO_{X_T}\to\cO_{X'_T}\to0.
\]
The resulting long exact sequences compare units and Picard groups on $X_T$ and $X'_T$. The only small point is that the left square obtained from units is not literally commutative. Nevertheless, one has the inclusion of images
\[
\operatorname{Im}\delta_1\subset\operatorname{Im}\delta_2
\]
for the two connecting homomorphisms. To see this, work after making $S$ and $T$ affine. Choose an affine covering $(V_\lambda)$ of $X_T$, and lift a unit $g$ on $X'_T$ to units $g_\lambda$ on the $V_\lambda$. The two boundary classes are represented by cocycles which differ by multiplication by the image of $g$ acting on $I_T$. The hypotheses on the nilradical and the fibrewise surjectivity in (v) allow this multiplicative discrepancy to be absorbed into the image of the other boundary map.

Using (iii) and (iv), the cohomology sequence becomes the functorial exact sequence
\[
0\to H^0(T,E'\otimes\cO_T)\to
H^1(X_T,\cO_{X_T}^*)\to
H^1(X'_T,\cO_{X'_T}^*)\to
H^0(T,Q\otimes\cO_T).
\]
After sheafification for the fppf topology this is precisely
\[
0\to\mathbb V(E')\to\underline{\Pic}_{X/S}\to
\underline{\Pic}_{X'/S}\to\mathbb V(Q).
\]
The kernel $P$ is the pullback of the zero section of the separated vector group $\mathbb V(Q)$, hence is represented by a closed immersion. The remaining statement says that the first part is a vector-group torsor, and it follows by fppf descent after trivializing the torsor.

\paragraph{Corollary 3.3.}
Under the hypotheses of Proposition 3.1:
\begin{enumerate}[label=\roman*)]
\item the corresponding connected-component Picard sheaves for $X$ and $X'$ are identified whenever one of the two maps is already known to be an isomorphism;
\item if $T$ is affine and $\Pic(X_T)\to \Pic(X'_T)$ is surjective in the relevant component, then the analogous surjectivity holds for $X$.
\end{enumerate}
Both assertions are immediate from the exact sequence above and the fact that vector groups have no higher obstruction on affine bases.

\paragraph{Corollary 3.4.}
Let $k$ be a field and let $X$ be a proper $k$-scheme. Suppose that $\underline{\Pic}_{X/k}$ possesses a universal invertible sheaf. Then over every affine open subset $V$ of $\underline{\Pic}_{X/k}$ there is a universal invertible sheaf on $X\times V$. This applies, by descent, when each connected component of $X$ has a rational point, for instance over an algebraically closed field.

\paragraph{Proposition 3.5.}
Let $S$ be integral and noetherian, and let $f:X\hookrightarrow Y$ be a nilpotent immersion of proper $S$-schemes. Then, after replacing $S$ by a non-empty open subset $V$, the morphism
\[
f^*:\underline{\Pic}_{Y/S}|_V\to\underline{\Pic}_{X/S}|_V
\]
is representable by an affine morphism of finite presentation.

\emph{Proof.}
Filter the nilpotent ideal defining the immersion by successive square-zero pieces satisfying the hypotheses of Proposition 3.1 after shrinking the base. Cohomological flatness in dimension $0$ and local freeness of the necessary cohomology modules are open conditions in a proper flat family. Applying Proposition 3.1 successively to the successive thickenings gives an affine morphism of finite presentation at each step; their composite is affine of finite presentation. This proves assertion I.

\section*{4. Proof of II: the descent part}

The remaining case is the most delicate. Assume $X$ and $Y$ are integral over $S$, and that
\[
\cO_Y\longto f_*\cO_X\rightrightarrows g_*\cO_{X\times_YX}
\tag{$*$}
\]
is exact. After shrinking $S$, one may also assume the auxiliary flatness and finite-presentation hypotheses needed below.

The proof is reduced to the representability of the fibre of $f^*$ over a given line bundle on $X$. After an fppf base change one may suppose that $X\to S$ has a section $\alpha$; then $Y$ has the section $\beta=f\alpha$. A point of $\underline{\Pic}_{X/S}$ is represented by an invertible sheaf $L$ on $X$ rigidified along $\alpha$. We must show that the functor of rigidified invertible sheaves $M$ on $Y$ together with an isomorphism
\[
f^*M\iso L
\]
is quasi-affine and of finite presentation over $S$.

\paragraph{Lemma 4.2.}
Under the exactness condition $(*)$, the morphism $f$ is a descent morphism for locally free sheaves. Equivalently, for locally free sheaves $L,M$ on $Y$, the sequence
\[
\Hom_Y(L,M)\to\Hom_X(f^*L,f^*M)
\rightrightarrows
\Hom_{X\times_YX}(f_1^*f^*L,f_2^*f^*M)
\]
is exact.

\emph{Proof.}
Put $N=\mathcal Hom_Y(L,M)$. The assertion is equivalent to exactness of
\[
\Gamma(Y,N)\to\Gamma(Y,f_*\cO_X\otimes N)
\rightrightarrows
\Gamma(Y,g_*\cO_{X\times_YX}\otimes N).
\]
This follows from $(*)$ after tensoring with the locally free sheaf $N$ and taking global sections.

Thus invertible sheaves on $Y$ pulling back to $L$ are the same as effective descent data on $L$ relative to $f$. Forgetting effectivity gives a monomorphism into the functor of gluing data
\[
\underline{\Isom}(p_1^*L,p_2^*L),
\]
where $p_1,p_2:X\times_YX\to X$ are the projections. By Lemma 2.4 this is represented by an affine scheme of finite presentation over $S$.

It remains to represent the subfunctor consisting of effective descent data. After the base change to the representing affine scheme of all gluing data, this becomes the following problem: given an invertible sheaf $L$ on $X$ equipped with a gluing isomorphism
\[
\varphi:p_1^*L\iso p_2^*L,
\]
determine when this datum descends to $Y$.

\paragraph{Lemma 4.3.}
The gluing datum on $L$ is effective if and only if the following conditions hold:
\begin{enumerate}[label=\arabic*)]
\item for all $i\geq0$, the sheaves $R^if_*L$ are flat over $S$;
\item for all $j\geq0$, the sheaves $R^jg_*p_1^*L$ are flat over $S$;
\item the cokernel of
\[
\pi_1-\pi_2:f_*L\to g_*p_1^*L\simeq g_*p_2^*L
\]
is flat over $S$;
\item the kernel $N$ of the same homomorphism is invertible on $Y$;
\item the natural composite
\[
f^*N\to f^*f_*L\to L
\]
is an isomorphism.
\end{enumerate}

\emph{Proof.}
The last two conditions are visibly sufficient: the kernel $N$ is then the desired descended invertible sheaf, and its pullback is $L$. Conversely, if $L\simeq f^*M$ comes from an invertible sheaf $M$ on $Y$, the assertion may be checked locally on $Y$ after trivializing $M$. It then reduces to the hypotheses imposed on $f$ and to the exactness of $(*)$.

Each of the five conditions is represented by a monomorphism of finite presentation. For the first two one uses the flattening stratification for higher direct images. For the third, once the first two hold, the cokernel commutes with base change and one applies the same flattening criterion. For the fourth and fifth one uses openness of the locus where a finitely presented module is invertible and where a morphism of finitely presented modules is an isomorphism.

\paragraph{Lemma 4.4.}
Let $f:X\to S$ be proper of finite presentation and let $F$ be an $S$-flat sheaf of finite presentation. For a fixed integer $n$, the subfunctor of $S$ on which $R^if_*F$ is flat for all $i\geq n$ is represented by a monomorphism of finite presentation, and over it the sheaves $R^if_*F$ commute with arbitrary base change for $i\geq n$.

The proof is by descending induction on $n$, using the theorem on cohomology and base change and the representability of the flatness functor.

\paragraph{Lemma 4.5.}
For a proper morphism of finite presentation $X\to S$ and a sheaf $F$ of finite presentation on $X$, the subfunctor of $S$ on which $F$ is flat over the base is represented by a monomorphism of finite presentation.

\paragraph{Lemma 4.6.}
Let $u:F\to G$ be a homomorphism of finitely presented modules on a scheme proper over $S$, with $G$ flat over $S$. The subfunctor of $S$ on which $u$ is an isomorphism is represented by an open subscheme of finite presentation. Indeed, first cut out the open on which the cokernel vanishes; over it, the kernel commutes with base change, and one cuts out the open on which the kernel vanishes.

Combining these lemmas, the subfunctor of effective descent data is represented by a monomorphism of finite presentation into the affine scheme of all gluing data. Hence it is quasi-affine of finite presentation. This proves assertion II and therefore Theorem 1.1.

\section*{Expose XIII -- Finiteness theorems for the Picard functor}

{\large\textbf{Finiteness theorems for the Picard functor}}\\[0.25em]
{\normalsize S. Kleiman}
\end{center}


\subsection*{1. Bounded sheaves}

Let $k$ be an algebraically closed field, and let $X$ be a projective $k$-scheme endowed with an ample invertible sheaf $\cO_X(1)$.

\paragraph{Definition 1.1 (Mumford).}
Let $m\in\ZZ$. A coherent sheaf $F$ on $X$ is called \emph{$m$-regular} if, at every point of the support of $F$, the sheaf $\cO_X(1)$ is generated by $H^0(\cO_X(1))$, and if, for every $q>0$, one has
\[
H^q(F(m-q))=0.
\]

\paragraph{Lemma 1.2.}
Let
\[
0\to F(-1)\to F\to G\to0
\]
be an exact sequence of coherent sheaves on $X$, where the first map is induced by tensoring with a section of $\cO_X(1)$. If $F$ is $m$-regular, then $G$ is $m$-regular.

Indeed, the corresponding cohomology sequence contains
\[
H^q(F(m-q))\to H^q(G(m-q))\to H^{q+1}(F(m-q-1)),
\]
and the two outside terms vanish by the definition of $m$-regularity.

\paragraph{Proposition 1.3.}
If $F$ is $m$-regular, then for every $n\geq m$:
\begin{enumerate}[label=\roman*)]
\item $F$ is $n$-regular;
\item the multiplication map
\[
H^0(F(n))\otimes H^0(\cO_X(1))\longto H^0(F(n+1))
\]
is surjective;
\item $F(n)$ is generated by its global sections.
\end{enumerate}

The proof is by induction on the dimension $s$ of the support of $F$. The case $s=-1$ is empty. Choose a section $\sigma\in H^0(\cO_X(1))$ which vanishes at no point of the finite set $\operatorname{Ass}(F)$. It gives an exact sequence
\[
0\to F(-1)\to F\to G\to0,
\]
and by Lemma 1.2 the sheaf $G$ is $m$-regular, with support of dimension $s-1$.

From the exact sequence
\[
H^q(F(n-q-1))\to H^q(F(n-q))\to H^q(G(n-q)),
\]
the induction hypothesis gives $H^q(G(n-q))=0$ for $n\geq m$; hence vanishing for $F(n-q-1)$ implies vanishing for $F(n-q)$. Starting from the defining vanishing at $n=m$, this proves (i).

For (ii), one uses the commutative diagram obtained by multiplying sections by $\sigma$ and comparing the exact rows
\[
0\to H^0(F(n))\to H^0(F(n+1))\to H^0(G(n+1))
\]
with the corresponding multiplication maps for $F$ and $G$. The first row is exact by (i), and the required surjectivity for $G$ is known by the induction hypothesis. This gives the surjectivity for $F$.

For (iii), write $M_X=M\otimes_k\cO_X$ for a vector space $M$. The maps
\[
H^0(F(n))_X\longto F(n)
\]
are compatible with multiplication by $H^0(\cO_X(1))$. Since the multiplication maps in (ii) are surjective, once $H^0(F(n))_X\to F(n)$ is surjective for one $n$, it is surjective for all larger $n$. Serre's theorem gives surjectivity for $n\gg0$, and hence for every $n\geq m$.

\paragraph{Proposition 1.4.}
Let
\[
0\to F(-1)\to F\to G\to0
\]
be an exact sequence on $X$. If $G$ is $m$-regular, then:
\begin{enumerate}[label=\roman*)]
\item $H^q(F(n))=0$ for $q\geq2$ and $n\geq m-(q-1)$;
\item the map $H^1(F(n-1))\to H^1(F(n))$ is an isomorphism for
\[
n\geq (m-1)+h^1(F(m-1));
\]
\item $H^1(F(n))=0$ for
\[
n\geq (m-1)+h^1(F(m-1)).
\]
\end{enumerate}
In particular, $F$ is $[(m-1)+h^1(F(m-1))]$-regular.

Indeed, from
\[
H^{q-1}(G(n))\to H^q(F(n-1))\to H^q(F(n))\to H^q(G(n))
\]
the two outside terms vanish for $q\geq2$ and $n\geq m-(q-1)$ by Proposition 1.3. Thus
\[
H^q(F(m-q))\simeq H^q(F(m-q+1))\simeq H^q(F(m-q+2))\simeq\cdots,
\]
and the common value is zero by Serre's theorem. This proves (i).

For $n\geq m-1$ one has the exact sequence
\[
0\to H^0(F(n-1))\to H^0(F(n))\to H^0(G(n))\to H^1(F(n-1))\to H^1(F(n))\to0.
\]
If the map $H^0(F(n))\to H^0(G(n))$ is surjective, then by the multiplication diagram of Proposition 1.3 the analogous map in degree $n+1$ is also surjective. Hence, once surjectivity has occurred, all subsequent $H^1$ groups are equal and finally vanish by Serre's theorem. If surjectivity has not yet occurred, the integer $h^1(F(n-1))$ strictly decreases when passing to $h^1(F(n))$. There can be at most $h^1(F(m-1))$ such failures. This proves (ii) and (iii).

\paragraph{Definition 1.5.}
Let $F$ be a coherent sheaf on $X$, let $r\geq \dim\operatorname{Supp}(F)$, and let $(b)=(b_0,\ldots,b_r)$ be a sequence of $r+1$ integers. We say that $F$ is a \emph{$(b)$-sheaf} if it satisfies the following equivalent condition. First, $\cO_X(1)$ is generated by $H^0(\cO_X(1))$ at every point of the support of $F$; secondly $h^0(F(-1))\leq b_0$; and, if $r\geq1$, there is a section $\sigma\in H^0(\cO_X(1))$ giving an exact sequence
\[
0\to F(-1)\to F\to G\to0
\]
where $G$ is a $(b_1,\ldots,b_r)$-sheaf.

Equivalently, there is an $F$-regular sequence of $r$ sections $\sigma_1,\ldots,\sigma_r\in H^0(\cO_X(1))$ such that, for $i=0,\ldots,r$, if $F_{\sigma_i}$ denotes the restriction of $F$ to the common zero scheme of $\sigma_1,\ldots,\sigma_i$, then
\[
h^0(F_{\sigma_i}(-1))\leq b_i.
\]

\paragraph{Proposition 1.6.}
Let $F$ be a $(b)$-sheaf on $X$. Then:
\begin{enumerate}[label=\roman*)]
\item for every sufficiently general sequence $\sigma_1,\ldots,\sigma_r$ of sections of $H^0(\cO_X(1))$, the restrictions $F_{\sigma_i}$ satisfy
\[
h^0(F_{\sigma_i}(-1))\leq b_i\qquad(0\leq i\leq r);
\]
\item every subsheaf $G\subset F$ is a $(b)$-sheaf.
\end{enumerate}

Let $T$ be the open subset of the affine space of $r$-tuples of sections corresponding to $F$-regular sequences. For fixed $i$, the sheaves $F_{\sigma_i}(-1)$ for $\sigma\in T(k)$ belong to a flat family over $T$. Since $T$ is non-empty by hypothesis, the upper semicontinuity theorem for $h^0$ proves (i). If $G\subset F$, then a sufficiently general sequence is also $G$-regular and $G_{\sigma_i}$ is a subsheaf of $F_{\sigma_i}$ for all $i$; (ii) follows.

\paragraph{Lemma 1.7.}
Let
\[
0\to F(-1)\to F\to G\to0
\]
be exact. If the Hilbert polynomial of $F$ is written
\[
\chi(F(n))=\sum_{i=0}^r a_i\binom{n+i}{i},
\]
then
\[
\chi(G(n))=\sum_{i=0}^{r-1}a_{i+1}\binom{n+i}{i}.
\]
Indeed,
\[
\chi(G(n))=\chi(F(n))-\chi(F(n-1))
 =\sum_i a_i\left(\binom{n+i}{i}-\binom{n-1+i}{i}\right).
\]

\paragraph{Proposition 1.8.}
Let $F$ be a $(b)$-sheaf on $X$, put $s=\dim\operatorname{Supp}(F)$, and write
\[
\chi(F(n))=\sum_{i=0}^s a_i\binom{n+i}{i}.
\]
Then:
\begin{enumerate}[label=\roman*)]
\item for $n\geq -1$,
\[
h^0(F(n))\leq \sum_{i=0}^s b_i\binom{n+i}{i};
\]
\item $a_i\leq b_i$ for every $i$, and $F$ is also a $(b_0,\ldots,b_{s-1},a_s)$-sheaf.
\end{enumerate}

The proof is by induction on $s$. When $s=0$, the assertion is immediate, since
\[
a_0=h^0(F)=h^0(F(-1))\leq b_0.
\]
For $s\geq1$, choose
\[
0\to F(-1)\to F\to G\to0
\]
with $G$ a $(b_1,\ldots,b_s)$-sheaf and $\dim\operatorname{Supp}(G)=s-1$. Then
\[
h^0(F(n))-h^0(F(n-1))=h^0(G(n)),
\]
and by induction
\[
h^0(G(n))\leq\sum_{i=0}^{s-1} b_{i+1}\binom{n+i}{i}.
\]
Together with $h^0(F(-1))\leq b_0$, this proves (i) by induction on $n$. Lemma 1.7 and the induction hypothesis give (ii).

\paragraph{Definition 1.9.}
The \emph{$(b)$-polynomials} are universal polynomials with non-negative rational coefficients, denoted
\[
P_r(b_0,\ldots,b_{r-1};x),
\]
defined recursively by the preceding estimates for regularity: $P_0=0$, and $P_r$ is obtained by applying Proposition 1.4 to a hyperplane section controlled by $P_{r-1}$. They are increasing in each argument.

\paragraph{Remark 1.10.}
For these polynomials one has, immediately by induction on $r\geq1$,
\[
P_r\bigl(b_0,\ldots,b_{r-1};P_{r-1}(b_1,\ldots,b_{r-1};0)\bigr)
 =P_r(b_0,\ldots,b_{r-1};0).
\]

\paragraph{Theorem 1.11.}
Let $F$ be a $(b)$-sheaf on $X$, and write
\[
\chi(F(n))=\sum_{i=0}^s a_i\binom{n+i}{i}.
\]
Let $(c)=(c_1,\ldots,c_s)$ be a sequence of integers with $(c)\geq (b)-(a)$, and put $m=P_s(c)$. Then $m\geq0$ and $F$ is $m$-regular. In particular, if $s=\dim\operatorname{Supp}(F)$, then $F$ is
\[
P_s(a_0,\ldots,a_{s-1};a_s)
\]
regular.

The proof is by induction on $s$. The case $s=0$ is clear: $m=0$, and a zero-dimensional sheaf is $n$-regular for every $n\geq0$. For $s\geq1$, choose
\[
0\to F(-1)\to F\to G\to0
\]
with $G$ a $(b_1,\ldots,b_s)$-sheaf. By Lemma 1.7 and the induction hypothesis, $G$ is $n$-regular for an integer
\[
n=P_{s-1}(c_2,\ldots,c_s)
\]
controlled by the same universal polynomials. Proposition 1.4 then shows that $F$ is
\[
n+h^1(F(n-1))
\]
regular, while the estimates of Proposition 1.8 bound $h^1(F(n-1))$ by a universal expression in the same data. This is precisely the recursive construction of $P_s$. The final assertion follows from Proposition 1.8(ii) and Remark 1.10.

\paragraph{1.12. Families of coherent sheaves on fibres.}
Let $S$ be noetherian and let $X$ be of finite type over $S$. A \emph{family of classes of coherent sheaves on the fibres of $X/S$} means the following data. For every point $s\in S$ and every field extension $K/k(s)$, one specifies coherent sheaves on $X_K$. Two such sheaves, defined over extensions $K$ and $K'$, are in the same class if, after extension to some common field $K''$ over $k(s)$, they become isomorphic on $X_{K''}$.

Such a family $\mathscr F$ is said to be \emph{bounded by} a coherent sheaf $F$ on $X_T=X\times_ST$, where $T$ is of finite type over $S$, if $\mathscr F$ is contained in the family of classes of the sheaves $F_{k(t)}$ for $t\in T$. It is called \emph{bounded} if such $T$ and $F$ exist.

Assume in addition that $X/S$ is projective and endowed with a relatively ample invertible sheaf $\cO_X(1)$. Then $\mathscr F$ is called a \emph{$(b)$-family}, for $(b)=(b_0,\ldots,b_r)$, if every class in $\mathscr F$ is represented, after extending the ground field to an algebraic closure, by a $(b)$-sheaf.

\paragraph{Theorem 1.13.}
Let $S$ be noetherian and let $X$ be projective over $S$, endowed with an ample invertible sheaf $\cO_X(1)$ whose restrictions $\cO_{X_s}(1)$ are generated by their global sections. Let $\mathscr F$ be a family of classes of coherent sheaves on the fibres of $X/S$. The following conditions are equivalent:
\begin{enumerate}[label=\roman*)]
\item $\mathscr F$ is bounded. Moreover, if all sheaves in $\mathscr F$ are locally free of rank $p$, then $\mathscr F$ may be bounded by a sheaf on some $X_T$ which is locally free of rank $p$.
\item The set of Hilbert polynomials $\chi(F_s(n))$ for $F\in\mathscr F$ is finite, and there is a sequence $(b)$ such that $\mathscr F$ is a $(b)$-family.
\item The set of Hilbert polynomials $\chi(F_s(n))$ is finite, and there is an integer $m$ such that all sheaves in $\mathscr F$ are $m$-regular.
\item The set of Hilbert polynomials is finite, and $\mathscr F$ is contained in the family of quotient classes of sheaves $E_T$, where $E$ is coherent on some $X_T$. In fact, one may take $T=S$ and $E=\cO_X(-m)^{\oplus M}$.
\item $\mathscr F$ is contained in the family of cokernel classes of homomorphisms $E'_T\to E_T$, where $E'$ and $E$ are coherent on some $X_T$. In fact, one may take $T=S$ and
\[
E'=\cO_X(-m')^{\oplus M'},\qquad E=\cO_X(-m)^{\oplus M}.
\]
\end{enumerate}

Suppose first that $\mathscr F$ is bounded by a coherent sheaf $F$ on $X_T$. Cutting $T$ into finitely many locally closed pieces and using generic flatness, we may assume that $F$ is flat over $T$. Then the number of Hilbert polynomials $\chi(F_t(n))$ is at most the number of connected components of $T$. If a fibre $F_t$ is locally free of rank $p$, then $F$ is locally free of rank $p$ near $t$, after possibly replacing $T$ by an open neighbourhood.

By the hypotheses on $\cO_X(1)$ and the semicontinuity theorem, after another finite stratification of $T$ there is a sequence of global sections $\sigma_1,\ldots,\sigma_r$ which is regular for the members of the family, and the dimensions
\[
h^0(F_{\sigma_i,t}(-1))
\]
are bounded. Thus (i) implies (ii).

The implication (ii) $\Rightarrow$ (iii) is exactly Theorem 1.11. If (iii) holds, Proposition 1.3(iii) gives a surjection
\[
\cO_X(-m)^{\oplus M}\longto F
\]
for every member of the family, where $M$ may be chosen as the maximum of the finitely many values $\chi(F(m))$. This proves (iii) $\Rightarrow$ (iv).

Assume (iv). For a quotient sequence
\[
0\to G_T\to E_T\to F_T\to0,
\]
the boundedness of the family of the $E_T$ and the finiteness of the Hilbert polynomials of the $F_T$ imply the corresponding finiteness for the Hilbert polynomials of the kernels $G_T$. By Proposition 1.6, the $G_T$ are again controlled by the same type of $(b)$-condition. Applying the already proved implication (ii) $\Rightarrow$ (iv) to the family of kernels gives (v).

It remains to see that (v) implies boundedness. After replacing $T$ by a finite stratification, one may suppose that $E$ and $E'$ are quotients of sums of twists $\cO_X(-m)$ and that the relevant kernels commute with passage to fibres. Take presentations
\[
L_1\longto E',\qquad L_2\longto E
\]
with $L_i$ finite direct sums of twists of $\cO_X$. Choosing the twists sufficiently negative kills the relevant $\operatorname{Ext}^1$ obstructions. Hence every morphism $E'_T\to E_T$ may be lifted, after replacing the presentation, to a morphism between fixed direct sums of twists, and the cokernel is unchanged. Thus the family is contained in the set of cokernels of morphisms
\[
\cO_{X_T}(-m')^{\oplus M'}\longto \cO_{X_T}(-m)^{\oplus M}.
\]

After one more stratification, the sheaves
\[
R^q(f_T)_*\mathcal Hom(E',E)
\]
are flat and commute with base change. The affine space
\[
\mathbb V\bigl((f_T)_*\mathcal Hom(E',E)^*\bigr)
\]
therefore carries a universal homomorphism $E'_R\to E_R$ on $X_R$, and its universal cokernel bounds the family. This proves (v) $\Rightarrow$ (i), and hence the theorem.

\section*{2. Several technical lemmas}

\paragraph{2.1.}
Let $k$ be a field and let $X$ be proper over $k$. If $L_1,\ldots,L_s$ are invertible sheaves on $X$, and if $Y$ is a closed subscheme of dimension at most $s$, the intersection number
\[
\left\langle c_1(L_1)\cdots c_1(L_s)\cdot Y\right\rangle
\]
is the coefficient of the monomial $n_1\cdots n_s$ in Snapper's polynomial
\[
\chi\bigl(L_1^{\otimes n_1}\otimes\cdots\otimes L_s^{\otimes n_s}\otimes\cO_Y\bigr).
\]

\paragraph{2.2.}
From now on in this section, $k$ is algebraically closed and $X$ is integral, projective, of dimension $r$, and endowed with a very ample invertible sheaf $H=\cO_X(1)$. For an invertible sheaf $L$ on $X$, the integer
\[
\left\langle c_1(L)c_1(H)^{r-1}\right\rangle
\]
is called the degree of $L$, relative to $H$, and is denoted by $d(L)$.

Write
\[
\chi(L(n))=\sum_{i=0}^r a_i\binom{n+i}{i},\qquad
\chi(\cO_X(n))=\sum_{i=0}^r b_i\binom{n+i}{i}.
\]

\paragraph{Lemma 2.3.}
One has $a_r=b_r=d(H)$. If $r\geq1$, the next coefficient is obtained from the corresponding intersection product with $c_1(L)$ and $c_1(H)^{r-2}$; equivalently, it is read off by comparing the Hilbert polynomials of $L$ and of $L\otimes H^{\otimes m}$ at $m=0,1$.

\paragraph{Lemma 2.4.}
If $d(L)\leq0$ and $H^0(L)\neq0$, then $L\simeq\cO_X$. Hence $d(L)=0$ and $h^0(L)=1$.

Indeed, a nonzero section identifies $L$ with $\cO_X(Y)$ for an effective divisor $Y$. If $Y>0$, then
\[
0<\left\langle c_1(Y)c_1(H)^{r-1}\right\rangle=d(L),
\]
contrary to the hypothesis.

\paragraph{Lemma 2.5.}
If $d(L)<0$, then $L$ is a $(0,\ldots,0,a_r)$-sheaf, with $r-1$ zeros.

For $r=0$ this is immediate. If $r\geq1$, then $h^0(L(-1))=0$ by Lemma 2.4. Choose an integral hyperplane section $Y$ when $r\geq2$, and use the exact sequence
\[
0\to L(-1)\to L\to L_Y\to0.
\]
For $r=1$, Lemma 1.7 gives $h^0(L_Y(-1))=a_1$; for larger $r$ the assertion follows by induction on the dimension.


\paragraph{Lemma 2.7.}
Assume $r=2$ and $d(L)\leq0$. Then:
\begin{enumerate}[label=\roman*)]
\item for every integer $n$,
\[
h^1(L(n))\leq a_1(a_1+1)a_2-a_0;
\]
\item $a_0\leq a_1(a_1+1)$.
\end{enumerate}

Let $Y$ be an integral hyperplane section of $X$. By Lemma 1.7 and the preceding lemma, $L_Y$ is a $(0,a_2)$-sheaf, with Hilbert polynomial
\[
\chi(L_Y(n))=a_1+a_2\binom{n+1}{1}.
\]
The bounded-regularity theorem gives that $L_Y$ is $[-a_1]$-regular. Applying cohomology to
\[
0\to L(n-1)\to L(n)\to L_Y(n)\to0
\]
shows that $h^1(L(n))$ is controlled by the single middle value $h^1(L(-a_1-2))$ together with the finite sum of the dimensions of the $H^0(L_Y(n))$. Since $L$ is a $(0,0,a_2)$-sheaf by Lemma 2.5, Proposition 1.8 bounds these dimensions. Expanding the resulting inequality gives the two asserted estimates.

\paragraph{Lemma 2.8.}
Assume $r=2$ and $d(L)<0$. Then the preceding estimate admits an explicit uniform polynomial refinement in terms of the degree $\delta=d(L)$ and the first Hilbert coefficients of $L$. In particular, the numbers $h^1(L(n))$ are bounded above by a universal expression depending only on the Hilbert polynomial of $L$ and on the fixed polarization $H$.

Indeed, write
\[
\chi(L^{\otimes m}(n-1))=a_0^{(m)}+a_1^{(m)}n+a_2\binom{n+1}{2}.
\]
By comparison of Snapper polynomials, $a_0^{(m)}$ and $a_1^{(m)}$ are expressed by the intersection numbers of $c_1(L)^m$ and $c_1(L)^m c_1(H)$, while $a_2=d(H)$. Applying Lemma 2.7 to the twists $L^{\otimes m}(n)$ gives the announced polynomial bound.

\paragraph{Lemma 2.9.}
Assume $r\geq2$, $d(L)<0$, and $X$ normal. Then:
\begin{enumerate}[label=\roman*)]
\item $H^0(L(-n))=0$ for
\[
n\geq 2+h^1(\omega_X\otimes L^{-1});
\]
\item $h^1(L(n))$ is bounded, for all $n$, by a universal polynomial depending only on the last Hilbert coefficients of $L$ and on the fixed polarization;
\item there are universal polynomials $\beta_i$ with rational coefficients, independent of $k$ and of $X$, such that
\[
|a_i|\leq \beta_i(a_0,\ldots,a_r)\qquad(0\leq i\leq r).
\]
\end{enumerate}

For $r=2$ this is Lemma 2.8, together with Serre duality. In higher dimension the proof proceeds by induction on $r$. Choose a normal hyperplane section $Y$. The exact cohomology sequence
\[
H^0(L_Y(-n))\to H^1(L(-n-1))\to H^1(L(-n))\to H^1(L_Y(-n))
\]
reduces the assertion to the estimates already known for $Y$. Since
\[
\chi(L_Y(n))=\sum_{i=0}^{r-1}a_{i+1}\binom{n+i}{i},
\]
the induction hypothesis bounds the terms occurring on the right. The eventual vanishing for $n\gg0$ then propagates the bound back to all $n$. Finally, once $L_Y$ is known to be regular with a universally bounded regularity, Proposition 1.4 and the formula for the Hilbert polynomial express the remaining coefficients of $L$ through bounded quantities.

\paragraph{Remark 2.10.}
For a fixed integer $m$, compare the two expansions
\[
\sum_{i=0}^r a_i^{(m)}\binom{n+i}{i}
 =\sum_{i=0}^r a_i\binom{n+m+i}{i}.
\]
An elementary induction on $r$, evaluating at $n=-1$, gives
\[
a_i^{(m)}=\sum_{j=i}^r a_j\binom{m-1+j-i}{j-i}.
\]
It follows in particular that, for fixed $m$, the coefficients after twisting are bounded by universal rational polynomials in the original coefficients.

\paragraph{Lemma 2.11.}
Let $S$ be noetherian and let $X$ be projective over $S$, with very ample sheaf $H=\cO_X(1)$. Assume that the fibres of $X/S$ are geometrically integral and of common dimension $r$. Let $\mathscr L$ be a family of classes of invertible sheaves $L_s$ on the fibres of $X/S$. Then $\mathscr L$ is bounded if and only if, in the Hilbert polynomials
\[
\chi(L_s(n))=\sum_{i=0}^r a_i(s)\binom{n+i}{i},
\]
all coefficients $a_i(s)$ remain bounded. If, moreover, the fibres are geometrically normal, it is enough that the three coefficients $a_{r-1}$, $a_1$, and $a_0$ remain bounded.

Let $\delta=\sup_s d(L_s)$. If $\delta>0$, replace $\mathscr L$ by the family of the $L_s(-\delta)$; the formulae comparing the Hilbert coefficients after a twist show that this changes only the boundedness of the coefficients in a controlled way. Thus one may reduce to the case $\delta<0$. In that case Lemma 2.5 shows that the sheaves form a $(b)$-family once the Hilbert coefficients are bounded, and boundedness follows from Theorem 1.13. In the geometrically normal case the deeper estimates of this section show that boundedness of the three indicated coefficients already bounds all the other coefficients; Theorem 1.13 then applies again.
