\section*{Exposé III B: Calculs de termes locaux}

\subsection*{2. Réduction à un théorème d'annulation}

\subsubsection*{2.1}
Sous les hypothèses de 1.1, notons $s$ (resp. $t$) l'image de $z$ dans $A$ (resp. $B$), $i:U=X-\{x\}\hookrightarrow X$ et $j:V=Y-\{y\}\hookrightarrow Y$ les inclusions canoniques. Comme $a_2$ et $b_1$ sont finis, on a $a_2^{-1}(y)=s$, $b_1^{-1}(x)=t$, donc
\[
a_1^{-1}(U)\subset a_2^{-1}(V),\qquad b_2^{-1}(V)\subset b_1^{-1}(U),
\]
et l'on a des diagrammes commutatifs
\[
\begin{tikzcd}[column sep=large,row sep=large]
U & a_1^{-1}(U) \arrow[l,"a'_1"'] \arrow[r,"a'_2"] \arrow[d] & V \arrow[d,"j"]\\
X & A \arrow[l,"a_1"] \arrow[r,"a_2"'] & Y
\end{tikzcd}
\qquad
\begin{tikzcd}[column sep=large,row sep=large]
U & b_2^{-1}(V) \arrow[l,"b'_1"'] \arrow[r,"b'_2"] \arrow[d] & V \arrow[d,"j"]\\
X & B \arrow[l,"b_1"] \arrow[r,"b_2"'] & Y ,
\end{tikzcd}
\]
où les carrés correspondant à $U\leftarrow a_1^{-1}(U)\to A\to X$ et à $b_2^{-1}(V)\to V\to Y$ sont cartésiens.

\begin{statement}{Lemme 2.2}
Pour prouver 1.2, on peut se borner à considérer les deux cas suivants:
\[
\text{a) } L_x=0,\quad M_y=0;\qquad
\text{b) } L|U=0,\quad M|V=0 .
\]
\end{statement}

Considérons les filtrations décroissantes sur $L$, $M$ définies par les sous-complexes
\[
F^nL=
\begin{cases}
L,& n\le 0,\\
i_!i^*L,& n=1,\\
0,& n>1,
\end{cases}
\qquad
F^nM=
\begin{cases}
M,& n\le 0,\\
j_!j^*M,& n=1,\\
0,& n>1.
\end{cases}
\]
Le complexe $a_{2*}a_1^*L$ (resp. $b_{1*}b_2^*M$) est alors filtré par
\[
F^n a_{2*}a_1^*L=a_{2*}a_1^*F^nL.
\]
On a $F^n a_{2*}a_1^*L=a_{2*}a_1^*L$ pour $n\le 0$, et $0$ pour $n>1$; de plus, comme $a_2$ est fini et le carré cartésien, on a
\[
F^1a_{2*}a_1^*L=j_!(a'_2)_!(a'_1)^*i^*L.
\]
On a de même
\[
F^n b_{1*}b_2^*M=
\begin{cases}
b_{1*}b_2^*M,& n\le 0,\\
i_!(b'_1)_!(b'_2)^*j^*M,& n=1,\\
0,& n>1.
\end{cases}
\]

On peut supposer $u$ (resp. $v$) défini par un homomorphisme de complexes
\[
u:a_{2*}a_1^*L\longrightarrow M
\quad
(\text{resp. }v:b_{1*}b_2^*M\longrightarrow L).
\]
Les formules ci-dessus montrent que le composé
\[
F^1a_{2*}a_1^*L\longrightarrow a_{2*}a_1^*L\xrightarrow{u}M
\]
(resp. $F^1b_{1*}b_2^*M\to b_{1*}b_2^*M\xrightarrow{v}L$) se factorise de manière unique à travers $F^1M$ (resp. $F^1L$); en d'autres termes, $u$ (resp. $v$) est un homomorphisme de complexes filtrés. Ainsi $u$ (resp. $v$) est sous-jacent à une correspondance filtrée, au sens de (III 4.13), que nous noterons encore
\[
u\in\Hom_{DF}(a_1^*L,a_2^!M),\qquad
v\in\Hom_{DF}(b_2^*M,b_1^!L).
\]
Par l'additivité des cup-produits ([3] V 3.8.7), (III 4.13.1), on a
\[
\langle u_z,v_z\rangle
 =\langle \operatorname{gr}^0u_z,\operatorname{gr}^0v_z\rangle
 +\langle \operatorname{gr}^1u_z,\operatorname{gr}^1v_z\rangle,
\]
et
\[
\langle u,v\rangle
 =\langle \operatorname{gr}^0u,\operatorname{gr}^0v\rangle
 +\langle \operatorname{gr}^1u,\operatorname{gr}^1v\rangle.
\]
Le lemme en résulte, car $\operatorname{gr}^1L=i_!i^*L$ (resp. $\operatorname{gr}^1M=j_!j^*M$) a une fibre nulle en $x$ (resp. $y$), tandis que $\operatorname{gr}^0L$ (resp. $\operatorname{gr}^0M$) est concentré en $x$ (resp. $y$).

\begin{statement}{Lemme 2.3}
Sous les hypothèses de 1.1, supposons $L|U=0$, $M|V=0$. Alors on a
\[
\langle u_z,v_z\rangle=\langle u,v\rangle .
\]
\end{statement}

L'idée de la démonstration est qu'alors $u$ et $v$ sont images directes de correspondances entre complexes concentrés aux points fermés et que la formule à établir ne fait qu'exprimer la formule de Lefschetz pour les projections $x\to S$, $y\to S$. Compte tenu de 1.1.3, on peut remplacer les données de 1.1 par les suivantes: $X$ (resp. $Y$) est une $S$-courbe lisse munie d'un point fermé $x$ (resp. $y$), $A$ (resp. $B$) est une $S$-courbe séparée, $a:A\to X\times_S Y$, $b:B\to X\times_S Y$ sont des $S$-morphismes finis tels que $A\times_{X\times_SY}B$ soit réduit à un point fermé $z$ au-dessus de $(x,y)$,
\[
u\in\Hom(a_1^*L,a_2^!M),\qquad v\in\Hom(b_2^*M,b_1^!L),
\]
avec
\[
L=h_{1*}L',\quad M=h_{2*}M',\quad
L'\in\Ob D_{\mathrm{ctf}}(\{x\}),\quad M'\in\Ob D_{\mathrm{ctf}}(\{y\}),
\]
où $h_1:\{x\}\to X$ et $h_2:\{y\}\to Y$ désignent les inclusions canoniques. Il s'agit de montrer que
\[
\langle u_z,v_z\rangle=\langle u,v\rangle_z,
\]
où
\[
u_z\in\Hom(\mathrm R\Gamma(X,L),\mathrm R\Gamma(Y,M)),\qquad
v_z\in\Hom(\mathrm R\Gamma(Y,M),\mathrm R\Gamma(X,L))
\]
sont définis comme en 1.1, et $\langle u,v\rangle_z$ est la valeur en $z$ du cup-produit (III 4.2.7). On a $\mathrm R\Gamma(X,L)=\mathrm R\Gamma_c(X,L)=L_x$, $\mathrm R\Gamma(Y,M)=\mathrm R\Gamma_c(Y,M)=M_y$, et $u_z$ (resp. $v_z$) n'est autre que l'image directe $u_*$ (resp. $v_*$) de $u$ (resp. $v$) sur $S$ au sens de (III 3.7.6), comme le montre le procédé de calcul (III 3.5 b)).

D'autre part, si $h=h_1\times_Sh_2:(x,y)\to X\times_SY$ est l'inclusion, et si $a':A'\to(x,y)$, $b':B'\to(x,y)$ désignent les fibres de $a$, $b$ en $(x,y)$, réduites chacune à un point, on a des isomorphismes canoniques
\[
h_*a'_*a'{}^!(DL_x\lten_{\Lambda}M_y)
 \xrightarrow{\sim} a_*a^!(DL\lten_{\Lambda}M),
\]
\[
h_*b'_*b'{}^!(L_x\lten_{\Lambda}DM_y)
 \xrightarrow{\sim} b_*b^!(L\lten_{\Lambda}DM),
\]
et les flèches
\[
h_*:\Hom(a_1'{}^*L_x,a_2'{}^!M_y)\to \Hom(a_1^*L,a_2^!M),
\]
\[
h_*:\Hom(b_2'{}^*M_y,b_1'{}^!L_x)\to \Hom(b_2^*M,b_1^!L)
\]
(III 3.7.6) sont des isomorphismes. Les correspondances $u$ et $v$ s'écrivent donc de manière unique
\[
u=h_*u',\qquad v=h_*v',
\]
avec
\[
u'\in\Hom(a_1'{}^*L_x,a_2'{}^!M_y),\qquad
v'\in\Hom(b_2'{}^*M_y,b_1'{}^!L_x).
\]
Par la formule de Lefschetz (III 4.5.1) pour $(h_1,h_2,A'\to A,B'\to B)$, on a
\[
\langle u,v\rangle_z=\langle u',v'\rangle_z .
\]
D'autre part, on a, d'après la formule de Lefschetz pour $\{x\}\to S$, $\{y\}\to S$,
\[
\langle u_*,v_*\rangle=\langle u',v'\rangle_z .
\]
Compte tenu de la remarque faite plus haut, la conclusion découle de la conjugaison de ces deux égalités.

\begin{statement}{Lemme 2.4}
Pour prouver 1.2, on peut supposer que $a$ et $b$ sont des immersions fermées, et il suffit de démontrer que, si $L_x=0$, $M_y=0$, la flèche de restriction
\[
H_A^0(X\times_SY,DL\lten_{\Lambda}M)
 \longrightarrow H_z^0(B,b^*(DL\lten_{\Lambda}M))
\tag{2.4.1}
\]
est nulle.
\end{statement}

Posons $X\times_SY=Z$. Les sous-schémas fermés $a':A'\to Z$, $b':B'\to Z$, images de $a$ et $b$, vérifient les mêmes hypothèses que $a$ et $b$. Soit $u'$ (resp. $v'$) la correspondance à support dans $A'$ (resp. $B'$) image directe de $u$ (resp. $v$) par la flèche déduite de la flèche d'adjonction $f_*f^!\to 1$ (resp. $g_*g^!\to 1$), où $f:A\to A'$ (resp. $g:B\to B'$) est la projection. Il est immédiat que $u_z=u'_z$, $v_z=v'_z$. D'autre part, il découle aisément de la définition du cup-produit 1.1.2 que
\[
\langle u,v\rangle=\langle u',v'\rangle,
\]
d'où la première assertion.

D'après 2.2 et 2.3, on peut supposer que $L_x=0$, $M_y=0$, et la conclusion de 1.2 s'écrit alors $\langle u,v\rangle=0$. Soient $P,Q\in\Ob D_{\mathrm{ctf}}^b(Z)$. Le cup-produit
\[
H_A^0(Z,P)\otimes H_B^0(Z,Q)\longrightarrow H_z^0(Z,P\lten_{\Lambda}Q)
\]
se décompose en
\[
H_A^0(Z,P)\otimes H_B^0(Z,Q)
 \xrightarrow{\,b^*\otimes 1\,}
H_z^0(B,b^*P)\otimes H_B^0(Z,Q)
 \longrightarrow H_z^0(Z,P\lten_{\Lambda}Q),
\]
où $b^*$ est la restriction et la seconde flèche, variante du produit considéré dans (SGA $4^{1/2}$ Cycle I 2.1), dérive du produit évident au niveau des sections de faisceaux. Cette assertion se vérifie aisément, par exemple à l'aide des résolutions flasques de (SGA 4 XVII 4.2); si $a':\{z\}\to B$, $b':\{z\}\to A$, $c:\{z\}\to Z$ sont les inclusions, on peut aussi invoquer la décomposition standard de la flèche de Künneth (III 4.2.1)
\[
a'{}^!P\lten_{\Lambda} b'{}^!Q\longrightarrow c^!(P\lten_{\Lambda}Q)
\]
en une flèche de changement de base
\[
a'{}^!P\lten_{\Lambda} b'{}^!Q\longrightarrow a'{}^!b^*P\lten_{\Lambda} a^*b'{}^!Q,
\]
suivie de flèches de projection
\[
a'{}^!b^*P\lten_{\Lambda} a^*b'{}^!Q
 \longrightarrow a'{}^!(b^*P\lten_{\Lambda} b'{}^!Q)
 \longrightarrow a'{}^!b'{}^!(P\lten_{\Lambda}Q).
\]
Prenant $P=DL\lten_{\Lambda}M'$, $Q=L\lten_{\Lambda}DM$, et revenant à la définition de $\langle u,v\rangle$ (III 4.2.5), on voit qu'il suffit de prouver que $b^*u=0$, d'où la conclusion.

Compte tenu de 2.4, 1.2 va donc découler du résultat plus précis suivant.

\begin{statement}{Proposition 2.5}
Sous les hypothèses de 1.2, on suppose que $a$ et $b$ sont des immersions fermées et que $L_x=0$, $M_y=0$. Désignons par $T$ le localisé strict de $X\times_SY$ en $z=(x,y)$, et identifions $A$, $B$ à des sous-schémas fermés de $T$. Alors les flèches de restriction
\[
b^*:\mathrm R\Gamma_A(T,(DL\lten_{\Lambda}M)|T)
 \longrightarrow \mathrm R\Gamma_z(B,b^*(DL\lten_{\Lambda}M)),
\]
\[
a^*:\mathrm R\Gamma_B(T,(L\lten_{\Lambda}DM)|T)
 \longrightarrow \mathrm R\Gamma_z(A,a^*(L\lten_{\Lambda}DM))
\]
sont nulles.
\end{statement}

\subsubsection*{2.6}
La démonstration de 2.5 va occuper le reste du numéro 2 et les numéros 3 et 4. Compte tenu de la symétrie de l'énoncé, on peut se borner à prouver l'assertion relative à $b^*$. Nous allons la réexprimer sous une forme qui nous sera plus commode. Notons d'abord que
\[
\mathrm R\Gamma(T,(DL\lten_{\Lambda}M)|T)
 =(DL\lten_{\Lambda}M)_z
 =(DL)_x\lten_{\Lambda}M_y=0,
\]
car $M_y=0$, et que de même
\[
\mathrm R\Gamma(B,b^*(DL\lten_{\Lambda}M))
 =(DL\lten_{\Lambda}M)_z=0.
\]
Par suite, les triangles de cohomologie relative donnent des isomorphismes
\[
\mathrm R\Gamma(T-A,(DL\lten_{\Lambda}M)|T-A)
 \xrightarrow{\sim}
\mathrm R\Gamma_A(T,(DL\lten_{\Lambda}M)|T)[1],
\]
\[
\mathrm R\Gamma(B-z,b^*(DL\lten_{\Lambda}M)|B-z)
 \xrightarrow{\sim}
\mathrm R\Gamma_z(B,b^*(DL\lten_{\Lambda}M))[1],
\]
par lesquels la flèche $b^*$ de 2.5, décalée de 1, s'identifie à la restriction
\[
b^*:\mathrm R\Gamma(T-A,(DL\lten_{\Lambda}M)|T-A)
 \longrightarrow \mathrm R\Gamma(B-z,(DL\lten_{\Lambda}M)|B-z).
\tag{2.6.1}
\]
Comme $L_x=0$, $M_y=0$, on peut écrire
\[
L=i_!P,\qquad M=j_!Q,
\tag{2.6.2}
\]
où $i:X-x\to X$, $j:Y-y\to Y$ désignent les inclusions, et $P\in\Ob D_{\mathrm{ctf}}(X-x)$, $Q\in\Ob D_{\mathrm{ctf}}(Y-y)$. Par l'isomorphisme de dualité (SGA 4 XVIII 3.19.6), plus un passage à la limite, on a
\[
DL=Ri_*DP=Ri_*P,
\]
donc
\[
DL\lten_{\Lambda}M=Ri_*P\lten_{\Lambda}j_!Q.
\]
Posons $X\times_SY=Z$, et identifions $X\times\{y\}$, $\{x\}\times Y$ à des sous-schémas fermés de $Z$ (resp. $T$), que nous noterons simplement $X$, $Y$. Notons
\[
f':Z-Y\hookrightarrow Z,\qquad
g':Z-(X\cup Y)\hookrightarrow Z-Y,
\]
\[
f:T-Z\hookrightarrow T,\qquad
g:T-(X\cup Y)\hookrightarrow T-Y
\]
les inclusions. La formule de Künneth (III 1.6.4) donne, par passage à la limite,
\[
DL\lten_{\Lambda}M=Rf'_*g'_!(P\lten_{\Lambda}Q),
\]
d'où
\[
(DL\lten_{\Lambda}M)|T=Rf_*g_!(P\lten_{\Lambda}Q)|T-(X\cup Y).
\tag{2.6.3}
\]
Avec cette identification, 2.6.1 s'écrit
\[
b^*:\mathrm R\Gamma(T-(Y\cup A),g_!(E)|T-(Y\cup A))
 \longrightarrow \mathrm R\Gamma(B-z,g_!(E)|B-z),
\tag{2.6.4}
\]
où
\[
E=(P\lten_{\Lambda}Q)|T-(X\cup Y).
\]
Dans les deux numéros qui suivent, nous allons montrer que la flèche 2.6.4 est nulle: nous nous ramènerons d'abord au cas où les faisceaux de cohomologie de $E$ ont une ramification modérée, l'action de l'inertie se faisant plus précisément à travers un $\ell$-groupe, pour un nombre premier $\ell\ne p$, puis traiterons directement ce cas à l'aide d'un théorème de pureté.

\subsection*{3. Réduction au cas modéré}

\subsubsection*{3.1}
Rappelons que l'anneau de base $\Lambda$ est annulé par un entier premier à $p$, donc est produit direct d'anneaux noethériens $\Lambda_i$, chaque $\Lambda_i$ étant annulé par une puissance d'un nombre premier $\ell_i\ne p$. Pour tout schéma $S'$ sur $S$, la catégorie dérivée $D(S',\Lambda)$ est produit des catégories $D(S',\Lambda_i)$, de sorte que, pour prouver la nullité de 2.6.4, on peut supposer que $\Lambda$ est annulé par une puissance d'un nombre premier $\ell\ne p$. On peut supposer, d'autre part, que $P$ (resp. $Q$) est borné et à composantes localement libres de type fini.

\subsubsection*{3.2}
Posons $X-x=U$, $Y-y=V$. Soient $U''/U$ un revêtement galoisien fini de groupe $G$ qui trivialise les composantes de $P$, $H$ un $\ell$-sous-groupe de Sylow de $G$, $U'=U''/H$ le revêtement quotient, $X'$ le trait strictement local normalisé de $X$ dans $U'$: $X'/X$ est un revêtement fini, connexe, de degré $d_1$ premier à $\ell$, et la représentation de $\pi_1(U')$ définie par les composantes de $P'=P|U'$ se factorise à travers un $\ell$-groupe fini. Procédant de même avec $Q$, on trouve un revêtement fini, connexe, $Y'/Y$ de degré $d_2$ premier à $\ell$, induisant sur $V$ un revêtement étale $V'$ tel que la représentation de $\pi_1(V')$ définie par les composantes de $Q'=Q|V'$ se factorise à travers un $\ell$-groupe fini. Soit
\[
d=d_1e_1=d_2e_2=\operatorname{ppcm}(d_1,d_2).
\]
Quitte à extraire une racine $e_1$-ième (resp. $e_2$-ième) d'une uniformisante de $X'$ (resp. $Y'$), on peut supposer que $X'$ (resp. $Y'$) a les mêmes propriétés que précédemment, avec de plus $d_1=d_2=d$.

Soient $x'$ (resp. $y'$) le point fermé de $X'$ (resp. $Y'$), $T'$ le localisé strict de $X'\times_S Y'$ en $(x',y')$, $z'$ le point fermé de $T'$, $A'$ (resp. $B'$) l'image inverse de $A$ (resp. $B$) dans $T'$, et
\[
E'=(P\lten_{\Lambda}Q)|T'-(X'\cup Y')
   =(P'\lten_{\Lambda}Q')|T'-(X'\cup Y').
\]
On a alors un carré commutatif de flèches de restriction
\[
\begin{tikzcd}[column sep=huge,row sep=large]
\mathrm R\Gamma(T-(Y\cup A),g_!(E)|T-(Y\cup A))
 \arrow[r,"b^*"] \arrow[d]
&
\mathrm R\Gamma(B-z,g_!(E)|B-z) \arrow[d,"(1)"]\\
\mathrm R\Gamma(T'-(Y'\cup A'),g'_!(E')|T'-(Y'\cup A'))
 \arrow[r,"b'{}^*"]
&
\mathrm R\Gamma(B'-z',g'_!(E')|B'-z') ,
\end{tikzcd}
\]
où $b':B'\to T'$, $g':T'-(X'\cup Y')\to T'-Y'$ sont les inclusions. Le revêtement $T'/T$ est fini, de degré $d^2$, et étale en dehors de $X\cup Y$, donc induit un revêtement étale de degré $d^2$ de $B'-z'$ sur $B-z$. Le composé de (1) et du morphisme trace
\[
\mathrm R\Gamma(B'-z',g'_!(E')|B'-z')
 \longrightarrow
\mathrm R\Gamma(B-z,g_!(E)|B-z)
\]
est donc la multiplication par $d^2$, qui est un isomorphisme, puisque $\ell$ ne divise pas $d$. Pour prouver que la flèche 2.6.4 est nulle, il suffit donc de prouver que $b'{}^*=0$. On sera donc ramené au cas où la représentation de $\pi_1(U)$ (resp. $\pi_1(V)$) définie par les composantes de $P$ (resp. $Q$) se factorise à travers un $\ell$-groupe fini, pourvu que l'on s'assure que $A'$, $B'$ vérifient les hypothèses de position générale de 1.2. Cela va résulter du lemme suivant:

\begin{statement}{Lemme 3.3}
Sous les hypothèses de 1.2, $a$ et $b$ étant des immersions fermées, soient $X'/X$, $Y'/Y$ des revêtements finis de $S$-traits strictement locaux, de même degré $d$. Notons $A'$ (resp. $B'$) l'image inverse de $A$ (resp. $B$) dans $X'\times_S Y'$. Alors les couples $(A',B')$, $(A',X')$, $(Y',B')$ sont chacun en position générale au sens de 1.2.
\end{statement}

La démonstration de 3.3 sera donnée après quelques préliminaires.

\begin{statement}{Lemme 3.4}
Soient $X\xleftarrow{s_1}W\xrightarrow{s_2}Y$ des morphismes finis de $S$-traits, $s=(s_1,s_2):W\to X\times_SY$, et $u,v,w$ des uniformisantes de $X,Y,W$ respectivement, de sorte que
\[
s_1^*u=aw^m\pmod {w^{m+1}},\qquad
s_2^*v=bw^n\pmod {w^{n+1}},
\]
avec $a,b\in k^*$. Alors: (i) si $m<n$, $s(W)$ est tangent à $X$ en $z=(x,y)$; (ii) si $m=n$, $s(W)$ est tangent au sous-schéma de $X\times_SY$ d'équation
\[
ap_2^*(v)-bp_1^*(u)=0.
\]
\end{statement}

Soient $Z=X\times_SY$, $W'$ le sous-schéma fermé réduit image de $s$. Dans l'espace tangent en $z$ à $Z$, la pente, par rapport à la tangente à $X$, de la tangente à $W'$ est la valeur à l'origine de la fonction induite sur $W'$ par $p_2^*(v)/p_1^*(u)$; c'est aussi la valeur à l'origine de la fonction induite par $p_2^*(v)/p_1^*(u)$ sur $W$, c'est donc $0$ (resp. $b/a$) dans le cas (i) (resp. (ii)).

Dans la situation de 3.4, nous définirons la pente de $W$ par rapport à $((X,u),(Y,v))$ comme l'élément de $k\cup\{\infty\}$ égal à $0$ si $m<n$, $\infty$ si $m>n$, $b/a$ si $m=n$. Cet élément ne dépend pas du choix de $w$.

\begin{statement}{Lemme 3.5}
Soient $f:X'\to X$, $g:Y'\to Y$ des morphismes finis de $S$-traits de même indice de ramification $d$, et $u,v,u',v'$ des uniformisantes de $X,Y,X',Y'$ respectivement. Il existe alors $C\in k^*$ tel que, pour tout diagramme commutatif de morphismes finis de $S$-traits
\[
\begin{tikzcd}[column sep=large,row sep=large]
X' & W' \arrow[l,"s'_1"'] \arrow[r,"s'_2"] \arrow[d,"h"] & Y' \arrow[d,"g"]\\
X & W \arrow[l,"s_1"] \arrow[r,"s_2"'] & Y ,
\end{tikzcd}
\]
on ait $\lambda=C{\lambda'}^d$, où $\lambda$ (resp. $\lambda'$) est la pente de $W$ (resp. $W'$) par rapport à $((X,u),(Y,v))$ (resp. $((X',u'),(Y',v'))$).
\end{statement}

Par hypothèse, on a
\[
f^*u=\alpha u'{}^d\pmod {u'{}^{d+1}},\qquad
g^*v=\beta v'{}^d\pmod {v'{}^{d+1}},
\]
avec $\alpha,\beta\in k^*$. Montrons que $C=\beta/\alpha$ a les propriétés voulues. Soit $w$ (resp. $w'$) une uniformisante de $W$ (resp. $W'$). On a
\[
s_1^*u=aw^m\pmod {w^{m+1}},\qquad
s_2^*v=bw^n\pmod {w^{n+1}},
\]
\[
s'_1{}^*u'=a'w'{}^{m'}\pmod {w'{}^{m'+1}},\qquad
s'_2{}^*v'=b'w'{}^{n'}\pmod {w'{}^{n'+1}},
\]
\[
h^*w=\gamma w'{}^r\pmod {w'{}^{r+1}},
\]
avec $a,b,a',b',\gamma\in k^*$. La commutativité des carrés du diagramme entraîne
\[
m'd=rm,\qquad n'd=rn,
\]
d'où $m'/n'=m/n$. Supposons d'abord $m=n$. On a $\lambda=b/a$, $\lambda'=b'/a'$. Si l'on pose $N=m'd=rm=n'd=rn$, les congruences ci-dessus fournissent
\[
(fs'_1)^*u=\alpha a'{}^d w'{}^N\pmod {w'{}^{N+1}},\qquad
(s_1h)^*u=a\gamma^m w'{}^N\pmod {w'{}^{N+1}},
\]
d'où $\alpha a'{}^d=a\gamma^m$. On obtient de même $\beta b'{}^d=b\gamma^m$, d'où $\lambda=C{\lambda'}^d$, pour $\lambda\ne 0,\infty$. Mais, comme $m'/n'=m/n$, $\lambda=0$ (resp. $\infty$) équivaut à $\lambda'=0$ (resp. $\infty$), ce qui achève la démonstration.

Prouvons maintenant 3.3. Soient $Z=X\times_SY$, $Z'=X'\times_SY'$, $\widetilde A$ (resp. $\widetilde B$) le normalisé de $A$ (resp. $B$), $\widetilde A'$ (resp. $\widetilde B'$) le normalisé de $A'$ (resp. $B'$). Comme $\widetilde A'$ (resp. $\widetilde B'$) est aussi le normalisé de $\widetilde A\times_ZZ'$ (resp. $\widetilde B\times_ZZ'$), on a des carrés commutatifs
\[
\begin{tikzcd}[column sep=large,row sep=large]
\widetilde A' \arrow[r] \arrow[d] & Z' & \widetilde B' \arrow[l] \arrow[d]\\
\widetilde A \arrow[r] & Z & \widetilde B . \arrow[l]
\end{tikzcd}
\]
Soient $(x_i)_{1\le i\le m}$ (resp. $(y_j)_{1\le j\le n}$) les points de $\widetilde A$ (resp. $\widetilde B$) au-dessus de $x$ (resp. $y$), $(x'_i)_{1\le i\le m'}$ (resp. $(y'_j)_{1\le j\le n'}$) les points de $\widetilde A'$ (resp. $\widetilde B'$) au-dessus de $x'$ (resp. $y'$). Comme $A$ et $B$ sont strictement locaux, on a des décompositions
\[
\widetilde A=\coprod_{1\le i\le m}A_i,\qquad
\widetilde B=\coprod_{1\le j\le n}B_j,\qquad
\widetilde A'=\coprod_{1\le i\le m'}A'_i,\qquad
\widetilde B'=\coprod_{1\le j\le n'}B'_j,
\]
où $A_i,B_j,A'_i,B'_j$ sont des traits strictement locaux de points fermés respectifs $x_i,y_j,x'_i,y'_j$. Choisissons des uniformisantes $u,v,u',v'$ de $X,Y,X',Y'$ respectivement; notons $\lambda_i$ (resp. $\mu_j$) la pente de $A_i$ (resp. $B_j$) par rapport à $((X,u),(Y,v))$, $\lambda'_i$ (resp. $\mu'_j$) celle de $A'_i$ (resp. $B'_j$) par rapport à $((X',u'),(Y',v'))$. Les hypothèses de position générale sur $A$ et $B$ s'expriment par
\[
\lambda_i\ne0\text{ pour tout }i,\qquad
\mu_j\ne\infty\text{ pour tout }j,\qquad
\lambda_i\ne\mu_j\text{ pour tous }i,j,
\tag{1}
\]
et il s'agit de montrer que l'on a
\[
\lambda'_i\ne0\text{ pour tout }i,\qquad
\mu'_j\ne\infty\text{ pour tout }j,\qquad
\lambda'_i\ne\mu'_j\text{ pour tous }i,j.
\tag{2}
\]
Soient $i'\in[1,m']$, $j'\in[1,n']$, $x_i$ (resp. $y_j$) l'image de $x'_{i'}$ (resp. $y'_{j'}$). Le diagramme ci-dessus fournit des carrés commutatifs de morphismes finis de $S$-traits
\[
\begin{tikzcd}[column sep=large,row sep=large]
X' & A'_{i'} \arrow[l] \arrow[r] \arrow[d] & Y' \arrow[d]\\
X & A_i \arrow[l] \arrow[r] & Y
\end{tikzcd}
\qquad
\begin{tikzcd}[column sep=large,row sep=large]
X' & B'_{j'} \arrow[l] \arrow[r] \arrow[d] & Y' \arrow[d]\\
X & B_j \arrow[l] \arrow[r] & Y .
\end{tikzcd}
\]
D'après 3.5, il existe $C\in k^*$ tel que
\[
\lambda_i=C{\lambda'_{i'}}^d,\qquad
\mu_j=C{\mu'_{j'}}^d.
\]
Donc (1) entraîne (2), ce qui achève la démonstration de 3.3.

\subsection*{4. Fin de la démonstration de 1.2}

\subsubsection*{4.1}
Pour achever la démonstration de 2.5, donc de 1.2, il reste à établir la nullité de 2.6.4 sous les hypothèses additionnelles suivantes: $\Lambda$ est annulé par une puissance d'un nombre premier $\ell\ne p$, $P$ et $Q$ sont bornés, à composantes localement libres de type fini, telles que les représentations correspondantes des groupes fondamentaux se factorisent à travers des $\ell$-groupes finis. On a
\[
g_!(E)|T-(Y\cup A)=g_{1!}(E_1),
\]
où $g_1:T-(X\cup Y\cup A)\hookrightarrow T-(Y\cup A)$ est l'inclusion, et
\[
E_1=(P\lten_{\Lambda}Q)|T-(X\cup Y\cup A).
\]
Or $E_1$ est un complexe à cohomologie bornée, constructible, localement constante, telle que la représentation correspondante de $\pi_1(T-(X\cup Y\cup A))$ se factorise à travers un $\ell$-groupe. L'assertion à démontrer va donc résulter de l'énoncé plus général suivant:

\begin{statement}{Proposition 4.2}
Soient $T$ le localisé strict en un point fermé d'un $S$-schéma lisse de dimension $2$, $t$ le point fermé de $T$, $A$, $B$, $X$ des sous-schémas fermés de dimension $1$ de $T$, $i:T-(A\cup X)\hookrightarrow T-A$ l'inclusion. On suppose que $X$ est régulier, et que $A$ est en position générale par rapport à $B$ et $X$ (i.e. que l'intersection des cônes tangents en $t$ à $A$ et $B\cup X$ est réduite à $t$). Soit d'autre part $E\in\Ob D^b(T-(A\cup X))$, tel que $H^*(E)$ soit constructible, localement constant, et que la représentation correspondante de $\pi_1(T-(A\cup X))$ se factorise à travers un $\ell$-groupe. Alors la flèche de restriction
\[
\mathrm R\Gamma(T-A,i_!(E))
\longrightarrow
\mathrm R\Gamma(B-t,i_!(E)|B-t)
\]
est nulle.
\end{statement}

La démonstration va s'appuyer sur le

\begin{statement}{Lemme 4.3}
Soient $Z$ un schéma noethérien régulier connexe de dimension $2$, de caractéristiques résiduelles premières à $\ell$, $X$, $Y$ des diviseurs réguliers se coupant transversalement en un point $z$, d'où un diagramme d'inclusions
\[
\begin{tikzcd}[column sep=large,row sep=large]
Z-(X\cup Y) \arrow[r,"i'"] \arrow[d,"j"'] & Z-Y \arrow[d,"j"]\\
Z-X \arrow[r,"i"'] \arrow[d,"m"'] & Z \arrow[d,"m"]\\
Y-z \arrow[r,"i_1"'] & Y .
\end{tikzcd}
\]
On fait l'hypothèse de pureté:
\[
(P)\qquad R^qj_*(\mathbb Z/\ell)=0\quad\text{pour }q>1.
\]
Soit d'autre part $L\in\Ob D^b(Z-(X\cup Y),\Lambda)$, à cohomologie constructible localement constante, telle que la représentation correspondante de $\pi_1(Z-(X\cup Y))$ se factorise à travers un $\ell$-groupe. Alors:
\begin{enumerate}[label=\alph*)]
\item On a un isomorphisme canonique
\[
i_{1!}m'^*Rj_*i'_!L\xrightarrow{\sim} m^*Rj_*(i'_!L),
\]
et le complexe $m'^*Rj_*i'_!L$ est à cohomologie constructible, localement constante.
\item Supposons $Y$ connexe, et soit $\bar s$ un point géométrique de $Y$ localisé en un point fermé $s\ne z$. Alors la flèche de restriction
\[
\mathrm R\Gamma(Y,m^*Rj_*(i'_!L))\longrightarrow (Rj_*(i'_!L))_{\bar s}
\tag{4.3.1}
\]
est nulle.
\item Sous les hypothèses de b), soit $C$ un sous-schéma fermé de dimension $1$ de $Z$ tel que $Y\cap C=s$, notons $\widetilde C$ le localisé strict de $C$ en $\bar s$. Alors la flèche de restriction
\[
\mathrm R\Gamma(Z-Y,i'_!L)
\longrightarrow
\mathrm R\Gamma(\widetilde C-\bar s,i'_!(L)|\widetilde C-\bar s)
\tag{4.3.2}
\]
est nulle.
\end{enumerate}
\end{statement}

L'isomorphisme trivial
\[
i^*Rj_*i'_!L\xrightarrow{\sim}Rj_*(i'^*i'_!L)=Rj_*L
\]
donne un isomorphisme
\[
m'^*Rj_*i'_!L\xrightarrow{\sim} i^*m^*Rj_*i'_!L,
\]
d'où, par adjonction, une flèche
\[
i_{1!}m'^*Rj_*i'_!L\longrightarrow m^*Rj_*(i'_!L),
\]
qui induit un isomorphisme hors de $z$. Montrons qu'elle est un isomorphisme, i.e. que, si $\bar z$ est un point géométrique au-dessus de $z$,
\[
(Rj_*(i'_!L))_{\bar z}=0.
\]
On peut pour cela remplacer $\Lambda$ par $\mathbb Z/\ell^\nu$, où $\ell^\nu$ annule $\Lambda$, et l'on se ramène, par dévissage et passage à la limite, au cas où $L$ est un $\mathbb Z/\ell^\nu$-module localement constant de type fini, correspondant à une représentation du $\pi_1$ à travers un $\ell$-groupe. Comme le seul groupe abélien fini de $\ell$-torsion qui est simple sous l'action d'un $\ell$-groupe est $\mathbb Z/\ell$ avec action triviale, on se ramène, par un nouveau dévissage, au cas où $L$ est le faisceau constant $\mathbb Z/\ell$.

Soient $\widetilde Z$ le localisé strict de $Z$ en $\bar z$, $\widetilde X=X\times_Z\widetilde Z$, $\widetilde Y=Y\times_Z\widetilde Z$, $i':\widetilde Z-(\widetilde X\cup\widetilde Y)\hookrightarrow \widetilde Z-\widetilde Y$ l'inclusion. On a
\[
(Rj_*(i'_!(\mathbb Z/\ell)))_{\bar z}
=\mathrm R\Gamma(\widetilde Z-\widetilde Y,i'_!(\mathbb Z/\ell)),
\]
de sorte qu'on est ramené à prouver que la flèche de restriction
\[
H^n(\widetilde Z-\widetilde Y,\mathbb Z/\ell)
\longrightarrow
H^n(\widetilde X-z,\mathbb Z/\ell)
\]
est un isomorphisme pour tout $n$. Or, pour $n\le1$, c'est une conséquence du lemme d'Abhyankar, et pour $n>1$ les deux membres sont nuls par l'hypothèse de pureté. La première assertion de a) est donc établie. Prouvons la seconde. Par dévissage, on peut supposer $L$ concentré en degré $0$. L'hypothèse (P) entraîne, par dévissage, $R^qj_*L=0$ pour $q>2$. Reste à prouver la constructibilité et la locale constance de $m'^*R^qj_*i'_!L$ pour $q\le1$, ce qui est standard. Pour la constructibilité, résolvant $L$ à droite par des faisceaux du type $p_*p^*L$ où $p:Z'\to Z$ est un revêtement fini modérément ramifié le long de $Y$, on se ramène à $L$ constant, de valeur $L_0$: on a alors
\[
m'^*Rj_*i'_!L=(L_0)_{Y-z},
\qquad
m'^*R^1j_*i'_!L=(L_0)_{Y-z}(-1)
\]
par Abhyankar. Pour la locale constance, il s'agit de voir que les flèches de spécialisation sont des isomorphismes. On peut, pour cela, remplacer, comme plus haut, $\Lambda$ par $\mathbb Z/\ell^\nu$, et l'on se ramène par dévissage au cas où $L$ est le faisceau constant $\mathbb Z/\ell$: on conclut à l'aide du lemme d'Abhyankar.

On a donc démontré a). Prouvons b). Soit $\bar\eta$ un point générique géométrique de $Y$. Choisissons des flèches de spécialisation
\[
a:\bar\eta\to Y(\bar z),\qquad b:\bar\eta\to Y(\bar s),
\]
où $Y(\bar z)$ (resp. $Y(\bar s)$) est le localisé strict de $Y$ en $\bar z$ (resp. $\bar s$). Posons
\[
M=m^*Rj_*(i'_!L).
\]
Le carré commutatif
\[
\begin{tikzcd}
Y \arrow[r] & Y(\bar s)\\
Y(\bar z) \arrow[u] \arrow[r,"a"'] & \bar\eta \arrow[u,"b"']
\end{tikzcd}
\]
induit un carré commutatif
\[
\begin{tikzcd}[column sep=large,row sep=large]
\mathrm R\Gamma(Y,M) \arrow[r,"4.3.1"] \arrow[d] & M_{\bar s} \arrow[d,"b^*"]\\
M_{\bar z} \arrow[r,"a^*"'] & M_{\bar\eta}.
\end{tikzcd}
\]
Comme $M|Y-z$ est à cohomologie localement constante, $b^*$ est un isomorphisme. D'autre part, comme $M=i_{1!}i_1^*M$, on a $M_{\bar z}=0$, donc la flèche 4.3.1 est nulle. Reste à prouver c). On a $\widetilde C=C\times_Z\widetilde Z$, où $\widetilde Z$ désigne, comme plus haut, le localisé strict de $Z$ en $\bar z$. Posons $\widetilde Y=Y\times_Z\widetilde Z$. On a alors un diagramme commutatif de flèches de restriction
\[
\begin{tikzcd}[column sep=large,row sep=large]
\mathrm R\Gamma(Z-Y,i'_!L)=\mathrm R\Gamma(Z,Rj_*(i'_!L))
 \arrow[r,"4.3.2"] \arrow[d]
&
\mathrm R\Gamma(\widetilde C-\bar s,i'_!(L)|\widetilde C-\bar s)\\
\mathrm R\Gamma(Y,m^*Rj_*(i'_!L)) \arrow[r,"4.3.1"']
&
(Rj_*(i'_!L))_{\bar s}
  =\mathrm R\Gamma(\widetilde Z-\widetilde Y,(i'_!L)|\widetilde Z-\widetilde Y) \arrow[u]
\end{tikzcd}
\]
La nullité de 4.3.2 découle donc de b), ce qui achève la démonstration de 4.3.

Prouvons 4.2. Soient $f:T'\to T$ l'éclatement de l'idéal maximal de $T$, $D=f^{-1}(z)(\simeq\mathbb P^1_S)$ le diviseur exceptionnel, $A'$, $B'$, $X'$ les transformés purs respectifs de $A$, $B$, $X$. $T'$ est donc régulier, connexe, de dimension $2$, $X'$ est un diviseur régulier coupant transversalement $D$ en un point $z'$, et les hypothèses de position générale sur $A$, $B$, $X$ signifient que $A'$ est disjoint de $B'\cup X'$. Notons $E'$ l'image inverse de $E$ sur $T'-(A'\cup D\cup X')$ et
\[
i':T'-(A'\cup D\cup X')\hookrightarrow T'-(A'\cup D)
\]
l'inclusion. Il s'agit de prouver que la flèche de restriction
\[
\mathrm R\Gamma(T'-(A'\cup D),i'_!(E'))
\longrightarrow
\mathrm R\Gamma(B'\cup D,i'_!(E')|B'-(B'\cup D))
\tag{*}
\]
est nulle. Mais $B'$ est réunion disjointe de schémas strictement locaux $B'_j$ $(1\le j\le n)$, de dimension $1$, tels que, pour chaque $j$, $B'_j\cap D$ soit réduit à un point $b_j$. Appliquant 4.3 c) à $(Z=T'-A', X=X', Y=D-(D\cap A'), L=E', C=B'_j)$, l'hypothèse (P) étant vérifiée en vertu du théorème de pureté relatif (SGA 4 XVI 3) car $(Z,Y)$ est limite projective filtrante de couples lisses de codimension $1$ sur $S$, à morphismes de transition affines étales, on trouve que la flèche de restriction
\[
\mathrm R\Gamma(T'-(A'\cup D),i'_!(E'))
\longrightarrow
\mathrm R\Gamma(B'_j-b_j,i'_!(E')|B'_j-b_j)
\]
est nulle. Donc (*) est nulle, ce qui achève la démonstration de 4.2, donc de 2.5 et 1.2.
