% 04844: \subsection{Structure de la categorie $\mathbf{BIEXT}(P,Q;G)$}\label{2.5}
% 04845: 
% 04846: Les considerations generales formelles de [4, n$^\circ$ 1] permettent de
% 04847: tirer quelques consequences utiles du fait que les categories

% 04850: \subsubsection{}
% 04851: \label{2.5.1}
% 04852: Tout d'abord, pour $G$ fixe, la categorie $\mathbf{BIEXT}(P,Q;G)$* est
% 04853: munie d'une structure de composition, permettant a deux biextensions
% 04854: $E, E'$ de $(P,Q)$ par $G$ de faire correspondre fonctoriellement une
% 04855: biextension produit, notee $E \cdot E'$. On peut considerer que cette
% 04856: construction ne fait que paraphraser la construction bien connue du produit
% 04857: de deux extensions d'un Groupe commutatif par un autre~; on notera de plus
% 04858: qu'elle est bien compatible avec cette derniere notion, quand on interprete
% 04859: $E$ et $E'$, soient comme des extensions de Groupes de $Q_P$ par $G_P$,
% 04860: soit comme des extensions de Groupes de $P_Q$ par $G_Q$. En particulier,
% 04861: $E \cdot E'$ est, en tant que torseur sous $G_{P \times Q}$, le produit des
% 04862: torseurs sous $G_{P \times Q}$ definis par $E$ et $E'$. La formation de
% 04863: $E \cdot E'$ est de plus associative et commutative a isomorphismes
% 04864: canoniques pres, et la loi de composition envisagee dans
% 04865: $\mathbf{BIEXT}(P,Q;G)$ admet un ``objet neutre bilatere'', appele la
% 04866: \emph{biextension triviale} de $(P,Q)$ par $G$. Celle-ci peut etre
% 04867: construite (loc.\ cit.) comme la biextension deduite de l'unique
% 04868: biextension (a isomorphisme pres) $E_0$ de $(P,Q)$ par le Groupe nul $0$,
% 04869: grace a l'homomorphisme $0 \to G$. On verifie aussitot sur cette
% 04870: description que la biextension triviale $E$ peut etre aussi decrite comme
% 04871: le torseur trivial
% 04872: \[
% 04873: E = P \times Q \times G,
% 04874: \]
% 04875: ses deux structures de Groupes sur $P$ resp.\ sur $Q$ etant deduites des
% 04876: structures de Groupes produits de $Q \times G$ resp.\ de $P \times G$ par
% 04877: changement de base $P \to e$ resp.\ $Q \to e$.
% 04878: 
% 04879: Enfin, pour toute biextension $E$ de $(P,Q)$ par $G$, on peut definir la
% 04880: \emph{biextension inverse}, notee $E^{-1}$, munie d'un isomorphisme de
% 04881: $E \cdot E^{-1}$ avec la biextension triviale, ce qui determine $E^{-1}$ a
% 04882: isomorphisme unique pres, et permet de definir la dependance fonctorielle
% 04883: par rapport a $E$.

% 04886: d'un objet-groupe de la categorie $(\Cat)$ des categories~; a cela pres que
% 04887: les associativites, commutativites, relations d'unite et d'inverses ne sont
% 04888: valables que modulo des isomorphismes donnes pres. On pourra dire que cette
% 04889: categorie est un \emph{pseudo-groupe} (de la 2-categorie $(\Cat)$).

% 04905: \begin{equation}\label{eq:2.5.2.2}
% 04906: \mathrm{Biext}^1(P,Q;G)
% 04907: \end{equation}
% 04908: des classes d'isomorphismes de biextensions de $(P,Q)$ by $G$ une loi de
% 04909: groupe commutatif, l'inverse de la classe de $E$ etant celle du bitorseur
% 04910: inverse $E^{-1}$.

% 05002: \subsection{Biextension symetrique d'une biextension donnee}\label{2.7}
% 05003: 
% 05004: Si $E$ est une biextension de $(P,Q)$ by $G$, il y a lieu de definir la

% 05012: et gauche de $E$. Il est evident que l'operation $E \mapsto {}^s\!E$ est un

% 05025: \subsection{Changement de topos. Localisation}\label{2.8}

% 05028: \[

% 05030: \]

% 05032: $(P,Q)$ by $G$ en une biextension $f^*(E)$ de $(f^*(P), f^*(Q))$ par
% 05033: $f^*(G)$, comme chaque fois qu'on a une espece de structure qui peut
% 05034: s'expliciter ``en termes de limites projectives finies et de limites
% 05035: inductives quelconques''. Pour montrer la compatibilite des deux
% 05036: structures d'extensions sur $f^*(E)$ deduites de celles de $E$, on
% 05037: exprimera cette compatibilite par la commutativite du diagramme
% 05038: (\ref{eq:2.1.1}) dans le cas ``universel'', i.e.\ pour
% 05039: $S = P \times P \times Q \times Q$, $p, p', q, q'$ etant les quatre
% 05040: projections de $S$ sur ces quatre facteurs. Comme de juste, on obtiendra
% 05041: par $E \mapsto f^*(E)$ un foncteur canonique
% 05042: \[
% 05043: f^* : \mathbf{BIEXT}(\mathcal{T}) \longrightarrow \mathbf{BIEXT}(\mathcal{T}').
% 05044: \]
% 05045: 
% 05046: Nous utiliserons par la suite sans autre mention que toutes les
% 05047: constructions que nous pourrons etre amenes a effectuer au moyen de
% 05048: biextensions sont compatibles aux images inverses par morphismes de topos.
% 05049: Il en est en particulier ainsi des operations de produit $E \cdot E'$ et
% 05050: de passage a l'inverse $E^{-1}$, considerees dans \ref{2.5}.
% 05051: 
% 05052: On pourra prendre en particulier pour $\mathcal{T}'$ un topos induit
% 05053: $\mathcal{T}_{/S'}$, et pour $f$ le morphisme canonique (dit ``de
% 05054: localisation'' (SGA 4 IV 5))
% 05055: \[
% 05056: f_{S'} : \mathcal{T}_{/S'} \longrightarrow \mathcal{T}.
% 05057: \]
% 05058: On appelle alors $f^*(E)$ la biextension induite par $E$ sur $S'$, et on
% 05059: la notera aussi, comme d'habitude, $E|S'$ ou $E_{S'}$. Comme l'operation
% 05060: de restriction possede evidemment les proprietes habituelles de
% 05061: transitivite, on en conclut que les categories
% 05062: $\mathbf{BIEXT}(\mathcal{T}_{/S})$, pour un objet variable $S$ de
% 05063: $\mathcal{T}$, forment les fibres d'une categorie fibree sur $\mathcal{T}$

% 05098: $\mathrm{inj}_i$ du $i$-eme facteur dans ledit produit. Si on se donne un

% 05103: \begin{array}{c}
% 05104: P \times Q \times Q, \quad P \times Q \times Q \times Q, \\

% 05106: P \times P \times Q \times Q.

% 05110: \begin{proposition}\label{prop:2.9.3}
% 05111: Supposons que tout morphisme d'un des produits (\ref{eq:2.9.2}) dans $G$
% 05112: qui est trivial sur chacun des facteurs de ce produit (plonges dans ledit
% 05113: grace aux sections unite comme on vient de l'expliquer) soit trivial.

% 05115: 
% 05116: \textup{(a)} Le foncteur canonique (\ref{eq:2.9.1}) est pleinement
% 05117: fidele, et les deux membres de (\ref{eq:2.9.1}) sont des categories
% 05118: rigides. En particulier, si $E$ est un torseur sous $G_{P \times Q}$
% 05119: birigidifie pour $(e_P, e_Q)$, il existe sur $E$ au plus une structure de
% 05120: biextension compatible avec sa structure de torseur birigidifie.
% 05121: 
% 05122: \textup{(b)} Pour qu'il existe une telle structure de biextension sur
% 05123: $E$, il faut et il suffit qu'il existe sur $E$ une structure d'extension
% 05124: de $Q_P$ by $G_P$ compatible avec sa structure de torseur sous $G_P$, et
% 05125: qu'il existe sur $E$ une structure d'extension de $P_Q$ by $G_Q$
% 05126: compatible avec sa structure de torseur sous $G_Q$ (cf.\ \ref{prop:1.3.5}~(b)).

% 05128: $Q_P$ by $G_P$ compatible avec les structures de torseur sous $G$ et la

% 05130: d'extension de $P_Q$ by $G_Q$ compatible avec la structure de torseur
% 05131: sous $G$ et la rigidification donnee sur $e_P \times Q$, et ces deux
% 05132: structures d'extension sont compatibles (\ref{def:2.1}).

% 05135: \noindent\textit{Demonstration.} (a) Si $E$ est un torseur birigidifie sur
% 05136: $P \times Q$, le groupe de ses automorphismes s'identifie au groupe des
% 05137: morphismes $P \times Q \to G$ qui sont triviaux sur les deux facteurs
% 05138: partiels, donc d'apres l'hypothese appliquee au produit $P \times Q$,
% 05139: c'est l'automorphisme identite. Cela prouve l'assertion de rigidite. Pour
% 05140: montrer que si $E, E'$ sont des biextensions de $(P,Q)$ by $G$, tout
% 05141: morphisme de bitorseurs birigidifies $f : E \to E'$ est un morphisme de
% 05142: biextensions, il suffit d'appliquer \ref{prop:1.3.5}~(a) aux deux
% 05143: structures d'extension definies par $E, E'$ sur $P$ resp.\ sur $Q$ (ce qui
% 05144: amene a utiliser l'hypothese pour les deux produits
% 05145: $P \times Q \times Q$ et $P \times P \times Q$ de (\ref{eq:2.9.2})).
% 05146: 
% 05147: (b) La necessite de la condition enoncee sur le torseur birigidifie $E$
% 05148: pour qu'il provienne d'une biextension etant claire, prouvons la
% 05149: suffisance. Il resulte d'abord de \ref{prop:1.3.5}~(b) que la condition
% 05150: enoncee implique l'existence de structures d'extensions uniques de
% 05151: $Q_P$ by $G_P$ et de $P_Q$ by $G_Q$, compatibles avec la structure de
% 05152: torseur et les rigidifications. (On utilise l'hypothese pour les produits

% 05155: d'extension sont compatibles, i.e.\ qu'on a commutativite dans le diagramme
% 05156: (\ref{eq:2.1.1}). Or les deux composes de ce diagramme sont des

% 05158: different par un morphisme $u : P \times P \times Q \times Q \to G$.

% 05395: que $\alpha' = u\alpha$~; on compose les homomorphismes en composant les
% 05396: homomorphismes d'extensions correspondants. On trouve ainsi la categorie
% 05397: annoncee $\mathcal{F}_{L_\bullet}$. Elle ne depend, a isomorphisme
% 05398: canonique pres, que du tronque a l'ordre $1$ $[L_\bullet]$ de $L_\bullet$.
% 05399: 
% 05400: On a un foncteur evident, $(X,\alpha) \mapsto A$~:
% 05401: \begin{equation}\label{eq:3.1.4}
% 05402: \mathcal{F}_{L_\bullet} \longrightarrow \mathcal{C}.
% 05403: \end{equation}
% 05404: Ce foncteur est cofibrant. En effet, soit $u : A \to B$ un morphisme de
% 05405: $\mathcal{C}$, et $(X,\alpha)$ un objet de la categorie
% 05406: $\mathcal{F}_{L_\bullet}(A)$ fibre en $A$. Donc $X$ est une extension de
% 05407: $L_0$ par $A$, et on peut considerer l'extension $u_*(X)$ de $L_0$ par
% 05408: $B$, image directe de $X$ par $u$, et le munir de la trivialisation
% 05409: partielle composee $\tilde{\alpha} = f\alpha$, ou $f : X \to u_*(X)$ est
% 05410: le morphisme canonique. On verifie alors immediatement que le morphisme
% 05411: $f : (X,\alpha) \to (u_*(X), \tilde{\alpha})$ est un morphisme cocartesien
% 05412: pour le foncteur (\ref{eq:3.1.4}), i.e.\ satisfait a la propriete
% 05413: universelle habituelle pour les $u$-morphismes dans
% 05414: $\mathcal{F}_{L_\bullet}$ de source $(X,\alpha)$~; il suffit pour ceci
% 05415: d'utiliser la propriete universelle analogue pour le morphisme
% 05416: $f : X \to u_*(X)$, qui est bien connue et triviale.
% 05417: 
% 05418: Enfin, la categorie cofibree $\mathcal{F}_{L_\bullet}$ est additive,
% 05419: i.e.\ elle satisfait les deux conditions~:
% 05420: \begin{equation}\label{eq:3.1.5}
% 05421: \mathcal{F}_{L_\bullet}(0) \text{ est equivalente a la categorie
% 05422: ponctuelle,}
% 05423: \end{equation}
% 05424: \begin{equation}\label{eq:3.1.6}
% 05425: \mathcal{F}_{L_\bullet}(A \times B) \longrightarrow
% 05426: \mathcal{F}_{L_\bullet}(A) \times \mathcal{F}_{L_\bullet}(B) \text{ est une
% 05427: equivalence,}
% 05428: \end{equation}
% 05429: pour deux objets quelconques $A$ et $B$ de $\mathcal{C}$. La verification
% 05430: est triviale, en termes des proprietes correspondantes pour la categorie
% 05431: cofibree des extensions de $L_0$ par des objets variables de $\mathcal{C}$.
% 05432: Nous utiliserons librement les consequences de l'additivite de
% 05433: $\mathcal{F}_{L_\bullet}$, etablies dans (loc.\ cit.\ 1) et rappelees
% 05434: plus haut (\ref{2.5}).
% 05435: 
% 05436: \subsubsection{}\label{3.1.7}
% 05437: Parmi ces consequences, mentionnons le fait que pour tout objet $A$ de
% 05438: $\mathcal{C}$, il y a dans $\mathcal{F}_{L_\bullet}(A)$ un objet nul, ou
% 05439: trivial (\ref{2.5.1})~; on verifie qu'il est defini par l'extension
% 05440: triviale $X = L_0 \oplus A$ de $L_0$ par $A$, munie de la trivialisation
% 05441: partielle definie par $\alpha : L_1 \to X$ de composantes $d_1$ et $0$.
% 05442: D'autre part (\ref{2.5.2}), pour tout objet $(X,\alpha)$ de
% 05443: $\mathcal{F}_{L_\bullet}(A)$, le groupe des automorphismes de cet objet
% 05444: est canoniquement isomorphe au groupe des automorphismes de l'objet
% 05445: trivial de $\mathcal{F}_{L_\bullet}(A)$, i.e.\ a
% 05446: $\mathcal{F}_{L_\bullet}^0(A)$, qui est un groupe commutatif. Bien
% 05447: entendu, on retrouve trivialement ici ce fait sur la description explicite
% 05448: de $\mathcal{F}_{L_\bullet}(A)$, et on trouve meme un isomorphisme
% 05449: canonique
% 05450: \begin{equation}\label{eq:3.1.8}
% 05451: \mathcal{F}_{L_\bullet}^0(A) \simeq
% 05452: \Ker(\Hom(L_0,A) \to \Hom(L_1,A)) \simeq \Hom(\H_0(L_\bullet), A),
% 05453: \end{equation}
% 05454: induit par l'isomorphisme habituel du groupe des automorphismes
% 05455: d'extension de $X$ avec le groupe $\Hom(L_0,A)$. L'isomorphisme
% 05456: (\ref{eq:3.1.8}) est evidemment fonctoriel en $A$.
% 05457: 
% 05458: \subsubsection{}\label{3.1.9}
% 05459: Rappelons enfin (\ref{2.5.2}) que grace a la structure multiplicative
% 05460: naturelle dans la categorie fibre $\mathcal{F}_{L_\bullet}(A)$, l'ensemble
% 05461: des classes d'isomorphie de celle-ci est munie de facon naturelle d'une
% 05462: structure de groupe, que nous noterons $\mathcal{F}_{L_\bullet}^1(A)$, et
% 05463: que ce dernier est un foncteur additif par rapport a $A$.
% 05464: 
% 05465: \subsection{Determination de $\mathcal{F}_{L_\bullet}^1(A)$}\label{3.2}
% 05466: 
% 05467: Considerons un objet $(X,\alpha)$ (\ref{eq:3.1.3}) de
% 05468: $\mathcal{F}_{L_\bullet}(A)$. Considerons le complexe de cochaines de
% 05469: longueur 1
% 05470: \begin{equation}\label{eq:3.2.1}
% 05471: [K^\bullet] : 0 \longrightarrow X \xrightarrow{j} L_0 \longrightarrow
% 05472: 0 \cdots
% 05473: \end{equation}
% 05474: defini par le morphisme $j : X \to L_0$, la composante de degre $0$
% 05475: etant $X$. C'est une resolution de $A$, donc peut-etre identifie a $A$ en
% 05476: tant qu'objet de la categorie derivee $\mathrm{D}(\mathcal{C})$ [5]. Ceci
% 05477: pose, le diagramme (\ref{eq:3.1.3}) nous fournit un homomorphisme de
% 05478: degre 1 de complexes
% 05479: \begin{equation}\label{eq:3.2.2}
% 05480: L_\bullet \longrightarrow [K^\bullet]
% 05481: \end{equation}
% 05482: explicite dans le diagramme a carres anticommutatifs
% 05483: \begin{equation}\label{eq:3.2.3}
% 05484: \xymatrix{
% 05485: L_1 \ar[r]^{d_1} \ar[d]_{\alpha} & L_0 \ar@{=}[d] \\
% 05486: X \ar[r]^{j} & L_0 \ar[r] & 0
% 05487: }
% 05488: \end{equation}
% 05489: (On utilisera la convention habituelle, associant au complexe de chaines
% 05490: $L_\bullet$ un complexe a operateur differentiel de degre $1$, dont la
% 05491: composante de degre $i$ est $L_{-i}$). On trouve donc ainsi un
% 05492: homomorphisme de degre $1$ dans la categorie derivee
% 05493: \begin{equation}\label{eq:3.2.3}
% 05494: c(X,\alpha) : L_\bullet \longrightarrow A \quad \text{(hom.\ de degre 1
% 05495:  dans $\mathrm{D}(\mathcal{C})$)}
% 05496: \end{equation}
% 05497: qui ne depend manifestement que de la classe a isomorphie pres de
% 05498: $(X,\alpha)$. On a defini ainsi une application canonique
% 05499: \begin{equation}\label{eq:3.2.4}
% 05500: \mathcal{F}_{L_\bullet}^1(A) \longrightarrow \Ext^1(L_\bullet, A) \overset{\mathrm{dfn}}{=} \Hom_{\mathrm{D}(\mathcal{C})}(L_\bullet, A[1]).
% 05501: \end{equation}
% 05502: 
% 05503: \begin{theorem}\label{thm:3.2.5}
% 05504: L'application (\ref{eq:3.2.4}) est un isomorphisme de groupes, fonctoriel
% 05505: en l'objet $A$ de $\mathcal{C}$.
% 05506: \end{theorem}
% 05507: 
% 05508: Nous laisserons au lecteur le soin d'etablir l'additivite de
% 05509: l'application (\ref{eq:3.2.4}), et sa fonctorialite pour $A$, et nous
% 05510: bornerons a prouver qu'elle est bijective. Pour l'injectivite, montrons
% 05511: d'abord que si l'homomorphisme de complexes $u$ de (\ref{eq:3.2.3}) est
% 05512: nul dans $\mathrm{D}(\mathcal{C})$, alors il est deja nul dans la
% 05513: categorie $\mathrm{K}(\mathcal{C})$ des complexes a homotopie pres, i.e.\ il existe un morphisme $s : L_0 \to X$ tel que l'on ait
% 05514: \begin{equation}\label{eq:3.2.5.1}
% 05515: j \circ s = \id_{L_0}, \quad s \circ d_1 = \alpha.
% 05516: \end{equation}
% 05517: En effet, l'hypothese signifie que la conclusion devient vraie apres
% 05518: avoir compose le morphisme de complexes envisage avec un
% 05519: quasi-isomorphisme convenable $v$
% 05520: \[
% 05521: \xymatrix{
% 05522: 0 \ar[r] & X \ar[r]^{j} & L_0 \ar[r] & 0 \\
% 05523: & & C^\bullet \ar[u]^{v}
% 05524: }
% 05525: \]
% 05526: ou on peut d'ailleurs (quitte a remplacer $C^i$ par $0$ pour $i > 2$,
% 05527: par $\Ker(C^1 \to C^2)$ pour $i = 1$) supposer que $C^i = 0$ pour
% 05528: $i \geq 2$. Mais alors, le morphisme $(*)$ induisant l'identite sur $A$,
% 05529: on voit qu'il identifie $X$ au produit fibre de $L_0$ et de $C^0$ sur
% 05530: $C^1$, et que la donnee d'un morphisme $s' : L_1 \to C^0$ definissant une
% 05531: homotopie $vu \sim 0$, donc satisfaisant $j's' = vu$, se factorise par
% 05532: un morphisme $s : L_1 \to X$ satisfaisant $js = u$ et $sd_1 = \alpha$. Mais
% 05533: alors $s$ definit un splittage de l'extension $X$ de $L_0$ par $A$, donc
% 05534: le complexe $X \to L_0$ est isomorphe dans $\mathrm{K}(\mathcal{C})$ au
% 05535: complexe $K'^\bullet$ reduit a $A$. Or pour ce dernier, il est evident que
% 05536: l'homomorphisme
% 05537: \[
% 05538: \Hom_{\mathrm{D}(\mathcal{C})}(L_\bullet, K'^\bullet) \longrightarrow
% 05539: \Hom_{\mathrm{D}(\mathcal{C})}(L_\bullet, K'^\bullet)
% 05540: \]
% 05541: est un isomorphisme, d'ou notre assertion.
% 05542: 
% 05543: Ceci dit, la donnee d'un homomorphisme $j : L_1 \to X$ satisfaisant les
% 05544: conditions (\ref{eq:3.2.5.1}) equivaut precisement a une trivialisation
% 05545: de l'extension $X$ de $L_0$ par $A$, compatible avec la trivialisation
% 05546: partielle $\alpha$. Il en resulte bien l'injectivite de (\ref{eq:3.2.4}).
% 05547: 
% 05548: Pour la surjectivite, partons d'un element du deuxieme membre, qui peut
% 05549: donc etre decrit, par definition, a l'aide d'un homomorphisme de degre
% 05550: $1$ de complexes
% 05551: \begin{equation}\label{eq:3.2.5.2}
% 05552: \xymatrix{
% 05553: L_1 \ar[r]^{d_1} \ar[d]_{u^1} & L_0 \ar[d]^{u^0} \\
% 05554: C^0 \ar[r]_{d} & C^1 \ar[r] & \cdots
% 05555: }
% 05556: \quad \text{(carres anticommutatifs)},
% 05557: \end{equation}
% 05558: ou $C^\bullet$ est une resolution de $A$, et ou on peut supposer comme
% 05559: ci-dessus que $C^i = 0$ pour $i \geq 2$. Ainsi $C^0$ apparait comme une
% 05560: extension de $C^1$ par $A$, et on peut considerer l'extension $X$ image
% 05561: inverse de celle-ci par $L_0 \to C^1$. L'anticommutativite du carre de
% 05562: (\ref{eq:3.2.5.2}) montre que $-u^1$ se factorise par un morphisme
% 05563: $\alpha : L_1 \to X$, qui est un splittage partiel i.e.\ satisfait
% 05564: $j\alpha = d_1$ ($j$ defini dans (\ref{eq:3.1.2})). D'ailleurs la
% 05565: relation $u^1 d_2 = 0$ dans (\ref{eq:3.2.5.2}) implique que
% 05566: $\alpha d_2 = 0$. Donc $(X,\alpha)$ est bien un element de
% 05567: $\mathcal{F}_{L_\bullet}(A)$. De plus, par construction, (\ref{eq:3.2.5.2})
% 05568: est le compose de l'homomorphisme (\ref{eq:3.2.3}) relatif a
% 05569: $(X,\alpha)$, et de l'homomorphisme
% 05570: \[
% 05571: [K^\bullet] \longrightarrow [C^\bullet]
% 05572: \]
% 05573: de composantes $(f, u^0)$, ou $f : X \to C^0$ est l'homomorphisme
% 05574: canonique deduit de la definition de l'extension $X$ comme image inverse
% 05575: de l'extension $C^0$). Comme l'homomorphisme precedent est un
% 05576: homomorphisme de resolutions de $A$ (induisant l'identite sur $A$), on en
% 05577: conclut, par definition de $\Ext^1 = \Hom_{\mathrm{D}}$, que l'element
% 05578: donne de $\Ext^1(L_\bullet, A)$, decrit par (\ref{eq:3.2.5.2}), n'est
% 05579: autre que $c(X,\alpha)$, ce qui etablit la surjectivite de
% 05580: (\ref{eq:3.2.4}) et acheve la demonstration de \ref{thm:3.2.5}.
% 05581: 
% 05582: \subsection{Proprietes de la categorie cofibree $\mathcal{F}_{L_\bullet}$}\label{3.3}
% 05583: 
% 05584: Dans le present numero, qui n'est pas essentiel pour la comprehension de
% 05585: la suite, nous utilisons librement les notions de [4], notre but etant de
% 05586: comparer les resultats precedents a ceux de loc.\ cit.
% 05587: 
% 05588: \begin{theorem}\label{thm:3.3.1}
% 05589: La categorie cofibree additive $\mathcal{F}_{L_\bullet}$ sur $\mathcal{C}$
% 05590: est exacte a gauche (loc.\ cit.\ 2.2), elle admet un objet quasi-maximal
% 05591: (loc.\ cit.\ 4.4), et pour tout objet $E = (X,\alpha)$ de cette categorie,
% 05592: il existe un complexe typique $L_\bullet^E$ pour $E$ (loc.\ cit.\ 3.1).
% 05593: De plus, le procomplexe typique de $\mathcal{F}_{L_\bullet}$ (loc.\ cit.\ 3.7
% 05594: et 4.4) est canoniquement isomorphe dans $\mathrm{D}(\mathcal{C})$ au
% 05595: complexe tronque $[L_\bullet]$ envisage dans \ref{3.1}.
% 05596: \end{theorem}
% 05597: 
% 05598: Nous explicitons ci-dessous la signification des diverses assertions de
% 05599: \ref{thm:3.3.1}.
% 05600: 
% 05601: \subsubsection{}\label{3.3.2}
% 05602: Dire la categorie cofibree $\mathcal{F}$ est exacte a gauche signifie
% 05603: qu'elle est additive et satisfait de plus la condition~:
% 05604: \begin{equation}\label{eq:3.3.2.1}
% 05605: \text{Pour toute suite exacte } 0 \to A' \xrightarrow{v} A \xrightarrow{w}
% 05606: A'' \to 0 \text{ de } \mathcal{C}, \text{ le foncteur naturel}
% 05607: \end{equation}
% 05608: \[
% 05609: \mathcal{F}(A') \longrightarrow \mathcal{F}([A \xrightarrow{w} A''])
% 05610: \]
% 05611: est une equivalence de categories.
% 05612: 
% 05613: Rappelons que, pour une categorie cofibree $\mathcal{F}$ sur $\mathcal{C}$,
% 05614: et un morphisme $v : A \to A''$ de $\mathcal{C}$,
% 05615: $\mathcal{F}([A \to A''])$ designe la categorie formee des couples
% 05616: $(E, p)$, ou $E$ est un objet de $\mathcal{F}(A)$ et ou $p$ est une
% 05617: ``trivialisation'' de l'objet correspondant $v_*(E)$ de
% 05618: $\mathcal{F}(A'')$, i.e.\ un isomorphisme de cet objet avec l'objet nul
% 05619: (\ref{2.5.1}) de $\mathcal{F}(A'')$. La definition du foncteur
% 05620: $\mathcal{F}(A') \to \mathcal{F}([A \to A''])$ de
% 05621: (\ref{eq:3.3.2.1}) est evidente. Dans le cas
% 05622: $\mathcal{F} = \mathcal{F}_{L_\bullet}$, les objets de
% 05623: $\mathcal{F}_{L_\bullet}([A \to A''])$ s'explicitent comme les triples
% 05624: $(X, \alpha, p)$, ou $(X,\alpha)$ est comme dans (\ref{eq:3.1.3}), et ou
% 05625: $p : X \to A''$ est un morphisme dans $\mathcal{C}$ satisfaisant
% 05626: \[
% 05627: p|_A = v, \quad p\alpha = 0.
% 05628: \]
% 05629: Lorsque $v : A \to A''$ s'insere dans une suite exacte courte comme dans
% 05630: (\ref{eq:3.3.2.1}), alors la donnee d'un morphisme satisfaisant la
% 05631: premiere des relations $(*)$ equivaut, comme il est bien connu
% 05632: (exactitude a gauche de la categorie cofibree des extensions de $L_0$ par
% 05633: des objets variables de $\mathcal{C}$) a la donnee d'une extension $X'$ de
% 05634: $L_0$ par $A' = \Ker v$ qui soit une sous-extension de $X$ (on prend $X' =
% 05635: \Ker p$), et la deuxieme relation $(*)$ exprime que $\alpha$ se factorise
% 05636: en une $\alpha' : L_1 \to X'$. Ceci etablit, modulo des compatibilites
% 05637: dont la verification est evidente, l'exactitude a gauche explicitee dans
% 05638: (\ref{eq:3.3.2.1}).
% 05639: 
% 05640: \subsubsection{}\label{3.3.3}
% 05641: Rappelons qu'un objet $E_0$ d'une categorie cofibree $\mathcal{F}$ sur
% 05642: $\mathcal{C}$ est dit \emph{quasi-maximal} si pour tout objet $E$ de
% 05643: $\mathcal{F}$, au-dessus d'un objet $A$ de $\mathcal{C}$, on peut trouver
% 05644: un morphisme $E \to E'$ dans $\mathcal{F}$, induisant un monomorphisme
% 05645: $A \to A'$ dans $\mathcal{C}$, de telle sorte que $E$ soit majore par
% 05646: $E'$ i.e.\ qu'il existe un morphisme $E_0 \to E'$. Lorsque dans
% 05647: $\mathcal{C}$ il existe suffisamment d'objets injectifs, il revient au
% 05648: meme de dire que tout objet de $\mathcal{F}$ au-dessus d'un objet injectif
% 05649: de $\mathcal{C}$ est majore par $E_0$. Dans le cas
% 05650: $\mathcal{F} = \mathcal{F}_{L_\bullet}$, on a l'objet quasi-maximal
% 05651: canonique $E_0$ suivant. On pose
% 05652: \[
% 05653: L_1^1 = \Coker(d_2 : L_2 \to L_1)
% 05654: \]
% 05655: et on designe par $E_0 = (X_0, \alpha_0)$ l'objet de
% 05656: $\mathcal{F}_{L_\bullet}(L_1^1)$, ou $X_0 = L_0 \oplus L_1^1$ est
% 05657: l'extension triviale de $L_0$ par $L_1^1$, et $\alpha_0 : L_1 \to X_0$ a
% 05658: comme composantes $d_1$ et la surjection canonique $L_1 \to L_1^1$ (deduit
% 05659: de $d_1 : L_1 \to L_0$ par passage au quotient).
% 05660: 
% 05661: Pour voir que $E_0$ est bien quasi-maximal, notons que pour tout objet
% 05662: $(X,\alpha)$ de $\mathcal{F}_{L_\bullet}(A)$, on peut trouver un morphisme
% 05663: de celui-ci dans un $(X'',\alpha'')$, induisant un monomorphisme
% 05664: $A \to A''$ dans $\mathcal{C}$, tel que l'extension $X''$ de $L_0$ par $A''$
% 05665: soit scindee~: il suffit en effet de prendre $A'' = X$ et
% 05666: $(X'',\alpha'') = i_*(X,\alpha)$, les notations etant celles de
% 05667: (\ref{eq:3.1.2}). D'autre part, il est immediat qu'un objet
% 05668: $(X',\alpha')$ de $\mathcal{F}_{L_\bullet}$ tel que l'extension $X'$ soit
% 05669: scindee, est majore par $E_0$.
% 05670: 
% 05671: \subsubsection{}\label{3.3.4}
% 05672: Soit $E$ un objet d'une categorie cofibree additive $\mathcal{F}$ sur
% 05673: $\mathcal{C}$, au-dessus d'un objet $A$ de $\mathcal{C}$, et considerons le
% 05674: foncteur en l'objet variable $B$ de $\mathcal{C}$
% 05675: \begin{equation}\label{eq:3.3.4.1}
% 05676: B \longmapsto \Hom(E, 0_B)
% 05677: \end{equation}
% 05678: ou $0_B$ designe l'objet trivial de $\mathcal{F}(B)$. On dit que $E$
% 05679: admet un \emph{complexe typique} si le foncteur precedent est
% 05680: representable par un objet $\mathcal{C}$ de $\mathcal{C}$. Comme le foncteur
% 05681: envisage s'envoie canoniquement dans le foncteur $B \mapsto \Hom(A,B)$
% 05682: represente par $A$, on conclut alors un homomorphisme canonique
% 05683: \begin{equation}\label{eq:3.3.4.2}
% 05684: A \longrightarrow \mathcal{C}
% 05685: \end{equation}
% 05686: et le complexe de chaines correspondant $A \to \mathcal{C}$ de
% 05687: $\mathcal{C}$ de longueur $1$ est appele le \emph{complexe typique} de
% 05688: $E$.
% 05689: 
% 05690: Dans le cas $\mathcal{F} = \mathcal{F}_{L_\bullet}$, $E = (X,\alpha)$, on
% 05691: verifie immediatement que le foncteur (\ref{eq:3.3.4.1}) est
% 05692: representable par $\Coker \alpha$, l'homomorphisme canonique
% 05693: (\ref{eq:3.3.4.2}) n'etant autre que le compose $A \xrightarrow{i} X \to
% 05694: \Coker \alpha$.
% 05695: 
% 05696: \subsubsection{}\label{3.3.5}
% 05697: Lorsque une categorie cofibree additive $\mathcal{F}$ sur $\mathcal{C}$
% 05698: admet un objet quasi-maximal (\ref{3.3.3}), et que les complexes
% 05699: typiques de ses objets existent, on prouve que le complexe typique
% 05700: $T_\bullet$ d'un objet quasi-maximal $E_0$ de $\mathcal{F}$, considere
% 05701: comme un objet de la categorie derivee $\mathrm{D}(\mathcal{C})$, ne depend
% 05702: pas, a isomorphisme canonique pres, du choix de $E_0$ (loc.\ cit.\ 4.4).
% 05703: On l'appelle le \emph{complexe typique} de la categorie cofibree additive
% 05704: $\mathcal{F}$. Lorsqu'on suppose enfin que $\mathcal{F}$ est de plus
% 05705: exacte a gauche (\ref{3.3.2}), on montre que $\mathcal{F}$ se
% 05706: reconstitue, a equivalence de categories fibrees sur $\mathcal{C}$ pres
% 05707: (unique a isomorphisme unique pres), en termes du complexe typique
% 05708: $T_\bullet$ (loc.\ cit.\ 6.10), et qu'on a des isomorphismes canoniques
% 05709: fonctoriels en l'objet $A$ de $\mathcal{C}$ (loc.\ cit.\ 5.4)
% 05710: \begin{equation}\label{eq:3.3.5.1}
% 05711: \mathcal{F}^0(A) \simeq \Ext^0(T_\bullet, A), \quad
% 05712: \mathcal{F}^1(A) \simeq \Ext^1(T_\bullet, A).
% 05713: \end{equation}
% 05714: 
% 05715: Pour achever de faire le lien avec loc.\ cit., nous allons prouver que
% 05716: dans le cas $\mathcal{F} = \mathcal{F}_{L_\bullet}$, le complexe typique
% 05717: de $\mathcal{F}_{L_\bullet}$ est le tronque $[L_\bullet]$. Cela redonnera
% 05718: une definition des isomorphismes (\ref{eq:3.1.8}) et (\ref{eq:3.2.4}) en
% 05719: termes de (\ref{eq:3.3.5.1}), dont on verifie qu'elle est compatible avec
% 05720: celle donnee dans \ref{3.1} et dans \ref{3.2}.
% 05721: 
% 05722: Pour prouver notre assertion, on peut evidemment se borner au cas ou
% 05723: $L_\bullet$ est deja tronque a l'ordre $1$ (car on a signale dans \ref{3.1}
% 05724: que le tronquage ne modifie pas, a isomorphisme de categories cofibrees
% 05725: pres, la categorie $\mathcal{F}_{L_\bullet}$). Mais alors il est evident

% 05998: \textbf{3.5. Une r\'{e}solution partielle plate canonique pour un $\bZ$-Module quelconque}
% 05999: 
% 06000: Nous allons appliquer les r\'{e}sultats pr\'{e}c\'{e}dents au cas o\`{u} $C$ est la cat\'{e}gorie des Groupes commutatifs du topos $T$ i.e.\ des $\bZ$-Modules de $T$. Soit $P$ un objet de $C$, i.e.\ un groupe commutatif du topos $T$. Nous allons d\'{e}finir un complexe de cha\^{i}nes $L_{\cdot}(P)$ augment\'{e} vers $P$, d\'{e}pendant fonctoriellement de $P$ :
% 06001: 
% 06002: %=====================================================
% 06003: % PAGE 204
% 06004: %=====================================================
% 06005: \begin{equation}\tag{3.5.1}
% 06006: 0 \longrightarrow \bZ[PxPxP] \xrightarrow{d_{2}} \bZ[PxP] \xrightarrow{d_{1}} \bZ[P] \xrightarrow{\epsilon} P \longrightarrow 0
% 06007: \end{equation}
% 06008: satisfaisant $L_{i}(P) = 0$ pour $i \geq 3$. Les composantes de ce complexe sont explicit\'{e}es dans (3.5.1) ci-dessus, en utilisant les notations de 1.4. Pour expliciter les op\'{e}rateurs diff\'{e}rentiels $d_{1}, d_{2}$ et l'augmentation $\epsilon$, on utilise la notation $[p]$ pour d\'{e}signer le point de $\bZ[P](S)$ d\'{e}fini par le point $p \in P(S)$ de $P$, et de m\^{e}me pour les notations $[p,q], [p,q,r]$, lorsque $p,q,r \in P(S)$, pour d\'{e}signer des \'{e}l\'{e}ments de $\bZ[P \times P](S)$ resp.\ de $\bZ[P \times P \times P](S)$. Avec ces notations, on d\'{e}finira $\epsilon$, $d_{1}$ et $d_{2}$ par les formules
% 06009: \begin{equation}\tag{3.5.2}
% 06010: \begin{aligned}
% 06011: \epsilon[p] &= p \\
% 06012: d_{1}[p,q] &= [p+q] - [p] - [q] \\
% 06013: d_{2}[p,q,r] &= [p+q,r] - [p,q+r] + [p,r] - [q,r]
% 06014: \end{aligned}
% 06015: \end{equation}
% 06016: 
% 06017: On v\'{e}rifie facilement que les homomorphismes de Groupes commutatifs ainsi d\'{e}finis d\'{e}finissent effectivement un complexe $L_{\cdot}(P)$ augment\'{e} vers $P$, i.e.\ qu'on a les relations
% 06018: \[ \epsilon d_{1} = 0, \quad d_{1}d_{2} = 0. \]
% 06019: En fait, la premi\`{e}re relation est triviale, et la deuxi\`{e}me se d\'{e}compose en deux, suivant les deux facteurs de $L_{1}(P)$, les deux relations en question exprimant respectivement la commutativit\'{e} et l'associativit\'{e} de la loi de composition donn\'{e}e sur $P$.
% 06020: 
% 06021: %=====================================================
% 06022: % PAGE 205
% 06023: %=====================================================
% 06024: \noindent Il est imm\'{e}diat que le complexe augment\'{e} ainsi d\'{e}fini d\'{e}pend fonctoriellement de $P$, un morphisme $u: P \to P'$ donnant naissance \`{a} un diagramme commutatif
% 06025: \[
% 06026: \begin{array}{ccc}
% 06027: L_{\cdot}(P) & \xrightarrow{L_{\cdot}(u)} & L_{\cdot}(P') \\
% 06028: \epsilon\downarrow & & \downarrow\epsilon' \\
% 06029: P & \xrightarrow{u} & P'
% 06030: \end{array}
% 06031: \]
% 06032: Notons aussi que les composantes du complexe $L_{\cdot}$ sont plates, comme tout $\bZ$-Module libre.
% 06033: 
% 06034: \begin{proposition}[3.5.3]
% 06035: On a $H_{1}(L_{\cdot}(P)) = 0$, et l'augmentation $\epsilon$ induit un isomorphisme $H_{0}(L_{\cdot}(P)) \xrightarrow{\sim} P$.
% 06036: \end{proposition}
% 06037: 
% 06038: Pour d\'{e}montrer 3.5.3, on peut se r\'{e}duire par le proc\'{e}d\'{e} standard au cas o\`{u} $T$ est la cat\'{e}gorie des ensembles, et o\`{u} $P$ est donc un $\bZ$-Module ordinaire, o\`{u} (3.5.1) a des chances d'\^{e}tre ``bien connu''. Nous allons donner une d\'{e}monstration directe, n'utilisant pas cette r\'{e}duction, en appliquant 3.4.7 au morphisme d'augmentation
% 06039: \[ \epsilon: L_{\cdot}(P) \longrightarrow P, \]
% 06040: o\`{u} $P$ est comme d'habitude consid\'{e}r\'{e} comme un complexe de cha\^{i}nes r\'{e}duit \`{a} la dimension z\'{e}ro. On est donc ramen\'{e} \`{a} prouver que pour tout objet $G$ de $C$, le foncteur
% 06041: \begin{equation}\tag{*}
% 06042: \underline{\Upsilon}_{L_{\cdot}(P)}(G) \longrightarrow \underline{\Upsilon}_{P}(G)
% 06043: \end{equation}
% 06044: est une \'{e}quivalence de cat\'{e}gories. Or le premier membre de $(*)$ est \'{e}videmment isomorphe \`{a} la cat\'{e}gorie des extensions de $\bZ$-Modules de $P$ par $G$.
% 06045: 
% 06046: %=====================================================
% 06047: % PAGE 206
% 06048: %=====================================================
% 06049: \noindent Pour expliciter le deuxi\`{e}me membre, on interpr\`{e}te les cat\'{e}gories d'extensions de $L_{0}, L_{1}, L_{2}$ par $G$, comme la cat\'{e}gorie des torseurs sous $G_{P}$, resp.\ la cat\'{e}gorie des couples d'un torseur sous $G_{P \times P}$ et d'un torseur sous $G_{P \times P \times P}$ gr\^{a}ce \`{a} 1.4. Donc un objet du second membre de $(*)$ s'identifie \`{a} un couple $(E,\varphi)$, o\`{u} $E$ est un torseur sous $G_{P}$, et $\varphi$ une trivialisation du torseur $E'$ sur $P \times P$ qui correspond \`{a} l'image inverse par $d_{1}$ de l'extension de $\bZ[P]$ par $G$ d\'{e}finie par $E$, --- $\varphi$ \'{e}tant soumise \`{a} une condition de compatibilit\'{e} avec $d_{1}d_{2} = 0$ \`{a} pr\'{e}ciser.
% 06050: 
% 06051: Or la d\'{e}finition de $d_{1}$ (3.5.2) donne aussit\^{o}t
% 06052: \[ E' = \pr_{1}^{*}(E) \otimes \pr_{2}^{*}(E)^{-1} \otimes \mu^{*}(E) \]
% 06053: et une trivialisation de ce torseur peut s'interpr\'{e}ter comme un isomorphisme (\'{e}galement not\'{e} $\varphi$)
% 06054: \[ \mu^{*}(E) \xrightarrow{\sim} \pr_{1}^{*}(E) \otimes \pr_{2}^{*}(E). \]
% 06055: D'autre part, en explicitant la condition de compatibilit\'{e} de $\varphi$ avec $d_{1}d_{2} = 0$, on trouve qu'elle s'exprime par la condition de commutativit\'{e} de (1.2.1), et par la condition d'associativit\'{e} de (1.1.4.1), ces deux conditions correspondant respectivement aux deux facteurs de $L_{2}$. Le fait que $(*)$ est une \'{e}quivalence de cat\'{e}gories n'est alors autre que le dictionnaire de 1.1.6. Cela ach\`{e}ve la d\'{e}monstration de 3.5.3.

% 06093: \textbf{3.6. Application \`{a} la description de la cat\'{e}gorie cofibr\'{e}e des biextensions d'un couple $(P,Q)$ donn\'{e} de Groupes commutatifs}
% 06094: 
% 06095: Soient $P$, $Q$ deux $\bZ$-Modules du topos $T$. Choisissons arbitrairement une r\'{e}solution plate $L_{\cdot}^{!}(P)$ de $P$, co\"{i}ncidant en dimension $\leq 2$ avec le complexe augment\'{e} $L_{\cdot}(P)$ de 3.5, --- ce qui est possible gr\^{a}ce \`{a} 3.5.3 et gr\^{a}ce au fait que tout $\bZ$-Module sur $T$ est quotient d'un $\bZ$-Module libre, donc plat. Choisissons de m\^{e}me une r\'{e}solution plate $L_{\cdot}^{!}(Q)$ de $Q$ co\"{i}ncidant avec $L_{\cdot}(Q)$ en dimensions $\leq 2$. On a donc des homomorphismes canoniques
% 06096: \begin{equation}\tag{3.6.1}
% 06097: L_{\cdot}(P) \longrightarrow L_{\cdot}^{!}(P), \qquad L_{\cdot}(Q) \longrightarrow L_{\cdot}^{!}(Q),
% 06098: \end{equation}
% 06099: induisant un homomorphisme canonique
% 06100: \begin{equation}\tag{3.6.2}
% 06101: L_{\cdot}(P,Q) = L_{\cdot}(P) \otimes L_{\cdot}(Q) \longrightarrow L_{\cdot}^{!}(P,Q) = L_{\cdot}^{!}(P) \otimes L_{\cdot}^{!}(Q) \longrightarrow P \otimes Q,
% 06102: \end{equation}
% 06103: qui est un isomorphisme sur les composantes de dimension $\leq 2$. Par suite on a, en vertu des d\'{e}finitions,
% 06104: \begin{equation}\tag{3.6.3}
% 06105: \underline{\Upsilon}_{L_{\cdot}^{!}(P,Q)}^{1} \xrightarrow{\sim} \underline{\Upsilon}_{L_{\cdot}(P,Q)}^{1}.
% 06106: \end{equation}
% 06107: 
% 06108: \begin{theorem}[3.6.4]
% 06109: La cat\'{e}gorie cofibr\'{e}e additive, sur la cat\'{e}gorie $C = \mathbf{GRAB}(T)$ des Groupes commutatifs de $T$, form\'{e}e des biextensions de $(P,Q)$ par un Groupe ab\'{e}lien $G$ variable (2.4), est canoniquement \'{e}quivalente \`{a} la cat\'{e}gorie cofibr\'{e}e additive $\underline{\Upsilon}_{L_{\cdot}(P,Q)}^{1}$, d\'{e}finie en termes de $L_{\cdot}(P,Q) = L_{\cdot}(P) \otimes L_{\cdot}(Q)$ comme en 3.1.
% 06110: \end{theorem}
% 06111: 
% 06112: De ceci, et de 3.2.5, on d\'{e}duit le r\'{e}sultat principal du pr\'{e}sent num\'{e}ro, annonc\'{e} plus haut :
% 06113: 
% 06114: %=====================================================
% 06115: % PAGE 210
% 06116: %=====================================================
% 06117: \begin{corollary}[3.6.5]
% 06118: Pour $P$ et $Q$ fix\'{e}s, on a un isomorphisme canonique, fonctoriel en $G$ :
% 06119: \begin{equation}\tag{3.6.5.1}
% 06120: \Biext^{1}(P,Q;G) \xrightarrow{\sim} \Ext^{1}(P \mathop{\otimes}^{\bL} Q, G)
% 06121: \end{equation}
% 06122: \end{corollary}
% 06123: 
% 06124: D\'{e}montrons 3.6.5, et pour ceci explicitons d'abord les composantes de dimension 0, 1 et 2 de $L_{\cdot}(P,Q)$ :
% 06125: \begin{equation}\tag{3.6.6}
% 06126: \begin{aligned}
% 06127: L_{0}(P,Q) &= L_{0}(P) \otimes L_{0}(Q) = \bZ[P \times Q] \\
% 06128: L_{1}(P,Q) &= L_{1}(P) \otimes L_{0}(Q) + L_{0}(P) \otimes L_{1}(Q) = \bZ[P \times Q \times Q] + \bZ[P \times P \times Q] \\
% 06129: L_{2}(P,Q) &= L_{2}(P) \otimes L_{0}(Q) + L_{1}(P) \otimes L_{1}(Q) + L_{0}(P) \otimes L_{2}(Q) \\
% 06130: &= \bZ[P \times Q \times Q] + \bZ[P \times Q \times Q \times Q] \\
% 06131: &\quad + \bZ[P \times P \times Q \times Q] + \bZ[P \times P \times Q] + \bZ[P \times P \times P \times Q]
% 06132: \end{aligned}
% 06133: \end{equation}
% 06134: 
% 06135: Pour expliciter la cat\'{e}gorie cofibr\'{e}e $\underline{\Upsilon}_{L_{\cdot}(P,Q)}^{1}$ via ses cat\'{e}gories fibres $\underline{\Upsilon}_{L_{\cdot}(P,Q)}^{1}(G)$, on utilise 1.4 pour expliciter les extensions des $L_{i}(P,Q)$ (pour $i = 0,1,2$) par $G$, en termes de torseurs. Ainsi, une extension de $L_{0}(P \times Q)$ par $G$ s'identifie \`{a} un torseur sous $G_{P \times Q}$, une extension de $L_{1}(P,Q)$ par $G$ s'identifie \`{a} un couple d'un torseur sous $G_{P \times Q \times Q}$ et d'un autre sous $G_{P \times P \times Q}$ ; et de m\^{e}me une extension de $L_{2}(P,Q)$ par $G$ s'identifie \`{a} un syst\`{e}me de cinq torseurs, sous les groupes d\'{e}duits de $G$ par changement de base sur les bases $P \times Q \times Q$, $P \times Q \times Q \times Q$, $P \times P \times Q \times Q$, $P \times P \times Q$, $P \times P \times P \times Q$ respectivement. Utilisant ce dictionnaire, on peut interpr\'{e}ter un couple $(X,\alpha)$ d'une extension de $L_{0}(P,Q)$ par $G$, et d'une trivialisation de $d_{1}^{*}(X)$ sur $L_{1}(P,Q)$, comme un torseur $E$ sur $P \times Q$, muni d'un couple $(\varphi,\psi)$ de trivialisations du couple de torseurs $(E_{1},E_{2})$ sur $P \times Q \times Q$ resp.\ $P \times P \times Q$, d\'{e}duit de $E$ par image inverse
% 06136: 
% 06137: %=====================================================
% 06138: % PAGE 211
% 06139: %=====================================================
% 06140: \noindent \`{a} l'aide de $d_{1}: L_{1}(P,Q) \to L_{0}(P,Q)$. Explicitant l'op\'{e}rateur diff\'{e}rentiel $d_{1}d_{2}$ on trouve comme dans la d\'{e}monstration de 3.5.3 que $\varphi$ et $\psi$ peuvent s'interpr\'{e}ter comme des isomorphismes
% 06141: \[ \mu_{Q}^{*}(E) \xrightarrow{\sim} \pr_{1}^{*}(E) \otimes \pr_{2}^{*}(E), \qquad \mu_{P}^{*}(E) \xrightarrow{\sim} \pr_{1}^{*}(E) \otimes \pr_{2}^{*}(E) \]
% 06142: o\`{u} $\mu_{Q}, \pr_{1}, \pr_{2}: P \times Q \times Q \to Q$ sont les trois morphismes d\'{e}duits respectivement des trois morphismes $Q \times Q \to Q$ loi de composition, premi\`{e}re et deuxi\`{e}me projection, et o\`{u} $\mu_{P}, \pr_{1}, \pr_{2}: P \times P \times Q \to P \times Q$ sont d\'{e}finis de fa\c{c}on sym\'{e}trique. Donc $\varphi$ et $\psi$ peuvent s'interpr\'{e}ter comme des donn\'{e}es du type (2.0.4) et (2.0.7) qui figurent dans 2.1. De fa\c{c}on sym\'{e}trique, les deux derni\`{e}res conditions expriment que $\psi$ d\'{e}finit sur $E$ une structure d'extension commutative de $P$ par $G$. Enfin, la condition m\'{e}diane, portant sur des torseurs sur $P \times P \times Q \times Q$, n'est autre que la commutativit\'{e} du diagramme (2.1.1), qui exprime que $\varphi$ et $\psi$ sont compatibles, i.e.\ (compte tenu des autres conditions) qu'ils d\'{e}finissent sur $E$ une structure de biextension de $(P,Q)$ par $G$. La v\'{e}rification de ces interpr\'{e}tations est essentiellement triviale, et laiss\'{e}e au lecteur. Le lecteur admettra sans doute avec nous que les discours pr\'{e}c\'{e}dents d\'{e}finissent bien une \'{e}quivalence de cat\'{e}gories cofibr\'{e}es sur $C = \mathbf{GRAB}(T)$, qui est celle annonc\'{e}e dans 3.6.4.
% 06143: 
% 06144: \textbf{Remarque 3.6.7.} Chacune des trois g\'{e}n\'{e}ralisations signal\'{e}es dans 2.10 de la notion de biextension permet de donner une variante correspondante de 3.6.4 (donc aussi de 3.6.5). La g\'{e}n\'{e}ralisation au cas d'un nombre quelconque de facteurs (cf.\ 2.10.2) est triviale, et la g\'{e}n\'{e}ralisation au cas de biextensions de $A$-Modules (cf.\ 2.10.3) est imm\'{e}diate, une fois qu'on a obtenu la variante de la construction de $L_{\cdot}(P)$ dans ce cas, signal\'{e}e dans 3.5.4 b). Plus int\'{e}ressante du point de vue de l'alg\`{e}bre homologique semblerait la g\'{e}n\'{e}ralisation au cas o\`{u} $P$ et $Q$ ne sont pas suppos\'{e}s commutatifs (cf.\ 2.10.1). Notons d'ailleurs que la premi\`{e}re relation (1.6.1) implique \'{e}videmment que $\Hom(P \otimes Q, G) = 0$, donc que la cat\'{e}gorie $\EXT(P,\Hom(Q,G))$ des extensions (commutatives) de $P$ par $\Hom(Q,G)$ est rigide (tout automorphisme d'un objet est l'identit\'{e}), et qu'il en est de m\^{e}me de $\BIEXT(P,Q;G)$. La structure de ces cat\'{e}gories est donc connue quand on conna\^{i}t le groupe des classes d'isomorphie d'objets, qui en vertu de (1.6.2) est canoniquement isomorphe \`{a} $\Hom(Q, \Ext^{1}(P,G))$.

% 06532: \[ \Ext^{1}(P,Q') \times Q \to G \]
% 06533: (o\`{u} $Q' = \Hom(Q,G)$), en l'interpr\'{e}tant comme un homomorphisme $Q \to \Hom(Q',G)$, d'o\`{u} un compos\'{e}
% 06534: \begin{equation}\tag{1.6.3}
% 06535: \Ext^{1}(P,Q') \times Q \to \Ext^{1}(P,Q') \times \Hom(Q',G) \to \Ext^{1}(P,G),
% 06536: \end{equation}
% 06537: qui redonne (1.6.2).

% 06542: \textbf{2. Accouplements d\'{e}finis par une biextension}
% 06543: 
% 06544: \textbf{2.1.} Soient $P$ et $Q$ deux Groupes ab\'{e}liens sur $T$, et $n > 0$ un entier. Rappelons qu'on d\'{e}signe par ${}_{n}P$ le noyau de la multiplication par $n$ $\id_{n}$. Nous allons d\'{e}finir un homomorphisme canonique
% 06545: \begin{equation}\tag{2.1.1}
% 06546: a = a_{P,Q}^{n}: {}_{n}P \otimes Q \longrightarrow \Tor_{1}(P,Q)
% 06547: \end{equation}
% 06548: qui sera le compos\'{e}
% 06549: \begin{equation}\tag{2.1.2}
% 06550: {}_{n}P \otimes Q \xrightarrow{\sim} {}_{n}P \otimes {}_{n}Q \longrightarrow \Tor_{1}(P,{}_{n}Q) \longrightarrow \Tor_{1}(P,Q)
% 06551: \end{equation}
% 06552: o\`{u} la premi\`{e}re fl\`{e}che est $\id \otimes \inj$ et la deuxi\`{e}me est celle d\'{e}duite des inclusions ${}_{n}P \to P$, ${}_{n}Q \to Q$. % 06553: \[ P \mathop{\otimes}^{\bL} Q = L_{\cdot} \otimes_{\bZ} Q = (L_{\cdot} \otimes_{\bZ} \bZ/n\bZ) \otimes_{\bZ/n\bZ} Q \]
% 06554: qu'on peut \'{e}crire comme un isomorphisme dans la cat\'{e}gorie d\'{e}riv\'{e}e
% 06555: \begin{equation}\tag{2.1.3}
% 06556: P \mathop{\otimes}^{\bL} Q \xrightarrow{\sim} (P \mathop{\otimes}^{\bL} \bZ/n\bZ) \otimes_{\bZ/n\bZ}^{\bL} Q
% 06557: \end{equation}
% 06558: donnant naissance \`{a} la suite spectrale de type homologique ``de K\"{u}nneth'' (cf.\ SGA 1 IV 5.2) :
% 06559: \begin{equation}\tag{2.1.4}
% 06560: \Tor_{q}(P,Q) \Leftarrow E_{pq}^{2} = \Tor_{p}^{\bZ/n\bZ}(\Tor_{q}(P,\bZ/n\bZ), Q)
% 06561: \end{equation}
% 06562: Rappelons d'ailleurs qu'on a des isomorphismes canoniques
% 06563: \[ \Tor_{0}(P,\bZ/n\bZ) = P/nP = P, \quad \Tor_{1}(P,\bZ/n\bZ) = {}_{n}P, \quad \Tor_{i}(P,\bZ/n\bZ) = 0 \ \text{si} \ i \geq 2, \]
% 06564: d\'{e}duits du calcul des $\Tor_{i}(P,\bZ/n\bZ)$ \`{a} l'aide de la r\'{e}solution plate canonique suivante de $(\bZ/n\bZ)_{T}$ :
% 06565: \[ 0 \longrightarrow \bZ_{T} \xrightarrow{n} \bZ_{T} \longrightarrow 0 \]
% 06566: l'augmentation $L_{0} \to (\bZ/n\bZ)_{T}$ \'{e}tant d\'{e}finie en envoyant $1$ sur $1$. Par suite, la suite spectrale (2.1.4) se r\'{e}duit essentiellement \`{a} une suite exacte
% 06567: \begin{equation}\tag{2.1.5}
% 06568: 0 \longrightarrow \Tor_{1}^{\bZ/n\bZ}({}_{n}P, Q) \longrightarrow {}_{n}P \otimes Q \longrightarrow \Tor_{1}(P,Q) \longrightarrow \Tor_{1}^{\bZ/n\bZ}(P/nP, Q) \longrightarrow 0,
% 06569: \end{equation}
% 06570: o\`{u} on a pos\'{e} $P_{n} = P \otimes \bZ/n\bZ = P/nP$. L'homomorphisme cherch\'{e} est l'homomorphisme-coin ${}_{n}P \otimes Q \to \Tor_{1}(P,Q)$ qui figure dans la derni\`{e}re suite exacte.
% 06571: 
% 06572: Cette derni\`{e}re \'{e}tant \'{e}videmment fonctorielle en $P$ et en $Q$, la fonctorialit\'{e} en $P$ montre que l'homomorphisme envisag\'{e} est bien le compos\'{e}
% 06573: \[ {}_{n}P \otimes Q \to \Tor_{1}(P,Q) \to \Tor_{1}(P,Q) \]
% 06574: o\`{u} le premier homomorphisme est l'homomorphisme $a$ d\'{e}fini comme pr\'{e}c\'{e}demment, mais avec $(P,Q)$ remplac\'{e} par $({}_{n}P,Q)$, et o\`{u} le deuxi\`{e}me homomorphisme provient de l'inclusion ${}_{n}P \to P$.

% 06576: \textbf{2.1.6.} Ceci d\'{e}finit donc bien, pour $P$ et $Q$ quelconques, un homomorphisme (2.1.1) satisfaisant \`{a} la condition de factorisation (2.1.2).
