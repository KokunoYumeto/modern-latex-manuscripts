\section[The functors Rq u* and the Cartan--Leray spectral sequence relative to a morphism of topoi]{The functors $R^q u_*$ and the Cartan--Leray spectral sequence relative to a morphism of topoi}

\subsection*{5.0}
Let
\[
        u:(E,A)\longrightarrow (E',A')
\]
be a morphism of ringed topoi, and let $u_*$ denote the direct-image functor. The same notation $u_*$ will also denote the induced functor on modules. For modules, say left modules to fix ideas, the functor $u_*$ is left exact. Its right derived functors are denoted
\[
        R^q u_* ,\qquad q\geq 0.
\]

\begin{statement}{Proposition 5.1}
\begin{enumerate}[label=\arabic*)]
\item For every $A$-module $M$, the sheaf $R^q u_*M$ is the sheaf associated with the presheaf
\[
        X'\longmapsto H^q(u^*X',M),\qquad X'\in\ob(E').
\]
\item The formation of the functors $R^q u_*$ commutes with restriction of scalars.
\item The formation of the functors $R^q u_*$ commutes with localizations. More precisely, for every object $X'$ of $E'$, if
\[
        u_{/X'}:E_{/u^*X'}\longrightarrow E'_{/X'}
\]
denotes the morphism deduced from $u$ by localization \emph{(IV, 8)}, then, for every $A$-module $M$, there is a canonical isomorphism
\begin{equation*}
        R^q(u_{/X'})_*(M|u^*X')\simeq (R^q u_*M)|X',\qquad q\geq 0 . \tag{5.1.1}
\end{equation*}
\end{enumerate}
\end{statement}

Let
\[
        u_*^{\wedge}:E^{\wedge}\longrightarrow E'^{\wedge}
\]
be the direct-image functor for $\U$-presheaves, so that $u_*^{\wedge}M=M\circ u^*$. Since $u^*$ and $u_*$ are adjoint, one has an isomorphism
\[
        u_*\simeq \underline a\,u_*^{\wedge}{}^{0},
\]
where $\underline a$ is the associated-sheaf functor for $E'$. Since the functors $\underline a$ and $u_*^{\wedge}$ are exact, one has
\[
        R^q u_*\simeq \underline a\,u_*^{\wedge}\Hh^q,
\]
which is another way of stating 1). Assertion 2) then follows from 1) and 3.5. By the definition of the localized morphism $u_{/X'}$, there is the canonical isomorphism (5.1.1) for $q=0$. The general case follows by observing that localization functors \emph{(IV, 8)} are exact and send injective objects to injective objects (4.11).

\begin{statement}{Proposition 5.2}
Let $u:E\to E'$ be a morphism of topoi, and let $S'$ be a generating family of $E'$. The sheaves $M$ which are acyclic for the functors
\[
        H^0(u^*X',-),\qquad X'\in S',
\]
are acyclic for the functor $u_*$. In particular, flasque sheaves are acyclic for $u_*$.
\end{statement}

This follows from 5.1.1).

\begin{statement}{Proposition 5.3}
Let $u:E\to E'$ be a morphism of topoi, and let $M$ be an abelian group object of $E$. There is a spectral sequence
\begin{equation*}
        E_2^{p,q}=H^p(E',R^q u_*M)\Longrightarrow H^{p+q}(E,M).\tag{5.3.1}
\end{equation*}
More generally, for every object $X'$ of $E'$, there is a spectral sequence
\begin{equation*}
        E_2^{p,q}=H^p(X',R^q u_*M)\Longrightarrow H^{p+q}(u^*X',M).\tag{5.3.2}
\end{equation*}
\end{statement}

By definition of the direct- and inverse-image functors, there is an isomorphism
\[
        H^0(X',u_*M)\simeq H^0(u^*X',M).
\]
The functor $u_*$ sends injective objects to flasque sheaves (4.6 and 4.9). The spectral sequences displayed above are therefore the spectral sequences of composed functors (0.3).

\begin{statement}{Proposition 5.4}
Let $u:E\to E'$ and $v:E'\to E''$ be two morphisms of topoi. There is a spectral sequence
\[
        R^p v_*\,R^q u_*\Longrightarrow R^{p+q}(v\circ u)_* .
\]
\end{statement}

One has $v_*u_*\simeq (vu)_*$, and $u_*$ sends injective objects to flasque sheaves (4.9), hence to objects acyclic for $v_*$ (5.2). Thus this is the spectral sequence of composed functors (0.3).

\section{Local Ext and cohomology with supports}

\subsection*{6.0}
Let $(E,A)$ be a ringed topos, and let $M$ be an $A$-module, say a left module to fix ideas. The functor
\[
        N\longmapsto \Hhom_A(M,N),
\]
from left $A$-modules to abelian groups, is left exact (IV, 12). Its right derived functors are denoted
\begin{equation*}
        \Eext_A^q(M,N).\tag{6.0.1}
\end{equation*}
In particular,
\begin{equation*}
        \Eext_A^0(M,N)=\Hhom_A(M,N).\tag{6.0.2}
\end{equation*}
By definition, one has canonical isomorphisms (2.1 and IV, 12)
\begin{equation*}
        H^0(E,\Eext_A^0(M,N))=\Ext_A^0(E;M,N)=\Hom_A(M,N).\tag{6.0.3}
\end{equation*}
More generally, for every object $X$ of $E$, one has (2.2)
\begin{equation*}
        H^0(X,\Eext_A^0(M,N))=\Ext_A^0(X;M,N)=\Hom_{A|X}(M|X,N|X).\tag{6.0.4}
\end{equation*}

\begin{statement}{Proposition 6.1}
\begin{enumerate}[label=\arabic*)]
\item The formation of the functors $\Eext_A^q$ commutes with localization. More precisely, for every object $X$ of $E$ one has functorial isomorphisms
\begin{equation*}
        \Eext_A^q(M,N)|X\simeq \Eext_{A|X}^q(M|X,N|X).\tag{6.1.1}
\end{equation*}
\item The sheaf $\Eext_A^q(M,N)$ is isomorphic to the sheaf associated with the presheaf
\[
        X\longmapsto \Ext_A^q(X;M,N).
\]
\item There is a spectral sequence
\begin{equation*}
        \Ext_A^{p+q}(E;M,N)\Longleftarrow E_2^{p,q}=H^p(E,\Eext_A^q(M,N)).\tag{6.1.2}
\end{equation*}
\end{enumerate}
More generally, for every object $X$ of $E$, there is a spectral sequence, functorial in $X$ and in the arguments $M$ and $N$,
\begin{equation*}
        \Ext_A^{p+q}(X;M,N)\Longleftarrow E_2^{p,q}=H^p(X,\Eext_A^q(M,N)).\tag{6.1.3}
\end{equation*}
\end{statement}

By definition, (6.1.1) is an isomorphism for $q=0$ (IV, 12). The general case follows from the fact that localization functors are exact and send injective modules to injective modules (2.2). The functor
\[
        N\longmapsto \Hhom_A(M,N)=\Eext_A^0(M,N)
\]
sends injective modules to flasque sheaves, hence to sheaves acyclic for $H^0(X,-)$ (4.10). The spectral sequences (6.1.2) and (6.1.3) are therefore spectral sequences of composed functors, using (6.0.3) and (6.0.4). When $X$ varies in $E$, (6.1.3) gives a spectral sequence of presheaves and hence, after passage to associated sheaves, a spectral sequence of sheaves. Since the sheaves associated with the presheaves $X\mapsto H^p(X,-)$ vanish for $p\neq 0$ (3.1), this spectral sequence degenerates and gives the isomorphism asserted in 2).

\begin{statement}{Proposition 6.2}
The functors $(M,N)\mapsto \Eext_A^q(M,N)$, $q\geq 0$, form a $\delta$-functor in the variable $M$ and in the variable $N$. Explicitly, for every exact sequence $0\to N'\to N\to N''\to 0$ (\resp\ $0\to M'\to M\to M''\to 0$), one has long exact sequences
\begin{equation*}
\cdots\to \Eext_A^q(M,N')\to \Eext_A^q(M,N)\to \Eext_A^q(M,N'')\xrightarrow{\delta}\Eext_A^{q+1}(M,N')\to\cdots,\tag{6.2.1}
\end{equation*}
respectively
\begin{equation*}
\cdots\to \Eext_A^q(M'',N)\to \Eext_A^q(M,N)\to \Eext_A^q(M',N)\xrightarrow{\delta}\Eext_A^{q+1}(M'',N)\to\cdots .\tag{6.2.2}
\end{equation*}
\end{statement}

It follows from the general properties of the functors $\Ext^q$ that, for every object $X$ of $E$, the functors $(M,N)\mapsto \Ext_A^q(X;M,N)$, $q\geq 0$, form a $\delta$-functor in each argument. The assertion follows by letting $X$ vary and taking the associated sheaf (6.1).

\subsection*{6.3}
Let $(E,A)$ be a ringed topos, let $M$ be an $A$-module, let $Z$ be a closed subtopos of $E$ \emph{(IV, 9)}, and let $U$ be its open complement. We denote by $H_Z^0(E,M)$ the group of sections of $M$ whose support is contained in $Z$ (IV, 14), and by $\Hh_Z^0(M)$ the subsheaf of $M$ defined by the sections of $M$ ``with support in $Z$'' (IV, 14). The functors $H_Z^0(E,-)$ and $\Hh_Z^0(-)$ are left exact (IV, 14). Their derived functors are denoted
\[
        H_Z^q(E,-),\qquad \Hh_Z^q(-),
\]
respectively, and are called the \emph{groups} (\resp\ \emph{sheaves}) \emph{of cohomology of $M$ with supports in $Z$}.\footnote{Compare SGA 2, Expos\'e I for the case of ordinary topological spaces, and the expos\'e of Hartshorne cited later in the original source.}

One has canonical isomorphisms (IV, 14)
\begin{align*}
        H_Z^0(E,M)&\simeq \Hom_A(A_Z,M)\simeq \Ext_A^0(E;A_Z,M),\tag{6.3.1}\\
        \Hh_Z^0(M)&\simeq \Hhom_A(A_Z,M)\simeq \Eext_A^0(A_Z,M),\tag{6.3.2}
\end{align*}
and hence isomorphisms, for every $q\geq 0$,
\begin{align*}
        H_Z^q(E,M)&\simeq \Ext_A^q(E;A_Z,M),\tag{6.3.3}\\
        \Hh_Z^q(M)&\simeq \Eext_A^q(A_Z,M).\tag{6.3.4}
\end{align*}
Notice that, since $A_Z$ is a bimodule, the sheaves
\[
        \Eext_A^q(A_Z,M)\simeq \Hh_Z^q(M)
\]
are endowed canonically with $A$-module structures.

For every object $X$ of $E$, denote by $Z_{/X}$ the closed subtopos of $E_{/X}$ deduced from $Z$ by localization; it is the complement of $U\times X$. By definition, one has canonical isomorphisms
\begin{equation*}
        H^0(X,\Hh_Z^0(M))\simeq \Hh_{Z_{/X}}^0(E_{/X},M|X).\tag{6.3.5}
\end{equation*}
Put
\begin{equation*}
        H_Z^q(X,M)=H_{Z_{/X}}^q(E_{/X},M|X).\tag{6.3.6}
\end{equation*}
Taking account of (6.3.2), one has canonical isomorphisms
\begin{equation*}
        H_Z^q(X,M)\simeq \Ext_A^q(X;A_Z,M).\tag{6.3.7}
\end{equation*}

\begin{statement}{Proposition 6.4}
\begin{enumerate}[label=\arabic*)]
\item The formation of the functors $\Hh_Z^q$ commutes with localization. More precisely, for every object $X$ of $E$ and every $A$-module $M$, one has canonical isomorphisms
\begin{equation*}
        \Hh_Z^q(M)|X\simeq \Hh_{Z_{/X}}^q(M|X).\tag{6.4.1}
\end{equation*}
\item The sheaf $\Hh_Z^q(M)$ is the sheaf associated with the presheaf
\[
        X\longmapsto H_Z^q(X,M).
\]
\item There is a spectral sequence
\begin{equation*}
        H_Z^{p+q}(E,M)\Longleftarrow H^p(E,\Hh_Z^q(M)).\tag{6.4.2}
\end{equation*}
More generally, for every object $X$ of $E$, there is a spectral sequence
\begin{equation*}
        H_Z^{p+q}(X,M)\Longleftarrow H^p(X,\Hh_Z^q(M)).\tag{6.4.3}
\end{equation*}
\end{enumerate}
\end{statement}

This is just Proposition 6.1 translated by means of the dictionary of 6.3.

\begin{statement}{Proposition 6.5}
With the notation of 6.3, let $j:U\to E$ be the canonical morphism. For every $A$-module $M$, there is an exact sequence of sheaves
\begin{equation*}
        0\to \Hh_Z^0(M)\to M\to j_*(M|U)\to \Hh_Z^1(M)\to 0,\tag{6.5.1}
\end{equation*}
and, for $q\geq 2$, isomorphisms
\begin{equation*}
        \Hh_Z^q(M)\simeq \Eext_A^{q-1}(A_U,M)\simeq R^{q-1}j_*(M|U).\tag{6.5.2}
\end{equation*}
Moreover, there is a long exact sequence
\begin{equation*}
\cdots\to H_Z^q(E,M)\to H^q(E,M)\to H^q(U,M)\to H_Z^{q+1}(E,M)\to\cdots,\tag{6.5.3}
\end{equation*}
\par\noindent\emph{Note.} This exact sequence makes precise the role of the global cohomological invariants $H_Z^q(E,N)$ as the cohomology groups of $E$ \emph{modulo} the open $U$, with coefficients in $M$.
and more generally, for every object $X$ of $E$, there is a long exact sequence
\begin{equation*}
\cdots\to H_Z^q(X,M)\to H^q(X,M)\to H^q(X\times U,M)\to H_Z^{q+1}(X,M)\to\cdots.\tag{6.5.4}
\end{equation*}
\end{statement}

By the definition of $A_Z$, there is an exact sequence (IV, 14)
\begin{equation*}
        0\to A_U\to A\to A_Z\to 0.\tag{6.5.5}
\end{equation*}
The functors $\Eext_A^q(A,-)$ vanish for $q>0$. One has
\[
        \Hhom_A(A_U,M)\simeq j_*(M|U)
\]
(IV, 14), and therefore (2.2)
\[
        \Eext_A^q(A_U,M)\simeq R^qj_*(M|U).
\]
Finally, $\Eext_A^q(A_Z,M)\simeq \Hh_Z^q(M)$ by (6.3.4). The long exact sequence (6.2.2) gives in this case (6.5.1) and (6.5.2). The exact sequences (6.5.3) and (6.5.4) follow from the dictionary of 6.3 and from the long exact sequence of the $\delta$-functor $\Ext_A^q(X;- ,M)$, $q\geq 0$, associated with (6.5.5).

\begin{statement}{Proposition 6.6}
\begin{enumerate}[label=\arabic*)]
\item Flasque sheaves are acyclic for the functors $\Hh_Z^0$ and $H_Z^0(X,-)$.
\item The functors $\Hh_Z^0$ and $H_Z^0(X,-)$ commute with restriction of scalars.
\end{enumerate}
\end{statement}

Let $M$ be a flasque sheaf. The exact sequence (6.5.4) gives equalities $H_Z^q(X,M)=0$ for every $X$ and every $q\geq 2$, and an exact sequence
\[
0\to H_Z^0(X,M)\to H^0(X,M)\to H^0(X\times U,M)\to H_Z^1(X,M)\to 0.
\]
Since $M$ is flasque, the morphism $H^0(X,M)\to H^0(X\times U,M)$ is surjective (4). Hence $H_Z^1(X,M)=0$, and $M$ is acyclic for $H_Z^0(X,-)$. Passing to associated sheaves, it follows that $M$ is acyclic for the functor $\Hh_Z^0$ (6.4). It is clear that the functors $\Hh_Z^0$ and $H_Z^0(X,-)$ commute with restriction of scalars; since flasque sheaves are acyclic for these two functors, assertion 2) follows.

\begin{statement}{Proposition 6.7}
Let $(E,A)$ be a ringed topos, let $Z$ be a closed subtopos of $E$, let $U$ be its open complement, and let $M,N$ be two $A$-modules. There are isomorphisms, functorial in $M$ and $N$ and compatible with change of closed subtopos,
\[
        H_Z^0(\Hhom_A(M,N))\simeq \Hhom_A(M,\Hh_Z^0(N))\simeq \Hhom_A(M\otimes_A A_Z,N).
\]
\end{statement}

This follows from (IV, 14), taking (6.3.2) into account.

\subsection*{6.8}
Put
\begin{equation*}
        \Hhom_{A,Z}(M,N)=\Hh_Z^0(\Hhom_A(M,N)).\tag{6.8.1}
\end{equation*}
The functor $N\mapsto \Hhom_{A,Z}(M,N)$ is left exact. Its derived functors are denoted $\Eext_{A,Z}^q(M,N)$; these are the \emph{sheaves $\Eext$ with supports in $Z$}. Thus
\begin{equation*}
        \Eext_{A,Z}^0(M,N)=\Hhom_{A,Z}(M,N).\tag{6.8.2}
\end{equation*}
Put $M_Z=M\otimes_A A_Z$. It follows from IV, 14 that $M_Z$ is the direct image on $E$ of the inverse image of $M$ on $Z$. Therefore, taking account of 6.7, there are isomorphisms
\begin{equation*}
        \Eext_{A,Z}^q(M,N)\simeq \Eext_A^q(M_Z,N).\tag{6.8.3}
\end{equation*}

Now pass to global invariants. Put
\begin{equation*}
        \Hom_{A,Z}(M,N)=\Ext_{A,Z}^0(E;M,N)=H_Z^0(E,\Hhom_A(M,N)).\tag{6.8.4}
\end{equation*}
The group $\Hom_{A,Z}(M,N)$ is the subgroup of the group of morphisms from $M$ to $N$ whose support is in $Z$ (IV, 14), i.e. which are zero on $U$. More generally, for every object $X$ of $E$, put
\begin{equation*}
        \Ext_{A,Z}^0(X;M,N)=H_Z^0(X,\Hhom_A(M,N)).\tag{6.8.5}
\end{equation*}
The functors $N\mapsto \Ext_{A,Z}^0(X;M,N)$ are left exact. Their derived functors are denoted $\Ext_{A,Z}^q(X;M,N)$. These are the \emph{groups $\Ext$ with supports in $Z$}. The definitions (6.3.6), (6.8.1), and (6.8.5), together with the isomorphisms of 6.7, give isomorphisms
\begin{equation*}
\Ext_{A,Z}^0(X;M,N)\simeq
\begin{cases}
H^0(X,\Hhom_{A,Z}(M,N)),\\[3pt]
\Ext^0(X;M,\Hh_Z^0(N)),\\[3pt]
\Ext^0(X;M_Z,N).
\end{cases}\tag{6.8.6}
\end{equation*}
The last isomorphism in (6.8.6) gives isomorphisms
\begin{equation*}
        \Ext_{A,Z}^q(X;M,N)\simeq \Ext_A^q(X;M_Z,N).\tag{6.8.7}
\end{equation*}

\begin{statement}{Proposition 6.9}
\begin{enumerate}[label=\arabic*)]
\item There are two spectral sequences, functorial in $M$ and $N$ and compatible with change of closed subtopos,
\begin{equation*}
\Eext_{A,Z}^{p+q}(M,N)\Longleftarrow
\begin{cases}
E_2^{p,q}=\Hh_Z^p(\Eext_A^q(M,N)),\\[3pt]
{}'E_2^{p,q}=\Eext_A^p(M,\Hh_Z^q(N)).
\end{cases}\tag{6.9.1}
\end{equation*}
\item There are three spectral sequences, functorial in $X,M,N$ and compatible with change of closed subtopos,
\begin{equation*}
\Ext_{A,Z}^{p+q}(X;M,N)\Longleftarrow
\begin{cases}
E_2^{p,q}=H_Z^p(X,\Eext_A^q(M,N)),\\[3pt]
{}'E_2^{p,q}=H^p(X,\Eext_{A,Z}^q(M,N)),\\[3pt]
{}''E_2^{p,q}=\Ext_A^p(X;M,\Hh_Z^q(N)).
\end{cases}\tag{6.9.2}
\end{equation*}
\item The sheaves $\Eext_{A,Z}^q(M,N)$ are canonically isomorphic to the sheaves associated with the presheaves
\[
        X\longmapsto \Ext_{A,Z}^q(X;M,N).
\]
\end{enumerate}
\end{statement}

When $N$ is injective, the sheaf $\Hhom_A(M,N)$ is flasque (4.10), hence acyclic for $\Hh_Z^0$ (6.6). The first spectral sequence in (6.9.1) is the spectral sequence of the composed functors in (6.8.1). Likewise, when $N$ is injective, $\Hh_Z^0(N)$ is injective (4.11), and the second spectral sequence in (6.9.1) is the spectral sequence of composed functors deduced from 6.7. The spectral sequences in (6.9.2) are spectral sequences of composed functors deduced from the definition (6.8.5) and from the first two isomorphisms of (6.8.6). Finally, letting $X$ vary in the second spectral sequence of (6.9.2) and taking associated sheaves, one obtains a spectral sequence of sheaves which degenerates by 3.1 and gives the isomorphisms of 3).

\begin{statement}{Proposition 6.10}
The functors $(M,N)\mapsto \Eext_{A,Z}^q(M,N)$, $q\geq 0$, and $(M,N)\mapsto \Ext_{A,Z}^q(X;M,N)$, $q\geq 0$, are $\delta$-functors in each of the variables $M$ and $N$. If $M_U$ denotes the sheaf $M\otimes_A A_U$ \emph{(cf. IV, 11)}, then there is a long exact sequence
\begin{equation*}
\cdots\xrightarrow{\delta}\Eext_{A,Z}^q(M,N)\to \Eext_A^q(M,N)\to \Eext_A^q(M_U,N)\xrightarrow{\delta}\Eext_{A,Z}^{q+1}(M,N)\to\cdots,\tag{6.10.1}
\end{equation*}
and an analogous exact sequence for the groups $\Ext$.
\end{statement}

The functors $\Eext_A^q(-,-)$ and $\Ext_A^q(X;-,-)$ form $\delta$-functors in each variable (6.2), and the functor $M\mapsto M_Z$ is exact because $A_Z$ is flat (1.3.3). The first assertion follows from the isomorphisms (6.8.3) and (6.8.7). The exact sequence (6.10.1), and the analogous exact sequence for the groups $\Ext$, is the long exact sequence of the $\delta$-functor
\[
        \Eext_A^q(-,N),\quad q\geq 0
        \quad(\resp\ \Ext_A^q(X;-,N),\quad q\geq 0)
\]
relative to the exact sequence
\[
        0\to M_U\to M\to M_Z\to 0,
\]
taking (6.8.3) and (6.8.7) into account.

\subsection*{6.11}
Let us briefly indicate how these results may be extended to the case of families of supports. Let $E$ be a topos. For every object $X$ of $E$, denote by $\Fer(X)$ the set of closed subtopoi of $E_{/X}$. This set is in bijection, by passage to complements, with the set of open subtopoi of $E_{/X}$, i.e. with the set of subobjects of $X$ (IV, 8). The presheaf $\Fer:X\mapsto \Fer(X)$ is in fact a $\U$-sheaf, as is checked immediately; hence it is representable. In other words, there is an object of $E$, again denoted $\Fer$, and a closed subtopos $Z_{\Fer}$ of $E_{/\Fer}$ such that, for every $X$, every element $Z$ of $\Fer(X)$ is deduced from $Z_{\Fer}$ by base change along a morphism
\[
        u_Z:X\to \Fer
\]
uniquely determined by $Z$.

\begin{statement}{Definition 6.12}
A \emph{family of supports} of $E$ is a subset $\Phi$ of the set of closed subtopoi of $E$ having the following properties:
\begin{itemize}
\item[(S1)] the union of a finite family of elements of $\Phi$ belongs to $\Phi$;
\item[(S2)] every closed subtopos of $E$ contained in an element of $\Phi$ is an element of $\Phi$.
\end{itemize}
\end{statement}

\subsubsection*{6.12.1}
Let $E$ be a topos, let $\Phi$ be a family of supports of $E$, and let $X$ be an object of $E$. Denote by $\Phi(X)$ the smallest family of supports of $E_{/X}$ containing the closed subtopoi of $E_{/X}$ deduced from closed subtopoi in $\Phi$ by the base change $X\to e$, where $e$ is a final object of $E$. The functor $X\mapsto \Phi(X)$ is a subpresheaf of the sheaf $\Fer$, hence is separated.

\subsubsection*{6.12.2}
A family $\Phi$ of supports of $E$ is said to be \emph{of local character} if it has the following property:

\medskip
\noindent(CL) For every epimorphic family $(X_i\to e)_{i\in I}$, where $e$ is the final object of $E$, the sequence of sets
\[
        \Phi\longrightarrow \prod_i\Phi(X_i)\rightrightarrows \prod_{i,j}\Phi(X_i\times X_j)
\]
is exact.

Let $a\Phi$ denote the sheaf associated with the presheaf $X\mapsto \Phi(X)$ (6.12.1). Condition (CL) is equivalent to the condition that the canonical morphism
\[
        \Phi\to a\Phi(e)
\]
be a bijection (cf. the construction of the sheaf associated with a separated presheaf in Expos\'e II). To verify (CL), one may therefore restrict to a final family of epimorphic families $(X_i\to e)_{i\in I}$.

\begin{remarktext}{Examples 6.12.3}
\begin{enumerate}[label=\arabic*)]
\item Let $Z$ be a closed subtopos of $E$. The set of closed subtopoi of $E$ contained in $Z$ is a family of supports of $E$. It is of local character.
\item Let $T$ be a topological space and let $\Phi$ be a paracompactifying family of supports on $T$ [7]. In general, $\Phi$ is not of local character.
\item Let $T$ be a topological space and let $p$ be an integer. The family $\Phi_p$ of closed subsets of $T$ of Krull codimension $\geq p$ is a family of supports of $T$. It is of local character.
\item Suppose $E$ is a topos with the following property: every epimorphic family $(X_i\to e)_{i\in I}$ is refined by a finite epimorphic family. Then every family of supports of $E$ is of local character. Indeed, using 1), it is enough to show that a filtered inductive limit $\Phi_\lambda$ of families of local character is a family of local character. This follows immediately by passing to the inductive limit in the exact sequence of sets
\[
\Phi_\lambda\to \prod_i\Phi_\lambda(X_i)\rightrightarrows \prod_{i,j}\Phi_\lambda(X_i\times X_j),
\]
where $(X_i\to e)_{i\in I}$ is a finite epimorphic family. By 6.12.2 and the hypothesis on $E$, one may restrict to finite epimorphic families to verify (CL), and filtered inductive limits commute with finite projective limits (I, 2); hence exactness of these sequences of sets is preserved by passage to the inductive limit.
\item If $E$ is a coherent topos (VI, 2.3), then, by the preceding discussion, every family of supports of $E$ is of local character.
\end{enumerate}
\end{remarktext}

\subsection*{6.13}
Let $(E,A)$ be a ringed topos, let $\Phi$ be a family of supports of $E$, let $N$ be an $A$-module, say a left module to fix ideas, and let $X$ be an object of $E$. Put
\begin{align*}
        \Hh_\Phi^0(N)&=\varinjlim_{Z\in\Phi}\Hh_Z^0(N),\tag{6.13.1}\\
        H_\Phi^0(E,N)&=\varinjlim_{Z\in\Phi}H_Z^0(E,N).\tag{6.13.2}
\end{align*}
If $M$ is a left $A$-module, put
\begin{gather*}
        \Eext_{A,\Phi}^0(M,N)=\Hhom_{A,\Phi}(M,N)=\varinjlim_{Z\in\Phi}\Eext_{A,Z}^0(M,N),\tag{6.13.3}\\
        \Ext_{A,\Phi}^0(X;M,N)=\Hom_{A|X,\Phi(X)}(M|X,N|X)=\varinjlim_{Z\in\Phi}\Ext_{A,Z}^0(X;M,N).\tag{6.13.4}
\end{gather*}
There are canonical isomorphisms
\begin{equation*}
        \Eext_{A,\Phi}^0(M,N)\simeq \Hh_\Phi^0(\Eext_A^0(M,N)),\tag{6.13.5}
\end{equation*}
and
\begin{equation*}
        \Ext_{A,\Phi}^0(X;M,N)\simeq H_\Phi^0(X,\Eext_A^0(M,N)).\tag{6.13.6}
\end{equation*}
When $\Phi$ is the family of closed subtopoi contained in a closed subtopos $Z$, there are isomorphisms
\begin{equation*}
\begin{cases}
\Hh_\Phi^0\simeq \Hh_Z^0,\\
H_\Phi^0\simeq H_Z^0,\\
\Eext_{A,\Phi}^0(-,-)\simeq \Eext_{A,Z}^0(-,-),\\
\Ext_{A,\Phi}^0(X;-,-)\simeq \Ext_{A,Z}^0(X;-,-).
\end{cases}\tag{6.13.7}
\end{equation*}
The inductive limits defining the preceding functors are filtered. Hence these functors are left exact. Their right derived functors are denoted
\begin{equation*}
        \Hh_\Phi^q,
        \quad H_\Phi^q,
        \quad \Eext_{A,\Phi}^q(-,-),
        \quad \Ext_{A,\Phi}^q(X;-,-).\tag{6.13.8}
\end{equation*}
They too are computed by inductive limits of the functors $\Hh_Z^q$, $H_Z^q$, and so on, as $Z$ runs through $\Phi$.

From the isomorphisms (6.13.5) and (6.13.6), by passing to the inductive limit in the first spectral sequence of (6.9.1) and in the first spectral sequence of (6.9.2), one obtains two spectral sequences:
\begin{gather*}
        \Eext_{A,\Phi}^{p+q}(M,N)\Longleftarrow E_2^{p,q}=\Hh_\Phi^p(\Eext_A^q(M,N)),\tag{6.13.9}\\
        \Ext_{A,\Phi}^{p+q}(X;M,N)\Longleftarrow E_2^{p,q}=H_\Phi^p(X,\Eext_A^q(M,N)).\tag{6.13.10}
\end{gather*}

\subsection*{6.14}
With the notation of 6.13, and without any hypothesis on $\Phi$, there is a functorial morphism
\begin{equation*}
        \theta:H_\Phi^0(E,N)\to H^0(E,\Hh_\Phi^0(N)).\tag{6.14.1}
\end{equation*}
This morphism is always injective, but is not in general an isomorphism. However, if $\Phi$ is \emph{of local character}, then $\theta$ is an isomorphism, as is checked immediately. Similarly, if $\Phi$ is \emph{of local character}, there is an isomorphism
\begin{equation*}
        \theta':\Ext_{A,\Phi}^0(E;M,N)\simeq H^0(E,\Eext_{A,\Phi}^0(M,N)).\tag{6.14.2}
\end{equation*}

\begin{statement}{Proposition 6.15}
Let $(E,A)$ be a ringed topos such that $E$ is coherent \emph{(VI)}, let $\Phi$ be a family of supports of $E$, and let $M,N$ be two $A$-modules. There are two spectral sequences:
\begin{align*}
        H_\Phi^{p+q}(E,N)\Longleftarrow E_2^{p,q}&=H^p(E,\Hh_\Phi^q(N)),\tag{6.15.1}\\
        \Ext_{A,\Phi}^{p+q}(E;M,N)\Longleftarrow E_2^{p,q}&=H^p(E,\Eext_{A,\Phi}^q(M,N)).\tag{6.15.2}
\end{align*}
\end{statement}

These spectral sequences are deduced from (6.4.2) and from the second spectral sequence of (6.9.2) by passage to the inductive limit over the closed subtopoi $Z$ in $\Phi$, taking account of the fact that the cohomology of a coherent topos commutes with inductive limits of sheaves (IV, 5).

\subsection*{6.16}
Let us mention, without proof, that the second spectral sequence of (6.9.1) and the third spectral sequence of (6.9.2), with $X$ the final object of $E$, may be extended to the case of families of supports, provided the topos $E$ is coherent and the module $M$ is perfect [13].

Finally, one may also generalize the notion of a family of supports by introducing presheaves of families of supports, sheaves of families of supports, and the corresponding cohomology groups and sheaves.\footnote{Cf. [12], Chapter IV, \S1 (Lecture Notes in Mathematics 20, Springer) for the development of these notions in the form of a fugue with variations.}

\section{Appendix: \Cech\ cohomology}

Developing an idea due to P. Cartier, this section shows how, in an arbitrary topos, one may compute the cohomology of a sheaf by means of coverings. For the classical theory of paracompact spaces, see [7]; for another method applying to certain topoi, in particular to the \'{e}tale topos, see [14].

\subsection{Skeleton and coskeleton}

\subsubsection*{7.1.0}
Let $\Delta$ be the category of standard simplexes: its objects are the ordered sets
\[
        \Delta_n=[0,\ldots,n],
\]
and its morphisms are increasing maps. Let $E$ be a $\U$-topos, and let $\Delta(E)$ be the category of presheaves on $\Delta$ with values in $E$, in other words the category of \emph{semi-simplicial objects} of $E$. Denote by $\Delta[n]$ the full subcategory of $\Delta$ defined by the objects $\Delta_p$, $p\leq n$, by
\[
        i_n:\Delta[n]\to\Delta
\]
the inclusion functor, and by $\Delta E[n]$ the category of presheaves on $\Delta[n]$ with values in $E$, in other words the category of \emph{semi-simplicial objects of $E$ truncated at order $n$}. By I, 5.1, there is a triple of adjoint functors (I, 5.3)
\[
\xymatrix@C=3cm@M=.5cm{
\Delta E[n] \ar@<17pt>[r]^{i_{n!}} \ar@<-19pt>[r]_{i_{n*}}&
\Delta E \ar@<1pt>[l]_{i_n^*}
}
\]
where $i_n^*$ is the restriction functor to $\Delta[n]$, and where $i_{n*}$ and $i_{n!}$ are respectively its right and left adjoints. Note that $\Delta E$ and $\Delta E[n]$ are $\U$-topoi (IV, 1.2), and that $(i_n^*,i_{n*})$ is a morphism of topoi (IV, 3.1) which is an embedding of $\Delta E[n]$ into $\Delta E$ (I, 5.6 and IV, 9.1.1).

\begin{statement}{Definition 7.1.1}
The functor
\[
        \sk_n=i_{n!}i_n^*
\]
(\resp\ $\cosk_n=i_{n*}i_n^*$) from $\Delta E$ to $\Delta E$ is called the \emph{skeleton functor of order $n$} (\resp\ the \emph{coskeleton functor of order $n$}).
\end{statement}

Skeleton and coskeleton functors are in constant use in the theory of semi-simplicial sets [3].

\begin{statement}{Proposition 7.1.2}
\begin{enumerate}[label=\arabic*)]
\item The adjunction morphism $\sk_n\to\id$ is a monomorphism. The canonical morphisms
\[
        \sk_n\sk_m\to\sk_n
        \quad(\resp\ \cosk_n\to\cosk_n\cosk_m)
\]
are isomorphisms when $n\leq m$ (\resp\ $n\geq m$).
\item Let $u:E\to E'$ be a morphism of topoi. The functor $u^*$, extended to semi-simplicial and truncated semi-simplicial objects, commutes with the functors $i_{n!}$, $i_n^*$, $\sk_n$, and $\cosk_n$.
\end{enumerate}
\end{statement}

The first assertion follows immediately from the definitions. To prove 2), it suffices to observe that, by I, 5.1, the functors $i_{n!},\ldots,\cosk_n$ are computed by means of inductive limits and finite projective limits.

\subsection{An acyclicity lemma}

\subsubsection*{7.2.0}
Let $M$ be an abelian group of $E$. The \emph{homology} of a semi-simplicial object $K$ of $E$, with coefficients in $M$, is defined in the usual way. First form $A(K)$, the free abelian semi-simplicial group generated by $K$, and then form the tensor product $M\otimes_{\ZZ}A(K)$; this gives an abelian semi-simplicial group of $E$. One then considers the associated complex, obtained by taking the alternating sum of the boundary operators, and takes its homology. The homology objects are denoted
\[
        H_i(K,M).
\]
They are abelian groups in $E$. When $M=\ZZ_E$, the free abelian group generated by the final object of $E$, we write simply $H_i(K)$. The groups $H_i(K)$ are called the \emph{homology groups} of $K$. A semi-simplicial object $K$ is called \emph{acyclic} if, for every abelian group $M$ of $E$, the groups $H_i(K,M)$ vanish for $i>0$; equivalently, if the groups $H_i(K)$ vanish for $i>0$.

The formation of homology commutes with inverse images by morphisms of topoi.

The aim of this subsection is to prove the following lemma.

\begin{statement}{Lemma 7.2.1}
Let $E$ be a topos, let $K_\bullet$ and $L_\bullet$ be two semi-simplicial objects of $E$, let
\[
        v_\bullet:K_\bullet\to L_\bullet
\]
be a morphism of semi-simplicial objects, and let $n\geq 0$ be an integer. Assume that $v$ has the following properties:
\begin{enumerate}[label=\arabic*)]
\item for every integer $p<n$, $p\geq 0$, the morphism $v_p:K_p\to L_p$ is an isomorphism;
\item the morphism $v_n:K_n\to L_n$ is an epimorphism;
\item the canonical morphisms $K_\bullet\to \cosk_n(K_\bullet)$ and $L_\bullet\to \cosk_n(L_\bullet)$ are isomorphisms.
\end{enumerate}
Then, for every integer $p$, the morphism $v_p$ is an epimorphism, and the morphism $v_\bullet$ induces an isomorphism on homology objects.
\end{statement}

\begin{remarktext}{Remark}
The proof in fact shows that, with a suitable definition of homotopy in topoi, the morphism $v_\bullet$ is a homotopy equivalence.
\end{remarktext}

\begin{prooftext}
One may assume that $E$ is the topos of sheaves on a small site $C$ (IV, 1), and hence that $E$ is a subtopos of a topos $E'$ having enough fiber functors, for example the presheaf topos $C^{\wedge}$. Let
\[
        a^*:E'\to E
\]
be the inverse-image functor for the embedding morphism $E\hookrightarrow E'$. Let
\[
        v_\bullet[n]:K_\bullet[n]\to L_\bullet[n]
\]
be the morphism of truncated semi-simplicial objects obtained by truncating $v_\bullet$ at order $n$. Let $L_\bullet[n]'$ be the image, in $E'$, of $K_\bullet[n]$ by $v_\bullet[n]$, and let
\[
        v_\bullet[n]':K_\bullet[n]\to L_\bullet[n]'
\]
be the morphism induced by $v_\bullet[n]$. Put
\[
        L!=i_{n*}L_\bullet[n]',\qquad K!=i_{n*}K_\bullet[n],\qquad v!=i_{n*}v_\bullet[n]',
\]
where here $i_{n*}$ is relative to $E'$. By 1), the morphism $v!:K!\to L!$ has properties 1), 2), and 3) of the lemma. Moreover, by 2), 3), and 7.1.2, the morphism $a^*v!$ is none other than $v_\bullet$. It is therefore enough to prove the lemma for $v!$. Thus one may assume that $E$ has enough fiber functors, and by using these fiber functors one is reduced to the case where $E$ is the category of sets.
\end{prooftext}

First observe that the hypotheses of the lemma on the morphism $v_\bullet$ are stable under every base change $M_\bullet\to L_\bullet$ where $M_\bullet$ is a coskeleton of order $n$. Hence the fiber $P_\bullet$ of $v_\bullet$ at an arbitrary base point of $L_\bullet$ is of the form $i_{n*}P_\bullet[n]$, where $P_\bullet[n]$ is a non-empty complex truncated at order $n$ of the form
\begin{equation*}
        P_n\to e\to e\to e\to\cdots\to e.\tag{7.2.2}
\end{equation*}
For every integer $i$, let $\delta\Delta_i$ be the semi-simplicial boundary of the simplex $\Delta_i$. Every morphism from $\delta\Delta_i$ to $P_\bullet$ extends to a morphism from $\Delta_i$ to $P_\bullet$. Indeed, this is evident if $i\geq n$, since $P_\bullet$ is a coskeleton of order $n$, and it is clear for $i<n$ from the description (7.2.2). Since $P_\bullet$ is a Kan complex, $P_\bullet$ is homotopically trivial [6]. Note that $v_\bullet$ is a fibration in the sense of Kan [6]. Since the fibers of this fibration are homotopically trivial, $v_\bullet$ induces an isomorphism on the homotopy groups of $K_\bullet$ and $L_\bullet$ at every base point of $K_\bullet$. By the Hurewicz theorem, this implies that $v_\bullet$ induces an isomorphism on homology [6]. This proves the second assertion of the lemma. To prove the first assertion, note that every morphism from a complex truncated at order $n$ into $L_\bullet[n]$ lifts to a morphism into $K_\bullet[n]$. Hence, by 3), every morphism from a complex into $L_\bullet$ lifts to a morphism into $K_\bullet$. Thus $v_\bullet$ is surjective.

\subsection{Hypercoverings}

\subsubsection*{7.3.0}
Let $C$ be a site belonging to the universe $\U$, and suppose that fiber products and products of two objects are representable. Denote by $C^{\wedge}$ the topos of $\U$-presheaves on $C$, and by $C^{\sim}$ the topos of $\U$-sheaves on $C$. An object $K$ of $C^{\wedge}$ is called \emph{semi-representable} if it is isomorphic to a direct sum of representable presheaves.

Let $\SR(C)$ be the full subcategory of $C^{\wedge}$ defined by the semi-representable objects. In $\SR(C)$, projective limits indexed by finite non-empty categories are representable, and the inclusion functor $\SR(C)\hookrightarrow C^{\wedge}$ commutes with these finite projective limits. By a remark already made, it follows that if $K_\bullet$ is a semi-simplicial object truncated at order $n$ in $C^{\wedge}$ and all its objects are semi-representable, then the prolongation $i_n^+(K_\bullet)$ has the same property. Hence, if $K_\bullet$ is a semi-simplicial object of $C^{\wedge}$ all of whose objects are semi-representable, then, for every $n$, the object $\cosk_n(K_\bullet)$ has the same property.

\subsubsection{Definitions and notation}

\paragraph*{7.3.1.1}
Let $p>0$ be an integer. A semi-simplicial object $K_\bullet$ of $C^{\wedge}$ is called a \emph{hypercovering of type $p$} of $C$, or, when no confusion can result, a hypercovering of type $p$, if it has the following properties:
\begin{itemize}
\item[HR1)] For every integer $n\geq 0$, $K_n$ is semi-representable.
\item[HR2)] The canonical morphism $K_\bullet\to\cosk_p(K_\bullet)$ is an isomorphism.
\item[HR3)] For every integer $n\geq 0$, the canonical morphism of presheaves
\[
        K_{n+1}\to (\cosk_n(K_\bullet))_{n+1}
\]
is a covering isomorphism of presheaves (II, 6.2). The canonical morphism of presheaves $K_0\to e$, where $e$ is the final object of $C^{\wedge}$, is a covering morphism of presheaves.
\end{itemize}

\paragraph*{7.3.1.2}
A hypercovering of type $p$ is a hypercovering of type $q$ for every $q\geq p$. A semi-simplicial object $K_\bullet$ will be called a \emph{hypercovering} of $C$ if it has properties HR1) and HR3).

\paragraph*{7.3.1.3}
We denote by $\HR_p$ (\resp\ $\HR$), and call the \emph{category of hypercoverings of type $p$} (\resp\ the \emph{category of hypercoverings}), the following category:
\begin{enumerate}[label=\alph*)]
\item the objects of $\HR_p$ (\resp\ $\HR$) are the hypercoverings of type $p$ (\resp\ the hypercoverings);
\item if $K_\bullet$ and $L_\bullet$ are two objects of $\HR_p$ (\resp\ $\HR$), a morphism from $K_\bullet$ to $L_\bullet$ is a morphism $v_\bullet:K_\bullet\to L_\bullet$ of semi-simplicial objects of $C^{\wedge}$.
\end{enumerate}

\paragraph*{7.3.1.4}
Let $C$ be a site in which fiber products are representable, and let $X$ be an object of $C$. A hypercovering of $X$ (\resp\ a hypercovering of type $p$ of $X$) means a hypercovering of the site $C/X$ (\resp\ a hypercovering of type $p$ of $C/X$). The category of hypercoverings of $X$ (\resp\ hypercoverings of type $p$ of $X$) is defined similarly. There is a sequence of inclusion functors
\[
        \cdots\HR_p\hookrightarrow \HR_{p+1}\hookrightarrow\cdots\hookrightarrow \HR .
\]
These functors are fully faithful. We denote by $\HR_\infty$ the inductive limit of the categories $\HR_p$.

\paragraph*{7.3.1.6}
For every integer $n\geq 0$, let $\Delta_n^c$ denote the standard semi-simplicial object of dimension $n$ with values in the category of sets, i.e. the functor on $\Delta$ represented by the ordered set $[0,n]$. We also denote by $\underline{\Delta}_n^c$ the presheaf on $C$ with values in semi-simplicial sets which is constant with value $\Delta_n$. Thus $\underline{\Delta}_n^c$ is a presheaf of semi-simplicial sets. The two canonical injections of $\Delta_0$ into $\Delta_1$ define two morphisms of semi-simplicial presheaves
\[
\xymatrix@C=2cm@M=.25cm{
\underline{\Delta}_0^c \ar@<3pt>[r]^{e_0} \ar@<-3pt>[r]_{e_1} & \underline{\Delta}_1^c,
}
\]
and consequently, for every semi-simplicial presheaf $K_\bullet$, two canonical injections
\[
\xymatrix@C=2cm@M=.25cm{
K_\bullet \ar@<3pt>[r]^{e_0} \ar@<-3pt>[r]_{e_1} & K_\bullet\times \underline{\Delta}_1^c .
}
\]
Two morphisms
\[
\xymatrix@C=1cm@M=.25cm{
K_\bullet \ar@<3pt>[r]^{u_0} \ar@<-3pt>[r]_{u_1} & L_\bullet
}
\]
of semi-simplicial presheaves are said to be \emph{homotopic} if there exists a morphism
\[
        v:K_\bullet\times \underline{\Delta}_1^c\to L_\bullet
\]
such that the diagrams
\[
\xymatrix@R=.5cm@C=1.25cm{
K_\bullet \ar[rr]^{e_i} \ar[rdd]_{u_i}&&K_\bullet\times\underline{\Delta}_1^c\ar[ldd]^v\\
&&&\txt{$i=0,1$}\\
&L_\bullet&
}
\]
commute. The morphism $v$ is called a \emph{homotopy joining $u_0$ to $u_1$}. The relation that $u_0$ and $u_1$ are homotopic morphisms from $K_\bullet$ to $L_\bullet$ is not, in general, an equivalence relation. However, the equivalence relation generated by it is compatible with composition of morphisms. This allows us to define the category of semi-simplicial presheaves up to homotopy by quotienting by the equivalence relation generated by homotopy.

\paragraph*{7.3.1.7}
In this way one defines the categories $\sheaf{HR}_p$, $\sheaf{HR}_\infty$, and $\sheaf{HR}$, the \emph{categories of hypercoverings up to homotopy}.

\begin{statement}{Theorem 7.3.2}
Let $C$ be a site satisfying the conditions of 7.3.0.
\begin{enumerate}[label=\arabic*)]
\item The category $\sheaf{HR}_p^0$ (\resp\ $\sheaf{HR}^0$) is filtered \emph{(I, 2.7)}.
\item Let $K_\bullet$ be an object of $\HR_p$ (\resp\ of $\HR$), let $n$ be an integer with $0\leq n\leq p$ (\resp\ $0\leq n$), and let
\[
        u:X\to K_n
\]
be a covering morphism of presheaves. There exists an object $L_\bullet$ of $\HR_p$ (\resp\ of $\HR$) and a morphism $f:L_\bullet\to K_\bullet$ such that
\[
        f_n:L_n\to K_n
\]
factors through $u$.
\item The semi-simplicial sheaves associated \emph{(II, 3.5)} with hypercoverings are acyclic (7.2.0) in strictly positive degrees. The zeroth homology sheaf is isomorphic to the sheaf associated with the constant sheaf $\ZZ$.
\end{enumerate}
\end{statement}

\begin{prooftext}
We first prove 3). Let $K_\bullet^{\sim}$ be the semi-simplicial sheaf associated with $K_\bullet$. The associated-sheaf functor commutes with $\cosk_n$ (7.1.2.2)). Hence $K_\bullet^{\sim}$ has the following properties:
\begin{enumerate}[label=\alph*)]
\item the canonical morphism $K_0\to e$ is an epimorphism of sheaves;
\item for every integer $n\geq 0$, the canonical morphism
\[
        K_{n+1}\to (\cosk_n(K_\bullet))_{n+1}
\]
is an epimorphism of sheaves.
\end{enumerate}
\end{prooftext}

For $n\geq 0$, put $L_{\bullet,n}=\cosk_nK_\bullet^{\sim}$. Then, for every $n$, there is a sequence of morphisms of semi-simplicial sheaves
\[
        K_\bullet^{\sim}\xrightarrow{u_n}L_{\bullet,n}\xrightarrow{v_n}L_{\bullet,n-1}\cdots\to L_{\bullet,0}\xrightarrow{v_0}e_\bullet,
\]
and, for every $j$, $0\leq j\leq n$, the morphism $v_j$ has the properties of 7.2.1. Moreover, the morphism $u_n$ induces an isomorphism on components of degree $\leq n$. It follows from 7.2.1, by induction on $n$, that $K_\bullet$ is acyclic.

To prove 1) and 2), we introduce terminology.

\begin{statement}{Definition 7.3.3}
Let $f:X_\bullet\to Y_\bullet$ be a morphism of semi-simplicial presheaves. One says that $f$ is \emph{special of type $p$} (\resp\ \emph{special}) if:
\begin{enumerate}[label=\arabic*)]
\item The objects $X_\bullet$ and $Y_\bullet$ are canonically isomorphic to their coskeleta of order $p$, and $\cosk_p(f)=f$ (\resp\ no condition is imposed on $f$, $X_\bullet$, or $Y_\bullet$).
\item For every integer $n$ such that $0\leq n\leq p$ (\resp\ $0\leq n$), the morphism $\Phi_{n+1}$ appearing in the diagram below is covering:
\[
\xymatrix@C=1cm@R=1.25cm{
X_{n+1} \ar[rrr]^{f_{n+1}} \ar[rd]^{\Phi_{n+1}} \ar[dd]&&&Y_{n+1}\ar[dd]\\
&P_{n+1}\ar[ld]\ar[rru]&&\\
(\cosk_nX_\bullet)_{n+1}\ar[rrr]_{(\cosk_nf)_{n+1}}&&&(\cosk_nY_\bullet)_{n+1}.
}
\]
Here the vertical arrows are defined by the canonical morphisms $X_\bullet\to\cosk_nX_\bullet$ and $Y_\bullet\to\cosk_nY_\bullet$; the object $P_{n+1}$ is the fiber product, and $\Phi_{n+1}$ is the unique arrow making the diagram commute.
\item The morphism $f_0:X_0\to Y_0$ is covering.
\end{enumerate}
A semi-simplicial presheaf $K_\bullet$ is said to be special of type $p$ (\resp\ special) if the canonical morphism
\[
        K_\bullet\to e_\bullet
        \qquad(e_\bullet\text{ is the final semi-simplicial object})
\]
is special of type $p$ (\resp\ special), i.e. if $K_\bullet$ satisfies conditions HR2) and HR3) (\resp\ HR3)) of 7.3.1.1.
\end{statement}

\begin{statement}{Lemma 7.3.4}
\begin{enumerate}[label=\arabic*)]
\item The composite of two special morphisms (\resp\ special morphisms of type $p$) is special (\resp\ special of type $p$).
\item Let $K_\bullet$ be a special semi-simplicial presheaf (\resp\ special of type $p$), let $X_\bullet\xrightarrow{f}Y_\bullet$ be a special morphism (\resp\ special of type $p$), and let $u:K_\bullet\to Y_\bullet$ be a morphism of complexes. Then the fiber product
\[
        P_\bullet=K_\bullet\times_{Y_\bullet}X_\bullet
\]
is special (\resp\ special of type $p$).
\end{enumerate}
\end{statement}

The proof of this lemma is left to the reader.

\subsubsection*{7.3.5. Proof of assertion 2) of 7.3.2}
One may first assume that $X$ is semi-representable. For every semi-simplicial object $L_\bullet$, let $\Hom_{(X)}(L_\bullet,K_\bullet)$ denote the set of morphisms of semi-simplicial objects equipped with a factorization
\[
        L_n\to X\xrightarrow{u}K_n.
\]
The functor $\Hom_{(X)}(-,K_\bullet)$ is representable. Indeed, this functor sends inductive limits to projective limits, and $\Delta(C^{\wedge})$ is a topos. We denote by $P$ an object representing it. Similarly, for an object $Z$ of $C^{\wedge}$, denote by $j_n^+(Z)$ the object representing the functor
\[
        L_\bullet\longmapsto \Hom(L_n,Z)
\]
on $\Delta C^{\wedge}$; its construction by $\varprojlim$ is made explicit in III, 1.1. Here one observes that this construction involves only finite projective limits. Thus $j_n^+$ sends semi-representable presheaves to semi-simplicial presheaves with semi-representable components, and covering morphisms $Z'\to Z$ to special morphisms (7.3.3).

By the definition of $P_\bullet$, there is a cartesian square
\[
\xymatrix@C=2cm@R=1.5cm{
P_\bullet\ar[r]\ar[d]&j_n^+(X)\ar[d]^{j_n^+(u)}\\
K_\bullet\ar[r]&j_n^+(K_n).
}
\]
By what has just been observed, the semi-simplicial objects $j_n^+(X)$ and $j_n^+(K_n)$ are semi-representable, and the morphism $j_n^+(u)$ is special of type $n$. It remains only to apply 3.4 and the formalities of 7.3.0.

\subsubsection*{7.3.6.0}
Let $M_\bullet$ and $N_\bullet$ be two semi-simplicial presheaves. The functor
\[
        L_\bullet\longmapsto \Hom(L_\bullet\times M_\bullet,N_\bullet)
\]
is representable. The semi-simplicial presheaf representing it will be denoted
\[
        \Hhom_\bullet(M_\bullet,N_\bullet).
\]
The degree-$n$ component of this object will be denoted $\Hhom_n(M_\bullet,N_\bullet)$. There is a canonical isomorphism
\[
        \Hhom_n(M_\bullet,N_\bullet)\xrightarrow{\sim}\Hhom_0(M_\bullet\times\Delta_n^c,N_\bullet).
\]

\begin{statement}{Lemma 7.3.6}
Let
\[
\xymatrix@C=2cm@R=1.5cm{
M_\bullet\ar@{^{(}->}[r]&N_\bullet\\
\sk_n\Delta_{n+1}^c\ar[u]\ar@{^{(}->}[r]^{u}&\Delta_{n+1}^c\ar[u]&\kern-1.5cm\txt{($u$ the canonical injection)}
}
\]
be a cocartesian diagram of semi-simplicial presheaves. For every hypercovering $L_\bullet$, the morphism
\[
        \Hhom_0(N_\bullet,L_\bullet)\to \Hhom_0(M_\bullet,L_\bullet)
\]
is covering.
\end{statement}

\begin{prooftext}
There is a cartesian diagram
\[
\xymatrix{
\Hhom_0(N_\bullet,L_\bullet)\ar[d]\ar[r]&\Hhom_0(M_\bullet,L_\bullet)\ar[d]\\
\Hhom_0(\Delta_{n+1}^c,L_\bullet)\ar[r]^{(*)}&\Hhom_0(\sk_n\Delta_{n+1}^c,L_\bullet).
}
\]
It is therefore enough to show that the morphism $(*)$ is covering. But $(*)$ is isomorphic to the morphism
\[
        L_{n+1}\to (\cosk_nL_\bullet)_{n+1},
\]
which is covering by hypothesis.
\end{prooftext}

\subsubsection*{7.3.7. Proof of assertion 1) of 7.3.2}
The product of two objects of $\HR_p$ (\resp\ $\HR$) is again an object of $\HR_p$ (\resp\ of $\HR$), by 7.3.4. To prove that the category $\sheaf{HR}_p^0$ (\resp\ $\sheaf{HR}^0$) is filtered, it is enough to show that, given two morphisms in $\HR_p$ (\resp\ $\HR$)
\[
\xymatrix@C=2cm@M=.2cm{
K_\bullet\ar@<3pt>[r]^{u_0}\ar@<-3pt>[r]_{u_1}&L_\bullet,
}
\]
there exists a morphism $v:M_\bullet\to K_\bullet$ in $\HR_p$ (\resp\ in $\HR$) such that the morphisms
\[
\xymatrix@C=2cm@M=.2cm{
M_\bullet\ar@<3pt>[r]^{u_0v}\ar@<-3pt>[r]_{u_1v}&L_\bullet
}
\]
are homotopic, i.e. such that there is a $w:M_\bullet\times\Delta_1^c\to L_\bullet$ making the diagrams
\[
\xymatrix@C=2cm@R=1.5cm{
M_\bullet\ar[d]_v\ar[r]^{e_i}&M_\bullet\times\Delta_1^c\ar[d]^w\\
K_\bullet\ar[r]^{u_i}&L_\bullet&\kern-1.5cm\txt{$i=0,1$}
}
\tag{x}
\]
commute.

For every semi-simplicial object $M_\bullet$ of $C^{\wedge}$, not necessarily a hypercovering, let $F(M_\bullet)$ be the set of pairs $(v,w)$ with $v:M_\bullet\to K_\bullet$ and $w:M_\bullet\times\Delta_1^c\to L_\bullet$ such that the diagrams (x) commute. The functor $M_\bullet\mapsto F(M_\bullet)$ is contravariant in $M_\bullet$, sends inductive limits to projective limits, and hence is representable by a semi-simplicial object $F_\bullet$. Evidently $F_\bullet$ is the vertex of a cartesian diagram
\[
\xymatrix@=2cm{
F_\bullet\ar[r]\ar[d]&\Hhom_\bullet(\Delta_1^c,L_\bullet)\ar[d]^\pi\\
K_\bullet\ar[r]_{u_0,u_1}&L_\bullet\times L_\bullet,
}
\]
where $\pi$ is defined by the two inclusions of $\Delta_0^c$ into $\Delta_1^c$. To prove the assertion, it is therefore enough to show that $F_\bullet$ is an object of $\HR_p$ (\resp\ of $\HR$). By 7.3.4.2) and the formalities of 7.3.0, it is enough to show that
\begin{enumerate}[label=\alph*)]
\item $\Hhom_\bullet(\Delta_1^c,L_\bullet)$ is semi-representable;
\item the morphism $\pi$ is special of type $p$ (\resp\ special).
\end{enumerate}
First, one verifies immediately that $\Hhom_\bullet(\Delta_1^c,L_\bullet)$ is a semi-representable semi-simplicial object. Indeed, its components are computed by finite projective limits from the $L_n$, and are therefore semi-representable. Let us verify b). If $L_\bullet$ is special of type $p$, first one shows that the morphism
\[
        \Hhom_\bullet(\Delta_1^c,L_\bullet)\to\cosk_p\Hhom_\bullet(\Delta_1^c,L_\bullet)
\]
is an isomorphism; this verifies property 1) of 7.3.3. Next, the morphism
\[
        \pi_0:\Hhom_0(\Delta_1^c,L_\bullet)\to L_0\times L_0
\]
is isomorphic to the canonical morphism
\[
        L_1\xrightarrow{(d_0,d_1)}L_0\times L_0,
\]
which is covering by hypothesis, since $L_\bullet$ is a hypercovering.

It remains to verify property 2) of 7.3.3, i.e. to verify that, for all $n$, in the diagram
\[
\xymatrix@C=.75cm@R=1.25cm{
\Hhom_{n+1}(\Delta_1^c,L_\bullet)\ar[rr]^{\pi_{n+1}}\ar[dd]\ar[rd]_{\Phi_{n+1}}&&L_{n+1}\times L_{n+1}\ar[dd]\\
&P_{n+1}\ar[ru]\ar[ld]&\\
(\cosk_n\Hhom_\bullet(\Delta_1^c,L_\bullet))_{n+1}\ar[rr]^{(\cosk_n\pi)_{n+1}}&&
(\cosk_nL_\bullet)_{n+1}\times(\cosk_nL_\bullet)_{n+1}
}
\]
where $P_{n+1}$ is the fiber product, the morphism $\Phi_{n+1}$ is covering. A straightforward adjoint-functor chase shows that
\begin{align*}
P_{n+1}&\xrightarrow{\sim}\Hhom_0(\sk_{n+1}(\Delta_{n+1}^c\times\Delta_1^c),L_\bullet),\\
\Hhom_{n+1}(\Delta_1^c,L_\bullet)&\xrightarrow{\sim}\Hhom_0(\Delta_{n+1}^c\times\Delta_1^c,L_\bullet),
\end{align*}
and that $\Phi_{n+1}$ is isomorphic to the morphism
\[
\Hhom_0(\Delta_{n+1}^c\times\Delta_1^c,L_\bullet)
\to
\Hhom_0(\sk_{n+1}(\Delta_{n+1}^c\times\Delta_1^c),L_\bullet)
\]
coming from the injection
\[
        \sk_{n+1}(\Delta_{n+1}^c\times\Delta_1^c)\hookrightarrow \Delta_{n+1}^c\times\Delta_1^c.
\]
Now there is a sequence of subobjects of $\Delta_{n+1}^c\times\Delta_1^c$,
\[
\sk_{n+1}(\Delta_{n+1}^c\times\Delta_1^c)=M_0\hookrightarrow M_1\hookrightarrow\cdots\hookrightarrow M_k=\Delta_{n+1}^c\times\Delta_1^c,
\]
such that, for every $0\leq i\leq k$, the morphism
\[
        M_i\hookrightarrow M_{i+1}
\]
fits into a cocartesian diagram
\[
\xymatrix@C=1.5cm@R=1.5cm{
M_i\ar@{^{(}->}[r]&M_{i+1}\\
\sk_{n+1}\Delta_{n+2}^c\ar@{^{(}->}[r]\ar[u]&\Delta_{n+2}^c\ar[u].
}
\]
One adds one after another the non-degenerate simplexes of dimension $n+2$ of $\Delta_{n+1}\times\Delta_1$. By 7.3.6, the morphism $\Phi_{n+1}$ is a composite of covering morphisms, hence is itself covering. This completes the proof of 7.3.2.

\subsection{The isomorphism theorem}

\subsubsection*{7.4.0}
Let $F$ be a presheaf of abelian groups on a site $C$ satisfying the condition of 7.3.0. For every hypercovering $K_\bullet$, put
\[
        H^q(K_\bullet,F)=H^q(\Hom_{C^{\wedge}}(K_\bullet,F)).
\]
The $H^q(K_\bullet,F)$ form a $\delta$-functor on the category of abelian presheaves on $C$, but in general they are not the derived functors of $H^0(K_\bullet,F)$.

Put
\begin{equation*}
        \clH{r}{q}(C,F)=\varinjlim_{\HR_r}H^q(-,F)
        \qquad (r\text{ may be infinite}),\tag{7.4.0.1}
\end{equation*}
and
\begin{equation*}
        \clHall{q}(C,F)=\varinjlim_{\HR}H^q(-,F).\tag{7.4.0.2}
\end{equation*}
The $\clH{r}{q}(C,F)$ (\resp\ $\clHall{q}(C,F)$) still form a $\delta$-functor, because the category $\sheaf{HR}_r^0$ (\resp\ $\sheaf{HR}^0$) is filtered (7.3.2). Since the categories $\sheaf{HR}_p$ are not necessarily $\U$-categories, these $\delta$-functors take values in the category of $\V$-abelian groups, for a suitable universe $\V$.

Assume now that the presheaf $F$ is a sheaf. Since the semi-simplicial sheaf associated with a hypercovering is acyclic (7.3.2), there is a spectral sequence, functorial in $F$ and in $K_\bullet$,
\begin{equation*}
        H^p(K_\bullet,\Hh^q(F))\Longrightarrow H^{p+q}(C^{\sim},F).\tag{7.4.0.3}
\end{equation*}
Passing to the limit gives spectral sequences
\begin{equation*}
\begin{split}
        \clH{r}{p}(C,\Hh^q(F))&\Longrightarrow H^{p+q}(C^{\sim},F),\\
        \clHall{p}(C,\Hh^q(F))&\Longrightarrow H^{p+q}(C^{\sim},F).
\end{split}\tag{7.4.0.4}
\end{equation*}

\begin{statement}{Theorem 7.4.1}
Let $C$ be a site satisfying the condition of 7.3.0, and let $F$ be an abelian sheaf on $C$.
\begin{enumerate}[label=\arabic*)]
\item The spectral sequences (7.4.0.4) define isomorphisms
\[
        \clH{r}{q}(C,F)\xrightarrow{\sim}H^q(C^{\sim},F),\qquad q\leq r+1,
\]
and a monomorphism
\[
        \clH{r}{r+2}(C,F)\longrightarrow H^{r+2}(C^{\sim},F).
\]
\item There is an isomorphism of $\delta$-functors
\[
        \clHi{q}(C,F)\xrightarrow{\sim}\clHall{q}(C,F)\xrightarrow{\sim}H^q(C^{\sim},F),\qquad \text{for all }q.
\]
\item Let $G$ be a presheaf of abelian groups and let $F$ be the associated sheaf. There are isomorphisms of functors
\[
        \clH{r}{q}(C,G)\xrightarrow{\sim}H^q(C^{\sim},F),\qquad q\leq r-1,
\]
and a monomorphism
\[
        \clH{r}{r}(C,G)\longrightarrow H^r(C^{\sim},F).
\]
When $G$ is a separated presheaf, this latter morphism is an isomorphism, and there is a monomorphism
\[
        \clH{r}{r+1}(C,G)\longrightarrow H^{r+1}(C^{\sim},F).
\]
\item There are isomorphisms
\[
        \clHi{q}(C,G)\xrightarrow{\sim}\clHall{q}(C,G)\xrightarrow{\sim}H^q(C^{\sim},F),\qquad \text{for all }q.
\]
\end{enumerate}
\end{statement}

\begin{prooftext}
It is enough to show that, if $N$ is an abelian presheaf whose associated sheaf is zero, then
\[
        \clH{r}{q}(C,N)=0\quad\text{for }q\leq r
        \qquad(\resp\ \clHall{q}(C,N)=0\text{ for all }q),
\]
which is done immediately using 3.2.
\end{prooftext}

\begin{remarktext}{Remark 7.4.2}
\begin{enumerate}[label=\arabic*)]
\item Note that, for sites satisfying the condition of 7.3.0, hypercoverings of type $0$ are ordinary coverings, and the functors $\clHi{q}$ are none other than the functors $\check H^q$ introduced in (2.4.5.1).
\item Let $C$ be a site with representable finite projective limits such that, for every object $X$ of $C$ and every covering family $(X_i\to X)_{i\in I}$, there is an $i_0\in I$ such that $X_{i_0}\to X$ is covering. Then one can show that the hypercoverings of type $p$ whose components are representable are cofinal in $\sheaf{HR}_p$. Similarly, hypercoverings $K_\bullet$ for which there exists an integer $p$ such that $K_\bullet$ is of type $p$, and whose components are representable, are cofinal in $\sheaf{HR}$. This can be proved by following 3.5. These hypercoverings with representable components therefore suffice, in the case under consideration, to compute the cohomology of the sheaves associated with presheaves.
\end{enumerate}
\end{remarktext}


\section{Appendix: Local inductive limits (by P. Deligne)}

The editor recommends that the reader, as a rule, avoid reading this appendix. It presents a technique which sometimes makes it possible to extend to topoi without enough points assertions which are made trivial by the existence of points. This technique gives a sheaf-theoretic variant of a theorem of D. Lazard asserting that flat modules over a ring are inductive limits of finite free modules.

\subsection{Locally filtered categories}

\subsubsection*{8.1.0}
If $\Bb$ is a category and $p:\Aa\to\Bb$ is a category over $\Bb$, we shall use the following notation. For $U\in\ob(\Bb)$, $\Aa_U$ is the fiber category $p^{-1}(U)$. For $f:U\to V$ in $\Bb$ and objects $\lambda,\mu\in\Aa$ with $p(\lambda)=U$ and $p(\mu)=V$, put
\[
        \Hom_f(\lambda,\mu)=p^{-1}(f),\qquad
        p:\Hom(\lambda,\mu)\to\Hom(U,V).
\]

\begin{statement}{Definition 8.1.1}
Let $\Ss$ be a site. A \emph{locally cofiltered category} over $\Ss$ (or \emph{left locally filtered category}) is a category $p:\Ll\to\Ss$ over $\Ss$ satisfying:
\begin{itemize}
\item[($\Ll0$)] for every $f:U\to V$ in $\Ss$ and every $\mu\in\ob(\Ll_V)$, locally on $U$ there is an object $\lambda\in\ob(\Ll_U)$ with $\Hom_f(\lambda,\mu)\neq\varnothing$;
\item[($\Ll1$)] for every $U\in\ob(\Ss)$ and every finite family $(\lambda_i)$ of objects of $\Ll_U$, there is a covering $f_j:U_j\to U$ and objects $\mu_j\in\ob(\Ll_{U_j})$ such that, for all $i,j$, one has $\Hom_{f_j}(\mu_j,\lambda_i)\neq\varnothing$;
\item[($\Ll2$)] for every $f:U\to V$ in $\Ss$ and every double arrow $(\phi_0,
\phi_1):\lambda\rightrightarrows\mu$ above $f$, there is a covering $f_j:U_j\to U$ and arrows $\psi_j$ with target $\lambda$, lying above $f_j$, such that $\phi_0\psi_j=\phi_1\psi_j$.
\end{itemize}
\end{statement}

\paragraph*{8.1.1.1}
This definition simplifies when $\Ll$ is fibered over $\Ss$. Then axiom $(\Ll0)$ is satisfied, and $(\Ll1)$ and $(\Ll2)$ may be stated as follows:
\begin{itemize}
\item[($\Ll'1$)] for every $U\in\ob(\Ss)$ and every finite family $\lambda_i$ of objects of $\Ll_U$, locally on $U$ there is an object $\mu$ mapping to all the $\lambda_i$;
\item[($\Ll'2$)] for every $U\in\ob(\Ss)$, every double arrow $(\phi_0,
\phi_1):\lambda\rightrightarrows\mu$ in $\Ll_U$ can, locally on $U$, be equalized by an arrow with target $\lambda$.
\end{itemize}

\subsubsection*{8.1.2}
Let $\Ss$ be a site, let $\Ll$ be a category over $\Ss$, and let $\mathsf e$ be a stack over $\Ss$. Denote by $\Gamma\mathsf e$ the category of global sections $\Hhom_{\Ss}^{\mathrm{cart}}(\Ss,\mathsf e)$ of $\mathsf e$. If $\Ss$ has a final object $S$, then $\Gamma\mathsf e$ is, up to the usual convention, just $\mathsf e_S$.

\begin{statement}{Definition 8.1.2.1}
The category of \emph{local projective systems of objects of $\mathsf e$, indexed by $\Ll$}, is the category
\[
        \Hhom_{\Ss}(\Ll,\mathsf e).
\]
\end{statement}

Let $c$ denote the composite functor, ``associated constant local projective system'',
\[
\Gamma\mathsf e=\Hhom_{\Ss}^{\mathrm{cart}}(\Ss,\mathsf e)
\hookrightarrow \Hhom_{\Ss}(\Ss,\mathsf e)
\xrightarrow{u\mapsto u\circ p}\Hhom_{\Ss}(\Ll,\mathsf e).
\]
When the dependence on $\Ll$ has to be made explicit, we write $c_\Ll$.

\begin{statement}{Definition 8.1.3}
The \emph{local projective-limit functor}, denoted $\varprojlim_{\lambda\in\Ll}$, is the partially defined functor right adjoint to $c$.
\end{statement}

Thus
\[
        \Hom(X,\varprojlim_{\lambda\in\Ll}X_\lambda)
        \simeq
        \Hom(c(X),(X_\lambda)_{\lambda\in\Ll}).
\]

\begin{statement}{Definition 8.1.4}
An $\Ss$-functor $F:\Ll\to\Mm$ between locally cofiltered categories over $\Ss$ is called \emph{cofinal} if it satisfies:
\begin{itemize}
\item[($C\mathfrak F1$)] for every $U\in\ob(\Ss)$ and every $\mu\in\ob(\Mm_U)$, there is a covering $f_j:U_j\to U$ and objects $\lambda_j\in\ob(\Ll_{U_j})$ such that $\Hom_{f_j}(F(\lambda_j),\mu)\neq\varnothing$;
\item[($C\mathfrak F2$)] for every $f:U\to V$ in $\Ss$ and every double arrow $(\phi_0,
\phi_1):F(\lambda)\rightrightarrows\mu$ above $f$, there is a covering $f_j:U_j\to U$ and arrows $\psi_j$ with target $\lambda$ in $\Ll$, with $p(\psi_j)=f_j$, such that $\phi_0F(\psi_j)=\phi_1F(\psi_j)$.
\end{itemize}
\end{statement}

The composite of two cofinal functors is cofinal; equivalences of locally filtered categories are cofinal. The proof is left to the reader.

\begin{statement}{Proposition 8.1.5}
Let $F:\Ll\to\Mm$ be a cofinal functor between locally cofiltered categories over $\Ss$, let $\mathsf e$ be a stack over $\Ss$, and let $(X_\mu)_{\mu\in\Mm}$ be a local projective system indexed by $\Mm$. Then the morphism of functors, in $X\in\Gamma\mathsf e$ and with values in sets,
\[
F^*:\Hom(c_\Mm(X),(X_\mu)_{\mu\in\Mm})
\longrightarrow
\Hom(c_\Ll(X),(X_{F(\lambda)})_{\lambda\in\Ll})
\]
is an isomorphism. Hence
\[
        F^*:\varprojlim_\Mm X_\mu\longrightarrow
        \varprojlim_\Ll X_{F(\lambda)}
\]
is an isomorphism, the two sides being defined or undefined simultaneously.
\end{statement}

The proof uses the following lemma, left to the reader.

\begin{statement}{Lemma 8.1.6}
If $F:\Ll\to\Mm$ is cofinal, then, for every $f:U\to V$ in $\Ss$ and every pair of arrows above $f$,
\[
        \phi_i:F(\lambda_i')\to\mu,
        \qquad i=1,2,
\]
there is a covering $f_j:U_j\to U$, objects $\lambda_j'\in\ob(\Ll_{U_j})$, and commutative diagrams
\[
\xymatrix@=1.25cm{
&F(\lambda_1')\ar[rd]^{\phi_1}&\\
F(\lambda_j')\ar@{--}[ru]\ar@{--}[rd]&&\mu\\
&F(\lambda_2')\ar[ru]_{\phi_2}&
}
\]
lying above $U_j\xrightarrow{f_j}U\to V$.
\end{statement}

For $U\in\ob(\Ss)$ and $\mu\in\Mm_U$, condition $(C\mathfrak F1)$ gives a covering $f_j:U_j\to U$ and arrows $\phi_j:F(\lambda_j)\to\mu$ above $f_j$. If
\[
\psi=(\psi_\lambda)\in\Hom(c_\Ll X,(X_{F(\lambda)})_{\lambda\in\Ll})
\]
is the image of
\[
\phi=(\phi_\mu)\in\Hom(c_\Mm X,(X_\mu)_{\mu\in\Mm}),
\]
then the diagrams
\[
\xymatrix@=1.5cm{
&X_{F(\lambda_j)}\ar[ld]_{\psi_{\lambda_j}}\ar[rd]^{\phi_j}&\\
X|U_j\ar[rr]_{\phi_\mu|U_j}&&X_\mu|U_j
}
\]
commute. Thus the $\phi_\mu$ are locally determined by $\psi$; since $\mathsf e$ is a stack, $F^*$ is injective. For surjectivity, start from $\psi$ and construct $\phi$ with $F^*\phi=\psi$. The preceding construction gives local composites
\[
        \phi_{\mu,j}=\phi_j\circ\psi_{\lambda_j}:X|U_j\to X_\mu|U_j.
\]
By Lemma 8.1.6 and $(\Ll0)$ these morphisms agree locally on overlaps, hence glue to a morphism $\phi_\mu:X|U\to X_\mu$. The compatibility of the $\phi_\mu$ with arrows in $\Mm$ follows again locally, and gives the desired morphism of functors.

If $E$ is an ordered set, we shall also denote by $E$ the category with object set $E$ and with $\Hom(i,j)$ reduced to one element or empty according as $i\leq j$ or not.

\begin{statement}{Proposition 8.1.7}
Let $\Ll$ be a locally cofiltered category over a site $\Ss$. There is an ordered set $E$ and a functor $F:E\to\Ll$ such that:
\begin{enumerate}[label=(\roman*)]
\item $E$, viewed as a category over $\Ss$ by means of $p\circ F:E\to\Ss$, is locally cofiltered;
\item the functor $F$ is cofinal.
\end{enumerate}
\end{statement}

Let $\Ll'$ be the product of $\Ll$ with the category defined by the ordered set $\ZZ$. The projection $\Ll'\to\Ll$ is cofinal, so it is enough to prove the proposition for $\Ll'$. If $X$ is a finite set of arrows in $\Ll'$, let $X^0$ be the set of endpoints of arrows of $X$. Let $E$ be the set of finite non-empty subsets $X$ of $\operatorname{Fl}(\Ll')$ such that there is an object $\lambda_X$ of $\Ll'$ satisfying:
\begin{enumerate}[label=\alph*)]
\item no arrow of $X$, except $1_{\lambda_X}$, has target $\lambda_X$;
\item if $\mu\in X^0$, then $1_\mu\in X$ and $\Hom(\lambda_X,\mu)\cap X\neq\varnothing$;
\item every diagram of arrows in $X$ of the form
\[
\xymatrix@=1.75cm{\lambda_X\ar[r]\ar[rd]&\mu\ar[d]\\&\mu'}
\]
is commutative.
\end{enumerate}
Order $E$ by the relation opposite to inclusion. If $X\in E$ and $\mu\in X^0$, there is a unique arrow from $\lambda_X$ to $\mu$ lying in $X$. If $X,Y\in E$ and $X\supset Y$, let $\lambda_{X,Y}$ be the unique arrow in $X$ from $\lambda_X$ to $\lambda_Y$. The assignments $X\mapsto\lambda_X$ and $(X\supset Y)\mapsto\lambda_{X,Y}$ define a functor $E\to\Ll'$.

\begin{statement}{Lemma 8.1.8}
The category $E$ is locally cofiltered over $\Ss$, and the functor $\lambda:E\to\Ll'$ is cofinal.
\end{statement}

One verifies the axioms in sequence. For $(\Ll0)$, given $f:V\to U$ and $X\in E_U$, choose $\lambda\in\ob(\Ll'_V)$ and $\phi\in\Hom_f(\lambda,\lambda_X)$, with $\lambda\notin X^0$; put $X'=X\cup\{1_\lambda\}\cup Y$, where $Y$ consists of the composites
\[
        \lambda\xrightarrow{\phi}\lambda_X\xrightarrow{\psi}\mu,
        \qquad \psi\in X.
\]
Then $X'\in E$, $\lambda_{X'}=\lambda$, and $\Hom(X',X)\neq\varnothing$. For $(\Ll1)$, given a finite family $X_i$ over $U$, choose after refinement arrows from objects $\lambda_j$ to the $\lambda_{X_i}$, and refine again so that arrows landing in common endpoints are equalized. Adding to $\bigcup_iX_i$ the identity of $\lambda_j$ and all composites from $\lambda_j$ to endpoints in the $X_i$ gives objects $X_j\in E$ satisfying $(\Ll1)$. Axiom $(\Ll2)$ is trivial because there are no non-trivial double arrows in the ordered category $E$. Condition $(C\mathfrak F1)$ is trivial because $\lambda$ is surjective. Finally, $(C\mathfrak F2)$ follows by the same equalization construction: for a double arrow $\lambda_X\rightrightarrows\lambda$, refine and add the equalizing arrows to $X$, so that the induced arrow $X_j\to X$ equalizes the given pair.

\paragraph*{8.1.9.0}
Suppose $\Ss$ is the site of non-empty open subsets of the one-point topological space. A locally cofiltered category over $\Ss$ is then just a left filtered category, i.e. one whose opposite is filtered in the sense of I, 2.7. Cofinal functors are the usual cofinal functors between left filtered categories (I, 8). Proposition 8.1.7 becomes:

\begin{statement}{Proposition 8.1.9}
For every cofiltered category $\Ll$, there is a cofiltered ordered set $E$ and a cofinal functor $E\to\Ll$.
\end{statement}

In this special case, the proof of 8.1.7 reduces essentially to the proof of 8.1.9 given in I, 8.

\subsubsection*{8.1.10}
Let $q:\Tt\to\Ss$ be a morphism of sites, and let $p:\Ll\to\Ss$ be a locally cofiltered category over $\Ss$. Define a category $\Ll^*$ as follows. An object is a quadruple $(V,U,\phi,\lambda)$ with $V\in\ob(\Tt)$, $U\in\ob(\Ss)$, $\phi\in\Hom(V,q^*U)$, and $\lambda\in\ob(\Ll_U)$. An arrow
\[
        (V,U,\phi,\lambda)\to(V',U',\phi',\lambda')
\]
is a triple $(f_1,f_2,f_3)$ with $f_1:V\to V'$, $f_2:U\to U'$, $f_3:\lambda\to\lambda'$, satisfying
\[
        q^*f_2\circ\phi=\phi'\circ f_1,
        \qquad p(f_3)=f_2.
\]
The functor $(f_1,f_2,f_3)\mapsto f_1$ makes $\Ll^*$ a category over $\Tt$.

\begin{statement}{Lemma 8.1.11}
The category $\Ll^*$ is locally cofiltered over $\Tt$.
\end{statement}

Axiom $(\Ll0)$ is trivial. For $(\Ll1)$, start with a finite family $L_i=(V,U_i,\phi_i,\lambda_i)$. The $\phi_i$ define a morphism from $V$ to the sheaf $q^*a\prod U_i=a\prod q^*U_i$; hence, locally on $V$, they factor through an object $q^*U_j$ mapping to all $q^*U_i$. After applying $(\Ll1)$ in $\Ll$ and refining, one obtains objects $L_j$ mapping to all $L_i$. The verification of $(\Ll2)$ is similar: a double arrow gives a morphism into the kernel sheaf of a double arrow in $\Ss$; after a local refinement, and then applying $(\Ll2)$ in $\Ll$, the double arrow is equalized.

\begin{statement}{Proposition 8.1.12}
Let $t^*:\Ss^*\to\Ss$ be a locally cofiltered category over $\Ss$, and endow $\Ss^*$ with the topology induced from that of $\Ss$ by $t^*$. Then $t^*$ is an equivalence of sites $t:\Ss\to\Ss^*$.
\end{statement}

By construction, pullback by $t^*$ sends sheaves on $\Ss$ to sheaves on $\Ss^*$. Axioms $(\Ll0)$ and $(\Ll1)$ applied to the empty family imply that every sufficiently small object of $\Ss$ lies in the image of $t^*$, i.e. that the image of $t^*$ is a covering sieve in $\Ss$. By the comparison lemma (III, 4), the inclusion into $\Ss$ of the full subcategory defined by $t^*(\ob\Ss^*)$ is an equivalence of sites; hence one may suppose that $t^*$ is surjective on objects.

Let $\Ff$ be a sheaf on $\Ss^*$. The proof constructs, functorially in $\Ff$, a presheaf $\Ff_0$ on $\Ss$ whose direct image is $\Ff$. The essential point is that if two local presentations of an arrow in $\Ss$ are chosen in $\Ss^*$, the induced maps on values of $\Ff$ agree after a common refinement, by $(\Ll2)$ and the sheaf condition. This gives a well-defined transition morphism $\overline f$ for every arrow $f$ of $\Ss$, independent of all choices. A further refinement argument, using $(\Ll0)$, $(\Ll1)$, and $(\Ll2)$, shows that $\overline f$ is functorial in $f$. Thus every sheaf $\Ff$ on $\Ss^*$ is the direct image of a uniquely determined presheaf $\Ff_0$ on $\Ss$, and morphisms of sheaves are induced by unique morphisms of such presheaves. Hence $t_*$ is fully faithful. Finally, if $f_j:U_j\to U$ is a covering in $\Ss$, it is the image of a covering in $\Ss^*$; therefore $\Ff_0(U)$ injects into $\prod_j\Ff_0(U_j)$, and compatible local sections glue because they already glue in $\Ss^*$. Thus $\Ff_0$ is a sheaf, completing the proof.

\paragraph*{8.1.13.0}
Let $q:\Tt\to\Ss$, $p:\Ll\to\Ss$, and $\Ll^*$ be as in 8.1.10, and let $\mathsf e$ be a stack over $\Tt$. The stack $q_*\mathsf e$ over $\Ss$ is the stack defined by
\[
        (q_*\mathsf e)_U=\mathsf e_{q^*U}.
\]
If $(X_\lambda)_{\lambda\in\Ll}$ is a local projective system indexed by $\Ll$ and with values in $\mathsf e$, it defines a local projective system $(X_L)_{L\in\Ll^*}$ by
\[
        X_L=\phi^*X_\lambda
        \qquad\text{for }L=(V,U,\phi,\lambda).
\]
The reader is left to verify the following statement.

\begin{statement}{Proposition 8.1.13.1}
With the preceding notation, one has
\[
        \varprojlim_{\lambda\in\Ll}X_\lambda
        \simeq
        \varprojlim_{L\in\Ll^*}X_L,
\]
the two sides being defined or undefined simultaneously.
\end{statement}

Let $\Ss$ be a site and let $p^*:\Ss^*\to\Ss$ be a locally filtered category over $\Ss$. If $\mathsf e$ is a stack over $\Ss$, then $p_*\mathsf e$ is a stack over $\Ss^*$ by 8.1.13.0; moreover the identity functor $\Ss^*\to\Ss^*$ is a locally filtered category over $\Ss^*$, endowed with the induced topology, and every local inductive system indexed by $\Ss^*$ defines a local inductive system over $\Ss^*$ indexed by $\Ss^*$ itself.

\begin{statement}{Proposition 8.1.14}
With the preceding notation, one has
\[
        \varprojlim_{\lambda\in\Ss^*/\Ss}X_\lambda
        \simeq
        \varprojlim_{\lambda\in\Ss^*/\Ss^*}X_\lambda,
\]
the two sides being defined or undefined simultaneously.
\end{statement}

\subsection{Local inductive limits in categories of sheaves}

\subsubsection*{8.2.0}
Let $p:\Tt\to\Ss$ be a morphism of sites. Let $\mathsf e$ be the following stack over $\Ss$. An object of $\mathsf e$ is a pair $(U,\Ff)$ consisting of an object $U$ of $\Ss$ and a sheaf $\Ff$ on $p^*U$. An arrow
\[
        f:(U,\Ff)\to(V,\Gg)
\]
is a pair consisting of an arrow $f_0:U\to V$ and a morphism of sheaves on $p^*U$,
\[
        f_1:p^*(f_0)^*\Gg\to\Ff.
\]
The functor from $\mathsf e$ to $\Ss$ is given by $f\mapsto f_0$. Notice that $\mathsf e_U$ is the \emph{opposite category} of the category of sheaves on $p^*U$.

Let $\Ll$ be a locally cofiltered, left locally filtered, category over $\Ss$.

\begin{statement}{Definition 8.2.1}
With the preceding notation, the category of local inductive systems, indexed by $\Ll$, of sheaves on $\Tt$, is the category
\[
        \Hhom_{\Ss}(\Ll,\mathsf e)^{\op}.
\]
\end{statement}

The ``projective-limit'' functor of 8.1.3 is here called the \emph{local inductive-limit functor}. It is a functor from $\Hhom_{\Ss}(\Ll,\mathsf e)^{\op}$ to $(\Gamma\mathsf e)^{\op}=\Tt^{\sim}$.

\begin{statement}{Theorem 8.2.2}
The local inductive-limit functor, from the category of inductive systems of sheaves on $\Tt$ indexed by $\Ll$ to the category of sheaves on $\Tt$, is everywhere defined, commutes with finite projective limits, and commutes with arbitrary inductive limits.
\end{statement}

Propositions 8.1.13, 8.1.14, and 8.1.12 reduce the proof to the case $\Ss=\Tt=\Ll$. A local inductive system is then the data, for each $U\in\ob(\Ss)$, of a sheaf $\Ff_U$ on $\Ss/U$ and, for every arrow $f:V\to U$ in $\Ss$, a morphism $f^*\Ff_U\to\Ff_V$, functorial in $f$.

For such systems, the inductive limits are computed as follows.

\begin{statement}{Lemma 8.2.3}
Let $U\mapsto\Ff_U$ be a section of the category of presheaves over the objects of $\Ss$. Then
\[
        a\bigl(U\mapsto \Gamma(U,\Ff_U)\bigr)
        \simeq \varinjlim_U a\Ff_U,
\]
where $a$ denotes the associated-sheaf functor.
\end{statement}

By definition, the right-hand side represents the functor which assigns to a sheaf $\Ff$ on $\Ss$ the set of coherent systems of arrows $\phi_U:\Ff_U\to\Ff|U$. Such an arrow $\phi_U$ is in turn a coherent system of arrows $\phi_g:\Ff_U(V)\to\Ff(V)$, one for each $g:V\to U$. Applying 8.1.11 to the identity functor of $\Ss$, and then using cofinality of the functor $U\mapsto 1_U$, gives
\[
        \varinjlim_U \Ff_U(U)^{\sim}\simeq \varinjlim_U a\Ff_U.
\]
Returning to the definitions,
\[
        \varinjlim_U \Ff_U(U)^{\sim}=a\bigl(U\mapsto\Gamma(U,\Ff_U)\bigr).
\]
This proves the lemma. It also shows, in the reduced case, that the functor $\varinjlim$ is everywhere defined and is the composite of the associated-sheaf functor and the functor
\[
        (\Ff_U)\longmapsto (U\mapsto\Gamma(U,\Ff_U)).
\]
Both commute with finite projective limits, and the functor $\varinjlim$ also commutes with arbitrary inductive limits by its definition as an adjoint.

Lemma 8.2.3 gives a systematic procedure for proving properties of the local inductive-limit functor from the analogous properties of the associated-sheaf functor. Thus one obtains:

\begin{statement}{Proposition 8.2.4}
Let $\Tt$, $\Ss$, and $\Ll$ be as above. Let $\Aa_\lambda$ be a local inductive system of sheaves of rings on $\Tt$, and let $\Ll_\lambda$ (\resp\ $\Mm_\lambda$) be a local inductive system of sheaves of right (\resp\ left) modules over $\Aa_\lambda$, all three indexed by $\Ll$. Then
\[
\varinjlim(\Ll_\lambda\otimes_{\Aa_\lambda}\Mm_\lambda)
\simeq
(\varinjlim\Ll_\lambda)\otimes_{\varinjlim\Aa_\lambda}(\varinjlim\Mm_\lambda).
\]
\end{statement}

Reduce to the case $\Tt=\Ss=\Ll$, and note that the two sides coincide with the sheaf generated by the presheaf
\[
        \Ll_U(U)\otimes_{\Aa_U(U)}\Mm_U(U).
\]

\begin{statement}{Proposition 8.2.5}
Let $\Tt_1\xrightarrow q\Tt_2\to\Ss$ be two morphisms of sites, and let $(F_\lambda)$ be a local inductive system of sheaves on $\Tt_2$ indexed by a locally cofiltered category $\Ll$ over $\Ss$. Then
\[
        q^*\varinjlim_{\lambda\in\Ll}F_\lambda
        \simeq
        \varinjlim_{\lambda\in\Ll}q^*F_\lambda.
\]
\end{statement}

This follows immediately from the fact that $q^*$ admits a right adjoint $q_*$.

Let $\Ss$ and $\Tt$ be two sites in which finite projective limits are representable. Let $p:\Tt\to\Ss$ be a morphism of sites such that $p^*:\Ss\to\Tt$ commutes with finite projective limits, and let $\Ll$ be the locally cofiltered category over $\Tt$ whose objects are triples $(V,U,\phi)$ with $V\in\ob(\Tt)$, $U\in\ob(\Ss)$, and $\phi\in\Hom(V,p^*U)$. It is obtained by applying 8.1.11 to $1_\Ss:\Ss\to\Ss$.

\begin{statement}{Proposition 8.2.6}
For every sheaf $\Ff$ on $\Tt$, one has
\[
        \Ff\simeq \varinjlim_{\lambda\in\Ll}\phi^*\phi_*\Ff,
\]
where, by abuse of notation, $\phi$ denotes the morphism induced by $p$ from $\Tt/V$ to $\Ss/U$.
\end{statement}

Let $\phi'$ be the inverse-image functor in the sense of presheaves. From 8.2.3 one obtains
\[
\varinjlim_{\lambda\in\Ll}{\phi'}^*\phi_*\Ff
 =\varinjlim_{\lambda\in\Ll}a\phi'\phi_*\Ff
 =\varinjlim_{\lambda\in\Ll}\Ff(V)^{\sim}
 =\varinjlim_V\Ff(V)^{\sim}
 =\Ff.
\]

\begin{statement}{Proposition 8.2.7}
Let $p:T\to S$ be a morphism of topoi, let $\Oo_S$ be a sheaf of rings on $S$, and let $\Oo_T=p^*\Oo_S$. For a right $\Oo_T$-module $\Mm$ to be flat over $\Oo_T$, it is necessary and sufficient that the functor
\begin{equation*}
        \Nn\longmapsto \Mm\otimes_{\Oo_T}p^*\Nn\tag{8.2.7.1}
\end{equation*}
from left $\Oo_S$-modules to abelian sheaves on $T$ be exact.
\end{statement}

Necessity is trivial. Conversely, suppose that (8.2.7.1) is exact. One may suppose $p$ is defined by a morphism of sites $p:\Tt\to\Ss$ satisfying the hypotheses used above. Let $V\in\ob(\Tt)$, $U\in\ob(\Ss)$, and $\phi:V\to p^*U$; again denote by $\phi$ the induced morphism of sites $\Tt/V\to\Ss/U$.

\begin{statement}{Lemma 8.2.8}
The functor
\[
        \Nn\longmapsto \Mm|V\otimes_{\Oo_V}\phi^*\Nn,
\]
from $\Oo_S|U$-modules on $\Ss/U$ to $\Oo_T|V$-modules on $\Tt/V$, is exact.
\end{statement}

One reduces to the case $V=p^*U$. Let $j:\Ss/U\to\Ss$ and $j':\Tt/p^*U\to\Tt$ be the inclusion morphisms. Then
\[
        j'_!(\Mm|V\otimes_{\Oo_V}\phi^*\Nn)
        =\Mm\otimes_{\Oo_T}p^*j_!\Nn.
\]
Lemma 8.2.8 follows because $j_!$ and $j'_!$ are exact and faithful.

Let $Q$ be an $\Oo_T$-module. By 8.2.6,
\[
        Q=\varinjlim_{\phi:V\to p^*U}\phi^*\phi_*Q|V.
\]
Therefore, by 8.2.4,
\[
\Mm\otimes_{\Oo_T}Q
 =\varinjlim_{\phi:V\to p^*U}
   \Mm|V\otimes_{\Oo_V}\phi^*\phi_*Q|V.
\]
By Lemma 8.2.8, the functors
\[
        Q\longmapsto \Mm|V\otimes_{\Oo_V}\phi^*\phi_*Q|V
\]
are exact on the left. Hence $\Mm\otimes_{\Oo_T}Q$ is left exact in $Q$, and therefore exact.

\begin{statement}{Corollary 8.2.9}
Let $f:(T,\Oo_T)\to(S,\Oo_S)$ be a morphism of ringed topoi. Then the inverse image by $f$ of a flat sheaf of right modules is a flat sheaf of modules.
\end{statement}

Factoring $f$ through $(T,f^*\Oo_S)$, one reduces to the case $\Oo_T=f^*\Oo_S$. Let $\Mm$ be a flat sheaf of modules on $S$. By 8.2.7, to check that $f^*\Mm$ is flat it is enough to check exactness of the functor
\[
        \Nn\longmapsto f^*\Mm\otimes_{\Oo_T}f^*\Nn
        =f^*(\Mm\otimes_{\Oo_S}\Nn),
\]
which follows since $f^*$ is exact.

\begin{statement}{Lemma 8.2.10}
Let $(S,\Oo_S)$ be a ringed topos, let $\Ff$ be a sheaf of left modules locally of finite presentation, let $\Gg$ be a sheaf of bimodules, and let $\Hh$ be a flat sheaf of left modules. Then the canonical arrow
\[
        \Hhom(\Ff,\Gg)\otimes\Hh\longrightarrow
        \Hhom(\Ff,\Gg\otimes\Hh)
\]
is an isomorphism.
\end{statement}

The question is local on $S$, so one may suppose that $\Ff$ admits a finite presentation
\[
        \Ff_1\to\Ff_0\to\Ff\to 0.
\]
The first row of the following diagram is exact because $\Hh$ is flat:
\[
\xymatrix{
0\ar[r]&\Hhom(\Ff,\Gg)\otimes\Hh\ar[r]\ar[d]&
\Hhom(\Ff_0,\Gg)\otimes\Hh\ar[r]\ar[d]&
\Hhom(\Ff_1,\Gg)\otimes\Hh\ar[d]\\
0\ar[r]&\Hhom(\Ff,\Gg\otimes\Hh)\ar[r]&
\Hhom(\Ff_0,\Gg\otimes\Hh)\ar[r]&
\Hhom(\Ff_1,\Gg\otimes\Hh).
}
\]
The assertion follows. Taking $\Gg=\Oo_S$, one obtains:

\begin{statement}{Lemma 8.2.11}
Every morphism from a sheaf of modules locally of finite presentation to a flat sheaf of modules factors, locally on $S$, through a free sheaf $\Oo_S^n$, $n\in\mathbb N$.
\end{statement}

\begin{statement}{Theorem 8.2.12 (D. Lazard)}
For a sheaf of modules $\Mm$ on a ringed site $(S,\Oo_S)$ to be flat, it is necessary and sufficient that it be a local inductive limit of finite free modules.
\end{statement}

Sufficiency follows from 8.2.2 and 8.2.4. For every $U\in\ob(\Ss)$, let $\Ll_0(U)$ (\resp\ $\Ll_1(U)$) be the category of finite free (\resp\ finitely presented) sheaves of modules on $U$, endowed with a linear map to $\Mm|U$. For every arrow $f:V\to U$ in $\Ss$, let $f^*$ be the restriction functor from $\Ll_i(U)$ to $\Ll_i(V)$. These functors define a category $\Ll_i$, fibered over $\Ss$, whose fibers are the opposite categories of the categories $\Ll_i(U)$.

The reader checks that $\Ll_1$ is locally filtered over $\Ss$ and that
\[
        \Mm=\varinjlim_{(M,f)\in\Ll_1}M.
\]
If $\Mm$ is flat, the inclusion of $\Ll_0$ into $\Ll_1$ is fully faithful and, by 8.2.11, satisfies $(C\mathfrak F1)$ of 8.1.4. It therefore automatically satisfies $(C\mathfrak F2)$, and $\Ll_0$ is locally filtered. By 8.1.5,
\[
        \Mm=\varinjlim_{(M,f)\in\Ll_0}M,
\]
which proves the theorem.
