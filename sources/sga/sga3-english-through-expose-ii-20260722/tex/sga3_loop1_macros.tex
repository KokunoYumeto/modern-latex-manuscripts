% Loop-1 diagram helper.
% Usage:
% \SGAThreeDiagram[0.85\textwidth]
%   {figures/exp1/diagram-id.png}
%   {Caption}
%   {fig:stable-label}
%   {Polo-Gille combined reader p. N; component local p. M}
%
% PNGs are deliberate temporary stand-ins. If the named file is missing, the
% build displays a visible box rather than silently dropping the diagram.
\newcommand{\SGAThreeDiagram}[5][0.92\textwidth]{%
  \begin{figure}[H]
    \centering
    \IfFileExists{#2}{%
      \includegraphics[width=#1]{#2}%
    }{%
      \fbox{%
        \parbox[c][0.18\textheight][c]{0.82\textwidth}{%
          \centering
          \textbf{Loop-1 diagram image pending}\\[0.75em]
          \texttt{\detokenize{#2}}%
        }%
      }%
    }%
    \caption{#3}
    \label{#4}
    \par\smallskip
    {\footnotesize\itshape Temporary Loop-1 image; source locator: #5.}
  \end{figure}%
}
