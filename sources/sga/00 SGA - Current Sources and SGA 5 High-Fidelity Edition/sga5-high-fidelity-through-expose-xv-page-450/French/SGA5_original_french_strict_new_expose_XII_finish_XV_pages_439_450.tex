\section*{Exposé XII (suite et fin)}

\begin{statement}{Proposition 7.2}
Soient $X$ un schéma lisse de dimension $1$, $F$ constructible et
\[
(u,f):(X,F)\longrightarrow (X,F)
\]
un endomorphisme tel que le morphisme de faisceaux
\[
u:f^*(F)\longrightarrow F
\]
soit un isomorphisme. Sous ces conditions il existe $U,U',G,f',\varphi$ satisfaisant aux conditions de 5.1, 6.1, 6.2.
\end{statement}

\begin{prooftext}[Démonstration]
Soit $U$ le plus grand ouvert de $X$ tel que $F$ soit localement constant dans $U$ et désignons par $Y$ son complémentaire. L'ensemble $Y$ est stable par $f$:
\[
f(Y)\subset Y.
\]
En effet, supposons par l'absurde qu'il existe $y\in Y$ tel que $f(y)\in U$. Le faisceau $f^*(F)$ est alors localement constant au voisinage de $y$ et, en utilisant l'isomorphisme $u$, il résulte que $F$ est localement constant au voisinage de $y$, ce qui est absurde.

Si $f$ est constant, la démonstration de la proposition est triviale. On va donc supposer que $f$ n'est pas constant. Si $\eta$ est le point générique de $X$, on aura donc $f(\eta)=\eta$.

Soient $i_\eta:\Spec(k(\eta))\hookrightarrow X$ l'inclusion canonique, $F_\eta=i_\eta^*(F)$ et
\[
u_\eta:f_\eta^*(F_\eta)\longrightarrow F_\eta
\]
le morphisme, en fait l'isomorphisme, de faisceaux induit par $u$, $f_\eta$ étant bien-entendu la restriction de $f$ à $\Spec(k(\eta))$.

En traduisant tout en termes de $H$-modules:
\[
H=\Gal(k(\eta)_s/k(\eta)),\qquad k(\eta)_s\text{ une clôture séparable de }k(\eta),
\]
on obtient les isomorphismes
\[
\begin{tikzcd}[column sep=large]
H \arrow[r,"\Psi"] & H
\end{tikzcd}
\qquad
\begin{tikzcd}[column sep=large]
F_\eta \arrow[r,"u_\eta"] & F_\eta,
\end{tikzcd}
\]
où $F_\eta$ a une structure de $H$-module et la relation suivante est vérifiée:
\[
u_\eta(gx)=\Psi(g)\,u_\eta x,
\qquad g\in H,
\quad x\in F_\eta.
\]
Si $H_1$ est le plus grand sous-groupe distingué de $H$ d'indice fini tel que $F_\eta^{H_1}=F_\eta$, on a $\Psi(H_1)=H_1$. Un tel sous-groupe existe parce que le faisceau $F_\eta$ est constructible.

Soit $K\supset k(\eta)$ l'extension galoisienne de $k(\eta)$ correspondante au sous-groupe $H_1$, et $X'$ la normalisée de $X$ dans $K$. Évidemment on peut définir un endomorphisme
\[
f':X'\longrightarrow X'
\]
et un endomorphisme
\[
H/H_1=G\xrightarrow{\varphi}G
\]
tels que le système $U,U',f',G,\varphi$ satisfasse aux conditions voulues.
\end{prooftext}

\begin{statement}{Exemple 7.3}
Soient $X$ un préschéma défini sur le corps fini $\mathbb F_p$, $p$ étant un nombre premier, $\fr_X:X\to X$ l'endomorphisme de Frobenius (cf. XIV) et $F$ un faisceau de $\Lambda$-modules sur $X$.

Si $Y,S$ sont des schémas sur $\mathbb F_p$, et si $Y$ est un schéma sur $S$, nous désignons par
\[
\Fr_{Y/S}:Y\longrightarrow Y^{(p/S)}
\]
le morphisme de Frobenius relatif (cf. XIV 2).

On obtient un morphisme de faisceaux sur $X_{\et}$
\[
(\fr_X)_*(F)\longrightarrow F
\]
en posant, pour tout $X$-préschéma étale $U$,
\[
(\fr_X)_*(F)(U)=F(\fr_X^{-1}(U))=F(U^{(p/X)})
 \xrightarrow{\,F(\Fr_{U/X})\,}F(U).
\]
Ce morphisme est en fait un isomorphisme. Son inverse
\[
F(\Fr_{/X})^{-1}:F\longrightarrow (\fr_X)_*(F)
\]
définit par adjonction un morphisme
\[
\Fr_{F/X}^*:(\fr_X)^*(F)\longrightarrow F
\]
qui est un isomorphisme.

Si $X$ est en particulier lisse de dimension $1$ et $F$ constructible, la paire
\[
(\Fr_{F/X}^*,\fr_X):(X,F)\longrightarrow (X,F)
\]
est donc justiciable de la proposition 7.2.
\end{statement}

\subsection*{§ 1. Morphisme de Frobenius}

\subsubsection*{n\textsuperscript{o} 1. Endomorphisme de Frobenius}

\begin{statement}{Définition 1}
Soit $p$ un nombre premier. On dit qu'un préschéma $X$ est de caractéristique $p$ si $p\cdot 1_{\OO_X}=0$.
\end{statement}

Cette condition équivaut à la suivante: le morphisme unique $X\to\Spec(\bbZ)$ se factorise à travers $\Spec(\bF_p)\to\Spec(\bbZ)$; notons de plus qu'une telle factorisation est unique.

Nous désignerons par $(\Sch)_p$ la sous-catégorie pleine de la catégorie des préschémas dont les objets sont les préschémas de caractéristique $p$. D'après la remarque précédente, la catégorie $(\Sch)_p$ est isomorphe de manière naturelle à la catégorie des $\bF_p$-préschémas.

Si $X$ est un préschéma de caractéristique $p$, le couple
\[
\fr_X=(\id_{|X|},\psi),
\qquad \psi:\OO_X\longrightarrow\OO_X,
\qquad \psi(f)=f^p
\]
pour toute section $f$ de $\OO_X$ dans un ouvert, est un endomorphisme de $X$, car $\psi$ est visiblement un endomorphisme du faisceau d'anneaux $\OO_X$.

\begin{statement}{Définition 2}
Soit $X$ un préschéma de caractéristique $p$; on appelle endomorphisme de Frobenius de $X$ l'endomorphisme $\fr_X$ ainsi défini.
\end{statement}

L'endomorphisme de Frobenius $\fr_X$ dépend fonctoriellement du préschéma $X$ de caractéristique $p$, c'est-à-dire que, pour tout morphisme $g:X\to Y$ de préschémas de caractéristique $p$, le diagramme suivant est commutatif:
\[
\begin{tikzcd}[column sep=large,row sep=large]
X \arrow[r,"\fr_X"] \arrow[d,"g"'] & X \arrow[d,"g"]\\
Y \arrow[r,"\fr_Y"'] & Y.
\end{tikzcd}
\]
Autrement dit, on a défini un endomorphisme $\fr$ du foncteur identique $\id_{(\Sch)_p}$.

\subsubsection*{n\textsuperscript{o} 2. Morphisme de Frobenius relatif}

Soit $S$ un préschéma de caractéristique $p$; tous les $S$-préschémas sont aussi de caractéristique $p$. Si $X$ est un $S$-préschéma, on voit que $\fr_X$ se factorise à travers la projection
\[
\pi_{X/S}:X\times_S(S,\fr_S)\longrightarrow X,
\]
où $(S,\fr_S)$ désigne $S$ considéré comme $S$-préschéma au moyen du morphisme $\fr_S$. Nous désignerons le produit fibré $X\times_S(S,\fr_S)$ par $X^{(p/S)}$, ou même par $X^{(p)}$ si aucune confusion n'en résulte; l'unique $S$-morphisme de $X$ dans $X^{(p/S)}$ qui, composé avec $\pi_{X/S}$, donne $\fr_X$ est désigné par $\Fr_{X/S}$; ainsi on obtient un diagramme commutatif où le carré est cartésien.

Il est clair que $X^{(p/S)}$ dépend fonctoriellement du $S$-préschéma $X$; le foncteur
\[
p_S:X\longmapsto X^{(p/S)}
\]
est simplement le changement de base au moyen du morphisme $\fr_S:S\to S$. Comme tout changement de base, il commute aux limites projectives et aux sommes, et conserve les propriétés telles que: de type fini, séparé, affine, projectif, plat, étale, etc. La commutation aux limites projectives montre que, si $X$ est muni d'une structure algébrique, par exemple groupe, algèbre, etc., $X^{(p/S)}$ se trouve muni naturellement d'une structure algébrique de même espèce.

En utilisant la fonctorialité de $\fr_X$ par rapport à $X$, on voit que le morphisme $\Fr_{X/S}$ dépend fonctoriellement du $S$-préschéma $X$, de sorte qu'on a défini un morphisme fonctoriel
\[
\Fr_{X/S}:\id_{(\Sch)/S}\longrightarrow p_S.
\]

\begin{statement}{Définition 3}
Soient $S$ un préschéma de caractéristique $p$ et $X$ un $S$-préschéma. On dit que le $S$-morphisme
\[
\Fr_{X/S}:X\longrightarrow X^{(p/S)}
\]
est le morphisme de Frobenius de $X$ relativement à $S$.
\end{statement}

Les premières propriétés formelles du morphisme de Frobenius sont énoncées dans la proposition suivante.

\begin{statement}{Proposition 1}
\begin{enumerate}[label=\alph*)]
\item \emph{(Transitivité.)} Soient $S$ un préschéma de caractéristique $p$ et $g:X\to Y$ un $S$-morphisme de $S$-préschémas. Le diagramme naturel reliant $X$, $X^{(p/S)}$, $X^{(p/Y)}$, $Y$, $Y^{(p/S)}$ et $S$ est commutatif et ses trois carrés sont cartésiens.

\item \emph{(Changement de base.)} Soient $g:S'\to S$ un morphisme de préschémas de caractéristique $p$ et $X$ un $S$-préschéma. On a un isomorphisme canonique
\[
X^{(p/S)}\times_S S'\simeq (X\times_S S')^{(p/S')}
\]
fonctoriel en $X$, soit $(X^{(p/S)})'\simeq X'{}^{(p/S')}$ en affectant d'un $'$ le résultat du changement de base par $g$; de plus
\[
\Fr_{X'/S'}=(\Fr_{X/S})'
\]
modulo l'identification permise par l'isomorphisme précédent.
\end{enumerate}
\end{statement}

La démonstration est immédiate.

\begin{statement}{Proposition 2}
Soient $S$ un préschéma de caractéristique $p$ et $X$ un $S$-préschéma.
\begin{enumerate}[label=\alph*)]
\item Le morphisme de Frobenius relatif
\[
\Fr_{X/S}:X\longrightarrow X^{(p/S)}
\]
est entier, surjectif et radiciel, donc c'est un homéomorphisme universel.

\item Si $X\to S$ est une immersion ouverte, $X^{(p)}=X$ et $\Fr_{X/S}=\id_X$. Si $X\to S$ est une immersion fermée, identifiant $X$ à un sous-espace annelé fermé $(|X|,\OO_S/\JJ)$ de $S$, alors
\[
X^{(p)}=(|X|,\OO_S/\JJ^p)
\]
et $\Fr_{X/S}=(\id_{|X|},\varphi)$, où $\varphi:\OO_S/\JJ^p\to\OO_S/\JJ$ est l'augmentation évidente.

\item Supposons $X$ localement de présentation finie sur $S$.
\begin{enumerate}[label=\arabic*)]
\item Les trois conditions suivantes sont équivalentes:
\[
\begin{array}{ll}
\text{(i)}& X\text{ est non ramifié sur }S,\\
\text{(ii)}& \Fr_{X/S}\text{ est non ramifié},\\
\text{(iii)}& \Fr_{X/S}\text{ est un monomorphisme}.
\end{array}
\]
\item De même, les trois conditions suivantes sont équivalentes:
\[
\begin{array}{ll}
\text{(i')}& X\text{ est étale sur }S,\\
\text{(ii')}& \Fr_{X/S}\text{ est étale},\\
\text{(iii')}& \Fr_{X/S}\text{ est un isomorphisme}.
\end{array}
\]
\end{enumerate}
\end{enumerate}
\end{statement}

Démontrons a). Le composé $\pi_{X/S}\circ\Fr_{X/S}=\fr_X$ est entier, surjectif et radiciel de façon évidente; il en résulte que $\Fr_{X/S}$ est radiciel (EGA I 3.5.6 (ii)); de plus, $\pi_{X/S}$ est séparé et radiciel, car il se déduit de $\fr_S$ par le changement de base $X\to S$; il en résulte que $\Fr_{X/S}$ est entier (EGA II 6.1.5 (v)) et surjectif.

La démonstration de b) est évidente, en observant que
\[
\JJ\OO_{(S,\fr_S)}=\JJ^p
\]
pour tout idéal $\JJ$ de $\OO_S$.

Preuve de c). On sait que, pour qu'un morphisme soit un monomorphisme, respectivement un isomorphisme, il faut et il suffit qu'il soit radiciel et non ramifié, respectivement radiciel, étale et surjectif; d'où l'équivalence de (ii) et (iii), respectivement de (ii') et (iii'). Il reste à prouver que (i) équivaut à (ii) et (i') à (ii').

Pour cela on utilise le critère différentiel de non-ramification (SGA I 3) et l'isomorphisme
\[
\Omega^1_{X/S}\simeq \Omega^1_{X/X^{(p)}}
\]
donné par la suite exacte canonique
\[
\Fr_{X/S}^*(\Omega^1_{X^{(p)}/S})\longrightarrow \Omega^1_{X/S}
\longrightarrow \Omega^1_{X/X^{(p)}}\longrightarrow 0
\]
(SGA II 4), dans laquelle la première flèche est visiblement nulle, car $\Omega^1_{X^{(p)}/S}$ est engendré par les différentielles des sections de $\OO_{X^{(p)}}$, et la différentielle $d_{X/S}$ s'annule sur $(\OO_X)^p$ et sur $\OO_S$. Ainsi (i) est équivalente à (ii).

Supposons maintenant $X$ étale sur $S$; il en est de même de $X^{(p)}$ par changement de base, et par suite le morphisme $\Fr_{X/S}:X\to X^{(p)}$ est étale. Inversement, supposons $\Fr_{X/S}$ étale; il est non ramifié, donc $X$ est non ramifié sur $S$. Autrement dit, le morphisme structural $g:X\to S$ est localement sur $X$, pour la topologie étale, une immersion fermée (SGA I cor. 7.8). On peut donc supposer que $g$ est une immersion fermée de présentation finie, et il s'agit de voir que si $\Fr_{X/S}$ est un isomorphisme, $g$ est une immersion ouverte. Or $X$ est défini par un idéal $\JJ$ de type fini,
\[
X=(|X|,\OO_S/\JJ),
\]
et, d'après b),
\[
X^{(p)}=(|X|,\OO_S/\JJ^p),
\qquad
\Fr_{X/S}=(\id_{|X|},\OO_S/\JJ^p\to\OO_S/\JJ).
\]
Dire que $\Fr_{X/S}$ est un isomorphisme revient donc à dire que $\JJ=\JJ^p$; et, comme $\JJ$ est de type fini, on en déduit par le lemme de Nakayama que $\JJ_s=0$, ou $\JJ_s=\OO_{S,s}$, pour tout point $s\in S$, d'où la conclusion.

\subsubsection*{n\textsuperscript{o} 3. Effet sur un Module. Exemples}

Soient $S$ un préschéma de caractéristique $p$ et $X$ un $S$-préschéma; pour tout $\OO_X$-Module $\cE$, faisceau pour la topologie de Zariski, on pose
\[
\cE^{(p/S)}=\pi_{X/S}^*(\cE)=\cE\otimes_{\OO_X}\OO_{X^{(p)}}
 =\cE\otimes_{\OO_S}\OO_{(S,\fr_S)}.
\]
C'est un $\OO_{X^{(p)}}$-Module; on a
\[
\cE^{(p/X)}\simeq \Fr_{X/S}^*(\cE^{(p/S)}).
\]

Le cas le plus intéressant est celui où $X=S$; on peut alors écrire $\cE^{(p)}$ au lieu de $\cE^{(p/S)}$ sans risque de confusion; $\cE^{(p)}=\fr_S^*(\cE)$. Le morphisme d'adjonction
\[
\cE\longrightarrow \fr_{S*}\fr_S^*(\cE)
\]
est un morphisme de faisceaux abéliens $\cE\to\cE^{(p)}$ qui n'est pas $\OO_S$-linéaire, mais $p$-linéaire, en un sens évident; c'est visiblement un morphisme $p$-linéaire universel de $\cE$ dans un $\OO_S$-Module.

Il est clair que $\cE^{(p)}$ dépend fonctoriellement de $\cE$ et que le foncteur $\cE\mapsto\cE^{(p)}$ est exact à droite, mais pas exact en général; il commute au produit tensoriel:
\[
(\cE\otimes_{\OO_X}\cF)^{(p)}\simeq
\cE^{(p)}\otimes_{\OO_{X^{(p)}}}\cF^{(p)},
\]
et conserve les propriétés telles que: quasi-cohérent, de type fini, de présentation finie, plat, localement libre, etc. En particulier, si $\cL$ est un faisceau inversible sur $S$, on voit immédiatement que $\cL^{(p)}$ est canoniquement isomorphe à $\cL^{\otimes p}$, car le morphisme $\cL\to\cL^{\otimes p}$ défini par l'élévation à la puissance tensorielle $p$-ième des sections est un morphisme $p$-linéaire universel.

Si $\mathcal{A}$ est une $\OO_S$-Algèbre, $\mathcal{A}^{(p)}$ possède une structure de $\OO_S$-Algèbre déduite de celle de $\mathcal{A}$. Supposons $\mathcal{A}$ quasi-cohérente, respectivement graduée et quasi-cohérente; il en est de même de $\mathcal{A}^{(p)}$ et on a
\[
\Spec(\mathcal{A}^{(p)})=\Spec(\mathcal{A}\otimes_{\OO_S}\OO_{(S,\fr_S)})
 \simeq \Spec(\mathcal{A})\times_S(S,\fr_S)=\Spec(\mathcal{A})^{(p)},
\]
respectivement
\[
\Proj(\mathcal{A}^{(p)})=\Proj(\mathcal{A}\otimes_{\OO_S}\OO_{(S,\fr_S)})
 \simeq \Proj(\mathcal{A})\times_S(S,\fr_S)=\Proj(\mathcal{A})^{(p)}.
\]
La projection $\pi_{\Spec(\mathcal{A})/S}$, respectivement $\pi_{\Proj(\mathcal{A})/S}$, s'identifie au morphisme déduit de $\mathcal{A}\to\mathcal{A}^{(p)}$. D'autre part $\Fr_{\Spec(\mathcal{A})/S}$, respectivement $\Fr_{\Proj(\mathcal{A})/S}$, provient d'un homomorphisme de $\OO_S$-Algèbres
\[
\mathcal{A}^{(p)}=\mathcal{A}\otimes_{\OO_S}\OO_{(S,\fr_S)}\longrightarrow \mathcal{A}
\]
qui transforme $a\otimes f$ en $a^p f$ pour toute section $a$ de $\mathcal{A}$ et toute section $f$ de $\OO_S$. Pour le voir, on observe que le composé de cet homomorphisme avec $\mathcal{A}\to\mathcal{A}^{(p)}$, à droite, est l'élévation à la puissance $p$-ième dans $\mathcal{A}$.

Considérons le cas particulier où $\mathcal{A}$ est l'algèbre symétrique $\Sym(\cE)$ d'un $\OO_S$-Module quasi-cohérent $\cE$; il est clair que $\mathcal{A}^{(p)}\simeq\Sym(\cE^{(p)})$, donc
\[
\mathbb V(\cE^{(p)})\simeq\Spec(\mathcal{A}^{(p)})\simeq\Spec(\mathcal{A})^{(p)}=\mathbb V(\cE)^{(p)},
\]
\[
\mathbb P(\cE^{(p)})\simeq\Proj(\mathcal{A}^{(p)})\simeq\Proj(\mathcal{A})^{(p)}=\mathbb P(\cE)^{(p)}.
\]
Quant à $\Fr_{\mathbb V(\cE)/S}$, respectivement $\Fr_{\mathbb P(\cE)/S}$, il se factorise de la façon suivante:
\[
\mathbb V(\cE)\longrightarrow \mathbb V(\Sym^p(\cE))\longrightarrow \mathbb V(\cE^{(p)}),
\]
respectivement
\[
\mathbb P(\cE)\longrightarrow \mathbb P(\Sym^p(\cE))\longrightarrow \mathbb P(\cE^{(p)}),
\]
où la première flèche est définie par le morphisme canonique
\[
\Sym(\Sym^p(\cE))\longrightarrow \Sym(\cE)
\]
qui provient de $\Sym^p(\cE)\to\Sym(\cE)$, tandis que la seconde est définie par le morphisme de $\OO_S$-Modules
\[
\cE^{(p)}\longrightarrow \Sym^p(\cE)
\]
provenant du morphisme $p$-linéaire $\cE\to\Sym^p(\cE)$ donné par l'élévation à la puissance symétrique $p$-ième des sections. Dans le cas de $X=\mathbb P(\cE)$, calculons le faisceau fondamental $\OO_{X^{(p)}}(1)$:
\[
\OO_{X^{(p)}}(1)=(\Sym(\cE)^{(p)}(1))^{\sim}
=(\Sym(\cE)(1)^{(p)})^{\sim}
\simeq \OO_X(1)^{(p/S)}.
\]
Lorsque $\cE=\OO_S^n$, $\Sym(\cE)=\OO_S[T_1,\ldots,T_n]$ et $X=\mathbb V(\cE)=S[T_1,\ldots,T_n]$, respectivement $X=\mathbb P(\cE)=\mathbb P_S^{n-1}$, il est clair que $\cE^{(p)}\simeq\OO_S^n=\cE$, de sorte que $X^{(p)}=X$. Le morphisme de Frobenius $\Fr_{X/S}$ provient de l'endomorphisme de l'algèbre $\OO_S[T_1,\ldots,T_n]$ qui transforme $T_i$ en $T_i^p$ $(i=1,\ldots,n)$.

Considérons maintenant un préschéma $X$ affine de type fini, respectivement projectif, sur $S$, défini par un idéal, respectivement un idéal homogène,
\[
\JJ\subset \OO_S[T_1,\ldots,T_n].
\]
Ainsi $X=\Spec(\mathcal{A})$, respectivement $X=\Proj(\mathcal{A})$, où
\[
\mathcal{A}=\OO_S[T_1,\ldots,T_n]/\JJ.
\]
On voit que
\[
\mathcal{A}^{(p)}\simeq \OO_S[T_1,\ldots,T_n]/\JJ^{(p)},
\]
où $\JJ^{(p)}$ est l'image dans $\OO_S[T_1,\ldots,T_n]$ de $\JJ\otimes_{\OO_S}\OO_{(S,\fr_S)}$; l'idéal $\JJ^{(p)}$ se déduit de $\JJ$ en élevant les coefficients des polynômes à la puissance $p$-ième; on a $X^{(p)}=\Spec(\mathcal{A}^{(p)})$, respectivement $\Proj(\mathcal{A}^{(p)})$, et le morphisme de Frobenius
\[
\Fr_{X/S}:X\longrightarrow X^{(p)}
\]
est défini par l'élévation à la puissance $p$-ième des coordonnées $T_i$, ce qui détermine un morphisme $\mathcal{A}^{(p)}\to\mathcal{A}$.
