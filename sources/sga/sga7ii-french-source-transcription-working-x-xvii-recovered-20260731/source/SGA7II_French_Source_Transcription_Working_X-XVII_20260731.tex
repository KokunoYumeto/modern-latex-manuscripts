% Breakable underlining: a long underlined statement inside an unbreakable
% \underline box runs off the page. \uline breaks and hyphenates. Dispatch on
% \ifmmode because \underline is also used inside math.
\usepackage[normalem]{ulem}
\let\mathunderline\underline
\renewcommand{\underline}[1]{\ifmmode\mathunderline{#1}\else\uline{#1}\fi}

% Long inline formulas give TeX no break point; let it loosen spacing rather
% than push text past the margin.
