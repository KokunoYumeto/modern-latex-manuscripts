\section*{9. Subtopoi and gluing of topoi}

\subsection*{9.1. Subtopoi and embeddings of topoi}

\begin{statement}{Definition 9.1.1}
\begin{enumerate}[label=\alph*)]
\item A morphism of topoi \(f:F\to E\) is called an \emph{embedding} if the functor
\(f_*:F\to E\) is fully faithful, or equivalently if the adjunction morphism
\(f^*f_*\to \id_F\) is an isomorphism.

\item A \emph{subtopos} of a topos \(E\) is a strictly full subcategory \(E'\) of \(E\) which is a topos and for which the inclusion functor
\(\alpha:E'\to E\) is of the form \(i_*\), where \(i:E'\to E\) is a morphism of topoi; equivalently, \(\alpha\) has a left adjoint \(i^*\) which is left exact. The morphism \(i\), determined up to unique isomorphism, is then called the \emph{inclusion morphism} of the subtopos \(E'\) into \(E\).
\end{enumerate}
\end{statement}

\begin{remarktext}{Remark 9.1.2}
\begin{enumerate}[label=\alph*)]
\item The inclusion morphism of a subtopos is an embedding. A strictly full subcategory \(E'\subset E\) is a subtopos if and only if it is the essential image of the direct-image functor \(f_*\) attached to a morphism of topoi \(f:F\to E\) which is an embedding.

\item A morphism \(f:F\to E\) of topoi is an embedding if and only if it is isomorphic to a composite
\[
F\xrightarrow{g} E'\xrightarrow{i} E,
\]
where \(g\) is an equivalence of topoi and \(i\) is the inclusion of a subtopos \(E'\) of \(E\). The subtopos \(E'\), namely the essential image of \(f_*\), is uniquely determined, and the factorization is unique up to unique isomorphism. In practice one identifies \(F\), by means of \(g\), with the corresponding subtopos of \(E\).

\item An equivalence of topoi \(u:E\xrightarrow{\sim}E_1\) gives, in the evident way, a bijection between the subtopoi of \(E\) and the subtopoi of \(E_1\). Thus for the study of subtopoi one may reduce to the case \(E=\widetilde C\), where \(C\) is a small site. In that case a strictly full subcategory \(E'\subset E\) is a subtopos if and only if its inclusion has a left exact left adjoint. Moreover the subtopoi so obtained correspond bijectively to the topologies \(T'\) on \(C\) finer than the given topology \(T\): to \(T'\) one associates the strictly full subcategory \((C,T')^\sim\) of \(C^\sim\) consisting of the sheaves for \(T'\). Consequently, for any \(\U\)-topos \(E\), a strictly full subcategory \(E'\subset E\) is a subtopos if and only if its inclusion has a left exact left adjoint; the ordered set \(\mathfrak E(E)\) of subtopoi of \(E\) is \(\U\)-small, and every subset of it admits a supremum and an infimum.

\item If \(E'\) and \(E''\) are two subtopoi of \(E\), their intersection is again a subtopos. In the case \(E=C^\sim\), let \(T'\) and \(T''\) be the corresponding topologies on \(C\), both finer than \(T\). Then \(E'\cap E''\) is the category of sheaves for the topology \(T'''\) which is the supremum of \(T'\) and \(T''\). More generally, if \(T'\) is the supremum of a family \((T_i)\) of topologies, then the sheaves for \(T'\) form the intersection of the corresponding categories of sheaves.
\end{enumerate}
\end{remarktext}

\begin{statement}{Proposition 9.1.3}
For every family \((E_i)_{i\in I}\) of subtopoi of a topos \(E\), the intersection
\[
\bigcap_{i\in I}E_i
\]
is a subtopos of \(E\).
\end{statement}

\begin{statement}{Proposition 9.1.4}
Let \(i:E'\to E\) be an embedding of topoi, and let \(F\) be a topos. The functor
\[
f\longmapsto i\circ f:\Hhomtop(F,E')\longrightarrow \Hhomtop(F,E)
\tag{9.1.4.1}
\]
is fully faithful. Its essential image consists of those morphisms \(g:F\to E\) for which the essential image of \(g_*\) is contained in that of \(i_*\).
\end{statement}

\begin{prooftext}
Consider the evident commutative square
\[
\xymatrix@C=1.75cm@R=1cm{
\Hhomtop(F,E')\ar@{^{(}->}[d]\ar[r]&\Hhomtop(F,E)\ar@{^{(}->}[d]\\
\Hhom(F,E')\ar[r]&\Hhom(F,E).
}
\]
The vertical functors send a morphism of topoi to its direct image and are fully faithful by definition of \(\Hhomtop\). Since \(i_*\) is fully faithful, the lower horizontal functor is fully faithful, and so is the upper one. The characterization of the image follows from \((if)_*=i_*f_*\). Conversely, if \(g_*=i_*f_*\) for a functor \(f_*:F\to E'\), then \(f_*\) has the left adjoint \(g^*i_*\), which is left exact; hence \(f_*\) is the direct image of a morphism of topoi \(f:F\to E'\).
\end{prooftext}

\begin{statement}{Corollary 9.1.5}
If \(f:E'\to E\) is an embedding of topoi, then the induced functor on categories of points
\[
\Point(f):\Ppoint(E')\longrightarrow \Ppoint(E)
\tag{9.1.5.1}
\]
is fully faithful.
\end{statement}

\begin{remarktext}{Exercise 9.1.6. Embeddings of topoi and \(2\)-monomorphisms}
Let \(i:E'\to E\) be a morphism of topoi.
\begin{enumerate}[label=\alph*)]
\item If \(F\) and \(G\) are topoi over \(E'\), define the canonical functor
\[
f\longmapsto f*i:\mathcal{H}om_{\mathrm{top},E'}(F,G)\longrightarrow \mathcal{H}om_{\mathrm{top},E}(F,G).
\tag{9.1.6.1}
\]
\item Suppose that \(i\), as a \(1\)-arrow of the \(2\)-category of topoi, is a \(2\)-monomorphism, that is, for every topos \(F\) belonging to a sufficiently large universe the functor (9.1.4.1) is fully faithful. Show that for every pair of topoi \(F,G\) the functor (9.1.6.1) is fully faithful.
\item Show that \(i\) is an embedding if and only if \(i\) is a \(2\)-monomorphism.
\item Let \(g:F\to E\) be a morphism of topoi and assume the \(2\)-fiber product
\[
F'=F\times_E^2 E'
\]
exists. Let \(j:F'\to F\) be the projection. If \(i\) is an embedding, show that \(j\) is an embedding. When \(E'\) is a subtopos of \(E\), the essential image of \(j_*\) is called the \emph{inverse-image subtopos of \(F\)} of \(E'\) by \(g\).
\item With the preceding notation, let \(X\) be an object of \(F\). Show that if \(X\) belongs to the inverse-image subtopos \(F'\), then for every object \(U\) of \(F\), writing \(g_U:F_{/U}\to E\) for the induced morphism and \(X|U=(X\times U\to U)\), one has
\[
g_{U*}(X|U)\in \Ob E'.
\]
Ask conversely whether this condition is sufficient, or equivalently whether the strictly full subcategory so obtained is a subtopos of \(F\).
\item If \(X\in E\), put \(X'=i^*X\) and let \(i_{/X}:E'_{/X'}\to E_{/X}\) be the induced morphism of topoi. Show that if \(i\) is an embedding, then \(i_{/X}\) is an embedding. Conversely, if \((X_\alpha)_\alpha\) is a covering family of the final object and every \(i_{/X_\alpha}\) is an embedding, then \(i\) is an embedding.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.1.7. Subtopoi and multiplicative sets of arrows}
\begin{enumerate}[label=\alph*)]
\item Let \(i:E'\to E\) be a morphism of topoi and let \(S\) be the set of arrows \(u\) of \(E\) such that \(i^*(u)\) is an isomorphism. Let
\[
\rho:E[S^{-1}]\longrightarrow E'
\tag{9.1.7.1}
\]
be the canonical functor induced by \(i^*\). Show that \(i\) is an embedding if and only if \(\rho\) is an equivalence. In this case \(S\) admits both a left and a right calculus of fractions. The essential image of \(i_*\) consists of the objects \(X\in E\) such that, for every \(u:Y\to Y'\) in \(S\), the map
\[
\Hom(Y',X)\longrightarrow\Hom(Y,X)
\]
is bijective.

\item Assume now that \(i\) is an embedding. For any topos \(F\), show that the functor
\[
f^*\longmapsto f^*i^*:\Hhom(E',F)\longrightarrow\Hhom(E,F)
\]
induces an equivalence between inverse-image functors arising from morphisms \(F\to E'\) and inverse-image functors \(g^*:E\to F\) arising from morphisms \(F\to E\) which carry every arrow of \(S\) to an isomorphism.

\item Let \(S\subset \Fl(E)\). Show that \(S\) comes from a subtopos by the preceding construction if and only if it satisfies:
\begin{itemize}
\item[ST1)] \(S\) contains the isomorphisms and is stable under composition and base change.
\item[ST2)] If a composite \(vu\) lies in \(S\), then \(u\in S\) if and only if \(v\in S\).
\item[ST3)] If \(u:X\to Y\) becomes an element of \(S\) after base change by a covering family \(Y_i\to Y\), then \(u\in S\).
\end{itemize}
This gives a bijection between subtopoi of \(E\) and subsets \(S\subset\Fl(E)\) satisfying ST1--ST3. Given \(S\), associate to it the topology \(T'\) finer than the canonical topology for which \((X_i\to X)\) is covering precisely when the image of \(\coprod X_i\to X\) is included by a monomorphism belonging to \(S\).

\item \(S\) is determined by its monomorphisms. The assignment \(S\mapsto S_0\), where \(S_0\) is the subset of monomorphisms in \(S\), gives a bijection between subsets of \(\Fl(E)\) satisfying ST1--ST3 and subsets of monomorphisms satisfying the same conditions.

\item If \((E_i')_{i\in I}\) is a family of subtopoi and \(S_i\) are the corresponding subsets of \(\Fl(E)\), then \(\bigcap_i S_i\) satisfies ST1--ST3 and corresponds to the subtopos generated by the \(E_i'\), that is, to their supremum.

\item Given \(A\subset\Fl(E)\), there is a smallest subset \(S\supset A\) satisfying ST1--ST3. It is the union of subsets \(A_n\), \(n\ge 0\), obtained inductively by adjoining isomorphisms, composites, base changes, factors of composites whose other factor and composite are already present, small sums, and arrows which become present after an epimorphic base change.
\item With the notation of e), if \(A=\bigcup_iS_i\) and \(S\) is obtained from \(A\) as in f), then the corresponding subtopos is the intersection, or infimum, of the \(E_i'\).
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.1.7.2. Canonical factorization of a morphism of topoi}
\begin{enumerate}[label=\alph*)]
\item Let \(f:F\to E\) be a morphism of topoi. Show that, up to isomorphism, \(f\) factors as
\[
F\xrightarrow{f'}E'\xrightarrow{i}E,
\tag{9.1.7.3}
\]
where \(f'\) is conservative and \(i\) is an embedding; this factorization is unique up to equivalence. Introduce \(S_f\subset\Fl(E)\), the arrows \(u\) such that \(f^*u\) is an isomorphism, and take for \(E'\) the subtopos associated with \(S_f\).

\item The subtopos defined by \(i\) is called the \emph{image subtopos} of \(f\). It is the smallest subtopos through which \(f\) factors, or equivalently the smallest subtopos containing the essential image of \(f_*\). The morphism \(f\) is an embedding if and only if it induces an equivalence of \(F\) with its image; it is conservative if and only if its image is all of \(E\).

\item Suppose \(f=\Top(f_0)\) comes from a continuous map \(f_0:Y\to X\) of sober spaces. Let \(Y\to X'=f_0(Y)\to X\) be the usual factorization. The canonical factorization of \(f\) is the corresponding factorization
\[
\xymatrix@C=2cm{\Top(Y)\ar[r]&\Top(X')\ar@{^{(}->}[r]&\Top(X).}
\]
If \(Z\) is the smallest sober subspace of \(X\) containing \(f_0(Y)\), then \(f\) is conservative if and only if \(Z=X\), that is, if \(f_0(Y)\) is very dense in \(X\). Thus conservative morphisms of topoi generalize surjective continuous maps.

\item Give a \(2\)-cartesian square of topoi
\[
\xymatrix{F'\ar[r]\ar[d]&F\ar[d]^f\\E'\ar[r]&E}
\]
with \(f\) conservative but \(f'\) not conservative, for instance \(E'\) a punctual topos and \(F'\) the empty topos.

\item For a family \((E_i)_{i\in I}\) of subtopoi of \(E\), prove that \(E=\bigvee_iE_i\) if and only if the family of embedding morphisms \(E_i\to E\) is conservative.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.1.8. Subtopoi and points; the case of topological spaces}
Let \(E\) be a topos.
\begin{enumerate}[label=\alph*)]
\item Let \(Z\) be a full subcategory of \(\Ppoint(E)\), or the corresponding subset of isomorphism classes of points. Define \(E_Z\) to be the intersection of all subtopoi \(E'\) of \(E\) through which every point of \(Z\) factors. Show that \(E_Z\) is the smallest such subtopos.
\item If \((Z_i)\) is a family of such full subcategories, show that \(E_{\bigcup_iZ_i}\) is the subtopos generated by the \(E_{Z_i}\).
\item If \(E'\subset E\), describe the full subcategory of points of \(E\) coming from \(E'\), and show that \(E'\subset E_Z\) exactly when the points of \(Z\) factor through \(E'\).
\item Let \(E=\Top(X)\) for a sober space \(X\). For a sober subspace \(X'\subset X\), the associated subtopos \(E_{X'}\) is the essential image of \(\Top(X')\to\Top(X)\). Conversely, every subtopos with enough points is of this form.
\item If \((X_i')\) is a family of sober subspaces of \(X\), there is a smallest sober subspace \(X'\) containing them; it is the union when the family is finite or when \(X\) is separated. The subtopos \(E_{X'}\) is generated by the \(E_{X_i'}\).
\item Give an example of a subtopos of a topos of the form \(\Top(X)\) which is nonempty but has no points.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.1.9. Topoi defined by ordered sets}
Use the dictionary between topoi generated by their open subobjects and ordered sets admitting arbitrary suprema, finite infima, and distributivity of finite infima over arbitrary suprema.
\begin{enumerate}[label=\alph*)]
\item If \(f:E'\to E\) is an embedding and \((X_i)\) is a generating family of \(E\), then \((f^*X_i)\) is a generating family of \(E'\). Thus a subtopos of a topos generated by its opens is again generated by its opens.
\item Let \(f:E'\to E\) be a morphism of such topoi, corresponding to a morphism \(g:\Oo\to\Oo'\) of ordered sets. Show that \(f\) is an embedding if and only if \(g\) is surjective and also surjective on the graphs of the order relations; equivalently, the order on \(\Oo'\) is the quotient of the order on \(\Oo\).
\item If \(E=\Oo^\sim\), subtopoi correspond to equivalence relations \(R\) on \(\Oo\) compatible with finite infima and arbitrary suprema. Under this correspondence, the subtopos generated by a family corresponds to the intersection of the corresponding equivalence relations.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.1.10. Intersections of subspaces, point-free topoi, and nonexistence of complements}
Let \(X\) be a topological space and \(E=\Top(X)=\Ouv(X)^\sim\).
\begin{enumerate}[label=\alph*)]
\item Define \(R\) on \(\Ouv(X)\) by \((U,U')\in R\) if \(U\cap U'\) is dense in both \(U\) and \(U'\). Show that \(R\) satisfies the conditions of 9.1.9 and defines a subtopos \(E'\). A point \(x\) of \(X\) comes from \(E'\) if and only if it is \emph{thick}, that is, belongs to the closure of the interior of \(\overline{\{x\}}\). The subtopos \(E'\) is empty only when \(X\) is empty.
\item If \(X\) is sober, \(\Ppoint(E')\) is equivalent to the ordered set of thick points. If the points of \(X\) are closed, the thick points are precisely the isolated points; hence a nonempty separated space without isolated points gives a nonempty subtopos without points.
\item If \(X'\subset X\) and \(E_{X'}\) is as in 9.1.8, then for dense \(X'\), \(E_{X'}\) contains the preceding \(E'\). If \(X\) is sober and \(X'\), \(X''\) are sober subspaces, then \(X'\cap X''\) is sober and
\[
E_{X'\cap X''}\subset E_{X'}\cap E_{X''},
\]
and the inclusion may be strict. It is an equality exactly when the right-hand side has enough points; this holds if one of \(X'\), \(X''\) is locally closed.
\item Let \(X''=X-X'\). If \(X''\) is a union of locally closed subsets, then the subtopoi disjoint from \(E_{X'}\) have a largest element, the \emph{weak complement} of \(E_{X'}\), precisely when \(E_{X'}\cap E_{X''}\) is empty. If this fails, the infimum in the ordered set of subtopoi is not distributive over arbitrary suprema.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.1.11. Subtopoi of locally noetherian topoi}
\begin{enumerate}[label=\alph*)]
\item If \(f:E'\to E\) is an embedding and \(X\) is a prenoetherian object of \(E\), then \(f^*X\) is prenoetherian. Hence if \(E\) has a generating family of prenoetherian objects, or is noetherian or locally noetherian, then \(E'\) has the same property.
\item If \(E\) is locally noetherian, every subtopos of \(E\) is of the form \(E_Z\) for a suitable full subcategory \(Z\) of \(\Ppoint(E)\). In particular, if \(E=\Top(X)\) with \(X\) sober and locally noetherian, then \(Z\mapsto E_Z\) identifies sober subspaces of \(X\) with subtopoi of \(E\), and intersections of sober subspaces correspond to intersections of subtopoi.
\item For \(X\) sober locally noetherian and \(X'\subset X\), with complement \(X''\), the following are equivalent: \(X'\) is constructible; both \(X'\) and \(X''\) are sober; \(E_{X'}\) has a complementary subtopos; and \(E_{X'}\cap E_{X''}\) is empty.
\item For \(X\) sober noetherian, the following are equivalent: every sober subspace is constructible; it is enough to test \(X-\{x\}\) for maximal \(x\); \(X\) is finite; every subtopos of \(\Top(X)\) has a complement; and infima distribute over arbitrary suprema.
\item If \(X\) is locally noetherian, then in the ordered set of subtopoi of \(\Top(X)\) finite suprema distribute over binary infima.
\item For every sober subspace \(Z\) of a sober locally noetherian \(X\), define a topology \(T_Z\) on \(\Ouv(X)\) by declaring \((U_i\to U)\) covering when \(U\cap Z=\bigcup_i(U_i\cap Z)\). Then \(Z\mapsto T_Z\) is an order-reversing bijection from sober subspaces of \(X\) to topologies on \(\Ouv(X)\) finer than the canonical topology.
\item For a sober topological space \(X\), a subset is sober if and only if its complement is a union of locally closed subsets. If \(X\) is locally noetherian this is the same as proconstructibility. The constructible topology \(X_{\cons}\) identifies the ordered set of subtopoi of \(\Top(X)\) with the closed subsets of \(X_{\cons}\); complemented subtopoi correspond to clopen subsets.
\item If \(E\) is locally noetherian and has the property stated in 9.1.14 b), then the essential images of \(\Ppoint(E')\to\Ppoint(E)\), with \(E'\) running over all subtopoi, are the closed subsets of a topology \(T_{\cons}\) on the set of isomorphism classes of points, and the ordered set of subtopoi is identified with these closed subsets.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.1.12. Finite topoi}
A topos \(E\) is called \emph{finite} if it is equivalent to \(\widehat C\) for a finite category \(C\).
\begin{enumerate}[label=\alph*)]
\item There are quasi-inverse \(2\)-equivalences between Karoubian categories equivalent to finite categories and finite topoi:
\[
C\longmapsto\widehat C,\qquad E\longmapsto\Ppoint(E).
\]
\item A finite topos is noetherian.
\item If \(E\simeq\widehat C\), with \(C\) Karoubian and finite up to equivalence, then subtopoi of \(E\) correspond to strictly full Karoubian subcategories \(C'\subset C\). The subtopos is the essential image of \(\widehat{C'}\to\widehat C\).
\item The finite set \(X=\Point(E)\) of isomorphism classes of points carries a canonical sober topology, for which subtopoi of \(E\) correspond to closed subsets of \(X\). Thus the ordered set of subtopoi is finite.
\item For a category \(C\) equivalent to a finite category and a full subcategory \(C'\), define a topology \(T_{C'}\) by declaring \((X_i\to X)\) covering when, for every \(Y\in C'\), the maps \(\Hom(Y,X_i)\to\Hom(Y,X)\) are jointly surjective. This gives an order-reversing bijection from strictly full Karoubian subcategories of \(C\) to topologies on \(C\).
\item If \(f:C'\to C\) and \(C\) is equivalent to a finite category, then \(\widehat f:\widehat{C'}\to\widehat C\) is conservative if and only if every object of \(C\) is a direct factor of an object in the image of \(f\).
\item For the category with objects \(e,a\) and arrows \(f:a\to e\), \(g:e\to a\), \(p=gf\), \(p^2=p\), show that \(\widehat C\) has exactly one proper nonempty subtopos, namely \(\widehat{\{e\}}\). Thus any two nonempty subtopoi meet nontrivially.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.1.13. Complementary subtopoi}
Two subtopoi \(E'\), \(E''\) of \(E\) are called \emph{complementary} if
\[
E'\cap E''=\emptytopos,\qquad E'\vee E''=E.
\]
Assume binary infima distribute over finite suprema in the ordered set of subtopoi.
\begin{enumerate}[label=\alph*)]
\item If \(E'\) and \(E''\) are complementary, then \(E''\) is the largest subtopos disjoint from \(E'\).
\item The following are equivalent: every weak complement is a complement; every proper subtopos has a nonempty disjoint subtopos; every maximal subtopos disjoint from a given one is a complement. These conditions may fail even for finite topoi, but hold for \(\Top(X)\) with \(X\) locally noetherian.
\item Complemented subtopoi are stable under finite infima, finite suprema, and complements; complementation interchanges infima and suprema.
\item State the dual assertions.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{9.1.14. Open questions}
Let \(E\) be a topos.
\begin{itemize}
\item[a)] In the ordered set of subtopoi of \(E\), is the infimum of two elements distributive over finite suprema?
\item[b)] If \(E'\vee E''=E\), is every point of \(E\) isomorphic to the image of a point of \(E'\) or of \(E''\)?
\item[c)] If \(E'\) and \(E''\) are subtopoi of a topos \(F\) whose supremum is \(F\), equivalently if
\[
E'\amalg E''\longrightarrow F
\tag{9.1.14.1}
\]
is conservative, is this morphism universally conservative, that is, still conservative after every \(2\)-base change?
\item[d)] In the ordered set \(\Sigma\) of subtopoi of \(E\), is the infimum of two elements distributive over arbitrary suprema?
\item[e)] If \(F\) is noetherian and \((E_i)_{i\in I}\) is a family of subtopoi with intersection \(\emptytopos\), does a finite subfamily already have empty intersection?
\end{itemize}
Positive answers to a) and d) were later supplied in the literature cited in the French edition.
\end{remarktext}

\subsection*{9.2. The case of open subtopoi}

Let \(U\) be an open of \(E\), i.e. a subobject of the final object \(e\). The localization morphism
\[
j:E_{/U}\longrightarrow E
\tag{9.2.2}
\]
is an embedding. An embedding is called an \emph{open embedding} if it is isomorphic to one so obtained; the corresponding subtopos is an \emph{open subtopos}. The open \(U\) attached to an open embedding \(j:E'\to E\) is uniquely determined as the subobject \(j_!(e')\subset e\), where \(e'\) is the final object of \(E'\). Thus opens of \(E\) are in bijection with open subtopoi of \(E\).

\begin{remarktext}{Remark 9.2.3}
The open subtopos defined by \(U\) is the essential image of \(j_*:E_{/U}\to E\). It must not be confused with the essential image of \(j_!\), namely the objects whose structural morphism to \(e\) factors through \(U\). These two subcategories determine one another and are canonically equivalent, but the former is the subtopos in the sense fixed here.
\end{remarktext}

\begin{statement}{Proposition 9.2.4}
Let \(j:\Uu\to E\) be an open embedding. Then \(j_!\), left adjoint to \(j^*\), exists; hence there is a chain of adjoint functors
\[
j_!,\quad j^*,\quad j_*.
\]
Moreover:
\begin{enumerate}[label=\alph*)]
\item \(j_!\) and \(j_*\) are fully faithful.
\item \(j_!\) commutes with fiber products, with products indexed by nonempty small sets, and with projective limits over small cofiltered categories.
\item For every \(X\in E\), the adjunction morphism \(j_!j^*X\to X\) is a monomorphism.
\end{enumerate}
\end{statement}

\begin{prooftext}
For the localization defined by an open \(U\), the functor \(j_!\) is the forgetful functor. The assertions follow from the fact that \(U\to e\) is a monomorphism and monomorphisms are stable under base change.
\end{prooftext}

\begin{remarktext}{Remark 9.2.4.1}
There are embeddings \(j:E'\to E\) for which \(j_!\) exists, but \(j_!\) does not commute with binary products or cofiltered projective limits.
\end{remarktext}

Let \(g:E'\to E\) be any morphism of topoi and put \(U'=g^*U\). The square
\[
\xymatrix@C=2cm@R=1.5cm{
E'_{/U'}\ar[d]_{g_U}\ar[r]^{j'}&E'\ar[d]^g\\
E_{/U}\ar[r]^j&E
}
\]
is \(2\)-cartesian. Thus inverse images of open subtopoi are open subtopoi, and open embeddings are stable under \(2\)-base change. Moreover, \(g\) factors through \(E_{/U}\) if and only if \(U'=e_{E'}\).

For two opens \(U,V\) of \(E\), the \(2\)-fiber product of \(E_{/U}\) and \(E_{/V}\) over \(E\) is \(E_{/U\cap V}\). Hence finite intersections of open subtopoi are open. The assignment \(U\mapsto E_{/U}\) also commutes with arbitrary suprema.

\subsection*{9.3. Construction of the closed subtopos complementary to an open subtopos}

Let \(T\) be a topological space, \(U\subset T\) an open, and \(Y=T-U\) the closed complement. Then \(\Top(Y)\) may be regarded as the subtopos of \(\Top(T)\) consisting of objects whose restriction to \(U\) is the final object. This description makes sense for any topos \(E\) and open \(U\subset e\), and it defines the closed subtopos complementary to \(U\).

Let
\[
j:\Uu=E_{/U}\longrightarrow E
\tag{9.3.2.1}
\]
be the localization. For \(X\in E\), define
\[
X_{\complement U}=U\coprod_{U\times X}X,
\tag{9.3.2.2}
\]
where the amalgamation is along the projections \(U\times X\to U\) and \(U\times X\to X\). Thus
\[
\xymatrix@C=1.75cm@R=1.5cm{
U\times X\ar[r]^{\pr_2}\ar[d]_{\pr_1}&X\ar[d]\\
U\ar[r]&X_{\complement U}
}
\tag{9.3.2.3}
\]
is cocartesian.

\begin{statement}{Proposition 9.3.3}
With the preceding notation:
\begin{enumerate}[label=\alph*)]
\item For \(X\in E\), the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\item \(X\simeq Y_{\complement U}\) for some \(Y\in E\);
\item \(X\to X_{\complement U}\) is an isomorphism;
\item \(X\times U\to U\) is an isomorphism, equivalently \(j^*X\) is final in \(E_{/U}\).
\end{enumerate}
\item For a morphism \(m:X\to X'\), the following are equivalent:
\begin{enumerate}[label=(\roman*')]
\item the square
\[
\xymatrix{X\ar[r]^m\ar[d]&X'\ar[d]\\
X_{\complement U}\ar[r]&X'_{\complement U}}
\]
is cartesian;
\item \(m\times\id_U:X\times U\to X'\times U\) is an isomorphism, equivalently \(j^*m\) is an isomorphism.
\end{enumerate}
\item The functor \(X\mapsto X_{\complement U}\) is left exact, sends epimorphic families to epimorphic families, and commutes with filtered inductive limits and pushouts.
\end{enumerate}
\end{statement}

\begin{prooftext}
First prove the assertion when \(E=\widehat C\) and reduce, by the usual arguments, to the punctual topos \(\Set\), where it is immediate. The general case follows by the standard sheafification procedure.
\end{prooftext}

\begin{statement}{Proposition 9.3.4}
Let \(\Ff\) be the strictly full subcategory of \(E\) consisting of objects \(X\) such that \(j^*X\) is final in \(\Uu\). Then \(\Ff\) is a subtopos of \(E\). Its inclusion \(i:\Ff\to E\) has inverse-image functor
\[
i^*X=X_{\complement U}.
\]
The adjunction morphism \(X\to i_*i^*X\) is the canonical morphism \(X\to X_{\complement U}\).
\end{statement}

\begin{prooftext}
Let \(u(X):X\to X_{\complement U}\). This morphism is functorial in \(X\), and \(u(X_{\complement U})\) is an isomorphism. These properties formally give an adjunction \(i^*\dashv i_*\), with \(i^*(X)=X_{\complement U}\). Since \(i^*\) is left exact, the general criterion for a full subcategory to be a topos applies.
\end{prooftext}

The subtopos \(\Ff\) is called the \emph{closed subtopos of \(E\) complementary to the open \(U\)}. Its inclusion \(i:\Ff\to E\) is a \emph{closed embedding}. Conversely a subtopos is called closed if it is obtained in this way. The complementary open is recovered as \(i_*(\varnothing_\Ff)\subset e_E\). For a group object \(G\) on \(E\), and for a section \(s\in\Hom(e_E,G)\), the \emph{support} of \(G\) or \(s\) is the closed subtopos complementary to its cosupport.

With \(j:\Uu\to E\) and \(i:\Ff\to E\), we have the two adjoint strings
\[
\xymatrix@C=2cm@R=1.5cm@M=.2cm{
\Uu\ar@<6pt>[r]^{j_!}\ar@<-13pt>[r]_{j_*}&
E\ar@<3pt>[r]^{i^*}\ar@<4pt>[l]_{j^*}&
\Ff\ar@<13pt>[l]_{i_*}.}
\tag{9.3.6.2}
\]
The functor
\[
\rho=i^*j_*:\Uu\longrightarrow\Ff
\tag{9.3.6.3}
\]
is called the \emph{gluing functor} relative to \(U\). More generally, a functor \(\rho:\Uu\to\Ff\) between topoi is called a gluing functor if it is left exact and accessible, equivalently if it admits a pro-adjoint.

\subsection*{9.4. First properties of the closed subtopos \(\Ff\)}

\begin{statement}{Proposition 9.4.1}
With the preceding notation:
\begin{enumerate}[label=\alph*)]
\item \(i_*:\Ff\to E\) sends epimorphic families to epimorphic families and commutes with pushouts and filtered inductive limits.
\item If \(X\in E\) and \(U\times X\to U\) is an epimorphism, then \(X\to i_*i^*X\) is an epimorphism.
\item The pair of functors \((i^*,j^*)\) from \(E\) to \(\Ff\) and \(\Uu\) is conservative.
\item The gluing functor \(\rho=i^*j_*:\Uu\to\Ff\) is left exact and accessible; equivalently it admits a pro-adjoint
\[
\sigma:\Pro(\Ff)\longrightarrow\Pro(\Uu).
\]
\end{enumerate}
\end{statement}

\begin{prooftext}
The assertions follow directly from 9.3.3 and the construction of \(X_{\complement U}\). For d), \(j_*\) is a direct image functor and is accessible, while \(i^*\) commutes with inductive limits; the composite is therefore accessible.
\end{prooftext}

\begin{statement}{Proposition 9.4.2}
For any topos \(E'\), the functor
\[
h\longmapsto i\circ h:\Homtop(E',\Ff)\longrightarrow\Homtop(E',E)
\]
is fully faithful. Its essential image consists of those morphisms \(g:E'\to E\) such that \(g^*(U)\) is an initial object of \(E'\).
\end{statement}

\begin{prooftext}
Full faithfulness is 9.1.4. The criterion follows by writing \(U'=g^*U\) and considering the square
\[
\xymatrix@C=1.5cm{
E'_{/U'}\ar[r]^{j'}\ar[d]_{g_U}&E'\ar[d]^g\\
E_{/U}\ar[r]^j&E.
}
\]
The condition that \(g_*(X')\) lie in \(\Ff\) for all \(X'\) is equivalent to \(g_{U*}(Y')\) being final for every \(Y'\in E'_{/U'}\), and by adjunction this is equivalent to \(U'\) being initial in \(E'\).
\end{prooftext}

\begin{statement}{Corollary 9.4.3}
Let \(g:E'\to E\) be a morphism of topoi, let \(U'=g^*U\), and let \(\Ff'\) be the closed subtopos of \(E'\) complementary to \(U'\). Then \(g\circ i':\Ff'\to E\) factors, up to isomorphism, through a morphism \(g_\Ff:\Ff'\to\Ff\), and the square
\[
\xymatrix@C=2cm@R=1.4cm{
\Ff'\ar[r]^{i'}\ar[d]_{g_\Ff}&E'\ar[d]^g\\
\Ff\ar[r]^i&E
}
\tag{9.4.3.1}
\]
is \(2\)-cartesian.
\end{statement}

Thus inverse images of closed subtopoi are closed subtopoi.

\begin{statement}{Corollary 9.4.4}
Let \(\Uu\) be an open subtopos of \(E\) and \(\Ff\) its closed complement.
\begin{enumerate}[label=\alph*)]
\item
\[
\Uu\cap\Ff\simeq\emptytopos,\qquad \Uu\vee\Ff=E.
\]
Thus \(\Uu\) and \(\Ff\) are complementary subtopoi.
\item \(\Uu\) is the largest subtopos \(E'\) with \(E'\cap\Ff\simeq\emptytopos\), and \(\Ff\) is the largest subtopos \(E'\) with \(\Uu\cap E'\simeq\emptytopos\).
\end{enumerate}
\end{statement}

\begin{prooftext}
The first assertion reduces to the criteria 9.2.5 and 9.4.2 for a morphism to factor through the open or through the closed complement. The equality \(\Uu\vee\Ff=E\) is equivalent to conservativity of \((i^*,j^*)\). The maximality statements follow by applying 9.4.3 to a subtopos \(E'\subset E\).
\end{prooftext}

\begin{statement}{Corollary 9.4.5}
If \((\Ff_i)_{i\in I}\) is a family of closed subtopoi of \(E\), and \(U_i\) is the complementary open of \(\Ff_i\), then \(\bigcap_i\Ff_i\) is closed and its complementary open is \(U=\Sup_i U_i\).
\end{statement}

\begin{statement}{Corollary 9.4.6}
If the family in 9.4.5 is finite, then the supremum \(\bigvee_i\Ff_i\) is closed and its complementary open is \(\bigcup_iU_i\).
\end{statement}

\begin{remarktext}{Exercise 9.4.7. Gluing subtopoi}
Let \(E'\) be a subtopos of \(E\), and let \((S_i)_{i\in I}\) be a covering family of the final object of \(E\). Write \(E_i=E_{/S_i}\) and \(E_i'\) for the inverse-image subtopos of \(E'\) in \(E_i\). Show that an object \(X\in E\) belongs to \(E'\) if and only if every restriction \(X|S_i\) belongs to \(E_i'\). Conversely, if subtopoi \(E_i'\subset E_i\) are compatible on overlaps \(E_{/S_i\times S_j}\), construct a unique subtopos \(E'\subset E\) inducing them.
\end{remarktext}

\begin{remarktext}{Exercise 9.4.8. Exterior, closure, interior, and boundary}
For a subtopos \(E'\subset E\), define its exterior as the largest open subtopos disjoint from \(E'\), its closure as the closed complement of this exterior, its interior as the largest open subtopos contained in \(E'\), and its boundary as the closed complement of the union of the exterior and the interior. Show the expected formal properties, including compatibility with induced topoi. The exterior of \(E'\) is empty exactly when the embedding \(E'\to E\) is dominant.
\end{remarktext}

\begin{remarktext}{Exercise 9.4.9. Locally closed subtopoi}
A subtopos \(F\) of \(E\) is \emph{locally closed} if it is the intersection of an open subtopos and a closed subtopos.
\begin{enumerate}[label=\alph*)]
\item Finite intersections of locally closed subtopoi are locally closed, and inverse images of locally closed subtopoi are locally closed.
\item For \(E=\Top(X)\), locally closed subsets \(X'\subset X\) give locally closed subtopoi \(T(X')\); conversely, under sobriety assumptions, every locally closed subtopos comes in this way.
\item A subtopos \(F\subset E\) is locally closed if and only if it is open in its closure. This is a local property on \(E\).
\item Among all expressions \(F=\Uu\cap\Ff\) with \(\Uu\) open and \(\Ff\) closed, there is a canonical one with \(\Uu\) largest and \(\Ff\) smallest, namely \(\Ff=\overline F\) and \(\Uu\) the largest open inducing \(F\) on \(\Ff\).
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.4.10}
Develop the notions of constructible, locally constructible, quasi-constructible, and locally quasi-constructible subtopoi on the model of the analogous notions for topological spaces. For \(E=\Top(X)\), these notions recover the corresponding classes of subsets of \(X\).
\end{remarktext}

\subsection*{9.5. The gluing theorem}

Let \(f:A\to B\) be a functor. Denote by \((B,A,f)\) the category whose objects are triples \((X,Y,u)\), with \(X\in B\), \(Y\in A\), and \(u:X\to f(Y)\); a morphism \((X,Y,u)\to(X',Y',u')\) is a pair \((m,m')\), with \(m:X\to X'\), \(m':Y\to Y'\), such that
\[
\xymatrix@C=1.5cm@R=1cm{
X\ar[r]^u\ar[d]_m&f(Y)\ar[d]^{f(m')}\\
X'\ar[r]^{u'}&f(Y')
}
\]
commutes. This construction is functorial in \(A,B,f\); in particular a natural transformation \(f\to f'\) gives a functor \((B,A,f)\to(B,A,f')\), an isomorphism if the transformation is one.

Return to \(E\), an open subtopos \(\Uu\), its closed complement \(\Ff\), and \(\rho=i^*j_*:\Uu\to\Ff\). To \(X\in E\) attach
\[
(i^*X,\ j^*X,\ h(X):i^*X\to \rho j^*X),
\]
where \(h(X)\) is obtained by applying \(i^*\) to \(X\to j_*j^*X\). This defines a functor
\[
\Phi:E\longrightarrow(\Ff,\Uu,\rho).
\tag{9.5.3.1}
\]

\begin{statement}{Theorem 9.5.4}
\begin{enumerate}[label=\alph*)]
\item Let \(E\) be a topos, \(\Uu\) an open subtopos, and \(\Ff\) its closed complement. Then \(\rho=i^*j_*:\Uu\to\Ff\) is left exact and accessible, and
\[
\Phi:E\longrightarrow(\Ff,\Uu,\rho)
\]
is an equivalence of categories.
\item Conversely, if \(\Uu\), \(\Ff\) are topoi and \(\rho:\Uu\to\Ff\) is a gluing functor, then \(E=(\Ff,\Uu,\rho)\) is a topos. The object
\[
U=(\varnothing_\Ff,e_\Uu,\varnothing_\Ff\to\rho(e_\Uu))
\]
defines an open subtopos equivalent to \(\Uu\), and the functor
\[
Y\longmapsto (Y,e_\Uu,Y\to\rho(e_\Uu))
\]
identifies \(\Ff\) with the closed complement. The resulting gluing functor is compatible with the given \(\rho\).
\end{enumerate}
\end{statement}

\begin{statement}{Definition 9.5.5}
For topoi \(\Uu,\Ff\) and a gluing functor \(\rho:\Uu\to\Ff\), the topos \((\Ff,\Uu,\rho)\) is called the topos obtained by \emph{gluing} \(\Uu\) and \(\Ff\) by means of \(\rho\).
\end{statement}

\begin{prooftext}\emph{Proof of 9.5.4 a).} 
The pair \((i^*,j^*)\) is conservative, so \(\Phi\) is faithful. For every \(X\in E\), the square
\[
\xymatrix@C=2.5cm{
X\ar[r]\ar[d]&j_*j^*X\ar[d]\\
i_*i^*X\ar[r]&i_*i^*j_*j^*X
}
\tag{*}
\]
is cartesian, by 9.3.3. Given a morphism between \(\Phi(X)\) and \(\Phi(Y)\), the universal property of this cartesian square produces a unique \(X\to Y\), showing full faithfulness. Conversely, an object \((Y,X,u:Y\to i^*j_*X)\) of \((\Ff,\Uu,i^*j_*)\) is obtained from the fiber product
\[
\xymatrix@C=2.75cm@R=1.25cm{
W\ar[r]\ar[d]&j_*X\ar[d]\\
i_*Y\ar[r]^{i_*u}&i_*i^*j_*X.
}
\]
Then \(\Phi(W)\simeq(Y,X,u)\), so \(\Phi\) is essentially surjective.
\end{prooftext}

\begin{prooftext}\emph{Proof of 9.5.4 b).} 
It remains chiefly to see that \((\Ff,\Uu,\rho)\) is a topos. Finite projective limits are computed componentwise, since \(\rho\) is left exact. Sums are constructed by
\[
\coprod_i(Y_i,X_i,u_i)=
\left(\coprod_iY_i,\ \coprod_iX_i,\ \coprod_iY_i\to\rho(\coprod_iX_i)\right),
\]
using the canonical map \(\coprod_i\rho(X_i)\to\rho(\coprod_iX_i)\). Equivalence relations are effective and universal by applying the corresponding assertions in \(\Ff\) and \(\Uu\). Finally, accessibility of \(\rho\), or equivalently the existence of a pro-adjoint \(\sigma:\Pro(\Ff)\to\Pro(\Uu)\), supplies a small generating family: take generators \(C'\) of \(\Uu\), \(C''\) of \(\Ff\), represent each \(\sigma(X'')\) by a small filtered projective system, and use the objects
\[
(\varnothing_\Ff,X',\varnothing_\Ff\to\rho X')\quad\text{and}\quad
(X'',\sigma(X'')_i,X''\to\rho(\sigma(X'')_i)).
\]
\end{prooftext}

For \(E=(\Ff,\Uu,\rho)\), the structural functors are explicitly:
\[
i^*(X,Y,u)=X,\qquad
i_*(X)=(X,e_\Uu,X\to\rho(e_\Uu)),
\]
\[
j^*(X,Y,u)=Y,\qquad
j_!(Y)=(\varnothing_\Ff,Y,\varnothing_\Ff\to\rho(Y)),
\]
\[
j_*(Y)=(\rho(Y),Y,\id_{\rho(Y)}).
\]
The adjunction morphisms are the evident morphisms represented by the arrows \(u:X\to\rho(Y)\) and by the maps to \(\rho(e_\Uu)\).

\begin{remarktext}{Exercise 9.5.9. Exactness properties of a glued topos}
Let \(E=(\Ff,\Uu,\rho)\).
\begin{enumerate}[label=\alph*)]
\item Show that \(j_!\) preserves all inductive limits and \(j^*\) preserves both all inductive limits and finite projective limits. Show that \(i^*\) preserves inductive limits, and \(i_*\) preserves filtered inductive limits and pushouts. If \(\rho\) preserves a given class of projective limits, then \(j_*\) preserves it.
\item Give an example where \(j_!\) is not exact on the right for arbitrary embeddings, and one where \(j_!\) fails to preserve products.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.5.10}
Let \(E=(\Ff,\Uu,\rho)\), and let \(A\) be an object of \(E\) corresponding to \((B,C,u)\). Describe the induced topos \(E_{/A}\) as a glued topos built from \(\Ff_{/B}\) and \(\Uu_{/C}\), and identify its gluing functor.
\end{remarktext}

\begin{remarktext}{Exercise 9.5.11. Morphisms from a glued topos}
Let \(F\) be a category or a topos.
\begin{enumerate}[label=\alph*)]
\item Describe \(\Hhom(F,E)\) as a category glued from \(\Hhom(F,\Uu)\) and \(\Hhom(F,\Ff)\), with gluing functor induced by \(\rho\).
\item A functor \(f:F\to E\) commutes with a given class of inductive limits if and only if \(j^*f\) and \(i^*f\) do; for projective limits also require \(\rho\) to commute with them.
\item If \(F\) is a topos, then \(f\) is an inverse-image functor of a morphism \(E\to F\) if and only if \(j^*f\) and \(i^*f\) are. Hence \(\Hhomtop(E,F)\) is equivalent to the category of triples \((f',f'',u)\), where \(f'\in\Hhomtop(\Uu,F)\), \(f''\in\Hhomtop(\Ff,F)\), and \(u:{f''}^*\to\rho\,{f'}^*\) is a morphism of functors \(F\to\Ff\).
\item Spell out in what sense this gives a characterization of \(E\), up to equivalence in the \(2\)-category of topoi, by the triple \((\Uu,\Ff,\rho)\).
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.5.12. Gluing of topological spaces}
\begin{enumerate}[label=\alph*)]
\item Let \(X\) be a topological space and \(\rho:\Top(X)\to\Set\) a gluing functor, represented by a pro-object \(A\in\Pro(\Top(X))\). Show that the glued topos is of the form \(\Top(Y)\) if and only if \(A\) is isomorphic to an object of \(\Pro(\Ouv(X))\). If \(X\ne\varnothing\), there are choices of \(\rho\) for which the glued topos is not of the form \(\Top(Y)\).
\item More generally, for \(\rho:\Top(X)\to\Top(Y)\), use the pro-adjoint \(\sigma\) to obtain a map
\[
\phi:X\longrightarrow \Ob\Pro(\Top(Y)).
\]
If the glued topos is \(\Top(Z)\), then \(\phi\) takes values in the essential image of \(\Pro(\Ouv(Y))\).
\end{enumerate}
\end{remarktext}

\subsection*{9.6. The gluing sheaf}

The data of a topos \(E\) with an open \(U\) is, by 9.5.4, essentially the same as the data of two topoi \(\Uu\), \(\Ff\) and a gluing functor
\[
\rho:\Uu\longrightarrow\Ff,
\tag{9.6.1}
\]
or equivalently a pro-adjoint
\[
\sigma:\Pro(\Ff)\longrightarrow\Pro(\Uu).
\tag{9.6.2}
\]
The functor \(\sigma\) is determined by its restriction
\[
\sigma_\circ:\Ff\longrightarrow\Pro(\Uu),
\tag{9.6.3}
\]
extended to filtered projective limits. Conversely, \(\sigma_\circ\) arises from a pro-adjoint exactly when, for every \(X\in\Uu\), the functor
\[
Y\longmapsto \Hom_{\Pro(\Uu)}(\sigma_\circ(Y),X)
\]
on \(\Ff^{\circ}\) is representable, equivalently a sheaf for the canonical topology. Thus the opposite functor
\[
\sigma_\circ^{\circ}:\Ff^\circ\longrightarrow\Pro(\Uu)^\circ\subset\check{\Uu}=\Hhom(\Uu,\Set)
\tag{9.6.4}
\]
is a sheaf on \(\Ff\) with values in \(\Pro(\Uu)^\circ\). The data of a gluing functor is therefore equivalent to such a sheaf. This sheaf is called the \emph{gluing sheaf} attached to \(\rho\), or to the open \(U\) of \(E\).

\begin{remarktext}{Exercise 9.6.5}
Refine 9.5.4 to a statement of \(2\)-equivalence between the \(2\)-category of pairs \((E,U)\), where \(E\) is a topos and \(U\) an open, and the \(2\)-category of triples \((\Uu,\Ff,\rho)\) with \(\rho\) a gluing functor. Give the corresponding variant where \(\rho\) is replaced by the gluing sheaf \(\phi\) on \(\Ff\) with values in \(\Pro(\Uu)^\circ\).
\end{remarktext}

\subsection*{9.7. Points of a glued topos}

Let \(C\) be a category, and let \(D'\to C\), \(D''\to C\) be fully faithful functors with essential images \(C'\), \(C''\), such that the objects of \(C\) are partitioned by \(C'\) and \(C''\), and such that
\[
\Hom(X'',X')=\varnothing\qquad(X'\in C',\ X''\in C'').
\]
Then \(C\) is reconstructed from \(C'\), \(C''\), and the functor
\[
(X',X'')\longmapsto\Hom(X',X''):{C'}^\circ\times C''\to\Set.
\]
This applies to the points of a glued topos.

\begin{statement}{Proposition 9.7.2}
Let \(E\) be a topos, \(U\) an open, \(\Uu=E_{/U}\), and \(\Ff\) the closed complement. The inclusion morphisms \(j:\Uu\to E\), \(i:\Ff\to E\) induce
\[
u:\Ppoint(\Uu)\longrightarrow\Ppoint(E),\qquad
v:\Ppoint(\Ff)\longrightarrow\Ppoint(E).
\tag{9.7.2.1}
\]
Then:
\begin{enumerate}[label=\alph*)]
\item \(u\) and \(v\) are fully faithful, and every point of \(E\) belongs to exactly one of their essential images.
\item If \(p\) comes from \(\Uu\) and \(q\) comes from \(\Ff\), then
\[
\Hom(q,p)=\varnothing.
\tag{9.7.2.2}
\]
\item For \(p\in\Ppoint(\Uu)\) and \(q\in\Ppoint(\Ff)\), there are functorial isomorphisms
\[
\Hom(u(p),v(q))\simeq\Hom(\Pro(\rho)(p),q)\simeq\Hom(p,\sigma(q)),
\tag{9.7.2.3}
\]
where \(\sigma:\Pro(\Ff)\to\Pro(\Uu)\) is pro-adjoint to \(\rho\).
\end{enumerate}
\end{statement}

\begin{prooftext}
Full faithfulness is 9.1.4. A point factors through \(\Uu\) or \(\Ff\) according as \(p^*U\) is final or initial in the punctual topos; these are the only two possibilities. For c), use the commutative diagram
\[
\xymatrix@C=2cm@R=.55cm{
\Ppoint(\Uu)\ar@{^{(}->}[d]\ar[r]^u&\Ppoint(E)\ar@{^{(}->}[d]&\Ppoint(\Ff)\ar@{^{(}->}[d]\ar[l]_v\\
\Pro(\Uu)\ar[r]^{\Pro(j_*)}&\Pro(E)&\Pro(\Ff)\ar[l]_{\Pro(i_*)}
}
\]
and the adjunction \(\Pro(i^*)\dashv\Pro(i_*)\).
\end{prooftext}

\begin{statement}{Lemma 9.7.2.4}
For a morphism of topoi \(f:F\to E\), the diagram
\[
\xymatrix@C=2cm@R=.5cm{
\Ppoint(F)\ar[r]^{p\mapsto f\circ p}\ar[d]&\Ppoint(E)\ar[d]\\
\Pro(F)\ar[r]^{\Pro(f_*)}&\Pro(E)
}
\tag{9.7.2.4.1}
\]
commutes up to canonical isomorphism.
\end{statement}

\begin{statement}{Corollary 9.7.3}
If \(E\) has enough points, then every closed subtopos \(\Ff\) of \(E\) has enough points. More precisely, if \((p_i)_{i\in I}\) is a conservative family of points of \(E\), then the subfamily consisting of those \(p_i\) which come from \(\Ff\) is conservative on \(\Ff\).
\end{statement}

\begin{remarktext}{Remark 9.7.4}
If \(E\) has enough points, an open \(U\) is determined by the full subcategory of \(\Ppoint(E)\) consisting of points factoring through \(E_{/U}\). Therefore a closed subtopos is also determined by the essential image of its category of points in \(\Ppoint(E)\); it even suffices to know this intersection with any conservative family of points.
\end{remarktext}

\begin{remarktext}{Exercise 9.7.5}
For a subtopos \(F\subset E\), let \(P(F)\subset\Point(E)\) be the set of isomorphism classes of points factoring through \(F\). If \(F\) is open, closed, or locally closed, then \(P(F)\) is respectively open, closed, or locally closed in \(\Point(E)\). If \(E\) has enough points, this gives bijections between open, closed, locally closed subtopoi of \(E\) and open, closed, locally closed subsets of \(\Point(E)\). Generalize to any subspace of \(\Point(E)\) coming from a conservative family of points.
\end{remarktext}

\subsection*{9.8. Complements on certain topoi attached to topological spaces}

The following exercises are meant to accustom the reader to the point of view of topoi in classical-looking situations.

\begin{remarktext}{Exercise 9.8.1. Descent of sheaves on a topological space}
Let \(f:X\to Y\) be a continuous map and \(f^*:\Top(Y)\to\Top(X)\).
\begin{enumerate}[label=\alph*)]
\item The following are equivalent: \(f^*\) is faithful; \(f^*\) is conservative; and \(f(X)\) is very dense in \(Y\). It is enough that \(f\), or \(f_{\sob}\), be surjective. Give an example where \(f^*\) is faithful but \(f_{\sob}\) is not surjective.
\item Let \(R=X\times_YX\rightrightarrows X\). Show that a sheaf on \(Y\) is the same as a sheaf on \(X\) with descent datum relative to this equivalence relation exactly when \(f^*\) is fully faithful. Relate this to very density.
\item Study when the usual descent category is effective for open coverings, for surjective local homeomorphisms, and more generally for maps satisfying suitable local section conditions.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.8.2. \'Etendues and spaces with operators}
\begin{enumerate}[label=\alph*)]
\item Define an \emph{\'etendue} as a topos locally equivalent to the topos of a topological space. Show that if \(X\) is a space with a discrete group \(G\) of operators, the topos \(\Top(X/G)\) is an \'etendue.
\item For \(x\in X\), identify the monoid of endomorphisms of the corresponding point of \(\Top(X,G)\) with the stabilizer group \(G_x\). Thus points of an \'etendue may have nontrivial automorphism groups.
\item Determine \(\Ppoint(T)\) for the \'etendue defined by an \'etale pre-equivalence relation on \(X\), and show that every endomorphism of a point is an automorphism.
\item A topos \(T\) is \emph{pointwise rigid}, or simply \emph{rigid}, if between any two of its points there is at most one morphism. Show that for an \'etale pre-equivalence relation \(R\) on \(X\), the quotient \'etendue \(\Top(X)/R\) is rigid if and only if \(R\) is an equivalence relation.
\item Let \(G\) act on \(X\), let \(H\triangleleft G\) act properly and freely, put \(X'=X/H\), \(G'=G/H\). Show that \(\Top(X,G)\to\Top(X',G')\) is an equivalence of topoi. Identify opens of \(\Top(X/G)\) with \(G\)-stable opens of \(X\), and express connectedness and \(\pi_0\) in terms of the \(G\)-action.
\item If \(X\) is locally connected and locally simply connected and \(T=\Top(X,G)\) is connected, show that there is a realization \(\Top(X',G')\simeq T\) with \(X'\) connected and simply connected, unique up to nonunique isomorphism. This is equivalent to choosing a universal covering of the final object of \(T\), and \(G'\) is the corresponding fundamental group.
\item Characterize the connected, locally connected, locally simply connected \'etendues which arise from such spaces with operators. Give an example of a connected simply connected \'etendue locally isomorphic to \(\mathbb R\) which is not a topological space, for instance from the universal covering of the line with doubled origin.
\item Define differentiable, real analytic, and complex analytic \'etendues as ringed topoi locally of the corresponding form. Give their construction by spaces with operators and examples which are not ringed topoi of ordinary spaces.
\end{enumerate}
\end{remarktext}

\begin{remarktext}{Exercise 9.8.3. Topoi associated with local equivalence relations and foliations}
This exercise, in the French text, was explicitly marked as provisional.
\begin{enumerate}[label=\alph*)]
\item For every open \(U\subset X\), let \(\Quot(U)\) be the set of equivalence relations on \(U\). These form a presheaf, and its associated sheaf is denoted \(Q_X\). A section \(r\) of \(Q_X\) is a \emph{local equivalence relation} on \(X\). The order by fineness on equivalence relations makes \(Q_X\) a sheaf of ordered sets.
\item For a global equivalence relation \(R\), let \(\loc(R)\) be the associated section. For a local relation \(r\), define \(\glob(r)\) as the supremum of all global equivalence relations whose localization is no finer than \(r\). Describe \(\glob(r)\) using local representatives \(R_x\) and restrictions to smaller neighborhoods.
\item Define \(r\) to be precoherent, coherent, and globally coherent by the conditions \(\glob(r)\in E(r)\), stability under restriction, and \(r=\loc\glob(r)\). Define the corresponding notions for global equivalence relations. Show that globally coherent local relations correspond bijectively to coherent global relations.
\item Give local connectedness criteria ensuring coherence, and discuss examples with connected locally connected fibers which are not locally coherent.
\item For a continuous map \(f:X'\to X\), define the inverse image of a local equivalence relation. A map is a \emph{fiber map} for \(r\) if the inverse image of \(r\) is the coarse local equivalence relation. Show that \(X'\mapsto\operatorname{Hom}_{\mathrm{fib},r}(X',X)\) is contravariantly functorial.
\item If \(r\) is coherent, show that the preceding functor is represented by a space \(X^r\) over \(X\). The universal fiber map \(X^r\to X\) is bijective and locally a homeomorphism onto its image; connected components of \(X^r\) are open and correspond to fibers of \(R=\glob(r)\).
\item Define open and strictly open local equivalence relations. If \(r\) is open, give sufficient, and in the locally closed-fiber case necessary, conditions for strict openness.
\item For a sheaf \(F\) on \(X\), define \(Q(F/X)\), whose sections describe equivalence relations on \(F\) compatible with equivalence relations on the base. An \(r\)-structure on \(F\) is a section of \(Q(F/X)\) above \(r\). Define the category \(\Top(X/r)\) of \(r\)-sheaves and the faithful conservative forgetful functor \(\Top(X/r)\to\Top(X)\). Prove finite limits, finite colimits, and a small generating family in the appropriate cases.
\item If \(f:X\to Y\) is compatible with \(R=\glob(r)\), define
\[
f_r^*:\Top(Y)\longrightarrow \Top(X/r).
\]
If \(r\) is globally coherent and strictly open and \(f:X\to X/R\) is the quotient map, then \(f_r^*\) is an equivalence.
\item If \(r\) is strictly open, conclude that \(\Top(X/r)\) is a topos and that the forgetful functor is the inverse image of a morphism
\[
p:\Top(X)\longrightarrow \Top(X/r).
\]
\item Establish functoriality of \(\Top(X/r)\) in \((X,r)\). If \(X'\to X\) is \'etale and \(r'\) is the inverse image of \(r\), then \(\Top(X'/r')\to\Top(X/r)\) is a localization.
\item Show that \(\Top(X/r)\) is a rigid \'etendue. If \(X\) is sober, the set of isomorphism classes of points is homeomorphic to \(X/\glob(r)\), and there is at most one morphism between two points.
\item Study the canonical morphism \(\Top(X)\to\Top(X/r)\); over an object \(U\) for which the induced topos is an ordinary space, the corresponding map of spaces has fibers homeomorphic to components of \(X^r\).
\item For a differentiable manifold \(X\) with a foliation, define the associated strictly open local equivalence relation, and put a ring on \(T=\Top(X/r)\) making it a differentiable \'etendue. Identify locally free modules on \(T\) with locally free modules on \(X\) equipped with a connection along the foliation. Give real and complex analytic variants.
\end{enumerate}
\end{remarktext}
