% 0372: \begin{enonce*}{Définition 1.1}\etiquette[1.1]{V.def1.1}
% 0373: Soit $(E, A)$ un topos annelé \emph{(IV 11.1.1)}. %% étiquette
% 0374: Un $A$-Module à droite
% 0375: (\resp à gauche) $M$ est dit plat si le foncteur $M
% 0376: \otimes_A ⋅$ (\resp $⋅\otimes_A M$) de la catégorie des $A$-Modules
% 0377: à gauche (\resp à droite) dans la catégorie des faisceaux
% 0378: abéliens de $E$\sisi{,}{} est exact.
% 0379: \end{enonce*}

% 0381: \begin{enonce*}{Proposition 1.2}\etiquette[1.2]{V.prop1.2}
% 0382: Soit $M$ un $B$-$A$ bi-Module.
% 0383: \begin{enumerate}
% 0384: \renewcommand{\labelenumi}{\theenumi)}
% 0385: \item Les propriétés suivantes sont équivalentes:
% 0386: \renewcommand{\theenumii}{\roman {enumii}}
% 0387: \renewcommand{\labelenumii}{\theenumii)}
% 0388: \begin{enumerate}
% 0389: \item Le module $M$ est $A$-plat à gauche\sisi{}{.}
% 0390: \item Pour tout $B$-Module injectif $I$, le  $A$-Module à gauche
% 0391:   $\Hhom_{B}(M, I)$ est injectif.
% 0392: \end{enumerate}
% 0393: \item Un Module $M$, limite inductive pseudo-filtrante \emph{(I 2.7.1)} %% étiquette
% 0394: de Modules plats, est plat.
% 0395: 
% 0396: \item Enfin, si $M_∙ = \cdots \sisi{}{\to} M_{i + 1} \to M_i \sisi{}{\to} \cdots$ est un complexe
% 0397:   acyclique de Modules plats $(M_i = 0$ pour $i < i_0)$, alors
% 0398:   pour tout Module $F$, le complexe:
% 0399: $$
% 0400: M_∙ \otimes_A F = \cdots \sisi{}{\to} M_{i + 1} \otimes_A F \to M_i \otimes_A F \sisi{}{\to} \cdots
% 0401: $$
% 0402: est\pageoriginale acyclique.
% 0403: \end{enumerate}
% 0404: \end{enonce*}
% 0405: 
% 0406: \begin{enonce*}[remark]{Preuve}
% 0407: D'après (IV 12.12) %% étiquette
% 0408: on a un isomorphisme d'adjonction :
% 0409: $$
% 0410: \Hom_B (M \otimes_A \cdot , \cdot) \xrightarrow{\sim} \Hom_A (\cdot ,
% 0411: \Hhom_B(M , \cdot) ){\quoi}.
% 0412: $$
% 0413: Il suffit alors d'appliquer \hyperref[V.prop0.2]{0.2} pour obtenir
% 0414: l'équivalence (i) $\Leftrightarrow$ (ii). Cet isomorphisme
% 0415: d'adjonction montre par ailleurs que le produit tensoriel commute
% 0416: aux limites inductives. Le fait que les limites inductives
% 0417: pseudo-filtrantes soient exactes (\hyperref[V.prop0.1]{0.1}) entraîne la
% 0418: deuxième assertion. Pour montrer que le complexe $M_∙ \otimes_A F$
% 0419: est acyclique, il suffit de montrer que pour tout faisceau abélien
% 0420: injectif I le complexe $\Hom^{∙}_{\ZZ} (M_∙ \otimes_A F, I)$ est
% 0421: acyclique. Ce complexe est isomorphe, en vertu des formules
% 0422: d'adjonction, au complexe $\Hom^{∙}_A(F, \Hhom_{\ZZ}(M_∙ , I))$. Or
% 0423: d'après l'équivalence (i) $\Leftrightarrow$ (ii), le complexe de
% 0424: faisceaux
% 0425: $$
% 0426: \Hhom_{\ZZ}(M_∙ , I)
% 0427: $$
% 0428: est un complexe acyclique dont les objets sont injectifs, d'où la conclusion.
% 0429: \end{enonce*}

% 0432: Soient $(E, A)$ un topos annelé, $H$ un objet de $E$, $M$ un
% 0433: $A_{/H}$-Module plat. Alors $j_{H!}M$ est un $A$-Module plat. En
% 0434: particulier $A_H$ est plat à droite et à gauche.
% 0435: \end{enonce*}

% 0438: droite. Pour tout $A$-Module à gauche $P$, on a un isomorphisme
% 0439: canonique (IV 12) %% étiquette
% 0440: $$
% 0441: P \otimes_A j_{H!} M \simeq j_{H!} (P \otimes_{A_{/H}} M){\quoi}.
% 0442: $$
% 0443: Les foncteurs $j_{H!}$ et $j^{\ast}_H$ sont exacts (IV 11.3.1 et 11.12.2)
% 0444: et par hypothèse, le foncteur $-\otimes_{A_{/H}} M$ est exact. Par
% 0445: suite le foncteur $P \to P \otimes_A j_{H!}M$ est exact et $j_{H!}M$
% 0446: est plat.

% 0451: Soient $(E,A)$ un topos annelé, $i : F \to E$ un sous-topos fermé de
% 0452: $E$. Posons $i^{\ast} A = A_{/F}$. Pour tout $A_{/F}$-Module (à droite) $M$ et
% 0453: tout $A$-Module (à gauche) $P$, on\pageoriginale a un isomorphisme fonctoriel
% 0454: $$
% 0455: i_\ast (M \otimes_{A_{/F}} i^\ast P) \simeq (i_\ast M) \otimes_A P
% 0456: $$
% 0457: \end{enonce*}
% 0458: 
% 0459: Soient $U$ l'ouvert complémentaire de $F$ et $j : U \to E$ le
% 0460: morphisme canonique d'immersion. On a $j^{\ast} (i_\ast M \otimes_A P)
% 0461: \simeq 0 \otimes_{A_{/U}} j^\ast (P)$ (IV 12); par suite $i_{\ast} M
% 0462: \otimes_{A} P$ a son support dans $F$. Donc le morphisme d'adjonction
% 0463: $i_{\ast} M \otimes_{A} P \to i_\ast i^\ast (i_\ast M_A P)$ est un
% 0464: isomorphisme (IV 14). On a $i^{\ast} (i_\ast M \otimes_{A} P) \simeq
% 0465: i^\ast i_\ast M \otimes_{A_{/F}} i^\ast P$ (IV 12) et comme $i : F \to
% 0466: E$ est une immersion fermée, $i^{\ast} i_{\ast} M \simeq M$ (IV 14);
% 0467: d'où l'isomorphisme annoncé.

% 0593: j^{\ast}_V M$ est exact. Comme le foncteur prolongement par zéro
% 0594: $j_{V!}$ est exact et fidèle (IV 11.3.1), il suffit de montrer que

% 0597: \otimes_{A/V} j^{\ast}_V M) \simeq j_{V!} (P) \otimes_A M$ (IV

% 0606:   gauche plat et soit $x : P \to E$ un point de $E$. Montrons que

% 0626: \to P''_{x_i} \otimes_{A_{x_i}} M_{x_i} \to 0$ et comme $M_{x_i}$ est

% 0629: \begin{enonce*}{Proposition 1.7}\etiquette[1.7]{V.prop1.7}
% 0630: Soient $(E, A)$ un topos annelé, $u : E' \to E$ un morphisme de
% 0631: topos et posons $A' = u^{\ast} A$. Soit $M$ un $A'$-Module à droite

% 0633: \begin{enumerate}
% 0634: \renewcommand{\theenumi}{\roman{enumi}}

% 0638:   à gauche (\resp à droite) dans la catégorie des faisceaux
% 0639:   abéliens de $E'$ est exact.

% 0646: Montrons que i) $\Rightarrow$ ii). Nous ne ferons la démonstration
% 0647: que dans le cas où $E'$ possède une famille conservatrice de
% 0648: points $(x_i)_{i \in I}$. Le cas général est traité en
% 0649: (\hyperref[V.prop8.2.7]{8.2.7}). Dans ce cas particulier, qui couvre la plupart
% 0650: des applications, on est ramené \pageoriginale au cas où $E'$
% 0651: est le topos ponctuel et où, par suite, le morphisme $u$, qu'on
% 0652: notera $x$, est un point de $E$ (\hyperref[V.prop1.6]{1.6} et IV 11.3.1). Nous nous
% 0653: bornerons au cas où $M$ est un $A'$-Module à droite. Le cas
% 0654: où $M$ est un $A'$-Module à gauche s'y ramène en passant aux
% 0655: anneaux opposés. Soient $V$ un voisinage de $x$ et $V_x$ sa fibre en
% 0656: $x$. On a, avec les notations de \hyperref[V.subsec1.4]{1.4}, un diagramme
% 0657: essentiellement commutatif de morphismes de topos:
% 0658: \[
% 0659: \xymatrix@C=2.25cm@R=1.5cm{
% 0660: P \ar[r]^{j} \ar[rd]_{x_V}& P/V_{x}\ar[r]^{j_{V_x}} \ar[d]^{x/V}& P\ar[d]^{x}\\
% 0661:   &  E/V \ar[r]   & E\quad \sisi{,}{}
% 0662: }
% 0663: \]
% 0664: où $x /V$ est le morphisme déduit de $x$ par localisation sur
% 0665: $V$ (IV 5.10) et $j$ est le morphisme déduit du point marqué de
% 0666: $V_x$. Montrons que le foncteur $N' \mapsto M \otimes_{A'} x^{\ast}_V
% 0667: N'$ est exact. On a $x^\ast_V - j^\ast(x / V)^{\ast}$ et $\id = j^*
% 0668: j^*_{V_x}$ et par suite, on a un isomorphisme canonique $M
% 0669: \otimes_{A'} x^*_V N' \simeq j^*(j^*_{V_x} M \otimes_{A'/V_x} (x/
% 0670: V)^*N')$. Comme le foncteur $j^*$ est exact et comme le foncteur
% 0671: $j_{V_x !}$ est exact et fidèle (IV 11.3.1), il suffit de montrer
% 0672: que le foncteur $N'\mapsto j_{V_{x}!}$ $(j^{\ast}_{V_{x}}M\otimes
% 0673: (x/V)^{\ast}N')$ est exact, et par suite (IV 12.11), il suffit de
% 0674: montrer que le foncteur $N' \mapsto M \otimes_{A'} x^* j_{V!}N'$ est
% 0675: exact. Mais il résulte de la définition de l'image inverse d'un topos
% 0676: induit (\hyperrefext[IV][IV.eq5.10.2]{5.10.2}) que $j_{V_{x}!}(x/V)^{\ast}N'$
% 0677: est égal à $x^*j_{V!}N'$ et comme le foncteur $j_{V!}$ est exact
% 0678: (\hyperrefext[IV][IV.prop11.3.1]{11.3.1}), le foncteur $N' \mapsto M \otimes
% 0679: j_{V_{x}!}(x/V)^{\ast}N'$ est exact par hypothèse.
% 0680: Pour tout $A'$-module $Q$, on a un isomorphisme
% 0681: fonctoriel (\hyperref[V.prop1.5]{1.5}):
% 0682: $$
% 0683: Q \simeq \varinjlim_{\Vois(x)^{\circ}} x^*_V x_{V*} Q{\quoi}.
% 0684: $$
% 0685: D'après ce qui précède, le foncteur $Q \mapsto M \otimes_{A'} Q$ est
% 0686: limite inductive filtrante de foncteurs exacts à gauche. Il est donc
