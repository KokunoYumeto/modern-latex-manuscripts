\subsection*{6. Appendix: Study of $(b)$-sheaves on $P=\mathbf P^N_k$.}

\textbf{6.1.} Let $k$ be an algebraically closed field and let $P=\mathbf P^N_k$. Let $F$ be a coherent sheaf on $P$, with Hilbert polynomial
\[
  \chi(F(n))=\sum_{i=0}^{r}a_i\binom{n+i}{i},
\]
and let
\[
  (c)=(c_0,\ldots,c_r)
\]
be a sequence of integers with $(c)\geq(a)$. Put
\[
  a_{m,i}=\sum_{j=0}^{r-i}a_{j+i}\binom{m-1+j}{j}
  \quad\text{and}\quad
  c_{m,i}=\sum_{j=0}^{r-i}c_{j+i}\binom{m-1+j}{j}.
\]
Finally, for every integer $q\geq1$, denote by $N_q$ the largest coherent subsheaf of $F$ whose support has dimension at most $q-1$, and put $F_q=F/N_q$.

\textbf{Lemma 6.2.} \emph{Under the conditions of 6.1, consider an exact sequence
\[
  0\longrightarrow I\longrightarrow\cO_P^{\oplus M}
  \longrightarrow F(m)\longrightarrow0
\]
with $m\geq0$. Then $I$ is $p$-regular, where
\(p=P_r(c_{m,0},\ldots,c_{m,r})\), and $P_r$ is the $r$-th
$(b)$-polynomial of 1.9.}

Indeed, $\cO_P$ is plainly a $(0,\ldots,0,1)$-sheaf ($N+1$ terms), so
$I$ is a $(0,\ldots,0,M)$-sheaf by 1.6. By 2.10,
\(\chi(I(n))=M\binom{n+N}{N}
-\sum_{i=0}^{r}a_{m,i}\binom{n+i}{i}\). The assertion follows from
1.11 and 1.10.

\textbf{Proposition 6.3.} \emph{Under the conditions of 6.1, suppose that $F$ is a $(b)$-sheaf. Then, by means of universal polynomials in the $a_i$, the $b_i$, and $N$, one can calculate integers $m,M,m_1,M_1$ for which there is an exact sequence
\[
  \cO_P(-m_1)^{\oplus M_1}\longrightarrow
  \cO_P(-m)^{\oplus M}\longrightarrow F\longrightarrow0.
\]
Moreover, $N$ enters only into the calculation of $M_1$.}

% [SGA6 current-rescribe idx667 | printed 654 | source-PDF 657 | high-res page 668]
Indeed, take $m=P_{r-1}(c_0,\ldots,c_{r-1})$, with $c_i=b_i-a_i$,
and $M=\sum_i a_i\binom{m+i}{i}$, so that, by 1.11 and 1.3, there is
an exact sequence
\[
  0\longrightarrow I\longrightarrow\cO_P^{\oplus M}
  \longrightarrow F(m)\longrightarrow0.
\]
Likewise, taking account of 6.2, take $m_1=m+p$, where
$p=P_r(a_{m,0},\ldots,a_{m,r})$, and
$M_1=M\binom{p+N}{N}-\sum_i a_i\binom{m_1+i}{i}
\quad(=\chi(I(p)))$.

\textbf{Theorem 6.4.} \emph{Under the conditions of 6.1, suppose that
$F$ is a quotient of $\cO_P(-m)^{\oplus M}$, with $m\geq0$. Put
$b_i=P_{r-i}(c_{m,i},\ldots,c_{m,r})$ for $i=0,\ldots,r$, where $P_s$
is the $s$-th $(b)$-polynomial of 1.9; put $b=b_1+m-1$, and put
$B=\sum_{i=1}^{r}a_i\binom{b+i}{i}$. Then:}
\begin{enumerate}[label=(\roman*)]
\item \emph{$F$ is $b$-regular.}
\item \emph{$-B\leq a_0=h^0(F(b))-B$}
\item \emph{$F$ is a $(b_0,\ldots,b_r)$-sheaf.}
\end{enumerate}

Indeed, choose a section $\sigma$ of $\cO_P(1)$ for which one obtains the following commutative diagram with exact rows and columns:
\[
\begin{tikzcd}[column sep=large,row sep=large]
&0 \arrow[d]&0 \arrow[d]&0 \arrow[d]&\\
0 \arrow[r]&I(-1) \arrow[r] \arrow[d]&I \arrow[r] \arrow[d]&J \arrow[r] \arrow[d]&0\\
0 \arrow[r]&\cO_P(-1)^{\oplus M} \arrow[r] \arrow[d]&\cO_P^{\oplus M} \arrow[r] \arrow[d]&\cO_Y^{\oplus M} \arrow[r] \arrow[d]&0\\
0 \arrow[r]&F(m-1) \arrow[r] \arrow[d]&F(m) \arrow[r] \arrow[d]&G(m) \arrow[r] \arrow[d]&0\\
&0&0&0&
\end{tikzcd}
\]

% [SGA6 current-rescribe idx668 | printed 655 | source-PDF 658 | high-res page 669]
Since $\chi(G(n))=\sum_{i=0}^{r-1}a_{i+1}\binom{n+i}{i}$ by 1.7,
$J$ is $b_1$-regular by 6.2; hence, by 1.4(i), $H^q(I(p))=0$ for
$q\geq2$ and $p\geq b_1-q$. The exact sequence
$H^q\bigl(\cO_P^{\oplus M}(p)\bigr)\longrightarrow
H^q(F(m+p))\longrightarrow H^{q+1}(I(p))$ then shows that $F(m)$ is
$(b_1-1)$-regular; this proves (i).

Assertion (i) implies
$0\leq h^0(F(b))=\chi(F(b))
=\sum_{i=0}^{r}a_i\binom{b+i}{i}$, which proves (ii). By 2.10 and 1.9,
\[
  \chi(F(b))
  =\sum_{i=0}^{r}a_{m,i}\binom{b_1-1+i}{i}
  \leq b_1+\sum_{i=0}^{r}c_{m,i}\binom{b_1-1+i}{i}
  =b_0.
\]
Thus $h^0(F(-1))\leq h^0(F(b))\leq b_0$. By induction on $r$, one may suppose that $G$ is a $(b_1,\ldots,b_r)$-sheaf, so (iii) follows from 1.6.2.

\textbf{Lemma 6.5.} \emph{Under the conditions of 6.1, suppose that $F$
has no associated prime cycle of dimension $0$.}
\begin{enumerate}[label=(\roman*)]
\item \emph{If $h^0(F)\geq1$, then $h^0(F(-1))\leq h^0(F)-1$.}
\item \emph{$H^0(F(-n))=0$ for $n\geq h^0(F)$.}
\item \emph{Let $0\to F(-1)\to F\to G\to0$ be exact, and let
$n_0\geq0$ satisfy $H^0(G(-n_0))=0$. Then $H^0(F(-n_0))=0$ and
$h^0(F)\leq n_0\cdot h^0(G)$.}
\end{enumerate}

For (i), let $0\ne\sigma\in H^0(F)$. The section $\sigma$ defines a nonzero subsheaf $F'=\cO_X\cdot\sigma$ of $F$, so $s=\dim\Supp(F')\geq1$. Consider the commutative diagram with exact rows and columns
\[
\begin{tikzcd}[column sep=large,row sep=large]
&0 \arrow[d]&0 \arrow[d]&0 \arrow[d]&\\
0 \arrow[r]&F'(-1) \arrow[r] \arrow[d]&F' \arrow[r] \arrow[d]&G' \arrow[r] \arrow[d]&0\\
0 \arrow[r]&F(-1) \arrow[r]&F \arrow[r]&G \arrow[r]&0.
\end{tikzcd}
\]

% [SGA6 current-rescribe idx669 | printed 656 | source-PDF 659 | high-res page 670]
Since $s\geq1$, one has $G'\ne0$; hence $F'$ is not contained in $F(-1)$. In other words, the section $\sigma$ is not in the image of the injection $H^0(F(-1))\to H^0(F)$, which proves (i).

Assertion (ii) follows immediately from (i), bearing in mind that
$h^0(F)=0$ plainly implies $h^0(F(-1))=0$, since
$h^0(F(-1))\leq h^0(F)$. In (iii) one obtains
$0\leq h^0(F(n))-h^0(F(n-1))\leq h^0(G(n))$.
Thus, for $n\geq n_0$, $h^0(F(-n_0))=h^0(F(-n))$, since $h^0(G(-n_0))=0$; hence $h^0(F(-n_0))=0$ by (ii). Consequently, for $p\geq1$,
$h^0(F(p-n_0))
\leq h^0(G(1-n_0))+\cdots+h^0(G(p-n_0))
\leq p\cdot h^0(G(p-n_0))$,
which gives (iii).

\textbf{Proposition 6.6.} \emph{Under the conditions of 6.1, suppose
that $F$ is a $(b_0,\ldots,b_r)$-sheaf. Then $F_q$ is a
$\bigl((b_{q-1})^q,\ldots,(b_{q-1})^2,b_{q-1},b_q,\ldots,b_r\bigr)$-sheaf.}

Indeed, let $\sigma$ be a ``general'' section of $\cO_X(1)$. It defines a commutative diagram with exact rows and columns:
\[
\begin{tikzcd}[column sep=large,row sep=large]
&0 \arrow[d]&0 \arrow[d]&0 \arrow[d]&\\
0 \arrow[r]&N_q(-1) \arrow[r] \arrow[d]&N_q \arrow[r] \arrow[d]&G' \arrow[r] \arrow[d]&0\\
0 \arrow[r]&F(-1) \arrow[r] \arrow[d]&F \arrow[r] \arrow[d]&G \arrow[r] \arrow[d]&0\\
0 \arrow[r]&F_q(-1) \arrow[r] \arrow[d]&F_q \arrow[r] \arrow[d]&G'' \arrow[r] \arrow[d]&0\\
&0&0&0&
\end{tikzcd}
\]
where $G$ is a $(b_1,\ldots,b_r)$-sheaf by 1.6(i), and where
$G''=G_{q-1}$ (a depth question).

% [SGA6 current-rescribe idx670 | printed 657 | source-PDF 660 | high-res page 671]
Suppose first that $q=1$. Then $\dim\Supp(N_q)=0$, so
$H^1(N_q(-1))=0$ and $h^0(F_q(-1))\leq h^0(F(-1))\leq b_0$.
Moreover, $G'=0$, and hence $G\simeq G''$. Thus $F_1$ is indeed a $(b_0,\ldots,b_r)$-sheaf. In addition, 6.5(ii) shows that
$H^0(F_1(-b_0))=0$. If $q\geq2$, argue by induction on $q$. One may
suppose that $G_{q-1}$ is a
$\bigl((b_{q-1})^{q-1},\ldots,b_{q-1},\ldots,b_r\bigr)$-sheaf and that
$H^0(G_{q-1}(-b_{q-1}))=0$. Lemma 6.5 then gives
$H^0(F_q(-b_{q-1}))=0$ and $h^0(F_q(-1))\leq(b_{q-1})^q$,
which proves the assertion.

\textbf{Theorem 6.7.} \emph{There are two sequences
$\{A_i(X_0,\ldots,X_i;Y)\}$ and
$\{A_i^{(q)}(X_0,\ldots,X_q;Y)\}$ of polynomials having the following
properties. Assume the conditions of 6.1 and suppose that $F$ is a
quotient of $\cO_P(-m)^{\oplus M}$, with $m\geq0$.}
\begin{enumerate}[label=(\roman*)]
\item \emph{If $F$ is a $(b_0,\ldots,b_r)$-sheaf, then, for
$i=0,\ldots,r$,}
\[
  |a_i|\leq A_{r-i}(b_i,\ldots,b_r;m).
\]
\item \emph{If
$\chi(F_q(n))=\sum_{i=0}^{r}a_i^{(q)}\binom{n+i}{i}$ is the Hilbert
polynomial of $F_q$, then, for $i=0,\ldots,q-1$,}
\[
  |a_i^{(q)}|\leq A_{r-i}^{(r-q)}(c_{q-1},\ldots,c_r;m).
\]
\end{enumerate}
\emph{(Notice that $a_{q-1}^{(q)}\leq a_{q-1}$ and
$a_q^{(q)}=a_q,\ldots,a_r^{(q)}=a_r$, since
$\dim\Supp(N_q)\leq q-1$.)}

To prove (i), use induction on $r$ and apply 6.4 to an exact sequence
$0\longrightarrow F(-1)\longrightarrow F\longrightarrow G\longrightarrow0$
in which $G$ is a $(b_1,\ldots,b_r)$-sheaf. By induction on $r$, one may
suppose that $A_0,\ldots,A_{r-1}$ have been defined, taking 1.7 into
account. Then 6.4(ii), together with 1.8, displays the definition of
$A_r$. Finally, (ii) follows formally from 6.4(iii), 6.6, and assertion
(i), applied to $F_q$ in place of $F$.

\textbf{Corollary 6.8. (Grothendieck).} \emph{Let $X$ be projective over
a noetherian scheme $S$, let $\cO_X(1)$ be very ample for $X/S$, and let
$\mathcal J$ be a family of classes of coherent sheaves on the fibers of
$X/S$. Suppose that:}
\begin{enumerate}[label=(\alph*)]
\item \emph{There is a coherent sheaf $E$ on $X$ such that $\mathcal J$
is contained in the family of classes of quotients of sheaves of the
form $E_K$.}
\item[$(b_q)$] \emph{In the Hilbert polynomials $\chi(F_K(n))$, for
$F_K\in\mathcal J$, the coefficients in degrees at least $q-1$ remain
bounded above.}
\end{enumerate}
\emph{Then the $(F_K)_q$ (as $F_K$ ranges over $\mathcal J$) form a
bounded family; moreover, the coefficients of $\chi(F_K(n))$ in degrees
at least $q-2$ remain bounded below.}

Indeed, one may plainly suppose that $X=\mathbf P^N_S$ and
$E=\cO_X(-m)^{\oplus M}$. The first assertion follows formally from
6.7(ii) and 1.13. The second follows by induction, using an exact sequence
$0\longrightarrow F(-1)\longrightarrow F\longrightarrow G\longrightarrow0$
and 6.4(ii).

\textbf{Definition 6.9.} \emph{Let $k$ be a field. We call a
\emph{special positive $k$-cycle of dimension $r$} any projective
$k$-scheme $X$, equipped with a very ample sheaf $\cO_X(1)$, which is a
union of closed subschemes $X_j$ of dimension $r$, each $X_j$ being
obtained by base change $\Spec(k)\to\Spec(k_j)$ as the inverse image of
an integral $k_j$-scheme $X'_j$ (equipped with $\cO_{X'_j}(1)$). We call
the \emph{degree} of $X$ the coefficient $a_r$ in the Hilbert polynomial
$\chi(\cO_X(n))=\sum_i a_i\binom{n+i}{i}$.}

\textbf{Lemma 6.10.} \emph{Let $k$ be an algebraically closed field, and let $X$ be a special positive $k$-cycle of dimension $r$ and degree $d$. Then $\cO_X$ is a $(0,\ldots,0,d)$-sheaf.}

% [SGA6 current-rescribe idx672 | printed 659 | source-PDF 662 | high-res page 673]
Indeed, view $X$ as a subscheme of $\mathbf P^N_k$. After replacing $k$ by the algebraic closure of a purely transcendental extension, which is permissible by 1.6(i), one may suppose that $k$ contains $r(N+1)$ elements algebraically independent over all the $k_j$. The $r$ successive ``general'' hyperplane sections of $X$ defined by these elements are then always special positive $k$-cycles of degree $d$.

Arguing by induction on $r$, it remains only to observe that, when $r=0$,
one has $h^0(\cO_X(-1))=d$, while, when $r\geq1$, one has
$h^0(\cO_X(-1))=0$. For the latter point, the injection
$\cO_X\longrightarrow\coprod_j\cO_{X_j}$ allows one to suppose that $X$
itself comes from an integral scheme $X'$. But then
$h^0(\cO_X(-1))=h^0(\cO_{X'}(-1))=0$, because $H^0(\cO_{X'})$ is a
field.

\textbf{Corollary 6.11.} (i) \emph{Let $r,d$ be two integers, put
$c_i=A_{r-i}(0,\ldots,0,d;0)$ for $i=0,\ldots,r$ (6.7(i)), and put
$p=P_r(c_0,\ldots,c_r)$ (1.9). Let $k$ be an algebraically closed field,
and let $X$ be a special positive $k$-cycle of dimension at most $r$ and
degree at most $d$, with Hilbert polynomial
$\chi(\cO_X(n))=\sum_{i=0}^{r}a_i\binom{n+i}{i}$. Then, for
$i=0,\ldots,r$,}
\[
  |a_i|\leq c_i.
\]
\emph{Moreover, $X$ embeds as a subscheme of $\mathbf P^N_k$, with
$N=d(r+1)-1$, defined by (at most) $\binom{N+p}{N}$ equations of degree
$p$.}

(ii) (Chow) \emph{Let $S$ be a noetherian scheme. As $K$ ranges over
the algebraically closed extensions of the $k(s)$, for $s\in S$, the
special positive $K$-cycles of bounded dimension and degree form a
``bounded family'' (either with the $X$ regarded as subschemes of a fixed
$\mathbf P^N_S$, or in an evident abstract sense).}

% [SGA6 current-rescribe idx673 | printed 660 | source-PDF 663 | high-res page 674]
Indeed, in (i), the first assertion follows from 6.7(i); the second
follows from 6.10 and 1.8(i), and from 6.2, applied to
$0\longrightarrow I\longrightarrow\cO_P\longrightarrow\cO_X\longrightarrow0$,
$P=\mathbf P^N_k$, together with 1.3. Finally, (ii) follows from (i) and
1.13.

\textbf{Remark 6.12.} Let $k$ be an algebraically closed field and let
$X$ be a projective $k$-scheme, purely of dimension $r$, with no embedded
associated prime cycle. If arbitrary nilpotent elements are allowed in
$\cO_X$, the coefficients in the Hilbert polynomial
$\chi(\cO_X(n))=\sum_{i=0}^{r}a_i\binom{n+i}{i}$ can no longer be bounded
in terms of $a_r$ alone.

For example, let $Z$ be a smooth curve of degree $d$ in $\mathbf P^3_k$.
For every $n\gg0$, there is a smooth surface $Y$ of degree $n$ containing
$Z$. Then $Z$ is a Cartier divisor on $Y$; put
$\cO_{X_n}=\cO_Y/\cO_Y(-2Z)$. There is an exact sequence
% SOURCE-EMENDATION: the print defines X_n but then prints O_X(-Z) and O_X;
% O_{X_n} is retained here as the mathematically required reading.
$0\longrightarrow\cO_{X_n}(-Z)\longrightarrow
\cO_{X_n}\longrightarrow\cO_Z\longrightarrow0$ and the formula
$\langle Z^2\rangle_Y=2p_a(Z)-2-(n-4)d$. It follows that
$a_0=\chi(\cO_{X_n}(-1))\longrightarrow\infty$ as $n\longrightarrow
\infty$, although $a_1=2d$.

\textbf{Corollary 6.13.} \emph{Under the conditions of 6.1, consider an exact sequence
\[
  \cO_P(-m_1)^{\oplus M_1}\longrightarrow
  \cO_P(-m)^{\oplus M}\xrightarrow{\alpha}F\longrightarrow0
\]
with $m_1\geq m\geq0$. Then, for $i=0,\ldots,N$,
\[
  |a_i|\leq A_{N-i}(0,\ldots,0,M;m_1)
  +M\binom{m}{N-i}.
\]}

Indeed, let $I=\Ker(\alpha)$. Plainly, $I$ is a
$(0,\ldots,0,M)$-sheaf, and
$\chi(I(n))=\allowbreak\sum_{i=0}^{N}
\bigl[M\tbinom{-m-1+N-i}{N-i}-a_i\bigr]\allowbreak\tbinom{n+i}{i}$.
The assertion follows from 6.7(i) and the formula
$\tbinom{-m-1+Q}{Q}=(-1)^Q\tbinom{m}{Q}$.

\textbf{Corollary 6.14. (I. Hermann).} \emph{For every $N\geq0$, there
is a polynomial $R_N(m)$ such that, for every field $k_0$ and every ideal
$I$ of the polynomial ring $k_0[T_1,\ldots,T_N]$ generated by elements
of degree at most $m$, the radical $\sqrt I$ is generated by elements of
degree at most $R_N(m)$.}

Indeed, let $k$ be the algebraic closure of $k_0$. After introducing an
auxiliary indeterminate, let $Y$ (respectively $X$) be the closed
subscheme of $P=\mathbf P^N_k$ defined by the homogenized ideal of $I$
(respectively of $\sqrt I$). In the Hilbert polynomial
$\chi(\cO_Y(n))=\sum_{i=0}^{N}b_i\binom{n+i}{i}$, one has
$|b_i|\leq c_i$, where $c_i=A_{N-i}(0,\ldots,0,1;m)$, by 6.13.

Let $X=\bigcup X^q$ be the decomposition by dimension. Then
$\deg(X^q)\leq e_q$, where $e_q=P_{N-q}(c_q,\ldots,c_N)$.
Indeed, after cutting $X$ (respectively $Y$) by a ``general'' linear subspace of codimension $q$, one may suppose that $q=0$. Then
$\deg(X^q)=h^0(\cO_{X^q})$ and
$h^0(\cO_{X^q})\leq h^0(\cO_Y)$, and 6.4(iii) gives
$h^0(\cO_Y)\leq e_q$.

Consequently, $\cO_{X^q}$ is a $(0,\ldots,0,e_q)$-sheaf by 6.10. It
follows from the injection $\cO_X\longrightarrow\prod_q\cO_{X^q}$ that
$\cO_X$ is an $(e_0,\ldots,e_N)$-sheaf. Hence, in the Hilbert polynomial
% SOURCE-EMENDATION: the print has upper bound n; N is required by the
% ambient projective dimension and is retained deliberately.
$\chi(\cO_X(n))=\sum_{i=0}^{N}a_i\binom{n+i}{i}$, one has
$|a_i|\leq f_i$, where $f_i=A_{N-i}(e_i,\ldots,e_N;0)$, by 6.7.
Finally, the ideal $\sqrt I$ of $X$ is $R_N(m)$-regular, where
$R_N(m)=P_N(f_0,\ldots,f_N)$, and the assertion follows from 1.3.

\subsection*{7. Appendix: The Hodge index theorem}

Here we recover and complete some earlier results by a direct argument.

\textbf{Theorem 7.1.} \emph{Let $X$ be an irreducible surface, proper
over an algebraically closed field. Suppose that there is an invertible
sheaf $H$ on $X$ such that $\langle c_1(H)^2\rangle>0$ (for example,
$H$ ample). Then an invertible sheaf $L$ on $X$ satisfying
\[
  \langle c_1(L)\cdot c_1(H)\rangle=0,
  \qquad
  \langle c_1(L)^2\rangle\geq0,
\]
is numerically equivalent to $0$. Consequently, by 4.6, the intersection
form on $A^1_{\mathrm{num}}(X)_{\Q}
=\bigl[\Pic(X)/\Pic^\tau(X)\bigr]\otimes\Q$ is nondegenerate of type
$(1,\rho-1)$.}

\textbf{7.1.1.} First suppose that $X$ is reduced (hence integral), that
$H$ is ample, that $\langle c_1(L)\cdot c_1(H)\rangle=0$, and that
$\langle c_1(L)^2\rangle>0$. Arguing by contradiction, choose $m>0$
such that $H_1=L\otimes H^{\otimes m}$ is ample, and choose $n$ such
that, on putting $N=L^{\otimes n}\otimes H^{-1}$, one has
$\langle c_1(N)\cdot c_1(H_1)\rangle
=n\langle c_1(L)^2\rangle-m\langle c_1(H)^2\rangle>0$. One also has
$\langle c_1(N)\cdot c_1(H)\rangle=-\langle c_1(H)^2\rangle<0$ and
$\langle c_1(N)^2\rangle
=n^2\langle c_1(L)^2\rangle+\langle c_1(H)^2\rangle>0$, contradicting
the following lemma.

\textbf{Lemma 7.1.2.} \emph{Let $X$ be an integral surface, projective over an algebraically closed field. For an invertible sheaf $N$ on $X$ such that $\langle c_1(N)^2\rangle>0$, the following conditions are equivalent:
\begin{enumerate}[label=(\roman*)]
\item for every ample sheaf $H$, one has $\langle c_1(N)\cdot c_1(H)\rangle>0$;
\item[(i$'$)] there is an ample sheaf $H$ such that $\langle c_1(N)\cdot c_1(H)\rangle>0$;
\item there is an integer $n>0$ such that $H^0(X,N^{\otimes n})\ne0$.
\end{enumerate}}

% [SGA6 current-rescribe idx676 | printed 663 | source-PDF 666 | high-res page 677]
Indeed, (ii) implies that there is an effective divisor $D$ such that
$N^{\otimes n}\simeq\cO_X(D)$, and $\langle c_1(N)^2\rangle>0$ implies
$D>0$; hence (i).

To see that (i$'$) implies (ii), one may plainly suppose that
$H=\cO_X(1)$ is very ample. There is then a closed integral locally
principal curve $Y$ such that $H=\cO_X(Y)$. By the Riemann theorem,
(i$'$) implies $H^1\bigl(N_Y^{\otimes n}(p)\bigr)=0$ for $p\geq0$ and
$n\geq n_0$, where $n_0=2h^1(\cO_Y)-1$.
Since $H^2(N^{\otimes n}(p))=0$ for $p\gg0$, the exact sequence
\[
  H^1\bigl(N_Y^{\otimes n}(p)\bigr)\longrightarrow
  H^2\bigl(N^{\otimes n}(p-1)\bigr)\longrightarrow
  H^2\bigl(N^{\otimes n}(p)\bigr)\longrightarrow0
\]
gives $H^2(N^{\otimes n})=0$ for $n\geq n_0$. Thus, for $n\gg0$,
\[
  h^0(N^{\otimes n})
  =\chi(N^{\otimes n})+h^1(N^{\otimes n})
  \geq\chi(N^{\otimes n})
  =\frac12\langle c_1(N)^2\rangle n^2+\cdots>0,
\]
which completes the proof of the lemma.

\textbf{7.1.3.} Now suppose in 7.1 that there is an invertible sheaf $M$
on $X$ such that $\langle c_1(L)\cdot c_1(M)\rangle\ne0$, and argue by
contradiction. Replace $X$ by the underlying reduced scheme of a Chow
presentation of $X$, which is permissible by the projection formula X
4.4.4, and let $G$ be ample on $X$. Choose $m>0$ such that
$H_1=G^{\otimes m}\otimes M$ is ample and
$\langle c_1(L)\cdot c_1(H_1)\rangle\ne0$. Choose $p,q$, with $q\ne0$,
such that, on putting $L_1=L^{\otimes p}\otimes H^{\otimes q}$, one has
$\langle c_1(L_1)\cdot c_1(H_1)\rangle
=p\langle c_1(L)\cdot c_1(H_1)\rangle
+q\langle c_1(H)\cdot c_1(H_1)\rangle=0$. Then one also has
$\langle c_1(L_1)^2\rangle=p^2\langle c_1(L)^2\rangle
+q^2\langle c_1(H)^2\rangle>0$. This is the case 7.1.1, with $L=L_1$
and $H=H_1$.

\textbf{7.1.4.} Finally, consider the general case of 7.1. Still arguing by contradiction, suppose that $L$ is not numerically equivalent to $0$. Replace $X$ by the normalization of $X_{\red}$, so that $X$ has

% [SGA6 current-rescribe idx677 | printed 664 | source-PDF 667 | high-res page 678]
only finitely many singular points; this reduction is justified by the projection formula X 4.4.4 and by 4.5(ii). There is therefore a closed integral curve $Y$ on $X$ such that
$\langle c_1(L)\cdot Y\rangle\ne0$, and $Y$ is locally principal
outside a finite set $Z$. Let $f:X'\to X$ be the blow-up of the ideal
% SOURCE-EMENDATION: the print says I is an ideal of O_Y; O_X is required
% for the blow-up of X and is retained deliberately.
$I\subset\cO_X$ defining $Y$. The ideal $\mathcal M=I\cO_{X'}$ is
invertible and defines a subscheme of $X'$ which is the union of two
subschemes $Y'$ and $Y''$ such that $f(Y')=Y$ and $f(Y'')\subset Z$.
By the projection formula X 4.4.4,
$\langle c_1(f^*L)\cdot c_1(\mathcal M)\rangle
=-\langle c_1(f^*L)\cdot Y'\rangle
-\langle c_1(f^*L)\cdot Y''\rangle
=-\langle c_1(L)\cdot Y\rangle-0\ne0$. Thus one may replace $X,L,H$
by $X',f^*L,f^*H$. This reduces the problem to the preceding case and
completes the proof of 7.1.

\textbf{Remark 7.2.} It is not known whether an irreducible proper surface over an algebraically closed field which possesses an invertible sheaf $H$ satisfying $\langle c_1(H)^2\rangle>0$ is necessarily projective. Nor is it known whether, for every irreducible proper surface, the intersection form on $A^1_{\mathrm{num}}(X)_{\Q}$ is nondegenerate. It follows from 7.1 that, when it is degenerate, it is necessarily negative (non-definite).

an invertible sheaf on $X$ which is not numerically equivalent to $H$,
and put $d=\langle c_1(M)\cdot c_1(H)\rangle$ and
$h=\langle c_1(H)^2\rangle$. Then
$\langle c_1(M)^2\rangle<d^2/h$.}

\textbf{Corollary 7.4.} \emph{Let $X$ be an irreducible scheme of
dimension $r\geq2$, projective over an algebraically closed field, and
let $H$ be an ample sheaf on $X$. Then:}
\begin{enumerate}[label=(\roman*)]
\item \emph{An invertible sheaf $L$ on $X$ is numerically equivalent to
$0$ if and only if}
\[
  \langle c_1(L)\cdot c_1(H)^{r-1}\rangle=0,
  \qquad
  \langle c_1(L)^2\cdot c_1(H)^{r-2}\rangle\geq0;
\]
\item \emph{For every invertible sheaf $M$ on $X$ which is not
numerically equivalent to $H$,}
\[
  \langle c_1(M)^2\cdot c_1(H)^{r-2}\rangle
  <\frac{\langle c_1(M)\cdot c_1(H)^{r-1}\rangle^2}
  {\langle c_1(H)^r\rangle}.
\]
\end{enumerate}

Assertion (ii) follows immediately from (i), by the argument of 7.3. To
prove (i), let $Y$ be a closed integral curve in $X$; we wish to prove
that $\langle c_1(L)\cdot Y\rangle=0$. Embed $X$ in $\mathbf P^N_k$ by
means of $H^{\otimes n}$, with $n>0$. Let $I=\bigoplus I_\nu$ be the
homogeneous ideal of $Y$, and let $\nu_0$ be an integer such that
$I_{\nu_0}$ generates $I_\nu$ for $\nu\geq\nu_0$. Then the
hypersurfaces $Z$ of degree $\nu_0$ containing $Y$ have no common points
outside $Y$. By induction on $r=\dim X$ and Bertini's theorem, one finds
$r-2$ hypersurfaces $Z_1,\ldots,Z_{r-2}$ of degree $\nu_0$ containing
$Y$ such that $X'=Z_1\cap\cdots\cap Z_{r-2}\cap X$ is irreducible.
Since $\langle c_1(L)\cdot Y\rangle
=\langle c_1(L|_{X'})\cdot Y\rangle$, it is enough to prove that
$L|_{X'}$ is numerically equivalent to $0$ on $X'$. By the projection
formula, $\langle c_1(L|_{X'})\cdot c_1(H|_{X'})\rangle=0$ and
$\langle c_1(L|_{X'})^2\rangle\geq0$, so the assertion follows from 7.1.

\textbf{Corollary 7.5.} \emph{Let $k$ be an algebraically closed field,
let $Y$ be a projective $k$-scheme equipped with an ample sheaf $H$, and
let $X$ be the zero scheme of a section $\varphi$ of $H$. Suppose that,
for every irreducible component $Y^i$ of $Y$, the irreducible components
of $X\cap Y^i=V(\varphi|_{Y^i})$

\section*{Expos\'e XIV. Open problems in intersection theory}
\begin{center}
by A. Grothendieck
\end{center}

In the present expos\'e we review some of the problems connected with questions encountered in this seminar which deserve to be studied. We leave aside the problems connected with Serre's \emph{local intersection theory}~[14], which we have not had to use in the seminar, and refer in particular to \emph{loc. cit.} for the two fundamental conjectures which remain open in that direction (\emph{loc. cit.}, Chapter V, \S4). We also pass over the problems, very important for arithmetic applications, connected with the \emph{structure of groups of cycle classes}, especially on a smooth projective algebraic scheme. For discussion of these problems, we refer to S. Kleiman, ``Algebraic cycles and the Weil conjectures,'' in \emph{Dix expos\'es sur la cohomologie des sch\'emas} (Masson--North Holland Publishing Co.). In no. 4 of the present expos\'e we also give some supplements to the seminar, concerning in particular the \emph{Chow ring}, which had not found a place in the preceding expos\'es.

\section*{1. The operations $\Lambda^i$ in the derived category $D(X)$\footnote{Cf. the note on p. 23.}}

Let $X$ be a locally ringed topos, and let $K^\bullet(X)$ be the ring of classes of perfect complexes on $X$ (Expos\'e IV). One would like to define on $K^\bullet(X)$ a $\lambda$-ring structure (Expos\'e V), functorial in $X$, such that the natural homomorphism
\[
  K^\circ_{\mathrm{naive}}(X)\longrightarrow K^\bullet(X)
\]
(where the first member is the ring of classes of locally free modules on $X$, equipped with the $\lambda$-structure defined in Expos\'e V) is a $\lambda$-homomorphism. For this purpose, one would have to define functors, in general nonadditive, of ``exterior power''
\[
  \Lambda^i:D^-(X)\longrightarrow D^-(X),
\]
transforming pseudo-coherent (respectively perfect) complexes into pseudo-coherent (respectively perfect) complexes and having the formal properties with respect to exact triangles needed to define a $\lambda$-structure on $K^\bullet(X)$ by the formula
\begin{equation}\tag{1.1}
  \lambda^i\bigl(\cl(L^\bullet)\bigr)
  =\cl\bigl(\Lambda^iL^\bullet\bigr).
\end{equation}

% [SGA6 current-rescribe idx681 | printed 668 | source-PDF 671 | high-res page 682]
When $X$ is of characteristic zero, that is, when its structure sheaf is a sheaf of $\Q$-algebras, the problem is solved simply by putting
\begin{equation}\tag{1.2}
  \Lambda^iL^\bullet
  =\left({}^{\otimes i}L^\bullet\right)^{\mathrm{alt}},
\end{equation}
where the $i$-th tensor power, in the sense of derived categories, of
$L^\bullet$ is regarded as an object of $D^-(\mathcal O_i)$, with
$\mathcal O_i=\mathcal O_X[\mathfrak S_i]$ the algebra of the symmetric
group $\mathfrak S_i$ over $\mathcal O_X$, and where the superscript
\(\mathrm{alt}\) denotes passage to the ``alternating part under the
action of $\mathfrak S_i$,'' an operation which makes sense because of
the characteristic-zero hypothesis. A formula of this type was already
used in [8] (cited below as [RRR]) in the first proof of the
Riemann--Roch theorem.

Without a characteristic hypothesis, when $L^\bullet$ is zero in degrees $>0$, Dold--Puppe~[6] gives a canonical homotopy from $L^\bullet$ to the complex associated with a semisimplicial module $L^{\prime\bullet}$ canonically attached to $L^\bullet$. For $L^\bullet$ flat, say, a fairly natural candidate for $\Lambda^iL^\bullet$ would be the complex associated with the semisimplicial module $\Lambda^iL^{\prime\bullet}$, whose degree-$n$ component is $\Lambda^iL'_n$. But to extend the definition to complexes of $D^-(X)$ which are no longer assumed to be zero in degrees $>0$, one would have to study the behavior of the preceding operation with respect to translation $L^\bullet\mapsto L^\bullet[1]$, or \emph{suspension} in Dold--Puppe terminology~[6]. Unfortunately, there does not seem to be an isomorphism $\Lambda^i\bigl(L^\bullet[1]\bigr)\simeq\Lambda^i(L^\bullet)[i]$ which would make the desired extension trivial.

\section*{2. The Riemann--Roch formula without projectivity hypotheses}

\textbf{2.1.} Once one has a natural $\lambda$-structure on the $K^\bullet(X)$, it is possible to formulate the Riemann--Roch theorem, in the form contemplated in Expos\'e VIII, for any proper complete-intersection morphism $f:X\to Y$, and even, more generally, for a relative scheme $X$ over a locally ringed topos $Y$, with $X$ still subject to the corresponding relative conditions. The Todd class of $f$ is defined by means of the class in $K^\bullet(X)$ of the relative cotangent complex $T_f$ constructed in Expos\'e 0, 4.4; that complex is perfect here by the local hypothesis on $f$. It seems clear that a proof of the Riemann--Roch formula in this general case will require essentially new ideas, and it is possible that these same ideas could provide the key

% [SGA6 current-rescribe idx682 | printed 669 | source-PDF 672 | high-res page 683]
to a proof of the analogous formula in the complex analytic (or rigid-analytic!) case. Recall that, in the complex analytic case, only the case of a locally free sheaf $F$ on a compact nonsingular variety $X$, over a one-point $Y$, is currently known, by the Atiyah--Singer index formula; the case in which $F$ is only assumed coherent still seems to escape presently known techniques.

\textbf{2.2.} As an essential ingredient of the Riemann--Roch formula,
recall also the \emph{finiteness problem} raised in Expos\'e III: if
$f:X\to Y$ is a proper morphism and $L^\bullet$ is a complex on $X$
which is pseudo-coherent relative to $Y$, is $Rf_*(L^\bullet)$
pseudo-coherent? A general proof of this conjecture will probably use
algebraic ideas which will yield a proof of Grauert's ``finiteness
theorem''~[7] under appreciably more general conditions than those of
\emph{loc. cit.}, working from the outset over topologically ringed
spaces (or topoi) in $\mathbb C$-algebras $Y$ more general than analytic
spaces. This would include the cases in which $Y$ is a locally compact
space with the sheaf of continuous complex-valued functions, a
$C^m$-manifold with the sheaf of complex-valued $C^m$-functions, a real
analytic space, and so forth.

\textbf{2.3.} Finally, in these questions the properness hypothesis on $f$ should be replaced by the hypothesis that the supports of the cohomology sheaves $H^i(L^\bullet)$ of the complexes under consideration on $X$ are proper over $Y$.

\section*{3. A Riemann--Roch formula ``without denominators'' for an immersion}

\textbf{3.1.} Let $f:Y\to X$ be a regular closed immersion (Expos\'e VII), with conormal sheaf $\underline N$. For simplicity, suppose that $X$ admits an ample invertible sheaf. The Riemann--Roch formula for $y\in K^\bullet(Y)$,
\[
  \ch_X(f_*(y))
  =f_*\bigl(\ch_Y(y)\Todd(-\check{\underline N})\bigr),
\]
together with the general formula (Expos\'e VII)
\[
  f_*(y')f_*(y'')
  =f_*\bigl(y'y''c_d(\underline N)\bigr),
  \qquad y',y''\in\Gr^\bullet(Y)_{\Q},
\]
allows one, assuming $f$ has codimension $d$, to express the Chern classes of $f_*(y)$ in $\Gr^\bullet(X)_{\Q}$ in the form

% [SGA6 current-rescribe idx683 | printed 670 | source-PDF 673 | high-res page 684]
\begin{equation}\tag{3.1}
  c_i(f_*(y))
  =f_*\bigl(P_i((c_\alpha(y))_\alpha,(c_\beta(\underline N))_\beta)\bigr),
\end{equation}
where $P_i$ is a universal polynomial in the $c_\alpha(y)$ and the $c_\beta(\underline N)$, a polynomial which a priori has rational coefficients. In fact, one finds in [RRR] that these polynomials have \emph{integer coefficients}. This raises the question of the validity of a Riemann--Roch formula ``without denominators.''

In fact, such a formula for $c_i(f_*(y))$ cannot in general be sought in $\Gr^\bullet(X)$ itself, since the homomorphism $f_*:K^\bullet(Y)\longrightarrow K^\bullet(X)$ is not itself compatible with the $\lambda$-filtrations, modulo a shift by $d$ (4.6), but is compatible only up to torsion. Thus it is not possible to define a homomorphism $f_*:\Gr^\bullet(Y)\longrightarrow\Gr^\bullet(X)$ inducing the natural homomorphism $f_*:\Gr^\bullet(Y)_{\Q}\longrightarrow\Gr^\bullet(X)_{\Q}$ and making it possible to interpret (3.1) as a formula in $\Gr^\bullet(X)$ rather than only in $\Gr^\bullet(X)_{\Q}$.

When $X$ is regular, however, so that $K^\bullet(X)\simeq K_\bullet(X)$ (IV 2.5), one can equip $K^\bullet(X)$ with the \emph{topological filtration} by codimension of supports, inherited from the corresponding filtration of $K_\bullet(X)$. It is plausible that this filtration is compatible with the ring structure. This is proved at least in [RRR] when $X$ is smooth over a field $k$, using the theory of the Chow ring [4]. When this is so, the associated graded
\[
  \Gtop^\bullet(X)
\]
is a \emph{graded ring}; and, if $Y$ is also regular, one has precisely what is needed to obtain a natural homomorphism of degree $d$ of graded groups
\[
  f_*=f_*^{\mathrm{top}}:\Gtop^\bullet(Y)
  \longrightarrow\Gtop^\bullet(X).
\]
Since the $\lambda$-filtration of $K^\bullet(X)$ is finer than the topological filtration by codimension just considered (X 1.3.3), one obtains a canonical homomorphism of graded groups (and even of graded rings),
\[
  \Gr^\bullet(X)\longrightarrow\Gtop^\bullet(X),
\]
which is, moreover, an isomorphism modulo torsion (Expos\'e VIII). This makes it possible, as in [RRR], to consider a theory of Chern classes on $X$ with values in $\Gtop^\bullet(X)$, and likewise on $Y$, and thus to interpret (3.1) as a formula in $\Gtop^\bullet(X)$. In this form, the formula is proved in [RRR] when $X$ is of \emph{characteristic zero}, at least if $X$ is smooth over a field (an assumption used only to ensure that the topological filtration of $K^\bullet(X)$ is compatible with the ring structure). The characteristic-zero hypothesis is used essential%
% [SGA6 current-rescribe idx684 | printed 671 | source-PDF 674 | high-res page 685]
ly to have formulas of the type (1.1), (1.2), and essentially the same proof should apply without a characteristic hypothesis once the question raised in no. 1 has been answered affirmatively.

\textbf{3.2.} When $X$ is smooth over a field, one also has the Chow ring
\[
  A(X),
\]
and a theory of Chern classes on $X$ with values in $A(X)$~[9], finer than the theory with values in $\Gtop^\bullet(X)$ just introduced. The latter Chern classes are the images of the Chow-style Chern classes under the canonical surjective homomorphism (cf. 4.1)
\[
  \varphi:A(X)\longrightarrow\Gtop^\bullet(X).
\]
This homomorphism is an isomorphism modulo torsion, but not an isomorphism in general (4.7). That said, if $X$ and $Y$ are smooth, the inclusion $f:Y\to X$ still defines a homomorphism of degree $d$ of graded groups
\[
  f_*:A(Y)\longrightarrow A(X),
\]
which makes it possible to interpret (3.1) as a formula in $A^i(X)$.
This formula reduces to $0=0$ for $i<d$ and is therefore of interest
only for $i>d$. Even for $i=d=2$, this formula has not been proved at
present (cf. 4.3), nor has the less precise formula deduced from it by
replacing rational equivalence of algebraic cycles by algebraic
equivalence. Unlike the analogous formula in $\Gtop^\bullet(X)$, there
does not seem to be any convincing reason to suppose that the formula
without denominators should be true in $A(X)$. A good way to test it,
given our almost total ignorance of the nature of the groups
$A^i(X)$\footnote{The formula, as well as the variant mentioned in 3.3,

to be to test the formulas that can be deduced from it in groups
$A^1(Z)=\Pic(Z)$, for example by applying to both sides of (3.1)
$g_*:A(X)\longrightarrow A(Z)$,
where $g:X\to Z$ is a proper morphism to a smooth quasi-projective $Z$ of dimension $\dim X-i+1$.

\]
(Cf. no. 6.) This is the case, at least, if $Y$ and $X$ are both smooth
over a scheme $S$ (by the relative cohomological purity theorem

\section*{4. Relations between $K^\bullet(X)$ and the Chow ring $A(X)$}

\textbf{4.1.} Suppose for simplicity that $X$ is a smooth quasi-projective
scheme over a field $k$ (see no.~8 for a more general case), so that one has

% Current-rescribe idx686 | printed volume page 673 | source-PDF page 676
the Chow ring $A(X)$ of cycle classes modulo rational equivalence [4]. As
variants of this ring, defined under appreciably more general conditions,
one has the rings $\operatorname{Gr}^\bullet(X)$ (the graded ring associated
with the $\lambda$-filtration of $K^\bullet(X)$) and
$\Gtop^\bullet(X)$ (the graded ring associated with the filtration of
$K^\bullet(X)\simeq K_\bullet(X)$ by codimension of supports). There are
theories of Chern classes with values in each of these three rings, and also
an $\ell$-adic cohomological theory (for $\ell$ prime to the characteristic
of $k$),
\[
 c_i(E)\in H^{2i}(X,\Zell(i)),
\]
where $E$ is a vector bundle on $X$, or more generally any element of
$K^\bullet(X)$. These theories are related by the canonical homomorphisms
compatible with the multiplicative structures:
\begin{equation}\tag{4.1}
\begin{tikzcd}[column sep=large,row sep=large]
 H^{2\bullet}(X,\Zell(\bullet)) & {} \\
 A(X) \arrow[u,"\gamma"] \arrow[r,"\varphi"] & \Gtop^\bullet(X) \\
 {} & \operatorname{Gr}^\bullet(X) \arrow[u,"j"']
\end{tikzcd}
\end{equation}
where
\[
 H^{2\bullet}(X,\Zell(\bullet))
   =\bigoplus_i H^{2i}(X,\Zell(i)).
\]
The homomorphism $j$ comes from the fact that the $\lambda$-filtration of
$K^\bullet(X)$ is finer than the topological filtration by codimension of
support (Exposé X), and it is an isomorphism modulo torsion (Exposé VII
4.11). The homomorphism $\varphi$, defined for example in RRR, associates
to the class of the codimension-$i$ cycle carried by a closed subscheme
$Y\subset X$ the element $\cl^i(\mathcal O_Y)$ of
\[
 \Fil^i_{\mathrm{top}}K_\bullet(X)
   /\Fil^{i+1}_{\mathrm{top}}K_\bullet(X).
\]
It is surjective, and we shall see again in 4.2 that it is an isomorphism
modulo torsion. Hence, modulo torsion, the three rings $A(X)$,
$\Gtop^\bullet(X)$, and $\operatorname{Gr}^\bullet(X)$ are canonically
isomorphic; consequently the Riemann--Roch formulas expressed using any one
of these rings after tensoring with $\Q$ are equivalent. The graded ring
$\operatorname{Gr}^\bullet(X)$, used in this seminar chiefly as a substitute
for the Chow ring, has the considerable advantage of being defined for
every locally ringed topos, and in particular for every scheme

% Current-rescribe idx687 | printed volume page 674 | source-PDF page 677
(regular or not). Its disadvantage with respect to $A(X)$ and
$\Gtop^\bullet(X)$ is that it is not covariant as it stands for proper
morphisms, even between smooth quasi-projective schemes over a field; it
acquires that covariance only after tensoring with $\Q$.

Another advantage of $A(X)$ over its competitors $\Gtop^\bullet(X)$ and
$\operatorname{Gr}^\bullet(X)$ is the homomorphism
$A(X)\to H^{2\bullet}(X,\Zell(\bullet))$, which interprets cohomological
Chern classes as the images of Chern classes with values in $A(X)$. In
general this homomorphism does not factor through $\Gtop^\bullet(X)$ (4.7),
and it is at least doubtful that there is in general a homomorphism
$\operatorname{Gr}^\bullet(X)\to H^{2\bullet}(X,\Zell(\bullet))$ carrying
Chern classes to Chern classes (see no.~5). These difficulties disappear
after tensoring with $\Q$, and one obtains a homomorphism
\begin{equation}\tag{4.2}
 \Gtop^\bullet(X)\longrightarrow H^{2\bullet}(X,\Qell(\bullet))
  =H^{2\bullet}(X,\Zell(\bullet))\otimes_{\Zell}\Qell
\end{equation}
that carries Chern classes to Chern classes.

\textbf{4.2.} To show that the surjective homomorphism $\varphi$ in (4.1)
is an isomorphism modulo torsion, one uses the Chern homomorphism
\[
 K_\bullet(X)\longrightarrow \Ch(A(X)),
\]
with values in the Chern ring of $A(X)$. It is compatible with the
topological filtration of $K^\bullet(X)$ and the natural filtration of
$\Ch(A(X))$, and therefore induces a homomorphism of graded groups
\begin{equation}\tag{4.3}
 \psi:\Gtop^\bullet(X)\longrightarrow A(X).
\end{equation}
One proves the formulas
\begin{equation}\tag{4.4}
\left\{
\begin{aligned}
 \psi\varphi(x)&=(-1)^{i-1}(i-1)!\,x
   &&\bmod\text{ torsion},\quad x\in A^i(X),\\
 \varphi\psi(x)&=(-1)^{i-1}(i-1)!\,x
   &&\bmod\text{ torsion},\quad x\in \Gtop^i(X).
\end{aligned}
\right.
\end{equation}
The first implies the second because $\varphi$ is surjective. These formulas
show that $\varphi$ and $\psi$ are isomorphisms modulo torsion and, up to
the factor $(-1)^{i-1}(i-1)!$ in degree $i$, are inverse to one another.

To prove the first formula in (4.4), it is enough to suppose that $x$ is
represented by an irreducible closed subscheme $Y\subset X$ of codimension
$i$. The formula is then equivalent to

% Current-rescribe idx688 | printed volume page 675 | source-PDF page 678
\begin{equation}\tag{4.5}
 c_i(\mathcal O_Y)=(-1)^{i-1}(i-1)!\,\cl^i(Y)
   \quad\bmod\text{ torsion}.
\end{equation}
On the left, $\mathcal O_Y$ is regarded as a coherent sheaf on $X$ and
hence defines an element of $K^\bullet(X)$; on the right,
$\cl^i(Y)$ denotes the class of $Y$ in $A^i(X)$. Since $A^i(X)$ is
unchanged upon deleting a closed subset of codimension $>i$, one reduces
to the case where $Y$ is regular, and therefore smooth over $k$ if $k$ is
perfect. In that case (4.5) is precisely the Riemann--Roch formula for the
inclusion $Y\to X$ and the element $1\in K^\bullet(Y)$, written using
Chern classes with values in the Chow ring $A(X)$. Under the present
hypotheses this formula is proved in [2] by a method very close to the one
used in this seminar for the analogous formula with values in
$\Gtop^\bullet(X)$. Thus, at least when $k$ is perfect, (4.4) follows, and
so does the fact that $\varphi$ is an isomorphism modulo torsion. It follows
in turn, a posteriori, that Riemann--Roch ``with denominators'' for a proper
morphism of smooth algebraic schemes over $k$ is essentially the same
whether it is written using $A(X)$ as in [2] or using $\Gtop^\bullet(X)$ as
in this seminar. The case of an arbitrary $k$ reduces to the algebraically
closed case: for a finite extension $k'/k$ of degree $n$, the kernel of
\[
 A(X)\xrightarrow{f_X^*}A(X\otimes_k k')
\]
is annihilated by $n$, as follows from the trace homomorphism
\[
 f_{X*}:A(X\otimes_k k')\longrightarrow A(X),
 \qquad f_{X*}f_X^*=n\,\id.
\]
\footnote{Thus the formula is established, through P.~Jouanolou, for $X$
smooth and quasi-projective over a field (see the note on p.~5).}

\textbf{4.3.} It seems quite plausible that the formulas (4.4) hold as
stated, not merely modulo torsion; equivalently, that (4.5) holds without
passing modulo torsion. This is a very special case of the denominator-free
Riemann--Roch problem (3.1) in $A(X)$ for an immersion $Y\to X$, namely
the case $i=d$. Its validity may be regarded as fairly plausible. Indeed,
as noted in no.~3, in characteristic zero the denominator-free formula is
valid at least in $\Gtop^\bullet(X)$; hence the two sides of (4.5) have the
same image there, or equivalently the second formula in (4.4) holds. This
result will extend automatically to arbitrary characteristic once

% Current-rescribe idx689 | printed volume page 676 | source-PDF page 679
the question raised in no.~1 has been satisfactorily resolved. As support
for (4.5) in $A(X)$ itself, note that the restriction of the left-hand side
to $X-Y$ is plainly zero. Since $Y$ is irreducible, that side must be an
integral multiple of $\cl^i(Y)$:
\[
 c_i(Y)=n(Y)\cl^i(Y).
\]
This already proves exact formula (4.5) whenever $\cl^i(Y)$ is not torsion
in $A^i(X)$. That is certainly so when $X$ is projective over $k$: if
$\xi\in A^1(X)$ is a polarization, then $\deg(Y\cdot\xi^i)>0$. Thus the
formula holds in that case. It also holds whenever $X$ is an open subscheme
of a smooth projective scheme over $k$, as happens when a suitable
resolution-of-singularities theorem is available; in particular it holds
in characteristic zero by Hironaka [11].

The validity of exact formula (4.5), and hence of (4.4), would imply that
the kernel of
\[
 \varphi:A^i(X)\longrightarrow\Gtop^i(X)
\]
is annihilated by $(i-1)!$, and in particular that
$\varphi:A^2(X)\to\Gtop^2(X)$ is an isomorphism. That $\varphi^1$ is an
isomorphism follows immediately from X 5.3.2. Unless there is an error,
the assertion for $\varphi^2$ is not presently proved, even in the special
case $\dim X=3$. The editor's earlier assertion of (4.4) and (4.5) in
[9, no.~4, 3o] was therefore premature.

\textbf{4.4.} At present the chief advantage of $\Gtop^\bullet(X)$ (or of
$\operatorname{Gr}^\bullet(X)$) over the Chow ring $A(X)$ is that it is
better suited to explicit calculations of intersection formulas in cases
of excess intersection, that is, intersection of too large a dimension.
The definition of an intersection class by means of the Chow lemma which
moves a cycle sufficiently within its class is poorly adapted to
calculation. A striking example is the following fundamental formula
[Exposé VII 2.7], for a closed immersion $f:Y\to X$ of smooth
quasi-projective schemes over $k$:
\begin{equation}\tag{4.6}
 f^*f_*(y)=y\,c_d(N),
\end{equation}
valid in $\Gtop^\bullet(Y)$ for $y\in\Gtop^\bullet(Y)$, where $N$ is the
conormal sheaf

% Current-rescribe idx690 | printed volume page 677 | source-PDF page 680
of $Y$ in $X$, assumed everywhere of rank $d$. In the cited passage this
formula follows from a very simple calculation of Tor sheaves. By 4.2 it
implies the same formula in $A(Y)$, but only modulo torsion. It is quite
remarkable that (4.6) in $A(Y)$ itself has not been proved at present,
\footnote{See the note on p.~23.} nor even the special case $y=1$, which
gives the self-intersection of $Y$:
\begin{equation}\tag{4.7}
 f^*(\cl(Y))=c_d(N),
\end{equation}
even when $k$ is the field of complex numbers.

An analogous problem appears when one tries to calculate the Chow ring of
the blow-up $X'$ of $X$ along $Y$, following the method used to calculate
$\operatorname{Gr}^\bullet(X')_\Q$ [Exposé VII 4.8], which also works
without tensoring with $\Q$ for $\Gtop^\bullet(X')$. With the notation of
the cited passage, the key formula is
\begin{equation}\tag{4.8}
 f^*(i_*(y))=j_*\bigl(g^*(y)c_{d-1}(\check P)\bigr).
\end{equation}
It is presently established in $\Gtop^\bullet(X')$ for
$y\in\Gtop^\bullet(Y)$ by a Tor calculation (Exposé VII 3.4); its validity
in $A(X')$ for $y\in A(Y)$ remains to be proved. Even the $\ell$-adic
analogues of (4.6), (4.7), and (4.8) are not presently proved. When $y$ is
the cohomology class of an algebraic cycle, (4.6) and (4.8) are at least
true modulo torsion.

\textbf{4.5.} In this section and the following ones we give examples where,
in diagram (4.1), respectively $j$ and $\varphi$ are not isomorphisms. For
a fixed $X$, it is equivalent to say that
\[
 j:\operatorname{Gr}^\bullet(X)\longrightarrow\Gtop^\bullet(X)
\]
is an isomorphism, a monomorphism, or an epimorphism. Its image is the
subring of $\Gtop^\bullet(X)$ generated by the Chern classes of locally free
sheaves on $X$. If $K^\bullet(X)$ is generated as a $\Z$-algebra by the
classes of invertible sheaves, that subring is also the subring generated
by $\Gtop^1(X)$, the degree-one component of $\Gtop^\bullet(X)$.

% Current-rescribe idx691 | printed volume page 678 | source-PDF page 681
Now take for $X$ a flag variety
\[
 X=G/B,
\]
where $G$ is a simply connected semisimple algebraic group over an
algebraically closed field $k$, and $B$ is a Borel subgroup [5]. If
$k=\C$, the Bruhat cell decomposition of $X$ [5, Exposé 13] implies that
the canonical homomorphism
\[
 (*)\qquad K^\bullet(X)\longrightarrow K(X(\C))
\]
is an isomorphism, as is the canonical homomorphism
\[
 (**)\qquad A(X)\longrightarrow H^\bullet(X(\C),\Z),
\]
where $X(\C)$ is the compact space of complex points of $X$ with its usual
topology. L.~Hodgkin [12] proved that $K(X(\C))$ is generated by the
classes of invertible sheaves on $X$. The same is therefore true of
$K^\bullet(X)$; a specialization argument using semisimple groups split
over $\Z$ (SGA 3) shows that the result remains valid over every
algebraically closed field $k$. The cell decomposition (or the bijectivity
of $(**)$ for $k=\C$) also shows that $A(X)$ is torsion-free, and hence by
4.2
\[
 \varphi:A(X)\longrightarrow\Gtop^\bullet(X)
\]
is an isomorphism. It follows that $j$ is an isomorphism if and only if
$A(X)$ is generated by its degree-one elements; over $\C$, equivalently,
the integral cohomology of $X(\C)$ is generated by its degree-two elements,
or the classifying space $B_{G(\C)}$ has no torsion. By [15, p.~217,
Theorem B, $(1)\Leftrightarrow(2)$, and 2.5], this is so exactly when $G$ is
``without torsion'', equivalently when $G$ is a product of groups
$\operatorname{SL}(n)$ and $\operatorname{Sp}(m)$. Thus one obtains an
example where $j$ is not an isomorphism by taking $G=\operatorname{SO}(n)$
with $n\geq7$, or a simple exceptional group.

\textbf{4.6.} In this example there can be no doubt that the ``good'' ring
is the torsion-free ring
\[
 A(X)\simeq\Gtop^\bullet(X),
\]
with basis the cells of the Bruhat decomposition, and, for $k=\C$, also
isomorphic to the integral cohomology of $X(\C)$. By contrast,
$\operatorname{Gr}^\bullet(X)$ appears as a ring with torsion of a rather
pathological kind. This confirms the view that
$\operatorname{Gr}^\bullet(X)$ is hardly interesting

except modulo torsion, that is, after tensoring with $\Q$. The same example
shows that $\operatorname{Gr}^\bullet(X)$ is covariant for proper morphisms
only after tensoring with $\Q$, even if one restricts to closed immersions
of smooth quasi-projective schemes. Indeed, if $j$ is not an isomorphism,
there is an integral closed subscheme $Y\subset X$ such that
\[
 \cl^i(Y)\in K^\bullet(X)
\]
does not lie in $\Fil^i(X)$, where $i=\operatorname{codim}(Y,X)$. We may
choose $Y$ of minimal dimension, so that
\[
 \Fil^{i+1}(X)=\Fil_{\mathrm{top}}^{i+1}(X).
\]
If $Z$ is the singular locus of $Y$, and $X'=X-Z$, $Y'=Y-Z$, it follows
that
\[
 \cl^i(Y')\in K^\bullet(X')
\]
does not lie in $\Fil^i(X')$. Here one uses the surjectivity of
$K^\bullet(X)\to K^\bullet(X')$, its compatibility with the augmentations,
and
\[
 \operatorname{Im}\bigl(K(Z)\longrightarrow K(X)\bigr)
 \subset\Fil^{i+1}(X)=\Fil_{\mathrm{top}}^{i+1}(X).
\]
Consider the inclusion
\[
 f:Y'\longrightarrow X'.
\]
It is a closed immersion of codimension $i$ between smooth
quasi-projective schemes over $k$, and
\[
 f_*:K^\bullet(Y')\longrightarrow K^\bullet(X')
\]
is not compatible with the $\lambda$-filtrations, even after shifting by
$i$, since $f_*(1)\notin\Fil^i(X')$.

\textbf{4.7.} Retaining the notation of 4.5, Exposé IX shows that Hodgkin's
theorem is equivalent to
\[
 K^\bullet(G)\simeq\Z,
 \qquad\text{hence}\qquad
 \Gtop^\bullet(G)\simeq\Z.
\]
The homomorphism $K^\bullet(G/B)\to K^\bullet(G)$ is a priori surjective,
with kernel generated by the elements $\cl'(L)-1$, where $L$ ranges over
the invertible sheaves on $X=G/B$ associated with representations
$B\to\Gm$ (which in fact give all invertible sheaves on $X$ [5,
Exposé 15]). Similarly,
\[
 A(G/B)\longrightarrow A(G)
\]
is surjective, and its kernel is the ideal generated by the elements of
$A^1(G/B)$ defined by representations $B\to\Gm$, hence by all of
$A^1(G/B)$.\footnote{See IX 3.4, 2.2.} Thus $A(G)$ is reduced to
$A^0(X)=\Z$ if and

only if $G$ is ``without torsion'', that is, if and only if $G$ is a
product of groups $\operatorname{SL}(n)$ and $\operatorname{Sp}(m)$. In
that case, and only in that case, the homomorphism
\[
 \varphi:A(G)\longrightarrow\Gtop^\bullet(G)
\]
is an isomorphism. Thus, if $G$ has torsion, the underlying scheme gives an
example of a non-bijective $\varphi$.

\textbf{4.8.} One likewise obtains an example in which the homomorphism
$\gamma$ of (4.1),
\begin{equation}\tag{4.9}
 \gamma:A(G)\longrightarrow H^{2\bullet}(G,\Zell(\bullet)),
\end{equation}
does not factor through $\Gtop^\bullet(G)$, equivalently here, an example
where it is nonzero in positive degrees. By suitable specialization
theorems it is enough to take $k=\C$ and study
\begin{equation}\tag{4.10}
 A(G)\longrightarrow H^\bullet(G(\C),\Z),
\end{equation}
which is induced, by passing to the quotient, from the canonical map
\[
 A(G/B)\simeq H^\bullet(G/B(\C),\Z)
   \longrightarrow H^\bullet(G(\C),\Z).
\]
It is tempting to conjecture that (4.10) is injective, since the torsion
primes of $G(\C)$ are known to be exactly those dividing the order of the
finite group
\[
 A^+(G)=\bigoplus_{i>0}A^i(G).
\]
If so, whenever $G$ had torsion, (4.10) would not factor through
$\varphi$, and the same would hold for (4.9) for a suitable $\ell$ (more
precisely, for $\ell$ dividing the order of $A^+(G)$, that is, a torsion
prime of $G$). At least for $G=G_2$ over $\C$, (4.10) is nonzero in
positive degree: an element of $A^3(G)$ has image of order $2$ in
$H^6(G(\C),\Z)$. Hence (4.9) does not factor through $\varphi$ for
$\ell=2$. The assertion about $G_2$ is equivalent to the existence of an
element of $H^6(G/B(\C),\Z)$ whose image in $H^6(G(\C),\Z)$ is nonzero;
it follows from [3], via the usual dictionary between complex semisimple
algebraic groups and compact Lie groups, and was kindly communicated to me
by J.-P. Serre.

\section*{5. Relations between $\operatorname{Gr}^\bullet(X)$ and
$H^{2\bullet}(X,\Zell(\bullet))$}

\textbf{5.1.} The example of 4.8 does not rule out the possible existence,
for an arbitrary scheme $X$, of a ring homomorphism
\[
 g':\operatorname{Gr}^\bullet(X)
   \longrightarrow H^{2\bullet}(X,\Zell(\bullet))
\]
such that for every locally free module $E$ on $X$, its component $g^i$
carries the Chern class $c_i(E)$ of Exposé V to the cohomological Chern
class of SGA 5 VII. Here $\ell$ is a prime distinct from all residue
characteristics of $X$. Since $\operatorname{Gr}^\bullet(X)$ is generated
by the $c^i(E)$, compatibility with Chern classes determines $g'$
uniquely.

\textbf{5.2.} The Chern homomorphism
\[
 K^\bullet(X)\longrightarrow
 \Ch\bigl(H^{2\bullet}(X,\Zell(\bullet))\bigr)
\]
induces, upon passing to the associated gradeds for the filtration on the
source and the degree filtration on the target, a homomorphism of graded
groups (but not of rings)
\[
 \psi:\operatorname{Gr}^\bullet(X)
   \longrightarrow H^{2\bullet}(X,\Zell(\bullet))
\]
satisfying
\[
 \psi(c_i(E))=(-1)^{i-1}(i-1)!\,c_i^\ell(E),
\]
where $c_i^\ell(E)$ is the $\ell$-adic Chern class. Division among the
components $\psi^i$ therefore gives a ring homomorphism
\[
 g_\Q:\operatorname{Gr}^\bullet(X)
   \longrightarrow H^{2\bullet}(X,\Qell(\bullet))
   =H^{2\bullet}(X,\Zell(\bullet))\otimes_{\Zell}\Qell
\]
carrying Chern classes to Chern classes, with image contained in the image
of $H^{2\bullet}(X,\Zell(\bullet))$. Thus the existence question for $g'$
has an affirmative answer if $H^{2\bullet}(X,\Zell(\bullet))$ is replaced
by its quotient by the torsion subgroup. The first doubtful case for the
components $g^i$ of the hypothetical homomorphism is $g^3$: a priori one
can define only
\[
 2g^3=\psi^3.
\]
The answer would be negative if, for example, one could find three
invertible sheaves $L_1,L_2,L_3$ on $X$ satisfying
\[
 (L_1-1)(L_2-1)(L_3-1)=0\quad\text{in }K^\bullet(X),
\]

% Current-rescribe idx695 | printed volume page 682 | source-PDF page 685
but such that
\[
 c_1(L_1)c_1(L_2)c_1(L_3)\ne0
   \quad\text{in }H^6(X,\mu_\ell^{\otimes3}).
\]

\textbf{5.3.} When $X$ is smooth and quasi-projective over a field $k$,
one may similarly ask for a ring homomorphism
\[
 g:\operatorname{Gr}^\bullet(X)\longrightarrow A(X)
\]
with values in the Chow ring and carrying Chern classes to Chern classes.
The preceding discussion applies and shows in particular how to construct
$(-1)^{i-1}(i-1)!g^i$ directly when $g^i$ itself is unavailable.

\textbf{5.4.} Essentially the same question arises if, instead of a scheme,
$X$ is a compact topological space. One can still form the graded ring
$\operatorname{Gr}^\bullet(X)$ of $K^\bullet(X)$ with its
$\lambda$-filtration and ask for a ring homomorphism
\[
 \operatorname{Gr}^\bullet(X)\longrightarrow H^{2\bullet}(X,\Z)
\]
carrying Chern classes to Chern classes.

\section*{6. Cohomological Riemann--Roch and the Gysin homomorphism}

\textbf{6.1.} For simplicity, restrict to schemes $X$ admitting an ample
invertible sheaf, so that one need not distinguish naive $K'(X)$ from the
other $K'(X)$. Begin with Riemann--Roch ``with denominators'' for a proper
complete-intersection morphism $f:X\to Y$. For a prime $\ell$ invertible on
$Y$, one can define the $\ell$-adic Chern classes of a locally free module
$E$,
\[
 c_i(E)\in H^{2i}(X,\Zell(i))
\]
(SGA 5 VII), corresponding to a homomorphism
\begin{equation}\tag{6.1}
 K'(X)\longrightarrow\Ch\bigl(H^{2\bullet}(X,\Zell(\bullet))\bigr).
\end{equation}
The problem is to give a cohomological version of the Riemann--Roch
formula. The only missing ingredient needed to state it is a Gysin
homomorphism

% Current-rescribe idx696 | printed volume page 683 | source-PDF page 686
\begin{equation}\tag{6.2}
 H^n(X,\Qell(i))\longrightarrow H^{n-2d}(Y,\Qell(i-d)),
\end{equation}
where $d$ is the virtual relative dimension of $f$. Assuming this
homomorphism is defined, and using Riemann--Roch in the form of Exposé VIII
with values in $\operatorname{Gr}'(Y)_\Q$, the validity of $\ell$-adic
Riemann--Roch for $f$ is exactly equivalent to the commutativity of
\begin{equation}\tag{6.3}
\begin{tikzcd}[column sep=large,row sep=large]
 \operatorname{Gr}'(X)_\Q \arrow[r] \arrow[d]
   & H^{2\bullet}(X,\Qell(\bullet)) \arrow[d] \\
 \operatorname{Gr}'(Y)_\Q \arrow[r]
   & H^{2\bullet}(Y,\Qell(\bullet)).
\end{tikzcd}
\end{equation}
The horizontal arrows are obtained from (6.1) by passing to associated
gradeds and dividing so that Chern classes map to Chern classes; the
vertical arrows are respectively the map of Exposé VIII and the assumed
cohomological Gysin homomorphism.

\textbf{6.2.} The Gysin homomorphism is defined whenever $Y$ is smooth over
a field $k$ (SGA 5 IV), using relative cohomological purity over $k$. The
same construction applies when $Y$ is excellent, regular, and of
characteristic zero, by absolute cohomological purity (SGA 4 XIX); it will
apply automatically to every excellent regular $Y$ without a restriction
on the characteristic once absolute cohomological purity is proved in full
generality. It also applies, by relative cohomological purity (SGA 4 XVI
3.7), when $X$ and $Y$ are both smooth over the same scheme $S$ and $f$ is
an $S$-morphism. One should conjecture that (6.3) is commutative in each of
these cases. The only case presently verified is when $X$ and $Y$ are both
smooth over a field $k$; it follows from the commutativity of the analogous
diagram
\begin{equation}\tag{6.4}
\begin{tikzcd}[column sep=large,row sep=large]
 A(X) \arrow[r] \arrow[d]
   & H^{2\bullet}(X,\Zell(\bullet)) \arrow[d] \\
 A(Y) \arrow[r]
   & H^{2\bullet}(Y,\Zell(\bullet)),
\end{tikzcd}
\end{equation}

% Current-rescribe idx697 | printed volume page 684 | source-PDF page 687
which involves the cycle-class rings modulo rational equivalence $A(X)$ and
$A(Y)$ and whose commutativity is proved in SGA 5 IV.

\textbf{6.3.} In every case above where a Gysin homomorphism can be defined,
the construction in fact has coefficients in $\Zell(\bullet)$, not merely
in $\Qell(\bullet)$. When $f$ is an immersion this permits one to ask for
the denominator-free Riemann--Roch formula as explained in no.~3. If $X$
and $Y$ are both smooth over a field $k$, the commutativity of (6.4) shows
that it would be enough to prove the denominator-free formula in the Chow
ring $A(Y)$; see (3.3).

\textbf{6.4.} Without a regularity hypothesis on $Y$, even when $X$ and $Y$
are of finite type over $\C$, there is no apparent plausible definition of
a cohomological Gysin homomorphism (6.2), even for a regular immersion of
codimension $1$. Can one nevertheless define one with reasonable
properties, in particular making (6.3) commutative? This is plainly related
to defining, when $X$ is a complete intersection of codimension $i$ in
$Y$, ``its cohomology class''
\[
 \gamma(X)\in H^{2i}(Y,\Zell(i))
\]
so that intersections without excess components become cup-products. This
question has an affirmative answer at least modulo torsion, that is, in
$H^{2\bullet}(Y,\Qell(\bullet))$, by setting
\[
 \gamma(X)=\frac{1}{(-1)^{i-1}(i-1)!}\,c_i(\mathcal O_X).
\]
But it is not known whether $c_i(\mathcal O_X)$ is divisible by $(i-1)!$
in $H^{2i}(Y,\Zell(i))$. In particular, let $Y$ be a projective scheme of
dimension $3$ over $\C$, and let $X$ be a singular point of $Y$. Is
$c_3(\mathcal O_X)$ divisible by $2$ in $H^6(Y,\mathbf Z_2(3))$ and, in
particular, is its Chern class zero in $H^6(Y,\Z/2\Z)$? Similarly, let $Y$
be a singular projective scheme of dimension $4$, and let $X$ be a
one-dimensional closed subscheme which globally is the intersection of
three positive divisors $D_1,D_2,D_3$. Consider their product modulo $2$:
\[
 c_1(D_1)c_1(D_2)c_1(D_3)\in H^6(Y,\Z/2\Z).
\]

% Current-rescribe idx698 | printed volume page 685 | source-PDF page 688
Is this element of $H^6$ independent of the particular divisors $D_i$
whose intersection is $X$?

\section*{7. Chern classes of perfect complexes}

\textbf{7.1.} To develop a cohomological Riemann--Roch formula, with or
without denominators, without the rather artificial assumption that the
schemes admit an ample invertible sheaf, one is necessarily led to define,
for a perfect complex of modules $L^\bullet$ on a scheme $X$, Chern classes
\[
 c_i(L^\bullet)\in H^{2i}(X,\Zell(i))
\]
generalizing the Chern classes of locally free modules in SGA 5 VII. The
same question arises for general locally ringed topoi (compare [10]) and
for perfect complexes on paracompact spaces ringed by continuous
complex-valued functions, using ordinary integral cohomology. For a compact
space, $K^\bullet(X)$ built from perfect complexes agrees with the ``naive''
group built from locally free modules (Exposé IV), so one returns trivially
to the classical definition. Already for a locally compact space, however,
the problem appears nontrivial, because integral cohomology need not be the
inverse limit of the cohomologies of its compact subsets.

\textbf{7.2.} More precisely, let $Y$ be a closed subset of $X$ such that
$L^\bullet|_{X-Y}$ is zero in the derived category, that is, all cohomology
sheaves of $L^\bullet$ are supported in $Y$. One would like to define Chern
classes with support in $Y$,
\[
 c_i^Y(L^\bullet)\in H_Y^{2i}(X,\Zell(i))
 \qquad\text{respectively}\qquad
 H_Y^{2i}(X,\Z),
\]
whose images under the canonical homomorphism
\[
 H_Y^\bullet(X,-)\longrightarrow H^\bullet(X,-)
\]
are the ``ordinary'' Chern classes of 7.1.

% Current-rescribe idx699 | printed volume page 686 | source-PDF page 689
Even when the latter reduce to the Chern classes of locally free modules
(when $X$ is a scheme with an ample invertible sheaf, or a compact space),
the construction of these Chern classes ``with supports'' has not been
made, and will probably have to be carried out at the same time as that of
the general Chern classes postulated in 7.1.\footnote{Added in October
1968. The problems raised here have been solved in Hodge cohomology by
L.~Illusie; see [13].}

Chern classes with supports will probably have a role in proving the
denominator-free cohomological Riemann--Roch formula (see 3.5). They should
also make possible an improved version of Riemann--Roch, with or without
denominators, involving families of supports.

\section*{8. The Chow ring of regular schemes}

The problem is to define rational equivalence of cycles and study the
resulting ring of classes, in particular its functorial properties, for a
noetherian regular scheme $X$ admitting an ample invertible sheaf,
generalizing the theory known when $X$ is smooth over a field [4]. The key
technical result should again be a ``moving lemma'', ensuring that a cycle
can be varied within its class so as to meet a given cycle in the proper
dimension.

The problem is natural, especially because 4.5--4.8 show that the
substitutes $\operatorname{Gr}'(X)$ and $\Gtop^\bullet(X)$ for the Chow ring
$A(X)$ are not very satisfactory when one wants more than a theory
``modulo torsion''. Suppose $X$ is obtained from a smooth projective
algebraic scheme $Y$ over a field $k$ by taking the local scheme at the
vertex of the projective cone $C$ and deleting the vertex. Assuming that
$A(X)$ is the inductive limit of the Chow rings of $U-(O)$, where $U$
ranges over open neighbourhoods of the vertex $O$ in $C$, one finds that
$A(X)$ is canonically isomorphic to
\[
 A(Y)/\xi A(Y),
\]
where $\xi\in A^1(Y)$ is induced by the chosen projective embedding of $Y$.
Thus knowing the nonclassical Chow ring $A(X)$ is equivalent to knowing
$A(Y)/\xi A(Y)$, which may be called the ``primitive part'' of the Chow
ring of $Y$, in a sense suggested by Lefschetz--Hodge cohomology for smooth
projective algebraic varieties. This example shows that the ``local Chow
rings'' of local rings with isolated

% Current-rescribe idx700 | printed volume page 687 | source-PDF page 690
singularities are highly nontrivial geometric invariants and should have a
role in the ``arithmetic'' study of these local rings.

Once a reasonable notion of Chow ring has been obtained for a noetherian
regular scheme admitting an ample invertible sheaf, one should generalize
to this case the relations established or conjectured in no.~4 between
that ring and $\Gtop^\bullet(X)$, and formulate in terms of it the
denominator-free Riemann--Roch problem stated in no.~3.

\subsection*{Note for page 1}

The theory of the operations $\Lambda^i$ in $D^-(X)$ and the operations
$\lambda^i$ in $K'(X)$ was written down by P.~Deligne in 1968. He proves
that $K'(X)$ thereby becomes a pre-$\lambda$-ring and, when $\mathcal O_X$
is a sheaf of algebras over a field, even a $\lambda$-ring. It is plausible
that a refinement of Deligne's method will remove this restrictive
hypothesis on $\mathcal O_X$.
