\section{The sheaf associated to a presheaf}

\begin{statement}{Definition 3.0.1}
Let \(C\) be a site. A \emph{topologically generating family} of \(C\), or simply a \emph{generating family} when no confusion can result, is a set \(G\) of objects of \(C\) such that every object of \(C\) is the target of a covering family of morphisms of \(C\) (1.2) whose sources are elements of \(G\).
\end{statement}

\begin{statement}{Definition 3.0.2}
Let \(\U\) be a universe. A \(\U\)-\emph{site} is a site \(C\) whose underlying category is a \(\U\)-category (Expos\'e~I, 1.1), and which has a small topologically generating family. If \(C\) is a \(\U\)-category, a \(\U\)-\emph{topology} on \(C\) is a topology on \(C\) making \(C\) into a \(\U\)-site. A site \(C\) is said to be \(\U\)-\emph{small}, or, by abuse of language, \emph{small}, if the underlying category of \(C\) is small (Expos\'e~I, 1.0).
\end{statement}

\subsection*{3.0.3}
It follows immediately from the definitions that every topology finer than a \(\U\)-topology is again a \(\U\)-topology, and that a small site is a \(\U\)-site.

\begin{statement}{Proposition 3.0.4}
Let \(C\) be a \(\U\)-site and let \(G\) be a small topologically generating family of \(C\). For \(X\in\Ob(C)\), let \(J_G(X)\) denote the set of covering sieves on \(X\) generated by a family of morphisms
\[
(Y_\alpha \xrightarrow{u_\alpha} X)_{\alpha\in A},\qquad Y_\alpha\in G.
\]
Then:
\begin{enumerate}[label=\arabic*)]
\item the set \(J_G(X)\) is small;
\item \(J_G(X)\) is cofinal in the set \(J(X)\) of all covering sieves on \(X\), ordered by inclusion;
\item for every \(R\in J_G(X)\), there is a small epimorphic family \((u_\alpha:Y\to R)_{\alpha\in A}\) with \(Y\in G\) (Expos\'e~I, 10.2).
\end{enumerate}
\end{statement}

\begin{prooftext}
For (1), put
\[
A(X)=\coprod_{Y\in G}\Hom(Y,X).
\]
The set \(A(X)\) is small (Expos\'e~I, 0), and \(\operatorname{card} J_G(X)\le 2^{\operatorname{card} A(X)}\). Hence \(J_G(X)\) is small.

For (2), let \(R\in J(X)\). Put
\[
A(R)=\coprod_{Y\in G}\Hom(Y,R),
\]
and let \(R'\) be the sieve on \(X\) generated by the family
\[
Y\xrightarrow{u}R\hookrightarrow X,\qquad u\in A(R).
\]
We have \(R'\subset R\), and it remains only to prove that \(R'\) is covering. By axiom (T2) of 1.1, it is enough to show that, for every morphism \(Z\to R\) with \(Z\in\Ob(C)\), the sieve \(R'\times_X Z\) on \(Z\) is covering. But this sieve contains the sieve generated by the family of all morphisms \(Y\to Z\) with \(Y\in G\), which is covering by the hypothesis on \(G\). Hence \(R'\times_XZ\) is covering, again by axiom (T2).

For (3), let \(R\in J_G(X)\). The family
\[
(Y\xrightarrow{u}R),\qquad u\in A(R)=\coprod_{Y\in G}\Hom(Y,R),
\]
is epimorphic by construction. For every \(Y\in\Ob(C)\), \(\Hom(Y,R)\) is contained in \(\Hom(Y,X)\), which is small. Hence \(A(R)\) is small. \qedtext
\end{prooftext}

\subsection*{3.0.5}
Let \(C\) be a \(\U\)-site, let \(\V\) be a universe such that \(C\in\V\) and \(\U\subset \V\), and let \(G\) be a \(\U\)-small topologically generating family of \(C\). The category \(\widehat C\) of presheaves of \(\U\)-sets on \(C\) is a \(\V\)-category (Expos\'e~I, 1.1.1). Let \(X\) be an object of \(C\). The set \(J(X)\) of covering sieves on \(X\), ordered by inclusion, is \(\V\)-small and cofiltered (1.1.1). For every \(\U\)-presheaf \(F\), the inductive limit
\[
\varinjlim_{R\in J(X)} \Hom_{\widehat C}(R,F)
\]
is therefore representable by an element of \(\V\) (Expos\'e~I, 2.4.1). Moreover, by 3.0.4(3), for every \(R\in J_G(X)\), the set \(\Hom_{\widehat C}(R,F)\) is \(\U\)-small; since \(J_G(X)\) is a \(\U\)-small cofinal subset of \(J(X)\) (loc. cit.), it follows from Expos\'e~I, 2.4.2 that
\[
\varinjlim_{R\in J(X)} \Hom_{\widehat C}(R,F)
\]
is \(\U\)-small. We choose, for each \(F\) and \(X\), an element of \(\U\) representing this inductive limit, and put
\[
LF(X)=\varinjlim_{R\in J(X)}\Hom_{\widehat C}(R,F).
\]
If \(g:Y\to X\) is a morphism of \(C\), the base-change functor \(g^\ast:J(X)\to J(Y)\) defines a map
\[
LF(g):LF(X)\longrightarrow LF(Y),
\]
making \(X\mapsto LF(X)\) into a presheaf on \(C\).

Since the morphism \(\id_X:X\to X\) is an element of \(J(X)\), for every object \(X\) of \(C\) one has a map
\[
\ell(F)(X):F(X)\longrightarrow LF(X),
\]
and these maps plainly define a morphism of presheaves
\[
\ell(F):F\longrightarrow LF.
\]
It is also clear that \(F\mapsto LF\) is functorial in \(F\), and that the morphisms \(\ell(F)\) define a natural transformation
\[
\ell:\operatorname{Id}\longrightarrow L.
\]

Finally, let \(R\hookrightarrow X\) be a refinement of \(X\). The definition of \(LF(X)\) and Expos\'e~I, 1.4 give a map
\[
Z_R:\Hom_{\widehat C}(R,F)\longrightarrow\Hom_{\widehat C}(X,LF).
\]
For every morphism \(g:Y\to X\) of \(C\), the definition of \(LF\) shows that the square
\[
\begin{array}{ccc}
\Hom_{\widehat C}(R,F) & \xrightarrow{\ Z_R\ } & \Hom_{\widehat C}(X,LF)\\
\downarrow && \downarrow\\
\Hom_{\widehat C}(R\times_XY,F) & \xrightarrow{\ Z_{R\times_XY}\ } & \Hom_{\widehat C}(Y,LF)
\end{array}
\tag{*}
\]
is commutative, where the vertical arrows are the evident ones.

We now copy the standard argument of Demazure.

\begin{statement}{Lemma 3.1}
\begin{enumerate}[label=\arabic*)]
\item For every refinement \(i_R:R\hookrightarrow X\) and every \(u:R\to F\), the diagram
\[
\begin{array}{ccc}
R&\xrightarrow{i_R}&X\\
{\scriptstyle u}\downarrow&&\downarrow{\scriptstyle Z_R(u)}\\
F&\xrightarrow{\ell(F)}&LF
\end{array}
\tag{**}
\]
is commutative.

\item For every morphism \(v:X\to LF\), there exists a refinement \(R\) of \(X\) and a morphism \(u:R\to F\) such that \(Z_R(u)=v\).

\item Let \(Y\) be an object of \(C\), and let \(u,v:Y\rightrightarrows F\) be two morphisms such that \(\ell(F)\circ u=\ell(F)\circ v\). Then the equalizer, or kernel, of the pair \((u,v)\) is a refinement of \(Y\).

\item Let \(R\) and \(R'\) be two refinements of \(X\), and let \(u:R\to F\) and \(u':R'\to F\) be two morphisms. Then \(Z_R(u)=Z_{R'}(u')\) if and only if \(u\) and \(u'\) coincide on some refinement
\[
R''\hookrightarrow R\times_XR'.
\]
\end{enumerate}
\end{statement}

\begin{prooftext}
Only assertion (1) is nontrivial. We must prove
\[
Z_R(u)\circ i_R=\ell(F)\circ u.
\]
By Expos\'e~I, 3.4 it is enough to prove equality after composition with every morphism \(g:Y\to R\), where \(Y\) is an object of \(C\). Let \(f=i_R\circ g\), and consider \(R\times_XY\). The canonical monomorphism \(R\times_XY\to Y\) has a section, hence is an isomorphism by 1.3(3). By definition of \(\ell(F)\), the morphism \(Z_{R\times_XY}(u\circ g')\) is \(\ell(F)\circ u\circ g\). On the other hand, the commutativity of \((*)\) gives
\[
Z_{R\times_XY}(u\circ g')=Z_R(u)\circ f.
\]
Thus \(\ell(F)\circ u\circ g=Z_R(u)\circ i_R\circ g\), as required. \qedtext
\end{prooftext}

\begin{statement}{Proposition 3.2}
\begin{enumerate}[label=\arabic*)]
\item The functor \(L\) is left exact (Expos\'e~I, 2.3.2).
\item For every presheaf \(F\), \(LF\) is a separated presheaf.
\item A presheaf \(F\) is separated if and only if the morphism \(\ell(F):F\to LF\) is a monomorphism. In that case \(LF\) is a sheaf.
\item The following properties are equivalent:
\begin{enumerate}[label=(\roman*)]
\item \(\ell(F):F\to LF\) is an isomorphism;
\item \(F\) is a sheaf.
\end{enumerate}
\end{enumerate}
\end{statement}

\begin{prooftext}
For (1), it is enough, by Expos\'e~I, 3.1, to prove that for every object \(X\) of \(C\), the functor \(F\mapsto LF(X)\) commutes with finite projective limits. But for every \(R\hookrightarrow X\) in \(J(X)\), the functor
\[
F\longmapsto \Hom_{\widehat C}(R,F)
\]
commutes with projective limits, by the definition of projective limit; and the filtered inductive limit \(\varinjlim_{J(X)}\) commutes with finite projective limits because \(J(X)\) is a cofiltered set (Expos\'e~I, 2.7).

For (2), let \(X\) be an object of \(C\), and let \(f,g:X\rightrightarrows LF\) be two morphisms which coincide on a refinement \(R\hookrightarrow X\). By Lemma~3.1(2), there is a covering sieve \(R'\hookrightarrow X\), which we may suppose contained in \(R\), and two morphisms \(u,v:R'\rightrightarrows F\) such that \(Z_{R'}(u)=f\) and \(Z_{R'}(v)=g\). By Lemma~3.1(1), \(\ell(F)\circ u=\ell(F)\circ v\). Hence, by Lemma~3.1(4), \(u\) and \(v\) coincide on a refinement \(R''\hookrightarrow R'\). If \(w\) denotes the restriction of \(u\) to \(R''\), then
\[
Z_{R''}(w)=Z_{R'}(u)=Z_{R'}(v),
\]
and consequently \(f=g\). Thus \(LF\) is separated.

For (3), if \(F\) is separated, then \(\ell(F)\) is a monomorphism, because a filtered inductive limit of monomorphisms is a monomorphism. Conversely, if \(\ell(F)\) is a monomorphism, then \(F\) is a subpresheaf of a separated presheaf, hence is separated. We now prove that \(LF\) is then a sheaf. Let \(i:R\hookrightarrow X\) be a covering sieve of an object \(X\) of \(C\), and let \(u:R\to LF\) be a morphism. It is enough to prove that \(u\) factors through \(X\). Put
\[
R'=F\times_{LF}R,\qquad v=Z_{R'}(\operatorname{pr}_1).
\]
To prove \(u=v\circ i\), it is enough, by Expos\'e~I, 3.4, to test after every morphism \(m:Y\to R\) with \(Y\) an object of \(C\). Put \(R''=Y\times_RR'\), with projections \(p_1,p_2\). The projection \(R''\to Y\) is a monomorphism making \(R''\) into a covering sieve of \(Y\) by Lemma~3.1(2). We have
\[
v\circ i\circ p_2=\ell(F)\circ \operatorname{pr}_1
\]
by Lemma~3.1(1), and hence
\[
v\circ i\circ p_2=u\circ m\circ p_2.
\]
Pulling back along \(p_1\) gives
\[
v\circ i\circ m\circ p_2=u\circ m\circ p_2.
\]
Since \(LF\) is separated, \(v\circ i\circ m=u\circ m\). This proves the sheaf condition.

Assertion (4) is clear from (3). \qedtext
\end{prooftext}

\begin{remarktext}{Remark 3.3}
Let \(J'(X)\) be a cofinal subset of \(J(X)\). Then
\[
LF(X)=\varinjlim_{R\in J'(X)}\Hom_{\widehat C}(R,F).
\]
In particular, if the topology of \(C\) is defined by a pretopology \(X\mapsto \Cov(X)\) (1.1.3), then the functor \(L\) can be described by means of the covering families of \(\Cov(X)\) (Expos\'e~I, 2.12 and 3.5). Making the formulae explicit, one recovers the usual construction.
\end{remarktext}

\begin{statement}{Theorem 3.4}
Let \(C\) be a \(\U\)-site. The inclusion functor
\[
i:\widetilde C\hookrightarrow \widehat C
\]
from sheaves to presheaves has a left adjoint
\[
\underline a:\widehat C\longrightarrow \widetilde C,
\]
which is left exact (Expos\'e~I, 2.3.2). The functor \(i\circ\underline a\) is canonically isomorphic to \(L\circ L\) (cf. 3.0.5). For every presheaf \(F\), the adjunction morphism
\[
F\longrightarrow i\,\underline a(F)
\]
is, through the preceding isomorphism, the composite
\[
F\xrightarrow{\ell(F)}LF\xrightarrow{\ell(LF)}L(LF).
\]
\end{statement}

\begin{statement}{Definition 3.5}
The sheaf \(\underline aF\) is called the \emph{sheaf associated to the presheaf \(F\)}, or the \emph{sheafification} of \(F\).
\end{statement}

Theorem~3.4 follows immediately from Proposition~3.2.

\begin{statement}{Proposition 3.6}
Let \(C\) be a \(\U\)-site and let \(\V\supset \U\) be a universe. Denote by
\[
\widehat C_{\U},\quad \widetilde C_{\U}
\]
the categories of \(\U\)-presheaves and \(\U\)-sheaves, and by
\[
\widehat C_{\V},\quad \widetilde C_{\V}
\]
the categories of \(\V\)-presheaves and \(\V\)-sheaves. Let
\[
\underline a_{\U}:\widehat C_{\U}\to \widetilde C_{\U},\qquad
\underline a_{\V}:\widehat C_{\V}\to \widetilde C_{\V}
\]
be the corresponding associated-sheaf functors. Then the diagram
\[
\begin{array}{ccc}
\widehat C_{\U} & \xrightarrow{\underline a_{\U}} & \widetilde C_{\U}\\
\downarrow && \downarrow\\
\widehat C_{\V} & \xrightarrow{\underline a_{\V}} & \widetilde C_{\V}
\end{array}
\]
where the vertical functors are the canonical inclusions, commutes up to canonical isomorphism.
\end{statement}

\begin{prooftext}
This follows from the construction of the functors \(\underline a_{\U}\) and \(\underline a_{\V}\) (3.4 and 3.0.5). \qedtext
\end{prooftext}

\section{Exactness properties of the category of sheaves}

The exactness properties of the category of sheaves are deduced from those of the category of presheaves by means of Theorem~3.4. This section makes that philosophy explicit in a number of typical statements which are among the most useful.

\begin{statement}{Theorem 4.1}
Let \(C\) be a \(\U\)-site, let \(\widetilde C\) be the category of sheaves, let
\[
\underline a:\widehat C\longrightarrow \widetilde C
\]
be the associated-sheaf functor, and let
\[
i:\widetilde C\longrightarrow \widehat C
\]
be the inclusion functor.
\begin{enumerate}[label=\arabic*)]
\item The functor \(\underline a\) commutes with inductive limits and is exact.
\item \(\U\)-inductive limits in \(\widetilde C\) are representable. For every small category \(I\) and every functor \(E:I\to\widetilde C\), the canonical morphism
\[
\varinjlim_I E\longrightarrow \underline a\!\left(\varinjlim_I iE\right)
\]
is an isomorphism.
\item \(\U\)-projective limits in \(\widetilde C\) are representable. For every object \(X\) of \(C\), the functor \(F\mapsto F(X)\) on \(\widetilde C\) commutes with projective limits; equivalently, the inclusion \(i:\widetilde C\to \widehat C\) commutes with projective limits.
\end{enumerate}
\end{statement}

\begin{prooftext}
These properties follow essentially from Theorem~3.4 and Expos\'e~I, 2.11. \qedtext
\end{prooftext}

Thus, in the category of sheaves, products indexed by an element of \(\U\), fiber products, sums indexed by an element of \(\U\), amalgamated sums, kernels, cokernels, images, and coimages are representable.

\begin{statement}{Corollary 4.1.1}
Let \(C\) be a \(\U\)-site and let \(F\) be a sheaf of sets on \(C\). The canonical homomorphism
\[
\varinjlim_{C/F}\underline aX\longrightarrow F
\]
is an isomorphism.
\end{statement}

\begin{prooftext}
This follows from Expos\'e~I, 3.4 and the fact that \(\underline a\) commutes with inductive limits. \qedtext
\end{prooftext}

\begin{statement}{Proposition 4.2}
Every morphism of \(\widetilde C\) which is both an epimorphism and a monomorphism is an isomorphism.
\end{statement}

\begin{prooftext}
Let \(u:G\to H\) be a morphism of \(\widetilde C\) which is both an epimorphism and a monomorphism. First note that \(u\) is a monomorphism of presheaves (4.1(1)). Form the amalgamated sum \(H\amalg_GH=K\) and the two canonical morphisms \(i_1,i_2:H\rightrightarrows K\) in the category of presheaves. Since \(u\) is a monomorphism of presheaves, the square
\[
\begin{array}{ccc}
G&\xrightarrow{u}&H\\
{\scriptstyle u}\downarrow&&\downarrow{\scriptstyle i_2}\\
H&\xrightarrow{i_1}&K
\end{array}
\]
is immediately verified to be both cartesian and cocartesian (Expos\'e~I, 10.1). Applying the associated-sheaf functor, we get a cartesian and cocartesian square in the category of sheaves:
\[
\begin{array}{ccc}
G&\xrightarrow{u}&H\\
{\scriptstyle u}\downarrow&&\downarrow{\scriptstyle \underline a i_2}\\
H&\xrightarrow{\underline a i_1}&\underline aK.
\end{array}
\tag{*}
\]
Since \(u\) is an epimorphism of sheaves, \(\underline a i_1\) is an isomorphism; since the square \((*)\) is cartesian, \(u\) is an isomorphism. \qedtext
\end{prooftext}

\begin{statement}{Proposition 4.3}
\begin{enumerate}[label=\arabic*)]
\item Representable inductive limits in \(\widetilde C\) are universal (Expos\'e~I, 2.5).
\item Every epimorphic family of morphisms is an effective universal epimorphic family (2.6).
\item Every equivalence relation is effective and universal (Expos\'e~I, 10.6).
\item \(\U\)-filtered inductive limits commute with finite projective limits (Expos\'e~I, 2.6).
\end{enumerate}
\end{statement}

\begin{prooftext}
Assertions (1)--(4) hold in the category of sets and hence in the category of presheaves of sets (Expos\'e~I, 3.1). Assertions (1) and (4) then follow at once from the corresponding assertions for presheaves and from Theorem~4.1.

We prove (3). Let \(p_1,p_2:R\rightrightarrows X\) be an equivalence relation in \(\widetilde C\). It is also an equivalence relation of presheaves (4.1.1). Hence there exists a morphism of presheaves \(u:X\to Y\) such that
\[
\begin{array}{ccc}
R&\xrightarrow{p_2}&X\\
{\scriptstyle p_1}\downarrow&&\downarrow{\scriptstyle u}\\
X&\xrightarrow{u}&Y
\end{array}
\]
is both cartesian and cocartesian in the category of presheaves. Applying the associated-sheaf functor, one obtains a cartesian and cocartesian square in the category of sheaves (4.1(1)). Thus \(R\rightrightarrows X\) is an effective equivalence relation. Since every equivalence relation is effective by the preceding argument, every equivalence relation is universally effective.

We prove (2). Let \((u_i:X_i\to X)_{i\in I}\) be an epimorphic family in \(\widetilde C\). By 2.6 and Expos\'e~I, 2.12, it is enough to prove:
\begin{enumerate}[label=\alph*)]
\item for every morphism of sheaves \(v:Y\to X\), the family
\[
\operatorname{pr}_{2,i}:X_i\times_XY\to Y,\qquad i\in I,
\]
is epimorphic in \(\widetilde C\);
\item for every sheaf \(Z\), the diagram of sets
\[
\Hom(X,Z)\longrightarrow \prod_{i\in I}\Hom(X_i,Z)
\rightrightarrows \prod_{(i,k)\in I\times I}\Hom(X_i\times_XX_k,Z)
\]
is exact.
\end{enumerate}

Let \(X'\) be the union of the images of the morphisms \(u_i\), in the sense of presheaves, and let \(j:X'\hookrightarrow X\) be the canonical injection. For every \(i\), the morphism \(u_i:X_i\to X\) factors as \(X_i\xrightarrow{u_i'}X'\xrightarrow{j}X\), and the family \(u_i':X_i\to X'\) is epimorphic in \(\widehat C\). Since \(j\) is a monomorphism, the morphism \(\underline a j:\underline aX'\to X\) is a monomorphism of sheaves (4.1). Since \(u_i=\underline a j\,\underline a u_i'\) for each \(i\), \(\underline a j\) is also an epimorphism of sheaves. Hence \(\underline a j\) is an isomorphism by Proposition~4.2.

Let \(v:Y\to X\) be a morphism of sheaves. After base change we obtain a morphism \(j_v:X'\times_XY\to Y\) and morphisms \(u_{i,v}':X_i\times_XY\to X'\times_XY\). Since \(\underline a\) commutes with fiber products, \(\underline a j_v\) is an isomorphism. Since epimorphic families of presheaves remain epimorphic after base change, the family \(u'_{i,v}\) is epimorphic in \(\widehat C\). As \(\underline a\) commutes with inductive limits, the family
\[
\underline a u'_{i,v}:X_i\times_XY\to \underline a(X'\times_XY)
\]
is epimorphic in \(\widetilde C\). This proves (a).

For (b), let \(Z\) be a sheaf. Since \(\underline a j\) is an isomorphism, the map
\[
\Hom(X,Z)\longrightarrow \Hom(X',Z)
\]
is a bijection. Since the \(u_i'\) form an epimorphic family in \(\widehat C\), the diagram
\[
\Hom(X',Z)\longrightarrow \prod_{i\in I}\Hom(X_i,Z)
\rightrightarrows \prod_{(i,k)\in I\times I}\Hom(X_i\times_{X'}X_k,Z)
\]
is exact. Finally, since \(X'\) is a subpresheaf of \(X\), the presheaf \(X_i\times_{X'}X_k\) is canonically isomorphic to \(X_i\times_XX_k\). This gives (b). \qedtext
\end{prooftext}

\begin{remarktext}{Remarks 4.3.1}
\begin{enumerate}[label=\arabic*)]
\item Proposition~4.3(2) formally gives a second proof of Proposition~4.2. In any category, a morphism which is both an effective universal epimorphism and a monomorphism is plainly an isomorphism.

\item From the preceding propositions it follows that every morphism in the category of sheaves factors uniquely as an effective epimorphism followed by an effective monomorphism. For non-additive categories, this property is the natural analogue of axiom (AB2) for abelian categories, namely \(\operatorname{coim}\simeq\operatorname{im}\) [\textsc{Tohoku}].
\end{enumerate}
\end{remarktext}

\subsection*{4.4.0}
Let \(C\) be a \(\U\)-site. The functor \(h:C\to \widehat C\) (Expos\'e~I, 1.3.1), composed with the associated-sheaf functor, gives a functor
\[
\epsilon_C:C\longrightarrow \widetilde C,
\]
called the \emph{canonical functor} from \(C\) to \(\widetilde C\). It will be used constantly below. The functor \(\epsilon_C\) commutes with finite projective limits. When the topology of \(C\) is less fine than the canonical topology, \(\epsilon_C\) is fully faithful and commutes with projective limits; in that case it is, moreover, defined even if \(C\) need not have a small topologically generating family. We shall not study in detail the behavior of \(\epsilon_C\) with respect to inductive limits (cf. SGA~3, IV). We shall, however, need the following propositions.

\begin{statement}{Theorem 4.4}
Let \(C\) be a \(\U\)-site and let \((u_i:X_i\to X)_{i\in I}\) be a family of morphisms of \(C\) with target \(X\). The following conditions are equivalent:\footnote{When the site \(C\) need not have a small topologically generating family and when the topology of \(C\) is less fine than the canonical topology, one has (ii) \(\Rightarrow\) (i); see the proof of 4.4.}
\begin{enumerate}[label=(\roman*)]
\item The family
\[
(\epsilon_Cu_i:\epsilon_CX_i\to \epsilon_CX)_{i\in I}
\]
is an epimorphic family in \(\widetilde C\) (Expos\'e~I, 10.2).
\item The family \((u_i:X_i\to X)_{i\in I}\) is a covering family of \(C\) (1.2).
\end{enumerate}
\end{statement}

\begin{prooftext}
(ii) \(\Rightarrow\) (i). Let \(R\hookrightarrow X\) be the sieve generated by the \(u_i\) (Expos\'e~I, 4.3.3), and let \(u_i':X_i\to R\) be the morphisms induced by the \(u_i\). The family \(u_i'\) is epimorphic in \(\widehat C\), and \(R\) is a covering sieve on \(X\). Hence, for every sheaf \(F\), the map
\[
\Hom(X,F)\longrightarrow \Hom(R,F)
\]
is bijective, while
\[
\Hom(R,F)\longrightarrow \prod_i\Hom(X_i,F)
\]
is injective. Thus
\[
\Hom(X,F)\longrightarrow \prod_i\Hom(X_i,F)
\]
is injective, and consequently
\[
\Hom(\underline aX,F)\longrightarrow \prod_i\Hom(\underline aX_i,F)
\]
is injective. This is exactly (i).

(i) \(\Rightarrow\) (ii). With the notation just introduced, let \(J_R:R\hookrightarrow X\) be the canonical inclusion of the sieve generated by the \(u_i\). From (i) it follows that the morphism of sheaves
\[
\underline aJ_R:\underline aR\longrightarrow \underline aX
\]
is both an epimorphism and a monomorphism. Hence it is an isomorphism by Proposition~4.2. With the notation of Section~3, we have a commutative diagram
\[
\begin{array}{ccccc}
X&\xrightarrow{\ell(X)}&LX&\xrightarrow{\ell(LX)}&\underline aX\\
\uparrow J_R&&\uparrow LJ_R&&\uparrow\wr\\
R&\xrightarrow{\ell(R)}&LR&\xrightarrow{\ell(LR)}&\underline aR.
\end{array}
\]

By Lemma~3.1(2), there exists a covering sieve \(J_{S_1}:S_1\to X\) and a morphism \(u_1:S_1\to LR\) such that
\[
\ell(LX)\circ \ell(X)\circ J_{S_1}
=
\underline aJ_R\circ \ell(LR)\circ u_1
=
\ell(LX)\circ LJ_R\circ u_1.
\tag{4.6.1}
\]
Put \(m=\ell(X)\circ J_{S_1}\) and \(n=LJ_R\circ u_1\). These are morphisms \(S_1\to LX\). Let \(S_2\hookrightarrow S_1\) be the equalizer of the pair \((m,n)\). For every object \(Y\) of \(C\) and every morphism \(\alpha:Y\to S_1\), equation (4.6.1) gives
\[
\ell(LX)m\alpha=\ell(LX)n\alpha.
\]
By Lemma~3.1(3), the equalizer \(S_2\times_{S_1}Y\) of \(m\alpha\) and \(n\alpha\) is a covering sieve on \(Y\). Hence, by axiom (T2), \(J_{S_2}:S_2\hookrightarrow X\) is a covering sieve on \(X\). Thus we have a morphism \(u_2:S_2\to LR\) and a commutative diagram
\[
\begin{array}{ccc}
S_2&\xrightarrow{J_{S_2}}&X\\
{\scriptstyle u_2}\downarrow&&\downarrow{\scriptstyle \ell(X)}\\
LR&\xrightarrow{LJ_R}&LX
\end{array}
\tag{4.6.2}
\]
with \(R\hookrightarrow X\) inserted in the evident way through \(J_R\).

Let \(Y\) be an object of \(C\) and let \(\beta:Y\to S_2\) be a morphism. By Lemma~3.1(1), there exists a covering sieve \(J_{Q_1}:Q_1\hookrightarrow Y\) and a morphism \(v_1:Q_1\to R\) such that
\[
u_2\beta J_{Q_1}=\ell(R)v_1.
\]
Put
\[
f=J_{S_2}\beta J_{Q_1},\qquad g=J_Rv_1.
\]
Let \(Q_2\) be the equalizer of the pair \(f,g:Q_1\rightrightarrows X\), and let \(J_{Q_2}:Q_2\hookrightarrow Y\) be the canonical inclusion. For every object \(Z\) of \(C\) and every morphism \(\gamma:Z\to Q_1\), the commutativity of (4.6.2) gives
\[
\ell(X)f\gamma
=
LJ_R\,u_2\beta J_{Q_1}\gamma
=
LJ_R\,\ell(R)v_1\gamma
=
\ell(X)J_Rv_1\gamma
=
\ell(X)g\gamma.
\]
By Lemma~3.1(3), the equalizer \(Q_2\times_{Q_1}Z\) of \(f\gamma\) and \(g\gamma\) is a covering sieve on \(Z\). By axiom (T2), \(J_{Q_2}:Q_2\hookrightarrow Y\) is a covering sieve on \(Y\). Hence, for every morphism \(\beta:Y\to S_2\), there exists a covering sieve \(Q_2\hookrightarrow Y\) and a morphism \(v_2:Q_2\to R\) making
\[
\begin{array}{ccc}
Q_2&\xrightarrow{v_2}&R\\
\downarrow&&\downarrow J_R\\
Y&\xrightarrow{\beta}&S_2\xrightarrow{J_{S_2}}X
\end{array}
\]
commutative.

Now let \(S_2\cap R\) denote the fiber product of \(S_2\) and \(R\) over \(X\). The sieve
\[
(S_2\cap R)\times_{S_2}Y
\]
on \(Y\) contains \(Q_2\); hence it is covering. Axiom (T2) implies that \(S_2\cap R\) is a covering sieve on \(X\). Since \(R\) contains \(S_2\cap R\), \(R\) is itself a covering sieve on \(X\). \qedtext
\end{prooftext}

\begin{statement}{Corollary 4.4.4}
Let \(T\) be the topology of the \(\U\)-site \(C\). Let \(T'\) (respectively \(T''\)) be the finest topology on \(C\) among those for which the objects of \(\widetilde C\) are sheaves (respectively separated presheaves) (2.2). Then
\[
T=T'=T''.
\]
\end{statement}

Indeed, trivially \(T\le T'\le T''\), while Theorem~4.4 and Proposition~2.2 imply \(T''\le T\). \qedtext

\begin{statement}{Definition 4.5}
An \emph{initial object} of a category \(A\) is an object \(\phi_A\) representing the empty inductive limit, i.e. such that for every \(X\in\Ob(A)\) there is exactly one arrow \(\phi_A\to X\). An object \(\phi_A\) is called \emph{strict initial} if it is initial and every morphism with target \(\phi_A\) is an isomorphism.

Let \((S_i)_{i\in I}\) be a family of objects of a category \(A\). Suppose that the sum \(S=\coprod_{i\in I}S_i\) is representable. The sum \(S\) is said to be \emph{disjoint} if the structural morphisms \(S_i\to S\) are base-changeable monomorphisms and, for every pair \(i,j\) with \(i\ne j\), the fiber product \(S_i\times_SS_j\) is an initial object of \(A\). The sum is said to be \emph{universally disjoint} if it is disjoint and remains a disjoint sum after every base change \(T\to S\). It follows that, for every \(i\ne j\), the objects \(S_i\times_SS_j\) are strict initial objects of \(A\).
\end{statement}

\begin{remarktext}{Example 4.5.1}
In the category of sets, direct sums are disjoint and universal. Hence the same is true in every category of presheaves of sets (Expos\'e~I, 3.1), and therefore in every category of sheaves of sets on a \(\U\)-site (4.1). In particular, the initial object of \(\widetilde C\) is strict.
\end{remarktext}

\begin{statement}{Proposition 4.6}
Let \(C\) be a \(\U\)-category and let \((s_i:X_i\to X)_{i\in I}\) be a family of morphisms of \(C\). For every \(\U\)-topology \(\mathcal T\) on \(C\) (3.0.2), let \(\widetilde C_{\mathcal T}\) be the corresponding category of sheaves, and let
\[
\epsilon_{\mathcal T}:C\to\widetilde C_{\mathcal T}
\]
be the corresponding canonical functor.
\begin{enumerate}[label=\arabic*)]
\item Let \(\mathcal T\) be a \(\U\)-topology such that
\[
\coprod_{i\in I}\epsilon_{\mathcal T}(X_i)\xrightarrow{(\epsilon_{\mathcal T}(s_i))}\epsilon_{\mathcal T}(X)
\]
is an isomorphism. Then, for every topology \(\mathcal T'\) finer than \(\mathcal T\), the morphism
\[
\coprod_{i\in I}\epsilon_{\mathcal T'}(X_i)\xrightarrow{(\epsilon_{\mathcal T'}(s_i))}\epsilon_{\mathcal T'}(X)
\]
is an isomorphism.

\item Let \(\mathcal T\) be a \(\U\)-topology on \(C\). The following properties (i) and (ii) are equivalent:
\begin{description}[leftmargin=2.7em]
\item[(i)] The following three conditions hold:
\begin{description}[leftmargin=2.7em]
\item[(a)] The family \((s_i:X_i\to X)_{i\in I}\) is covering for \(\mathcal T\).
\item[(b)] For every \(i\in I\), the diagonal morphism of presheaves
\[
\Delta_i:X_i\hookrightarrow X_i\times_XX_i
\]
is transformed by the associated-sheaf functor for \(\mathcal T\) into an isomorphism; this is automatically the case if the \(s_i\) are monomorphisms.
\item[(c)] For every pair \(i\ne j\), the presheaf \(X_i\times_XX_j\) is transformed by the associated-sheaf functor for \(\mathcal T\) into the initial object of \(\widetilde C_{\mathcal T}\).
\end{description}
\item[(ii)] \(\epsilon_{\mathcal T}(X)\) is the sum of the \(\epsilon_{\mathcal T}(X_i)\).
\end{description}
\end{enumerate}
\end{statement}

\begin{prooftext}
For (1), a sheaf for \(\mathcal T'\) is a sheaf for \(\mathcal T\). Hence, for every sheaf \(F\) for \(\mathcal T'\), the morphism
\[
\Hom_{\widehat C}(X,F)\longrightarrow \prod_{i\in I}\Hom_{\widehat C}(X_i,F)
\]
is an isomorphism. This implies the assertion.

For (2), by definition property (ii) holds if and only if the morphism of presheaves
\[
\Phi=(h(s_i))_{i\in I}:\coprod_i h(X_i)\longrightarrow h(X)
\]
is transformed by the associated-sheaf functor into an isomorphism; equivalently, by Proposition~4.2, \(\underline a(\Phi)\) is both an epimorphism and a monomorphism of sheaves. By Theorem~4.4, \(\underline a(\Phi)\) is an epimorphism if and only if (a) holds. The diagonal
\[
\Delta:\coprod_i h(X_i)\longrightarrow
\left(\coprod_i h(X_i)\right)\times_{h(X)}\left(\coprod_i h(X_i)\right)
\]
is the direct sum of morphisms \(\Delta_{i,j}\), indexed by \(I\times I\), defined as follows:
\begin{itemize}
\item if \(i=j\), \(\Delta_{i,i}\) is the diagonal
\[
h(X_i)\to h(X_i)\times_{h(X)}h(X_i);
\]
\item if \(i\ne j\), \(\Delta_{i,j}\) is the morphism
\[
\phi_{\widehat C}\to h(X_i)\times_{h(X)}h(X_j),
\]
where \(\phi_{\widehat C}\) denotes the initial object of \(\widehat C\).
\end{itemize}
The morphism \(\underline a(\Phi)\) is a monomorphism if and only if \(\underline a(\Delta)\) is an isomorphism; and by Theorem~4.1, \(\underline a(\Delta)\) is an isomorphism if and only if each \(\underline a(\Delta_{i,j})\) is an isomorphism, i.e. if and only if (b) and (c) hold. \qedtext
\end{prooftext}

\begin{statement}{Corollary 4.6.1}
Let \(C\) be a \(\U\)-category and let \(X\) be an object of \(C\).
\begin{enumerate}[label=\arabic*)]
\item Let \(\mathcal T\) be a \(\U\)-topology on \(C\). The object \(X\) of \(C\) is transformed by the associated-sheaf functor for \(\mathcal T\) into the initial object of \(\widetilde C_{\mathcal T}\) if and only if the empty sieve covers \(X\).
\item When \(X\) is a strict initial object of \(C\), the sheaf associated to \(X\), for every \(\U\)-topology finer than the canonical topology, is an initial object.
\end{enumerate}
\end{statement}

\begin{prooftext}
For (1), take the indexing set \(I\) in Proposition~4.6(2) to be empty. For (2), by (1) and Proposition~4.6(1), it is enough to prove that the empty sieve covers \(X\) for the canonical topology; equivalently, by 2.6, it is enough to prove that every object \(Y\to X\) over \(X\) is initial, which follows from the definition of strict initial object. \qedtext
\end{prooftext}

\begin{statement}{Corollary 4.6.2}
Let \(C\) be a \(\U\)-site, and let \((s_i:X_i\to X)_{i\in I}\) be a family of base-changeable morphisms in \(C\), all with target \(X\), satisfying:
\begin{description}[leftmargin=2.5em]
\item[\(\alpha\)] the family \(s_i\) is covering;
\item[\(\beta\)] each \(s_i:X_i\to X\) is a monomorphism;
\item[\(\gamma\)] for each pair \(i\ne j\), the object \(X_i\times_XX_j\) is covered by the empty sieve, i.e. for every sheaf \(F\) on \(C\), the set \(F(X_i\times_XX_j)\) is a singleton.
\end{description}
Then \(\epsilon_C(X)\) is the sum of the \(\epsilon_C(X_i)\), \(i\in I\).
\end{statement}

\begin{prooftext}
This follows immediately from Proposition~4.6(2) and Corollary~4.6.1(1). \qedtext
\end{prooftext}

\begin{statement}{Corollary 4.6.3}
Let \(C\) be a \(\U\)-category and let \((s_i:X_i\to X)_{i\in I}\) be a family of base-changeable morphisms with the same target. Suppose that the canonical topology of \(C\) is a \(\U\)-topology. The following conditions are equivalent:
\begin{enumerate}[label=(\roman*)]
\item There exists a \(\U\)-topology \(\mathcal T\) on \(C\), less fine than the canonical topology, such that \(\epsilon_{\mathcal T}(X)\) is the sum of the \(\epsilon_{\mathcal T}(X_i)\), \(i\in I\).
\item The object \(X\) is the disjoint universal sum, in \(C\), of the \(X_i\) (4.5).
\end{enumerate}
\end{statement}

\begin{prooftext}
(ii) \(\Rightarrow\) (i). Since \(X\) is the universal sum of the \(X_i\), the family \(s_i\) is covering for the canonical topology of \(C\) (2.6). Thus condition \(\alpha\) of Corollary~4.6.2 holds for the canonical topology. Condition \(\beta\) is immediate, and condition \(\gamma\) follows from Proposition~4.6(1),(2). Hence (i) holds.

(i) \(\Rightarrow\) (ii). Let \(\mathcal T\) be a topology on \(C\), less fine than the canonical topology, such that \(\epsilon_{\mathcal T}(X)\) is the sum of the \(\epsilon_{\mathcal T}(X_i)\). For every sheaf \(F\) for \(\mathcal T\), the canonical map
\[
\Hom(X,F)\to \prod_i\Hom(X_i,F)
\]
is a bijection. Since every representable presheaf is a sheaf, it follows at once that \(X\) is the sum in \(C\) of the \(X_i\). Applying the same argument when \(I\) is empty, we see that if an object of \(C\) is transformed into the initial object of \(\widetilde C_{\mathcal T}\), then it is an initial object of \(C\). The functor \(\epsilon_{\mathcal T}:C\to\widetilde C_{\mathcal T}\) is fully faithful and commutes with finite projective limits. Hence condition (b) of Corollary~4.6.2 implies that the \(s_i\) are monomorphisms in \(C\), and condition (c), by what has just been proved, implies that \(X_i\times_XX_j\) is an initial object of \(C\) for \(i\ne j\). Thus \(X\) is a disjoint sum of the \(X_i\). Since \(\epsilon_{\mathcal T}\) commutes with fiber products, this property is stable under every base change \(Y\to X\), and therefore \(X\) is the disjoint universal sum of the \(X_i\). \qedtext
\end{prooftext}

\begin{statement}{Corollary 4.6.4}
Let \(C\) be a \(\U\)-category and let \(X\) be an object of \(C\). Suppose that the canonical topology on \(C\) is a \(\U\)-topology. The following properties are equivalent:
\begin{enumerate}[label=(\roman*)]
\item There exists on \(C\) a topology less fine than the canonical topology such that \(\epsilon_{\mathcal T}(X)\) is an initial object of \(\widetilde C_{\mathcal T}\).
\item \(X\) is a strict initial object of \(C\).
\end{enumerate}
\end{statement}

\begin{prooftext}
Take \(I\) to be empty in Corollary~4.6.3. \qedtext
\end{prooftext}

\begin{statement}{Proposition 4.7}
Let \(C\) be a \(\U\)-category, let \(\widetilde C\) be the category of sheaves on \(C\) for the canonical topology, let
\[
\epsilon_C:C\to\widetilde C
\]
be the canonical functor (4.4.0), and let \(R\rightrightarrows X\) be an equivalence relation in \(C\) admitting a cokernel \(Y\) in \(C\). Let \(\pi:X\to Y\) be the canonical morphism and suppose it is base-changeable. Consider:
\begin{enumerate}[label=(\roman*)]
\item \(\epsilon_C(Y)\) is the quotient of the equivalence relation
\[
\epsilon_C(R)\rightrightarrows \epsilon_C(X).
\]
\item The equivalence relation \(R\rightrightarrows X\) is effective and universal (cf. no.~7).
\end{enumerate}
Then (ii) implies (i). If \(C\) has a \(\U\)-topology less fine than the canonical topology, then (i) implies (ii).
\end{statement}

\begin{prooftext}
(ii) \(\Rightarrow\) (i). The canonical morphism \(R\to X\times_YX\) is an isomorphism. Let \(S\to Y\) be the sieve generated by \(\pi:X\to Y\). For every presheaf \(F\), the diagram
\[
\Hom(S,F)\longrightarrow \Hom(X,F)\rightrightarrows \Hom(R,F)
\]
is exact (Expos\'e~I, 2.12). It is therefore enough to show that, for every sheaf \(F\) for the canonical topology, the map
\[
\Hom(Y,F)\longrightarrow \Hom(S,F)
\]
is a bijection; equivalently, that \(S\) is covering for the canonical topology, or that the sieve generated by \(\pi:X\to Y\) is covering. This follows from Definition~2.5.

(i) \(\Rightarrow\) (ii). The functor \(\epsilon_C:C\to\widetilde C\) commutes with finite projective limits and is fully faithful. Since the canonical morphism
\[
\epsilon_C(R)\to \epsilon_C(X)\times_{\epsilon_C(Y)}\epsilon_C(X)
\]
is an isomorphism (4.3), the canonical morphism \(R\to X\times_YX\) is an isomorphism. The morphism \(\epsilon_C(X)\to\epsilon_C(Y)\) is an epimorphism; hence, by Theorem~4.4, \(\pi:X\to Y\) is a covering morphism for the canonical topology. \qedtext
\end{prooftext}

\begin{statement}{Proposition 4.8}
Let \(C\) be a \(\U\)-site. The category of sheaves on \(C\) has the following properties:
\begin{enumerate}[label=\alph*)]
\item finite projective limits are representable;
\item direct sums indexed by an element of \(\U\) are representable, and are disjoint and universal (4.5);
\item equivalence relations are effective and universal.
\end{enumerate}
\end{statement}

These facts were proved in Theorem~4.1(2),(3), Proposition~4.3(3), and Example~4.5.1.

These properties have been singled out because they will later make it possible to characterize the \(\U\)-categories equivalent to categories of sheaves on categories belonging to \(\U\) (Expos\'e~IV, 1.2).

\begin{remarktext}{Remark}
Recall that property (a) is equivalent to the following: there exists a final object, and fiber products are representable (Expos\'e~I, 2.3.1).
\end{remarktext}

\subsection*{4.9}
Let \(C\) be a site whose underlying category is a \(\U\)-category. Then the category of \(\U\)-sheaves on \(C\) satisfies the conditions considered in Expos\'e~I, 7.3, either (i) or (ii); hence, by loc. cit., the various variants considered in Expos\'e~I, 7.1 for the notion of a \emph{generating family} in \(C\) coincide. With this understood:

\begin{statement}{Proposition 4.10}
With the preceding notation, consider the canonical functor
\[
\epsilon:C\to\widetilde C
\]
(4.4.0), and let \(G\subset\Ob(C)\) be a topologically generating family in \(C\) (3.0.1). Then the family \((\epsilon(X))_{X\in G}\) of objects of \(\widetilde C\) is a generating family. In particular, the family \((\epsilon(X))_{X\in\Ob(C)}\) is generating.
\end{statement}

We must prove that every morphism \(u:F\to F'\) in \(\widetilde C\) such that
\[
\Hom(\epsilon(X),F)\longrightarrow \Hom(\epsilon(X),F')
\tag{4.10.1}
\]
is a bijection for every \(X\in G\), is an isomorphism. The preceding map is identified with
\[
u(X):F(X)\longrightarrow F'(X),
\tag{4.10.2}
\]
and it remains to show that, if this map is bijective for every \(X\in G\), then it is bijective for every \(X\in\Ob(C)\). Let \((X_i\to X)_{i\in I}\) be a covering family of \(X\) by objects of \(G\). For every pair \((i,j)\) in \(I\times I\), consider the set of all morphisms
\[
X_{ijk}\to X_i\times_XX_j
\]
in \(\widehat C\), with \(X_{ijk}\in G\), where \(k\) ranges over an index set \(I_{i,j}\). We obtain a morphism of exact diagrams of sets
\[
\begin{array}{ccccc}
F(X)&\to&\displaystyle\prod_iF(X_i)&\rightrightarrows&\displaystyle\prod_{i,j,k}F(X_{ijk})\\
\downarrow&&\downarrow\wr&&\downarrow\wr\\
F'(X)&\to&\displaystyle\prod_iF'(X_i)&\rightrightarrows&\displaystyle\prod_{i,j,k}F'(X_{ijk}),
\end{array}
\]
where the last two vertical arrows are bijections by hypothesis. Hence the first vertical arrow is a bijection as well. \qedtext

\begin{statement}{Corollary 4.11}
Suppose that \(C\) is a \(\U\)-site (3.0.2). Then \(\widetilde C\) is a \(\U\)-category and has a \(\U\)-small generating family.
\end{statement}

Indeed, by hypothesis one may take in Proposition~4.10 a small generating family \(G\), which proves the existence of a small generating family. On the other hand, if \(F,F'\in\Ob(\widetilde C)\), a homomorphism \(F\to F'\) is known once the maps (4.10.1), or equivalently (4.10.2), are known for every \(X\in G\) (Expos\'e~I, 7.1.1). Hence the map
\[
\Hom(F,F')\longrightarrow \prod_{X\in G}\Hom(F(X),F'(X))
\]
is injective. Since the right-hand side is \(\U\)-small, the left-hand side is \(\U\)-small, which proves that \(\widetilde C\) is a \(\U\)-category.

\begin{statement}{Corollary 4.12}
Let \(C\) be a \(\U\)-site. Then for every object \(X\) of \(\widetilde C\), both the set of subobjects of \(X\) and the set of quotient objects of \(X\) are \(\U\)-small.
\end{statement}

This follows from Expos\'e~I, 7.4 and 7.5, respectively, which apply by Corollary~4.11.

\section{Extension of a topology from \(C\) to \(\widehat C\)}

\begin{statement}{Proposition 5.1}
Let \(C\) be a \(\U\)-site and let \(f:H\to K\) be a morphism of \(\widehat C\). The following conditions are equivalent:
\begin{description}[leftmargin=2.8em]
\item[(i)] For every \(X\to K\), with \(X\in\Ob(C)\), the corresponding morphism
\[
H\times_KX\to X
\]
has as image a covering sieve of \(X\).
\item[(ii)] The morphism of associated sheaves
\[
\underline a(f):\underline aH\to\underline aK
\]
is an epimorphism in \(\widetilde C\).
\item[(ii bis)] For every sheaf \(F\) on \(C\), the map
\[
\Hom(K,F)\to \Hom(H,F)
\]
deduced from \(f\) is injective.
\end{description}
\end{statement}

\begin{prooftext}
It is clear that (ii) is equivalent to (ii bis).

We prove (i) \(\Rightarrow\) (ii). By Theorem~4.4, (i) means that
\[
\underline a(H\times_KX)\to \underline a(X)
\]
is an epimorphism. Since
\[
\underline a(H\times_KX)\simeq
\underline a(H)\times_{\underline a(K)}\underline a(X),
\]
because \(\underline a\) commutes with finite projective limits (4.1), the morphism in question is obtained from
\[
\underline a(f):\underline a(H)\to \underline a(K)
\]
by base change. As epimorphisms in \(\widetilde C\) are universal, this shows that (ii) implies (i) after base change; conversely, since the family of all \(X\to K\) is epimorphic and the corresponding family \(\underline a(X)\to\underline a(K)\) is epimorphic in \(\widetilde C\) (4.1), to verify that \(\underline a(f)\) is an epimorphism it is enough to check it after every such base change. This proves (i) \(\Rightarrow\) (ii), and the equivalence follows. \qedtext
\end{prooftext}

\begin{statement}{Definition 5.2}
\begin{enumerate}[label=\arabic*)]
\item A morphism \(f:H\to K\) satisfying the equivalent conditions of Proposition~5.1 is called a \emph{covering morphism}. A family of morphisms \((f_i:H_i\to K)_{i\in I}\) with common target is called \emph{covering} if the corresponding morphism
\[
f:\coprod_iH_i\to K
\]
is covering.
\item A morphism \(f:H\to K\) is called \emph{bicov\-ering} if it is covering and if its diagonal morphism
\[
H\to H\times_KH
\]
is covering. A family \((f_i:H_i\to K)_{i\in I}\) with common target is called \emph{bicov\-ering} if the corresponding morphism \(\coprod_iH_i\to K\) is bicov\-ering.
\end{enumerate}
\end{statement}

By condition (i) of Proposition~5.1, to say that a family \((f_i:H_i\to K)_{i\in I}\) is covering means that, for every morphism \(X\to K\), with \(X\in\Ob(C)\), the family
\[
H_i\times_KX\to X,\qquad i\in I,
\]
has as image a covering sieve of \(X\); or equivalently, by condition (ii), that the family of morphisms
\[
\underline a(f_i):\underline aH_i\to \underline aK
\]
in \(\widetilde C\) is epimorphic, taking into account that \(\underline a\) commutes with direct sums (4.1).\footnote{The property of being covering (respectively bicov\-ering), for a morphism or family of morphisms of \(\widehat C\), is stable under base change.}

\begin{statement}{Proposition 5.3}
Let \(C\) be a \(\U\)-site and let \(f:H\to K\) be a morphism of \(\widehat C\). The following conditions are equivalent:
\begin{description}[leftmargin=2.9em]
\item[(i)] \(f\) is bicov\-ering (5.2).
\item[(i bis)] \(f\) is covering, and for every object \(X\) of \(C\) and every pair of morphisms \(u,v:X\rightrightarrows H\) such that \(fu=fv\), the equalizer of \((u,v)\) is a covering sieve of \(X\).
\item[(ii)] \(\underline a(f):\underline aH\to\underline aK\) is an isomorphism in \(\widetilde C\).
\item[(ii bis)] For every sheaf \(F\) on \(C\), the map
\[
\Hom(K,F)\to \Hom(H,F)
\]
is a bijection.
\end{description}
\end{statement}

\begin{prooftext}
The equivalence (i) \(\Leftrightarrow\) (i bis) follows from condition (i) of Proposition~5.1, applied to the diagonal morphism \(H\to H\times_KH\). The equivalence (ii) \(\Leftrightarrow\) (ii bis) is trivial.

(i) \(\Rightarrow\) (ii). The morphism
\[
\underline a(f):\underline aH\to \underline aK
\]
is an epimorphism by Proposition~5.1. Since \(\underline a\) commutes with fiber products (4.1), the diagonal
\[
\underline aH\to \underline aH\times_{\underline aK}\underline aH
\]
is an epimorphism by Proposition~5.1. A diagonal morphism is always a monomorphism; hence it is an isomorphism by Proposition~4.2. Thus \(\underline a(f)\) is a monomorphism. It is therefore an isomorphism by Proposition~4.2.

(ii) \(\Rightarrow\) (i). Since \(\underline a\) commutes with fiber products, both \(f\) and the diagonal \(H\to H\times_KH\) are transformed by \(\underline a\) into isomorphisms. In particular they are transformed into epimorphisms. Hence they are covering. \qedtext
\end{prooftext}

\subsection*{5.3.1}
It follows from Proposition~5.3 and from the fact that \(\underline a\) commutes with direct sums that a family
\[
(f_i:H_i\to K)_{i\in I}
\]
of morphisms of \(\widehat C\) is bicov\-ering if and only if the \(f_i\) induce, on associated sheaves, an isomorphism
\[
\coprod_i\underline aH_i\xrightarrow{\sim}\underline aK,
\]
or equivalently if and only if, for every sheaf \(F\), the map
\[
\Hom(K,F)\longrightarrow \prod_i\Hom(H_i,F)
\]
is bijective.

\begin{statement}{Proposition 5.4}
Let \(C\) be a \(\U\)-site. There exists on \(\widehat C\) a topology, clearly unique, such that a family \(H_i\to K\) of arrows of \(\widehat C\) with common target is covering for this topology if and only if it is covering in the sense of Definition~5.2. It is also the least fine topology on \(\widehat C\) among the topologies \(T\) satisfying:
\begin{enumerate}[label=\alph*)]
\item \(T\) is finer than the canonical topology of \(\widehat C\), i.e. every epimorphic family in \(\widehat C\) is covering for \(T\);
\item every covering family in \(C\) is covering in \(\widehat C\).
\end{enumerate}
\end{statement}

\begin{prooftext}
We only give indications. It is left to the reader to show that the families covering in the sense of Definition~5.2 are the covering families of a topology \(T_C\) on \(\widehat C\); one uses Theorem~4.1. The covering families for the canonical topology on \(\widehat C\) are the epimorphic families of \(\widehat C\) (2.6 and Expos\'e~I, 3.1). Since \(\underline a\) commutes with inductive limits (4.1), the topology \(T_C\) is finer than the canonical topology of \(\widehat C\). Moreover, the covering families of \(C\) are covering families for \(T_C\) (5.1).

Let \(T'\) denote the least fine topology on \(\widehat C\) having properties (a) and (b). Then \(T_C\) is finer than \(T'\). Let
\[
(f_i:H_i\to K)_{i\in I}
\]
be a \(T_C\)-covering family. We show that it is a covering family for \(T'\). Let \(s_i:H_i\to H=\coprod_iH_i\) be the canonical monomorphisms, and let \(f=(f_i):H\to K\) be the morphism defined by the \(f_i\). The family \(s_i\) is covering for \(T'\). Thus, by axiom (T2), to prove that \((f_i)\) is covering for \(T'\), it is enough to prove that the morphism \(f:H\to K\) is covering for \(T'\).

There exists an epimorphic family in \(\widehat C\)
\[
u_\lambda:X_\lambda\to K,\qquad \lambda\in\Lambda,\quad X_\lambda\in\Ob(C)
\]
(Expos\'e~I, 3.4). By axiom (T2), to prove that \(f:H\to K\) is covering for \(T'\), it is enough to prove that, for every \(\lambda\), the projection
\[
H\times_KX_\lambda\to X_\lambda
\]
is covering for \(T'\). Let
\[
v_j:Y_j\to H\times_KX_\lambda,\qquad j\in J,\quad Y_j\in\Ob(C),
\]
be an epimorphic family in \(\widehat C\). The family \((\operatorname{pr}_2\circ v_j)_{j\in J}\) is covering for \(T_C\). Hence it is a covering family in \(C\) by Proposition~5.1, and therefore is covering for \(T'\). Consequently the sieve generated by \(H\times_KX_\lambda\to X_\lambda\) contains a \(T'\)-covering sieve; it is therefore covering for \(T'\). This proves the proposition. \qedtext
\end{prooftext}

\begin{remarktext}{Remark 5.4.1}
The proof of Proposition~5.4 actually shows that every topology \(T'\) on \(\widehat C\) which is finer than the canonical topology of \(\widehat C\), is the least fine of the topologies \(T\) on \(\widehat C\) satisfying:
\begin{enumerate}[label=\alph*)]
\item \(T\) is finer than the canonical topology of \(\widehat C\);
\item every \(T'\)-covering family of the form
\[
u_i:X_i\to X,\qquad i\in I,
\]
where \(X\) and the \(X_i\) are objects of \(C\), is covering for \(T\).
\end{enumerate}
In particular, every topology \(T'\) on \(\widehat C\) finer than the canonical topology is uniquely determined by the families \(u_i:X_i\to X\), \(i\in I\), with \(X_i\) and \(X\) objects of \(C\), which are covering for \(T'\).
\end{remarktext}

\begin{remarktext}{Remark 5.4.2}
One can easily show that, for every topology \(T'\) on \(\widehat C\) finer than the canonical topology, the set of families of morphisms
\[
(X_i\to X)_{i\in I}
\]
with common target in \(C\), which are covering for \(T'\), is the set of covering families of a topology on \(C\). Thus Proposition~5.4 and Remark~5.4.1 establish a one-to-one correspondence between topologies on \(C\) and topologies on \(\widehat C\) finer than the canonical topology.
\end{remarktext}

\subsection*{5.5.0}
Let \(C\) be a small category. Let \(\mathcal C\) be the set of strictly full subcategories of \(\widehat C\) (strictly full means that every object isomorphic to an object of the subcategory is itself an object of the subcategory) whose inclusion functor admits a left adjoint commuting with finite projective limits. Also let \(\mathcal T\) denote the set of topologies on \(C\). Theorem~3.4 defines a map
\[
\Phi:\mathcal T\longrightarrow \mathcal C.
\]

We now define a map in the other direction. For this, to every element
\[
i':C'\hookrightarrow \widehat C,\qquad \underline a'\dashv i',
\]
with \(\underline a'\) left adjoint to \(i'\), we associate a topology \(T_e\) on \(C\). For every object \(X\) of \(C\), \(J_e(X)\) is defined to be the set of subobjects of \(X\) whose inclusion morphism is transformed by \(\underline a'\) into an isomorphism. Using the hypotheses on \(\underline a'\), one immediately checks that this defines a topology \(T_e\) on \(C\). We have therefore defined a map
\[
\Psi:\mathcal C\longrightarrow \mathcal T.
\]

We then have the following result, due to J. Giraud.

\begin{statement}{Theorem 5.5}
The map \(\Phi\) is a bijection, and \(\Psi\) is its inverse.
\end{statement}

\begin{prooftext}
The composite \(\Psi\circ\Phi\) is the identity. Indeed, this follows immediately from Proposition~5.1.

The composite \(\Phi\circ\Psi\) is the identity. Let
\[
i':C'\hookrightarrow \widehat C,\qquad \underline a'\dashv i',
\]
be an element of \(\mathcal C\), let \(T_e\) be the topology corresponding to it under \(\Psi\), and let \(C_e\) be the category of sheaves for \(T_e\). Using the definition of the topology \(T_e\) and the definition of bicov\-ering morphisms (5.2), one proves the equivalence of the following assertions:
\begin{enumerate}[label=(\roman*)]
\item a morphism \(u\) of \(\widehat C\) is bicov\-ering for the topology \(T_e\);
\item the morphism \(u\) of \(\widehat C\) is transformed by \(\underline a'\) into an isomorphism.
\end{enumerate}
It is clear that \(C'\) is a full subcategory of \(C_e\). It remains only to prove that every sheaf \(F\) for the topology \(T_e\) is an object of \(C'\). By the equivalence above, the morphism
\[
F\longrightarrow i'\underline a'(F)
\]
is bicov\-ering; since its source and target are sheaves, Proposition~5.3(ii) implies that it is an isomorphism. \qedtext
\end{prooftext}

\section{Sheaves with values in a category}

\subsection*{6.0}
Let \(C\) and \(D\) be two categories. A contravariant functor
\[
F:C^\circ\to D
\]
is called a \emph{presheaf on \(C\) with values in \(D\)}.

\begin{statement}{Definition 6.1}
Let \(C\) be a site and let \(D\) be a category. A presheaf \(F:C^\circ\to D\) is called a \emph{sheaf on \(C\) with values in \(D\)}, or more briefly a \emph{\(D\)-valued sheaf}, if for every object \(S\) of \(D\), the presheaf of sets
\[
X\longmapsto \Hom_D(S,F(X)),\qquad X\in\Ob(C),
\]
is a sheaf. The full subcategory of \(\operatorname{Hom}(C^\circ,D)\) consisting of sheaves on \(C\) with values in \(D\) will be denoted
\[
\operatorname{Hom}^{\sim}(C^\circ,D).
\]
\end{statement}

\begin{remarktext}{Remarks 6.2}
\begin{enumerate}[label=\arabic*)]
\item Condition 6.1 means that for every object \(X\) of \(C\), every covering sieve \(R\) of \(X\), and every object \(S\) of \(D\), one has an isomorphism
\[
\Hom_D(S,F(X))\xrightarrow{\sim}
\varprojlim_{Y\to R}\Hom_D(S,F(Y))
\]
(Expos\'e~I, 3.5 and 2.1). When \(D\) is the category of \(\U\)-sets, this recovers the definition of sheaves of sets.

\item Let \(D'\) be a \(\U\)-category and let \(G:D\to D'\) be a functor commuting with projective limits. By composition, \(G\) sends sheaves with values in \(D\) to sheaves with values in \(D'\).
\end{enumerate}
\end{remarktext}

\subsection*{6.3.0}
Let \(\gamma\) be a species of algebraic structure defined by finite projective limits (Expos\'e~I, 2.9), and let \(\U\)-\(\gamma\) be the category of \(\gamma\)-sets which belong to \(\U\). Denote by
\[
\operatorname{For}:\U\text{-}\gamma\to \U\text{-}\Set
\]
the forgetful functor to the underlying set. The functor \(\operatorname{For}\) commutes with projective limits and therefore, by the preceding remark, defines by composition functors
\[
\operatorname{For}^{\widehat{\phantom{,}}}:
\operatorname{Hom}(C^\circ,\U\text{-}\gamma)\longrightarrow \widehat C,
\]
and
\[
\operatorname{For}^{\widetilde{\phantom{,}}}:
\operatorname{Hom}^{\sim}(C^\circ,\U\text{-}\gamma)\longrightarrow \widetilde C.
\]
The latter is called the \emph{underlying sheaf of sets} functor. In fact, it factors canonically through the category \(\gamma\text{-}\widetilde C\) of \(\gamma\)-objects of \(\widetilde C\). Denoting the resulting functor again by
\[
\widetilde{\operatorname{For}}:
\operatorname{Hom}^{\sim}(C^\circ,\U\text{-}\gamma)\longrightarrow \gamma\text{-}\widetilde C,
\]
one may state:

\begin{statement}{Proposition 6.3.1}
The functor \(\operatorname{For}^{\widetilde{\phantom{,}}}\) establishes equivalences of categories between:
\begin{enumerate}[label=\arabic*)]
\item sheaves on \(C\) with values in \(\U\)-\(\gamma\);
\item \(\gamma\)-objects of the category of sheaves of sets;
\item \(\gamma\)-objects of the category of presheaves of sets whose underlying presheaf is a sheaf.
\end{enumerate}
\end{statement}

\begin{prooftext}
The proof uses essentially Expos\'e~I, 3.2 and the fact that a finite projective limit of sheaves, computed in the category of presheaves, is again a sheaf (4.1). \qedtext
\end{prooftext}

\subsection*{6.3.2}
The preceding proposition justifies the abuse of language by which one identifies a sheaf with values in \(\U\)-\(\gamma\) with the corresponding \(\gamma\)-object in \(\widetilde C\). We shall henceforth use this abuse of language systematically.

We shall use the classical terminology: \emph{sheaves of groups}, \emph{sheaves of commutative groups}, which we shall most often call \emph{abelian sheaves}, \emph{sheaves of rings}, \emph{sheaves of \(A\)-modules}, and so on.

\subsection*{6.3.3}
We denote by
\[
\widetilde C_{\gamma}\quad(\text{respectively } \widehat C_{\gamma})
\]
the category of \(\gamma\)-objects of \(\widetilde C\) (respectively of \(\widehat C\)). When \(\gamma\) is the species of structure ``\(A\)-module'', we also write \(\widetilde C_A\), or simply \(\widetilde C_{\mathrm{ab}}\) when \(A=\mathbb Z\).

Assume that \(C\) is a \(\U\)-site. The associated-sheaf functor is left exact (4.1), and therefore:

\begin{statement}{Proposition 6.4}
The inclusion functor
\[
i_{\gamma}:\widetilde C_{\gamma}\to \widehat C_{\gamma}
\]
has a left adjoint
\[
a_{\gamma}:\widehat C_{\gamma}\to\widetilde C_{\gamma}
\]
which is left exact; in other words, there is an isomorphism
\[
\Hom_{\widetilde C_{\gamma}}(a_{\gamma}(\blank),\blank)
\xrightarrow{\sim}
\Hom_{\widehat C_{\gamma}}(\blank,i_{\gamma}(\blank)).
\]

Let \(X\) be an object of \(\widehat C_{\gamma}\). The underlying sheaf of sets of \(a_{\gamma}(X)\) is canonically isomorphic to the sheaf of sets associated to the underlying presheaf of sets of \(X\). The morphism of underlying presheaves of sets associated to the adjunction morphism
\[
\operatorname{id}\to i_{\gamma}a_{\gamma}
\]
is identified with the adjunction morphism applied to the underlying presheaf of sets.
\end{statement}

\begin{statement}{Proposition 6.5}
Suppose that the forgetful functor
\[
\operatorname{For}:\gamma\text{-}\U\to \U\text{-}\Set
\]
has a left adjoint\footnote{One proves that such a left adjoint always exists in the usual examples: free group, free abelian group, free \(A\)-module, and so on.}
\[
\operatorname{Lib}:\U\text{-}\Set\to\gamma\text{-}\U,
\]
i.e.
\[
\Hom_{\U\text{-}\Set}(\blank,\operatorname{For}(\blank))
\xrightarrow{\sim}
\Hom_{\gamma\text{-}\U}(\operatorname{Lib}(\blank),\blank).
\]
Then the functor
\[
\operatorname{For}^{\widetilde{\phantom{,}}}:\widetilde C_{\gamma}\to \widetilde C
\]
(respectively
\[
\operatorname{For}^{\widehat{\phantom{,}}}:\widehat C_{\gamma}\to \widehat C)
\]
has a left adjoint
\[
\operatorname{Lib}^{\widetilde{\phantom{,}}}
\quad(\text{respectively } \operatorname{Lib}^{\widehat{\phantom{,}}}),
\]
and there is a canonical isomorphism
\[
\operatorname{Lib}^{\widetilde{\phantom{,}}}
=
a_{\gamma}\operatorname{Lib}^{\widehat{\phantom{,}}}i.
\]
\end{statement}

\begin{prooftext}
The proof is formal and is left to the reader. \qedtext
\end{prooftext}

\begin{statement}{Corollary 6.6}
Let \((X_i)_{i\in I}\) be a generating family of \(\widetilde C\) (4.9). Then the family
\[
\operatorname{Lib}^{\widetilde{\phantom{,}}}(X_i)
\]
is a generating family of \(\widetilde C_{\gamma}\).
\end{statement}

\begin{prooftext}
The proof is formal once one has observed that the functor \(\operatorname{For}^{\widetilde{\phantom{,}}}\) commutes with projective limits and is conservative: \(\operatorname{For}^{\widetilde{\phantom{,}}}(u)\) is an isomorphism if and only if \(u\) is an isomorphism. \qedtext
\end{prooftext}

\begin{statement}{Proposition 6.7}
Let \(A\) be a \(\U\)-sheaf of rings, or a small ring. Let
\[
\widetilde C_A\quad(\text{respectively } \widehat C_A)
\]
be the category of sheaves (respectively presheaves) of unital \(A\)-modules (6.3.3) on a \(\U\)-site \(C\). Then \(\widetilde C_A\) is a \(\U\)-abelian category satisfying the axioms (AB5), ``existence of left exact filtered inductive limits'', and (AB3)\({}^{\ast}\), ``existence of infinite products'', of \textsc{Tohoku}. It has a family of generators indexed by an element of \(\U\).
\end{statement}

\begin{prooftext}
It is clear that \(\widehat C_A\) is an abelian category satisfying axioms (AB3), (AB4), and (AB5) (Expos\'e~I, 2.8 and 3.3). From Proposition~6.4 it follows that \(\widetilde C_A\) is an additive category in which kernels and cokernels are representable.

More explicitly, let \(i_A\) and \(a_A\) be the inclusion and associated-sheaf functors, and let \(u:X\to Y\) be a morphism of \(\widetilde C_A\). The functor \(a_A\) is left exact and commutes with inductive limits. The functor \(i_A\) commutes with projective limits. Hence the functor
\[
i_Aa_A:\widehat C_A\to \widehat C_A
\]
is additive and left exact. We get canonical isomorphisms
\[
i_A(\Ker(u))\xrightarrow{\sim}\Ker(i_A(u)),
\tag{*}
\]
and
\[
\coker(u)\xrightarrow{\sim}a_A\,\coker(i_A(u)).
\tag{**}
\]
It follows that there is a canonical isomorphism
\[
\coim(u)\xrightarrow{\sim}a_A\,\coim(i_A(u)).
\tag{***}
\]

We prove that the canonical morphism \(\coim(u)\to\im(u)\) is an isomorphism. By \((*)\), it is enough to prove exactness of the sequence
\[
0\longrightarrow i_A(\coim(u))\longrightarrow i_A(Y)\longrightarrow i_A(\coker(u)).
\]
Using the isomorphisms \((**)\) and \((***)\), and observing that \(a_Ai_A(Y)\) is isomorphic to \(Y\), this sequence is identified with
\[
0\longrightarrow i_Aa_A(\coim(i_A(u)))\longrightarrow i_Aa_Ai_A(Y)\longrightarrow i_Aa_A(\coker(i_A(u))),
\]
which is simply the image under the left exact functor \(i_Aa_A\) of the exact sequence
\[
0\longrightarrow \coim(i_A(u))\longrightarrow i_A(Y)\longrightarrow \coker(i_A(u)).
\]
The category \(\widetilde C_A\) is therefore abelian.

The functor
\[
a_A:\widehat C_A\to\widetilde C_A
\]
is left exact and commutes with arbitrary inductive limits. Hence it is additive, exact, and commutes with inductive limits. Since \(\widehat C_A\) satisfies axiom (AB5), so does \(\widetilde C_A\). Finally, it is clear that in \(\widetilde C_A\), products indexed by an element of \(\U\) are representable; hence axiom (AB3)\({}^{\ast}\) holds. This completes the proof of the first assertion.

For every ring \(A\) which is an element of \(\U\), the functor
\[
\operatorname{For}:\U\text{-}A\text{-}\mathrm{Mod}\longrightarrow \U\text{-}\Set
\]
has a left adjoint \(X_A\), the free \(A\)-module generated by \(X\). It remains only to apply Proposition~4.10 and Corollary~6.6. \qedtext
\end{prooftext}

\begin{remarktext}{Notation 6.8}
Let \(H\) be a sheaf of sets. The sheaf of \(A\)-modules associated to the presheaf (cf. notation 6.5)
\[
X\longmapsto \operatorname{Lib}_A H(X),\qquad X\in\Ob(C),
\]
is denoted \(A_H\). When \(H=\underline a(X)\), the sheaf associated to the presheaf represented by \(X\), one sometimes simply writes \(A_X\), by abuse of notation. The proof of Proposition~6.7 shows that the family of sheaves of \(A\)-modules \(A_X\), \(X\in\Ob(C)\), is a generating family, indexed by an element of \(\U\), for the category of sheaves of \(A\)-modules.
\end{remarktext}

\begin{remarktext}{Remark 6.9}
By \textsc{Tohoku}, Proposition~6.7 shows that when \(A\) is an element of \(\U\), the category \(\widetilde C_A\) has enough injectives. It is known, moreover, that infinite products are not necessarily exact in \(\widetilde C_A\), and therefore that the category \(\widetilde C_A\) does not in general have enough projectives [\textsc{Roos}].
\end{remarktext}
