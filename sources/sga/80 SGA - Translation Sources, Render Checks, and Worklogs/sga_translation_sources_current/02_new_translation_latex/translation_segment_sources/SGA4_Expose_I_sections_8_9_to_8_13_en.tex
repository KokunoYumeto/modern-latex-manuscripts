\subsection{Exactness properties of \texorpdfstring{$\Ind(C)$}{Ind(C)}}

\begin{statement}{Proposition 8.9.1}
Let $C$ be a $\U$-category.
\begin{enumerate}[label=\alph*)]
\item The canonical functors $L$ (8.2.4.8) and $c$ (8.4.1)
\[
 C\xrightarrow{c}\Ind(C)\xrightarrow{L}\widehat C
\]
commute with projective limits.

\item Suppose that finite inductive limits are representable in $C$, and that $C$ is equivalent to a small category. Then small projective limits are representable in $\Ind(C)$.

\item If small projective limits (respectively finite projective limits) are representable in $C$, then the same is true in $\Ind(C)$. Let $J$ be a category which is finite and rigid, or discrete. If projective limits of type $J$ are representable in $C$, the same type of projective limits is representable in $\Ind(C)$.

\item Small filtered inductive limits in $\Ind(C)$, which are representable by 8.5.1, are left exact, that is, they commute with finite projective limits.
\end{enumerate}
\end{statement}

\begin{prooftext}
\begin{enumerate}[label=\alph*)]
\item The fact that $L$ commutes with projective limits follows from the fact that it is fully faithful and from the computation of projective limits in $\widehat C$ argument by argument. Thus, in order to verify that a projective system of morphisms $F\to F_i$ $(i\in I)$ in $\widehat C$ makes $F$ the projective limit of the $F_i$, it is enough to verify the analogous assertion for the projective systems of sets
\[
 \Hom(X,F)\longrightarrow \Hom(X,F_i)
\]
for all $X\in\Ob C$. But $C\subset\Ind(C)$.

The fact that $c$ commutes with projective limits follows formally from the fact that $L$ does so, together with the fact that the composite $L\circ c:C\to\widehat C$ does so.

\item In view of a), the assertion says that every projective limit of ind-representable presheaves is ind-representable. This follows immediately from criterion 8.3.3(v), since a projective limit of left exact functors is left exact; this is just the fact that projective limits commute with projective limits (2.5.0).

\item This follows formally from the analogous property of $\widehat C$ (3.3), together with the facts that $L$ is conservative, being fully faithful, and that it commutes with the limits of the type in question, by a) and 8.5.1.

\item It is well known (cf. 2.3) that representability of finite projective limits is equivalent to representability of projective limits of the special finite ordered types consisting of the empty type and the elementary fiber-product type. Representability of small projective limits is equivalent to representability of products and finite projective limits. Hence the second assertion made in c) implies the first.
\end{enumerate}

First suppose that $I$ is finite. Using 8.8.5 on the representability of functors $\varphi:J\to\Ind(C)$ in indexed form (8.5.4)
\[
 \Phi:J\times I\longrightarrow C,
\tag{8.9.1.1}
\]
the existence of $\varprojlim\varphi$ is a special case of the following more precise and more general result.
\end{prooftext}

\begin{statement}{Corollary 8.9.2}
Let $\varphi:J\to\Ind(C)$ be a functor given in indexed form $\Phi$ as in (8.9.1.1), where $J$ is a finite category. If, for every $i\in\Ob I$, the partial functor $j\mapsto\Phi(j,i)$ has a projective limit (respectively an inductive limit) representable in $C$, then $\varphi$ has a projective limit (respectively an inductive limit) representable in $\Ind(C)$, and there is a canonical isomorphism
\[
 \varprojlim\varphi\simeq \indlimI{i}\,\varprojlim_j\Phi(j,i),
\tag{8.9.2.1}
\]
respectively
\[
 \varinjlim\varphi\simeq \indlimI{i}\,\varinjlim_j\Phi(j,i),
\tag{8.9.2.2}
\]
where $\varprojlim_j$ (respectively $\varinjlim_j$) is computed in $C$.
\end{statement}

For the first formula, one uses only that in $\Ind(C)$ filtered inductive limits commute with $\varprojlim_j$, and that $c$ commutes with $\varprojlim_J$ by 8.9.1(d) and a):
\[
\begin{aligned}
\varprojlim\varphi
 &= \varprojlim_j \indlimI{i}\,\Phi(j,i) \\
 &\xleftarrow{\sim} \indlimI{i}\,{\varprojlim_j}^{\Ind(C)}\Phi(j,i)
 \simeq \indlimI{i}\,\varprojlim_j^{C}\Phi(j,i).
\end{aligned}
\]
The second is proved in the same way, using the commutation of inductive limits of type $I$ with inductive limits of type $J$ (2.5.0) and the fact that $c$ commutes with inductive limits of type $J$, proved in 8.9.4(a) below.

\emph{The case where $J$ is discrete.} This case is contained in the following more precise assertion.

\begin{statement}{Corollary 8.9.3}
Let $I_\alpha$ be essentially small filtered categories indexed by a small set $A$, and, for each $\alpha\in A$, let
\[
 X(\alpha)=(X(\alpha)_{i_\alpha})_{i_\alpha\in I_\alpha}
\]
be an ind-object of $C$ indexed by $I_\alpha$. Let $I$ be the product category of the $I_\alpha$, and suppose that, for every $\mathbf i=(i_\alpha)_{\alpha\in A}\in I$, the product
\[
 \prod_{\alpha\in A}X(\alpha)_{i_\alpha}
\]
is representable in $C$. Then the product $\prod_{\alpha\in A}X(\alpha)$ is representable in $\Ind(C)$, and is canonically isomorphic to the ind-object indexed by $I$ given by the formula
\[
 \prod_{\alpha\in A}\;\indlimI{i_\alpha\in I_\alpha}\,X(\alpha)_{i_\alpha}
 \simeq
 \indlimI{\mathbf i=(i_\alpha)_{\alpha\in A}\in I}\;
 \left(\prod_{\alpha\in A}X(\alpha)_{i_\alpha}\right),
\tag{8.9.3.1}
\]
where the product on the right is computed in $C$. Notice that $I$ is filtered and essentially small because the $I_\alpha$ are.
\end{statement}

Indeed, it is well known, and immediate after reducing to the category of sets, that the displayed formula is valid when the limits are computed in $\widehat C$. The conclusion then follows from the fact that $L$ commutes with the limits under consideration (8.9.1(a) and 8.5.1).

\begin{statement}{Remarks 8.9.4}
The proof just given for 8.9.2 and 8.9.3 shows, more generally, that if, for every $i\in\Ob I$, the limit $\varprojlim_j\Phi(j,i)$ (respectively $\varinjlim_j\Phi(j,i)$), computed in $\Ind(C)$, is representable, then the corresponding limit of $\varphi$ is representable. This and the argument in c) show that, for a given category $J$ arising from a finite ordered set or a discrete category, or more generally one which is $C$-admissible in the sense of 8.3.1, projective limits (respectively inductive limits) of type $J$ are representable in $\Ind(C)$ if and only if, for every functor $\varphi:J\to C$, the projective limit (respectively inductive limit) of $\varphi$, computed in $\Ind(C)$, is representable. In the non-parenthesized case this also means, by a), that every projective limit of type $J$ of representable presheaves on $C$ is ind-representable.
\end{statement}

\begin{statement}{Proposition 8.9.5}
Let $C$ be a $\U$-category.
\begin{enumerate}[label=\alph*)]
\item The canonical functor
\[
 c:C\longrightarrow\Ind(C)
\]
is right exact, hence exact in view of 8.9.1(a).

\item If finite inductive limits (respectively finite sums) are representable in $C$, then small inductive limits (respectively small sums) are representable in $\Ind(C)$. Let $J$ be a finite or discrete preordered set. If inductive limits of type $J$ are representable in $C$, then the same is true in $\Ind(C)$.
\end{enumerate}
\end{statement}

\begin{prooftext}
\begin{enumerate}[label=\alph*)]
\item Suppose that in $C$ one has $X\simeq\varinjlim_JX_j$, with $J$ finite and with $X$ and the $X_j$ in $C$. Then, for every ind-object $\mathcal Y=(Y_i)_{i\in I}$ of $C$,
\[
\begin{aligned}
\Hom(X,\mathcal Y)
&\simeq \varinjlim_i\Hom(X,Y_i)
 \simeq \varinjlim_i\varprojlim_j\Hom(X_j,Y_i)\\
&\simeq \varprojlim_j\varinjlim_i\Hom(X_j,Y_i)
 \simeq \varprojlim_j\Hom(X_j,\mathcal Y),
\end{aligned}
\]
where the middle isomorphism is 2.8. The desired conclusion follows by comparing the extreme terms.

\item It was already observed in 8.8.4 that the assertions follow from a) and 8.9.2, at least for finite inductive limits. To prove the conclusions in the infinite cases, one is reduced to the case of sums, which may be interpreted as a filtered inductive limit of finite sums; the conclusion follows from 8.5.1.
\end{enumerate}
\end{prooftext}

\begin{statement}{Remark 8.9.6}
We have already observed that the functor $c$ does not in general commute with filtered inductive limits, and therefore not with infinite sums either, in contrast with what happens for $L:\Ind(C)\to\widehat C$. Conversely, apart from commutation with filtered inductive limits (8.5.1), $L$ practically never has commutation properties with any other type of inductive limit: initial object, sum of two objects, or cokernels of double arrows. Thus, if $C$ has an initial object $\varnothing_C$, which is then an initial object of $\Ind(C)$ by a), $\varnothing_C$ is never an initial object of $\widehat C$, that is, never identical with the constant presheaf of value $\varnothing$, since $\Hom(\varnothing_C,\varnothing_C)\neq\varnothing$.
\end{statement}

\begin{statement}{Proposition 8.9.7}
Let
\[
 f:C\longrightarrow C'
\]
be a functor between $\U$-categories, and let $J$ be a finite and rigid category (8.8.5). If inductive limits (respectively projective limits) of type $J$ are representable in $C$ and if $f$ commutes with them, then
\[
 \Ind(f):\Ind(C)\longrightarrow\Ind(C')
\]
also commutes with this type of limit.
\end{statement}

This follows immediately from 8.8.5 and from the calculation, in 8.9.2, of finite limits in a category of ind-objects, for a functor represented in indexed form.

\begin{statement}{Corollary 8.9.8}
If finite inductive limits (respectively finite projective limits) are representable in $C$, and if $f$ is right exact (respectively left exact), then the same is true of $\Ind(f)$. In the non-parenthesized case, $\Ind(f)$ even commutes with arbitrary small inductive limits.
\end{statement}

The first assertion follows from 8.9.7. The second follows from the first, taking into account that $\Ind(f)$ commutes with filtered inductive limits by 8.6.3.

Let $C$ be a $\U$-category.

Show that $\Ind(C)$ satisfies the same property. In the last parenthesized case one concludes that every epimorphism of $\Ind(C)$ is effective, every monomorphism of $\Ind(C)$ is effective, and every bimorphism of $\Ind(C)$ is an isomorphism. Show that, in this case, every epimorphism (respectively monomorphism) of $\Ind(C)$ can be represented by an inductive system of epimorphisms (respectively monomorphisms) of $C$. If, moreover, every epimorphism (respectively every monomorphism) in $C$ is universal, then the same property holds in $\Ind(C)$.

\item Suppose $C$ is additive (respectively abelian). Then $\Ind(C)$ is additive (respectively abelian).

\item Let $X=\indlim_I X_i$ be an ind-object of $C$, and let $R\rightrightarrows X$ be an equivalence relation in $X$. For every $i\in\Ob I$, let $R_i\rightrightarrows X_i$ be the equivalence relation in $X_i$, considered as an object of $\Ind(C)$, induced by $R$ via $X_i\to X$. Here one assumes that the fiber product
\[
 R_i=R\times_{X\times X}(X_i\times X_i)
\]
in $\Ind(C)$ is representable by an object of $\Ind(C)$, which is the case if finite projective limits are representable in $C$. Show that, for $R$ to be effective, it is enough that the $R_i$ be effective.

For $R$ to be effective, it is necessary that $R$ be of the form $\indlim_i R_i$, where the $R_i$ are equivalence relations in $X$, viewed as an object of $C$; this condition is sufficient if equivalence relations are effective in $C$.

\item Suppose every morphism in $C$ factors as an effective epimorphism followed by an effective monomorphism, and suppose finite projective limits are representable. \item Suppose finite projective limits and finite sums are representable in $C$, every equivalence relation in $C$ is effective, and every morphism in $C$ factors as an effective epimorphism followed by an effective monomorphism. \begin{description}[leftmargin=2em]
\item[ST)] For every object $X$ of $C$ and every subobject $Z$ of $X\times X$ in $C$, if one defines recursively a sequence of subobjects $Z_n$ $(n\geq 0)$ of $X\times X$ in $C$ by $Z_0=Z$ and
\[
 Z_{n+1}=\Sup(Z_n,s(Z_n),Z_n\circ Z_n),
\]
where the supremum is taken in the set of subobjects of $X\times X$, which exists by the hypotheses on $C$, then the sequence $(Z_n)_{n\geq 0}$ is stationary.
\end{description}

\setcounter{enumi}{10}
\item Show that in $\Ind(\Set)$ equivalence relations are not necessarily effective.
\end{enumerate}

\end{statement}

\subsection{Dual notions: pro-objects and pro-representable functors}

Let $C$ be a $\U$-category. A \emph{pro-object} of $C$ is a functor
\[
 \varphi:I^\op\longrightarrow C,
\tag{8.10.1}
\]
where $I$ is an essentially small filtered category, called the index category, and where, as usual, $I^\op$ denotes the opposite category. Let $\V$ be a universe such that $\V\supset\U$. Notice that pro-objects of $C$ indexed by $I\in\V$ are in bijective correspondence with ind-objects of $C^\op$ indexed by $I\in\V$: to such an ind-object $\psi:I\to C^\op$ one associates the pro-object $\varphi=\psi^\op:I^\op\to C$. Following this correspondence, one defines, by transport of structure and reversal of arrows, the notion of morphism between pro-objects of $C$ from the analogous notion (8.2.4.2, 8.2.5.1) for ind-objects. One accordingly obtains the category of pro-objects of $C$ indexed by $I\in\V$, together with a canonical isomorphism
\[
 \Pro_\V(C,\U)\simeq \Ind_\V(C^\op,\U)^\op.
\tag{8.10.2}
\]
For $\V=\U$ one simply speaks of the \emph{category of pro-objects of} $C$, denoted $\Pro_\U(C)$ or simply $\Pro(C)$; it is therefore defined by
\[
 \Pro(C)=\Pro_\U(C,\U)\simeq \Ind(C^\op)^\op.
\tag{8.10.3}
\]
If two pro-objects are given in indexed form
\[
 \mathcal X=(X_i)_{i\in I},\qquad \mathcal Y=(Y_j)_{j\in J},
\tag{8.10.4}
\]
then formula (8.2.5.1) takes here the form
\[
 \Hom(\mathcal X,\mathcal Y)\simeq \varprojlim_j\varinjlim_i\Hom(X_i,Y_j).
\tag{8.10.5}
\]
The functor (8.4.1), applied to $C^\op$, gives, after passage to opposite categories, a canonical functor, denoted by the same letter $c$ when no confusion is possible, which allows one to identify $C$ with a full subcategory of $\Pro(C)$:
\[
 c:C\longrightarrow\Pro(C).
\tag{8.10.6}
\]
The arrows were reversed in the definition of morphisms of pro-objects precisely in order to obtain such a functor, rather than a functor $C\to\Pro(C)^\op$. Pro-objects in the essential image of (8.10.6) are again called \emph{essentially constant pro-objects}.

Put
\[
 \check C=(C^\op)^{\widehat{\ }}=\Hhom(C,\Set).
\tag{8.10.7}
\]
Then the functor (8.2.4.7) may be regarded as a canonical fully faithful functor
\[
 L:\Pro_\V(C,\U)\longrightarrow \check C^\op,
\tag{8.10.8}
\]
and in particular one obtains
\[
 L:\Pro(C)\hookrightarrow \check C^\op.
\tag{8.10.9}
\]

We shall leave to the reader the task of translating, as needed, the results of the preceding sections and of the following ones from the language of ind-objects into that of pro-objects. Depending on the mathematical context, one or the other language is the more useful, not to mention the cases in which both notions occur simultaneously, for instance when one has to consider complex categories such as $\Pro(\Ind(C))$ or $\Ind(\Pro(C))$. We content ourselves with giving a few additional indications, in order to fix terminology and notation and to clarify the yoga.

\subsubsection*{8.10.10}
A \emph{covariant functor $F:C\to\Set$} is called a \emph{pro-representable functor} if it lies in the essential image of (8.10.9), or equivalently of 8.10.8. The full subcategory of $\check C$ consisting of such functors is therefore equivalent to $\Pro(C)^\op$. One generally prefers to work in the opposite category, as the essential image as a subcategory of (8.10.9), hence equivalent to $\Pro(C)$ itself. In accord with this usage, it is often preferable to regard covariant functors
\[
 F:C\longrightarrow\Set
\]
as objects of $\Hhom(C,\Set)^\op=\check C^\op$, and consequently to write the category of morphisms
\[
 X\longrightarrow F\quad\text{in }\check C^\op,
\]
that is, the pairs $(X,u)$ consisting of an object $X$ of $C$ and an element $u\in F(X)$, as
\[
 {}_F\backslash C.
\tag{8.10.11}
\]
With this convention one obtains a covariant functor, the source functor,
\[
 {}_F\backslash C\longrightarrow C,
\tag{8.10.12}
\]
and 3.4 is rewritten in the form
\[
 F\xrightarrow{\sim}\prolimI{({}_F\backslash C)^\op}X.
\tag{8.10.13}
\]

\subsubsection*{8.10.14}
Criterion 8.3.3, applied to $C^\op$, becomes a criterion of pro-representability for $F$: it is necessary and sufficient that $({}_F\backslash C)^\op$ be filtered and essentially small. The latter condition is superfluous if $C$ is equivalent to a small category. If, moreover, finite projective limits are representable in $C$, this is equivalent to saying that $F$ commutes with them, that is, that $F$ is \emph{left exact}.

\subsubsection*{8.10.15}
In $\Pro(C)$, small filtered projective limits are representable, and the functor (8.10.9) commutes with them. In other words, this functor transforms projective limits in $\Pro(C)$ into inductive limits in $\check C=\Hhom(C,\Set)$, just as does its composite $C\to\check C^\op$ with (8.10.6), which shares with it the unfortunate property of being contravariant when it is regarded as taking values in $\check C$. Notice carefully that pro-representable functors from $C$ to $\Set$ are functors which are small filtered inductive limits of representable functors, and not projective limits, as the terminology might perhaps suggest.

\subsection{Ind-adjoints and pro-adjoints}

\subsubsection*{8.11.1}
Let
\[
 f:C\longrightarrow C'
\tag{8.11.1.1}
\]
be a functor between $\U$-categories. It induces the functor $F\mapsto F\circ f$
\[
 f^*:\widehat{C'}\longrightarrow\widehat C.
\tag{8.11.1.2}
\]
We say that $f$ \emph{admits an ind-adjoint} if this functor sends ind-representable functors to ind-representable functors. Since $f^*$ commutes with inductive limits, and since the full subcategory of $\widehat{C'}$ formed by the ind-representable functors is stable under small filtered inductive limits, it is equivalent to say that $f$ admits an ind-adjoint, or that $f^*$ sends the representable objects of $\widehat{C'}$ to ind-representable functors on $C$, that is,
\[
 X\longmapsto \Hom(f(X),Y')\quad\text{is ind-representable for every }Y'\in\Ob C'.
\tag{8.11.1.3}
\]
Another way to express the condition that $f$ admit an ind-adjoint is to say that there is a functor
\[
 g:\Ind(C')\longrightarrow\Ind(C)
\tag{8.11.1.4}
\]
which is essentially induced by $f^*$, that is, such that there is an isomorphism of bifunctors
\[
 \Hom_{\Ind(C)}(X,g(\mathcal Y'))\simeq
 \Hom_{\Ind(C')}(f(X),\mathcal Y'),
\tag{8.11.1.5}
\]
where $X\in\Ob C$ and $\mathcal Y'\in\Ob\Ind(C')$. In fact it is enough to have a functor
\[
 g_0:C'\longrightarrow\Ind(C)
\tag{8.11.1.6}
\]
and an isomorphism of bifunctors
\[
 \Hom_{\Ind(C)}(X,g_0(Y'))\simeq \Hom_{C'}(f(X),Y')
\tag{8.11.1.7}
\]
for $X\in\Ob C$ and $Y'\in\Ob C'$. The functor $g$, respectively $g_0$, is plainly determined by $f$ up to unique isomorphism, and conversely $f$ can be reconstructed up to canonical isomorphism from $g$, respectively from $g_0$. Formula (8.11.1.5) makes it clear that $g$ commutes with inductive limits and extends $g_0$; this determines it up to unique isomorphism in terms of $g_0$ by 8.7.2. The functor $g$, and sometimes also the functor $g_0$ that it extends, is called the \emph{ind-adjoint functor of} $f$. There is no need here to specify ``right'', since the other one, if it exists, will be called the pro-adjoint (cf. 8.11.5 below).

Of course, if $f$ admits a right adjoint
\[
 f^{\ad}:C'\longrightarrow C,
\]
then it admits an ind-adjoint $g$, and this is canonically isomorphic to the canonical extension of $f^{\ad}$ to ind-objects:
\[
 g\simeq \Ind(f^{\ad}).
\tag{8.11.1.8}
\]
The notion of ind-adjoint is therefore a natural generalization of the notion of right adjoint.

Now consider the canonical extension
\[
 \Ind(f):\Ind(C)\longrightarrow\Ind(C').
\tag{8.11.1.9}
\]
We then have the following proposition.

\begin{statement}{Proposition 8.11.2}
For the functor $f:C\to C'$ to admit an ind-adjoint, it is necessary and sufficient that the functor $\Ind(f)$ in (8.11.1.9) admit a right adjoint. This right adjoint is canonically isomorphic to the ind-adjoint (8.11.1.4).
\end{statement}

The sufficiency and the last assertion are trivial from the ind-adjunction formula (8.11.1.5). For necessity, note that, by passage to the projective limit in this formula over the $X_i$, one deduces an adjunction isomorphism for the ind-objects $\mathcal X=(X_i)_{i\in I}$ and $\mathcal Y'$:
\[
 \Hom_{\Ind(C)}(\mathcal X,g(\mathcal Y'))\simeq
 \Hom_{\Ind(C')}(\Ind(f)(\mathcal X),\mathcal Y').
\tag{8.11.2.1}
\]
\qedtext

\begin{statement}{Corollary 8.11.3}
If $f$ admits an ind-adjoint, then $\Ind(f)$ commutes with inductive limits, and the ind-adjoint $g$ commutes with projective limits.
\end{statement}

\begin{statement}{Proposition 8.11.4}
For the functor $f:C\to C'$ to admit an ind-adjoint, it is necessary that $f$ be right exact, and this condition is sufficient when $C$ is equivalent to a small category.
\end{statement}

This follows from criterion (8.11.1.3), and from 8.3.1 and 8.3.3(iv). For another criterion in terms of the notion of accessible functor, see 8.13.3.

\subsubsection*{8.11.5}
Now consider the functor $F\mapsto F\circ f$
\[
 f^{\op *}:\check{C'}\longrightarrow\check C
\tag{8.11.5.1}
\]
induced by $f$. We shall say that $f$ \emph{admits a pro-adjoint} if the preceding functor sends pro-representable functors to pro-representable functors, that is, if there exists a functor, called the \emph{pro-adjoint functor of $f$},
\[
 g:\Pro(C')\longrightarrow\Pro(C),
\tag{8.11.5.2}
\]
and an isomorphism of bifunctors
\[
 \Hom_{\Pro(C)}(g(\mathcal Y'),\mathcal X)\simeq
 \Hom_{\Pro(C')}(\mathcal Y',\Pro(f)(\mathcal X)).
\tag{8.11.5.3}
\]
Of course, saying that $f$ admits a pro-adjoint means that $f^\op$ admits an ind-adjoint, so the notions and results for ind-adjoints translate trivially into terms of pro-adjoints. Let us merely point out that $f$ admits a pro-adjoint if and only if
\[
 \Pro(f):\Pro(C)\longrightarrow\Pro(C')
\]
admits a left adjoint, and that in this case the preceding functor $g$ is such a left adjoint of $\Pro(f)$; and that for this it is necessary that $f$ be left exact, this condition also being sufficient when $C$ is equivalent to a small category. In this case, $f$ is exact if and only if it admits both an ind-adjoint and a pro-adjoint.

\begin{statement}{Example 8.11.6}
Consider the case of a functor
\[
 f:C\longrightarrow \Set.
\]
If this functor admits a pro-adjoint, then it is pro-representable. The converse is true if and only if the full subcategory of $\check C$ formed by pro-representable functors is stable under small products; this is the case in particular if $C$ is equivalent to a small category (8.10.14), or if small products are representable in $C$ (8.9.5(b) applied to $C^\op$). Thus, up to minor qualifications, one may say that, for a functor $f:C\to C'$, the existence of a pro-adjoint is the \emph{natural generalization of pro-representability of $f$}, which is defined when $C'=\Set$.
\end{statement}

\subsection{Strict ind-objects and pro-objects. Application to a representability criterion}

\subsubsection*{8.12.1}
Let
\[
 \mathcal X=(X_i)_{i\in I}
\]
be an ind-object of the $\U$-category $C$, and let $F$ be the presheaf which it ind-represents. It is immediate that the following two conditions are equivalent: the canonical morphisms
\[
 X_i\longrightarrow F=\indlimI{i}X_i
\]
are monomorphisms of $\Ind(C)$, or equivalently of $\widehat C$, that is, monomorphisms of functors argument by argument; and, for every arrow $i\to j$ of $I$, the corresponding transition morphism
\[
 X_i\longrightarrow X_j
\]
is a monomorphism. When these conditions hold, and when moreover $I$ is an ordered category, $\mathcal X$ is called a \emph{strict ind-object}. Notice that this condition is not invariant under isomorphism of ind-objects. An ind-object will be called \emph{essentially strict} if it is isomorphic to a strict ind-object. A presheaf will be called \emph{strictly ind-representable} if it is ind-representable by a strict ind-object; thus $F=\indlimI{i}X_i$ is strictly ind-representable if and only if $\mathcal X=(X_i)_{i\in I}$ is essentially strict.

\paragraph{8.12.1.1.}
Let $F$ be a presheaf on $C$, and consider the full subcategory of $C_{/F}$ formed by the arrows $X\to F$ with source in $C$ which are monomorphisms. We shall call it the \emph{category of representable subfunctors of $F$}; it is the category associated with the ordered set of representable subfunctors of $F$, ordered by the order induced from the set of subobjects of $F$. The following then follows immediately from the definitions.

\begin{statement}{Proposition 8.12.2}
For a presheaf $F$ on $C$ to be strictly ind-representable, it is necessary and sufficient that the ordered category $I$ of representable subfunctors of $F$ be filtered and essentially small and that one have
\[
 \varinjlim_I X_i\xrightarrow{\sim}F.
\tag{8.12.2.1}
\]
When, for every object $X$ of $C$, the set of subobjects of $X$ in $C$ is small, for example when $C$ admits a small generating subcategory (7.4), this implies that the category of representable subfunctors is even small.
\end{statement}

\paragraph{8.12.2.2.}
If $F$ is strictly ind-representable, there is therefore a preferred way to ind-represent it by an ind-object, and that ind-object is even strict: take the representation (8.12.2.1). A strict ind-object $\mathcal Y$ is called \emph{saturated} if it is isomorphic to an ind-object of the form $(X_i)$ appearing in (8.12.2.1), for a suitable $F$. Thus there exists, up to unique isomorphism, only one saturated strict ind-object isomorphic to the given strict ind-object, namely the one considered in 8.12.2, taking $F=\varinjlim\mathcal Y$ as a limit in $\widehat C$.

\paragraph{8.12.2.3.}
Suppose one already knows that a small cofinal subset can be found in $\Ob C_{/F}$, which is the case in particular if $F$ is ind-representable. Then the same condition follows in the full subcategory $I$ considered in 8.12.2, so in the criterion 8.12.2 one may omit the condition that $I$ be essentially small.

\subsubsection*{8.12.3}
Let $F$ be a presheaf on $C$, and consider an object
\[
 u:X\longrightarrow F
\]
of $C_{/F}$. We say that $u$, or the pair $(X,u)$, is \emph{minimal} if, for every factorization of $u$ as
\[
 X\xrightarrow{p}X'\xrightarrow{u'}F,
\tag{8.12.3.1}
\]
with $p$ a strict epimorphism (10.2), the morphism $p$ is an isomorphism. Considering $u$ as an element of $F(X)$ (1.4), to say that $u$ is minimal therefore means that every strict epimorphism $p:X\to X'$ such that
\[
 u\in\Imn\bigl(F(p):F(X')\to F(X)\bigr)
\]
is an isomorphism. This notion is clarified by part a) of the following lemma.

\begin{statement}{Lemma 8.12.4}
Let $F$ be a presheaf on $C$.
\begin{enumerate}[label=\alph*)]
\item Suppose that $F$ transforms cokernels into kernels, and consider a morphism
\[
 u:X\longrightarrow F
\]
with $X\in\Ob C$. For $u$ to be a monomorphism, it is sufficient, when cokernels of double arrows are representable in $C$, that $u$ be minimal; this condition is also necessary if fiber products are representable in $C$.

\item Suppose finite inductive limits are representable in $C$. For the subcategory of representable subfunctors of $F$ (8.12.1.1) to be filtered, not necessarily small, and to have inductive limit $F$, it is sufficient that $F$ be left exact and that every morphism $u:X\to F$, with $X\in\Ob C$, factor as
\[
 X\longrightarrow X'\xrightarrow{u'}F
\]
with $u'$ minimal. This condition is also necessary if fiber products are representable in $C$.

\item Suppose that $C$ admits a small generating subcategory (7.1), and that $I$, the category of representable subfunctors of $F$, is filtered. Then $I$ is small.
\end{enumerate}
\end{statement}

\begin{prooftext}
\begin{enumerate}[label=\alph*)]
\item Suppose $u$ is minimal; we prove that $u$ is a monomorphism, that is, that for every double arrow $v,v':Y\rightrightarrows X$ such that $uv=uv'$, one has $v=v'$. Indeed, if $X'=\Coker(v,v')$, then $u$ factors as $X\xrightarrow{p}X'\xrightarrow{u'}F$, since $F$ transforms cokernels into kernels. As $p$ is a strict epimorphism by construction, it follows that $p$ is an isomorphism, that is, $v=v'$. Conversely, suppose $u$ is a monomorphism, and prove that it is minimal. Consider a factorization (8.12.3.1). Since $p$ is a strict epimorphism, it is the cokernel of the canonical double arrow
\[
 v,v': X\times_{X'}X\rightrightarrows X.
\]
Since $(u'p)v=(u'p)v'$ and $u'p=u$ is a monomorphism, we have $v=v'$, hence $p$ is an isomorphism.

\item Sufficiency: since $F$ is left exact, $C_{/F}$ is filtered. Since we know that $F=\varinjlim_{C_{/F}}X$, we are reduced by 8.1.3(c) to proving that the full subcategory $I$ of $C_{/F}$ formed by the subobjects of $F$ is cofinal in $C_{/F}$. By the sufficiency assertion in a), this is exactly what is ensured by the hypothesis that every object of $C_{/F}$ is dominated by a minimal object. Necessity: since every filtered inductive limit of left exact functors is again left exact, the first condition is trivially necessary. The second then follows from the necessity assertion in a).

\item A representable subfunctor $X\hookrightarrow F$ of $F$ is known once one knows the full subcategory $C'_{/X}$ of $C'_{/F}$, where $C'$ is a fixed small generating subcategory of $C$. Use the hypothesis that $I$ is filtered. Since $C'_{/F}$ is small, the set of its full subcategories is small, whence the conclusion.
\end{enumerate}
\end{prooftext}

\begin{statement}{Proposition 8.12.5}
Let $C$ be a $\U$-category in which finite inductive limits are representable and which admits a small generating subcategory (7.1). Let $F$ be a presheaf on $C$. For $F$ to be strictly ind-representable, it is sufficient that it satisfy the following two conditions; and these conditions are also necessary if fiber products are representable in $C$:
\begin{enumerate}[label=\alph*)]
\item $F$ is left exact.

\item Every pair $(X,u)$, with $X\in\Ob C$ and $u\in F(X)$, is dominated in $C_{/F}$ by a minimal pair (8.12.3). In other words, there is a minimal pair $(X',u')$, with $X'\in\Ob C$ and $u'\in F(X')$, and a morphism $f:X\to X'$ such that $u=F(f)(u')$.
\end{enumerate}
\end{statement}

Sufficiency follows from 8.12.2 and 8.12.4(b),(c); necessity follows from 8.3.1 and 8.12.4(a).

\begin{statement}{Remark 8.12.6}
When $F$ transforms amalgamated sums in $C$ into fiber products, one sees immediately that, for a given pair $(X,u)$ as in b), the set of strict quotients $X'$ of $X$ such that
\[
 u\in\Imn\bigl(F(X')\to F(X)\bigr)
\]
is decreasing filtered. It follows that if the set of strict quotients of $X$ is Artinian, then condition b) of 8.12.5 is automatically satisfied. Thus, if the preceding condition on $X$ is satisfied for every object $X$ of $C$ -- in which case one sometimes also says that the objects of $C^\op$ are \emph{Artinian} -- then 8.12.5 implies that $F$ is strictly ind-representable if and only if $F$ is left exact. In this case, $F$ is strictly ind-representable as soon as it is ind-representable (8.3.1).
\end{statement}

\begin{statement}{Corollary 8.12.7}
Let $C$ be a $\U$-category in which small projective limits are representable, and which admits a small cogenerating subcategory (7.9, 7.13). Then a functor $F:C\to\Set$ is representable if and only if $F$ commutes with small projective limits.
\end{statement}

Necessity is clear. For sufficiency, apply 8.12.5 to the opposite category $C^\op$. To prove first that $F$ is strictly pro-representable, it remains to prove that every pair $(X,u)$, with $X\in\Ob C$ and $u\in F(X)$, is dominated by a minimal pair. But the family $(X_i)_{i\in I}$ of strict subobjects of $X$ such that
\[
 u\in\Imn\bigl(F(X_i)\to F(X)\bigr)
\]
is small (7.5 in its dual form). Since $F$ is left exact, this family is cofiltered (8.12.6), and since $F$ commutes with small projective limits one sees that $X'=\varprojlim_i X_i$ is a smallest object of this family. Use the fact that $F$, being left exact, transforms monomorphisms into monomorphisms. If $u'\in F(X')$ is the unique element whose image in $F(X)$ is $u$, it follows that $(X',u')$ is a minimal pair dominating $(X,u)$. This proves that $F$ is pro-representable, and the conclusion follows from the following lemma.

\begin{statement}{Lemma 8.12.8.1}
Let $F:C\to\Set$ be a functor, where $C$ is a $\U$-category in which small projective limits are representable. For $F$ to be representable, it is necessary and sufficient that it be pro-representable and commute with small projective limits.
\end{statement}

Necessity is clear. We prove sufficiency. To say that $F$ is representable plainly means that ${}_F\backslash C$ has an initial object; indeed, the initial objects of ${}_F\backslash C$ are precisely the isomorphisms $F\to X$, that is, the data of a representation for $F$. Since $F$ is pro-representable, $({}_F\backslash C)^\op$ is filtered and equivalent to a small category. Thus
\[
 X=\varprojlim_{({}_F\backslash C)^\op} X
\]
is representable in $C$, and since $F$ commutes with the limit in question, it follows that $X$ is an initial object of ${}_F\backslash C$. \qedtext

\begin{statement}{Corollary 8.12.8}
With $C$ satisfying the hypotheses of 8.12.7, let $f:C\to C'$ be a functor from $C$ to a $\U$-category $C'$. For $f$ to admit a left adjoint, it is necessary and sufficient that $f$ commute with small projective limits.
\end{statement}

This reduces trivially to 8.12.7 by applying that statement to the composite functors of the form
\[
 X\longmapsto \Hom(Y',f(X)).
\]

\begin{statement}{Examples 8.12.9}
As noted in 7.13, the hypotheses on $C$ in 8.12.7 and 8.12.8 are satisfied when $C$ is the category of $\U$-sheaves of sets on a topological space $X\in\U$, or more generally on a $\U$-site (II 3.0.2). We give an instructive example, due to H. Bass, showing that the hypothesis of existence of a small cogenerating subcategory $D$ of $C$ is not redundant in 8.12.7. Take $C$ to be the category of groups belonging to $\U$. Let $J$ be the set of isomorphism classes of simple groups in $C$. For each $j\in J$, choose a simple group $G_j$ in the class $j$, and let $I$ be the filtered ordered set of $\U$-small subsets of $J$. For $i\in I$, let
\[
 X_i=\amalg_{j\in i}G_j.
\]
The $X_i$ then form a projective system $(X_i)_{i\in I}$ in $C$, with epimorphic transition morphisms, and the corresponding functor
\[
 F(X)=\varinjlim_i\Hom(X_i,X)
\tag{$*$}
\]
takes values in $\U$-sets, although the index set $I$ plainly does not have cardinal in $\U$. More precisely, let us show that for every small full subcategory $C_0$ of $C$, there exists $i_0\in I$ such that the restriction of $F$ to $C_0$ is represented by $X_{i_0}$. This will prove both that $F$ is $\U$-set-valued and that it commutes with small projective limits. To prove the assertion, it is enough to note that, for every $X\in\Ob C_0$, the cardinal of the set $J(X)$ of $j\in J$ for which there exists a nontrivial homomorphism from $G_j$ to $X$ is necessarily small, since such a morphism is necessarily a monomorphism, $G_j$ being simple. Hence, if $i_0$ is the subset of $J$ which is the union of the $J(X)$ for $X\in\Ob C_0$, then $i_0$ is small, that is, $i_0\in I$, and it does the job. On the other hand, it is clear that $F$ is not representable, since $\card I\notin\U$. From this and 8.12.7 one therefore concludes that \emph{the category $C$ of groups in $\U$ admits no small full cogenerating subcategory}. Since the object $\ZZ$ of $C$ is nevertheless a generator, it follows from the proof of 7.12 that there exists a two-generator group $G$ which does not embed into an injective object of the category $C$ of groups in $\U$. It also seems plausible that $C$ has no injective objects other than the unit groups.
\end{statement}

\subsection{Pro-representable functors and accessible functors}

\subsubsection*{8.13.1}
In this section we use some notions and results from the following paragraph, notably 9.11 and 9.13, in order to obtain a criterion of pro-representability which will be used incidentally in IV 9.16. In what follows, $C$ denotes a $\U$-category satisfying condition L of 9.1(b). This condition is satisfied, for example, if small filtered inductive limits are representable in $C$.

\begin{statement}{Proposition 8.13.2}
Let $C$ be as above, and let
\[
 f:C\longrightarrow\Set
\]
be a functor.
\begin{enumerate}[label=\alph*)]
\item Suppose every object of $C$ is accessible (9.3). If $f$ is pro-representable, then $f$ is accessible (9.2) and left exact.

\item Suppose finite projective limits are representable in $C$, and that $C$ admits a cardinal filtration (9.12). If $f$ is accessible and left exact, then $f$ is pro-representable.
\end{enumerate}
\end{statement}

\begin{prooftext}
\begin{enumerate}[label=\alph*)]
\item The hypothesis on $C$ means that the covariant representable functors from $C$ to $\Set$ are accessible. Therefore the same is true of every small inductive limit of such functors (9.6(i)), and hence of every pro-representable functor.

\item By 8.3.3(iii), it remains to prove that $\Ob({}_F\backslash C)^\op$ contains a small cofinal subcategory. By hypothesis there exists a cardinal $\Pi$ such that $f$ is $\Pi$-accessible. Let $(X,u)$, $u\in F(X)$, be an object of ${}_F\backslash C$. With the notation of 9.12(c), one then has $X=\varinjlim_i X_i$, with $I$ filtered and large relative to $\Pi$, and with the $X_i$ in $C'=\Filt^{\Pi}(C)$. Hence
\[
 F(X)\xleftarrow{\sim}\varinjlim_i F(X_i).
\]
This shows that the small subcategory $({}_F\backslash C')^\op$ is cofinal in $({}_F\backslash C)^\op$, and completes the proof.
\end{enumerate}
\end{prooftext}

\begin{statement}{Corollary 8.13.3}
Let $C$ be a $\U$-category satisfying the following conditions:
\begin{enumerate}[label=\alph*)]
\item finite projective limits are representable in $C$;
\item small filtered inductive limits are representable in $C$;
\item the functor $\Ker$ on the category of double arrows of $C$ is accessible, for example it commutes with small filtered inductive limits;
\item every strict epimorphism of $C$ is strictly universal (10.2);
\item there exists a small subcategory of $C$ generating by strict epimorphisms (7.1).
\end{enumerate}
Under these conditions, a functor $f:C\to\Set$ is pro-representable if and only if it is left exact and accessible (9.2).
\end{statement}

Indeed, the hypotheses on $C$ in 8.13.2(a) and (b) are satisfied by 9.11 and 9.13, respectively.

\begin{statement}{Corollary 8.13.4}
Let $f:C\to C'$ be a functor between $\U$-categories satisfying conditions a) through e) of 8.13.3. For $f$ to admit a pro-adjoint, it is necessary and sufficient that $f$ be left exact and accessible.
\end{statement}

Since the representable functors
\[
 h_{Y'}:X'\longmapsto \Hom(Y',X')\,,\qquad C'\to\Set,
\]
are left exact and accessible (9.11), if $f$ has these same properties then so do their composites with $f$, and these composites are therefore pro-representable by 8.13.3; that is, $f$ admits a pro-adjoint. Conversely, suppose that $f$ admits a pro-adjoint, and let $C'_1$ be a small generating subcategory of $C'$. Choose a cardinal $\Pi$ such that the objects $Y'\in\Ob C'_1$ are $\Pi$-accessible. To prove that $f$ is $\Pi$-accessible, it is enough to prove this for its composites with the $h_{Y'}$, $Y'\in\Ob C'_1$. But by hypothesis these composites are pro-representable, hence accessible by 8.13.3, and therefore $\Pi$-accessible after increasing $\Pi$ if necessary. \qedtext
