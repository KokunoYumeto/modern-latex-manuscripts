\subsubsection*{n\textsuperscript{o} 2. Efecto sobre la cohomología}

\begin{statement}{Proposición 2}
Sea $X$ un preesquema de característica $p$.
\begin{enumerate}[label=\alph*)]
\item Para todo haz de conjuntos $F$ sobre $X_{\et}$, el endomorfismo de $H^0(X,F)$ definido por $(\fr_X,\Fr^*_{F/X})$, es decir, el compuesto
\[
H^0(X,F)\longrightarrow H^0(X,\fr_X^*(F))\longrightarrow H^0(X,F),
\]
donde la primera flecha es la aplicación canónica (carácter contravariante de $H^0(X,F)$ respecto de $X$) y la segunda $H^0(X,\Fr^*_{F/X})$, es la identidad de $H^0(X,F)$.
\item Para todo haz de grupos $F$ sobre $X_{\et}$, el endomorfismo de $H^1(X,F)$ definido por $(\fr_X,\Fr^*_{F/X})$, es decir,
\[
H^1(X,F)\longrightarrow H^1(X,\fr_X^*F)\longrightarrow H^1(X,F),
\]
es la identidad.
\item Para todo complejo acotado por la izquierda $F$ de $\Lambda$-módulos sobre $X_{\et}$, el endomorfismo de $\RGamma_X(F)$ definido por $(\fr_X,\Fr^*_{F/X})$, es decir, el compuesto
\[
\RGamma_X(F)\longrightarrow \RGamma_X(\fr_X^*F)\longrightarrow \RGamma_X(F),
\]
es la identidad.
\end{enumerate}
\end{statement}
