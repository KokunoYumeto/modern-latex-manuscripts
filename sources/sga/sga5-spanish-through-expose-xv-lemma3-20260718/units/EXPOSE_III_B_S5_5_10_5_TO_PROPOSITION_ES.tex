Para $E\in\ob D^-(B)$, $M\in\ob D^+(K)$, $F\in\ob D^-(B)$, se define una flecha de $D(K)$\footnote{En el n\textsuperscript{o}~6 se aplicarán las construcciones que siguen al caso en que $M$ es un complejo dualizante (6.7).}
\begin{equation}\tag{5.10.5}
\uRHom_B(E,B\lten_KM)\lten_BF\longrightarrow\uRHom_A(E,A\lten_KM)\lten_AF
\end{equation}
como la composición de las flechas
\[
\begin{tikzcd}[column sep=small]
\uRHom_B(E,B\lten_KM)\lten_BF \arrow[r,"(1)"] & \uRHom_B(E,\uHom_A(B,A)\lten_A(B\lten_KM))\lten_BF \\
\arrow[r,"(2)"] & \uRHom_B(E,\uHom_A(B,A)\lten_KM)\lten_A B\lten_BF \\
\arrow[r,"(3)"] & \uRHom_B(E,\uHom_A(B,A\lten_KM))\lten_AF \\
\arrow[r,"(4)"] & \uRHom_A(E,A\lten_KM)\lten_AF,
\end{tikzcd}
\]
donde (1) viene dada por 5.10.1, (2) por el isomorfismo inverso de 5.10.4 con $P=\uHom_A(B,A)\lten_KM$, (3) por el isomorfismo canónico
\[
\uHom_A(B,A)\lten_KM\xrightarrow{\sim}\uHom_A(B,A\lten_KM)
\]
(5.4.2), y (4) por el isomorfismo de adjunción 5.2.2.

\begin{statement}{Proposición 5.10.6}
Supongamos que $M\lten_KF\in\ob D^+(B)$. Se tiene entonces un cuadrado conmutativo en $D(K)$
\begin{equation}\tag{5.10.7}
\begin{tikzcd}[column sep=large,row sep=large]
\uRHom_B(E,B\lten_KM)\lten_BF \arrow[r,"5.4.2"] \arrow[d,"5.10.5"'] & \uRHom_B(E,M\lten_KF) \arrow[d]\\
\uRHom_A(E,A\lten_KM)\lten_AF \arrow[r,"5.4.2"'] & \uRHom_A(E,M\lten_KF),
\end{tikzcd}
\end{equation}
donde la flecha vertical de la derecha es la restricción de escalares.
\end{statement}
