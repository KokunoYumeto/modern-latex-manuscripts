\begin{statement}{Proposición 2.2.2}
Bajo las hipótesis anteriores, se tiene:
\begin{enumerate}[label=\alph*)]
\item
\[
R^{2i+1}p_*(A_P)=0\quad\text{para todo }i\in\mathbb N.
\]
\item Cuando se dota a $\bigoplus_i R^{2i}p_*(\mu_P^{\otimes i})$ de su estructura de $A_S$-álgebra deducida del producto cup, el morfismo de $A_S$-álgebras
\[
A_S[T]\longrightarrow \bigoplus_i R^{2i}p_*(\mu_P^{\otimes i})
\]
definido por $\eta$ tiene por núcleo el ideal $(T^{r+1})$ y define, por tanto, un isomorfismo de $A_S$-álgebras
\[
\theta:A_S[T]/(T^{r+1})\xrightarrow{\sim}\bigoplus_i R^{2i}p_*(\mu_P^{\otimes i}).
\]
\item El morfismo
\[
A_S\longrightarrow R^{2r}p_*(\mu^{\otimes r})
\]
obtenido componiendo $\theta$ con la restricción a $A_S$ de la multiplicación por $T^r$ de $A_S[T]/(T^{r+1})$ en sí mismo es el inverso del morfismo de traza (SGA 4 XVIII)
\[
R^{2r}p_*(\mu^{\otimes r})\longrightarrow A_S;
\]
en otros términos, se tiene
\[
\operatorname{Tr}(\eta^r)=1.
\]
\end{enumerate}
\end{statement}
