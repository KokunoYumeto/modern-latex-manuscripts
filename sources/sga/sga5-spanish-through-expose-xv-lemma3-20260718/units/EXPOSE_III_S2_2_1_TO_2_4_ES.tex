\subsection*{2.1}
Sean $f:X\to Y$ un $S$-morfismo, $P\in\ob D(Y)$, $Q\in\ob D^{+}(Y)$. La flecha canónica evidente a nivel de los complejos
\[
f^{*}\mathcal Hom^{\bullet}(P,Q)\longrightarrow\mathcal Hom^{\bullet}(f^{*}P,f^{*}Q)
\]
proporciona por derivación una flecha canónica
\begin{equation}\tag{2.1.1}
f^{*}\uRHom(P,Q)\longrightarrow\uRHom(f^{*}P,f^{*}Q).
\end{equation}
Sean $L\in\ob D(X)$, $M\in\ob D^{+}(X)$. Pongamos
\begin{equation}\tag{2.1.2}
\Hom_f^{(i,j)}((L,M),(P,Q))=\Hom_f^{(i)}(L,P)\times\Hom_f^{(j)}(M,Q),
\end{equation}
con las notaciones de 1.4. Para $(u,v)\in\Hom_f^{(2,1)}((L,M),(P,Q))$ se define
\begin{equation}\tag{2.1.3}
\uRHom(u,v)\in\Hom_f^{(1)}(\uRHom(L,M),\uRHom(P,Q))
\end{equation}
componiendo 2.1.1 con la flecha $\uRHom(f^{*}P,f^{*}Q)\to\uRHom(L,M)$ definida por $u$ y $v$. Para $L=f^{*}P$, $Q=f_{*}M$, $u$ (resp. $v$) dado por la identidad de $L$ (resp. $Q$), $\uRHom(u,v)$ es el isomorfismo de ``dualidad trivial''
\begin{equation}\tag{2.1.4}
\uRHom(P,f_{*}M)\xrightarrow{\sim}f_{*}\uRHom(f^{*}P,M).
\end{equation}
Para $(u,v)\in\Hom_f^{(3,4)}((L,M),(P,Q))$, se define
\begin{equation}\tag{2.1.5}
\uRHom(u,v)\in\Hom_f^{(1)}(\uRHom(L,M),\uRHom(P,Q))
\end{equation}
como la compuesta de las flechas
\[
\uRHom(P,Q)\xrightarrow{a}\uRHom(f_{!}L,Q)\xrightarrow{b}f_{*}\uRHom(L,f^{!}Q)\xrightarrow{c}f_{*}\uRHom(L,M),
\]
donde $a$ (resp. $c$) está definida por $u$ (resp. $v$), y $b$ es la inversa del isomorfismo de dualidad (SGA~4~XVIII~3.1.9.6).

Se deduce fácilmente de las definiciones que la formación de $\uRHom(u,v)$ es compatible con la composición: para $(u,v)$, $(u',v')$ componibles y de tipo $(2,1)$ (resp. $(3,4)$), se tiene
\begin{equation}\tag{2.1.6}
\uRHom(u'u,v'v)=\uRHom(u',v')\,\uRHom(u,v).
\end{equation}

\subsection*{2.2}
Sea $X$ un $S$-esquema. Recordemos que, para $L\in\ob D^{-}(X)$, $M\in\ob D(X)$, $N\in\ob D^{+}(X)$, se tiene el isomorfismo de adjunción (``querido por Cartan'', como dice Grothendieck)
\begin{equation}\tag{2.2.1}
\Hom(L\otimes^{L}M,N)=\Hom(L,\uRHom(M,N)).
\end{equation}
Para $E\in\ob D^{-}(X)$, $F\in\ob D^{+}(X)$, la flecha identidad de $\uRHom(E,F)$ define mediante 2.2.1 una flecha
\begin{equation}\tag{2.2.2}
E\otimes^{L}\uRHom(E,F)\longrightarrow F,
\end{equation}
llamada \emph{flecha de evaluación}: deriva de la flecha a nivel de los complejos
\[
\mathcal Hom^{\bullet}(E,F)\otimes E\longrightarrow F,
\qquad f\otimes x\longmapsto f(x).
\]
Sean, para $i=1,2$, $E_i,F_i\in\ob D_{\ctf}(X)$. Por adjunción (2.2.1), la flecha
\[
E_1\otimes^{L}E_2\otimes^{L}\uRHom(E_1,F_1)\otimes^{L}\uRHom(E_2,F_2)\longrightarrow F_1\otimes^{L}F_2,
\]
compuesta de un isomorfismo de simetría y del producto tensorial de las flechas de evaluación relativas a $(E_1,F_1)$ y $(E_2,F_2)$, proporciona una flecha
\begin{equation}\tag{2.2.3}
\uRHom(E_1,F_1)\otimes^{L}\uRHom(E_2,F_2)\longrightarrow\uRHom(E_1\otimes^{L}E_2,F_1\otimes^{L}F_2).
\end{equation}
Sean, para $i=1,2$, $X_i$ un $S$-esquema, $X=X_1\times_SX_2$, $p_i:X\to X_i$ la proyección canónica, $E_i,F_i\in\ob D_{\ctf}(X_i)$, $E=E_1\otimes^{L}_{S}E_2$, $F=F_1\otimes^{L}_{S}F_2$. Se tiene una flecha canónica
\begin{equation}\tag{2.2.4}
\uRHom(E_1,F_1)\otimes^{L}_{S}\uRHom(E_2,F_2)\longrightarrow\uRHom(E,F),
\end{equation}
compuesta del producto tensorial de las flechas $p_i^{*}\uRHom(E_i,F_i)\to\uRHom(p_i^{*}E_i,p_i^{*}F_i)$ y de 2.2.3 para $(p_i^{*}E_i,p_i^{*}F_i)$.

\subsection*{Proposición 2.3}
La flecha 2.2.4 es un isomorfismo.

Por localización y devissage, nos reducimos a $E_i=f_{i!}A_{U_i}$ para $f_i:U_i\to X_i$ étale. Se tienen isomorfismos canónicos (dualidad)
\begin{equation}\tag{1}
\uRHom(f_{i!}A_{U_i},F_i)\xrightarrow{\sim}f_{i*}f_i^{*}F_i,
\qquad
\uRHom(E,F)\xrightarrow{\sim}f_{*}f^{*}F\quad(f=f_1\times_S f_2).
\end{equation}
Teniendo en cuenta la fórmula de Künneth 1.6.4, basta, para concluir, verificar que estos isomorfismos hacen conmutativo el cuadrado
\begin{equation}\tag{2}
\begin{tikzcd}[column sep=large,row sep=large]
\uRHom(f_{1!}A_{U_1},F_1)\otimes^{L}_{S}\uRHom(f_{2!}A_{U_2},F_2) \arrow[r,"2.2.3"] \arrow[d,"\sim"'] & \uRHom(f_{!}A_U,F) \arrow[d,"\sim"]\\
f_{1*}f_1^{*}F_1\otimes^{L}_{S}f_{2*}f_2^{*}F_2 \arrow[r,"1.6.4"'] & f_{*}f^{*}F,
\end{tikzcd}
\end{equation}
donde $U=U_1\times_S U_2$. Ahora bien, como $f_i$ es étale, el isomorfismo (1) relativo a $f_i$ (resp. $f$) es el adjunto de la flecha compuesta
\begin{equation}\tag{3}
f_i^{*}\uRHom(f_{i!}A_{U_i},F_i)\xrightarrow{\sim}\uRHom(f_i^{*}f_{i!}A_{U_i},f_i^{*}F_i)\longrightarrow f_i^{*}F_i,
\end{equation}
donde la segunda flecha se deduce de la adjunción $A_{U_i}\to f_i^{*}f_{i!}A_{U_i}$ (resp. la flecha compuesta análoga con $f$). La conmutatividad de (2) equivale a la del cuadrado
\begin{equation}\tag{4}
\begin{tikzcd}[column sep=large,row sep=large]
f_1^{*}\uRHom(f_{1!}A_{U_1},F_1)\otimes^{L}_{S}f_2^{*}\uRHom(f_{2!}A_{U_2},F_2) \arrow[r] \arrow[d] & f^{*}\uRHom(f_{!}A_U,F) \arrow[d]\\
f_1^{*}F_1\otimes^{L}_{S}f_2^{*}F_2 \arrow[r,"\sim"] & f^{*}F,
\end{tikzcd}
\end{equation}
donde las flechas verticales vienen dadas por (3). Ahora bien, la conmutatividad de (4) se deduce del hecho de que 2.2.3 es funtorial en $E_i$ y se reduce a la identidad para $E_i=A$, de donde se sigue la proposición.

\subsection*{2.4}
Sean, para $i=1,2$, $f_i:X_i\to Y_i$ un $S$-morfismo,
\[
f=f_1\times_S f_2:X\to Y,
\]
$E_i,F_i\in\ob D_{\ctf}(X_i)$, $P_i,Q_i\in\ob D_{\ctf}(Y_i)$,
\[
E=E_1\otimes^{L}_{S}E_2,
\quad F=F_1\otimes^{L}_{S}F_2,
\quad P=P_1\otimes^{L}_{S}P_2,
\quad Q=Q_1\otimes^{L}_{S}Q_2.
\]

\begin{equation}\tag{2.4.1}
\resizebox{0.92\textwidth}{!}{%
\begin{tikzcd}[ampersand replacement=\&,column sep=huge,row sep=large]

  \& (P_1\otimes^L_S E_2,Q_1\otimes^L_S F_2) \arrow[l,"{(P_1\otimes^L_S u_2,\,Q_1\otimes^L_S v_2)}"']

donde las flechas verticales (resp. horizontales) son de tipo $(3,4)$ (resp. $(2,1)$).

Ahora podemos enunciar la compatibilidad principal de este número, que desempeñará un papel esencial en la verificación de la fórmula de Lefschetz--Verdier.
