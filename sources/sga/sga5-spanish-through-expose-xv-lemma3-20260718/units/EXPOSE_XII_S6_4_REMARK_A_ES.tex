\subsection*{6.4. Observaciones sobre los términos locales}
\subsubsection*{a)} Es fácil ver que, si $g\in G$, se tiene la relación:

\begin{equation}\tag{6.4.1}
\alpha_x(f,u,g^{-1}f',G,\varphi_g)=\alpha_x(f,u,f',G,\varphi)
\end{equation}
donde se ha puesto

\begin{equation}\tag{6.4.2}
\varphi_g(\xi)=(\operatorname{int}_g\circ\varphi)(\xi)=g^{-1}\varphi(\xi)g,
\quad \forall\,\xi\in G.
\end{equation}
