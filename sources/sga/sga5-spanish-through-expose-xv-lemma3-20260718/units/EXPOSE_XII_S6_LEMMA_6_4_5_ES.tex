\subsection*{Lema 6.4.5}
\emph{Si se verifican las relaciones siguientes:}
\begin{align*}
\nu_{x'}(f') &> 1 \text{,} \\
\nu_{x'}(g^{-1}f') &> 1 \quad ,
\end{align*}
\emph{entonces $g$ pertenece al primer grupo de ramificación $G_1$ (cf. 4.7):}
\[
g \in G_1
\]
\emph{es decir, $g$ induce la identidad sobre el espacio vectorial $\fm_{x'}/\fm_{x'}^2$.}

\subsection*{Demostración}
Sea $\widehat{\mathcal{O}}_{x'} = k[[T]]$. El automorfismo de $\fm/\fm^2$ inducido por $g^{-1}$ está determinado por una raíz de la unidad $\varepsilon(g^{-1}) \in k$ mediante la condición
\[
g^{-1}T = \varepsilon(g^{-1})T \mod \fm^2 \quad ,
\]
donde $\fm$ es el ideal maximal del anillo $k[[T]]$.

Por otra parte, la hipótesis $\nu_{x'}(f') > 1$ da
\[
f'(T) = T + a_\nu T^\nu + a_{\nu+1} T^{\nu+1} + \ldots \text{, } \quad \nu \geq 2 \quad .
\]

Por consiguiente,
\[
g^{-1}f^{\prime}(T) = \varepsilon(g^{-1})T + b_\nu T^\nu + b_{\nu+1} T^{\nu+1} + \ldots \text{,}
\]
y la hipótesis $\nu(g^{-1}f^{\prime}) > 1$ implica $\varepsilon(g^{-1}) = 1$, c.q.d.
