\subsection*{6.20}
Sean, para $i=1,2$, $X_i$ un $G$-$S$-esquema,
\[
X=X_1\times_SX_2,\qquad c:C\to X
\]
un $G$-$S$-morfismo tal que $c_2$ sea cuasifinito y de dimensión Tor finita. Sean $L_i\in\Ob D^+(G,{}_AX_i)$. El morfismo de traza
\[
\Tr_{c_2}:c_{2!}c_2^*L_2\longrightarrow L_2
\]
define por adjunción una flecha de $D(G,X)$ que seguiremos denotando por
\begin{equation}
\Tr_{c_2}:c_2^*L_2\longrightarrow c_2^!L_2,
\tag{6.20.1}
\end{equation}
que permite asociar a todo morfismo $G$-equivariante
\[
u\in\Hom(c_1^*L_1,c_2^*L_2)
\]
una correspondencia $G$-equivariante
\begin{equation}
u^! = \Tr_{c_2}\circ u\in\Hom(c_1^*L_1,c_2^!L_2).
\tag{6.20.2}
\end{equation}

\begin{statement}{Proposición 6.21}
Sean $f:X\to Y$ un $G$-$S$-morfismo propio y un cuadrado conmutativo equivariante 6.11.1, con acción trivial de $G$ sobre $Y$ y $D$. Supongamos que $c$ y $d$ son propios, que $c_2$ y $d_2$ son cuasifinitos y de dimensión Tor finita, y que el cuadrado
\[
\begin{tikzcd}[column sep=large,row sep=large]
C \arrow[r,"c_2"] \arrow[d,"h"'] & X \arrow[d,"f"]\\
D \arrow[r,"d_2"'] & Y
\end{tikzcd}
\]
sea cartesiano y Tor-independiente (SGA 6 III 1.5). Sean $M\in\Ob D_{\mathrm{ctf}}(Y)$,
\[
v\in\Hom(d_1^*M,d_2^*M),
\]
de donde se obtiene un homomorfismo $G$-equivariante
\[
h^*v\in\Hom(c_1^*f^*M,c_2^*f^*M),
\]
y unas correspondencias $G$-equivariantes
\[
(h^*v)^!\in\Hom(c_1^*f^*M,c_2^!f^*M),
\qquad
f_*((h^*v)^!)\in\Hom(d_1^*(f_*f^*M),d_2^!(f_*f^*M)).
\]
Se tiene entonces un cuadrado conmutativo de flechas de $D({}_{\Lambda[G]}Y)$, considerándose $M$ como objeto de $D({}_{\Lambda[G]}Y)$ mediante la acción trivial,
\[
\begin{tikzcd}[column sep=huge,row sep=large]
d_1^*M \arrow[r,"v^!"] \arrow[d] & d_2^!M \arrow[d] & {}\\
d_1^*f_*f^*M \arrow[r,"f_*((h^*v)^!)"'] & d_2^!f_*f^*M,
\end{tikzcd}
\]
donde las flechas verticales vienen dadas por la flecha de adjunción $M\to f_*f^*M$.
\end{statement}

Pongamos $h^*v=u$, $f^*M=L$, y consideremos el diagrama
\[
\begin{tikzcd}[column sep=3.2em,row sep=large]
d_1^*M \arrow[r,"v"] \arrow[d,"{(1)}"'] &
d_2^*M \arrow[rr,"\Tr_{d_2}"] \arrow[d,"{(2)}"'] &&
d_2^!M \arrow[d,"{(3)}"] & {}\\
d_1^*f_*L \arrow[r,"{(4)}"'] &
h_*c_1^*L \arrow[r,"h_*u"'] &
h_*c_2^*L \arrow[r,"h_*\Tr_{c_2}"'] &
h_*c_2^!L \arrow[r,"{(5)}"'] &
d_2^!f_*L,
\arrow[from=1-1,to=2-2, phantom, "\scriptstyle (A)"]
\arrow[from=1-2,to=2-4, phantom, "\scriptstyle (B)"]
\end{tikzcd}
\]
Aquí las flechas (1) y (3) vienen dadas por la flecha de adjunción $M\to f_*L$, (2) por la flecha de adjunción $d_2^*M\to h_*h^*(d_2^*M)=h_*c_2^*L$, (4) por la flecha de cambio de base, y (5) es el isomorfismo canónico (SGA 4 XVIII 3.1.12.3). Por definición, $v^!$ es la compuesta de la fila superior. Por otra parte, de la interpretación de la imagen directa de una correspondencia (III 3.5 b), 3.7) se deduce que $f_*u^!$ es la compuesta de la fila inferior. Basta, pues, demostrar la conmutatividad de los cuadrados (A) y (B). La de (A) resulta inmediatamente de la definición de la flecha de cambio de base. En cuanto a la de (B), se sigue del hecho de que, teniendo en cuenta la hipótesis de Tor-independencia, $\Tr_{d_2}$ es compatible con el cambio de base por $f$ (SGA $4^{1/2}$ Ciclo 2.3.3), de donde se sigue la conclusión.

Aplicando 6.19, se deduce:

\begin{statement}{Corolario 6.22}
Bajo las hipótesis de 6.21, supongamos que
\[
f_*f^*M\in\Ob D_{\mathrm{ctf}}({}_{\Lambda[G]}Y),
\]
y que la flecha de adjunción
\[
M\to f_*f^*M
\]
induce un isomorfismo
\[
M\xrightarrow{\sim}\uRHom_{\Lambda[G]}(\Lambda,f_*f^*M).
\]
Entonces se tiene
\begin{equation}
\Tr_{\Lambda}(v^!)=
\sum_{g\in G_{\natural}}\Tr_{\Lambda[G]}(f_*((f^*v)^!))(g^{-1}).
\tag{6.22.1}
\end{equation}
\end{statement}
