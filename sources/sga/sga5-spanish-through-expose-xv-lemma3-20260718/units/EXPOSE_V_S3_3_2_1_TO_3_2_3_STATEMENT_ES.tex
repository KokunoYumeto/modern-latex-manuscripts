\subsubsection*{3.2. Sistemas proyectivos AR-$J$-ádicos}
\begin{statement}{Proposición 3.2.1}
Las afirmaciones siguientes, para un objeto $X$ de $P$ tal que $J^{n+1}X_n=0$ para todo $n\ge0$, son equivalentes:
\begin{enumerate}[label=(\roman*)]
\item Existe un sistema proyectivo $J$-ádico $Y$ y un AR-isomorfismo $u:Y\to X$ (2.4.5).
\item Existe un sistema proyectivo $J$-ádico $Y$, isomorfo a $X$ en la categoría $P_{AR}$.
\end{enumerate}
\end{statement}

\begin{prooftext}[Demostración]
Hay que demostrar que (ii) implica (i). Un $P_{AR}$-isomorfismo de $Y$ en $X$ está representado, para cierto entero $r$, por un AR-isomorfismo $v:Y[r]\to X$. Como $J^{n+1}X_n=0$ para todo $n\ge0$ e $Y$ es $J$-ádico, $v$ es la compuesta de un $P$-morfismo $u:Y\to X$ y del morfismo canónico de $Y[r]$ en $Y$. Entonces es claro que $u$ es un AR-isomorfismo, de donde se sigue la afirmación.
\end{prooftext}

\begin{statement}{Definición 3.2.2}
Un sistema proyectivo indexado por $\NN$ con valores en $\cC$ se llama AR-$J$-ádico si:
\begin{enumerate}[label=(\alph*)]
\item $J^{n+1}X_n=0$ para todo $n\ge0$;
\item verifica las condiciones equivalentes (i) y (ii) de la proposición 3.2.1.
\end{enumerate}
\end{statement}

\begin{statement}{Proposición 3.2.3}
Sea $X$ un sistema proyectivo indexado por $\NN$ con valores en $\cC$. Se supone que $J^{n+1}X_n=0$ para todo $n\ge0$. Entonces, para que $X$ sea AR-$J$-ádico, es necesario y suficiente que verifique la propiedad (MLAR) y que, denotando por $X'$ su sistema proyectivo de las imágenes universales, exista un entero $r\ge0$ tal que, para todo par $(m,n)$ de enteros con $m\ge n+r$, el «morfismo de transición» que aparece a continuación sea un isomorfismo:
\[
X'_m/J^{n+1}X'_m\longrightarrow X'_{n+r}/J^{n+1}X'_{n+r}.
\]
\end{statement}
