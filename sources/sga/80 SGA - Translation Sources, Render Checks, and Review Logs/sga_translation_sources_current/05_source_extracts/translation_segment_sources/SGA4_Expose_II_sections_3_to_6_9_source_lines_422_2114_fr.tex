\section{Faisceau associé à un préfaisceau}\etiquette[3]{II.sec3}

\begin{enonce*}{Définition 3.0.1}\etiquette[3.0.1]{II.def3.0.1}
Soit $C$ un site. On appelle famille topologiquement génératrice
(où lorsqu'aucune confusion n'en résulte, famille
génératrice) de $C$, un ensemble $G$ d'objets de $C$ tel que
tout objet de $C$ soit but d'une famille couvrante de morphismes de
$C$ (\hyperref[II.def1.2]{1.2}) dont les sources sont des éléments de $G$.
\end{enonce*}

\begin{enonce*}{Définition 3.0.2}\etiquette[3.0.2]{II.def3.0.2}
Soit $𝒰$ un univers. On appelle $𝒰$-site un site $C$
dont la catégorie sous-jacente est une $𝒰$-catégorie
{\rm(\hyperrefext[I][I.def1.1]{1.1})}, qui possède une petite famille
topologiquement génératrice. Soit $C$ une
$𝒰$-catégorie; on appelle $𝒰$-topologie sur $C$
une topologie sur $C$ faisant de $C$ un $𝒰$-site. On dit
qu'un site $C$ est $𝒰$-petit, ou par abus de langage, petit,
si la catégorie sous-jacente à $C$ est petite
{\rm(\hyperrefext[I][I.sec1.0]{1.0})}.
\end{enonce*}

\setcounter{subsubsection}{2}
\subsubsection{}\etiquette[3.0.3]{II.subsubsec3.0.3}
Il résulte immédiatement des définitions que toute topologie
plus fine qu'une $𝒰$-topologie est une
$𝒰$-topologie, et qu'un petit site est un $𝒰$-site.

\begin{enonce*}{Proposition 3.0.4}\etiquette[3.0.4]{II.prop3.0.4}
Soient\pageoriginale $C$ un $𝒰$-site, $G$ une petite famille
topologiquement génératrice de $C$. Pour tout $X\in \ob C$,
désignons par $J_G(X)$ l'ensemble des cribles couvrant $X$
engendrés par une famille de morphismes
$$
(Y_{\alpha}\xrightarrow{u_{\alpha}} X), \alpha\in A, \mbox{ où }
Y_{\alpha} \in G{\quoi}.
$$
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item L'ensemble $J_G(X)$ est petit.

\item L'ensemble $J_G(X)$ est cofinal dans l'ensemble $J(X)$ de tous
  les cribles couvrant $X$, ordonné par inclusion.

\item Pour tout $R\in J_G(X)$, il existe une petite famille
  épimorphique {\rm(\hyperrefext[I][I.sec10.2]{10.2})} $(u_{\alpha}:Y\rightarrow R)$, $\alpha
  \in A$, avec $Y\in G$.
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Posons $A(X)=\amalg_{Y\in G}\hom(Y,X)$. L'ensemble $A(X)$ est
  petit (\hyperrefext[I][I.sec0]{0}) et $\card(J_G(X))\leq 2^{\card(A(X))}$. Par suite
  $J_G(X)$ est petit.

\item Soit $R\in J(X)$. Posons $A(R)=\amalg_{Y\in G}\Hom(Y,R)$ et soit
  $R'$ le crible de $X$ engendré par la famille
  $(Y\xrightarrow{u}R\hookrightarrow X)$, $u\in A(R)$. On a $R'\subset
  R$ et il suffit de montrer que $R'$ est couvrant. D'après l'axiome
  (T 2) de \hyperref[II.def1.1]{1.1}, il suffit de montrer que pour tout morphisme
  $Z\rightarrow R$, $Z\in \ob C$, le crible $R'\times_X Z$ de $Z$ est
  couvrant. Or le crible $R'\times_X Z$ contient le crible engendré
  par la famille de tous les morphismes $Y\rightarrow Z$ avec $Y\in
  G$, famille qui est, par hypothèse, couvrante. Le crible
  $R'\times_X Z$ de $Z$ est donc couvrant (axiome T2 de \hyperref[II.def1.1]{1.1}).

\item Soit $R\in J_G(X)$. La famille $(Y\xrightarrow{u}R)$, $u\in
  A(R)=\amalg_{Y\in G}\Hom(Y,R)$ est, par hypothèse,
  épimorphique. Or, pour tout $Y\in \ob C$, $\Hom(Y,R)$ est contenu
  dans $\Hom(Y,X)$ qui est petit. Par suite $A(R)$ est petit.
\end{enumerate}
\end{enonce*}

\setcounter{subsubsection}{4}
\subsubsection{}\etiquette[3.0.5]{II.subsubsec3.0.5}
Soient $C$ un $𝒰$-site, $𝒱$ un univers tel que
$C\in𝒱$ et $𝒰\subset 𝒱$, $G$ une
$𝒰$-petite famille topologiquement génératrice de $C$.
La catégorie $C^{\widehat{\phantom{,}}}$ des préfaisceaux de
$𝒰$-ensembles sur $C$ est une $𝒱$-catégorie (I
1.1.1). Soit $X$\pageoriginale un objet de $C$. L'ensemble $J(X)$
des cribles couvrant $X$ est $𝒱$-petit et, ordonné par
inclusion, il est cofiltrant (\hyperref[II.subsubsec1.1.1]{1.1.1}). Pour tout
$𝒰$-préfaisceau $F$, la limite inductive
$\varinjlim_{J(X)} \Hom_{C^{\widehat{\phantom{,}}}}(R,F)$ est donc
représentable par un élément de $V$ (\hyperrefext[I][I.def2.4.1]{2.4.1}). De
plus, il résulte de (3.0.4 3)) que, pour tout $R\in J_G(X)$,
$\Hom_{C^{\widehat{\phantom{,}}}}(R,F)$ est $𝒰$-petit, et
comme $J_G(X)$ est un $𝒰$-petit ensemble cofinal dans $J(X)$
(\loccit), il résulte de (\hyperrefext[I][I.sec2.4.2]{2.4.2}) que
$\varinjlim_{J(X)}\Hom_{C^{\widehat{\phantom{,}}}}(R,F)$ est
$𝒰$-petit. Choisissons alors, pour tout $F$ et pour tout
$X$, un élément de $𝒰$ qui représente cette limite
inductive et posons
$$
LF(X)=\varinjlim_{J(X)}\Hom_{C^{\widehat{\phantom{,}}}}(.,F)
$$
Soit $g:Y\rightarrow X$ un morphisme de $C$. Le foncteur changement
de base $g^\ast:J(X)\rightarrow J(Y)$ définit une application:
$$
LF(g):LF(X)\longrightarrow LF(Y){\quoi}.
$$
qui fait de $X\mapsto LF(X)$ un préfaisceau sur $C$.

Le morphisme $\id_X:X\rightarrow X$ étant un élément de $J(X)$
on a, pour tout objet $X$ de $C$, une application:
$$
\ell(F)(X):F(X)\longrightarrow LF(X){\quoi},
$$
définissant visiblement un morphisme de foncteurs:
$$
\ell(F):F\longrightarrow LF{\quoi}.
$$
Il est clair de plus que $F\mapsto LF$ est un foncteur en $F$ et que
les morphismes $\ell(F)$ définissent un morphisme
$$
\ell:\Id\longrightarrow L{\quoi}.
$$
Enfin soit $R\hookrightarrow X$ un raffinement de $X$. La
définition de $LF(X)$ et (\hyperrefext[I][I.prop1.4]{1.4}) nous fournissent une
application:
$$
Z_R:\Hom_{C^{\widehat{\phantom{,}}}}(R,F)\longrightarrow
\Hom_{C^{\widehat{\phantom{,}}}}(X,LF){\quoi},
$$
et\pageoriginale pour tout morphisme de $C$, $Y\xrightarrow{g}X$, la
définition du foncteur $LF$ nous montre que le diagramme ci-après,
est commutatif:
\[\vcenter{
\xymatrix{ \qquad Z_R :& \Hom_{C^{\widehat{\phantom{a}}}}(R,F)
\ar[r]\ar[d] &
\Hom_{C^{\widehat{\phantom{a}}}}(X,LF)\ar[d]\\
Z_{\fprod{R}{Y}{X}}:&  \Hom_{C^{\widehat{\phantom{a}}}}(\fprod{R}{Y}{X},F) \ar[r]&  \Hom_{C^{\widehat{\phantom{a}}}}(Y,LF)\\
}}\leqno{(*)}
\]
(les flèches verticales sont les flèches évidentes).

\noindent Copions alors [2].

\begin{enonce*}{Lemme 3.1}\etiquette[3.1]{II.lem3.1}
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Pour tout raffinement $i_R:R\hookrightarrow X$ et tout
  $u:R\rightarrow F$, le diagramme:
\[\vcenter{
\xymatrix@L=5pt@R=1.35cm@C=3.5cm@M=.2cm{
F\ar[r]^{\ell(F)} & LF\\
R\ar[u]^{u}\ar@{^{(}->}_{i_R}[r] & X \ar[u]_{Z_R(u)} }}\leqno{(**)}
\]
est commutatif.

\item Pour tout morphisme $v:X\rightarrow LF$, il existe un
  raffinement $R$ de $X$ et un morphisme $u:R\rightarrow F$ tel que
  $Z_R(u)=v$.

\item Soient $Y$ un objet de $C$ et $u$, $v:Y\rightrightarrows F$ deux
  morphismes tels que $\ell(F)\circ u=\ell(F)\circ v$. Alors le noyau
  du couple $(u,v)$ est un raffinement de $Y$.

\item Soient $R$ et $R'$ deux raffinements de $X$, $u:R\rightarrow F$
  et $u':R'\rightarrow F$ deux morphismes. Pour que
  $Z_R(u)=Z_{R'}(u')$, il faut et il suffit que $u$ et $u'$
  co{ï}ncident sur un raffinement $R''\hookrightarrow R\times_X R'$.
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
La seule assertion non triviale est l'assertion 1). Il faut montrer
que $Z_R(u)\circ i_R=\ell(F)\circ u$. Pour cela il suffit de montrer
(\hyperrefext[I][I.prop3.4]{3.4}) que les composés de ces morphismes avec tout
morphisme $Y\xrightarrow{g}R$ ($Y$ objet de\pageoriginale $C$) sont
égaux. Or, considérons $f=i_R\circ g$ et le produit fibré $R\times_X
Y$:
\[\vcenter{
\xymatrix@L=5pt@R=1cm@C=3.5cm@M=.2cm{
F  \ar[r]^{\ell(F)}   & LF \\
R \ar[u]^{u}        \ar@{^{(}->}_{i_R}[r]  & X \ar[u]_{Z_R(u)} \\
\fprod{R}{Y}{X} \ar[u]^{g'} \ar[r]^(.35){\sim}_{i'} & Y
\ar[lu]^{g}\ar[u]_{f}. }}
\]

Dans ce diagramme ci-dessus, $i'$ est un monomorphisme qui admet une
section. Par suite (\hyperref[II.def1.3]{1.3} 3) $i'$ est un isomorphisme. Or
par définition du morphisme $\ell(F)$, le morphisme $Z_{R\times_X
  Y}(u\circ g')$ est égal à $\ell(F)\circ u\circ g$. D'autre part
la commutativité du diagramme (*) nous fournit l'égalité
$Z_{R\times_X Y}(u\circ g')=Z_R(u)\circ f$, et par suite on a bien
l'égalité $\ell(F)\circ u\circ g=Z_R(u)\circ i_R \circ g$.
\end{enonce*}

\begin{enonce*}{Proposition 3.2}\etiquette[3.2]{II.prop3.2}
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Le foncteur $L$ est exact à gauche {\rm(\hyperrefext[I][I.cor2.3.2]{2.3.2}).}

\item Pour tout préfaisceau $F$, $LF$ est un préfaisceau
  séparé.

\item Le préfaisceau $F$ est séparé si et seulement si le
  morphisme $\ell(F):F\rightarrow LF$ est un monomorphisme. Le
  préfaisceau $LF$ est alors un faisceau.

\item Les propriétés suivantes sont équivalentes:
\begin{enumerate}
\renewcommand{\theenumii}{\roman{enumii}}
\renewcommand{\labelenumii}{(\theenumii)}
\item $\ell(F):F\rightarrow LF$ est un isomorphisme.

\item $F$ est un faisceau.
\end{enumerate}
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Il suffit de montrer (\hyperrefext[I][I.prop3.1]{3.1}) que pour tout objet $X$ de $C$, le
  foncteur $F\mapsto LF(X)$ commute aux limites projectives finies. Or
  par définition de la limite projective, le foncteur
$$
F\longmapsto \Hom_{C^{\widehat{\phantom{,}}}}(R,F)\quad
R\hookrightarrow X\in \ob (J(X))
$$
commute\pageoriginale aux limites projectives, et la limite
inductive $\varinjlim_{J(X)}$ commute aux limites projectives finies
car $J(X)$ est une ensemble cofiltrant (\hyperrefext[I][I.def2.7]{2.7}).

\item Soient $X$ un objet de $C$ et $f$, $g:X\rightrightarrows LF$
  deux morphismes qui coïncident sur un raffinement
  $R\hookrightarrow X$ de $X$. En vertu de \hyperref[II.lem3.1]{3.1}\quad
  2), il existe un crible couvrant $R'\hookrightarrow
  X$, qu'on peut toujours supposer contenu dans $R$, et deux
  morphismes $u$, $v:R'\rightrightarrows F$ tels que $Z_{R'}(u)=f$ et
  $Z_{R'}(v)=g$. En vertu de \hyperref[II.lem3.1]{3.1} 1) on a alors $\ell(F)\circ
  u=\ell(F)\circ v$. Par suite (\hyperref[II.lem3.1]{3.1} 4)) $u$ et $v$ coïncident sur un
  raffinement $R''\hookrightarrow R'$. Soit $w$ le
  restriction de $u$ à $R''$. On a $Z_{R''}(w)=Z_{R'}(u)=Z_{R'}(v)$
  et par suite $f=g$. Le préfaisceau $LF$ est donc séparé.

\item Si $F$ est séparé, le morphisme $\ell(F)$ est un
  monomorphisme car une limite inductive filtrante de monomorphismes
  est un monomorphisme. Si $\ell(F)$ est un monomorphisme, le

  séparé, donc il est séparé. Montrons que $LF$ est alors un
  faisceau. Soient $i:R\hookrightarrow X$ un crible couvrant un objet

3.4) que pour tout morphisme $m:Y\rightarrow R$ ($Y$ objet de $C$),

$p_1$ et $p_2$ les projections. La projection $\xymatrix@M=5pt{ R''
\ar@{^{(}->}[r]^{p_2}& Y }$ est un monomorphisme et fait de $R''$ un
crible couvrant $Y$ (\hyperref[II.lem3.1]{3.1} 2)). On a $v\circ i\circ

\section{Propriétés d'exactitude de la catégorie des faisceaux}\etiquette[4]{II.sec4}

Les propriétés d'exactitude de la catégorie des faisceaux se
déduisent des propriétés d'exactitude de la catégorie des
préfaisceaux via le théorème \hyperref[II.theo3.4]{3.4}. La présent
numéro explicite cette philosophie sur des énoncés-types parmi
les plus utiles.

\begin{enonce*}{Théorème 4.1}\etiquette[4.1]{II.theo4.1}
Soient $C$ un $𝒰$-site, $\widetilde{C}$ la catégorie des
faisceaux, $\underline{a}:C^{\widehat{\phantom{,}}}\rightarrow
\widetilde{C}$ le foncteur faisceau associé,
$i:\widetilde{C}\rightarrow C^{\widehat{\phantom{,}}}$ le foncteur
d'inclusion.
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Le foncteur $\underline{a}$ commute aux limites inductives et est exact.

\item Les\pageoriginale $𝒰$-limites inductives dans
$\widetilde{C}$ sont
  représentables. Pour toute petite catégorie $I$ et pour tout
  foncteur $E:I\rightarrow \widetilde{C}$, le morphisme canonique
$$
\varinjlim_I E\longrightarrow \underline{a}(\varinjlim_I i\circ E)
$$
est un isomorphisme.

\item Les $𝒰$-limites projectives dans $\widetilde{C}$ sont
  représentables. Pour tout objet $X$ de $C$, le foncteur sur
  $\widetilde{C}:F\mapsto F(X)$ commute aux limites projectives, \ie
  le foncteur d'inclusion $i:\widetilde{C}\rightarrow \widehat{C}$
  commute aux limites projectives.
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
Ces propriétés résultent essentiellement du théorème
\hyperref[II.theo3.4]{3.4} et de (\hyperrefext[I][I.prop2.11]{2.11}).
\end{enonce*}

Ainsi, dans la catégorie des faisceaux, les produits indexés par
un élément de $𝒰$, produits fibrés, sommes indexées
par un élément de $𝒰$, sommes amalgamées, noyaux,
conoyaux, images, coimages sont représentables.

\begin{enonce*}{Corollaire 4.1.1}\etiquette[4.1.1]{II.cor4.1.1}
Soient $C$ un $𝒰$-site et $F$ un faisceau d'ensembles sur
$C$. L'homomorphisme canonique
$$
\varinjlim_{C/F} \underline{a}X\longrightarrow F
$$
est un isomorphisme.
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
Résulte de (\hyperrefext[I][I.prop3.4]{3.4}) et du fait que $\underline{a}$ commute
aux limites inductives.
\end{enonce*}

\begin{enonce*}{Proposition 4.2}\etiquette[4.2]{II.prop4.2}
Tout morphisme de $\widetilde{C}$, qui est à la fois un
épimorphisme et un monomorphisme, est un isomorphisme.
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
Soit $u:G\rightarrow H$ un morphisme de $\widetilde{C}$ qui est un
épimorphisme et un monomorphisme. Remarquons tout d'abord que la
morphisme $u$ est un monomorphisme de\pageoriginale préfaisceaux
(\hyperref[II.theo4.1]{4.1} 1)). Construisons la somme amalgamée $H\amalg_G
H=K$ et les deux morphismes canoniques $i_1$,
$i_2:H\rightrightarrows K$ dans la catégorie des préfaisceaux.
Comme $u$ est un monomorphisme de préfaisceaux, on vérifie
immédiatement que le diagramme ci-après est cartésien et
cocartésien (\hyperrefext[I][I.sec10.1]{10.1}):
\[
\xymatrix@C=3cm@R=1.5cm{
G \ar[r]^{u} \ar[d]_{u}& H \ar[d]^{i_2}\\
H \ar[r]^{i_1}& K . }
\]
En appliquant le foncteur « faisceau associé », on obtient
donc un diagramme cartésien et cocartésien de la catégorie des
faisceaux (\hyperref[II.theo4.1]{4.1} 1)):
\[\vcenter{
\xymatrix@C=3cm@R=1.5cm{
G \ar[r]^{u} \ar[d]_{u}& H \ar[d]^{\underline{a}i_2}\\
H \ar[r]^{\underline{a}i_1}& \underline{a}K . }}\leqno{(*)}
\]
Comme $u$ est un épimorphisme de faisceaux, le morphisme
$\underline{a}i_1$ est un isomorphisme et comme le diagramme (*) est
cartésien, le morphisme $u$ est un isomorphisme.
\end{enonce*}

\begin{enonce*}{Proposition 4.3}\etiquette[4.3]{II.prop4.3}
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Les limites inductives dans $C^{\widetilde{\phantom{,}}}$ qui
  sont représentables, sont universelles {\rm (\hyperrefext[I][I.def2.5]{2.5}).}

\item Toute famille épimorphique {\rm(\hyperrefext[I][I.sec10.2]{10.2})} de morphismes est
  épimorphique effective universelle (\hyperref[II.prop2.6]{2.6}).

\item Toute relation d'équivalence est effective universelle {\rm(I
  10.6).}

\item Les $𝒰$-limites inductives filtrantes commutent aux
  limites projectives finies {\rm (\hyperrefext[I][I.prop2.6]{2.6})}.
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
Les\pageoriginale assertions 1) à 4) sont vraies dans la catégorie
des ensembles donc dans la catégorie des préfaisceaux
(\hyperrefext[I][I.prop3.1]{3.1}). Les assertions 1) et 4) résultent alors
immédiatement des assertions correspondantes pour les préfaisceaux
et de \hyperref[II.theo4.1]{4.1}. Démontrons 3). Soit $p_1$, $p_2:$
$\xymatrix{R\ar@<2pt>[r] \ar@<-2pt>[r]& X}$ une relation
d'équivalence de $C^{\widetilde{\phantom{,}}}$. Le diagramme
$\xymatrix{R \ar@<2pt>[r]^{p_1} \ar@<-2pt>[r]_{p_2}&X}$ est alors
une relation d'équivalence de préfaisceaux (\hyperref[II.cor4.1.1]{4.1.1})). Il
existe donc un morphisme de préfaisceaux
       $u:X\rightarrow Y$ tel que le diagramme
\[
\xymatrix@C=3cm@R=1.25cm{
R \ar[r]^{p_2} \ar[d]_{p_1}& X\ar[d]^{u}\\
X \ar[r]^{u}& Y }
\]
soit cartésien et cocartésien dans la catégorie des
préfaisceaux. En appliquant le foncteur « faisceau associé », on obtient un diagramme cartésien et cocartésien dans la
catégorie des faisceaux (\hyperref[II.theo4.1]{4.1} 1)). Par suite
$\xymatrix{R \ar@<2pt>[r]^{p_1} \ar@<-2pt>[r]_{p_2}&X}$ est une
relation d'équivalence effective. Comme toute relation
d'équivalence est effective d'après ce qui précède, toute
relation d'équivalence est effective universelle. Démontrons 2).
Soit $(u_i:X_i\rightarrow X)$, $i\in I$, une famille épimorphique
de $C^{\widetilde{\phantom{,}}}$. D'après (II\ \hyperref[II.prop2.6]{2.6}) et
(\hyperrefext[I][I.prop2.12]{2.12}), il suffit de démontrer les assertions
suivantes:
\begin{enumerate}
\renewcommand{\theenumi}{\alph{enumi}}
\renewcommand{\labelenumi}{\theenumi)}
\item Pour tout morphisme de faisceaux $v:Y\rightarrow X$, la famille

  une famille épimorphique de $C^{\widetilde{\phantom{,}}}$.

\item Pour tout faisceau $Z$ le diagramme d'ensembles:

\end{enumerate}

Soient\pageoriginale $X'$ la réunion des images des morphismes
$u_i$ au sens des préfaisceaux et $j:X'\hookrightarrow X$
l'injection canonique. Pour tout $i\in I$, le morphisme
$u_i:X_i\rightarrow X$ se factorise en un morphisme
$u'_i:X_i\rightarrow X'$ et le morphisme $j$, et la famille
$u'_i:X_i\rightarrow X'$, $i\in I$, est une famille épimorphique
de $C^{\widehat{\phantom{,}}}$. Comme $j$ est un monomorphisme, le
morphisme $\underline{a}i:\underline{a}X'\rightarrow X$ est un monomorphisme

faisceaux. Par suite $\underline{a}j$ est un isomorphisme
(\hyperref[II.prop4.2]{4.2}). Soit $v:Y\rightarrow X$ un morphisme de
faisceaux. On en déduit par changement de base un morphisme
$j_v:X'\times_X Y\rightarrow Y$ et des morphismes
$u'_{i,v}:X_i\times_X Y\rightarrow X'\times_X Y$. Comme $\underline{a}$
commute aux produits fibrés, $\underline{a}j_v$ est un isomorphisme.

conservent ce caractère par changement de base, la famille
$u'_{i,v}$, $i\in I$, est épimorphique dans
$C^{\widehat{\phantom{,}}}$. Comme $\underline{a}$ commute aux limites
inductives, la famille $\underline{a}u'_{i,v}:X_i\times_X Y\rightarrow

un faisceau. Comme $\underline{a}j$ est un isomorphisme, l'application

canoniquement isomorphe à $X_i\times_X X_k$, d'où b).

\begin{enonce*}{Théorème 4.4}\etiquette[4.4]{II.theo4.4}
Soient $C$ un $𝒰$-site et $(u_i:X_i\rightarrow X)$, $i\in
I$, une famille de morphismes de $C$ de but $X$. Les conditions
suivantes sont équivalentes \footnote[1]{Lorsque le site $C$ ne
possède pas nécessairement de petite famille topologiquement
génératrice et lorsque la topologie de $C$ est moins fine que la

de \hyperref[II.theo4.4]{4.4}).}:
\begin{enumerate}
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\theenumi)}
\item La famille $(\epsilon_C u_i:\epsilon_C X_i\rightarrow \epsilon_C
  X)$, $i\in I$, est une famille épimorphique de
  $C^{\widetilde{\phantom{,}}}$ {\rm(\hyperrefext[I][I.sec10.2]{10.2}).}\pageoriginale

\item La famille $(u_i:X_i\rightarrow X)$, $i\in I$, est une famille
  couvrante de $C$ (\hyperref[II.def1.2]{1.2}).
\end{enumerate}

\begin{enonce*}[remark]{Preuve}
ii) $\Rightarrow$ i). Soient $R\hookrightarrow X$ le crible
engendré par les $u_i$, $i\in I$ (\hyperrefext[I][I.sec4.3.3]{4.3.3}), et
$u'_i:X_i\rightarrow R$ les morphismes induits par les $u_i$. La
famille des $u'_i$, $i\in I$, est une famille épimorphique de
$C^{\widehat{\phantom{,}}}$, et $R$ est un crible couvrant $X$. Par
suite, pour tout faisceau $F$, l'application
$$
\Hom(X,F)\longrightarrow \Hom(R,F)
$$
est bijective et l'application
$$
\Hom(R,F)\longrightarrow \prod_i \Hom(X_i,F)
$$
est injective. Donc l'application
$$
\Hom(X,F)\longrightarrow \prod_i\Hom(X_i,F)
$$
est injective et par suite l'application
$$
\Hom(\underline{a}X,F)\longrightarrow \prod_i \Hom(\underline{a}X_i,F)
$$
est injective, d'où i).

\noindent i) $\Rightarrow$ ii). Avec les notations introduites
précédemment, soit $J_R:R\hookrightarrow X$ l'injection
canonique dans $X$ du crible engendré par les $u_i$, $i\in I$. Il
résulte de i) que le morphisme de faisceaux
$\underline{a}J_R:\underline{a}R\rightarrow \underline{a}X$ est à la fois un
épimorphisme de faisceaux et un monomorphisme de faisceaux. C'est
donc un isomorphisme de faisceaux (\hyperref[II.prop4.2]{4.2}). On a, avec les
notations du ${\rm n^{\circ}}$ 3, un diagramme commutatif:
\[
\xymatrix@C=2.5cm@R=1.5cm@M=.25cm{
X \ar[r]^{\ell(x)} & LX \ar[r]^{\ell(LX)} & \mathsf{\underline{a}}X\\
R \ar[r]^{\ell(R)} \ar@{^{(}->}[u]^{J_R}& LR \ar[r]^{\ell(LR)}
\ar@{^{(}->}[u]^{LJ_R}& \mathsf{\underline{a}}R
\ar[u]^{\wr}_{\mathsf{\underline{a}}J_R}. }

Tout\pageoriginale d'abord, d'après \hyperref[II.lem3.1]{3.1} 2), il existe un
crible couvrant $J_{S_1}:S_1\rightarrow X$ et un morphisme
$u_1:S_1\rightarrow LR$ tel que
\begin{equation}
\ell(LX)\circ \ell (X)\circ J_{S_1}=\underline{a}J_R\circ \ell (LR)\circ
u_1=\ell(LX)\circ LJ_R\circ u_1{\quoi}.\tag{4.6.1}

Posons $m=\ell(X)\circ J_{S_1}$ et $n=LJ_R\circ u_1$. Les morphismes
$m$ et $n$ sont des morphismes de $S_1$ dans $LX$. Soit
$S_2\hookrightarrow S_1$ le noyau du couple de flèches $(m,n)$. Pour

X$ est un crible couvrant $X$. On a donc un morphisme
$u_2:S_2\rightarrow LR$ et un diagramme commutatif:
\[\vcenter{
\xymatrix@C=2.5cm@R=.8cm@M=.2cm{
S_2 \ar@{^{(}->}[rr]^{J_{S_{2}}} \ar[dd]_{u_2}&  &X \ar[dd]^{\ell(X)}\\
                  &R\ar@{^{(}->}[ru]_{J_R} \ar[ld]^{\ell(R)}&\\
LR \ar[rr]^{LJ_R}  &  &LX&. }}\leqno{(4.6.2)}

\begin{enonce*}{Définition 4.5}\etiquette[4.5]{II.def4.5}
Un objet initial d'une catégorie $A$ est un objet $\phi_A$ qui
représente la limite inductive vide \ie tel que pour tout $X\in
\ob A$, il existe une flèche et une seule $\phi_A\rightarrow X$. On
dit qu'un objet $\phi_A$ est \emph{initial strict} s'il est initial
et si tout morphisme de but $\phi_{A}$ est un isomorphisme. Soit
$(S_i)$, $i\in I$, une famille d'objets d'une catégorie $A$.
Supposons que la somme $s=\amalg_{i\in I} S_i$ soit représentable.
On dit que la somme $S$ est disjointe si les morphismes structuraux
$S_i\rightarrow S$ sont quarrables, s'ils sont des monomorphismes et
si pour tout couple $i$, $j$, $i\neq j$, le produit $S_i\times_S
S_j$ est un objet initial de $A$. On dit que la somme $S$ est
disjointe universelle\pageoriginale si elle est disjointe et si elle
reste somme disjointe après tout changement de base $T\rightarrow
S$; il en résulte que pour tout couple $i$, $j$, $i\neq j$, les
objets $S_i\times_S S_j$ sont des objets initiaux stricts de $A$.
\end{enonce*}

\begin{enonce*}[remark]{Exemple 4.5.1}\etiquette[4.5.1]{II.exem4.5.1}
Dans la catégorie des ensembles, les sommes directes sont
disjointes et universelles. Il en est donc de même dans toute
catégorie de préfaisceaux d'ensembles (\hyperrefext[I][I.prop3.1]{3.1}) et par
suite dans toute catégorie de faisceaux d'ensembles sur un
$𝒰$-site (\hyperref[II.theo4.1]{4.1}); en particulier, l'objet initial
de $C^{\widetilde{\phantom{,}}}$ est strict.
\end{enonce*}

\begin{enonce*}{Proposition 4.6}\etiquette[4.6]{II.prop4.6}
Soient $C$ une $𝒰$-catégorie et $(s_i:X_i\rightarrow X)$,
$i\in I$, une famille de morphismes de $C$. Pour toute
$𝒰$-topologie $\Tt$ sur $C$ (\hyperref[II.def3.0.2]{3.0.2}), on désigne
par $C^{\widetilde{\phantom{,}}}_{\Tt}$ la catégorie de faisceaux
correspondante et par $\epsilon_{\Tt}:C\rightarrow
C^{\widetilde{\phantom{,}}}_{\Tt}$ le foncteur correspondant.
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Soit $\Tt$ une $𝒰$-topologie telle que
$$
\coprod_{i\in I}
\epsilon_{\Tt}(X_i)\xrightarrow{(\epsilon_{\Tt}(s_i))}\epsilon_{\Tt}(X)
$$
soit un isomorphisme. Alors pour toute topologie $\Tt'$ plus fine
que $\Tt$, le morphisme:
$$
\coprod_{i\in
  I}\epsilon_{\Tt'}(X_i)\xrightarrow{(\epsilon_{\Tt'}(s_i))}\epsilon_{\Tt'}(X)
$$
est un isomorphisme.

\item Soit $\Tt$ une $𝒰$-topologie sur $C$. Les
  propriétés suivantes {\rm(i)} et {\rm(ii)} sont équivalentes:
\begin{itemize}
\item[i)]
\begin{itemize}
\item[a)] La famille $(s_i:X_i\rightarrow X)$, $i\in I$, est couvrante
  pour $\Tt$.

\item[b)] Pour tout $i\in I$, le morphisme diagonal de préfaisceaux
  $\Delta_i:X_i\hookrightarrow X_i\times_X X_i$\pageoriginale
  est transformé par
  le foncteur « faisceau associe » (pour $\Tt$) en un
  isomorphisme (ce qui est le cas si les $s_i$ sont des
  monomorphismes).

\item[c)] Pour tout couple $(i,j)$, $i\neq j$, d'éléments de $I$, le
  préfaisceau. $X_i\times_X X_j$ est transformé par le foncteur
  « faisceau associé » (pour $\Tt$) en l'objet initial
  de $C^{\widetilde{\phantom{,}}}_{\Tt}$.
\end{itemize}
\item[ii)] $\epsilon_{\Tt}(X)$ est somme de $\epsilon_{\Tt}(X_i)$.
\end{itemize}
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Un faisceau $F$ pour $\Tt'$ est un faisceau pour
  $\Tt$. Pas suite, pour tout faisceau $F$ pour
  $\Tt'$, le morphisme
$$
\Hom_{C^{\widehat{\phantom{,}}}}(X,F)\longrightarrow \prod_{i\in I}
\Hom_{C^{\widehat{\phantom{,}}}}(X_i,F)
$$
est un isomorphisme, ce qui entraîne l'assertion.

\item Par définition, on a la propriété ii) si et seulement si
  le morphisme de préfaisceaux
$$
\Phi=(h(s_i),i\in I):\coprod_i h(X_i)\longrightarrow h(X)
$$
est transformé par le foncteur « faisceau associé » en un
isomorphisme \ie (\hyperref[II.prop4.2]{4.2}) si et seulement si
$\underline{a}(\Phi)$ est un épimorphisme et monomorphisme de faisceaux.
D'après (\hyperref[II.theo4.4]{4.4}), le morphisme $\underline{a}(\Phi)$ est un
épimorphisme si et seulement si on a la propriété a). Le morphisme
diagonal
$$
\Delta:\coprod_i h(X_i)\longrightarrow (\coprod_i
h(X_i))\times_{h(X)}(\coprod_i h(X_i))
$$
est somme directe d'une famille $\Delta_{i,j}$, $(i,j)\in i\times
I$, de morphismes définis comme suit:
\begin{itemize}
\item[$\beta)$] Lorsque $i=j$, $\Delta_{i,i}$ est le morphisme
  diagonal $h(X_i)\rightarrow h(X_i)\times_{h(X)}h(X_i)$

\item[$\gamma)$] Lorsque $i\neq j$, $\Delta_{i,j}$ est le morphisme
  $\phi_{C^{\widehat{\phantom{,}}}}\rightarrow
  h(X_i)\times_{h(X)}h(X_j)$ ($\phi_{C^{\widehat{\phantom{,}}}}$
  désigne l'objet initial de $C^{\widehat{\phantom{,}}}$).
\end{itemize}
\end{enumerate}
\end{enonce*}

Le\pageoriginale morphisme $\underline{a}(\Phi)$ est un monomorphisme si
et seulement si $\underline{a}(\Delta)$ est un isomorphisme, et d'après
(\hyperref[II.theo4.1]{4.1}) $\underline{a}(\Delta)$ est isomorphisme si et
seulement si les $\underline{a}(\Delta_{i,j})$ sont des isomorphismes
i.e. si et seulement si on a les propriétés b) et c).

\begin{enonce*}{Corollaire 4.6.1}\etiquette[4.6.1]{II.cor4.6.1}
Soient $C$ une $𝒰$-catégorie et $X$ un objet de $C$.
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Soit $\Tt$ une $𝒰$-topologie sur $C$. L'objet
  $X$ de $C$ est transformé par le foncteur « faisceau associé
  » (pour la topologie $\Tt$) en l'objet initial de
  $C^{\widetilde{\phantom{,}}}_{\Tt}$ si et seulement si le
  crible vide recouvre $X$.

\item Lorsque $X$ est un objet initial strict de $C$, le faisceau
  associé à $X$, pour toute $𝒰$-topologie plus fine que la
  topologie canonique, est un objet initial.
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}

\renewcommand{\labelenumi}{\theenumi)}

\item D'après 1) et \hyperref[II.prop4.6]{4.6} 1), il suffit de montrer que le crible vide
  recouvre $X$ pour la topologie canonique \ie (\hyperref[II.prop2.6]{2.6}) il suffit de

\begin{enonce*}{Corollaire 4.6.2}\etiquette[4.6.2]{II.cor4.6.2}

\begin{itemize}

\item[$\beta)$] Pour tout $i\in I$, $s_i:X_i\rightarrow X$ est un

\item[$\gamma)$] Pour tout couple $(i,j)$, $i\neq j$, d'éléments
  de $I$, $X_i\times_X X_J$ est recouvert par le crible vide, \ie pour
  tout faisceau $F$ sur $C$, $F(X_i\times_X X_J)$ est un ensemble
  réduit à un élément.
\end{itemize}
Alors $\epsilon_C(X)$ est somme des $\epsilon_C(X_i)$, $i\in I$.
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
Résulte\pageoriginale immédiatement de \hyperref[II.prop4.6]{4.6} 2) et de
\hyperref[II.cor4.6.1]{4.6.1} 1).
\end{enonce*}

\begin{enonce*}{Corollaire 4.6.3}\etiquette[4.6.3]{II.cor4.6.3}
Soient $C$ une $𝒰$-catégorie et $(s_i:X_i\rightarrow X)$,
$i\in I$, une famille de morphisme quarrables de même but.
Supposons que la topologie canonique de $C$ soit une
$𝒰$-topologie. Les conditions suivantes sont équivalentes:
\begin{enumerate}
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\theenumi)}
\item Il existe une $𝒰$-topologie $\Tt$ sur $C$, moins
  fine que la topologie canonique, telle que
  $\epsilon_{\Tt}(X)$ soit somme des
  $\epsilon_{\Tt}(X_i)$, $i\in I$.

\item L'objet $X$ est somme disjointe et universelle (\hyperref[II.def4.5]{4.5}) (dans
  $C$) des $X_i$.
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
ii) $\Rightarrow$ i). Comme $X$ est somme universelle des $X_i$,
$i\in I$, la famille des $s_i$, $i\in I$, est couvrante pour la
topologie canonique de $C$ (\hyperref[II.prop2.6]{2.6}). La condition
$\alpha$) de \hyperref[II.cor4.6.2]{4.6.2} est donc satisfaite lorsqu'on munit
$C$ de la topologie canonique. La condition $\beta$) est
évidemment satisfaite et la propriété $\gamma$) résulte de
\hyperref[II.prop4.6]{4.6} 1) 2), d'où i).

\noindent i) $\Rightarrow$ ii). Soit $\Tt$ une topologie sur $C$
moins fine que la topologie canonique telle que $\epsilon_{\Tt}(X)$
soit somme des $\epsilon_{\Tt}(X_i)$. Pour tout faisceau $F$, pour
$\Tt$, l'application canonique $\Hom(X,F)\rightarrow
\prod_i\Hom(X_i,F)$ est une bijection. De plus, tout préfaisceau
représentable est un faisceau. Il en résulte aussitôt que $X$ est
somme dans $C$ des $X_i$, $i\in I$. Appliquant ceci au cas où
l'ensemble $I$ est vide, on voit que si un objet de $C$ est
transformé en l'objet initial de $C^{\widetilde{\phantom{,}}}$, cet
objet est un objet initial de $C$. Le foncteur
$\epsilon_{\Tt}:C\rightarrow C^{\widetilde{\phantom{,}}}$ est
pleinement fidèle et commute aux limites projectives finies. Par
suite la condition b) de \hyperref[II.cor4.6.2]{4.6.2}) entraîne que les $s_i$,
$i\in I$, sont des monomorphismes de $C$, et la condition c)
entraîne, d'après de qui précède, que les $X_i\times_X X_j$, $i\neq
j$, sont des objets initiaux de $C$. Par suite $X$ est somme
disjointe des\pageoriginale $X_i$. Comme $\epsilon_{\Tt}$ commute
aux produits fibrés, cette dernière propriété est stable par tout
changement de base $Y\rightarrow X$, et par suite $X$ est somme
disjointe et universelle des $X_i$, $i\in I$.
\end{enonce*}

\begin{enonce*}{Corollaire 4.6.4}\etiquette[4.6.4]{II.cor4.6.4}
Soient $C$ une $𝒰$-catégorie et $X$ un objet de $C$.
Supposons que la topologie canonique sur $C$ soit une
$𝒰$-topologie. Les propriétés suivantes sont
équivalentes:
\begin{enumerate}
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\theenumi)}
\item Il existe sur $C$ une topologie moins fine que la topologie
  canonique telle que $\epsilon_{\Tt}(X)$ soit un objet
  initial de $C^{\widetilde{\phantom{,}}}_{\Tt}$.

\item $X$ est un objet initial strict de $C$.
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
On prend pour ensemble $I$ l'ensemble vide dans 4.6.3.
\end{enonce*}

\begin{enonce*}{Proposition 4.7}\etiquette[4.7]{II.prop4.7}
Soient $C$ une $𝒰$-catégorie,
$C^{\widetilde{\phantom{,}}}$ la catégorie des faisceaux sur $C$
pour la topologie canonique, $\epsilon_C:C\rightarrow
C^{\widetilde{\phantom{,}}}$ le foncteur canonique
(\hyperref[II.subsubsec4.4.0]{4.4.0}), $R\rightrightarrows X$ une relation
d'équivalence dans $C$ admettant un conoyau $Y$ dans $C$. Soit
$\Pi:X\rightarrow Y$ le morphisme canonique et supposons-le
quarrable. Considérons alors les propriétés suivantes:
\begin{enumerate}
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\theenumi)}
\item $\epsilon_C(Y)$ est le quotient de la relation d'équivalence
  $\epsilon_C(R)\rightrightarrows \epsilon_C(X)$.

\item La relation d'équivalence $R\rightrightarrows X$ est effective
  universelle {\rm(\cf ${\rm n^{\circ}}$ 7).}
\end{enumerate}
On a {\rm{ii) $\Rightarrow$ i).}}. Lorsque $C$ possède une
$𝒰$-topologie moins fine que la topologie canonique, on a
{\rm{i) $\Rightarrow$ ii).}}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
ii) $\Rightarrow$ i). Le morphisme canonique $R\rightarrow X\times_X
X$ est un isomorphisme. Soit $S\rightarrow Y$ le crible engendré
par $\pi:X\rightarrow Y$. Pour tout préfaisceau $F$, on a un
diagramme exact (\hyperrefext[I][I.prop2.12]{2.12}):
$$
\Hom(S,F)\rightarrow \Hom(X,F)\rightrightarrows \Hom(R,F){\quoi}.
$$
Il\pageoriginale suffit donc de montrer que pour tout faisceau $F$
pour la topologie canonique, l'application
$$
\Hom(X,F)\longrightarrow \Hom(S,F)
$$
est une bijection \ie il suffit de montrer que $S$ est couvrant pour
la topologie canonique, ou encore que le crible engendré par
$\pi:X\rightarrow Y$ est couvrant, ce qui résulte de la
définition 2.5.

\noindent i) $\Rightarrow$ ii). Le foncteur $\epsilon_C:C\rightarrow
C^{\widetilde{\phantom{,}}}$ commute aux limites projectives finies
et est pleinement fidèle. Or le morphisme canonique
$\epsilon_C(R)\rightarrow J_C(X)\times_{J_C(Y)}J_C(X)$ est un
isomorphisme (\hyperref[II.prop4.3]{4.3}). Par suite le morphisme canonique
$R\rightarrow X\times_Y X$ est un isomorphisme. Le morphisme
$\epsilon_C(X)\rightarrow \epsilon_C(Y)$ est un épimorphisme, ce
qui entraîne (\hyperref[II.theo4.4]{4.4}) que $\pi:X\rightarrow Y$ est un
morphisme couvrant de $C$ pour la topologie canonique, \cqfd
\end{enonce*}

\begin{enonce*}{Proposition 4.10}\etiquette[4.10]{II.prop4.10}
Avec les notations précédentes, considérons le foncteur
canonique $\epsilon:C\rightarrow \widetilde{C}$
(\hyperref[II.subsubsec4.4.0]{4.4.0}), et soit $G\subset \ob C$ une famille
topologiquement génératrice dans $C$ (\hyperref[II.def3.0.1]{3.0.1}). Alors
la famille $(\epsilon(X))_{X\in G}$ d'objets de $\widetilde{C}$ est
une famille génératrice. En particulier la famille
$(\epsilon(X))_{X\in \ob C}$ d'objets de $C$ est génératrice.
\end{enonce*}

Il faut prouver que tout morphisme $u:F\rightarrow F'$ dans
$\widetilde{C}$ tel que
\begin{equation}
\Hom(\epsilon(X),F)\longrightarrow \Hom(\epsilon(X),F')\tag{4.10.1}
\end{equation}\etiquette[4.10.1]{II.eq4.10.1}
soit une bijection pour tout $X\in G$, est un isomorphisme. Or
l'application précédente s'identifie à l'application
\begin{equation}
u(X):F(X)\longrightarrow F'(X){\quoi},\tag{4.10.2}
\end{equation}\etiquette[4.10.2]{II.eq4.10.2}
et il faut montrer que si celle-ci est bijective pour tout $X\in G$,
il en est de même pour tout $X\in \ob C$. Or soit $(X_i\rightarrow
X)_{i\in I}$ une famille couvrante de $X$ par des objets de $G$, et
pour tout couple d'indices $(i,j)$ de $I$, considérons l'ensemble
de tous les morphismes $X_{ijk}\rightarrow X_i\times_XX_j$ dans
$\widehat{C}$, avec $X_{ijk}\in G$ ($k$ variant dans un ensemble
d'indices $I_{i,j}$). On obtient alors un homomorphisme de
diagrammes exacts d'ensembles
\[
\xymatrix{ F(X) \ar[r] \ar[d]& \displaystyle{{\prod_i}} F(X_i)
\ar[d]^{\wr} \ar@<3pt>[r] \ar@<-3pt>[r] & \displaystyle{\prod_{ijk}}
F(X_{ijk}) \ar[d]^{\wr}\\
F(X) \ar[r] & \displaystyle{{\prod_i}} F(X_i) \ar@<3pt>[r]
\ar@<-3pt>[r] & \displaystyle{\prod_{ijk}} F(X_{ijk})&, }
\]
où\pageoriginale par hypothèse les deux dernières flèches
verticales sont des bijections. Il en est donc de même de la
première flèche verticale, \cqfd

et admet une $𝒰$-petite famille génératrice.
\end{enonce*}

En effet, par hypothèse on peut prendre dans \hyperref[II.prop4.10]{4.10} pour

petite famille génératrice. D'autre part, si $F$, $F'\in \ob C$, un
homomorphisme de $F$ dans $F'$ est connu quand on connaît
l'homomorphisme \eqhyperref[II.eq4.10.1]{4.10.1} \ie \eqhyperref[II.eq4.10.2]{4.10.2} pour tout
$X\in \ob G$ (I 7.1.1), d'où s'ensuit que l'application
$$
\Hom(F,F')\longrightarrow \prod_{X\in G} \Hom(F(X),F'(X))
$$
est injective. Comme le deuxième membre est $𝒰$-petit, il en
est de même du premier, ce qui prouve que $\widetilde{C}$ est une

\section{Extension d'une topologie de $C$ à
  $C^{\shat{\;\hbox{}}}$}\etiquette[5]{II.sec5}

\begin{enonce*}{Proposition 5.1}\etiquette[5.1]{II.prop5.1}
Soient $C$ un $𝒰$-site et $f:H\rightarrow K$ un morphisme de
$C^{\widehat{\phantom{,}}}$. Les conditions suivantes sont
équivalentes:
\begin{itemize}
\item[i)] Pour tout $X\rightarrow K$, avec $X\in \ob C$, le morphisme
  correspondant $H\times_K X\rightarrow X$ a pour image un crible
  couvrant de $X$.

\item[ii)] Le morphisme $\underline{a}(f):\underline{a}H\rightarrow \underline{a}K$ sur
  les faisceaux associés est une épimorphisme\pageoriginale de
  $C^{\widetilde{\phantom{,}}}$.

\item[ii bis)] Pour tout faisceau $F$ sur $C$, l'application
  $\Hom(K,F)\rightarrow \Hom(H,F)$ déduite de $f$ est injective.
\end{itemize}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
Il est clair que ii) équivaut à ii bis).
\end{enonce*}

Prouvons que (i) $\Rightarrow$ (ii). En vertu de \hyperref[II.theo4.4]{4.4},
(i) signifie que $u:(H\times_K X)\rightarrow \underline{a}(X)$ est un
épimorphisme. Or, comme $\underline{a}(H\times_K X)\simeq
\underline{a}(H)\times_{\underline{a}(K)}\underline{a}(X)$, $\underline{a}$
commutant aux $\varprojlim$ finies (\hyperref[II.theo4.1]{4.1}), le morphisme
envisagé se déduit de $\underline{a}(f):\underline{a}(H)\rightarrow
\underline{a}(K)$ par changement de base. Comme les épimorphismes de
$\widetilde{C}$ sont universels, cela montre que (ii) $\Rightarrow$
(i). Inversement, comme la famille des $X_i\rightarrow K$ est
épimorphique, il en est de même de la famille des
$\underline{a}(X)\rightarrow \underline{a}(K)$ dans $\widetilde{C}$
(\hyperref[II.theo4.1]{4.1}), donc pour vérifier que
$\underline{a}(f):\underline{a}(H)\rightarrow \underline{a}(K)$ est
épimorphique, il suffit de le voir après tout changement de base
du type précédent $\underline{a}(X)\rightarrow \underline{a}(K)$, ce qui
prouve i) $\Rightarrow$ ii), \cqfd

\begin{enonce*}{Définition 5.2}\etiquette[5.2]{II.def5.2}
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item Un morphisme $f:H\rightarrow K$ satisfaisant aux trois
  conditions équivalentes de \hyperref[II.prop5.1]{5.1} est appelé un \emph{morphisme
  couvrant}. Une famille de morphismes $f_i:H_i\rightarrow K$, $i\in
  I$, de même but est dite \emph{couvrante} si le morphisme correspondant
  $f:\amalg_i H_i\rightarrow K$ est couvrant.

\item Un morphisme $f:H\rightarrow K$ est dit \emph{bicouvrant} s'il est
  couvrant et si le morphisme diagonal $H\rightarrow H\times_K H$ est
  couvrant. Une famille $f_i:H_i\rightarrow K$, $i\in I$, de même
  but est dite bicouvrante si le morphisme correspondant $f:\amalg_i
  H_i\rightarrow K$ est bicouvrant.
\end{enumerate}
\end{enonce*}

Par la condition i) de \hyperref[II.prop5.1]{5.1}, dire qu'une famille
$(f_i:H_i\rightarrow K)$, $i\in I$, est couvrante, signifie que pour
tout morphisme $X\rightarrow K$, avec $X\in \ob C$, la famille des
$H_i\times_K X\rightarrow X$, $i\in I$, a pour image un crible
couvrant de\pageoriginale $X$; ou encore, par ii), la famille des
morphismes $\underline{a}(f_i):\underline{a}H_i\rightarrow \underline{a}K$ dans
$C^{\widetilde{\phantom{,}}}$ est épimorphique (compte tenu de ce
que le foncteur $\underline{a}$ commute aux sommes directes
(\hyperref[II.theo4.1]{4.1})).\footnote[1]{Notons que la propriété pour un
morphisme ou
  une famille de morphismes de $C^{\widehat{\phantom{,}}}$ d'être
  couvrant (\resp bicouvrant) est stable par changement de base.}

\begin{enonce*}{Proposition 5.3}\etiquette[5.3]{II.prop5.3}
Soit $C$ un $𝒰$-site et $f:H\rightarrow K$ un morphisme de
$C^{\widehat{\phantom{,}}}$. Les conditions suivantes sont
équivalentes:
\begin{itemize}
\item[i)] Le morphisme $f$ est bicouvrant (\hyperref[II.def5.2]{5.2})).

\item[i bis)] Le morphisme $f$ est couvrant et pour tout objet $X$ de
  $C$ et tout couple de morphismes $u$, $v:X\rightrightarrows H$ tel
  que $fu=fv$, le noyau de $(u,v)$ est un crible couvrant de $X$.

\item[ii)] Le morphisme $\underline{a}(f):\underline{a}H\to \underline{a}K$ est un
  isomorphisme
  de $C^{\widetilde{\phantom{,}}}$.

\item[ii bis)] Pour tout faisceau $F$ sur $C$, l'application
  $\Hom(K,F)\rightarrow \Hom(H,F)$ est une bijection.
\end{itemize}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
L'équivalence i) $\Leftrightarrow$ i bis) résulte de la
condition i) de \hyperref[II.prop5.1]{5.1}, appliquée au morphisme diagonal
$H\rightarrow H\times_K H$; l'équivalence ii) $\Leftrightarrow$ ii
bis) est triviale.

\noindent i) $\Rightarrow$ ii). Le morphisme
$\underline{a}(f):\underline{a}H\rightarrow \underline{a}K$ est un
épimorphisme (\hyperref[II.prop5.1]{5.1}). Comme le foncteur $\underline{a}$
commute à la formation des produits fibrés (\hyperref[II.theo4.1]{4.1}), le
morphisme diagonal $\underline{a}H\rightarrow
\underline{a}H\times_{\underline{a}K}\underline{a}H$ est un épimorphisme
(\hyperref[II.prop5.1]{5.1}). Comme le morphisme diagonal est toujours un
monomorphisme, c'est un isomorphisme (\hyperref[II.prop4.2]{4.2}). Par suite
le morphisme $\underline{a}(f)$ est un monomorphisme. C'est donc un
isomorphisme (\hyperref[II.prop4.2]{4.2}).

\noindent ii) $\Rightarrow$ i). Comme le foncteur $\underline{a}$
commute à la formation des produits fibrés, le morphisme $f$ et le
morphisme diagonal $H\rightarrow H\times_K H$ sont transformés par
$\underline{a}$ en isomorphismes. En particulier, ils sont transformés
par $\underline{a}$ en épimorphismes. Ils sont donc couvrants, \cqfd
\end{enonce*}

\setcounter{subsection}{3} \setcounter{subsubsection}{0}
\subsubsection{}\etiquette[5.3.1]{II.subsubsec5.3.1}
Il\pageoriginale résulte de \hyperref[II.prop5.3]{5.3}, et du fait que
$\underline{a}$ commute aux sommes directes, qu'une famille
$(f_i:H_i\rightarrow K)$, $i\in I$, de morphismes de
$C^{\widehat{\phantom{,}}}$ est bicouvrante si et seulement si les
$f_i$ induisent sur les faisceaux associés un isomorphisme de la
somme directe $\amalg_i \underline{a}H_i$ sur $\underline{a}K$, ou encore si
et seulement si, pour tout faisceau $F$, l'application
$$
\Hom(K,F)\longrightarrow \prod_i \Hom(H_i,F)
$$
est bijective.

\begin{enonce*}{Proposition 5.4}\etiquette[5.4]{II.prop5.4}
Soit $C$ un $𝒰$-site. Il existe sur
$C^{\widehat{\phantom{,}}}$ une topologie (évidemment unique)
telle qu'une famille $H_i\rightarrow K$ de flèches de
$C^{\widehat{\phantom{,}}}$ de même but soit couvrante pour cette
topologie si et seulement si elle est couvrante au sens de
\hyperref[II.def5.2]{5.2}. C'est aussi la topologie la moins fine sur
$C^{\widehat{\phantom{,}}}$ parmi les topologies $T$ ayant les
propriétés suivantes:
\begin{enumerate}
\renewcommand{\theenumi}{\alph{enumi}}
\renewcommand{\labelenumi}{\theenumi)}
\item $T$ est plus fine que la topologie canonique de
  $C^{\widehat{\phantom{,}}}$ (\ie toute famille épimorphique de
  $C^{\widehat{\phantom{,}}}$ est couvrante pour $T$).

\item Toute famille couvrante dans $C$ est couvrante dans
  $C^{\widehat{\phantom{,}}}$.
\end{enumerate}
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
Nous nous bornerons à donner des indications. Nous laissons au
lecteur le soin de montrer que les familles couvrantes au sens de
\hyperref[II.def5.2]{5.2} sont les familles couvrantes d'une topologie $T_C$
sur $C^{\widehat{\phantom{,}}}$ (on utilise \hyperref[II.theo4.1]{4.1}). Les
familles couvrantes de la topologie canonique sur
$C^{\widehat{\phantom{,}}}$ sont les familles épimorphiques sur
$C^{\widehat{\phantom{,}}}$ (\hyperref[II.prop2.6]{2.6} et I 3.1). Comme
$\underline{a}$ commute aux limites inductives (\hyperref[II.theo4.1]{4.1}), la
topologie $T_C$ est plus fine que la topologie canoniques de
$C^{\widehat{\phantom{,}}}$. De plus les familles couvrantes de $C$
sont des familles couvrantes de $T_C$ (\hyperref[II.prop5.1]{5.1}). Si donc
$T'$ désigne la moins fine des topologie de
$C^{\widehat{\phantom{,}}}$ possédant les propriétés a) et b), $T_C$
est plus fine que $T'$. Soit $(f_i:H_i\rightarrow K)$, $i\in I$,
une\pageoriginale famille couvrante de $T_C$. Montrons que
$(f_i,i\in I)$ est une famille couvrante de $T'$. Soient
$s_i:H_i\rightarrow H=\amalg_i H_i$ les monomorphismes canoniques et
$f=(f_i,i\in I):H=\amalg_i H_i\rightarrow K$ le morphisme défini par
les $f_i$. La famille des $s_i$ est couvrante pour $T'$. Pour
montrer que $(f_i,i\in I)$ est une famille couvrante de $T'$, il
suffit donc de montrer, en vertu de l'axiome (T~2) des topologies,
que la morphisme $f:H\rightarrow K$ est couvrant pour $T'$. Il
existe une famille épimorphique de $C^{\widehat{\phantom{,}}}$,
$u_{\lambda}:X_{\lambda}\rightarrow K$, $\lambda\in \Lambda$,
$X_{\lambda}\in \ob C$ (\hyperrefext[I][I.prop3.4]{3.4}). Pour montrer que
$f:H\rightarrow K$ est couvrant pour $T'$, il suffit donc, en vertu
de l'axiome (T 2) des topologies, de montrer que pour tout
$\lambda\in \Lambda$, le morphisme $\pr_2:H\times_K
X_{\lambda}\rightarrow X_{\lambda}$ est couvrant pour $T'$. Soit
alors $v_j:Y_j\rightarrow H\times_K X_{\lambda}$, $j\in J$, $Y_j\in
\ob C$, une famille épimorphique de $C^{\widehat{\phantom{,}}}$. La
famille $(\pr_2\circ v_j, j\in J)$ est couvrante pour $T_{C}$. C'est
donc une famille couvrante de $C$ (\hyperref[II.prop5.1]{5.1}). C'est donc une
famille couvrante de $T'$. Par suite le crible engendré par
$\pr_2:H\times_K X\rightarrow X$ contient un crible couvrant pour
$T'$. Il est donc couvrant pour $T'$ et par suite $\pr_2:H\times_K
X\rightarrow X$ est couvrant, \cqfd
\end{enonce*}

\begin{enonce*}[remark]{Remarque 5.4.1}\etiquette[5.4.1]{II.rem5.4.1}
La démonstration de \hyperref[II.prop5.4]{5.4} montre en fait que toute
topologie $T'$ sur $C^{\widehat{\phantom{,}}}$, plus fine que la
topologie canonique de $C^{\widehat{\phantom{,}}}$, est la moins
fine des topologies $T$ sur $C^{\widehat{\phantom{,}}}$ qui
possèdent les propriétés suivantes:
\begin{enumerate}
\renewcommand{\theenumi}{\alph{enumi}}
\renewcommand{\labelenumi}{\theenumi)}
\item $T$ est plus fine que la topologie canonique de
  $C^{\widehat{\phantom{,}}}$.

\item Toute famille couvrante pour $T'$ de la forme
  $u_i:X_i\rightarrow X$, $i\in I$, où $X$ et les $X_i$ sont des
  objets de $C$, est couvrante pour $T$.
\end{enumerate}
\end{enonce*}

En particulier, toute topologie $T'$ sur
$C^{\widehat{\phantom{,}}}$, plus fine que la topologie canonique,
est uniquement déterminée par les familles $u_i:X_i\rightarrow
X$, $i\in I$, $X_i$ et $X$ objets de $C$, qui sont couvrantes pour
$T'$.

\begin{enonce*}[remark]{Remarque 5.4.2}\etiquette[5.4.2]{II.rem5.4.2}
On\pageoriginale peut facilement montrer que pour toute topologie
$T'$ sur $C^{\widehat{\phantom{,}}}$, plus fine que la topologie
canonique, l'ensemble des familles de morphismes $(X_i\rightarrow
X)$, $i\in I$, de même but de $C$, qui sont couvrantes pour $T'$,
est l'ensemble des familles couvrantes d'une topologie sur $C$. Donc
\hyperref[II.prop5.4]{5.4} et \hyperref[II.rem5.4.1]{5.4.1} permettent d'établir une
correspondance biunivoque entre les topologies sur $C$ et les
topologies sur $C^{\widehat{\phantom{,}}}$ plus fines que la
topologie canonique.
\end{enonce*}

\setcounter{subsection}{5} \setcounter{subsubsection}{-1}
\subsubsection{}\etiquette[5.5.0]{II.subsubsec5.5.0}
Soit $C$ une petite catégorie. Désignons par $\Caf$ l'ensemble des
sous-catégories strictement pleines (tout objet isomorphe à un objet
de la sous-catégorie est un objet de la sous-catégorie) de
$C^{\widehat{\phantom{,}}}$ dont le foncteur d'injection admette un
adjoint à gauche qui commute aux limites projectives finies.
Désignons aussi par $\Tt$ l'ensemble des topologies sur $C$. Le
théorème \hyperref[II.theo3.4]{3.4} nous définit une application:
$$
\Phi:\Tt\longrightarrow \Caf{\quoi}.
$$

Nous allons définir une application en sens inverse. Pour cela, il
faut associer à tout élément $\xymatrix{ e=i': C' \ar@<-4pt>[r]&
\ar@<-2pt>[l]_(.35){a'} C^{\widehat{\phantom{,}}} }$, $\underline{a}'$
adjoint à gauche de $i'$) une topologie $T_e$ sur $C$. Pour tout
objet $X$ de $C$, nous définirons $J_e(X)$ comme étant
l'ensemble des sous-objets de $X$ dont le morphisme d'injection est
transformé par $\underline{a}'$ en un isomorphisme. On vérifie
immédiatement, à l'aide des hypothèses faites sur $\underline{a}'$,
qu'on définit ainsi une topologie $T_e$ sur $C$. On a donc
défini une application:
$$
\Psi:\Caf\longrightarrow \Tt{\quoi}.
$$

On a alors le résultat (d\^u à J. GIRAUD):

\begin{enonce*}{Théorème 5.5}\etiquette[5.5]{II.theo5.5}
L'application\pageoriginale $\Phi$ est une bijection, et $\Psi$ est
l'application inverse.
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
L'application $\Psi\circ \Phi$ est l'identité. En effet, ceci
résulte immédiatement de \hyperref[II.prop5.1]{5.1}. L'application
$\Phi\circ \Psi$ est l'identité. En effet, soient ($\xymatrix{ i':
C' \ar@<-4pt>[r]& \ar@<-2pt>[l]_(.35){{\underline{a}'}}
C^{\widehat{\phantom{,}}} }$) un élément de $\Caf$, $T_e$ la
topologie qui lui correspond par $\Psi$, $C_e$ la catégorie des
faisceaux pour $T_e$. On démontre alors, en utilisant la
définition de la topologie $T_e$ et la définition des morphismes
bicouvrants (\hyperref[II.def5.2]{5.2}), l'équivalence des assertions
suivantes:
\begin{enumerate}
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\theenumi)}
\item Le morphisme $u$ de $C^{\widehat{\phantom{,}}}$ est bicouvrant
  pour la topologie $T_e$.

\item Le morphisme $u$ de $C^{\widehat{\phantom{,}}}$ est transformé
  par $\underline{a}'$ en un isomorphisme,
\end{enumerate}

Il est clair que $C'$ est une sous-catégorie pleine de $C_e$. Il
suffit donc de montrer que tout faisceau $F$ pour la topologie $T_e$
est un objet de $C'$. Mais, d'après l'équivalence ci-dessus, le
morphisme $F\rightarrow i'\circ a'(F)$ est bicouvrant, et sa source
et son but étant des faisceaux, on en déduit par
\hyperref[II.prop5.3]{5.3} (ii) que $F\rightarrow i'\circ a'(F)$ est un
isomorphisme, \cqfd
\end{enonce*}

\section{Faisceaux à valeurs dans une catégorie}\etiquette[6]{II.sec6}

\setcounter{subsection}{-1}
\subsection{}\etiquette[6.0]{II.subsec6.0}
Soient $C$ et $D$ deux catégories. Un foncteur contravariant de
$C$ dans $D$, $F:C^{\circ}\rightarrow D$, est appelé un
\emph{préfaisceau sur $C$ à valeurs dans $D$.}

\begin{enonce*}{Définition 6.1}\etiquette[6.1]{II.def6.1}
Soient $C$ un site, $D$ une catégorie. Un préfaisceau
$F:C^{\circ}\rightarrow D$ est appelé un \emph{faisceau sur $C$ à
valeurs dans} $D$ (ou, plus brièvement, un faisceau à valeurs dans
$D$) si pour tout objet $S$ de $D$, le préfaisceau d'ensembles:
$$
X\longmapsto \Hom_D(S,F(X))\quav X\in \ob(C)
$$
est un faisceau. La sous-catégorie pleine de $\Hhom(C^{\circ},D)$
formée des faisceaux sur $C$ à valeurs dans $D$ sera notée
$\Hhom^{\sim}(C^{\circ},D)$.
\end{enonce*}

\begin{enonce*}[remark]{Remarques 6.2}\etiquette[6.2]{II.rem6.2}
\begin{enumerate}
\renewcommand{\labelenumi}{\theenumi)}
\item La\pageoriginale condition \hyperref[II.def6.1]{6.1} signifie que pour
  tout objet $X$
  de $C$, tout crible couvrant $R$ de $X$, et tout objet $S$ de $D$,
  on a un isomorphisme (\hyperrefext[I][I.cor3.5]{3.5} et \hyperref[II.def2.1]{2.1}):
$$
\Hom_D(S,F(X))\xrightarrow{\sim}\varprojlim_{C/R}F(.){\quoi}.
$$
On retrouve ainsi, lorsque $D$ est la catégorie des
$𝒰$-ensembles, la définition des faisceaux d'ensembles
(\loccit).

\item Soient $D'$ une $𝒰$-catégorie et $G:D\rightarrow D'$
  un foncteur commutant aux limites projectives. Le foncteur $G$
  transforme, par composition, les faisceaux à valeurs dans $D$ en
  faisceaux à valeurs dans $D'$.
\end{enumerate}
\end{enonce*}

\setcounter{subsection}{3} \setcounter{subsubsection}{-1}
\subsubsection{}\etiquette[6.3.0]{II.subsubsec6.3.0}
Soient $\gamma$ une espèce de structure algébrique définie par
limites projectives finies (\hyperrefext[I][I.cor2.9]{2.9}) et $𝒰-\gamma$ la
catégorie des $\gamma$-ensembles qui appartiennent à $𝒰$.
Désignons par $\esj:𝒰-\gamma\rightarrow 𝒰-\Ens$ le
foncteur « ensemble sous-jacent ». Le foncteur $\esj$ commute
aux limites projectives et, par suite, d'après la remarque
précédente, définit par composition des foncteurs:
\begin{align*}
&\esj^{\widehat{\phantom{,}}} :
\Hhom(C^{\circ},𝒰-\gamma)\longrightarrow
C^{\widehat{\phantom{,}}}\\
&\esj^{\widetilde{\phantom{,}}} :
\Hhom^{\sim}(C^{\circ},𝒰-\gamma) \longrightarrow
C^{\widetilde{\phantom{,}}}{\quoi}.
\end{align*}

Le foncteur $\esj^{\widetilde{\phantom{,}}}$ est appelé le
foncteur « faisceau d'ensembles sous-jacent ». En fait, il se
factorise de façon canonique par la catégorie
$\gamma-\widetilde{C}$ des $\gamma$-objets de
$C^{\widetilde{\phantom{,}}}$, et désignant encore par
$\widetilde{\esj}$ le foncteur
$$
\Hhom^{\sim}(C^{\circ}, 𝒰-\gamma)\longrightarrow
\gamma-\widetilde{C}
$$
obtenu, on peut énoncer:

\begin{enonce*}{Proposition 6.3.1}\etiquette[6.3.1]{II.prop6.3.1}
Le foncteur $\esj^{\widetilde{\phantom{,}}}$ établit une
équivalence de catégories entre le catégorie des faisceaux sur
$C$ à valeurs dans $𝒰-\gamma$, celle des $\gamma$-objets de
la catégorie des faisceaux d'ensembles, et la catégorie des
$\gamma$-objets de la catégorie des préfaisceaux d'ensembles
dont le préfaisceau sous-jacent est un faisceau.
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
La\pageoriginale preuve utilise essentiellement (\hyperrefext[I][I.cor3.2]{3.2}) et
le fait qu'une limite projective finie de faisceaux dans la
catégorie des préfaisceaux est un faisceau (\hyperref[II.theo4.1]{4.1}).
\end{enonce*}

\setcounter{subsection}{3} \setcounter{subsubsection}{1}
\subsubsection{}\etiquette[6.3.2]{II.subsubsec6.3.2}
La proposition précédente justifie l'abus de langage qui
consiste à identifier un faisceau à valeurs dans $𝒰-\gamma$
et la $\gamma$-objet correspondant dans
$C^{\widetilde{\phantom{,}}}$. Nous ferons désormais
systématiquement cet abus de langage.

Nous emploierons la terminologie classique: \emph{faisceaux de
  groupes, faisceaux de groupes commutatifs} (que nous appellerons le
  plus souvent \emph{faisceaux abéliens}), \emph{faisceaux
  d'anneaux, faisceaux de $A$-modules} \ldotsn

\subsubsection{}\etiquette[6.3.3]{II.subsubsec6.3.3}
Nous désignons par
$$
C^{\widetilde{\phantom{,}}}_{\gamma}(\resp
C^{\widehat{\phantom{,}}}_{\gamma})
$$
la catégorie des $\gamma$-objets de $C^{\widetilde{\phantom{,}}}$
(\resp de $C^{\widehat{\phantom{,}}}$). Lorsque $\gamma$ est
l'espèce de structure « $A$-module », on écrit aussi
$C^{\widetilde{\phantom{,}}}_{A}$, ou simplement
$\widetilde{C}_{\ab}$ lorsque $A=\ZZ$.

Supposons que $C$ soit un $𝒰$-site. Le foncteur « faisceau
associé » est exact à gauche (\hyperref[II.theo4.1]{4.1}) et par suite:

\begin{enonce*}{Proposition 6.4}\etiquette[6.4]{II.prop6.4}
Le foncteur d'inclusion
$i_{\gamma}:C^{\widetilde{\phantom{,}}}_{\gamma}\rightarrow
C^{\widehat{\phantom{,}}}_{\gamma}$ admet un adjoint à gauche
$a_{\gamma}$ exact à gauche, \ie on a un isomorphisme:
$$
\Hom_{C^{\widetilde{\phantom{,}}}_{\gamma}}(a_{\gamma},\;)\xrightarrow{\sim}\Hom_{C^{\widehat{\phantom{,}}}_{\gamma}}(\;,i_{\gamma}){\quoi}.
$$

Soit $X$ un objet de $C^{\widehat{\phantom{,}}}_{\gamma}$. Le
faisceau d'ensembles sous-jacent à $a_{\gamma}(X)$ est canoniquement
isomorphe au faisceau d'ensembles associé au préfaisceau
d'ensembles sous-jacent à $X$. Le morphisme de préfaisceaux
d'ensembles sous-jacents au morphisme d'adjonction $\id\rightarrow
i_{\gamma} a_{\gamma}$ s'identifie au morphisme d'adjonction
appliqué au préfaisceau d'ensembles sous-jacent.
\end{enonce*}

\begin{enonce*}{Proposition 6.5}\etiquette[6.5]{II.prop6.5}
Supposons\pageoriginale que le foncteur
$\esj:\gamma-𝒰\rightarrow 𝒰-\Ens$ admette un adjoint
à gauche \footnote[1]{On démontre qu'un tel adjoint existe
  toujours [C.F.]: groupe libre, groupe abélien libre, $A$-module
  libre \ldotsn}:
$$
\Hom_{𝒰-\Ens}(.,\esj(.))\xrightarrow{\sim}\Hom_{\gamma-𝒰}(\Lib(.),.){\quoi}.
$$

Le foncteur
$\esj^{\widetilde{\phantom{,}}}:C^{\widetilde{\phantom{,}}}_{\gamma}\rightarrow
C^{\widetilde{\phantom{,}}}$ (\resp
$\esj^{\widehat{\phantom{,}}}:C^{\widehat{\phantom{,}}}_{\gamma}\rightarrow
C^{\widehat{\phantom{,}}}$) admet un adjoint à gauche
$\Lib^{\widetilde{\phantom{,}}}$ (\resp
$\Lib^{\widehat{\phantom{,}}}$) et on a un isomorphisme canonique
$$
\Lib^{\widetilde{\phantom{,}}}=a_{\gamma}\Lib^{\widehat{\phantom{,}}}
i{\quoi}.
$$
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
La preuve est formelle et est laissée au lecteur.
\end{enonce*}

\begin{enonce*}{Corollaire 6.6}\etiquette[6.6]{II.cor6.6}
Soit $(X_i)$, $i\in I$, une famille génératrice de
$C^{\widetilde{\phantom{,}}}$ (\hyperref[II.subsec4.9]{4.9}). Alors la famille
$\Lib^{\widetilde{\phantom{,}}}(X_i)$ est une famille
génératrice de $C^{\widetilde{\phantom{,}}}_{\gamma}$.
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
Le preuve est formelle une fois qu'on a remarqué que le foncteur
$\esj^{\widetilde{\phantom{,}}}$ commute aux limites projectives et
qu'il est conservatif ($\esj^{\widetilde{\phantom{,}}}(u)$ est un
isomorphisme $\Leftrightarrow$ $u$ est isomorphisme).
\end{enonce*}

\begin{enonce*}{Proposition 6.7}\etiquette[6.7]{II.prop6.7}
Soient $A$ un $𝒰$-faisceau d'anneaux, ou bien un petit
anneau, $C^{\widetilde{\phantom{,}}}_A$ (\resp
$C^{\widehat{\phantom{,}}}_A$) la catégorie des faisceaux (\resp
des préfaisceaux) de $A$-modules unitaires
$(\hyperref[II.subsubsec6.3.3]{6.3.3})$ sur un $𝒰$-site $C$. Alors
$C^{\widetilde{\phantom{,}}}_A$ est une $𝒰$-catégorie
abélienne vérifiant les axiomes {\rm (AB 5)} (« existence de
limites inductives filtrantes exactes à gauche ») et {\rm (AB
  3)$^{\ast}$} (« existence de produits infinis ») de
{\rm[\textsc{Tohoku}]}. Elle possède une famille de générateurs
indexée par un élément de $𝒰$.
\end{enonce*}

\begin{enonce*}[remark]{Preuve}
Il est clair que la catégorie $C^{\widehat{\phantom{,}}}_A$ est une
catégorie abélienne vérifiant les axiomes (AB 3), (AB 4), et (AB 5)
(\hyperrefext[I][I.prop2.8]{2.8} et \hyperrefext[I][I.cor3.3]{3.3}). On déduit\pageoriginale alors
de \hyperref[II.prop6.4]{6.4} que la catégorie $C^{\widetilde{\phantom{,}}}_A$
est une catégorie additive où les noyaux et conoyaux sont
représentables. Plus précisément, soient $i_A$ et $a_A$ les
foncteurs inclusion dans les préfaisceaux et faisceaux associés et
$u:X\rightarrow Y$ un morphisme de $C^{\widetilde{\phantom{,}}}_A$.
Le foncteur $a_A$ est exact à gauche et commute aux limites
inductives. Le foncteur $i_A$ commute aux limites projectives. Par
suite le foncteur $i_A a_A:C^{\widehat{\phantom{,}}}\rightarrow
C^{\widehat{\phantom{,}}}_A$ est un foncteur additif exact à gauche.
On en déduit des isomorphismes canoniques:
\begin{gather*}
i_a(\Ker(u))\xrightarrow{\sim} \Ker(i_A(u)){\quoi},\tag{$\ast$}\\
\coker(u)\xrightarrow{\sim} a_A \coker
(i_a(u)){\quoi}.\tag{${\ast}{\ast}$}
\end{gather*}
On a par suite un isomorphisme canonique:
\begin{equation}\coim(u)\xrightarrow{\sim} a_A \coim (i_a (u)){\quoi}.
\tag{${\ast}{\ast}{\ast}$}
\end{equation}
Montrons que le morphisme canonique $\coim(u)\rightarrow \im(u)$ est
un isomorphisme. Pour cela, il suffit de montrer, d'après (*), que
la suite:
$$
0\longrightarrow i_A(\coim(u))\longrightarrow i_A(Y)\longrightarrow
i_A(\coker(u))
$$
est exacte. Utilisant les isomorphismes $({\ast}{\ast})$ et
$({\ast}{\ast}{\ast})$ et remarquant que $a_A i_A(Y)$ est isomorphe
à $Y$, on voit que cette suite est isomorphe à la suite:
$$
0\longrightarrow i_A a_A(\coim(i_A(u))\longrightarrow i_A a_A i_A
(Y)\longrightarrow i_A a_A (\coker(i_A (u)){\quoi},
$$
qui n'est autre que la transformée par le foncteur $i_Aa_A$ de la
suite:
$$
0\longrightarrow \coim(i_A (u))\longrightarrow i_A(Y)\longrightarrow
\coker (i_A(u)){\quoi}.
$$
Or, cette suite est exacte et le foncteur $i_Aa_A$ est exact à
gauche. La catégorie $C^{\widetilde{\phantom{,}}}_A$ est donc une
catégorie abélienne. Le foncteur
$a_A:C^{\widehat{\phantom{,}}}_A\rightarrow
C^{\widetilde{\phantom{,}}}_A$\pageoriginale est exact à gauche et
commute aux limites inductives quelconques. Par suite il est
additif, exact et commute aux limites inductives. La catégorie
$C^{\widehat{\phantom{,}}}_A$ vérifiant l'axiome (AB 5), il en est
de même de la catégorie $C^{\widetilde{\phantom{,}}}_A$. Enfin
il est clair que dans la catégorie
$C^{\widetilde{\phantom{,}}}_A$, les produits indexés par un
élément de $𝒰$ sont représentables et, par suite, que
l'axiome (AB 3)$^{\ast}$ est vérifié, ce qui achève la
démonstration de la première assertion.
\end{enonce*}

Pour tout anneau $A$ élément de $𝒰$, le foncteur:
$$
\esj:𝒰-A-\mbox{module}\longrightarrow 𝒰-\Ens
$$
admet un adjoint à gauche $X_A$ ($A$-module libre engendré). Il
suffit donc d'appliquer \hyperref[II.prop4.10]{4.10} et \hyperref[II.cor6.6]{6.6} pour
achever la preuve.

\begin{enonce*}[remark]{Notation 6.8}\etiquette[6.8]{II.not6.8}
Soit $H$ un faisceau d'ensembles. Le faisceau de $A$-modules
associé au préfaisceau (\cf notation \hyperref[II.prop6.5]{6.5})
$$
X\longmapsto \Lib_A H(X)\qquad X\in \ob C
$$
est noté $A_H$. Lorsque $H=\underline{a}(X)$ (faisceau associé au
préfaisceau représenté par $X$) on écrit parfois simplement
$A_X$ (par abus de notation). La démonstration de \hyperref[II.prop6.7]{6.7}
montre que la famille de faisceaux de $A$-module $𝒜_X$,
$(X\in \ob(C))$, est une famille génératrice, indexée par un
élément de $𝒰$, de la catégorie des faisceaux de
$A$-modules.
\end{enonce*}

\begin{enonce*}[remark]{Remarque 6.9}\etiquette[6.9]{II.rem6.9}
D'après [\textsc{Tôhoku}], \hyperref[II.prop6.7]{6.7} montre que la catégorie
$C^{\widetilde{\phantom{,}}}_A$, lorsque $A$ est un élément de
$𝒰$, possède suffisamment d'injectifs. On sait, par
ailleurs, que  les produits infinis ne sont pas nécessairement
exacts dans $C^{\widetilde{\phantom{,}}}_A$, et, par suite, que la
catégorie $C^{\widetilde{\phantom{,}}}_A$ n'admet pas, en général,
suffisamment de projectifs [\textsc{Roos}].
\end{enonce*}
