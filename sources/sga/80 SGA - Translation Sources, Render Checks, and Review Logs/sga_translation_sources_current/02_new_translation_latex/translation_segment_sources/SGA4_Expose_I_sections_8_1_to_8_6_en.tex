\section{Ind-objects and pro-objects}

\subsection{Cofinal functors and cofinal subcategories}

\begin{statement}{Definition 8.1.1}
Let
\[
\varphi:I\longrightarrow I'
\tag{8.1.1.1}
\]
be a functor. We say that $\varphi$ is a \emph{cofinal functor} if, for every category $C$ and every functor $u:I'\to C$, considering $\varinjlim u$ and $\varinjlim u\circ\varphi$ as objects of $\Hhom(G(\V\text{-}\Set))^{\op}$ (2.1, 2.3.1), where $\V$ is a universe such that $I,I'\in\V$ and $C$ is a $\V$-category, the canonical morphism
\[
\varinjlim u\circ\varphi\longrightarrow \varinjlim u
\tag{8.1.1.2}
\]
is an isomorphism. When $\varphi$ is the inclusion functor of a subcategory of $I'$, we say that $I$ is a \emph{cofinal subcategory} if $\varphi$ is cofinal.
\end{statement}

\subsubsection*{8.1.2}
Notice that, for fixed $\varphi$ and $u$, the bijectivity of (8.1.1.2) does not depend on the choice of the universe $\V$.

Returning to the definition of the terms appearing in (8.1.1.2), one sees that to say that $\varphi$ is cofinal is also to say that, for every universe $\V$ such that $I$ and $I'$ are $\V$-small, and for every functor
\[
v:{I'}^{\op}\longrightarrow (\V\text{-}\Set),
\]
the canonical homomorphism
\[
\varprojlim v\longrightarrow \varprojlim v\circ\varphi
\tag{8.1.2.1}
\]
is an isomorphism, that is, a bijection. To see that this condition is indeed necessary, observe, putting $v=v^{\op}$, that it also says that the condition of Definition~8.1.1 is fulfilled when one takes $C=(\V\text{-}\Set)^{\op}$; this implies that the inductive limits considered in (8.1.1.2) are representable in $C$.

\subsubsection*{8.1.2.2}
It is immediate from the definition that \emph{the composite of two cofinal functors is cofinal}.

\begin{statement}{Proposition 8.1.3}
Let $\varphi:I\to I'$ be a functor.
\begin{enumerate}[label=\alph*)]
\item For $\varphi$ to be cofinal, it is necessary that $\varphi$ satisfy the condition:
\begin{description}[leftmargin=2.4em]
\item[\rm F 1)] For every object $i'$ of $I'$, there exists an object $i$ of $I$ such that $\varphi(i)$ dominates $i'$, i.e. such that $\Hom(i',\varphi(i))\neq\emptyset$.
\end{description}

\item Suppose that $I$ is filtered. For $\varphi$ to be cofinal, it is necessary and sufficient that $\varphi$ satisfy condition \textup{(F 1)} and the following condition:
\begin{description}[leftmargin=2.4em]
\item[\rm F 2)] For every object $i$ of $I$ and every double arrow
\[
\begin{tikzcd}[column sep=2.2em]
i' \arrow[r,shift left=.55ex,"{f'}"] \arrow[r,shift right=.55ex,swap,"{g'}"] & \varphi(i)
\end{tikzcd}
\]
in $I'$ with target $\varphi(i)$, there exists an arrow $h:i\to j$ in $I$ such that $\varphi(h)f'=\varphi(h)g'$.
\end{description}
Moreover, if $\varphi$ is cofinal, then $I'$ is filtered.

\item Suppose that $I'$ is filtered and that $\varphi$ is fully faithful. For $\varphi$ to be cofinal, it is necessary and sufficient that it satisfy condition \textup{F 1)} of \textup{a)}. This implies that $I$ is filtered.
\end{enumerate}
\end{statement}

\begin{proofsketch}
For necessity in a) and b), one uses Definition~8.1.1 only in the case where $u$ is the canonical inclusion functor
\[
u:I'\hookrightarrow \widehat{I'}
\]
(choosing a universe $\U$ such that $I$ and $I'$ are $\U$-small). Then (8.1.1.2) may be interpreted as an arrow of $\widehat{I'}$, whose target is the \emph{final} functor on $I'$. One sees immediately that condition F~1) expresses that this arrow is an epimorphism of $\widehat{I'}$, i.e. that it is surjective on each argument, since inductive limits in $\widehat{I'}$ are calculated argument by argument. This proves a).

Now suppose that $I$ is filtered and that $\varphi$ is cofinal. Then $I'$ is filtered: indeed, the condition that two objects of $I'$ be dominated by a third follows at once from the corresponding condition on $I$ and from F~1). It remains only to prove condition PS~2) of 2.7, i.e. that for every double arrow
\[
f',g':i'\rightrightarrows j'
\]
in $I'$, there exists an arrow $h':j'\to k'$ in $I'$ such that $h'f'=h'g'$. By F~1) we may suppose that $k'$ is of the form $\varphi(i)$. But by the hypothesis that $\varphi$ is cofinal, for every object $i'$ of $I'$ the set $\varinjlim_i\Hom(i',\varphi(i))$ has one element. It follows from the standard description of filtered inductive limits in $\Set$ (2.8.1) that there is an arrow $h:i\to j$ in $I$ such that $\varphi(h)f'=\varphi(h)g'$. This finishes the proof that $I'$ is filtered, and proves F~2) at the same time.

To prove that the conditions stated in b) are sufficient for $\varphi$ to be cofinal, use the form 8.1.2 of the definition. One immediately checks that F~1) implies that (8.1.2.1) is a monomorphism, i.e. injective on each argument, while F~2), together with F~1), ensures that it is bijective, since projective limits in $\widehat{{I'}^{\op}}$ are calculated argument by argument. This proves b).

Finally, if $I'$ is filtered and $\varphi$ is fully faithful, it is immediate that F~1) implies that $I$ is filtered and implies F~2). Thus c) follows from a) and b).
\end{proofsketch}

\subsubsection*{8.1.4}
When $I'$ is a filtered category and $I$ is a full subcategory of $I'$, Proposition~8.1.3 c) shows that the condition that $I$ be cofinal in $I'$ depends only on the subset $\Ob I$ of the preordered set $\Ob I'$ (for the preorder relation ``$x\leq y$'' iff ``$\Hom(x,y)\neq\emptyset$''). If $J'$ is a preordered set and $J$ is a subset of $J'$, we shall sometimes say that $J$ is a \emph{cofinal subset of} $J'$ when every element of $J'$ is dominated by an element of $J$. When $J'$ is filtered, this therefore means that the inclusion functor $\mathcal J\hookrightarrow\mathcal J'$ of the associated categories is cofinal.

\subsubsection*{8.1.5}
In what follows, we shall use the notion of cofinal functor only in cases where the categories $I$ and $I'$ are filtered. Classically, one even restricted oneself to categories associated with preordered sets, i.e. categories in which there is at most one arrow with given source and target. It turns out, however, that this restriction is inconvenient in applications, because the ``natural'' filtered categories which arise in many applications are \emph{not} ordered categories. The following result, due to P.~Deligne, nevertheless shows that there is no essential difference between the two points of view.

\begin{statement}{Proposition 8.1.6}
Let $I$ be a small filtered category. Then there exists a small ordered set $E$, and a cofinal functor
\[
\varphi:\mathcal E\longrightarrow I,
\]
where $\mathcal E$ denotes the category associated with $E$.
\end{statement}

First suppose that the preordered set $\Ob I$ has no greatest element. By a subdiagram of $I$ we mean a pair $D=(\mathcal O,F)$ consisting of a subset $F$ of $\Fl(I)$ and a subset $\mathcal O$ of $\Ob I$, such that for every $f\in F$, the source and target of $f$ belong to $\mathcal O$. An element $e$ of $\mathcal O$ is called a \emph{final object} of the subdiagram $D$ if, for every $x\in D$, the set $\Hom(x,e)\cap F$ of arrows of $D$ with source $x$ and target $e$ has exactly one element $f_x$, if for every arrow $g:x\to y$ of $D$ one has $f_x=f_y\circ g$, and if $f_e=\id_e$.

Let $E$ be the set of \emph{finite} subdiagrams $D$ of $I$ having a \emph{unique} final element $\varphi(D)$, and order $E$ by inclusion. If $D,D'$ are elements of $E$ and $D\subset D'$, then there is a unique arrow $\varphi(D',D)$ of $D'$ with source $\varphi(D)$ and target $\varphi(D')$; if $D\subset D'\subset D''$ are inclusions in $E$, then of course
\[
\varphi(D'',D)=\varphi(D'',D')\varphi(D',D),
\]
and also $\varphi(D,D)=\id_{\varphi(D)}$. Thus one obtains a functor $\varphi:\mathcal E\to I$, and it remains to prove that $E$ is filtered and that $\varphi$ is cofinal. By Proposition~8.1.3 b), three verifications are required.

\begin{enumerate}[label=\arabic*)]
\item Condition F~1) of 8.1.3. For every $i\in I$, take $D$ to be the subdiagram of $I$ reduced to the object $i$ and its identity arrow. Then $i\leq \varphi(D)=i$. This condition F~1) also shows that $E\neq\emptyset$.

\item Condition F~2). For every $i\in\Ob I$, every $D\in E$, and every double arrow $i\rightrightarrows \varphi(D)$, one must find a diagram $D'\in E$, containing $D$, such that
\[
\varphi(D',D):\varphi(D)\longrightarrow\varphi(D')
\]
\emph{equalizes} the double arrow. Since $I$ is filtered, there exists an object $j$ of $I$ and an arrow $f:\varphi(D)\to j$ equalizing the double arrow. Since $I$ has no greatest element, we may suppose that $j$ is a \emph{strict} upper bound of $\varphi(D)$, which implies that it is distinct from all the other objects of $D$. Let $D'$ be the subdiagram of $I$ whose set of objects is $\mathcal O\cup\{j\}$, and whose set of arrows consists of the arrows of $D$, together with the composites
\[
x\xrightarrow{f_x}\varphi(D)\xrightarrow{f}j,
\]
where $x$ is an object of $D$ and $f_x$ is the unique arrow of $D$ with source $x$ and target $\varphi(D)$, and together with $\id_j$. It is then clear that $D'$ is a finite subdiagram of $I$, that it has $j$ as its unique final object, hence that $D'\in E$, and that $D'$ satisfies the required condition.

\item Let $D$ and $D'$ be two subdiagrams in $E$. They are contained in some common $D''\in E$. Indeed, one can find a strict upper bound $j$ of $\varphi(D)$ and $\varphi(D')$,
\[
\begin{tikzcd}[column sep=2.5em,row sep=1.8em]
\varphi(D) \arrow[dr,"f"] & \\
& j \\
\varphi(D') \arrow[ur,swap,"f'"] & {}
\end{tikzcd}
\]
and take $D''$ to be the subdiagram whose set of objects is the union of the sets of objects of $D$ and $D'$ and of $\{j\}$, and whose set of arrows is the union of the sets of arrows of $D$ and $D'$, the composites $f\circ f_x$ with $x$ an object of $D$, the composites $f'\circ f'_{x'}$ with $x'$ an object of $D'$, and $\{\id_j\}$. This is indeed a finite subdiagram of $I$. We show that, after replacing $j,f,f'$ by $j',gf,gf'$ with a suitable $g:j\to j'$, we may arrange for $j$ to be a final object of $D''$. This is the same as saying that, for every $x$ which is an object of both $D$ and $D'$, one has
\[
ff_x=f'f'_{x'}.
\]
But this is only a \emph{finite} set of double arrows to be equalized by some $g:j\to j'$, and this is possible because $I$ is filtered.
\end{enumerate}

This completes the proof in the case considered. The general case is reduced to it by introducing the filtered category $\mathbb N$ associated with the ordered set $\mathbb N$ of natural numbers, and observing that the category $\mathbb N\times I$ is filtered, has no greatest element, and that the projection $\mathbb N\times I\to I$ is a cofinal functor.

\begin{statement}{Corollary 8.1.7}
Let $I$ be a $\U$-category. For there to exist a small filtered ordered set $E$ and a cofinal functor $\mathcal E\to I$, it is necessary and sufficient that $I$ be filtered and that $\Ob I$ admit a small cofinal subset $I'$ (8.1.4).
\end{statement}

This is necessary by 8.1.3 a), and sufficient by 8.1.6 applied to the full subcategory of $I$ defined by $I'$.

\begin{statement}{Definition 8.1.8}
Let $I$ be a filtered category. We say that $I$ is \emph{essentially small} if $I$ is a $\U$-category and if it satisfies the equivalent conditions of 8.1.7.
\end{statement}

\subsection{Ind-objects and ind-representable functors}

\subsubsection*{8.2.1}
In what follows we fix a $\U$-category $C$, which we always regard as embedded in the category $\widehat C$ of $\U$-presheaves by means of the canonical functor (1.3.1)
\[
h:C\hookrightarrow \widehat C.
\tag{8.2.1.1}
\]
We shall call an \emph{ind-object of} $C$ (or an \emph{inductive system} of $C$) any functor
\[
\varphi:I\longrightarrow C,
\tag{8.2.1.2}
\]
where $I$ is a \emph{filtered} category (2.7), called the \emph{index category} of the ind-object. Apart from this, $I$ is arbitrary; in particular we do not require $I$ to be a $\U$-category. It will often be convenient to denote an ind-object $\varphi$ by the indexed notation $(X_i)_{i\in\Ob I}$, or even, by a further abuse of notation, by $(X_i)_{i\in I}$, $(X_i)_i$, or $(X_i)$, where $X_i=\varphi(i)$ for $i\in\Ob I$. This notation is no more and no less abusive than the classical notation for inductive systems indexed by filtered ordered sets, where the transition morphisms are likewise not mentioned. The $X_i$ are called the \emph{component objects} of the ind-object $\mathcal X$. Notice that, in the general case considered here, the ``\emph{transition morphisms}'' $\varphi(f)$ of the inductive system are indexed by the set $\Fl(I)$ of arrows of $I$, which in general is not identified with a subset of $\Ob I\times\Ob I$. In other words, for given $i$ and $j$, with $j$ dominating $i$, there may exist more than one transition morphism from $X_i$ to $X_j$.

\subsubsection*{8.2.2}
The most useful ind-objects $\mathcal X=(X_i)_{i\in I}$ of $C$ are those for which the index category $I$ is essentially small (8.1.8); such an ind-object is called \emph{essentially small}. If $(X_i)_{i\in I}$ is such an object, using the fact that small inductive limits in $\widehat C$ are representable (3.1), one may consider
\[
L(\mathcal X)\mathrel{:=}\varinjlim h\cdot\mathcal X=\varinjlim_i X_i\in\Ob\widehat C,
\tag{8.2.2.1}
\]
where the last limit is taken \emph{in} $\widehat C$. Thus $L(\mathcal X)$ is the presheaf
\[
L(\mathcal X):Y\longmapsto \varinjlim_i\Hom(Y,X_i)
\tag{8.2.2.2}
\]
on $C$. We say that this presheaf is the \emph{presheaf ind-represented by the ind-object $\mathcal X=(X_i)_{i\in I}$ of $C$}. A presheaf $F$ on $C$ is called an \emph{ind-representable presheaf} if it is isomorphic to a presheaf $L(\mathcal X)$ ind-represented by an essentially small ind-object. It follows from 8.1.6 that, in this definition, one may suppose that $I$ is a small category, and even that $I$ is associated with an ordered set belonging to $\U$.

\subsubsection*{8.2.3}
Consider an ind-object $\mathcal X=(X_i)_{i\in I}$, given by a functor $\varphi:I\to C$, and let
\[
u:I'\longrightarrow I
\]
be a functor, where $I'$ is a second filtered category. Then the functor $\varphi\circ u$ is an ind-object $\mathcal X'$ of $C$ with index category $I'$, written in indexed notation as $(X_{u(i')})_{i'\in I'}$. It is called the ind-object of $C$ \emph{deduced from $(X_i)_{i\in I}$ by change of index categories} using the functor $u$. If $I$ and $I'$, equivalently $\mathcal X$ and $\mathcal X'$, are essentially small, one obtains a canonical morphism
\[
L(\mathcal X')=\varinjlim_{i'}X_{u(i')}\longrightarrow L(\mathcal X)=\varinjlim_i X_i
\tag{8.2.3.1}
\]
between the presheaves ind-represented by $\mathcal X'$ and $\mathcal X$. When $u$ is a cofinal functor (8.1.1), $\mathcal X$ is essentially small if and only if $\mathcal X'$ is so (8.1.7), and the preceding homomorphism (8.2.3.1) is an isomorphism
\[
L(\mathcal X')\simeq L(\mathcal X).
\tag{8.2.3.2}
\]

\subsubsection*{8.2.4}
Let
\[
\mathcal X=(X_i)_{i\in I},\qquad \mathcal Y=(Y_j)_{j\in J}
\tag{8.2.4.1}
\]
be two essentially small ind-objects of $C$, indexed by filtered essentially small categories $I$ and $J$. A \emph{morphism} from the ind-object $\mathcal X$ to the ind-object $\mathcal Y$ is by definition a morphism
\[
L(\mathcal X)\longrightarrow L(\mathcal Y)
\tag{8.2.4.2}
\]
between the presheaves they ind-represent. We put
\[
\Hom_{\text{ind-ob}}(\mathcal X,\mathcal Y)
=\Hom_{\widehat C}(L(\mathcal X),L(\mathcal Y)).
\tag{8.2.4.3}
\]
We suppress the subscript ``ind-ob'' on $\Hom$ if no confusion is likely. Composition of morphisms of ind-objects of $C$ is defined as composition of morphisms of the presheaves they ind-represent. If $\V\supset\U$ is a universe containing $\U$, and if one restricts to essentially small ind-objects which belong to $\V$, these form the set of objects of a category whose arrows are triples $(\mathcal X,\mathcal Y,u)$, where $\mathcal X$ and $\mathcal Y$ are essentially small ind-objects belonging to $\V$ and $u:\mathcal X\to\mathcal Y$ is a morphism of ind-objects, i.e. a morphism $L(\mathcal X)\to L(\mathcal Y)$. The category so obtained is denoted
\[
\Ind_{\V}(C,\U).
\tag{8.2.4.4}
\]
When $\V=\U$, it is denoted simply
\[
\Ind_{\U}(C)\quad\text{or}\quad\Ind(C),
\tag{8.2.4.5}
\]
and is called the \emph{category of ind-objects of} $C$ relative to $\U$, the mention of $\U$ being suppressed when no confusion is likely.

Of course, if $\V\subset\V'$ are two universes containing $\U$, then $\Ind_{\V}(C,\U)$ is a full subcategory of $\Ind_{\V'}(C,\U)$. It follows from 8.2.3 that the inclusion functor is an \emph{equivalence of categories}
\[
\Ind_{\V}(C,\U)\xrightarrow{\approx}\Ind_{\V'}(C,\U).
\tag{8.2.4.6}
\]
This shows in particular that \emph{every essentially small ind-object of $C$ defines an element of $\Ind(C)$, uniquely up to isomorphism}. This observation justifies the fairly common practical abuse of language which consists in identifying any essentially small ind-object of $C$ with an object of $\Ind(C)$.

It follows at once from the definitions that for every universe $\V$ we have a canonical functor
\[
L:\Ind_{\V}(C,\U)\longrightarrow \widehat C
\tag{8.2.4.7}
\]
which is fully faithful. These functors are of course known up to unique isomorphism, by (8.2.4.6), once one of them is known, in particular once one knows the canonical functor
\[
L:\Ind(C)\longrightarrow \widehat C.
\tag{8.2.4.8}
\]
Since this functor is fully faithful, it is often used to identify an object of the first category with the functor it pro-represents, or even to identify $\Ind(C)$ with its essential image in $\widehat C$, consisting of the ind-representable presheaves. Notice, however, that this identification has distinctly more drawbacks than the analogous identification of $C$ with a full subcategory of $\widehat C$ via the functor $h$ (8.2.1.1), because unlike the latter functor, the functor (8.2.4.8) is not in general injective on objects.

\subsubsection*{8.2.5}
Let us spell out definition (8.2.4.3) in terms of the indexed expressions (8.2.4.1) of the ind-objects under consideration. Taking account of the definition (8.2.2.1), one obtains
\[
\Hom(\mathcal X,\mathcal Y)
\simeq \varprojlim_i \Hom(X_i,\mathcal Y)
\mathrel{:=}\varprojlim_i \Hom(X_i,L(\mathcal Y))
=\varprojlim_i\varinjlim_j\Hom_C(X_i,Y_j),
\]
the last equality coming from (8.2.2.2) applied to $\mathcal Y$. Thus there is a canonical bijection
\[
\Hom((X_i)_{i\in I},(Y_j)_{j\in J})
\simeq \varprojlim_{i\in I}\varinjlim_{j\in J}\Hom_C(X_i,Y_j).
\tag{8.2.5.1}
\]
We leave to the reader the task of making the composition of morphisms of ind-objects explicit using this formula. Notice that it follows at once from this formula that the set $\Hom(\mathcal X,\mathcal Y)$ of homomorphisms of ind-objects is $\U$-\emph{small}; equivalently, the categories (8.2.4.4) are $\U$-categories. Equivalently again, by (8.2.4.6), the \emph{category $\Ind(C)$ is a $\U$-category}; that is, the right hand side of (8.2.5.1) is small when $I,J\in\U$, which follows immediately from the fact that $C$ is a $\U$-category, hence the $\Hom_C(X_i,Y_j)$ are small.

\subsubsection*{8.2.6}
Let $I$ be an essentially small filtered category. Then one sees either from (8.2.5.1), or from the functoriality of the inductive limit of a functor $I\to\widehat C$ with respect to that functor, that there is a canonical functor
\[
\Hhom(I,C)\longrightarrow\Ind(C).
\tag{8.2.6.1}
\]
Notice that this functor is \emph{not} in general fully faithful, or even faithful.

\begin{statement}{Remark 8.2.7}
The definitions of 8.2.4 make clear the notion of \emph{isomorphism} of two essentially small ind-objects, which also means the isomorphism of the presheaves they ind-represent. Notice that this is a very weak notion of isomorphism when compared with the notion of isomorphism in categories of the form $\Hhom(I,C)$. Thus many fairly natural relations one can impose on an ind-object, such as having its components in a given strictly full subcategory, or being strict (8.12 below), are \emph{not} stable under isomorphism. Therefore, if one wants to work with notions stable under isomorphism of ind-objects, one must ``saturate'' the notions under consideration by passing to the strictly full subcategory of $\Ind(C)$ generated by the subcategory of $\Ind(C)$ consisting of objects satisfying the condition in question. See, for example, the notion of essentially constant ind-object introduced below (8.4).
\end{statement}

\begin{statement}{Exercise 8.2.8}
Let $\U\subset\V$ be two universes and let $C$ be a $\U$-category belonging to $\V$. Let $\SysInd(C)$ denote the following category.
\begin{enumerate}[label=\alph*)]
\item The objects of $\SysInd(C)$ are functors $\varphi:I\to C$, where $I$ is an essentially $\U$-small filtered category belonging to $\V$.
\item If $(I,\varphi)$ and $(J,\psi)$ are two objects of $\SysInd(C)$, a morphism in $\SysInd(C)$ from $(I,\varphi)$ to $(J,\psi)$ is a pair $(m,u)$, where $m:I\to J$ is a functor and $u:\varphi\to\psi\circ m$ is a morphism of functors. The composition of morphisms in $\SysInd(C)$ is defined in the evident way.
\end{enumerate}
\end{statement}

Let $S$ be the set of morphisms $(m,u):(I,\varphi)\to(J,\Psi)$ of $\SysInd(C)$ such that $m$ is cofinal and $u$ is an isomorphism. Let $(I,\varphi)$ be an object of $\SysInd(C)$. The category $\Fl(I)$ of arrows of $I$ maps to $I$ by two natural functors: source and target. Moreover these two functors are related by the canonical morphism of functors $v:\mathrm{source}\to\mathrm{target}$. Hence there are two morphisms in $\SysInd(C)$ from $(\Fl(I),\varphi\circ\mathrm{source})$ to $(I,\varphi)$:
\[
p_1(I,\varphi)=(\mathrm{source},\id),
\]
and
\[
p_2(I,\varphi)=(\mathrm{target},\varphi^*v:\varphi\circ\mathrm{source}\to\varphi\circ\mathrm{target}).
\]
Let $p:\SysInd(C)\to\Ind(C)$ be the evident functor. Let $B$ be a category. Show that the functor $F\mapsto F\circ p$,
\[
\Hhom(\Ind(C),B)\longrightarrow\Hhom(\SysInd(C),B),
\]
is fully faithful, and that a functor $G:\SysInd(C)\to B$ belongs to its essential image if and only if it has the following two properties:
\begin{enumerate}[label=\arabic*)]
\item For every $s\in S$, $F(s)$ is an isomorphism of $B$.
\item For every object $(I,\varphi)$ of $\SysInd(C)$, $F(p_1(I,\varphi))=F(p_2(I,\varphi))$.
\end{enumerate}

\subsection{Characterization of ind-representable functors}

\begin{statement}{Proposition 8.3.1}
Let $F$ be an ind-representable presheaf on $C$. Then $F$ is left exact, i.e. (2.3.2), for every finite category $J$ and every functor $\varphi:J\to C$ such that $\varinjlim\varphi$ is representable (i.e. $\varprojlim\varphi^{\op}$ is representable), the canonical map
\[
F(\varinjlim\varphi)\longrightarrow\varprojlim F\circ\varphi
\]
is bijective.
\end{statement}

Indeed, representable presheaves are plainly left exact; hence by 2.8 so is every filtered inductive limit of such functors. \qedtext

\subsubsection*{8.3.2}
Given a presheaf $F$ on $C$, we shall often have to work with the category
\[
C_{/F}\hookrightarrow\widehat C_{/F},
\tag{8.3.2.1}
\]
which denotes the full subcategory of the second category consisting of arrows $X\to F$ \emph{whose source lies in} $C$. It follows from 1.4 that the objects of this category are identified with pairs $(X,u)$, where $X$ is an object of $C$ and $u\in F(X)$. A morphism from $(X,u)$ to $(X',u')$ is then interpreted as an arrow $f:X\to X'$ in $C$ such that $F(f)(u')=u$. The ``forget-the-target'' functor from $\widehat C_{/F}$ to $\widehat C$ induces a functor, called the \emph{canonical} functor,
\[
C_{/F}\longrightarrow C,
\tag{8.3.2.2}
\]
already considered in 3.4, where it is proved that the inductive limit of this functor exists in $\widehat C$ (with no smallness condition on $C_{/F}$) and is canonically isomorphic to $F$.

\subsubsection*{8.3.2.3}
Notice that if finite inductive limits are representable in $C$, and if $F$ is left exact, i.e. carries them to finite projective limits in $\Set$, then the same is true in $C_{/F}$, and \emph{a fortiori} (2.7.1) $C_{/F}$ is filtered. More generally, if sums of two objects and cokernels of double arrows are representable in $C$, and if $F$ carries them respectively to products and to kernels, then $C_{/F}$ is stable under the same types of finite inductive limits. Hence it is filtered if and only if it is nonempty, i.e. if and only if the functor $F$ is not the constant functor with value $\emptyset$.

\begin{statement}{Theorem 8.3.3}
Let $C$ be a $\U$-category and let $F\in\Ob\widehat C$, say $F:C^{\op}\to(\U\text{-}\Set)$ is a $\U$-presheaf on $C$. The following conditions are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $F$ is ind-representable (8.2.2).
\item The category $C_{/F}$ (8.3.2) is filtered and essentially small.
\item If finite inductive limits are representable in $C$: the functor $F$ is left exact, and $\Ob C_{/F}$ admits a small cofinal subset (8.1.4).
\item[(iii bis)] If sums of two objects and cokernels of double arrows are representable in $C$: the functor $F$ carries sums of two objects of $C$ to products and cokernels of double arrows of $C$ to kernels, the category $C_{/F}$ is nonempty, i.e. the functor $F$ is not the constant functor with value $\emptyset$, and finally there exists a small family of objects of $C_{/F}$ such that every object $X$ of $C_{/F}$ is dominated by some $X_i$, i.e. admits an $F$-morphism $X_i\to X$.
\item If the category $C$ is equivalent to a small category: the category $C_{/F}$ is filtered.
\item If the category $C$ is equivalent to a small category and if filtered inductive limits in $C$ are representable: the functor $F$ is left exact.
\end{enumerate}
\end{statement}

\begin{proofsketch}
(i) $\Rightarrow$ (ii). Suppose that $F$ is ind-represented by $(X_i)_{i\in I}$, with $I$ small. We prove that $C_{/F}$ is filtered. Let $X,X'$ be two objects of $C_{/F}$, i.e. objects of $C$ equipped with morphisms $X\to F$ and $X'\to F$. We prove that they are dominated by a third object of $C_{/F}$. The morphisms $X\to F$ and $X'\to F$ come from morphisms $X\to X_i$ and $X'\to X'_{i'}$, and after replacing $i,i'$ by a common upper bound in $\Ob I$, we may suppose $i=i'$. We then take as common upper bound of $X$ and $X'$ the object $X_i$ equipped with the canonical morphism $X_i\to F$.

Now let
\[
\begin{tikzcd}[column sep=2.2em]
X \arrow[r,shift left=.55ex,"f"] \arrow[r,shift right=.55ex,swap,"g"] & X' \arrow[r,"h"] & F
\end{tikzcd}
\]
be a double arrow in $C_{/F}$. We prove that it is equalized by an arrow $X'\to X''$ of $C_{/F}$. The morphism $h:X'\to F$ is given by a morphism $h_i:X'\to X_i$, and the condition $hf=hg$ says that there exists $\alpha:i\to j$ in $I$ such that
\[
\varphi(\alpha)(h_if)=\varphi(\alpha)(h_ig).
\]
Thus, after replacing $h_i$ by $\varphi(\alpha)h_i$, we equalize $f$ and $g$. This proves that $C_{/F}$ is filtered. It is then immediate that it is essentially small, since the objects $(X_i\to F)_{i\in\Ob I}$ form a small cofinal family in $\Ob C_{/F}$.

(ii) $\Rightarrow$ (i). Put $I=C_{/F}$ and consider the canonical functor (8.3.2.1)
\[
\varphi:I=C_{/F}\longrightarrow C.
\]
The presheaf represented by this ind-object of $C$ is known to be $F$ (3.4), so $F$ is ind-representable by definition (8.2.2).

(ii) $\Leftrightarrow$ (iii). Indeed, (ii) $\Rightarrow$ (iii) because an ind-representable functor is left exact (8.3.1), and (iii) $\Rightarrow$ (ii) because we have observed (8.3.2.3) that left exactness of $F$ implies that $C_{/F}$ is filtered, and one applies Definition~8.1.8 to conclude that this category is essentially small.

The equivalence (ii) $\Leftrightarrow$ (iii bis) is proved in the same way as (ii) $\Leftrightarrow$ (iii).

The equivalences (ii) $\Leftrightarrow$ (iv) and (iii) $\Leftrightarrow$ (v) are trivial, since if $C$ is equivalent to a small category, then $C_{/F}$ is evidently also equivalent to a small category and \emph{a fortiori} is automatically essentially small as soon as it is filtered. This completes the proof of 8.3.3.
\end{proofsketch}

\begin{statement}{Remark 8.3.4}
Let $F:C'\to C''$ be a functor between $\U$-categories. It appears that the notion of left exactness (2.3.2) is of practical interest only when finite projective limits are representable in $C'$. In the case where $C''=(\U\text{-}\Set)$ and $C'$ is $\U$-small, writing $C'$ in the form $C^{\op}$ (so $C={C'}^{\op}$), the ``right notion'' which in the general case replaces left exactness is that of ind-representability of $F$ when it is considered as a presheaf on $C^{\op}$; equivalently, it is the pro-representability of $F$ when it is considered as a functor on $C'$ (cf. 8.10). This notion does coincide with left exactness when finite projective limits are representable in $C'$, i.e. when finite inductive limits are representable in $C$ (8.3.3). When one no longer assumes that $C'$ is $\U$-small, or at least equivalent to a $\U$-small category, the two notions for a functor $F$, left exactness and pro-representability, no longer necessarily coincide, even if finite projective limits are representable in $C'$ (cf. 8.12.9). In fact, it seems that in this case left exact functors which are not ind-representable should be regarded as pathological; the ``good'' objects remain the ind-representable functors. Compare 8.13.3. For the case of functors $F:C'\to C''$, with $C'$ and $C''$ again arbitrary $\U$-categories, we shall develop below (8.11.5), more generally, a notion which ``improves'' that of left exactness, namely the notion of a functor admitting a pro-adjoint functor.
\end{statement}

\subsection{Constant and essentially constant ind-objects}

Choose a final category $e$, i.e. one such that $\Ob e$ and $\Fl e$ are each reduced to a single element; for example, one may take for $e$ the full subcategory of $\Set$ consisting of the empty set, if one wants a canonical choice. This is plainly a filtered and small category. If $X$ is an object of $C$, we associate with it the constant ind-object indexed by $e$, with value $X$, denoted $c(X)$. It is clear that, as $X$ varies, one obtains a fully faithful functor
\[
c:C\longrightarrow\Ind(C),
\tag{8.4.1}
\]
which is moreover injective on objects, and through which we shall identify $C$ with a full subcategory of $\Ind(C)$. Notice that the composite functor
\[
C\xrightarrow{c}\Ind(C)\xrightarrow{L}\widehat C
\tag{8.4.2}
\]
is the canonical functor $h$ (8.2.1.1).

\subsubsection*{8.4.3}
More generally, let $I$ be a filtered category. The constant functor on $I$ with value an object $X$ of $C$ is also called \emph{the constant ind-object of value $X$ indexed by $I$}. If $I$ is essentially small, it is clear that this ind-object ind-represents the functor $h(X)$ represented by $X$, hence this ind-object is isomorphic to $c(X)$.

\subsubsection*{8.4.4}
An ind-object $\mathcal X$ of $C$ is called \emph{essentially constant} if it is isomorphic (in a category $\Ind_{\V}(C,\V')$ for suitable $\V,\V'$) to an ind-object of the form $c(X)$, with $X\in\Ob C$. The object $X$, which is then determined up to canonical isomorphism, is called the \emph{value} of the essentially constant ind-object in question. Thus, by definition, the functor $c$ of (8.4.1) induces an equivalence of $C$ with the full subcategory of $\Ind(C)$ consisting of essentially constant ind-objects; this subcategory is the essential image of the functor $c$.

Obviously a constant ind-object is essentially constant; the converse is true only in the trivial case where $C$ is empty or a one-point category.

\subsection{Filtered inductive limits in $\Ind(C)$}

\begin{statement}{Proposition 8.5.1}
Let $C$ be a $\U$-category. In $\Ind(C)$, small filtered inductive limits are representable, and the canonical functor (8.2.4.8)
\[
L:\Ind(C)\longrightarrow\widehat C
\]
commutes with them.
\end{statement}

Taking account of the fact that $L$ is fully faithful, the assertions of the proposition are equivalent to the following one.

\begin{statement}{Corollary 8.5.2}
Every small filtered inductive limit, in $\widehat C$, of ind-representable presheaves is ind-representable.
\end{statement}

This follows easily from criterion 8.3.3(ii). The details of the verification are left to the reader.

\subsubsection*{8.5.3}
The ``tautological'' case of filtered inductive limits in $\Ind(C)$ is the case where one starts from an essentially small ind-object $\mathcal X=(X_i)_{i\in I}$ of $C$. One then has, in $\widehat C$ and hence also in $\Ind(C)$,
\[
\mathcal X=\varinjlim_I X_i.
\tag{8.5.3.1}
\]
In writing this formula, one should take care that this is not an inductive limit in $C$, and that even when the latter exists, it is not isomorphic in $\Ind(C)$ to $\mathcal X$: indeed, the canonical functor (8.4.1) $c:C\to\Ind(C)$ \emph{does not in general commute with filtered inductive limits}. See Exercise~8.5.5 below.

To avoid this possible confusion in writing (8.5.3.1), ``certain authors'' (= P.~Deligne) prefer to write it
\[
\mathcal X=\textrm{``}\varinjlim_I\textrm{''}\,X_i,
\tag{8.5.3.2}
\]
the role of the quotation marks being to indicate that the inductive limit is taken in a category of ind-objects $\Ind(C)$. By extension, one should then denote by $\textrm{``}\lim\textrm{''}$ every inductive-limit operation in $\Ind(C)$, without requiring the components of the inductive system considered in $\Ind(C)$ to lie in the image, or in the essential image, of $c:C\to\Ind(C)$.

\subsubsection*{8.5.4}
To compute the inductive or projective limit of a functor
\[
\varphi:J\longrightarrow\Ind(C),
\tag{8.5.4.1}
\]
where $J$ is now an arbitrary category, it is often convenient to make $\varphi$ explicit by means of a functor
\[
\Phi:J\times I\longrightarrow C,
\tag{8.5.4.2}
\]
where $I$ is an essentially small, or preferably small, filtered category. Such a $\Phi$ indeed defines a functor
\[
j\longmapsto (i\longmapsto \Phi(j,i)):
J\longrightarrow\Hhom(I,C),
\]
hence, after composing with the canonical arrow (8.2.6.1) $\Hhom(I,C)\to\Ind(C)$, a functor
\[
j\longmapsto (i\longmapsto \Phi(j,i)):
J\longrightarrow\Ind(C).
\tag{8.5.4.3}
\]
We may say that $\Phi$ is an \emph{indexed expression of $\varphi$, indexed by the filtered category $I$}, if the corresponding functor (8.5.4.3) is isomorphic to $\varphi$. We shall study below (8.8) general existence conditions for an indexed expression of a given functor $\varphi$. Here we shall merely start from a functor $\varphi$ given in indexed form (8.5.4.3), in the case where $J$ is a small filtered category, in order to indicate the calculation of
\[
\textrm{``}\varinjlim\textrm{''}\,\varphi=\textrm{``}\varinjlim_j\textrm{''}\,\varphi(j).
\]
Formula (8.5.3.2), together with the associativity formula for inductive limits (2.5.0), then immediately gives the \emph{tautological calculation} of $\textrm{``}\varinjlim\textrm{''}\,\varphi$:
\[
\textrm{``}\varinjlim_J\textrm{''}\,\varphi
=\textrm{``}\varinjlim_{J\times I}\textrm{''}\,\Phi,
\tag{8.5.4.4}
\]
i.e.
\[
\textrm{``}\varinjlim_j\textrm{''}\,(i\longmapsto\Phi(j,i))
=\textrm{``}\varinjlim_{j,i}\textrm{''}\,\Phi(j,i).
\tag{8.5.4.5}
\]
Thus the inductive system sought is none other than $\Phi$ itself, the index category being $J\times I$, which is indeed filtered since $J$ and $I$ are so.

\begin{statement}{Exercise 8.5.5}
Let $C$ be a $\U$-category.
\begin{enumerate}[label=\alph*)]
\item Prove that the following conditions are equivalent:
\begin{enumerate}[label=(\roman*)]
\item Small inductive limits are representable in $C$, and the functor $c:C\to\Ind(C)$ commutes with them.
\item Small inductive limits are representable in $C$, and for every $X\in\Ob C$, the functor $Y\mapsto\Hom(X,Y)$ commutes with these limits.
\item The functor $c$ is an equivalence of categories.
\item Small filtered inductive limits are representable in $C$, and the functor $\lim c:\Ind C\to C$ (cf. 8.7.1.5) is fully faithful, or equivalently an equivalence.
\end{enumerate}

\item If $C$ is a finite category, prove that the preceding conditions are equivalent to the following one: for every projector in $C$ (10.6), its image is representable in $C$.
\end{enumerate}
\end{statement}

\subsection{Extension of a functor to ind-objects}

\subsubsection*{8.6.1}
Let
\[
f:C\longrightarrow C'
\tag{8.6.1.1}
\]
be a functor between $\U$-categories. For every ind-object
\[
\varphi:I\longrightarrow C
\]
of $C$, where $I$ is a filtered category, the composite functor $f\circ\varphi$ is an ind-object of $C'$, also denoted $\Ind(f)(\varphi)$. In indexed notation, if $\varphi$ is written as $\mathcal X=(X_i)_{i\in I}$, one obtains
\[
\Ind(f)(X_i)_{i\in I}=(f(X_i))_{i\in I}.
\tag{8.6.1.2}
\]
Let $\mathcal Y=(Y_j)_{j\in J}$ be a second inductive system of $C$. Then one obtains an evident map, also denoted $u\mapsto\Ind(f)(u)$:
\[
\begin{aligned}
\Hom(\mathcal X,\mathcal Y)&=\varprojlim_i\varinjlim_j\Hom(X_i,Y_j)\\
&\longrightarrow
\Hom(\Ind(f)(\mathcal X),\Ind(f)(\mathcal Y))
\simeq\varprojlim_i\varinjlim_j\Hom(f(X_i),f(Y_j)).
\end{aligned}
\tag{8.6.1.3}
\]
One immediately checks that these maps are compatible with the composition of morphisms of ind-objects, referring to the unwritten formula for composition of morphisms of ind-objects (8.2.5). We have thus obtained, for every universe $\V$, a functor
\[
\Ind(f):\Ind_{\V}(C,\U)\longrightarrow\Ind_{\V}(C',\U)
\tag{8.6.1.4}
\]
(notation of (8.2.4.5)), and in particular a functor
\[
\Ind(f):\Ind(C)\longrightarrow\Ind(C').
\tag{8.6.1.5}
\]
If one has a second functor
\[
g:C'\longrightarrow C'',
\]
then of course there is an identity of functors between categories of ind-objects, or of $\Ind_{\V}$-objects if desired,
\[
\Ind(gf)=\Ind(g)\circ\Ind(f),
\tag{8.6.1.6}
\]
and for $C'=C$ one has the perennial formula
\[
\Ind(\id_C)=\id_{\Ind(C)}.
\tag{8.6.1.7}
\]
Thus, figuratively speaking, one may say that the category $\Ind(C)$ \emph{depends functorially on $C$}, covariantly.\footnote{For a contravariant dependence, under certain conditions, cf. 8.11.} We leave to the reader the task of making explicit even a 2-\emph{functorial} dependence, by defining, for every pair of $\U$-categories $C,C'$, a canonical functor
\[
\mathsf{Ind}:\Hhom(C,C')\longrightarrow\Hhom(\Ind(C),\Ind(C')),
\tag{8.6.1.8}
\]
and by specifying the functorial nature of the identity isomorphisms (8.6.1.6) and (8.6.1.7).

\subsubsection*{8.6.2}
Return to the functor (8.6.1.1), and consider the corresponding functor (5.1)
\[
f_!:\widehat C\longrightarrow\widehat{C'},
\tag{8.6.2.1}
\]
and the diagram of functors
\[
\begin{tikzcd}[column sep=4.2em,row sep=2.0em]
\Ind(C) \arrow[r,"{\Ind(f)}"] \arrow[d,"{L_C}"'] & \Ind(C') \arrow[d,"{L_{C'}}"] \\
\widehat C \arrow[r,"{f_!}"'] & \widehat{C'}.
\end{tikzcd}
\tag{8.6.2.2}
\]
Here the vertical arrows denote the canonical functors $L$ of (8.2.4.8). Since the functor $f_!$ ``extends $f$'' and commutes with inductive limits (5.4.3), and since the functors $L_C$ and $L_{C'}$ commute with small filtered inductive limits (8.5.1), it follows from (8.5.3.2) that for every ind-object $(X_i)_{i\in I}$ of $C$ there is, in $\widehat{C'}$, a canonical isomorphism
\[
f_!(L_C((X_i)_{i\in I}))\simeq \varinjlim_i f_!L_C(X_i)
\simeq \varinjlim_i L_{C'}f(X_i)=L_{C'}(f(X_i)_{i\in I}),
\]
i.e. a canonical isomorphism
\[
f_!L_C(\mathcal X)\simeq L_{C'}(\Ind(f)(\mathcal X)).
\]
We leave to the reader the verification that the latter is functorial, i.e. that it corresponds to a canonical isomorphism
\[
f_!L_C\simeq L_{C'}\circ\Ind(f).
\tag{8.6.2.3}
\]
In other words, \emph{the square (8.6.2.2) is commutative up to canonical isomorphism}. Taking account of the fact that $f_!$ commutes with small inductive limits, and of 8.5.1, we conclude as follows.

\begin{statement}{Proposition 8.6.3}
Let $f:C\to C'$ be a functor between $\U$-categories. Then the functor
\[
\Ind(f):\Ind(C)\longrightarrow\Ind(C')
\]
commutes with filtered inductive limits. Moreover, it makes the following diagram commutative:
\[
\begin{tikzcd}[column sep=3.2em,row sep=2.0em]
C \arrow[r,"f"] \arrow[d,"{c_C}"'] & C' \arrow[d,"{c_{C'}}"] \\
\Ind(C) \arrow[r,"{\Ind(f)}"'] & \Ind(C'),
\end{tikzcd}
\]
where the vertical arrows $c_C,c_{C'}$ are the canonical functors (8.4.1).
\end{statement}

The last assertion is trivial from the definitions and has been included for ease of reference. Note moreover that the properties stated in 8.6.3 \emph{characterize the functor} $\Ind(f)$ up to unique isomorphism as being induced by the functor $f_!$ (8.6.2.1), as follows from the proof just given of (8.6.2.3).

\begin{statement}{Proposition 8.6.4}
The notation is that of 8.6.3.
\begin{enumerate}[label=\alph*)]
\item For $\Ind(f)$ to be faithful, respectively fully faithful, it is necessary and sufficient that $f$ be so.
\item For $\Ind(f)$ to be an equivalence of categories, it is necessary and sufficient that $f$ be fully faithful and that every object of $C'$ be isomorphic to a direct factor (10.6) of an object in the image of $f$.
\end{enumerate}
\end{statement}

\begin{proofsketch}
\begin{enumerate}[label=\alph*)]
\item Necessity follows immediately from the fact that $f$ is induced by $\Ind(f)$. For sufficiency, it is enough to use the form (8.6.1.3) of $\Ind(f)$ on the sets $\Hom(\mathcal X,\mathcal Y)$, remembering that filtered inductive limits of sets, and arbitrary projective limits, carry monomorphisms to monomorphisms and isomorphisms to isomorphisms.

\item We may already suppose $f$, hence $\Ind(f)$, fully faithful. Since every object of $\Ind(C')$ is a small filtered inductive limit of objects of $C'$, full faithfulness of $\Ind(f)$ implies that for this functor to be essentially surjective it is equivalent that every object $X'$ of $C'$ lie in its essential image. If one has an isomorphism
\[
X'\xrightarrow{\sim}\textrm{``}\varinjlim\textrm{''}\,f(X_i),
\]
then this isomorphism factors through one of the $f(X_i)$; this shows that $X'$ is a direct factor of $f(X_i)$, proving necessity in b). For sufficiency, using the full faithfulness of $\Ind(f)$, the assertion follows at once from the following corollary.
\end{enumerate}
\end{proofsketch}

\begin{statement}{Corollary 8.6.5}
In $\Ind(C)$ the images of projectors (10.6) are representable, and the functor $\Ind(f):\Ind(C)\to\Ind(C')$ commutes with the formation of these images.
\end{statement}

Indeed, this follows from the fact that in $\Ind(C)$ small filtered inductive limits are representable (8.5.1) and that $\Ind(f)$ commutes with them (8.6.3), taking account of the fact that the image of a projector $p:X\to X$ may be interpreted as the inductive limit of the filtered inductive system indexed by $\mathbb N$
\[
X\xrightarrow{p}X\xrightarrow{p}X\longrightarrow\cdots,
\]
or, equivalently, as the inductive limit of the evident functor on the filtered category $P$ having a single object and a nonidentity arrow $\Pi$ such that $\Pi^2=\Pi$.
