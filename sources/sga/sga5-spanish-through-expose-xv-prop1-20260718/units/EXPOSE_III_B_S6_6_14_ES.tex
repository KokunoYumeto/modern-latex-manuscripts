\subsection*{6.14}
Sean $X$ un $G$-$S$-esquema y $j:U\to X$ una inmersión abierta equivariante,
\[
c:C\to X\times_SX
\]
un morfismo de $G$-$S$-esquemas tal que $c_2(=p_2c):C\to X$ sea cuasifinito y de dimensión Tor finita. Supongamos que
\[
c_1^{-1}(U)\subset c_2^{-1}(U).
\]
Se tienen, pues, unos cuadrados conmutativos
\begin{equation}
\begin{tikzpicture}[baseline=(current bounding box.center),>=stealth]
\node (U1) at (0,3.0) {$U$};
\node (C1U) at (2.8,3.0) {$c_1^{-1}(U)$};
\node (C2U) at (2.8,2.15) {$c_2^{-1}(U)$};
\node (U2) at (5.2,2.15) {$U$};
\node (X1) at (0,1.15) {$X$};
\node (C) at (2.8,1.15) {$C$};
\node (X2) at (5.2,1.15) {$X$};
\node (XU1) at (0,0) {$X-U$};
\node (C2XU) at (2.8,0) {$c_2^{-1}(X-U)$};
\node (XU2) at (5.2,0) {$X-U$};
\draw[->] (C1U) -- (U1);
\draw[->] (C1U) -- (C2U);
\draw[->] (C2U) -- (U2);
\draw[->] (U1) -- node[left] {$i$} (X1);
\draw[->] (U2) -- (X2);
\draw[->] (C2U) -- (C);
\draw[->] (C2XU) -- (C);
\draw[->] (C) -- node[above] {$c_1$} (X1);
\draw[->] (C) -- node[above] {$c_2$} (X2);
\draw[->] (XU1) -- node[left] {$j$} (X1);
\draw[->] (XU2) -- (X2);
\draw[->] (C2XU) -- (XU1);
\draw[->] (C2XU) -- (XU2);
\end{tikzpicture}
\tag{6.14.1}
\end{equation}
El morfismo de traza
\[
\Tr_{c_2}:c_{2!}\Lambda_C\longrightarrow \Lambda_X
\]
define entonces una correspondencia $G$-equivariante filtrada
\begin{equation}
\underline c\in\Hom(c_1^*\Lambda_X,c_2^!\Lambda_X),
\tag{6.14.2}
\end{equation}
estando $\Lambda_X$ provisto de la filtración definida por el submódulo $i_!\Lambda_U$, es decir,
\[
F^n\Lambda_X=\Lambda_X\quad(n\le0),\qquad
F^1\Lambda_X=i_!\Lambda_U,\qquad
F^n\Lambda_X=0\quad(n\ge1).
\]
El graduado asociado consta de las dos correspondencias $G$-equivariantes
\[
\operatorname{gr}^0\underline c=\underline c_{X-U,X}
\in\Hom(c_1^*\Lambda_{X-U,X},c_2^!\Lambda_{X-U,X}),
\]
\begin{equation}
\operatorname{gr}^1\underline c=\underline c_{U,X}
\in\Hom(c_1^*\Lambda_{U,X},c_2^!\Lambda_{X-U,X}),
\tag{6.14.3}
\end{equation}
donde hemos puesto
\[
\Lambda_{X-U,X}=j_*\Lambda_{X-U},\qquad
\Lambda_{U,X}=i_!\Lambda_U.
\]
Se tiene
\begin{equation}
\underline c_{X-U,X}=j_*\underline c_{X-U},
\tag{6.14.4}
\end{equation}
donde
\[
\underline c_{X-U}\in\Hom(c_1^*\Lambda_{X-U},c_2^!\Lambda_{X-U})
\]
es la correspondencia definida por $\Tr_{c_2,X-U}$. Escribiremos $\underline c$, $\underline c_{U,X}$, etc., cuando queramos precisar el anillo base.

Olvidemos la acción de $G$. Se tienen términos locales
\begin{equation}
\Tr_{\Lambda}(\underline c),\quad
\Tr_{\Lambda}(\underline c_{U,X}),\quad
\Tr_{\Lambda}(\underline c_{X-U,X})
\in H^0(X^c,K_{X^c}),
\tag{6.14.5}
\end{equation}
que, en virtud de la aditividad de las trazas (III 4.13.1), satisfacen la relación
\begin{equation}
\Tr_{\Lambda}(\underline c)
=
\Tr_{\Lambda}(\underline c_{U,X})
+
\Tr_{\Lambda}(\underline c_{X-U,X}).
\tag{6.14.6}
\end{equation}
Además, según 6.14.4, se tiene
\begin{equation}
\Tr_{\Lambda}(\underline c_{X-U,X})
=
j_*^c\Tr_{\Lambda}(\underline c_{X-U}),
\tag{6.14.7}
\end{equation}
donde
\[
j_*^c:H^0((X-U)^c,K_{(X-U)^c})\longrightarrow H^0(X^c,K_{X^c})
\]
es el morfismo de imagen directa definido por $j^c:(X-U)^c\to X^c$. Recordemos (III 4.3) que, si $X$ es liso, $c$ una inmersión cerrada y $X^c$ es finito, $\Tr_{\Lambda}(\underline c)$ es la imagen en $\Lambda$ de la multiplicidad de intersección $(C\cdot X)$ de $C$ con la diagonal; si además $X-U$ es liso, $\Tr_{\Lambda}(\underline c_{X-U})$ es la imagen en $\Lambda$ de $(c_2^{-1}(X-U)\cdot(X-U))$. Observemos también que, si $X-U$ es propio sobre $S$ y $c|X-U$ es la diagonal, entonces la fórmula de Lefschetz (III 4.7) implica que
\[
\int_{X-U}\Tr(\underline c_{X-U})\in\Lambda
\]
es la característica de Euler--Poincaré de $X-U$ con valores en $\Lambda$.
