\begin{statement}{Proposición 1}
\begin{enumerate}[label=\alph*)]
\item El isomorfismo $\Fr^*_{F/X}$ depende funtorialmente del haz $F$; dicho de otro modo, se tiene un isomorfismo funtorial
\[
\Fr^*_{/X}:\fr_X^*\longrightarrow \id_{\widetilde X_{\et}},
\]
donde $\widetilde X_{\et}$ es la categoría de los haces de conjuntos sobre $X_{\et}$.
\item La correspondencia de Frobenius $(\fr_X,\Fr^*_{F/X})$ sobre $(X,F)$ es compatible con el cambio de base en $X$. Es decir, si $\mathcal F$ denota la categoría fibrada sobre $(\Sch)_p$ de los haces, para la topología étale, sobre preesquemas variables de característica $p$, los funtores $\fr_X^*$, $X\in\Ob(\Sch)_p$, definen un $(\Sch)_p$-funtor cartesiano
\[
\fr^*: \mathcal F\longrightarrow\mathcal F,
\]
y los morfismos $\Fr^*_{F/X}$, $X\in\Ob(\Sch)_p$, $F\in\Ob\widetilde X_{\et}$, definen un $(\Sch)_p$-morfismo funtorial
\[
\Fr^*_{/}:\fr^*\longrightarrow\id_{\mathcal F}
\]
que es un isomorfismo.
\item Recíprocamente, estas propiedades caracterizan $\Fr^*_{/}$, es decir, existe un único $(\Sch)_p$-morfismo funtorial de $\fr^*$ a $\id_{\mathcal F}$.
\end{enumerate}
\end{statement}
