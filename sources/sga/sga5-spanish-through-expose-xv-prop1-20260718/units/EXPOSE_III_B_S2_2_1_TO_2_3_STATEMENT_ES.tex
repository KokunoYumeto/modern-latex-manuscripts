\subsection*{2. Reducción a un teorema de anulación}

\subsubsection*{2.1}
Bajo las hipótesis de 1.1, denotemos por $s$ (resp. $t$) la imagen de $z$ en $A$ (resp. $B$), y por $i:U=X-\{x\}\hookrightarrow X$ y $j:V=Y-\{y\}\hookrightarrow Y$ las inclusiones canónicas. Como $a_2$ y $b_1$ son finitos, se tiene $a_2^{-1}(y)=s$, $b_1^{-1}(x)=t$, y, por tanto,
\[
a_1^{-1}(U)\subset a_2^{-1}(V),\qquad b_2^{-1}(V)\subset b_1^{-1}(U),
\]
y se tienen diagramas conmutativos
\[
\begin{tikzcd}[column sep=large,row sep=large]
U \arrow[d,"i"] \arrow[rd,phantom,"(1)"{description}] & a_1^{-1}(U) \arrow[l,"a'_1"'] \arrow[r,"a'_2"] \arrow[d] \arrow[rd,phantom,"(2)"{description}] & V \arrow[d,"j"]\\
X & A \arrow[l,"a_1"] \arrow[r,"a_2"'] & Y
\end{tikzcd}
\qquad
\begin{tikzcd}[column sep=large,row sep=large]
U \arrow[d,"i"] \arrow[rd,phantom,"(3)"{description}] & b_2^{-1}(V) \arrow[l,"b'_1"'] \arrow[r,"b'_2"] \arrow[d] \arrow[rd,phantom,"(4)"{description}] & V \arrow[d,"j"]\\
X & B \arrow[l,"b_1"] \arrow[r,"b_2"'] & Y ,
\end{tikzcd}
\]
donde los cuadrados (1) y (4) son cartesianos.

\begin{statement}{Lema 2.2}
Para demostrar 1.2, basta considerar los dos casos siguientes:
\[
\text{a) } L_x=0,\quad M_y=0;\qquad
\text{b) } L|U=0,\quad M|V=0 .
\]
\end{statement}

Consideremos las filtraciones decrecientes sobre $L$ y $M$ definidas por los subcomplejos
\[
F^nL=
\begin{cases}
L,& n\le 0,\\
i_!i^*L,& n=1,\\
0,& n>1,
\end{cases}
\qquad
F^nM=
\begin{cases}
M,& n\le 0,\\
j_!j^*M,& n=1,\\
0,& n>1.
\end{cases}
\]
El complejo $a_{2*}a_1^*L$ (resp. $b_{1*}b_2^*M$) queda entonces filtrado por
\[
F^n a_{2*}a_1^*L=a_{2*}a_1^*F^nL.
\]
Se tiene $F^n a_{2*}a_1^*L=a_{2*}a_1^*L$ para $n\le 0$, y $0$ para $n>1$; además, como $a_2$ es finito y el cuadrado (1) es cartesiano, se tiene
\[
F^1a_{2*}a_1^*L=j_!(a'_2)_!(a'_1)^*i^*L.
\]
Del mismo modo, se tiene
\[
F^n b_{1*}b_2^*M=
\begin{cases}
b_{1*}b_2^*M,& n\le 0,\\
i_!(b'_1)_!(b'_2)^*j^*M,& n=1,\\
0,& n>1.
\end{cases}
\]

Podemos suponer que $u$ (resp. $v$) está definido por un homomorfismo de complejos
\[
u:a_{2*}a_1^*L\longrightarrow M
\quad
(\text{resp. }v:b_{1*}b_2^*M\longrightarrow L).
\]
Las fórmulas anteriores muestran que el homomorfismo compuesto
\[
F^1a_{2*}a_1^*L\longrightarrow a_{2*}a_1^*L\xrightarrow{u}M
\]
(resp. $F^1b_{1*}b_2^*M\to b_{1*}b_2^*M\xrightarrow{v}L$) se factoriza de manera única a través de $F^1M$ (resp. $F^1L$); en otras palabras, $u$ (resp. $v$) es un homomorfismo de complejos filtrados. Así, $u$ (resp. $v$) es el homomorfismo subyacente a una correspondencia filtrada, en el sentido de (III 4.13), que seguiremos denotando por
\[
u\in\Hom_{DF}(a_1^*L,a_2^!M),\qquad
v\in\Hom_{DF}(b_2^*M,b_1^!L).
\]
Por la aditividad de los productos cup ([3] V 3.8.7), (III 4.13.1), se tiene
\[
\langle u_z,v_z\rangle
 =\langle \operatorname{gr}^0u_z,\operatorname{gr}^0v_z\rangle
 +\langle \operatorname{gr}^1u_z,\operatorname{gr}^1v_z\rangle,
\]
y
\[
\langle u,v\rangle
 =\langle \operatorname{gr}^0u,\operatorname{gr}^0v\rangle
 +\langle \operatorname{gr}^1u,\operatorname{gr}^1v\rangle.
\]
De ello resulta el lema, pues $\operatorname{gr}^1L=i_!i^*L$ (resp. $\operatorname{gr}^1M=j_!j^*M$) tiene una fibra nula en $x$ (resp. $y$), mientras que $\operatorname{gr}^0L$ (resp. $\operatorname{gr}^0M$) está concentrado en $x$ (resp. $y$).

\begin{statement}{Lema 2.3}
Bajo las hipótesis de 1.1, supongamos que $L|U=0$, $M|V=0$. Entonces se tiene
\[
\langle u_z,v_z\rangle=\langle u,v\rangle .
\]
\end{statement}
