\begin{statement}{Corolario 1.2}
Sea $L$ un $A_S$-módulo. Para todo entero $i$, el morfismo de imagen inversa
\[
H^i(S,L)\longrightarrow H^i(X,p^*L)
\]
es un isomorfismo.
\end{statement}

\emph{Demostración.} Por (1.1), el morfismo de adjunción $L\to Rp_*(p^*L)$ es un isomorfismo. Como $H^*(X,p^*L)\simeq H^*(S,Rp_*p^*(L))$, se sigue la afirmación.

\begin{statement}{Proposición 1.3}
Supongamos que $E$ tiene rango constante $r$. Para todo $A_S$-módulo $L$, se verifican las propiedades siguientes.
\begin{enumerate}[label=(\roman*)]
\item
\begin{enumerate}[label=\alph*)]
\item
\[
R^i q_*(q^*L)=0\quad\text{si }i\ne 0,2r-1.
\]
\item El morfismo de adjunción
\[
L\longrightarrow q_*q^*(L)
\]
es un isomorfismo.
\item Denotando por $Y$ el complementario de $U$ en $X$, el morfismo
\[
R^{2r-1}q_*(q^*L)\longrightarrow L\otimes\mu_S^{\otimes -r}
\]
compuesto del morfismo de borde
\[
R^{2r-1}q_*(q^*L)\longrightarrow R_Y^{2r}p_*(p^*L)
\tag{1}
\]
y del isomorfismo de pureza (SGA 4 XVIII)
\[
R_Y^{2r}p_*(p^*L)\longrightarrow L\otimes\mu_S^{\otimes -r}
\]
es un isomorfismo.
\end{enumerate}
\item
\begin{enumerate}[label=\alph*)]
\item
\[
R_!^i q(q^*L)=0\quad\text{si }i\ne 1,2r.
\]
\item El morfismo de borde
\[
L\longrightarrow R_!^1 q(q^*L)
\]
procedente de la sucesión exacta larga, donde $f:Y\to S$ es la proyección canónica,
\[
\cdots\longrightarrow R_!^i q(q^*L)\longrightarrow R_!^i p(p^*L)
\longrightarrow R^i f_*(f^*L)\longrightarrow R_!^{i+1}q(q^*L)\longrightarrow\cdots
\]
es un isomorfismo.
\item El morfismo de traza
\[
R_!^{2r}q(q^*L)\longrightarrow L\otimes\mu_S^{\otimes -r}
\]
es un isomorfismo.
\end{enumerate}
\end{enumerate}
\end{statement}
