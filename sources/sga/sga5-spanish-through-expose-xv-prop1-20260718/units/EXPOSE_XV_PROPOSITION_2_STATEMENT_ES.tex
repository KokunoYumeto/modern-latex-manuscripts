\begin{statement}{Proposición 2}
Sean $S$ un preesquema de característica $p$ y $X$ un $S$-preesquema.
\begin{enumerate}[label=\alph*)]
\item El morfismo de Frobenius relativo
\[
\Fr_{X/S}:X\longrightarrow X^{(p/S)}
\]
es integral, sobreyectivo y radicial; es, por tanto, un homeomorfismo universal.

\item Si $X\to S$ es una inmersión abierta, $X^{(p)}=X$ y $\Fr_{X/S}=\id_X$. Si $X\to S$ es una inmersión cerrada que identifica $X$ con un subespacio anillado cerrado $(|X|,\OO_S/\JJ)$ de $S$, entonces
\[
X^{(p)}=(|X|,\OO_S/\JJ^p)
\]
y $\Fr_{X/S}=(\id_{|X|},\varphi)$, donde $\varphi:\OO_S/\JJ^p\to\OO_S/\JJ$ es la proyección canónica.

\item Supóngase que $X$ es localmente de presentación finita sobre $S$.
\begin{enumerate}[label=\arabic*)]
\item Las tres condiciones siguientes son equivalentes:
\[
\begin{array}{ll}
\text{(i)}& X\text{ es no ramificado sobre }S,\\
\text{(ii)}& \Fr_{X/S}\text{ es no ramificado},\\
\text{(iii)}& \Fr_{X/S}\text{ es un monomorfismo}.
\end{array}
\]
\item Del mismo modo, las tres condiciones siguientes son equivalentes:
\[
\begin{array}{ll}
\text{(i')}& X\text{ es étale sobre }S,\\
\text{(ii')}& \Fr_{X/S}\text{ es étale},\\
\text{(iii')}& \Fr_{X/S}\text{ es un isomorfismo}.
\end{array}
\]
\end{enumerate}
\end{enumerate}
\end{statement}
