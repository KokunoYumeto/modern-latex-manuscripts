Por consiguiente,

\begin{equation}\tag{5.6}
\Tr_{\bZ_\ell}(g_{X'}^{-1}f')_{H^\vee_{U',X'}(\ell)} = \sum_{x \in U^f} \sum_{x' \in {X'_x}^{g_{X'}^{-1}f'}} \nu_{x'}(g_{X'}^{-1}f') - \sum_{x \in Y^f} \sum_{x' \in {X'_x}^{g_{X'}^{-1}f'}} (1 - \nu_{x'}(g_{X'}^{-1}f'))
\end{equation}

Pero, dado $x \in U^f$, el conjunto ${X'_x}^{g_{X'}^{-1}f'}$ no es vacío si y solo si $\mathrm{NW}_{f}^{G,\varphi}(x) = 1$. Como se tiene además:

\begin{equation}\tag{5.7}
\card {X'_x}^{g_{X'}^{-1}f'} = c^\varphi(g) \quad , \quad x \in U \quad .
\end{equation}

\begin{equation}\tag{5.8}
\nu_{x'}(g_{X'}^{-1}f') = \nu_x(f) \quad , \quad x \in U \quad .
\end{equation}

las relaciones (5.2) y (5.6) dan:

\begin{equation}\tag{5.9}
\Tr_{\bZ_\ell[G],\varphi}(f'^\vee_{H_{U',X'}(\ell)})(g) = \sum_{x \in U^f} \nu_x(f) \mathrm{NW}_f^{G,\varphi}(x)(g) - \sum_{x \in Y^f} \sum_{x' \in {X'_x}^{g_{X'}^{-1}f'}} \frac{(1 - \nu_{x'}(g_{X'}^{-1}f'))}{c^\varphi(g)}
\end{equation}
es decir, la fórmula que se quería demostrar.
\end{prooftext}
