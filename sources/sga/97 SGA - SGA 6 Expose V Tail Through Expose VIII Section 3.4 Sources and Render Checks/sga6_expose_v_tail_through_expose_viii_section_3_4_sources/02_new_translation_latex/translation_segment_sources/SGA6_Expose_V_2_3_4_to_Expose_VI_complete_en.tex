\section*{2. Pre-$\lambda$-rings, continued}
\addcontentsline{toc}{section}{2. Pre-lambda-rings, continued}

\subsection*{2.3.4}
For every $x\in K$ and every integer $n\geq 1$ one has
\begin{equation}\label{eq:2.3.4}
(-1)^{n-1}n\lambda^n(x)=\sum_{i=1}^{n}(-1)^{i-1}\lambda^{n-i}(x)\psi^i(x).
\end{equation}
This formula makes it possible to express the operations $\lambda^n$ in terms of the operations $\psi^n$, and conversely.

\subsection*{2.3.5}
Put
\begin{equation}\label{eq:2.3.5a}
\lambda_{-1/t}(x)(-t)^n=\sum_{k=0}^{n}(-1)^k\lambda^k(x)t^{n-k}.
\end{equation}
Then
\begin{equation}\label{eq:2.3.5b}
\lambda^n\bigl(\lambda^m(x)\bigr)=P_{n,m}\bigl(\lambda^1(x),\ldots,\lambda^{nm}(x)\bigr),
\end{equation}
where $P_{n,m}$ is a universal isobaric polynomial of weight $nm$.

\subsection*{2.3.6}
Let $K$ be a pre-$\lambda$-ring. We define a filtration $\Fil^\bullet K$ of $K$ by putting
\begin{equation}\label{eq:2.3.6}
\Fil^0K=K,
\qquad
\Fil^1K=K,
\qquad
\Fil^nK=\text{the subgroup generated by the }x\text{ such that }\lambda^i(x)=0\text{ for }i<n.
\end{equation}
One checks that
\[
\Fil^pK\cdot \Fil^qK\subset \Fil^{p+q}K.
\]

\section*{Grothendieck operations}
\addcontentsline{toc}{section}{Grothendieck operations}

\subsection*{2.3.7}
Let $K$ be a pre-$\lambda$-ring. The \emph{Grothendieck operations} $\gamma^i$ are defined by
\begin{equation}\label{eq:2.3.7a}
\gamma_t(x)=\lambda_{t/(1-t)}(x)=\sum_{n\geq 0}\gamma^n(x)t^n.
\end{equation}
Equivalently,
\begin{equation}\label{eq:2.3.7b}
\gamma^n(x)=\lambda^n(x+n-1)=\sum_{i=0}^{n}(-1)^i\binom{n-1}{i}\lambda^{n-i}(x).
\end{equation}

\subsection*{2.3.8}
The following formulas hold:
\begin{equation}\label{eq:2.3.8}
\gamma^0(x)=1,
\qquad
\gamma^1(x)=x,
\qquad
\gamma^n(x+y)=\sum_{i+j=n}\gamma^i(x)\gamma^j(y).
\end{equation}
The operations $\gamma^i$ define a new pre-$\lambda$-ring structure, called the \emph{opposite} structure.

\subsection*{2.3.9}
One has
\begin{equation}\label{eq:2.3.9}
\gamma^n(xy)=P'_n\bigl(\gamma^1(x),\ldots,\gamma^n(x);\gamma^1(y),\ldots,\gamma^n(y)\bigr),
\end{equation}
where $P'_n$ is a universal polynomial.

\subsection*{2.4}
A \emph{$\lambda$-ring} is a pre-$\lambda$-ring $K$ such that the homomorphism
\[
\lambda_t:K\longrightarrow 1+K[[t]]_+
\]
is a $\lambda$-homomorphism, that is, is compatible with the pre-$\lambda$-ring structures. This means that
\begin{equation}\label{eq:2.4}
\lambda^n\bigl(\lambda^m(x)\bigr)=P_{n,m}\bigl(\lambda^1(x),\ldots,\lambda^{nm}(x)\bigr)
\end{equation}
for all $x$ and all $n,m$, where $P_{n,m}$ is the universal polynomial of 2.3.5.

\subsection*{2.4.1}
The following conditions are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $K$ is a $\lambda$-ring.
\item The Adams operations satisfy $\psi^n\circ\psi^m=\psi^{nm}$ for all $n,m$.
\item For every finite-dimensional element $x$, the elements $\lambda^n(x)$ satisfy the universal relations.
\end{enumerate}

\subsection*{2.4.2}
If $K$ is a $\lambda$-ring, the Adams operations $\psi^n$ are ring endomorphisms and satisfy
\begin{equation}\label{eq:2.4.2}
\psi^n\circ\psi^m=\psi^{nm}.
\end{equation}
If $K$ is also augmented, the operations $\psi^n$ are compatible with the augmentation.

\subsection*{2.5}
Let $K$ be a pre-$\lambda$-ring. Denote by $\mathfrak W(K)$ the multiplicative group $1+K[[t]]_+$. We have seen that $\mathfrak W(K)$ carries a pre-$\lambda$-ring structure. If $K$ is a $\lambda$-ring, then $\mathfrak W(K)$ is again a $\lambda$-ring.

\subsection*{2.5.1}
The element $1+t$ is the unit for the product $\circ$. One has
\begin{equation}\label{eq:2.5.1}
\lambda^n(1+t)=1\quad (n\geq 2),
\qquad
\psi^n(1+t)=1+t^n.
\end{equation}

\subsection*{2.6}
Let $K$ be a ring. We denote by $\mathfrak W_\circ(K)$ the functor $\mathfrak W$ equipped with the product $\circ$ defined in 2.3 and with its operations $\lambda^i$. We denote by $\mathfrak W_\gamma(K)$ the same functor equipped with the product and operations arising from the operations $\gamma^i$.

\subsection*{2.6.1}
There is a commutative diagram
\[
\begin{array}{ccc}
\mathfrak W_\circ(K) & \xrightarrow{\lambda_t} & \mathfrak W_\circ(\mathfrak W_\circ(K))\\
\downarrow && \downarrow\\
\mathfrak W_\gamma(K) & \xrightarrow{\gamma_t} & \mathfrak W_\gamma(\mathfrak W_\gamma(K)).
\end{array}
\]

\section*{3. The $\gamma$-operations}
\addcontentsline{toc}{section}{3. The gamma-operations}

\subsection*{3.1}
Let $K$ be a ring and let $\mathfrak W(K)=1+K[[t]]_+$. The opposite pre-$\lambda$-ring structure on $K$ defines operations $\gamma^i$, related to the operations $\lambda^i$ by
\begin{equation}\label{eq:3.1}
\gamma_t(x)=\lambda_{t/(1-t)}(x).
\end{equation}

\subsection*{3.2}
For every $x\in K$ and every integer $n\geq 0$,
\begin{equation}\label{eq:3.2}
\gamma^n(x)=\lambda^n(x+n-1).
\end{equation}
This follows from the definition and the binomial formula for exterior powers.

\subsection*{3.3}
Let $K$ be a pre-$\lambda$-ring. We say that $K$ is a $\gamma$-ring if the operations $\gamma^i$ define a $\lambda$-ring structure on $K$, that is, if
\[
\gamma_t:K\longrightarrow \mathfrak W_\gamma(K)
\]
is a $\lambda$-homomorphism.

\subsection*{3.4}
The following conditions are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $K$ is a $\gamma$-ring.
\item The operations $\gamma^i$ satisfy the universal relations.
\item The operations $\psi^n$ satisfy $\psi^n\circ\psi^m=\psi^{nm}$.
\end{enumerate}

\subsection*{3.5}
It is immediate that a $\lambda$-ring is a $\gamma$-ring and conversely; indeed the conditions expressed in terms of the $\psi^n$ are identical. Moreover,
\begin{equation}\label{eq:3.5.1}
\gamma^n(xy)=\sum_{i+j=n}P'_{i,j}\bigl(\gamma^1(x),\ldots;\gamma^1(y),\ldots\bigr),
\end{equation}
where the $P'_{i,j}$ are the universal polynomials of 2.3.9.

\subsection*{3.6}
Let $K$ be a ring. The functor $\mathfrak W_\gamma(K)$ is a $\gamma$-ring, just as $\mathfrak W_\circ(K)$ is a $\lambda$-ring. There is a canonical isomorphism
\begin{equation}\label{eq:3.6}
\mathfrak W_\circ(K)\simeq \mathfrak W_\gamma(K),
\end{equation}
given by the change of variable $t\mapsto t/(1-t)$.

\subsection*{3.7}
Let $K$ be a $\lambda$-ring and let $A$ be a $K$-$\lambda$-algebra. The canonical $\gamma$-filtration of $A$ is defined by
\begin{equation}\label{eq:3.7}
\Fil^n_\gamma A=	ext{the ideal generated by the products }\gamma^{i_1}(x_1)\cdots\gamma^{i_k}(x_k)
\end{equation}
with $\sum i_j\geq n$ and $\varepsilon(x_j)=0$. It is decreasing:
\[
A=\Fil^0_\gamma A\supset \Fil^1_\gamma A\supset\cdots\supset \Fil^n_\gamma A\supset\cdots,
\]
and
\begin{equation}\label{eq:3.7.1}
\Fil^0_\gamma A=A,
\qquad
\Fil^1_\gamma A=\Ker(\varepsilon),
\qquad
\Fil^pA\cdot\Fil^qA\subset\Fil^{p+q}A.
\end{equation}

\subsection*{3.8}
Let $K$ be a binomial ring and let $A$ be a $K$-$\lambda$-algebra. Then
\begin{equation}\label{eq:3.8}
\gamma_t(x)=(1/(1-t))^x=\exp(-x\log(1-t)).
\end{equation}
If $a\in K$, multiplication by $\gamma_t(a)$ for the $\gamma$-law is characterized by its value on $1+Xt$:
\begin{equation}\label{eq:3.8.1}
\gamma_t(a)(1+Xt)=(1+Xt)^a=\exp(a\log(1+Xt))=\sum_{n=0}^{\infty}\binom{a}{n}X^nt^n.
\end{equation}
In particular the $n$-th universal polynomial defining multiplication by $\gamma_t(a)$ for the $\gamma$-law is the polynomial $R_{n,a}$ of 2.9.

\subsection*{3.9--3.10}
Let $A$ be an augmented $K$-$\lambda$-algebra, with augmentation $\varepsilon:A\to K$. The \emph{canonical filtration} of $A$ is the ring filtration for which $\Fil^nA$ is the ideal generated by products
\begin{equation}\label{eq:3.10}
\gamma^{i_1}(x_1)\cdots\gamma^{i_k}(x_k),
\qquad
\sum i_j\geq n,
\qquad
\varepsilon(x_j)=0.
\end{equation}
Thus
\begin{equation}\label{eq:3.10.1}
\Fil^0A=A,
\qquad
\Fil^1A=\Ker\varepsilon.
\end{equation}
This filtration is functorial in $A$: every $\lambda$-homomorphism of augmented $K$-$\lambda$-algebras is compatible with the filtrations. Unless expressly stated otherwise, this is the filtration considered on an augmented $\lambda$-ring.

\subsection*{3.11}
Let $A$ be a $\lambda$-ring. Denote by $\Fil^n_\lambda A$ the filtration defined using the operations $\lambda^i$, equivalently the canonical $\gamma$-filtration of the opposite structure. One has
\begin{equation}\label{eq:3.11}
\Fil^n_\lambda A=\Fil^n_\gamma A.
\end{equation}
Indeed the two filtrations are defined by the same conditions, in view of the equivalence between $\lambda$-rings and $\gamma$-rings.

\subsection*{3.11.1--3.11.2}
Let $A$ be a $\lambda$-ring and let $x_1,\ldots,x_n$ be elements of $A$ with $\varepsilon(x_i)=0$. Then
\begin{equation}\label{eq:3.11.1}
\gamma^n(x_1+\cdots+x_n)\equiv x_1\cdots x_n\pmod{\Fil^{n+1}A}.
\end{equation}
If $x_1,\ldots,x_n$ are elements of filtration at least $1$, then
\begin{equation}\label{eq:3.11.2}
\gamma^n(x_1+\cdots+x_n)\equiv \sum_{i=1}^{n}x_i\pmod{\Fil^2A}.
\end{equation}

\section*{4. $\lambda$-rings generated by generators and relations}
\addcontentsline{toc}{section}{4. Lambda-rings generated by generators and relations}

\subsection*{4.1}
We work throughout over $\mathcal C_0$. Let $K$ be a $\lambda$-ring. Consider a family of indeterminates $(X_i)_{i\in I}$ and put
\[
A=K[(X_i), (\lambda^nX_i)_{i\in I,n\geq 2}].
\]
We wish to define on $A$ a $\lambda$-ring structure extending that of $K$. To give a homomorphism $\lambda_t:A\to\mathfrak W(A)$, it is enough to give the images of the $X_i$ and of the $\lambda^nX_i$; these determine a ring homomorphism $A\to\mathfrak W(A)_\circ$. We put
\begin{equation}\label{eq:4.1.1}
\lambda_t(X_i)=1+X_it+\sum_{n\geq 2}\lambda^nX_i\,t^n,
\qquad
\lambda_t(\lambda^nX_i)=\lambda^n(\lambda_t(X_i)).
\end{equation}
Since the relevant diagram is a diagram of ring homomorphisms, it is enough to check commutativity on the $X_i$ and $\lambda^nX_i$; this follows from the definitions and from the fact that $\lambda_u$ and $\mathfrak W^u(\lambda_t)$ are $\lambda$-homomorphisms.

\subsection*{4.2}
The $\lambda$-ring $A$ is characterized by the following universal property.

\begin{proposition}
For every $K$-$\lambda$-algebra $B$ and every family $(b_i)_{i\in I}$ of elements of $B$, there exists a unique $K$-$\lambda$-homomorphism $f:A\to B$ such that $f(X_i)=b_i$ for all $i\in I$.
\end{proposition}

For $f$ to be a $\lambda$-homomorphism it is necessary that $f(\lambda^jX_i)=\lambda^j(b_i)$ for all $i,j$, and this determines $f$ uniquely. The resulting homomorphism is a $\lambda$-homomorphism because the equality $\lambda_t\circ f=\mathfrak W(f)\circ\lambda_t$ need only be checked on generators of $A$ over $K$.

\subsection*{4.3--4.4}
One may make the same construction using the $\gamma$'s instead of the $\lambda$'s. Let
\[
A'=K[(X_i),(\gamma^nX_i)_{i\in I,n\geq 2}].
\]
The structure is defined by
\begin{equation}\label{eq:4.3.1}
\gamma_t(X_i)=1+X_it+\sum_{n\geq 2}\gamma^nX_i\,t^n,
\qquad
\gamma_t(\gamma^nX_i)=\gamma'^n(\gamma_t(X_i)).
\end{equation}
The same universal property as in 4.2 holds, hence $A'$ is canonically isomorphic, as a $\lambda$-ring, to $A$.

The ring $A$ is called the \emph{free $\lambda$-ring} generated over $K$ by the family $(X_i)_{i\in I}$.

Let $A$ be a $\lambda$-ring and $J$ an ideal of $A$. We say that $J$ is a \emph{$\lambda$-ideal} if for every $x\in J$ and every $n\geq 1$ one has $\lambda^n(x)\in J$. The kernel of a $\lambda$-homomorphism is a $\lambda$-ideal. Conversely, if $J$ is a $\lambda$-ideal, then $A/J$ inherits a $\lambda$-ring structure.

\subsection*{4.5--4.7}
An intersection of $\lambda$-ideals is a $\lambda$-ideal. Hence there is a smallest $\lambda$-ideal containing any given family $(x_i)_{i\in I}$; by the relations of 2.4.2 it is the ordinary ideal generated by the $\lambda^n(x_i)$, $n\geq 1$.

Let $(P_{i'})_{i'\in I'}$ be a family of elements in the free $\lambda$-ring $A=K[(X_i),(\lambda^nX_i)]$. If $J$ is the $\lambda$-ideal generated by the $P_{i'}$, then $A/J$ is called the \emph{$\lambda$-ring generated} over $K$ by the generators $(X_i)_{i\in I}$ subject to the relations $(P_{i'})_{i'\in I'}$. The same construction can be made with relations among the $X_i$ and the $\gamma^nX_i$, since an ideal is a $\lambda$-ideal if and only if it is stable under all operations $\gamma^n$.

If $B$ is a $K$-$\lambda$-algebra and $(b_i)$ is a family of elements of $B$, the relations $P_{i'}$ can be evaluated at $b_i,\lambda^n(b_i)$. If all these evaluations vanish, then there is a unique $K$-$\lambda$-homomorphism $A/J\to B$ sending $X_i$ to $b_i$.

\subsection*{4.8--4.9}
Let $n\geq 0$. In the free $\lambda$-ring on one indeterminate $X$ over $K$, the ideal generated by the $\lambda^kX$, $k>n$, is a $\lambda$-ideal. Indeed, if $A=K[X,\lambda^kX]_{k\geq 2}$ and $B=A/J$, one has to show that the composite $A\xrightarrow{\lambda_t}\mathfrak W(A)\to\mathfrak W(B)$ vanishes on $J$. This reduces to the following elementary fact: for any ring $C$ and any series $\varphi\in\mathfrak W(C)$ whose coefficients vanish in degrees $>n$, one has $\lambda^k(\varphi)=1$ for $k>n$. It is enough to prove this for
\[
C=K[X_1,\ldots,X_n],
\qquad
\varphi=1+X_1t+\cdots+X_nt^n.
\]
After adjoining roots by the elementary symmetric functions, one may write
\[
1+X_1t+\cdots+X_nt^n=\prod_{i=1}^{n}(1+T_it),
\]
and therefore
\[
\lambda_u(1+X_1t+\cdots+X_nt^n)=\prod_{i=1}^{n}\lambda_u(1+T_it)=\prod_{i=1}^{n}(1+\{1+T_it\}u),
\]
which has degree $n$ in $u$.

\begin{corollary}
The $\lambda$-ring generated over $K$ by one generator $X$ subject to the relations $\lambda^k(X)=0$ for $k>n$ is isomorphic to $K[X,\ldots,\lambda^nX]$.
\end{corollary}

The same assertions hold with $\lambda$ replaced by $\gamma$. One may also replace $X$ by a family $(X_i)$ subject to relations $\lambda^k(X_i)=0$ for $k>n_i$.

\subsection*{4.10--4.15}
Let $K$ be a binomial ring and $A$ an augmented $K$-$\lambda$-algebra. Let
\[
B=A[X,\gamma^kX]_{k\geq 2}
\]
be the free $\lambda$-ring on $X$ over $A$, augmented by sending $X$ to $0$. We grade $B$ by the weight in the variables $\gamma^kX$, giving $\gamma^kX$ weight $k$, and denote by $M_k$ the set of monomials of weight $k$.

\begin{lemma}
Let $a\in A$. The coefficient of a monomial of weight $n$ in
\[
\gamma^k(aX)=P'_k(\gamma^i(a),\gamma^jX)
\]
has filtration at least $k-n$ in $A$.
\end{lemma}

The polynomial $P'_k$ has no isobaric component of weight $<k$, so the lemma is clear when $a$ has augmentation zero. In general one writes
\[
\gamma_t(aX)=\gamma_t((a-\varepsilon(a))X)\gamma_t(\varepsilon(a)X),
\]
and uses
\[
\gamma_t(\varepsilon(a)X)=(\gamma_tX)^{\varepsilon(a)}=\sum_{n=0}^{\infty}R_{n,\varepsilon(a)}(\gamma^jX)t^n,
\]
with $R_{n,a}$ isobaric of weight $n$.

\begin{proposition}\label{prop:4.12}
One has
\begin{equation}\label{eq:4.12}
\Fil^kB=\Fil^kA+\Fil^{k-1}A\cdot M_1+\cdots+\Fil^0A\cdot M_k+B\cdot M_{k+1}.
\end{equation}
\end{proposition}

The inclusion of the right hand side in the left is clear. Conversely, since the right hand side defines a ring filtration, it suffices to treat the generators $\gamma^k(b)$ with $\varepsilon(b)=0$. Writing $b=\sum a_im_i$ with $m_i$ a monomial in the $\gamma^jX$, additivity reduces the assertion to $\gamma^k(am_i)$. Using the lemma and the $A$-$\lambda$-homomorphism sending $X$ to $m_i$, one reduces further to $\gamma^k(m_i)$. Finally the absence of components of weight $<k$ in the polynomials $P'_k$ and $Q'_{j,k}$ gives the result.

The decomposition in \eqref{eq:4.12} is a direct sum decomposition of $A$-modules: if
\[
b=\sum_{j\leq k}a_{i,j}m_{i,j}+b'
\]
with $m_{i,j}\in M_j$ and $b'\in B\cdot M_{k+1}$, then $b\in\Fil^kB$ implies $a_{i,j}\in\Fil^{k-j}A$. The assertion also remains valid after quotienting by any $\lambda$-ideal contained in the augmentation ideal, and with any family of indeterminates instead of the single $X$.

\begin{corollary}
\[
\Gr^kB\simeq \Gr^kA\oplus \Gr^{k-1}A\cdot M_1\oplus\cdots\oplus\Gr^0A\cdot M_k.
\]
\end{corollary}

If $B'$ is the $\lambda$-ring generated over $K$ by $X$ with relations $\gamma^k(X)=0$ for $k>n$, so $B'=K[X,\ldots,\gamma^nX]$, then its $\lambda$-ring filtration is the same as the weight filtration in the variables $\gamma^jX$.

\section*{5. The elements $\lambda^p(N,x)$}
\addcontentsline{toc}{section}{5. The elements lambda p of N,x}

This section will only be used in Expose VII, section 4, for the study of regular immersions; it may be omitted on a first reading.

\subsection*{5.1--5.3}
Let $d\geq 1$. Let $A$ be the $\ZZ$-$\lambda$-algebra generated by $N$ and $x$, subject to the relations $\lambda^i(N)=0$ for $i>d$. Thus
\[
A=\ZZ[N,\ldots,\lambda^dN;x,\ldots,\lambda^jx,\ldots].
\]
Put
\[
\lambda_{-1}(N)=\sum_{i=0}^{d}(-1)^i\lambda^i(N).
\]

\begin{lemma}
For every $p\geq 1$, $\lambda^p(x\lambda_{-1}(N))$ is divisible by $\lambda_{-1}(N)$.
\end{lemma}

Since $\lambda^p(x\lambda_{-1}(N))$ is isobaric of weight $p$ in the quantities $\lambda^j(\lambda_{-1}(N))$, it suffices to prove the lemma for $\lambda^p(\lambda_{-1}(N))$. Let $B$ be the $\lambda$-ring generated over $\ZZ$ by $N_1,\ldots,N_d$ with $\lambda^i(N_j)=0$ for $i\geq 2$. Sending $N$ to $\sum_iN_i$ defines a $\lambda$-homomorphism $A\to B$ which identifies $A$ with the symmetric polynomials in the $N_i$. Then
\[
\lambda_{-1}(N)=\prod_{i=1}^{d}(1-N_i).
\]
If $N_i=1$ for some $i$, then $\lambda_t(\lambda_{-1}(N))$ becomes $\lambda_t(0)=1$, so all $\lambda^p(\lambda_{-1}(N))$ vanish. Hence each is divisible by every $1-N_i$, and therefore by their product.

Since $A$ is integral, we define $\lambda^p(N,x)$ unambiguously by
\begin{equation}\label{eq:5.3.1}
\lambda^p(N,x)\lambda_{-1}(N)=\lambda^p(x\lambda_{-1}(N)),
\qquad p\geq 1.
\end{equation}
Thus $\lambda^p(N,x)$ is a polynomial in the $\lambda^i(x)$ and $\lambda^j(N)$, isobaric of weight $p$ in the $\lambda^i(x)$. If a $\lambda$-ring $B$ contains an element $N''$ with $\lambda^k(N'')=0$ for $k>d$, then $\lambda^p(N'',x'')$ is defined by substitution; it does not depend on the chosen bound $d$.

\subsection*{5.4}
In the $\lambda$-ring generated by $N,x,y$ subject to $\lambda^k(N)=0$ for $k>d$, formula \eqref{eq:5.3.1} gives
\begin{align*}
\lambda^p(N,x+y)\lambda_{-1}(N)
&=\lambda^p((x+y)\lambda_{-1}(N))\\
&=\sum_{i+j=p}\lambda^i(x\lambda_{-1}(N))\lambda^j(y\lambda_{-1}(N)).
\end{align*}
Since the ring is integral, this yields the identity
\begin{equation}\label{eq:5.4.1}
\lambda^p(N,x+y)=\sum_{\substack{i+j=p\\ i,j\neq 0}}\lambda^i(N,x)\lambda^j(N,y)\lambda_{-1}(N)+\lambda^p(N,x)+\lambda^p(N,y).
\end{equation}

If $N'$ is another element with $\lambda^k(N')=0$ for $k>d'$, then $\lambda^k(N+N')=0$ for $k>d+d'$, and additivity gives
\begin{equation}\label{eq:5.4.2}
\lambda^p(N+N',x)\lambda_{-1}(N)=\lambda^p(N',x\lambda_{-1}(N)).
\end{equation}

\subsection*{5.5}
Let $A$ be any $\lambda$-ring and let $N\in A$ satisfy $\lambda^k(N)=0$ for $k>d$. We associate to $A$ a new $\lambda$-ring $A_N$. First define on $A$ a new multiplication by
\begin{equation}\label{eq:5.5.1}
x\cdot_N y=xy\lambda_{-1}(N).
\end{equation}
This ring generally has no unit, so we adjoin a unit and consider $\ZZ\times A$, with
\[
(n,x)+(n',x')=(n+n',x+x')
\]
and
\begin{equation}\label{eq:5.5.2}
(n,x)(n',x')=(nn',nx'+n'x+x\cdot_N x').
\end{equation}
The ring $A$ embeds by $x\mapsto(0,x)$. Define a pre-$\lambda$-ring structure by
\begin{equation}\label{eq:5.5.3}
\lambda_t(n,x)=(1+t)^n\sum_{p=0}^{\infty}\lambda^p(N,x)t^p.
\end{equation}
It is a $\lambda$-ring: by substitution it suffices to treat the universal case in which $A$ is generated over $\ZZ$ by $N,X,Y$ with $\lambda^k(N)=0$ for $k>d$. There is then a homomorphism $A_N\to A$ sending $(1,0)$ to $1$ and $(0,x)$ to $x\lambda_{-1}(N)$; by \eqref{eq:5.5.1} and \eqref{eq:5.3.1} it is a $\lambda$-homomorphism, and it is injective in the universal integral case. Since $A$ is a $\lambda$-ring, so is $A_N$.

In $A_N$ one can form the $\gamma$-operations. For $x\in A$,
\[
\gamma^p(0,x)=\sum_{i=0}^{p-1}\binom{p-1}{i}\lambda^i(N,x),
\]
which lies in $A$; equivalently it is characterized by
\begin{equation}\label{eq:5.5.4}
\gamma^p(N,x)\lambda_{-1}(N)=\gamma^p(x\lambda_{-1}(N)).
\end{equation}

\subsection*{5.6}
Let $A$ be the $\lambda$-ring generated over $\ZZ$ by $N,L,x$ subject to $\lambda^k(N)=0$ for $k>d$ and $\lambda^k(L)=0$ for $k>1$. We compute $\lambda^p(N+L,x)$ in terms of the $\lambda^q(N,x)$. By definition,
\[
\lambda^p(N+L,x)\lambda_{-1}(N)\lambda_{-1}(L)=\lambda^p(x\lambda_{-1}(N)\lambda_{-1}(L)),
\]
hence
\begin{align*}
1+\lambda_{-1}(N)\lambda_{-1}(L)\sum_{k\geq 1}\lambda^k(N+L,x)t^k
&=\lambda_t(x\lambda_{-1}(N))\circ\lambda_t(1-L)\\
&=\frac{\lambda_t(x\lambda_{-1}(N))}{\lambda_{Lt}(x\lambda_{-1}(N))}\\
&=1+\frac{\sum_{k\geq 1}\lambda^k(N,x)\lambda_{-1}(N)(1-L^k)t^k}{1+\sum_{k\geq 1}\lambda^k(N,x)\lambda_{-1}(N)L^kt^k}.
\end{align*}

\subsection*{5.7}
Let $K$ be a binomial ring, $A$ an augmented $K$-$\lambda$-algebra, and $N\in A$ such that $\lambda^k(N)=0$ for $k>d$. Then for every $x\in A$ and every $p\geq 1$,
\begin{equation}\label{eq:5.7.1}
\gamma^p(N,x)\in\Fil^{p-d}A.
\end{equation}
Indeed, let $A_\circ$ be the $K$-$\lambda$-algebra generated by $N'_\circ,X'_\circ$ with $\lambda^k(N'_\circ)=0$ for $k>d$, and augmented by $\varepsilon(N'_\circ)=\varepsilon(X'_\circ)=0$. This is the polynomial ring over $K$ in the variables $\gamma^i(N'_\circ)$, $1\leq i\leq d$, and $\gamma^j(X'_\circ)$, $j\geq 1$; its $\lambda$-filtration is the weight filtration. Put
\[
N_\circ=N'_\circ+d,
\qquad
X_\circ=X'_\circ+\varepsilon(x).
\]
For $N_\circ,X_\circ$ the assertion is equivalent to
\[
\gamma^p(N_\circ,X_\circ)\gamma^d(N'_\circ)\in\Fil^pA_\circ.
\]
Since $\gamma^d(N'_\circ)=(-1)^d\lambda_{-1}(N_\circ)$, this becomes
\[
(-1)^d\gamma^p(X_\circ\gamma^d(N'_\circ))\in\Fil^pA_\circ,
\]
which holds because $\gamma^d(N'_\circ)$ has zero augmentation. A $K$-$\lambda$-homomorphism $A_\circ\to A$ compatible with augmentations sends $N'_\circ$ to $N-d$ and $X'_\circ$ to $x-\varepsilon(x)$; compatibility with filtrations gives the desired result.

\section*{6. Chern ring}
\addcontentsline{toc}{section}{6. Chern ring}

Throughout this section the base ring is a fixed binomial ring $K$. We shall use the category $\mathcal G$ of 1.1, and therefore return to the notation of section 1: $\Go$ denotes the functor defined on $\mathcal G$, and $\Goi$ its restriction to $\mathcal G_0$.

\subsection*{6.1}
In 3.5 and 3.6 we defined morphisms of functors
\[
\Goi\times\Goi\xrightarrow{\star}\Goi,
\qquad
\Got\circ i\xrightarrow{\gamma'}\Gou\circ i\circ\Got\us,
\qquad
\Goi\xrightarrow{a}\Goi,
\]
the last being multiplication, for the law $\circ$, by $\gamma^t(a)$ with $a\in K$. By 1.9 these yield morphisms
\[
\Go\times\Go\xrightarrow{\star}\Go_g,
\qquad
\Got\xrightarrow{\gamma'}\Gou\circ i\circ\Got\us,
\qquad
\Go\xrightarrow{a}\Go_g.
\]
The isobaricity of the polynomials $P'_n,Q'_{i,j},R_{n,a}$ implies that the images of the first and third functors lie in $\Go$, and that the image of the second lies in the subgroup of formal series in $u$ whose non-zero degree coefficients lie in $\Got$.

Composing the first and third morphisms with the canonical morphism $\Go\to 1+\Go^+$ gives two morphisms of functors on $\mathcal G$:
\[
(1+\Go^+)\times(1+\Go^+)\xrightarrow{\star}1+\Go^+,
\qquad
1+\Go^+\xrightarrow{a}1+\Go^+.
\]
They define on $1+\Go^+$ a multiplication and operations of $K$. The multiplication, denoted $*$, is determined by its value on $(1+X,1+Y)$:
\begin{equation}\label{eq:6.1.1}
(1+X)*(1+Y)=\frac{1+X+Y}{(1+X)(1+Y)}.
\end{equation}
The $n$-th universal polynomial for this law is the isobaric component of weight $n$ of $P'_n$, denoted $P''_n$. The operations of $K$ are again written exponentially; the universal polynomials defining them are the lowest-degree isobaric components of the $R_{n,a}$, and since the $R_{n,a}$ are themselves isobaric, these components are the $R_{n,a}$.

For $A\in\Obj(\mathcal G)$, the ring $1+\widehat A^+$ defined by these laws generally has no unit. We extend it by adjoining the scalars of $K$, obtaining the ring $K\times(1+\widehat A^+)$ with
\[
(a,\varphi)+(a',\varphi')=(a+a',\varphi\varphi')
\]
and
\begin{equation}\label{eq:6.1.2}
(a,\varphi)(a',\varphi')=(aa',\varphi^{a'}\varphi'^{a}(\varphi*\varphi')).
\end{equation}
The unit is $(1,1)$, and $1+\widehat A^+$ is the ideal $(0)\times(1+\widehat A^+)$. It is naturally a $K$-algebra, with
\begin{equation}\label{eq:6.1.3}
a\cdot(a',\varphi)=(a,1)(a',\varphi)=(aa',\varphi^a).
\end{equation}

The preceding construction of $\gamma'$ and the canonical morphism $\Go\to 1+\Go^+$ give a morphism
\[
\Got\xrightarrow{\gamma'}\Gou\circ i\circ\bigl(K\times(1+\Go^+)\bigr),
\]
and therefore a morphism of functors
\[
1+\Go^+\longrightarrow\Gou\circ i\circ\bigl(K\times(1+\Go^+)\bigr).
\]
The universal polynomials defining it are the coefficients of $\gamma'^i(1+X_1+\cdots+X_n+\cdots)$; the $j$-th coefficient is the isobaric component of weight $j$ of $Q'_{i,j}$, denoted $Q''_{i,j}$.

Extending this morphism to $K\times(1+\Go^+)$ using the $\lambda$-structure of $K$, one obtains a pre-$\lambda$-ring structure whose corresponding $\gamma$-operations satisfy
\begin{equation}\label{eq:6.1.4}
\gamma_t(a,\varphi)=\gamma_t(a)\left(1+\sum_{p\geq 1}\gamma'^p(\varphi)t^p\right).
\end{equation}
We write these operations simply as $\gamma$. The verification that this is a $\lambda$-ring structure reduces to the commutativity of the diagram of 3.7.2; it is known on $K$ and follows on $1+\Go^+$ from the corresponding diagram for $\Go$ by passing to the quotient.

\begin{definition}
The functor defined above on graded $K$-algebras, with values in augmented $K$-$\lambda$-algebras, is called the \emph{Chern ring}. Its value on $A$ is denoted $\Ch(A)$, or $\Ch_K(A)$ if the base must be specified.
\end{definition}

From \eqref{eq:6.1.1}, \eqref{eq:6.1.2}, \eqref{eq:6.1.4}, and 3.6.2, one obtains
\begin{equation}\label{eq:6.2.1}
(1,1+X)(1,1+Y)=(1,1+X+Y),
\qquad
\lambda^i(1,1+X)=0\quad(i\geq 2).
\end{equation}
These are the relations used to define the Chern ring in Grothendieck's treatment. Conversely, they determine \eqref{eq:6.1.1} and \eqref{eq:6.1.4}.

This construction is motivated as follows. Suppose that $A$ is the associated graded ring of the $K^{\bullet}$ of a commutatively locally ringed topos, for the filtration of 3.10, and that this $K^{\bullet}$ is a $\lambda$-ring. We shall define Chern classes of elements of an augmented $\lambda$-ring with values in its associated graded ring. To a locally free sheaf of finite type one then associates an element of $\Ch(A)$: its scalar component is the rank, and its $1+\widehat A^+$ component is the sum of its Chern classes. The formulas \eqref{eq:6.2.1} then become the usual formulas for tensor products of invertible sheaves and exterior powers of an invertible sheaf.

\subsection*{6.3. The Chern character}
For every graded $K$-algebra $A$, denote by $A_{\QQ}$ the graded $K$-algebra $A\otimes_{\ZZ}\QQ$. Define a functorial homomorphism
\[
1+\widehat A^+\xrightarrow{\ch}\widehat A_{\QQ}^+
\]
by its value on $1+X$:
\begin{equation}\label{eq:6.3.1}
\ch(1+X)=\exp(X)-1.
\end{equation}
This homomorphism is compatible with the multiplication of 6.1. Indeed, it is enough to check this on $1+X,1+Y$, and one computes
\begin{align*}
\ch((1+X)*(1+Y))
&=\ch\left(\frac{1+X+Y}{(1+X)(1+Y)}\right)\\
&=\ch(1+X+Y)-\ch(1+X)-\ch(1+Y)\\
&=\exp(X+Y)-\exp(X)-\exp(Y)+1\\
&=\ch(1+X)\ch(1+Y).
\end{align*}

Let $\eta$ be the functorial endomorphism of $\widehat A_{\QQ}^+$ defined by
\[
\eta\left(\sum_{k\geq 1}a_kt^k\right)=\sum_{k\geq 1}\frac{(-1)^{k-1}}{(k-1)!}a_kt^k.
\]
Then, for $\varphi\in 1+\widehat A^+$,
\begin{equation}\label{eq:6.3.2}
\ch(\varphi)=\eta(\Log\varphi).
\end{equation}
It follows that $\ch$ commutes with the operations of $K$, since this may again be checked on $1+X$:
\[
\ch((1+X)^a)=\eta(a\Log(1+X))=a\eta(\Log(1+X))=a\ch(1+X).
\]
Thus $\ch$ extends to a homomorphism of $K$-algebras
\begin{equation}\label{eq:6.3.3}
\ch:K\times(1+\widehat A^+)\longrightarrow K\oplus\widehat A_{\QQ}^+,
\qquad
\ch(a,\varphi)=a+\eta(\Log\varphi).
\end{equation}
This homomorphism is called the \emph{Chern character}. If $A$ has characteristic zero, the Chern character is an isomorphism, with inverse
\[
a+\psi_1\longmapsto (a,\exp(\eta^{-1}\psi_1)).
\]

\begin{proposition}\label{prop:6.5}
Let $A$ be a graded $K$-algebra. Give $\Ch(A)$ the degree filtration
\begin{equation}\label{eq:6.5.1}
(\Ch A)_0=\Ch A,
\qquad
(\Ch A)_n=0\times(1+\widehat A^{\geq n})\quad(n\geq 1),
\end{equation}
where $\widehat A^{\geq n}=\prod_{i\geq n}A^i$. Then:
\begin{enumerate}[label=(\roman*)]
\item If $\varphi\in(\Ch A)_n$, $n\geq 1$, and $\varphi=1+a_n+\cdots$, then
\begin{equation}\label{eq:6.5.2}
\ch(\varphi)=\frac{(-1)^{n-1}}{(n-1)!}a_n+\cdots.
\end{equation}
\item If $\varphi=1+a_m+\cdots\in(\Ch A)_m$ and $\psi=1+b_n+\cdots\in(\Ch A)_n$, then
\begin{equation}\label{eq:6.5.3}
\varphi*\psi=1-\frac{(m+n-1)!}{(m-1)!(n-1)!}a_mb_n+\cdots.
\end{equation}
In particular the degree filtration is a ring filtration.
\end{enumerate}
\end{proposition}

The first congruence follows immediately from \eqref{eq:6.3.2}. For the second, by substitution one reduces to a torsion-free universal polynomial ring, embeds it in its tensor product with $\QQ$, and uses the injectivity of $\ch$. If $\varphi*\psi=1+Z_p+\cdots$ is the first non-zero term, multiplicativity of $\ch$ and \eqref{eq:6.5.2} identify $Z_p$.

\subsection*{6.6. Comparison of filtrations}
The $\lambda$-ring $\Ch A$ is augmented over $K$ by $(a,\varphi)\mapsto a$. Hence it has its canonical filtration as an augmented $\lambda$-ring. We compare it with the degree filtration.

\begin{proposition}\label{prop:6.6.1}
Let $x\in\Ch A$ have the form $x=(n,1+\cdots+a_n)$. Then:
\begin{enumerate}[label=(\roman*)]
\item $\lambda^i(x)=0$ for $i>n$;
\item $\lambda^n(x)=(1,1+a_1)$;
\item $\sum_{i=0}^{n}(-1)^i\lambda^i(x)=(0,1-(n-1)!a_n+\cdots)$, where the omitted terms have degree $>n$.
\end{enumerate}
\end{proposition}

It is enough, by functoriality, to prove this for $A=K[X_1,\ldots,X_n]$ filtered by weight, with $X_i$ of weight $i$, and $x=(n,1+X_1+\cdots+X_n)$. Embedding by symmetric functions into $K[T_1,\ldots,T_n]$ gives
\[
(n,1+X_1+\cdots+X_n)=\sum_{i=1}^{n}(1,1+T_i).
\]
Then \eqref{eq:6.2.1} gives the assertions on $\lambda^k(x)$, and the product formula \eqref{eq:6.5.3} gives the final congruence for $\lambda_{-1}(x)$.

\begin{corollary}\label{cor:6.6.2}
The relation 6.6.1(iii) holds for every element $x$ of augmentation $n$.
\end{corollary}

For this one uses that each $(\Ch A)_n$ is a $\lambda$-ideal. If $x=(0,\varphi)\in(\Ch A)_n$, then the polynomials defining the coefficients of $\lambda^i(x)$ depend only on the first coefficients of $\varphi$; hence all coefficients of degree $<n$ vanish. Thus $\lambda^i(x)\in(\Ch A)_n$.

Consequently, for $x=(n,1+a_1+\cdots+a_n+\cdots)$,
\[
\lambda_{-1}(x)=(0,1-(n-1)!a_n+\cdots).
\]
Equivalently, if $\varepsilon$ is the augmentation of $\Ch A$, then
\begin{equation}\label{eq:6.6.4}
\lambda^n(x-\varepsilon(x))=(0,1+(-1)^{n-1}(n-1)!a_n+\cdots).
\end{equation}
It follows that the canonical $\lambda$-ring filtration satisfies
\begin{equation}\label{eq:6.6.5}
\Fil^n\Ch A\subset(\Ch A)_n.
\end{equation}
If $A$ is a finite graded $K$-algebra of characteristic zero, the two filtrations are in fact identical. Indeed, if $A^{\geq p}=0$ and $x=(0,1+a_n+\cdots+a_p)\in(\Ch A)_n$, put
\[
x'=\left(0,1+\frac{(-1)^{n-1}}{(n-1)!}a_n+a_{n+1}+\cdots+a_p\right).
\]
Then $\lambda^n(x')=(0,1+a_n+\cdots)$, and one proceeds by induction on $n$.

\subsection*{6.7. Chern classes in an augmented $\lambda$-ring}
Let $A$ be an augmented $\lambda$-ring over $K$, with augmentation $\varepsilon$. Since $A$ has its canonical filtration, one may form the Chern ring of $\Gr A$. For $x\in A$, the element $\gamma_t(x-\varepsilon(x))$ lies in $\Go(A)$; its image in
\[
\Go(A)/\Go'(A)=1+\widehat{\Gr A}^+
\]
is denoted $c(x)$ and is called the \emph{total Chern class} of $x$. Its $i$-th coefficient is denoted $c^i(x)$:
\begin{equation}\label{eq:6.7.1}
c^i(x)=\cl_{\Gr^iA}\bigl(\lambda^i(x-\varepsilon(x))\bigr),
\qquad
c(x)=1+c^1(x)+\cdots.
\end{equation}
Put
\begin{equation}\label{eq:6.7.2}
\widetilde c(x)=(\varepsilon(x),c(x)),
\end{equation}
called the \emph{completed Chern class} of $x$.

\begin{proposition}\label{prop:6.8}
The completed Chern class
\[
\widetilde c:A\longrightarrow \Ch(\Gr A)
\]
is a homomorphism of augmented $K$-$\lambda$-algebras.
\end{proposition}

Additivity is clear. For multiplicativity, write
\begin{align*}
\gamma_t(xy-\varepsilon(xy))&=\gamma_t(\varepsilon(y)(x-\varepsilon(x)))\gamma_t(\varepsilon(x)(y-\varepsilon(y)))\\
&\qquad\cdot\gamma_t((x-\varepsilon(x))(y-\varepsilon(y)))\\
&=(\gamma_t(x-\varepsilon(x)))^{\varepsilon(y)}(\gamma_t(y-\varepsilon(y)))^{\varepsilon(x)}\\
&\qquad\cdot\bigl(\gamma_t(x-\varepsilon(x))\overline{\circ}\gamma_t(y-\varepsilon(y))\bigr),
\end{align*}
and pass to the associated graded, using \eqref{eq:6.1.2}. Compatibility with scalars and augmentations is immediate. Compatibility with the $\lambda$-operations follows by checking the diagram
\[
\begin{array}{ccc}
A&\xrightarrow{\gamma_t}&\Go_g(A)\\
\downarrow\widetilde c&&\downarrow\Go_g(\widetilde c)\\
\Ch(\Gr A)&\xrightarrow{\gamma_t}&\Go_g(\Ch(\Gr A))
\end{array}
\]
on $K$ and on elements of augmentation zero.

\begin{proposition}\label{prop:6.9}
Let $A$ be an augmented $K$-$\lambda$-algebra and let $z\in\Fil^mA$. Then
\begin{equation}\label{eq:6.9.1}
\gamma^m(z)\equiv(-1)^{m-1}(m-1)!\,z\pmod{\Fil^{m+1}A}.
\end{equation}
Equivalently,
\[
c^m(z)=\cl_{\Gr^mA}\bigl((-1)^{m-1}(m-1)!\,z\bigr).
\]
\end{proposition}

One writes $z$ as a sum of products $a\gamma^{i_1}(x_1)\cdots\gamma^{i_k}(x_k)$ with $\varepsilon(x_i)=0$ and $i_1+\cdots+i_k\geq m$. Since $\widetilde c$ is a $\lambda$-homomorphism and \eqref{eq:6.6.4} controls the first non-zero term of the Chern class, formula \eqref{eq:6.5.3} gives the displayed congruence.

\begin{proposition}\label{prop:6.10}
Let $K$ be a binomial ring, $R$ an augmented $K$-$\lambda$-algebra, $N\in R$ such that $\lambda^k(N)=0$ for $k>d$, and $x\in\Fil^iR$. Then
\begin{equation}\label{eq:6.10}
\gamma^{d+i}(N,x)-(-1)^{d+i-1}(d+i-1)!\,x\in\Fil^{i+1}R.
\end{equation}
\end{proposition}

Choose a presentation of $x$ by products of $\gamma$-operations, and pass to the universal $K$-$\lambda$-algebra $R_0$ generated by $N'_0$ and the auxiliary variables, with $\gamma^k(N'_0)=0$ for $k>d$ and zero augmentation. The filtration of $R_0$ is the weight filtration. After replacing $N'_0$ by $N_0+d$, the desired assertion becomes
\[
\gamma^{d+i}(N_0,x_0)\gamma^d(N'_0)-(-1)^{d+i-1}(d+i-1)!x_0\gamma^d(N'_0)\in\Fil^{d+i+1}R_0.
\]
Since $\lambda_{-1}(N_0)=(-1)^d\gamma^d(N'_0)$, this is exactly Proposition 6.9 applied to $z=(-1)^dx_0\gamma^d(N'_0)$. Functoriality then gives the general case.

\begin{proposition}\label{prop:6.11}
Assume that $K$ has characteristic zero. Let $n\geq 0$. Then the functor $\Ch$ is an equivalence from the category of graded $K$-algebras $A$ with $A_0=K$ and $A_k=0$ for $k>n$ to the category of augmented $K$-$\lambda$-algebras $\Lambda$ with $\Fil^{n+1}\Lambda=0$. A quasi-inverse is $\Lambda\mapsto\Gr\Lambda$.
\end{proposition}

If $A$ is finite graded, the two filtrations on $\Ch A$ coincide. Define
\[
u:A\longrightarrow\Gr_{\deg}(\Ch A)
\]
by the identity on $A_0$ and, for $i>0$, by
\[
u(a_i)=\cl_{\Gr^i}(0,1+(-1)^{i-1}(i-1)!a_i).
\]
Formula \eqref{eq:6.5.3} shows that $u$ is a homomorphism of graded rings, compatible with the $K$-operations. Its inverse sends a class in $\Gr^i(\Ch A)$ to $(-1)^{i-1}a_i/(i-1)!$, where $a_i$ is the $i$-th coefficient of any lift.

For an augmented $\lambda$-algebra $\Lambda$, the isomorphism
\[
\Lambda\longrightarrow\Ch(\Gr\Lambda)
\]
is given by the completed Chern class $\widetilde c$. It is enough to check this on associated graded rings, and then on generators of the canonical filtration, i.e. the classes $\gamma^i(x)$ with $\varepsilon(x)=0$. Formula \eqref{eq:6.6.4} shows that the resulting composite is the identity.

\begin{corollary}\label{cor:6.12}
If $K$ has characteristic zero and $\Lambda$ is a $K$-$\lambda$-algebra with discrete filtration, then
\[
\widetilde c:\Lambda\longrightarrow\Ch(\Gr\Lambda)
\]
is an isomorphism.
\end{corollary}

\begin{definition}\label{def:6.13}
Let $A$ be a graded $K$-algebra and $\Lambda$ an augmented $K$-$\lambda$-algebra. A \emph{Chern theory for $\Lambda$ with values in $A$} is a $\lambda$-homomorphism
\[
\Lambda\longrightarrow\Ch(A)
\]
compatible with augmentations.
\end{definition}

The functor sending $A$ to the set of such Chern theories is representable. When $K$ has characteristic zero and $\Lambda$ has finite filtration, the representing graded algebra is $\Gr\Lambda$ by Proposition 6.11.

\section*{7. Appendix: Adams operations $\psi^k$}
\addcontentsline{toc}{section}{7. Appendix: Adams operations}

We follow here a presentation due to J.-P. Serre.

\subsection*{7.1}
Let $K$ be a pre-$\lambda$-ring. For $x\in K$ define elements $\psi^k(x)$ by
\begin{equation}\label{eq:7.1.1}
\frac{d}{dt}\lambda_t(x)\big/\lambda_t(x)=\sum_{k=1}^{\infty}(-1)^{k-1}\psi^k(x)t^{k-1}.
\end{equation}
The maps $\psi^k:K\to K$ are additive, since logarithmic derivation turns products into sums. They are isobaric polynomials of weight $k$ in $\lambda^1,\ldots,\lambda^k$; for example
\[
\psi^1(x)=x,
\qquad
\psi^2(x)=x^2-2\lambda^2(x).
\]

\subsection*{7.2--7.3}
Conversely, if $K$ is a ring of characteristic zero and additive maps $\psi^k:K\to K$ are given with $\psi^1=\id_K$, then
\begin{equation}\label{eq:7.2.1}
\lambda_t(x)=\exp\left(\sum_{k=1}^{\infty}(-1)^{k-1}\psi^k(x)t^k/k\right)
\end{equation}
defines a pre-$\lambda$-structure on $K$. Thus in characteristic zero giving a pre-$\lambda$-structure is equivalent to giving the corresponding operations $\psi^k$. It is also clear from \eqref{eq:7.1.1} that the $\psi^k$ commute with $\lambda$-homomorphisms; in characteristic zero a homomorphism of pre-$\lambda$-rings is a $\lambda$-homomorphism if and only if it commutes with all $\psi^k$.

\subsection*{7.4}
For $K=\ZZ$ with its $\lambda$-ring structure, all $\psi^k$ are the identity. Indeed
\[
\frac{d}{dt}\lambda_t(n)\big/\lambda_t(n)=n(1+t)^{n-1}/(1+t)^n=n\sum_{k\geq 1}(-1)^{k-1}t^{k-1}.
\]
On the category $\mathcal G_0$ of $\ZZ$-algebras, the operations $\psi^k$ define additive endomorphisms of the $\lambda$-ring functor $\Go$. They are determined by
\begin{equation}\label{eq:7.4.1}
\psi^k(1+Xt)=1+X^kt.
\end{equation}

\begin{proposition}\label{prop:7.5}
\begin{enumerate}[label=(\roman*)]
\item If $K$ is a $\lambda$-ring, the operations $\psi^k$ satisfy
\begin{equation}\label{eq:7.5.1}
\psi^k(1)=1,
\qquad
\psi^k(xy)=\psi^k(x)\psi^k(y),
\qquad
\psi^k\circ\psi^{k'}=\psi^{kk'}.
\end{equation}
\item Conversely, if the $\psi^k$ satisfy these identities and $K$ has characteristic zero, then $K$ is a $\lambda$-ring.
\end{enumerate}
\end{proposition}

For the first assertion, $\lambda_t(1)=1+t$ gives $\psi^k(1)=1$. To prove multiplicativity, consider the diagram expressing the product $K\times K\to K$ and the product $\circ$ on $\Go(K)$; after applying logarithmic derivative and extracting the $k-1$ coefficient, it is enough to check the universal pair $(1+Xt,1+Yt)$. Then
\[
\psi^k((1+Xt)\circ(1+Yt))=\psi^k(1+XYt)=(XY)^k,
\]
which equals $X^kY^k$.

For the composition law it is enough to check the universal element again:
\[
\psi^{k'}_u(\lambda_u(1+Xt))=\psi^{k'}_u(1+(1+Xt)u)=(1+Xt)^{\circ k'}=1+X^{k'}t,
\]
and applying $\psi^k$ gives $X^{kk'}$. If $K$ has characteristic zero, logarithmic derivation is an isomorphism from $\Go(K)$ to $K[[t]]$; hence the relations in \eqref{eq:7.5.1} force $\lambda_t$ to be a ring homomorphism and then a $\lambda$-homomorphism. The relation $\lambda_t(1)=1+t$ follows by integration from $\psi^k(1)=1$.

\section*{1. Computation of $K^*$ of a projective bundle: locally free sheaves of finite type}
\addcontentsline{toc}{section}{Expose VI, 1. Projective bundles: locally free sheaves}

In this section $K^*(X)$ denotes the Grothendieck group of locally free sheaves of finite type on a scheme $X$. The aim is the following theorem.

\begin{theorem}[1.1]\label{thm:VI.1.1}
Let $S$ be a quasi-compact scheme, let $E$ be a locally free $\cO_S$-module of rank $r+1$, and let $X=\mathbb P(E)$ be the associated projective bundle. The structural morphism $f:X\to S$ gives $K^*(X)$ the structure of a $K^*(S)$-algebra. Let $\xi\in K^*(X)$ be the class of $\cO_X(1)$. Then $K^*(X)$ is a free $K^*(S)$-module with basis
\[
1,\xi,\ldots,\xi^r.
\]
\end{theorem}

\subsection*{1.2. Plan of the proof}
The proof has three parts.
\begin{enumerate}[label=$\arabic*^\circ$)]
\item The powers of $\xi$ generate $K^*(X)$ as a $K^*(S)$-module. This is proved by showing:
\begin{enumerate}[label=(\alph*)]
\item If $\mathcal F$ is locally free of finite type on $X$, there is a graded $\Sym(E)$-module $\mathcal M$ of finite presentation, flat over $\cO_S$, such that $\mathcal F=\Proj_0(\mathcal M)$.
\item Such an $\mathcal M$ has a finite graded resolution by graded $\Sym(E)$-modules $\mathcal L_i$ locally free of finite type:
\[
0\to \mathcal L_{r+1}\to\cdots\to \mathcal L_0\to\mathcal M\to 0.
\]
\item Each $\mathcal L_i$ has a finite filtration whose quotients are of the form $\mathcal N_j\otimes_{\cO_S}\Sym(E)(j)$ with $\mathcal N_j$ locally free of finite type on $S$.
\end{enumerate}
\item The elements $1,\xi,\ldots,\xi^{r+1}$ are linearly dependent over $K^*(S)$.
\item The elements $1,\xi,\ldots,\xi^r$ are linearly independent over $K^*(S)$.
\end{enumerate}

\subsection*{1.3. Graded modules}
Let $S$ be a scheme, $\mathcal B$ a graded $\cO_S$-algebra, and $\mathcal M$ a graded $\mathcal B$-module. For a graded $\mathcal B$-module $\mathcal N$, write $\mathcal N(n)$ for the graded module with
\[
(\mathcal N(n))_k=\mathcal N_{n+k}.
\]
A graded module is locally free, respectively locally free of finite type, if locally on $S$ it is a direct sum, respectively a finite direct sum, of modules $\mathcal B(n)$. It is of finite presentation if locally it has a presentation by finite free graded modules.

For every $n$ there is a canonical homomorphism of graded $\mathcal B$-modules
\begin{equation}\label{eq:VI.1.2.1}
\mathcal M_n\otimes_{\cO_S}\mathcal B(-n)\longrightarrow\mathcal M,
\end{equation}
sending $x\otimes a$ to $a\cdot x$. A graded module free of finite type decomposes as
\begin{equation}\label{eq:VI.1.2.2}
\mathcal M\simeq\bigoplus_{n=n_0}^{n_1}(\mathcal B(-n))^{k_n}.
\end{equation}
The integers $k_n$ in such a decomposition are independent of the chosen decomposition. This is proved by induction, comparing the lowest non-zero homogeneous components and then passing to the quotient.

\begin{proposition}\label{prop:VI.1.3}
Let $S$ be quasi-compact, $E$ a locally free $\cO_S$-module of finite type, $X=\mathbb P(E)$, and $\mathcal F$ an $\cO_X$-module of finite presentation, flat over $S$. There exists an integer $n_0$ such that for all $n\geq n_0$:
\begin{enumerate}[label=(\roman*)]
\item $R^if_*(\mathcal F(n))=0$ for $i\geq 1$;
\item $f_*(\mathcal F(n))$ is locally free of finite type on $S$;
\item $\mathcal M=\bigoplus_{n\geq n_0}f_*(\mathcal F(n))$ is a graded $\Sym(E)$-module of finite presentation.
\end{enumerate}
\end{proposition}

The assertion is local on $S$, hence the quasi-compactness allows reduction to $S$ affine. In the noetherian case, (i) follows from Serre's theorem, (ii) from EGA III, 7.9.10, and (iii) from the comparison between $\mathcal F$ and a quasi-coherent graded module defining it. For the general affine case, write $S$ as a filtered projective limit of noetherian affine schemes and descend $E$ and $\mathcal F$ to some stage. The necessary flatness and finite-presentation hypotheses descend by EGA IV, and the noetherian case applies after base change.

\begin{corollary}\label{cor:VI.1.4}
Under the hypotheses of Proposition VI.1.3, there is a graded $\Sym(E)$-module $\mathcal M$ of finite presentation, flat over $\cO_S$, such that $\mathcal F=\Proj_0(\mathcal M)$.
\end{corollary}

\begin{lemma}\label{lem:VI.1.5}
Let $A$ be a ring, $B$ an $A$-algebra of finite presentation, and $M$ a $B$-module of finite presentation. Assume that $B$ and $M$ are flat over $A$, and that for every residue field $k$ of $A$, the module $M\otimes_Ak$ has projective dimension $\leq n$ over $B\otimes_Ak$. Then $M$ has a resolution of length $n$ by projective $B$-modules of finite type:
\[
0\to L_n\to\cdots\to L_0\to M\to 0.
\]
\end{lemma}
This follows immediately from EGA IV, 12.3.2.

\begin{proposition}\label{prop:VI.1.6}
Let $S$ be quasi-compact, $E$ locally free of rank $r+1$, $\mathcal B=\Sym(E)$ with its usual grading, and $\mathcal M$ a graded $\mathcal B$-module of finite presentation, flat over $S$. Then there is a graded resolution
\[
0\to\mathcal L_{r+1}\to\cdots\to\mathcal L_0\to\mathcal M\to 0,
\]
with the $\mathcal L_i$ graded $\mathcal B$-modules locally free of finite type.
\end{proposition}

First, any such $\mathcal M$ admits a graded surjection $\mathcal L\to\mathcal M$ with $\mathcal L$ locally free of finite type: choose $N$ such that $\mathcal M$ is generated in degrees $\leq N$ and put
\[
\mathcal L=\bigoplus_{i\leq N}\mathcal M_i\otimes_{\cO_S}\mathcal B(-i).
\]
The map is the one of \eqref{eq:VI.1.2.1}. Iterating this construction on kernels gives a global graded resolution. After $r+1$ steps, the kernel is locally projective over $\mathcal B$ on affine opens by Lemma VI.1.5, hence locally free as a graded $\mathcal B$-module; truncating gives the asserted resolution.

\begin{proposition}\label{prop:VI.1.7}
Let $S$ be quasi-compact, $E$ locally free of finite type, $\mathcal B=\Sym(E)$, and $\mathcal L$ a graded $\mathcal B$-module locally free of finite type. Then $\mathcal L$ has a decreasing finite filtration by graded submodules $\mathcal L^{(m)}$ such that
\begin{enumerate}[label=(\roman*)]
\item $\mathcal L^{(m)}=\mathcal L$ for $m\leq m_0$;
\item $\mathcal L^{(m)}=0$ for $m\geq m_1$;
\item $\gr^m\mathcal L$ is isomorphic, as a graded $\mathcal B$-module, to $\mathcal N_m\otimes_{\cO_S}\mathcal B(m)$ with $\mathcal N_m$ locally free of finite type on $S$.
\end{enumerate}
\end{proposition}

Since $S$ is quasi-compact, the degrees of local homogeneous bases lie between fixed integers $n_0$ and $n_1$. Induction on $n_1-n_0$ proves the assertion. The first step is the canonical map $\mathcal M_{n_0}\otimes\mathcal B(-n_0)\to\mathcal M$, which locally identifies the summand of lowest homogeneous degree; its cokernel has local generators in higher degrees, and induction applies.

\subsection*{1.8--1.9. Generation and the relation}
Let $\mathcal F$ be locally free of finite type on $X$. By Corollary VI.1.4 and Proposition VI.1.6, resolve the corresponding graded module by $\mathcal L_i$'s. Applying $\Proj_0$ gives an exact sequence and hence in $K^*(X)$
\[
\cl(\mathcal F)=\sum_i(-1)^i\cl(\Proj_0\mathcal L_i).
\]
Using the filtration of Proposition VI.1.7, each term is a sum of classes
\[
\cl(f^*\mathcal N_{ij}(j))=f^*(\cl\mathcal N_{ij})\xi^j.
\]
Thus $K^*(X)$ is generated as a $K^*(S)$-module by the powers of $\xi$.

To get a dependence relation, use the canonical surjection $f^*E\to\cO_X(1)$ with locally free kernel $\mathcal F$ of rank $r$. Since $\lambda^{r+1}(f^*E-\xi)=0$, and $\lambda_t(-\xi)=1/(1+\xi t)$, one obtains
\begin{equation}\label{eq:VI.1.9.1}
\sum_{i=0}^{r+1}(-1)^{r+1-i}\xi^{r+1-i}\lambda^i(E)=0.
\end{equation}

\subsection*{1.10--1.11. Independence}
The proof of independence is due to J.-P. Jouanolou.

\begin{lemma}\label{lem:VI.1.10}
With the hypotheses of Theorem VI.1.1, let $\mathcal F$ and $\mathcal G$ be locally free sheaves of finite type on $X$ with equal classes in $K^*(X)$. Then there is $n_0$ such that for all $n\geq n_0$:
\begin{enumerate}[label=(\roman*)]
\item $f_*(\mathcal F(n))$ and $f_*(\mathcal G(n))$ are locally free of finite type on $S$;
\item their classes in $K^*(S)$ are equal.
\end{enumerate}
\end{lemma}

This follows by writing the equality of classes in $K^*(X)$ as a finite integral combination of relations coming from short exact sequences, twisting by $\cO_X(n)$, and applying $f_*$. For $n$ sufficiently large all higher direct images vanish and all direct images are locally free, so the relations descend to $K^*(S)$.

Suppose
\begin{equation}\label{eq:VI.1.11.1}
f^*(x_0)+f^*(x_1)\xi+\cdots+f^*(x_r)\xi^r=0.
\end{equation}
Write each $x_j$ as a difference of classes of locally free sheaves, and construct locally free sheaves $\mathcal F,\mathcal F'$ on $X$ such that \eqref{eq:VI.1.11.1} says $\cl(\mathcal F)=\cl(\mathcal F')$. By the lemma, for large $n$,
\begin{equation}\label{eq:VI.1.11.2}
\sum_{j=0}^{r}x_j\sigma_{n+j}=0,
\end{equation}
where $\sigma_p$ is the class of $\Sym^p(E)$ in $K^*(S)$.

The Koszul complex
\[
0\to \lambda^{r+1}(E)\otimes\Sym(E)\to\lambda^r(E)\otimes\Sym(E)\to\cdots\to E\otimes\Sym(E)\to\Sym(E)\to 0
\]
is a graded resolution of $\cO_S$. Its homogeneous components give
\begin{equation}\label{eq:VI.1.11.3}
\sum_{i=0}^{r+1}(-1)^i\lambda^i(E)\sigma_{p-i}=0\quad(p\geq 1),
\end{equation}
with $\sigma_k=0$ for $k<0$. From \eqref{eq:VI.1.11.2} for all large $n$, relation \eqref{eq:VI.1.11.3} allows descending induction on $n$, and gives \eqref{eq:VI.1.11.2} for all $n\geq -r$. Taking $n=-r$ gives $x_r=0$, and continuing downward gives $x_j=0$ for all $j$. This proves independence and Theorem VI.1.1.

\begin{remark}
Let $\sigma_i(E)$ denote the class of $\Sym^i(E)$. If $\sigma_t(E)=1+\sum_{i\geq 1}\sigma_i(E)t^i$, relation \eqref{eq:VI.1.11.3} is equivalent to
\begin{equation}\label{eq:VI.1.12.1}
\lambda_t(E)\sigma_{-t}(E)=1.
\end{equation}
This shows the additivity of $\sigma_t$ and extends it to a homomorphism $K^*(S)\to 1+K^*(S)[[t]]^+$.
\end{remark}

\begin{remark}
Any sequence of $r+1$ consecutive powers of $\xi$ is a basis of $K^*(X)$ over $K^*(S)$. Put $x=1-\xi^{-1}$, the class of a hyperplane section. Then $1,x,\ldots,x^r$ is also a basis, and the dependence relation is
\begin{equation}\label{eq:VI.1.13.1}
\sum_{k=0}^{r+1}\lambda^{r+1-k}(\check E-r-1)x^k=0.
\end{equation}
For $E=\cO_S^{r+1}$, so $X=\mathbb P^r_S$, this reduces to
\begin{equation}\label{eq:VI.1.13.2}
x^{r+1}=0.
\end{equation}
\end{remark}

\begin{remark}
Using relative schemes in the sense of Hakim, Theorem VI.1.1 also makes sense when $S$ is a locally ringed topos. If the final object of the topos is quasi-compact, the proof transports without difficulty.
\end{remark}

\section*{2. Computation of $K^*$ of a projective bundle: perfect complexes}
\addcontentsline{toc}{section}{Expose VI, 2. Projective bundles: perfect complexes}

This section will not be used later in the seminar. Here $K^*(X)$ denotes the Grothendieck group of perfect complexes on $X$.

\begin{theorem}[2.1]\label{thm:VI.2.1}
Let $S$ be quasi-compact, let $E$ be a locally free $\cO_S$-module of rank $r+1$, and let $X=\mathbb P(E)$. Let $\xi$ be the class in $K^*(X)$ of the complex consisting of $\cO_X(1)$ in degree zero. Then $K^*(X)$ is a free $K^*(S)$-module with basis
\[
1,\xi,\ldots,\xi^r.
\]
\end{theorem}

Let $\mathcal B=\Sym(E)$. Let $\mathcal A$ be the category of $\cO_X$-modules, $\mathcal G$ the category of graded $\mathcal B$-modules, $\mathcal G_0$ its full subcategory of graded $\mathcal B$-modules locally free of finite type, and $\mathcal G'$ the category of $\cO_S$-modules. For a complex $L^*$ on $X$, write $L^*(n)=L^*\otimes\cO_X(n)$.

\subsection*{2.3. The functor $\underline{\Proj}_0$}
We need to apply $\Proj_0$ to arbitrary graded $\mathcal B$-modules, not only quasi-coherent ones. If $Y=\Spec(\mathcal A)$ and $f:Y\to S$, the isomorphism $\mathcal A\simeq f_*\cO_Y$ gives a morphism of ringed spaces
\[
\widetilde f:(Y,\cO_Y)\longrightarrow(S,\mathcal A)
\]
and an exact inverse-image functor
\[
\widetilde f^*(\mathcal M)=f^{-1}\mathcal M\otimes_{f^{-1}\mathcal A}\cO_Y.
\]
For a quasi-coherent $\mathcal A$-module this is the usual associated module.

Now let $\mathcal B$ be graded and quasi-coherent, $\mathcal M$ a graded $\mathcal B$-module, and $X=\Proj(\mathcal B)$. For an affine open $U\subset S$ and a homogeneous section $s$ of positive degree, put
\[
\mathcal M_{(s),U}=((\mathcal M|_U)\otimes_{\mathcal B|_U}(\mathcal B|_U)_s)_0.
\]
On $D_+(U,s)$ define
\[
\underline{\Proj}_0(\mathcal M)_{U,s}=\widetilde f_{U,s}^{*}(\mathcal M_{(s),U}).
\]
The usual localization isomorphisms
\[
\mathcal M_{(st),U}\simeq (\mathcal M_{(s),U})_{t^d/s^e}
\]
show that these modules glue. This gives an exact functor $\underline{\Proj}_0$ on graded modules and hence a functor
\[
\underline{\Proj}_0:D^+(\mathcal G)\longrightarrow D^+(\mathcal A),
\]
compatible with tensor products.

\subsection*{2.4--2.6. Acyclic approximation}
For an $\cO_X$-module $\mathcal L$ and an integer $k$, set
\begin{equation}\label{eq:VI.2.4.1}
F_k(\mathcal L)=\bigoplus_{n\geq k}f_*(\mathcal L(n)).
\end{equation}
The functor $F_k$ is left exact, hence has a derived functor
\begin{equation}\label{eq:VI.2.4.2}
\RR F_k:D^+(\mathcal A)\longrightarrow D^+(\mathcal G).
\end{equation}
If $\mathcal L$ is locally free of finite type, then for $k$ sufficiently large it is $F_k$-acyclic, by the vanishing theorem used in 1.3.

There is a functorial morphism
\begin{equation}\label{eq:VI.2.5.1}
\beta:\underline{\Proj}_0(F_k(\mathcal F))\longrightarrow\mathcal F
\end{equation}
constructed on each standard open $D_+(U,s)$ by sending a fraction $z/s^n$ to $z\alpha_d(s^n)^{-1}$. It agrees with the usual EGA morphism when $\mathcal F$ is quasi-coherent. Passing to derived categories gives
\begin{equation}\label{eq:VI.2.5.3}
\underline{\Proj}_0\circ\RR F_k\longrightarrow \id_{D^+(\mathcal A)}.
\end{equation}

\begin{proposition}\label{prop:VI.2.6}
Let $S$ be quasi-compact, $E$ locally free of finite type, $X=\mathbb P(E)$, and $L^*$ a perfect complex on $X$. For all sufficiently large $k$:
\begin{enumerate}[label=(\roman*)]
\item $\RR F_k(L^*)$ is $\mathcal G_0$-perfect;
\item the canonical morphism
\[
\underline{\Proj}_0\RR F_k(L^*)\longrightarrow L^*
\]
is an isomorphism in $D^+(\mathcal A)$.
\end{enumerate}
\end{proposition}

The assertion is local on $S$, so assume $S$ affine. Since $\cO_X(1)$ is ample, one may replace $L^*$ by a strictly perfect complex. For large $k$ all its finitely many terms are $F_k$-acyclic and the modules $F_k(L^i)$ are $\mathcal G_0$-perfect, giving (i); the morphism is then an isomorphism term by term, giving (ii).

\subsection*{2.7--2.8. Decomposition and proof}
For each $n$, the functor taking the $n$-th homogeneous component gives
\[
\Comp_n:D^+(\mathcal G)\longrightarrow D^+(\mathcal G').
\]
The canonical map \eqref{eq:VI.1.2.1} gives a morphism
\begin{equation}\label{eq:VI.2.7.2}
\Comp_n\otimes_{\cO_S}\mathcal B(-n)\longrightarrow\id_{D^+(\mathcal G)}.
\end{equation}
If $M^*$ is $\mathcal G_0$-perfect, one constructs a finite sequence of distinguished triangles
\[
N_i^*\to P_{i-1}^*\to P_i^*\to N_i^*[1]
\]
with $P_0^*=M^*$, final $P_k^*$ acyclic, and
\[
N_i^*=K_i^*\otimes_{\cO_S}\mathcal B(d_i)
\]
for perfect complexes $K_i^*$ on $S$. The construction starts with the smallest degree $d_0$ appearing locally in a bounded locally free representative, uses \eqref{eq:VI.2.7.2}, and iterates. Finiteness follows from the finite filtration of Proposition VI.1.7.

Now let $L^*$ be perfect on $X$. For large $k$, put $M^*=\RR F_k(L^*)$. Applying the preceding decomposition and then $\underline{\Proj}_0$, one gets in $K^*(X)$
\[
\cl(L^*)=\sum_i\cl(f^*K_i^*)\xi^{d_i}=\sum_if^*(\cl K_i^*)\xi^{d_i}.
\]
Thus the powers of $\xi$ generate $K^*(X)$ over $K^*(S)$. The dependence relation follows from \eqref{eq:VI.1.9.1}, viewed in the perfect-complex $K$-group. For independence one uses the $K^*(S)$-linear map $f_*:K^*(X)\to K^*(S)$. Since
\[
f_*(1)=1,
\qquad
f_*(\xi^n)=0\quad(-r\leq n<0),
\]
multiplying a relation by appropriate powers of $\xi^{-1}$ and applying $f_*$ successively gives all coefficients zero. This proves Theorem VI.2.1.

\section*{3. Consequences of the structure theorem}
\addcontentsline{toc}{section}{Expose VI, 3. Consequences of the structure theorem}

\begin{proposition}\label{prop:VI.3.1}
Let $S$ be quasi-compact, let $E_1,\ldots,E_n$ be locally free $\cO_S$-modules of finite type, and let $X$ be the fiber product over $S$ of the corresponding flag schemes. Then the canonical homomorphism
\[
K^*(S)\longrightarrow K^*(X)
\]
is injective, whether $K^*$ is defined using locally free sheaves of finite type or perfect complexes.
\end{proposition}

By constructing flag schemes by successive projective bundles, this reduces to the projective-bundle case, where it follows immediately from Theorem VI.1.1 or VI.2.1.

\begin{theorem}\label{thm:VI.3.2}
For every quasi-compact scheme $S$, the pre-$\lambda$-ring $K^*(S)$ defined from locally free sheaves of finite type is a $\lambda$-ring.
\end{theorem}

It is enough to verify the universal $\lambda$-relations on classes of locally free sheaves. After decomposing $S$ into finitely many disjoint opens one may assume constant ranks. Pulling back to $X=\mathbb P(E)\times_S\mathbb P(F)$ is injective on $K^*$, and on $X$ the relevant classes split into sums of invertible sheaves. Thus it suffices to check the relations for classes $\xi,\eta$ of invertible sheaves, where they become
\[
1+\xi\eta t=(1+\xi t)\circ(1+\eta t),
\qquad
\lambda_u(1+\xi t)=1+\{1+\xi t\}u,
\]
which are precisely the defining relations of the universal $\lambda$-ring structure of $1+K^*(S)[[t]]_+$.

More generally, if $(T,\mathcal A)$ is a commutatively ringed topos and $K$ is the Grothendieck group of locally free $\mathcal A$-modules of bounded rank on $T$, then the canonical pre-$\lambda$-structure on $K$ is a $\lambda$-ring. The proof is the same after reducing constant-rank locally free modules to representations of the group schemes $\mathrm{GL}(n)$ and using the fact that the representation ring $R_{\ZZ}(G)$ is a $\lambda$-ring for finite products of linear group schemes.

\section*{4. Computation of $K^{\bullet}$ of a flag bundle}
\addcontentsline{toc}{section}{Expose VI, 4. Flag bundles}

\subsection*{4.1}
Let $S$ be a scheme and let $\mathcal E$ be a locally free $\cO_S$-module of rank $n$. Let $\pi=(p_1,\ldots,p_k)$ be a sequence of integers $\geq 1$ with $\sum_i p_i=n$. The \emph{flag functor of type $\pi$ of $\mathcal E$} assigns to an $S$-scheme $X\to S$ the set of chains of quotients of $f^*\mathcal E$
\[
(\mathcal L_{k-1},\ldots,\mathcal L_1),
\]
where $\mathcal L_i$ is a quotient of $\mathcal L_{i+1}$, locally free of rank $p_1+\cdots+p_i$. This functor is represented by an $S$-scheme $\underline D_\pi(\mathcal E)$ equipped with the universal quotient flag. For $\pi=(1,n-1)$ this is $\mathbb P(\mathcal E)$; for $\pi=(r,n-r)$ it is the Grassmannian; if all $p_i=1$, it is the usual flag bundle $\underline D(\mathcal E)$.

\begin{lemma}[4.2]
Let $\pi=(p_1,\ldots,p_i,\ldots,p_k)$ and split $p_i=p'+p''$. The flag bundle of type
\[
\pi'=(p_1,\ldots,p_{i-1},p',p'',p_{i+1},\ldots,p_k)
\]
is obtained from $\underline D_\pi(\mathcal E)$ by taking the flag bundle of type $(p',p'')$ of the kernel of the relevant universal quotient. This identification is canonical.
\end{lemma}
The proof is immediate from the functorial definition of flag schemes.

\begin{lemma}[4.3]
Let $A$ be a commutative ring, and let $t_1,\ldots,t_m\in A$. Consider the algebra generated by variables $\xi_i^{(j)}$ subject to the elementary symmetric relations
\[
\sigma_r(\xi_1^{(j)},\ldots,\xi_m^{(j)})=e_r(t_1,\ldots,t_m).
\]
For a partition $p_1+\cdots+p_k=n$, the corresponding block monomials
\[
\prod_{j=1}^{k}\prod_{\alpha=1}^{p_j}(\xi_{p_1+\cdots+p_{j-1}+\alpha}^{(j)})^{a_{j,\alpha}},
\qquad
0\leq a_{j,\alpha}\leq n-\alpha,
\]
form a basis over the subalgebra generated by the block elementary symmetric functions, and the relations among those symmetric functions are generated by the isobaric relations from
\[
\prod_{j=1}^{k}\prod_{\alpha=1}^{p_j}(1+\eta_j^{(\alpha)}T)=\prod_{i=1}^{m}(1+t_iT).
\]
\end{lemma}

\begin{corollary}[4.4]
The same block monomials form a basis over $A$.
\end{corollary}
The lemma and corollary are borrowed from Chevalley's seminar.

\subsection*{4.5--4.8}
Let $S$ be quasi-compact, let $\mathcal E$ be locally free of rank $n$, and put $X=\underline D_\pi(\mathcal E)$. Let $\mathcal L_i$ be the universal quotients and $\mathcal F_i=\Ker(\mathcal L_i\to\mathcal L_{i-1})$. Denote by $F_i$ the class of $\mathcal F_i$.

\begin{theorem}[4.6]\label{thm:VI.4.6}
With this notation, the $K^{\bullet}(S)$-algebra $K^{\bullet}(X)$ is generated by the Chern roots $\xi_i^{(\alpha)}$ of the $F_i$. The ideal of relations among them is generated by the relations deduced from
\begin{equation}\label{eq:VI.4.6}
\prod_{i=1}^{k}\lambda_t(F_i)=\lambda_t(E).
\end{equation}
Equivalently, $K^{\bullet}(X)$ is the $K^{\bullet}(S)$-$\lambda$-algebra generated by elements $F_i^{(\alpha)}$ subject to
\begin{enumerate}[label=(\roman*)]
\item $\lambda^r(F_i^{(\alpha)})=0$ for $r>p_i$;
\item $\prod_{i,\alpha}(1+F_i^{(\alpha)}t)=\lambda_t(E)$.
\end{enumerate}
\end{theorem}

First consider the usual complete flag bundle. Induction on the rank, using the projective bundle theorem at each step, gives a $K^{\bullet}(S)$-basis
\[
\xi_1^{a_1}\xi_2^{a_2}\cdots\xi_n^{a_n},
\qquad
0\leq a_i\leq n-i.
\]
The relation $\prod_i(1+\xi_it)=\lambda_t(E)$ holds by additivity. The abstract algebra defined by this relation has the same basis by Lemma 4.3, hence is isomorphic to $K^{\bullet}(X)$.

For a general flag type $\pi$, pass from $\underline D_\pi(\mathcal E)$ to the complete flag bundle by successively splitting the $\mathcal F_i$. The complete flag $K$-ring is a free module both over the subalgebra generated by the $\xi_i^{(\alpha)}$ and over $K^{\bullet}(\underline D_\pi(\mathcal E))$, with the same basis. Therefore these subrings coincide. The relations follow from Lemma 4.3.

\begin{corollary}[4.8]
With the hypotheses above, put
\[
c_i^{(j)}=\gamma^i(F_j-p_j),
\qquad 1\leq j\leq k,
\qquad 1\leq i\leq p_j.
\]
Then $K^{\bullet}(X)$ is generated over $K^{\bullet}(S)$ by the $c_i^{(j)}$, and the ideal of relations is generated by those deduced from
\begin{equation}\label{eq:VI.4.8}
\gamma_t(E-n)=\prod_{j=1}^{k}\gamma_t(F_j-p_j).
\end{equation}
\end{corollary}
This is just Theorem VI.4.6 rewritten in terms of the $\gamma$-operations.

\subsection*{4.9--4.10. Infinite Grassmannians}
Let $\underline G_n^{r,n}$ be the Grassmannian of type $(n,r-n)$ of $\cO_S^r$, with $r\geq n$. As $r$ varies these form an inductive system, denoted $\underline G_n^{\infty,n}$, and
\[
K^{\bullet}(\underline G_n^{\infty,n})=\varinjlim_r K^{\bullet}(\underline G_n^{r,n}).
\]
The universal quotients give compatible classes $c_i$ in this limit.

\begin{proposition}[4.10]
For $S$ quasi-compact there is an isomorphism
\[
K^{\bullet}(\underline G_n^{\infty,n})\simeq K^{\bullet}(S)[[c_1,\ldots,c_n]]/I,
\]
where $I$ is the closed ideal generated by the relations arising from the universal $\gamma$-identities for the tautological quotient and its complement.
\end{proposition}

For finite $r$ the presentation of Theorem VI.4.6 gives
\[
K^{\bullet}(\underline G_n^{r,n})=K^{\bullet}(S)[c_1,\ldots,c_n]/I_r,
\]
with $I_r$ homogeneous and supported in weights $>r-n$. Passing to the limit amounts to completing for the topology defined by the $I_r$. By passing to the splitting algebra and using the nilpotence of the corresponding $\xi_j-1$, this topology is equivalent to the weight topology, which proves the assertion.

\section*{5. Applications to projective bundles; study of $f_*$}
\addcontentsline{toc}{section}{Expose VI, 5. Applications and f star}

Let $S$ be quasi-compact, $E$ locally free of rank $r+1$, and $X=\mathbb P(E)$ with structural morphism $f:X\to S$. On $X$ there is the canonical exact sequence
\[
0\to \mathcal F\to f^*E\to\cO_X(1)\to 0.
\]
Let $E,F,\xi$ denote the corresponding classes in $K^{\bullet}$. Put $x=\xi-1$.

\subsection*{5.2}
For a locally free sheaf $\mathcal G$ on $X$, the higher direct images $R^if_*\mathcal G$ need not be locally free of finite type on $S$, so the direct image $f_*:K^{\bullet}(X)\to K^{\bullet}(S)$ is not defined by alternating sums of higher direct images in this setting. Theorem VI.1.1 provides a substitute. Every $y\in K^{\bullet}(X)$ has a unique expansion
\begin{equation}\label{eq:VI.5.2.1}
y=a_0+a_1\xi+\cdots+a_r\xi^r,
\qquad a_i\in K^{\bullet}(S),
\end{equation}
and we put
\begin{equation}\label{eq:VI.5.2.2}
f_*(y)=a_0.
\end{equation}
This is additive and satisfies the projection formula. Moreover
\begin{equation}\label{eq:VI.5.2.3}
f_*(1)=1,
\qquad
f_*(\xi^n)=0\quad(-r\leq n<0),
\qquad
f_*(\xi^n)=s_n\quad(n\geq 0),
\end{equation}
where $s_n$ is the $n$-th complete symmetric function. The last relations follow by induction from the dependence relation and the Koszul relation \eqref{eq:VI.1.11.3}.

\begin{proposition}[5.3]\label{prop:VI.5.3}
For every $k\geq 0$,
\begin{equation}\label{eq:VI.5.3}
\Fil^kK^{\bullet}(X)=\Fil^kK^{\bullet}(S)+\Fil^{k-1}K^{\bullet}(S)x+\cdots+\Fil^{k-r}K^{\bullet}(S)x^r,
\end{equation}
as a direct sum of $K^{\bullet}(S)$-modules, where the filtration is the canonical filtration of augmented $\lambda$-algebras.
\end{proposition}

Here $x$ has zero augmentation and
\[
\gamma_t(x)=\lambda_{t/(1-t)}(1-\xi^{-1})=\frac{1}{1-xt},
\]
so $\gamma^i(x)=x^i$. The result follows from the filtration computation for free $\lambda$-algebras in V, 4.12 and 4.13, together with the relation \eqref{eq:VI.1.13.1}. Directness follows from the independence of $1,\ldots,x^r$.

\begin{corollary}[5.4]
The filtration induced on $K^{\bullet}(S)$ by $K^{\bullet}(X)$ is its original filtration.
\end{corollary}

\begin{corollary}[5.5]
If $X$ is a product over $S$ of flag bundles of locally free sheaves, then
\[
\Gr K^{\bullet}(S)\longrightarrow \Gr K^{\bullet}(X)
\]
is injective.
\end{corollary}

\begin{corollary}[5.6]
With the projective-bundle notation above, $\Gr K^{\bullet}(X)$ is a free $\Gr K^{\bullet}(S)$-module with basis $1,x,\ldots,x^r$, and
\[
\Gr^kK^{\bullet}(X)=\Gr^kK^{\bullet}(S)\oplus\Gr^{k-1}K^{\bullet}(S)x\oplus\cdots\oplus\Gr^{k-r}K^{\bullet}(S)x^r.
\]
The elements $1,x,\ldots,x^{r+1}$ satisfy
\begin{equation}\label{eq:VI.5.6}
\sum_{i=0}^{r+1}(-1)^ic_i(E)x^{r+1-i}=0,
\end{equation}
where $c_i(E)$ is the $i$-th Chern class of $E$.
\end{corollary}

\begin{corollary}[5.7]
For a flag bundle as in 4.5, the $\Gr K^{\bullet}(S)$-algebra $\Gr K^{\bullet}(X)$ is generated by the classes $\overline\xi_i^{(\alpha)}$ of $\xi_i^{(\alpha)}-1$, with relations generated by
\[
c_t(E)=\prod_{i,\alpha}c_t(\xi_i^{(\alpha)}),
\]
where $c_t$ denotes the total Chern class.
\end{corollary}

\begin{corollary}[5.8]
For every $k$,
\[
f_*(\Fil^kK^{\bullet}(X))\subset \Fil^kK^{\bullet}(S).
\]
\end{corollary}
This is immediate from Proposition VI.5.3 and the projection formula.

\begin{proposition}[5.9]
With the notation of 5.1, for every $y\in K^{\bullet}(X)$ one has
\[
f_*(\lambda_{-1}(\mathcal F)y)=0
\qquad\text{and}\qquad
f_*(\gamma^*(\mathcal F)y)=0.
\]
In particular,
\[
f_*(\lambda_{-1}(\mathcal F))=1.
\]
\end{proposition}

Define coefficientwise
\[
\xi_t:1+K^{\bullet}(X)[[t]]^+\to 1+K^{\bullet}(S)[[t]]^+,
\qquad
\xi_t\left(\sum a_it^i\right)=\sum f_*(a_i)t^i.
\]
Although not generally a group homomorphism, it satisfies
\[
\xi_t(mf^*y)=\xi_t(m)y.
\]
Using $E=F+\xi$ and $\lambda_t(E)=\lambda_t(F)(1+
\xi t)$ gives the asserted identities.

\begin{proposition}[5.10]
Let $z\in K^{\bullet}(X)$. The condition $z(1-\xi)=0$ is equivalent to the existence of $y\in K^{\bullet}(S)$ such that
\begin{enumerate}[label=(\roman*)]
\item $z=y\lambda_{-1}(\mathcal F)$;
\item $y\lambda_{-1}(E)=0$.
\end{enumerate}
\end{proposition}

The sufficiency follows from
\[
\lambda_{-1}(E)=\lambda_{-1}(F)\lambda_{-1}(\xi)=\lambda_{-1}(F)(1-\xi).
\]
Conversely write $z=\sum_{i=0}^{r}a_i\xi^i$. Combining the dependence relation for the $\xi^i$ with $z(1-\xi)=0$ and using the independence of $1,\ldots,\xi^r$ expresses $z$ as a multiple of $\lambda_{-1}(F)$, and the top relation gives $y\lambda_{-1}(E)=0$.

\begin{proposition}[5.11]
Let $z\in\Gr K^{\bullet}(X)$. The condition $zx=0$ is equivalent to the existence of $y\in\Gr K^{\bullet}(S)$ such that
\begin{enumerate}[label=(\roman*)]
\item $z=y\,c^{\mathrm{top}}(F)$;
\item $y\,c^{\mathrm{top}}(E)=0$.
\end{enumerate}
\end{proposition}
The proof is the same as that of 5.10, after passing to associated graded rings and using 5.6.

\section*{6. Study of the filtration of $K^{\bullet}(X)$ when $X$ has an ample sheaf}
\addcontentsline{toc}{section}{Expose VI, 6. Filtration when X has an ample sheaf}

Let $X$ be a scheme possessing an ample sheaf $\mathcal L$. Then the two definitions of $K^{\bullet}(X)$, by locally free sheaves of finite type or by perfect complexes, coincide. Let
\[
\varepsilon:K^{\bullet}(X)\to H^0(X,\ZZ)
\]
be the augmentation.

\begin{proposition}[6.1]
If $X$ is quasi-compact and has an ample sheaf, then $\Fil^1K^{\bullet}(X)$ is a nilideal.
\end{proposition}

It suffices to show that $E-\varepsilon(E)$ is nilpotent for the class $E$ of a locally free sheaf. By decomposing $X$ into finitely many disjoint opens and then passing to the flag bundle of $E$, one reduces to the case where $E$ is the class of an invertible sheaf. Let $L$ be the class of an ample invertible sheaf. For $n$ large, $E^{\vee}(n)$ is generated by global sections, giving a surjection $(E(-n))^m\to\cO_X$ with locally free kernel. Thus
\[
mEL^{-n}=1+F.
\]
The corresponding $\gamma$-relation shows that $1-EL^{-n}$, hence $L^n-E$, is nilpotent. Since $E-1=(L^n-1)-(L^n-E)$ and $L^n-1$ is also nilpotent for large $n$, the result follows.

\begin{proposition}[6.2]
Let $R$ be an augmented sub-$\lambda$-algebra of $K^{\bullet}(X)$ generated by the classes of finitely many sheaves. Then the $\lambda$-ring filtration of $R$ is discrete and $\Fil^1R$ is nilpotent. Moreover $R$ is a finite module over its image in $H^0(X,\ZZ)$.
\end{proposition}

After reducing to constant ranks, $R$ is generated by finitely many elements $x_i=E_i-\varepsilon(E_i)$ and their $\gamma$-operations. Proposition 6.1 makes all these elements nilpotent. Hence the augmentation ideal of $R$ is nilpotent. Since only finitely many non-zero monomials in the $x_i$ and $\gamma^j(x_i)$ remain, the filtration is discrete and $R$ is finite over the augmentation ring.

\begin{corollary}[6.3]
Let $A$ be a commutative ring. An element $y$ of $K^{\bullet}(X)\otimes_{\ZZ}A$ is invertible if and only if its augmentation is invertible. In particular, the class of any nowhere-zero locally free sheaf of finite type becomes invertible in $K^{\bullet}(X)\otimes_{\ZZ}A$.
\end{corollary}
The augmentation ideal is nil, so invertibility is detected after quotienting by it.

\begin{corollary}[6.4]
Let $X$ be quasi-compact with an ample sheaf, and let $f:X'\to X$ be proper and perfect. Assume either $f$ is projective or $X$ is noetherian, so that the finiteness theorem applies. If there is $x'\in K^{\bullet}(X'\otimes A)$ such that the augmentation of $f_*(x')$ is invertible, then
\[
f^*:K^{\bullet}(X\otimes A)\to K^{\bullet}(X'\otimes A)
\]
is injective.
\end{corollary}
Indeed, by the projection formula,
\[
f_*(f^*y\cdot x')=y\cdot f_*(x'),
\]
and $f_*(x')$ is invertible by 6.3.

\subsection*{6.5--6.6. The codimension filtration}
Assume now that $X$ is noetherian of dimension $d$ and has an ample sheaf. Define a filtration $K^{\bullet}(X)_i$ as follows: $x\in K^{\bullet}(X)_i$ if and only if for every closed subscheme $Y\subset X$ there is a perfect complex $M^{\bullet}(Y)$ with
\begin{enumerate}[label=(\alph*)]
\item $\cl(M^{\bullet})=x$;
\item $\codim(\Supp H^j(M^{\bullet})\cap Y,Y)\geq i+j$, with $H^j(M^{\bullet})=0$ for $j\gg 0$.
\end{enumerate}
These $K^{\bullet}(X)_i$ are additive subgroups.

\begin{proposition}[6.6]
The filtration of 6.5 is a ring filtration, and $K^{\bullet}(X)_{d+1}=0$.
\end{proposition}

If $x$ and $x'$ are represented by perfect complexes $M^{\bullet}$ and $M'^{\bullet}$ satisfying the codimension estimates, then $xx'$ is represented by $M^{\bullet}\otimes M'^{\bullet}$. The support of $H^n(M^{\bullet}\otimes M'^{\bullet})$ is contained in the union of the intersections of supports with total degree $n$, and the codimension estimate follows from EGA $0_{\mathrm{IV}}$, 14.2.2.3. Finally $K^{\bullet}(X)_{d+1}=0$ because a support in codimension $>d$ is empty.

\begin{lemma}[6.7]
Let $X=\Spec A$ be affine noetherian, let $x_1,\ldots,x_n$ be finitely many points of $X$, and let $\mathcal E$ be invertible. There is a section $s\in\Gamma(X,\mathcal E)$ which is a basis of $\mathcal E_{x_i}$ for all $i$.
\end{lemma}
The proof uses closed specializations $x'_i$ with distinct maximal ideals $\mathfrak m_i$, the isomorphisms $M/\mathfrak m_iM\simeq A/\mathfrak m_i$ for $M=\Gamma(X,\mathcal E)$, and Nakayama's lemma.

\begin{lemma}[6.8]
Let $X$ be noetherian with an ample sheaf $\mathcal L$, let $\mathcal E$ be invertible, and let $x_1,\ldots,x_n$ be points of $X$. There is an integer $N$ such that, for every $k\geq 1$, $\mathcal E(kN)$ has a global section which is a basis at every $x_i$.
\end{lemma}
Choose a power of the ample sheaf and a section whose non-vanishing locus is affine and contains all $x_i$, apply Lemma 6.7 there, and extend after multiplying by a further power.

\begin{theorem}[6.9]\label{thm:VI.6.9}
Let $X$ be a noetherian scheme of dimension $d$ with an ample sheaf. Then
\[
\Fil^{d+1}K^{\bullet}(X)=0,
\]
and therefore $(\Fil^1K^{\bullet}(X))^{d+1}=0$.
\end{theorem}

First, if $\mathcal E$ and $\mathcal E'$ are invertible, with classes $E,E'$, then $E-E'\in K^{\bullet}(X)_1$. For a closed $Y\subset X$, choose one point in each irreducible component. Lemma 6.8 gives morphisms from suitable negative powers of the ample sheaf to $\mathcal E$ and $\mathcal E'$ which are isomorphisms at those points. The two-term complex built from these morphisms has class $E-E'$ and is acyclic at the chosen generic points of $Y$, hence its cohomology supports have codimension at least $1$ in $Y$.

Now take $x\in\Fil^{d+1}K^{\bullet}(X)$. It suffices to treat products
\[
\gamma^{i_1}(E_1-F_1)\cdots\gamma^{i_k}(E_k-F_k),
\qquad
\sum i_j\geq d+1,
\]
with $E_i,F_i$ represented by locally free sheaves of the same constant rank. Pull back to the product of the flag bundles of these sheaves. The pullback is injective on $K^{\bullet}$ and splits each class into sums of invertible classes. For invertible classes $\xi,\eta$,
\[
\gamma_t(\xi-\eta)=\frac{\gamma_t(\xi-1)}{\gamma_t(\eta-1)}=\frac{1+(\xi-1)t}{1+(\eta-1)t},
\]
so $\gamma^i(\xi-\eta)=(-1)^{i-1}(\xi-\eta)(\eta-1)^{i-1}\in K^{\bullet}(X')_i$. Thus the pullback of $x$ belongs to $K^{\bullet}(X')_{d+1}$. Multiplying by a basis element of $K^{\bullet}(X')$ over $K^{\bullet}(X)$ of relative codimension $r$ gives an element of $K^{\bullet}(X')_{d+1+r}=0$, since $\dim X'=d+r$. By the basis property and injectivity, $x=0$.

\begin{problem}[6.10]
Under the hypotheses of Theorem VI.6.9, is it true that for every $k$ one has
\[
K^{\bullet}(X)_k=\Fil^kK^{\bullet}(X)?
\]
\end{problem}
