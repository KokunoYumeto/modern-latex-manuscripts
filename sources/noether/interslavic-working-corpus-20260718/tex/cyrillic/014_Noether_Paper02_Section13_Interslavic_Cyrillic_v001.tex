\begin{center}
\textbf{\S\ 13. \quad Формы осмого поряда (Систем III).}
\end{center}
\[
\text{Свијанье }\left\{\begin{array}{l}
\A\cdot j \text{ с }\nu,\\
\theta^2,\ f\cdot j,\ N \text{ с }H,\\
\theta \text{ с }L,\sigma.
\end{array}\right.
\]

\noindent\textit{Нередуцибилне формы:}
\[
\begin{array}{c@{\quad}l@{\qquad}c@{\quad}l}
\mathrm{A)}&\A_\nu(j\nu x)&
\mathrm{B)}&\begin{array}{c}(\vartheta^2\eta x)^3\\(\vartheta^2\eta x)^4\end{array}
\quad H_\vartheta(\vartheta^2\eta x)^3,
\ \theta_\eta(\vartheta^2\eta x)^3,\\[1em]
\mathrm{C)}&(aHu)H_j,\quad a_\eta(aHu)H_j&
\mathrm{D)}&H_n,\quad N_\eta.
\end{array}
\]
\[
\mathrm{E)}\quad
\begin{array}{c|c|c|c}
\cdot&(\theta Lu)^2&(\theta Lu)^3&\cdot\\
\theta_i&\cdot&L_\vartheta(\theta Lu)^2,
\ \theta_i(\theta Lu)^2&\theta_i(\theta Lu)^3\\
(\vartheta lx)^2&\theta_i^2&\cdot&\cdot\\
\cdot&\theta_i(\vartheta lx)^2&\cdot&\cdot
\end{array}
\]

\noindent\textit{Редукције:}

\noindent\mbox{A)} $\A_\nu(j\nu x)^3$: редуцент $a_\nu(j\nu x)^3$.

$\A_\nu(j\nu x)^2$ јест редуцибилна чрез разложиму форму $a_\nu(j\nu x)^2$ по \S\ 3. 2)\mbox{b} с уважаньем редуцента $\theta_{\vartheta j}$. Бо по \S\ 11 \mbox{B)} достајемо:
\[
\begin{aligned}
&\frac3{10}\A_\nu(j\nu x)^2+\frac35a_\nu(j\nu x)a_\vartheta(j\nu x)
+\frac1{10}a_\nu(j\nu x)^2\\
&\qquad=\frac35\theta_\nu^3\theta_{\vartheta j}
+\frac25\theta_\nu^3\theta_{\vartheta j}\theta_{\vartheta j},
\end{aligned}
\]
(редуценты $a_j$ и $\theta_{\vartheta j}$).

\noindent\mbox{B)} Формы $H_\vartheta(\vartheta'\eta x)^2$, $H_\vartheta^2(\vartheta'\eta x)$, $H_\vartheta^2(\vartheta'\eta x)^2$ разкладајут се како настале чрез последовательно једноразово свијанье с разложимыми формами $H_\vartheta$ и $H_\vartheta^2$, соответне $H_\vartheta a_\eta$ и $H_\vartheta^2a_\eta$, по \S\ 3. 2)\mbox{b)} и по релацији:
\[
H_\vartheta(\vartheta'\eta x)^2=2H_\vartheta a_\eta^2f-2H_\vartheta a_\eta b_\eta.
\]

\noindent\mbox{C)} Ряд форм $a_\eta(j\eta x)^2$: редуценты $a_j$ и $(j\eta x)^3$.

Форма $a_\eta(j\eta x)$ разкладаје се: редуцент $a_j$ и једноразово свијанье с разложимоју формоју $(j\eta x)^2$ по \S\ 3, 2)\mbox{a} (специализација $y=uv$).

Форма $a_\eta H_j$ разкладаје се: 1) чрез једноразово свијанье с разложимоју формоју $a_\nu(j\nu x)^2$;

2) чрез специално двојноразово свијанье с разложимоју формоју $(j\eta x)^2$; из того релација меджу разложимыми формами.

Из $a_\nu(j\nu x)^2$:
\[
0=\theta_\nu'\theta_\nu+2a_\eta H_j\pmod{(s,\R,t,i,J)}.
\]
Из $(j\eta x)^2$:
\[
0=\theta_\nu'\theta_\nu-\frac32a_\eta H_j\pmod{(s,\R,t,i,J)}
\]
(продукты с контравариантами сут изпушчене), по зачатној схеме:
\[
\begin{aligned}
0={}&\frac12a_\nu(abu)^2\theta_\nu^3
+\frac12u_x(ubu)\theta_{\nu_1}^3(\theta bu)
-\frac34(\widehat{j}\eta bu)^2\\
&+\frac34H_j(\widehat{j}\eta bu)-\frac18fH_j^2
\end{aligned}
\]
и заменоју симболов в $a_\nu(ubu)(\theta bu)\theta_{\nu_1}^3$.

\noindent\mbox{D)} Ряд форм $N_\eta(NHu)$: редуцент $\A_\nu^2$.

$(NHu)$, $(n\eta x)$, $N_\eta(n\eta x)$, $H_\eta(NHu)$ сут разложиме формы (срав. \S\ 12 \mbox{B)}); $(NHu)^2$ разкладаје се чрез једноразово свијанье с формоју $(\A Hu)^2$.

\noindent\mbox{E)} Ряды форм $\theta_iL_\vartheta$, $\theta_i^2(\theta Lu)^2$, $\theta_i^2(\vartheta lx)$:

редуценты: $\theta_\eta H_\vartheta$, $\theta_\eta^2(\theta Hu)^2$, $\theta_\eta^2(\vartheta\eta x)$.

Формы $(\theta Lu)$, $(\vartheta lx)$, $L_\vartheta$, $L_\vartheta(\theta Lu)$, $\theta_i(\theta Lu)$, $L_\vartheta(\vartheta lx)$, $\theta_i(\vartheta lx)$, $L_\vartheta^2$, $L_\vartheta^2(\theta Lu)$, $\theta_i^2(\theta Lu)$ разкладајут се. (Они сут дуалистично противоположне разложимым формам из \S\ 12 \mbox{B)}.)

\noindent\mbox{F)} Ряд форм $\theta_\sigma(\vartheta\sigma x)$: редуцент $a_\sigma^2$.

Формы $(\vartheta\sigma x)$ и $(\vartheta\sigma x)^2$ разкладајут се како настале чрез једноразово свијанье с разложимоју формоју $a_\sigma$; $\theta_\sigma$, настала чрез двојноразово свијанье, разкладаје се, бо ряд форм
\[
a_{\nu_2}^2(a\theta u)\theta_{\nu_1}^3\equiv H_j^2(j\eta x)\equiv0\pmod{u_x(s,\R,t,i,J)}:
\]
\[
\theta_\sigma=a_\sigma(abu)^2=a_\nu^2(abu)^2\theta_\nu^3.
\]

\noindent\textit{Последки:}

$\theta^3$ або $\theta^2K$ не входи в свијанье с $H$; $\theta$, разматривана једино како контравариант, входи в степени або продукты в свијаньи с $L$, але обче не входи в свијанье с $\sigma$ и $(\nu\sigma x)$.

$N$ не входи в продукты в свијаньи с којукоľвек формоју система II, так само не с степенами або продуктами форм система II. Из продуктов в систему I, с уважаньем последков последньих параграфов, оставајут једино продукты $f\cdot j$, $\A\cdot j$, степени $\theta$ и продукты $\theta\cdot K$.
