\textbf{\S\ 8. \quad Формы третјего поряда (Систем III).}
\end{center}

\noindent\textbf{Свијанье $f$ с $\nu$.}

\noindent\textit{Нередуцибилне формы:}
\[
\begin{array}{cccc}
a_\nu&\cdot&\cdot&\cdot\\
\cdot&a_\nu^2&\cdot&\cdot\\
\cdot&\cdot&\cdot&\cdot\\
\cdot&\cdot&\cdot&a_\nu^4=i.
\end{array}
\]

\noindent\textit{Редукција:}
\[
a_\nu^3=\frac13 i u_x\qquad\text{(формула (2.)).}
\]
\noindent\textit{Последки:} 1) $a_\nu^3$ јест редуцент. 2) $f$ входи в свијанье с $\nu$ само в продуктах с контрагредиентными формами $(j)$. То само важи за $\nu$ в односу к $f$.
