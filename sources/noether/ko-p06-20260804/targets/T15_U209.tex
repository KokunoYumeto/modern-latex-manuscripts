\emph{보조정리. $n$개의 양들의 행 $(yz\cdots t)$}
\[
\begin{array}{@{}l@{\quad}c@{}}
(1)&
\begin{array}{cccc}
  y_1&z_1&\cdots&t_1\\
  y_2&z_2&\cdots&t_2\\
  \vdots&\vdots&&\vdots\\
  y_n&z_n&\cdots&t_n
\end{array}
\end{array}
\]
\emph{에 대한 정수계수 다항 대칭함수들은 그 가운데 유한개의 함수로 된 정수계수 다항함수이다. 구체적으로 그 유한개의 함수는 기본대칭함수}
\[
\begin{array}{@{}l@{\quad}c@{}}
(2)&
\begin{gathered}
  \sigma_1(y),\sigma_2(y),\ldots,\sigma_n(y);\quad
  \sigma_1(z),\sigma_2(z),\ldots,\sigma_n(z);\quad\ldots;\\
  \sigma_1(t),\ldots,\sigma_n(t);
\end{gathered}
\end{array}
\]
\emph{와 함수}
\[
\begin{array}{@{}l@{\quad}c@{}}
(3)&
  \chi_1(y,z,\ldots,t),\ldots,\chi_k(y,z,\ldots,t),
\end{array}
\]
\emph{이며, 함수 (3)은 함수 (2)에 대하여 대수적으로 정수적이고 정수계수적으로 종속된다.}
