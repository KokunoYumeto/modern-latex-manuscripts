
\chapter{Матричні похідні, найменші квадрати і перевірка Якобіанів}

\section{Чому це входить до першого ядра}

Фільтри, факторні графи, калібрування, SLAM, оцінювання параметрів і керування всі зводяться до Якобіанів. Якщо переклад не дає читачеві стабільної мови матричних похідних, то пізніші розділи будуть виглядати як набір магічних формул.

\section{Лінеаризація}

Для функції
\[
f:\R^n\to\R^m
\]
поблизу $x$:
\[
f(x+\delta x)\simeq f(x)+J\delta x,
\qquad
J=\frac{\partial f}{\partial x}\in\R^{m\times n}.
\]
Розмірність $J$ завжди ``вихід на вхід''. Якщо $f$ має $m$ компонент, а $x$ має $n$ компонент, то $J$ має $m$ рядків і $n$ стовпців.

\section{Скалярна квадратична форма}

Нехай
\[
\phi(x)=\frac12\|Ax-b\|^2
=\frac12(Ax-b)\trans(Ax-b).
\]
Тоді
\[
\nabla\phi(x)=A\trans(Ax-b).
\]
Якщо є вагова матриця $W\succ0$:
\[
\phi(x)=\frac12(Ax-b)\trans W(Ax-b),
\]
то
\[
\nabla\phi(x)=A\trans W(Ax-b).
\]

\section{Нормальні рівняння}

Мінімум
\[
\min_x \|Ax-b\|^2
\]
задовольняє
\[
A\trans A x=A\trans b.
\]
Якщо $A\trans A$ погано обумовлена, явне розв'язання нормальних рівнянь може погіршити точність. Краще використовувати QR або SVD. Для зваженої задачі:
\[
A\trans W A x=A\trans Wb.
\]

\section{Гаус--Ньютон}

Для нелінійного залишку $r(x)$:
\[
\min_x \frac12\|r(x)\|^2.
\]
Лінеаризація:
\[
r(x+\delta x)\simeq r(x)+J\delta x.
\]
Крок Гауса--Ньютона:
\[
(J\trans J)\delta x=-J\trans r.
\]
Оновлення:
\[
x\leftarrow x+\delta x.
\]
На многовидах:
\[
X\leftarrow X\oplus\delta x.
\]

\section{Левенберг--Марквардт}

Демпфований крок:
\[
(J\trans J+\lambda I)\delta x=-J\trans r.
\]
Або масштабований варіант:
\[
(J\trans J+\lambda\operatorname{diag}(J\trans J))\delta x=-J\trans r.
\]
Велике $\lambda$ робить крок обережнішим; мале $\lambda$ наближає метод до Гауса--Ньютона.

\section{Скінченно-різницева перевірка Якобіана}

Для кожного базисного напрямку $e_i$:
\[
J_{:,i}^{\mathrm{num}}
\simeq
\frac{f(x+\epsilon e_i)-f(x-\epsilon e_i)}{2\epsilon}.
\]
Порівняння:
\[
\frac{\|J-J^{\mathrm{num}}\|}{1+\|J^{\mathrm{num}}\|}
\]
має бути малим. Типові значення $\epsilon$ для double precision лежать приблизно між $10^{-6}$ і $10^{-4}$, але це залежить від масштабу змінної.

\section{Перевірка Якобіана на групі Лі}

Якщо стан $X\in\SE(3)$, не можна тестувати $X+\epsilon e_i$. Треба тестувати через $\oplus$:
\[
J_{:,i}^{\mathrm{num}}
\simeq
\frac{f(X\oplus\epsilon e_i)-f(X\oplus(-\epsilon e_i))}{2\epsilon}.
\]
Якщо вихід також на многовиді, різницю виходів треба брати через $\ominus$:
\[
J_{:,i}^{\mathrm{num}}
\simeq
\frac{(f(X\oplus\epsilon e_i)\ominus f(X\oplus(-\epsilon e_i)))}{2\epsilon}.
\]
Це один з найважливіших практичних моментів для перекладу micro Lie матеріалу.

\section{Ознаки неправильної похідної}

\begin{itemize}
\item Фільтр працює лише при дуже великому $R$.
\item $P$ швидко втрачає симетрію або додатну напіввизначеність.
\item NIS систематично зашкалює.
\item Оптимізатор зменшує cost на першій ітерації, але далі розбігається.
\item Зміна знаку однієї орієнтаційної похибки ``виправляє'' усе. Це часто означає змішування лівої і правої домовленості.
\end{itemize}
