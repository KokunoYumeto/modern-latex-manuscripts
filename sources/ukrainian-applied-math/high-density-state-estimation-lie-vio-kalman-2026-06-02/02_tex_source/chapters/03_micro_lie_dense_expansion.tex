\section{Micro-Lie: групи Лі, збурення і Якобіани для оцінювання}
\sourceblock{Solà, Deray, Atchuthan, \textit{A micro Lie theory for state estimation in robotics}, файли \texttt{manifolds.tex}, \texttt{SO3.tex}, \texttt{SE3.tex}.}

\subsection{Навіщо потрібна групова нотація}

У навігації й робототехніці стан часто лежить не в плоскому просторі $\R^n$, а на многовиді: орієнтація в $\SO(3)$, поза в $\SE(3)$, комбінація швидкостей, зміщень і калібрувань у добутку груп і векторних просторів. Якщо переплутати евклідову різницю з груповою похибкою, лінеаризація стає залежною від координат і може псувати коваріацію.

Група Лі $\mathcal M$ має групову операцію $\circ$, одиницю $e$ і обернений елемент $X^{-1}$. Її дотичний простір в одиниці - алгебра Лі $\mathfrak m$, яку для обчислень ототожнюють із $\R^m$ через оператори hat/vee:
\begin{equation}
 \tau\in\R^m,\qquad \tau^\wedge\in\mathfrak m,
 \qquad (\tau^\wedge)^\vee=\tau.
\end{equation}

\subsection{Exp, Log, plus, minus}

Експоненціальна мапа переносить малий вектор з дотичного простору на групу:
\begin{equation}
 X = \Exp(\tau),\qquad \tau=\Log(X).
\end{equation}
Правий плюс і правий мінус часто визначають так:
\begin{align}
 X\oplus \tau &= X\circ\Exp(\tau),\\
 Y\ominus X &= \Log(X^{-1}\circ Y).
\end{align}
Лівий плюс/мінус мають інший порядок множення. У перекладі й коді це треба фіксувати явно, бо ліві та праві Якобіани відрізняються знаком і adjoint-перенесенням.

\subsection{$\SO(3)$}

Для $\phi\in\R^3$ позначимо $\theta=\norm{\phi}$. Hat-оператор:
\begin{equation}
 \phi^\wedge=\skew{\phi}=\bbm{0&-\phi_3&\phi_2\\ \phi_3&0&-\phi_1\\ -\phi_2&\phi_1&0}.
\end{equation}
Експонента Rodrigues:
\begin{equation}
 \Exp(\phi)=I+\frac{\sin\theta}{\theta}\phi^\wedge+
 \frac{1-\cos\theta}{\theta^2}(\phi^\wedge)^2.
\end{equation}
Логарифм, для $R\in\SO(3)$:
\begin{equation}
 \Log(R)=\frac{\theta}{2\sin\theta}(R-R\T)^\vee,
 \qquad \theta=\cos^{-1}\left(\frac{\operatorname{tr}(R)-1}{2}\right).
\end{equation}
Поблизу нуля треба застосовувати розклади Тейлора, а не наївне ділення на $\theta$.

Правий Якобіан $\SO(3)$:
\begin{equation}
 J_r(\phi)=I-\frac{1-\cos\theta}{\theta^2}\phi^\wedge+
 \frac{\theta-\sin\theta}{\theta^3}(\phi^\wedge)^2.
\end{equation}
Лівий Якобіан:
\begin{equation}
 J_l(\phi)=J_r(-\phi).
\end{equation}
Ці матриці потрібні для коректної лінеаризації збурень, інтеграції кутових швидкостей і перенесення коваріацій на многовиді.

\subsection{$\SE(3)$}

Елемент $T\in\SE(3)$ записують як
\begin{equation}
 T=\bbm{R&t\\0&1},\qquad R\in\SO(3),\quad t\in\R^3.
\end{equation}
Для $\xi=(\rho,\phi)\in\R^6$ маємо
\begin{equation}
 \xi^\wedge=\bbm{\phi^\wedge&\rho\\0&0}.
\end{equation}
Експонента:
\begin{equation}
 \Exp(\xi)=\bbm{\Exp(\phi)&V(\phi)\rho\\0&1},
\end{equation}
де $V(\phi)$ збігається з лівим Якобіаном $\SO(3)$ у поширеній конвенції:
\begin{equation}
 V(\phi)=I+\frac{1-\cos\theta}{\theta^2}\phi^\wedge+
 \frac{\theta-\sin\theta}{\theta^3}(\phi^\wedge)^2.
\end{equation}

Adjoint для $T=(R,t)$:
\begin{equation}
 \Ad_T=\bbm{R&\skew{t}R\\0&R}.
\end{equation}
Його головна роль - переносити мале збурення між системами координат:
\begin{equation}
 T\Exp(\xi)=\Exp(\Ad_T\xi)T.
\end{equation}

\subsection{Похибки, коваріації, композиція}

Якщо $X$ має праве мале збурення $X=\bar X\Exp(\epsilon)$, $\epsilon\sim\calN(0,P)$, і $Y=f(X)$, тоді локально
\begin{equation}
 \delta y \approx J\epsilon,
 \qquad P_Y\approx JPJ\T.
\end{equation}
Для композиції $Z=XY$ похибка часто має вигляд
\begin{equation}
 \delta z \approx \Ad_{Y^{-1}}\delta x + \delta y,
\end{equation}
залежно від правої/лівої конвенції. Саме тут автоматичні переклади часто стають небезпечними: вони можуть правильно перекласти слова, але залишити неоднозначним, чи похибка множиться зліва, чи справа.

\subsection{Фактор-графова форма}

Для фактору $r_i(x)$ задача найменших квадратів:
\begin{equation}
 \min_{\delta x}\sum_i \norm{r_i(x\oplus\delta x)}^2_{\Sigma_i^{-1}}.
\end{equation}
Лінеаризація:
\begin{equation}
 r_i(x\oplus\delta x)\approx r_i(x)+J_i\delta x.
\end{equation}
Нормальні рівняння:
\begin{equation}
 \left(\sum_i J_i\T\Sigma_i^{-1}J_i\right)\delta x
 = -\sum_i J_i\T\Sigma_i^{-1}r_i.
\end{equation}
Оновлення:
\begin{equation}
 x\leftarrow x\oplus\delta x.
\end{equation}
Це є спільна мова для ESKF, VIO, SLAM, bundle adjustment і калібрування.
