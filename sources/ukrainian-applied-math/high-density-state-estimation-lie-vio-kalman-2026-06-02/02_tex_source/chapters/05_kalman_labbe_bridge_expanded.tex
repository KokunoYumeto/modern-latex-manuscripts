\subsection{g-h filter}

Найпростіший фільтр пояснює ідею prediction-update без матриць. Для скалярного стану:
\begin{align}
 \hat x_{k|k-1} &= \hat x_{k-1}+\dot x_{k-1}\Delta t,\\
 r_k &= z_k-\hat x_{k|k-1},\\
 \hat x_k &= \hat x_{k|k-1}+g r_k,\\
 \dot x_k &= \dot x_{k-1}+h\frac{r_k}{\Delta t}.
\end{align}
Український модуль має пояснити, що $g$ коригує положення, а $h$ коригує швидкість. Це не ``gain'' як прибуток, а коефіцієнт підсилення/ваги корекції.

\subsection{Discrete Bayes filter}

Для дискретної сітки станів:
\begin{equation}
 \overline{bel}(x_t)=\sum_{x_{t-1}}p(x_t\mid x_{t-1},u_t)bel(x_{t-1}),
\end{equation}
\begin{equation}
 bel(x_t)=\eta\,p(z_t\mid x_t)\overline{bel}(x_t).
\end{equation}
Технічний переклад має зберегти різницю між prior/prediction $\overline{bel}$ і posterior $bel$.

\subsection{Добуток гаусіан}

Якщо
\begin{equation}
 x\sim\calN(\mu_1,\sigma_1^2),\qquad x\sim\calN(\mu_2,\sigma_2^2),
\end{equation}
то ненормований добуток пропорційний гаусіані з
\begin{align}
 \mu &= \frac{\sigma_2^2\mu_1+\sigma_1^2\mu_2}{\sigma_1^2+\sigma_2^2},\\
 \sigma^2 &= \frac{\sigma_1^2\sigma_2^2}{\sigma_1^2+\sigma_2^2}.
\end{align}
Це найкоротший шлях пояснити, чому точніший сенсор має більшу вагу.

\subsection{Лінійний фільтр Калмана}

Модель:
\begin{align}
 x_k &= F_kx_{k-1}+B_ku_k+w_k, & w_k&\sim\calN(0,Q_k),\\
 z_k &= H_kx_k+v_k, & v_k&\sim\calN(0,R_k).
\end{align}
Прогноз:
\begin{align}
 \hat x_{k|k-1}&=F_k\hat x_{k-1|k-1}+B_ku_k,\\
 P_{k|k-1}&=F_kP_{k-1|k-1}F_k\T+Q_k.
\end{align}
Оновлення:
\begin{align}
 y_k&=z_k-H_k\hat x_{k|k-1},\\
 S_k&=H_kP_{k|k-1}H_k\T+R_k,\\
 K_k&=P_{k|k-1}H_k\T S_k^{-1},\\
 \hat x_{k|k}&=\hat x_{k|k-1}+K_ky_k,\\
 P_{k|k}&=(I-K_kH_k)P_{k|k-1}(I-K_kH_k)\T+K_kR_kK_k\T.
\end{align}
