\section{IMU-кінематика}

Стан і стан похибки:
\[
\vect{x}=\begin{bmatrix}\vect{p}&\vect{v}&q&\vect{a}_b&\vect{\omega}_b&\vect{g}\end{bmatrix}\trans,
\qquad
\delta\vect{x}=\begin{bmatrix}\delta\vect{p}&\delta\vect{v}&\delta\vect{\theta}&\delta\vect{a}_b&\delta\vect{\omega}_b&\delta\vect{g}\end{bmatrix}\trans.
\]
Композиція істинного стану з номінального та похибкового:
\[
\vect{p}_t=\vect{p}+\delta\vect{p},
\quad
\vect{v}_t=\vect{v}+\delta\vect{v},
\quad
q_t=q\qmul\delta q,
\quad
\vect{g}_t=\vect{g}+\delta\vect{g}.
\]
Для малих кутів $\delta\vect{\theta}$:
\[
\delta q\simeq \begin{bmatrix}1\\ \delta\vect{\theta}/2\end{bmatrix},
\qquad
\delta R\simeq I+\hatmap{\delta\vect{\theta}}.
\]

Модель вимірювання IMU:
\begin{align}
\vect{a}_m &= R_t\trans(\vect{a}_t-\vect{g}_t)+\vect{a}_{bt}+\vect{a}_n,\\
\vect{\omega}_m &= \vect{\omega}_t+\vect{\omega}_{bt}+\vect{\omega}_n.
\end{align}
Звідси
\begin{align}
\vect{a}_t &= R_t(\vect{a}_m-\vect{a}_{bt}-\vect{a}_n)+\vect{g}_t,\\
\vect{\omega}_t &= \vect{\omega}_m-\vect{\omega}_{bt}-\vect{\omega}_n.
\end{align}

Істинна кінематика:
\begin{align}
\dot{\vect{p}}_t &= \vect{v}_t,\\
\dot{\vect{v}}_t &= R_t(\vect{a}_m-\vect{a}_{bt}-\vect{a}_n)+\vect{g}_t,\\
\dot{q}_t &= \frac{1}{2}q_t\qmul(\vect{\omega}_m-\vect{\omega}_{bt}-\vect{\omega}_n),\\
\dot{\vect{a}}_{bt} &= \vect{a}_w,\\
\dot{\vect{\omega}}_{bt} &= \vect{\omega}_w,\\
\dot{\vect{g}}_t &= 0.
\end{align}

Номінальна кінематика:
\begin{align}
\dot{\vect{p}} &= \vect{v},\\
\dot{\vect{v}} &= R(q)(\vect{a}_m-\vect{a}_b)+\vect{g},\\
\dot{q} &= \frac{1}{2}q\qmul(\vect{\omega}_m-\vect{\omega}_b),\\
\dot{\vect{a}}_b &= 0,\\
\dot{\vect{\omega}}_b &= 0,\\
\dot{\vect{g}} &= 0.
\end{align}

Лінеаризована локальна динаміка похибки:
\begin{align}
\dot{\delta\vect{p}} &= \delta\vect{v},\\
\dot{\delta\vect{v}} &\simeq -R(q)\hatmap{\vect{a}_m-\vect{a}_b}\delta\vect{\theta}-R(q)\delta\vect{a}_b-R(q)\vect{a}_n+\delta\vect{g},\\
\dot{\delta\vect{\theta}} &\simeq -\hatmap{\vect{\omega}_m-\vect{\omega}_b}\delta\vect{\theta}-\delta\vect{\omega}_b-\vect{\omega}_n,\\
\dot{\delta\vect{a}}_b &= \vect{a}_w,\\
\dot{\delta\vect{\omega}}_b &= \vect{\omega}_w,\\
\dot{\delta\vect{g}} &= 0.
\end{align}
