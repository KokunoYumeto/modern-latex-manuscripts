\chapter{РЧ та антени: від Максвелла до решіток і вимірюваних величин}
\label{chap:antenna-rf-stack}

\sourcebox{Bridge-розділ. Фактичний ZIP має лише три Maxwell-модулі. Заявлені Lecture 1, polarization, reciprocity, Chebyshev arrays, wire antennas, RCS/EIRP і duality не були в архіві.}

Антенна теорія починається з рівнянь Максвелла. У частотній області при часовій залежності $e^{j\omega t}$ маємо фазорну форму
\begin{align}
 \nabla\times \mathbf{E} &= -j\omega \mathbf{B}-\mathbf{M},\\
 \nabla\times \mathbf{H} &= \mathbf{J}+j\omega \mathbf{D},\\
 \nabla\cdot\mathbf{D} &= \rho_e,\\
 \nabla\cdot\mathbf{B} &= \rho_m.
\end{align}
Для лінійного ізотропного середовища
\begin{equation}
 \mathbf{D}=\epsilon\mathbf{E},\qquad \mathbf{B}=\mu\mathbf{H},\qquad \mathbf{J}=\sigma\mathbf{E}.
\end{equation}

\section{Плоска хвиля і хвильовий імпеданс}

Для плоскої хвилі з напрямом поширення $\hat{\mathbf{k}}$:
\begin{equation}
 \mathbf{E}(\mathbf{r})=\mathbf{E}_0 e^{-j\mathbf{k}\cdot\mathbf{r}},\qquad
 \mathbf{H}(\mathbf{r})=\frac{1}{\eta}\hat{\mathbf{k}}\times\mathbf{E}(\mathbf{r}),
\end{equation}
де $\eta=\sqrt{\mu/\epsilon}$ --- хвильовий імпеданс середовища.

\section{Поляризація}

Якщо
\begin{equation}
 \mathbf{E}(t)=\Re\{(E_x\hat{\mathbf{x}}+E_y e^{j\delta}\hat{\mathbf{y}})e^{j\omega t}\},
\end{equation}
то $\delta=0$ або $\pi$ дає лінійну поляризацію; рівні амплітуди і $\delta=\pm\pi/2$ дають кругову поляризацію; загальний випадок --- еліптична поляризація.

\section{Діаграма спрямованості, підсилення, EIRP}

Нехай $U(\theta,\varphi)$ --- інтенсивність випромінювання. Спрямованість
\begin{equation}
 D(\theta,\varphi)=\frac{4\pi U(\theta,\varphi)}{P_{\rm rad}}.
\end{equation}
Підсилення $G$ враховує також ефективність $\eta_a$:
\begin{equation}
 G(\theta,\varphi)=\eta_a D(\theta,\varphi),\qquad \mathrm{EIRP}=P_tG_t.
\end{equation}

\section{Решітки і множник решітки}

Для лінійної решітки з $N$ однакових елементів на відстані $d$ і фазовим кроком $\beta$ множник решітки
\begin{equation}
 AF(\theta)=\sum_{n=0}^{N-1} w_n e^{jn(kd\cos\theta+\beta)}.
\end{equation}
Ваги $w_n$ задають компроміс між шириною головної пелюстки та рівнем бокових пелюсток.

\section{Радарний переріз}

Радарний переріз $\sigma$ визначає еквівалентну площу розсіювання в дальньому полі. Для ідеалізованої моностатичної формули
\begin{equation}
 P_r \propto \frac{P_t G_t G_r \lambda^2 \sigma}{(4\pi)^3 R^4}.
\end{equation}
У цьому довіднику ця формула потрібна для dimensional literacy: які величини входять, які одиниці мають, чому дальність входить як $R^{-4}$ у простій моделі.

\section{Перевірки перекладу}
\begin{enumerate}[label=\arabic*.]
  \item $\mathbf{D}$ має одиниці $\mathrm{C/m^2}$, не $\mathrm{C/m^3}$.
  \item $\mathbf{J}$ має одиниці $\mathrm{A/m^2}$.
  \item $\mathbf{E}$ --- $\mathrm{V/m}$, $\mathbf{H}$ --- $\mathrm{A/m}$.
\end{enumerate}
