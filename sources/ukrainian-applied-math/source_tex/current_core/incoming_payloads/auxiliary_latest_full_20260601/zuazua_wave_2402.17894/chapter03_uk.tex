% p31 (54) III.1
\section{Опис методу HUM}
\label{sec:III.1}
Як зазначено у розділі~\ref{chapter01}, задачу точної внутрішньої керованості лінійного хвильового рівняння можна сформулювати так.

Нехай $\omega$ — непорожня відкрита підмножина $\Omega$. Розглянемо хвильове рівняння
\begin{equation}\label{eqIII01}
    \begin{cases}
    y^{\prime\prime} - \Delta y = h \chi_\omega \hspace{1.6cm} & \text{в} \hspace{0.3cm} Q
    \\
    y = 0 \hspace{0.5cm} & \text{на} \hspace{0.3cm} \Sigma
    \\
    y(0) = y^0,\ y^\prime(0) = y^1\hspace{0.5cm} & \text{в} \hspace{0.3cm} \Omega,
    \end{cases}
\end{equation}
де $\chi_\omega$ позначає характеристичну функцію $\omega$.

Нас цікавить властивість керованості в енергетичному просторі $H_0^1(\Omega) \times L^2(\Omega)$ з керуваннями в $L^2(\omega \times (0,T))$.

Задача полягає у знаходженні $T>0$ такого, що для кожної пари початкових даних $\{y^0, y^1\} \in H_0^1(\Omega) \times L^2(\Omega)$ існує керування $h \in L^2(\omega \times (0,T))$, для якого розв'язок~(\ref{eqIII01}) задовольняє
\begin{equation}\label{eqIII02}
y(T) = y^\prime(T) = 0.
\end{equation}

Як і в контексті задачі граничної керованості, такий результат вимагає, щоб $\omega$ задовольняла певні геометричні умови.

Метод HUM можна адаптувати до цієї задачі так.

Для заданих $\{\phi^0, \phi^1\} \in \mathcal{D}(\Omega) \times \mathcal{D}(\Omega)$ розглянемо систему
\begin{equation}\label{eqIII03}
    \begin{cases}
    \phi^{\prime\prime} - \Delta \phi = 0 \hspace{1.6cm} & \text{в} \hspace{0.3cm} Q
    \\
    \phi = 0 \hspace{0.5cm} & \text{на} \hspace{0.3cm} \Sigma
    \\
    \phi(0) = \phi^0,\ \phi^\prime(0) = \phi^1\hspace{0.5cm} & \text{в} \hspace{0.3cm} \Omega
    \end{cases}
\end{equation}
і потім розв'яжемо
\begin{equation}\label{eqIII04}
    \begin{cases}
    y^{\prime\prime} - \Delta y = -\phi \chi_\omega \hspace{1.6cm} & \text{в} \hspace{0.3cm} Q
    \\
    y = 0 \hspace{0.5cm} & \text{на} \hspace{0.3cm} \Sigma
    \\
    y(T) = 0,\ y^\prime(T) = 0 \hspace{0.5cm} & \text{в} \hspace{0.3cm} \Omega.
    \end{cases}
\end{equation}

Тоді визначимо оператор
\begin{equation}\label{eqIII05}
  \Lambda \{\phi^0, \phi^1\} = \{y^\prime(0),\ -y(0)\}.
\end{equation}

Множачи в~(\ref{eqIII04}) на $\phi$ та інтегруючи частинами, отримуємо
\begin{equation}\label{eqIII06}
  \langle \Lambda \{\phi^0, \phi^1\}, \{\phi^0, \phi^1\} \rangle =
  \int^T_0 \int_\omega |\phi|^2\, dx\, dt.
\end{equation}

Припустимо, що виконується такий результат єдиності:
\begin{equation}\label{eqIII07}
    \phi = 0 \hspace{0.2cm} \text{в} \hspace{0.2cm} \omega \times (0,T)
    \Rightarrow \phi^0 \equiv \phi^1 \equiv 0.
\end{equation}

Тоді
\begin{equation}\label{eqIII08}
    \|\{\phi^0, \phi^1\}\|_F = \left(\int^T_0 \int_\omega |\phi|^2\, dx\, dt \right)^{1/2}
\end{equation}
визначає норму в $\mathcal{D}(\Omega) \times \mathcal{D}(\Omega)$, де
\begin{equation}\label{eqIII09}
   F = \text{гільбертів простір} - \text{поповнення } \mathcal{D}(\Omega) \times \mathcal{D}(\Omega) \text{ за нормою } \|\cdot\|_F.
\end{equation}

З~(\ref{eqIII06}) випливає, що $\Lambda$ єдиним чином продовжується до ізоморфізму з $F$ у $F^\prime$.

Тому для заданих $\{y^0, y^1\}$ таких, що
$
\{y^1, -y^0\} \in F^\prime,
$
існує єдина пара $\{\phi^0, \phi^1\} \in F$ така, що
\begin{equation}\label{eqIII10}
    \Lambda \{\phi^0, \phi^1\} = \{y^1, -y^0\}.
\end{equation}

Зокрема, якщо застосовуємо керування $h = -\phi \in L^2(\omega \times (0,T))$, де $\phi$ — розв'язок~(\ref{eqIII03}) з початковими даними, що задовольняють~(\ref{eqIII10}), то розв'язок~(\ref{eqIII04}) задовольняє
$
y(0) = y^0,\ y^\prime(0) = y^1.
$



Теорема Гольмгрена гарантує, що для довільної непорожньої відкритої підмножини $\omega \subset \Omega$ існує $T_0(\omega, \Omega)$ таке, що якщо $T > T_0(\omega, \Omega)$, то виконується результат єдиності~(\ref{eqIII07}).

Тому ми показали, що якщо $T > T_0(\omega, \Omega)$, то система~(\ref{eqIII01}) керована в просторі початкових даних таких, що $\{y^1, -y^0\} \in F^\prime$, з керуваннями $h \in L^2(\omega \times (0,T))$.

Задача зводиться до знаходження достатніх умов на $\omega$ і $T$, щоб
$
F^\prime = L^2(\Omega) \times H^1_0(\Omega),
$
або, еквівалентно,
$
F = L^2(\Omega) \times H^{-1}(\Omega).
$

Отже, достатньо довести існування сталих $c>0,\ C>0$ таких, що
\begin{align}\label{eqIII11}
\begin{split}
   c \|\{\phi^0, \phi^1\}\|^2_{L^2(\Omega) \times H^{-1}(\Omega)} \leq \int^T_0 \int_\omega |\phi|^2\, dx\, dt \leq
   C \|\{\phi^0, \phi^1\}\|^2_{L^2(\Omega) \times H^{-1}(\Omega)},
   \\ \forall \{\phi^0, \phi^1\} \in \mathcal{D}(\Omega) \times \mathcal{D}(\Omega).
\end{split}
\end{align}


Добре відомо, що
$
\|\phi\|_{L^\infty(0, T; L^2(\Omega))} \leq
C \|\{\phi^0, \phi^1\}\|_{L^2(\Omega) \times H^{-1}(\Omega)},
$
звідки виводиться верхня оцінка в~(\ref{eqIII11}).

Залишається довести обернену нерівність спостережуваності виду
\begin{equation}\label{eqIII12}
    \|\{\phi^0, \phi^1\}\|_{L^2(\Omega) \times H^{-1}(\Omega)}
    \leq C \int^T_0 \int_\omega |\phi|^2\, dx\, dt,
\end{equation}
що буде предметом наступного розділу. Це вимагатиме накладення геометричних умов на спостережувальну підобласть $\omega$.

\section{Обернена нерівність}
\label{sec:III.2}
Нехай $x^0 \in \mathbb{R}^n$. Розглянемо розбиття межі $\{\Gamma(x^0), \Gamma_\star(x^0)\}$, введене в попередньому розділі.

Нехай $\omega \subset \Omega$ — окіл $\overline{\Gamma(x^0)}$, тобто для деякої відкритої множини $\Theta \subset \mathbb{R}^n$ такої, що
$
\overline{\Gamma(x^0)} \subset \Theta,
$
маємо
$
\omega = \Omega \cap \Theta.
$


\begin{figure}
\includegraphics[width=8cm]{fig02-chap03-sec3_2.eps}
\centering
\captionof{figure}{Геометрична конфігурація, у якій керування діє на околі $\omega$ множини $\Gamma(x^0)$.}
\end{figure}

У такому геометричному середовищі справджується наступний результат про керованість:
\begin{theorem}
\label{theorem:III.3.1}
Нехай $\Omega$ — обмежена область $\mathbb{R}^n$ з межею $\Gamma$ класу $C^2$. Нехай $x^0 \in \mathbb{R}^n$ та $\omega \subset \Omega$ — окіл $\overline{\Gamma(x^0)}$.

Якщо $T > T(x^0) = 2R(x^0)$, то існує стала $C = C(T) > 0$ така, що
\begin{equation} \label{eqIII13}
   \|\phi^0\|^2_{L^2(\Omega)} + \|\phi^1\|^2_{H^{-1}(\Omega)} \leq C \int^T_0 \int_\omega |\phi|^2\, dx\, dt
\end{equation}
для будь-якого розв'язку~(\ref{eqIII03}) з даними $\{\phi^0, \phi^1\} \in L^2(\Omega) \times H^{-1}(\Omega)$.
\end{theorem}

\label{proof:III.theorem.3.1}
\begin{proof}Доведення проводимо у кілька кроків.
\newline
\emph{Крок 1: Зведення до нерівності вищого порядку.}
Припустимо, що виконується оцінка
\begin{equation} \label{eqIII14}
   \|\phi^0\|^2_{H^1_{0}(\Omega)} + \|\phi^1\|^2_{L^2(\Omega)} \leq C \int^T_0 \int_\omega |\phi^\prime|^2\, dx\, dt.
\end{equation}
Метою є показати, що~(\ref{eqIII14}) тягне~(\ref{eqIII13}).

Для заданих $\{\phi^0, \phi^1\} \in L^2(\Omega) \times H^{-1}(\Omega)$ визначимо
\begin{equation}\label{eqIII15}
    X \in H^1_{0}(\Omega), \quad \Delta X = \phi^1 \text{ в } \Omega.
\end{equation}

Узявши розв'язок $\phi$ системи~(\ref{eqIII03}) з початковими даними $\{\phi^0, \phi^1\}$, визначимо
\begin{equation}\label{eqIII16}
    \psi(t) = \int^t_0 \phi(s)\, ds + X.
\end{equation}

Легко бачити, що $\psi^\prime = \phi$ і $\psi = \psi(x, t)$ розв'язує таку систему:
\begin{equation}\label{eqIII17}
    \begin{cases}
    \psi^{\prime\prime} - \Delta \psi = 0 \hspace{1.6cm} & \text{в} \hspace{0.3cm} Q
    \\
    \psi = 0 \hspace{0.5cm} & \text{на} \hspace{0.3cm} \Sigma
    \\
    \psi(0) = X,\ \psi^\prime(0) = \phi^0 & \text{в} \hspace{0.3cm} \Omega.
    \end{cases}
\end{equation}


Тепер застосовуємо нерівність~(\ref{eqIII14}) до $\psi$, щоб отримати
\begin{align} \label{eqIII18}
\begin{split}
   \|X\|^2_{H^1_{0}(\Omega)} + \|\phi^0\|^2_{L^2(\Omega)} \leq
   C \int^T_0 \int_\omega |\psi^\prime|^2\, dx\, dt = C \int^T_0 \int_\omega |\phi|^2\, dx\, dt.
\end{split}
\end{align}

З іншого боку,~(\ref{eqIII13}) виводиться з~(\ref{eqIII18}) завдяки класичним властивостям оператора Лапласа з умовами Діріхле, оскільки
\begin{equation}\label{eqIII19}
    \exists C > 0:\ \|\phi^1\|_{H^{-1}(\Omega)} \leq C \|X\|_{H^1_0}, \quad \forall \phi^1 \in H^{-1}(\Omega).
\end{equation}

Тепер ми готові довести~(\ref{eqIII14}) у кілька кроків.\\
% p60
\label{sec:III.2-c}
\emph{Крок 2: Оцінка нормальної похідної через локальний член.}
За теоремою~\ref{theorem:II.2.2} розділу~\ref{chapter02} показано, що якщо $T>T(x^0)$, то існує $C=C(T)>0$ таке, що
\begin{equation}\label{eqIII20}
    E_0 = \frac{1}{2} \int_\Omega \left[\left|\nabla \phi^0(x)\right|^2 + \left|\phi^1(x)\right|^2\right] dx \leq
    C \int^T_0 \int_{\Gamma(x^0)} \left|\frac{\partial \phi}{\partial \nu}\right|^2 d\Sigma.
\end{equation}

Нехай $\varepsilon > 0$ таке, що $\varepsilon < (T - T(x^0))/2$. Тоді з~(\ref{eqIII02}), інваріантності хвильового рівняння відносно зсувів за часом і збереження енергії випливає, що
\begin{equation}\label{eqIII21}
    E_0 \leq C \int^{T-\varepsilon}_\varepsilon \int_{\Gamma(x^0)} \left|\frac{\partial \phi}{\partial \nu}\right|^2 d\Sigma.
\end{equation}

%p61
Розглянемо векторне поле $q = q(x,t) \in (W^{1,\infty}(Q))^n$ таке, що
\begin{equation}\label{eqIII22}
\begin{cases}
q(x,t) = \nu(x) & \hspace{0.7cm} \forall(x,t) \in \Gamma(x^0) \times (\varepsilon, T-\varepsilon)
\\
q(x,t)\cdot \nu(x) \geq 0 & \hspace{0.7cm} \forall(x,t) \in \Gamma \times (0,T)
\\
q(x,0) = q(x,T) = 0 & \hspace{0.7cm} \forall x \in \Omega
\\
q(x,t) = 0 & \hspace{0.7cm} \forall(x,t) \in (\Omega \setminus \omega) \times (0,T).
\end{cases}
\end{equation}

Поле $q$ можна обрати у вигляді
$
q(x,t) = h(x) n(t),
$
де $n \in C^1([0,T])$ та $h = h(x) \in (W^{1, \infty}(\Omega))^n$ задовольняють
\begin{equation}\label{eqIII23}
    n(0) = n(T) = 0; \hspace{0.3cm} n(t) = 1 \hspace{0.3cm} \forall t \in (\varepsilon, T-\varepsilon),
\end{equation}
та
\begin{equation}\label{eqIII24}
    \begin{cases}
    h = \nu & \hspace{1.5cm} \text{на} \hspace{0.5cm} \Gamma(x^0)
    \\h \cdot \nu \geq 0 & \hspace{1.5cm} \text{на} \hspace{0.5cm} \Gamma
    \\ h = 0 & \hspace{1.5cm} \text{в} \hspace{0.5cm} \Omega \setminus \omega.
    \end{cases}
\end{equation}
(див. J.-L.~Lions~\cite{lions1988controlabilite}, гл.~I, розд.~3.1, для побудови цього векторного поля).

За тотожністю~(\ref{eqII18}) леми~\ref{lemma:II.2.1} з розділу~\ref{chapter02} отримуємо
\begin{align}\label{eqIII25}
\begin{split}
    \frac{1}{2} \int^{T-\varepsilon}_\varepsilon \int_{\Gamma(x^0)} \left|\frac{\partial \phi}{\partial \nu}\right|^2 d\Sigma \leq & \frac{1}{2} \int^{T-\varepsilon}_\varepsilon \int_\Gamma (q \cdot \nu) \left|\frac{\partial \phi}{\partial \nu}\right|^2 d\Sigma
    \\ \leq & C \int^{T}_0 \int_\omega \left(|\phi^\prime|^2 + |\nabla \phi|^2\right) dx\, dt,
\end{split}
\end{align}
де $C$ — додатна стала, що залежить від $\|q(x,t)\|_{W^{1, \infty}(Q)}$.

Оцінка~(\ref{eqIII25}) справджується для всіх $T > T(x^0)$. Тому, якщо $\varepsilon > 0$ таке, що $T - 2\varepsilon > T(x^0)$, за наведеним аргументом і поєднанням~(\ref{eqIII21}) та~(\ref{eqIII25}) виводимо
\begin{equation}\label{eqIII26}
    E_0 \leq C \int^{T-\varepsilon}_\varepsilon \int_\omega \left[|\phi^\prime|^2 + |\nabla \phi|^2\right] dx\, dt.
\end{equation}
% p62
\label{sec:III.2-d}
\emph{Крок 3: Локальний $\nabla\phi$-член.}
Нехай $\tilde{\omega}$ — окіл $\overline{\Gamma(x^0)}$ такий, що
\begin{equation}\label{eqIII27}
    \Omega \cap \overline{\tilde{\omega}} \subset \omega.
\end{equation}

Оскільки~(\ref{eqIII26}) виконується для будь-якого околу $\omega$ множини $\overline{\Gamma(x^0)}$, він виконується і для $\tilde{\omega}$, а отже
\begin{equation}\label{eqIII28}
    E_0 \leq C \int^{T-\varepsilon}_\varepsilon \int_{\tilde{\omega}} \left[|\phi^\prime|^2 + |\nabla \phi|^2\right] dx\, dt.
\end{equation}

Нехай $p = p(x,t) \in W^{1,\infty}(Q),\ p \geq 0$ — таке, що
\begin{equation}\label{eqIII29}
\begin{cases}
p(x,t) = 1 & \hspace{0.7cm} \forall(x,t) \in \tilde{\omega} \times (\varepsilon, T-\varepsilon)
\\
p(x,t) = 0 & \hspace{0.7cm} \forall(x,t) \in (\Omega \setminus \omega) \times (0,T)
\\
p(x,0) = p(x,T) = 0 & \hspace{0.7cm} \forall x \in \Omega
\\
|\nabla p|^2/p \in L^\infty(Q).
\end{cases}
\end{equation}
Функцію $p$ можна обрати у вигляді
\begin{equation}\label{eqIII30}
    p(x,t) = \rho^2(x) n(t),
\end{equation}
де $n \in C^1([0,T])$ та $\rho \in W^{1, \infty}(\Omega)$ задовольняє
\begin{equation}\label{eqIII31}
    \begin{cases}
    \rho(x) = 1 \hspace{2cm} \forall x \in \tilde{\omega}
    \\
    \rho(x) = 0 \hspace{2cm} \forall x \in \Omega \setminus \omega.
    \end{cases}
\end{equation}

Тепер, множачи рівняння~(\ref{eqIII03}) на $p\phi$ та інтегруючи частинами, отримуємо тотожність
\begin{equation}\label{eqIII32}
    \int_Q p |\nabla \phi|^2 = \int_Q p|\phi^\prime|^2
    - \int_\Omega p \phi^\prime \phi\bigg|^T_0
    + \int_Q p^\prime \phi^\prime \phi - \int_Q \nabla p \cdot \nabla \phi\, \phi.
\end{equation}

Враховуючи~(\ref{eqIII29}) і~(\ref{eqIII32}), отримуємо
\begin{equation}\label{eqIII33}
    \int_Q p |\nabla \phi|^2\, dx\, dt \leq \int^T_0 \int_\omega \left[|\phi^\prime|^2 + |\phi|^2\right] dx\, dt
    + \left|\int_Q (\nabla p \cdot \nabla \phi)\phi\right|.
\end{equation}

З іншого боку,
\begin{equation}\label{eqIII34}
    \left|\int_Q (\nabla p \cdot \nabla \phi)\phi\right| \leq
    \frac{1}{2} \int_Q p |\nabla \phi|^2 + \frac{1}{2} \int_Q \frac{|\nabla p|^2}{p} |\phi|^2.
\end{equation}

З~(\ref{eqIII33}) і~(\ref{eqIII34}), використовуючи~(\ref{eqIII29}), випливає
\begin{equation}\label{eqIII35}
    \int^{T-\varepsilon}_\varepsilon \int_\omega |\nabla \phi|^2 \leq C \int^T_0 \int_\omega \left[|\phi^\prime|^2 + |\phi|^2\right].
\end{equation}

Нарешті, поєднуємо~(\ref{eqIII28}) і~(\ref{eqIII35}):
\begin{equation}\label{eqIII36}
    E_0 \leq C \int^T_0 \int_{\omega} |\phi^\prime|^2\, dx\, dt + C \int_Q |\phi|^2\, dx\, dt.
\end{equation}
% p63
\label{sec:III.2-e}
\emph{Крок 4: $L^2$-член нижчого порядку.}
Щоб довести~(\ref{eqIII14}) (а отже й теорему), достатньо показати існування $C = C(T) > 0$ такого, що
\begin{equation}\label{eqIII37}
    \int_Q |\phi|^2\, dx\, dt \leq C \int^T_0 \int_\omega |\phi^\prime|^2\, dx\, dt
\end{equation}
виконується для кожного розв'язку~(\ref{eqIII03}).

Доводимо від супротивного, застосовуючи аргумент компактність–єдиність. Якщо~(\ref{eqIII37}) не виконується, існує послідовність розв'язків $\{\phi_n\}$ системи~(\ref{eqIII03}), що відповідає початковим даним $\{\phi^0_n, \phi^1_n\} \subset H^1_0(\Omega) \times L^2(\Omega)$, така, що
\begin{equation}\label{eqIII38}
    \int_Q |\phi_n|^2\, dx\, dt = 1 \hspace{0.5cm} \forall n \in \mathbb{N},
\end{equation}
\begin{equation}\label{eqIII39}
    \int^T_0 \int_\omega |\phi^{\prime}_n|^2\, dx\, dt \to 0,\ n \to \infty.
\end{equation}

З~(\ref{eqIII36}),~(\ref{eqIII38}) і~(\ref{eqIII39}) випливає, що $\{\phi^0_n, \phi^1_n\}$ обмежена в $H^1_0(\Omega) \times L^2(\Omega)$, тому
\begin{equation}\label{eqIII40}
    \{\phi_n\} \hspace{0.3cm} \text{обмежена в} \hspace{0.2cm} H^1(Q).
\end{equation}

З~(\ref{eqIII40}) і компактності вкладення $H^1(Q) \subset L^2(Q)$ маємо
\begin{equation}\label{eqIII41}
    \{\phi_n\} \hspace{0.3cm} \text{відносно компактна в} \hspace{0.2cm} L^2(Q).
\end{equation}

Виділяючи відповідну підпослідовність (яку для спрощення позначатимемо так само $\{\phi_n\}$), виводимо
\begin{equation}\label{eqIII42}
    \phi_n \to \phi \hspace{0.3cm} \text{слабко в} \hspace{0.2cm} H^1(Q),
\end{equation}
\begin{equation}\label{eqIII43}
    \phi_n \to \phi \hspace{0.3cm} \text{сильно в} \hspace{0.2cm} L^2(Q),
\end{equation}
де границя $\phi = \phi(x,t)$ — розв'язок~(\ref{eqIII03}). З іншого боку, з~(\ref{eqIII38}),~(\ref{eqIII39}),~(\ref{eqIII42}) і~(\ref{eqIII43}), $\phi$ задовольняє
\begin{equation}\label{eqIII44}
    \phi^\prime = 0 \hspace{0.3cm} \text{в} \hspace{0.3cm} \omega \times (0,T),
\end{equation}
\begin{equation}\label{eqIII45}
    \|\phi\|_{L^2(Q)} = 1.
\end{equation}

Зауважимо, що функція $\psi = \phi^\prime$ також є розв'язком~(\ref{eqIII03}), і вона задовольняє
\begin{equation}\label{eqIII46}
    \psi = 0 \hspace{0.3cm} \text{в} \hspace{0.3cm} \omega \times (0,T).
\end{equation}

Оскільки $T > T(x^0) \geq$ діаметр $\Omega$, з теореми єдиності Гольмгрена випливає
$
\psi \equiv 0
$ у $Q$.
Це означає, що
$
\phi = \phi(x)
$
— стаціонарний розв'язок~(\ref{eqIII03}). Звідси випливає, що
$
\phi \equiv 0,
$
що суперечить~(\ref{eqIII45}).
\end{proof}

\section{Основний результат}
\label{sec:III.3}

\begin{theorem}
\label{theorem:III.3.2}
Нехай $\Omega$ — обмежена область $\mathbb{R}^n$ з межею $\Gamma$ класу $C^2$. Нехай $x^0 \in \mathbb{R}^n$ і $\omega \subset \Omega$ — окіл $\overline{\Gamma(x^0)}$.

\end{theorem}


\begin{proof}
\label{proof:III.theorem.3.2}
Це безпосередній наслідок адаптації методу HUM, розвиненої у розділі~\ref{sec:III.1}, та оцінки спостережуваності~(\ref{eqIII13}), що означає, що збудований гільбертів простір $F$ збігається з $L^2(\Omega) \times H^{-1}(\Omega)$ — завдяки еквівалентності норм.
\end{proof}

% p66 III.4
\section{Змінні коефіцієнти в одновимірних областях}
\label{sec:III.4}
Розглянемо таке одновимірне хвильове рівняння зі змінним коефіцієнтом, залежним від просторової змінної:
\begin{equation}\label{eqIII47}
    \begin{cases}
    y^{\prime\prime} - (a(x) y_x)_x = h \chi_\omega \hspace{0.3cm} & \text{в} \hspace{0.3cm} \Omega \times (0,T)
    \\
    y(0,t) = y(1,t) = 0 \hspace{0.3cm} & \text{при} \hspace{0.3cm} t \in (0,T)
    \\
    y(x,0) = y^0(x),\ y^\prime(x,0) = y^1(x) \hspace{0.3cm} & \text{при} \hspace{0.3cm} x \in \Omega
    \end{cases}
\end{equation}
з $\Omega = (0,1)$ і $\omega = (l_1, l_2) \subset \Omega$.

Поєднуючи техніки попереднього розділу з техніками розділу~\ref{sec:II.7}, отримуємо такий результат.

\begin{theorem}
\label{theorem:III.3.3}
Припустимо, що коефіцієнт $a \in BV(0,1)$ задовольняє
\begin{equation}\label{eqIII48}
    a(x) \geq a_0 > 0, \hspace{0.5cm} \forall x \in (0,1).
\end{equation}

Тоді, якщо $T > 2R$ з $R = R(l_1, l_2) = \max\left(l_1/\sqrt{a_0},\ (1-l_2)/\sqrt{a_0}\right)$, то для кожної пари початкових даних $\{y^0, y^1\} \in H^1_0(0,1) \times L^2(0,1)$ існує керування $h \in L^2((l_1, l_2) \times (0,T))$ таке, що розв'язок $y = y(x,t)$ системи~(\ref{eqIII47}) задовольняє~(\ref{eqIII02}).
\end{theorem}


\begin{proof}
За методом HUM задача зводиться до отримання оцінки спостережуваності
\begin{equation}\label{eqIII49}
    \|\phi^0\|^2_{L^2(\Omega)} + \|\phi^1\|^2_{H^{-1}(\Omega)}
    \leq C \int^T_0 \int^{l_2}_{l_1} |\phi|^2\, dx\, dt
\end{equation}
для всіх розв'язків $\phi$ однорідної системи
\begin{equation}\label{eqIII50}
    \begin{cases}
    \phi^{\prime\prime} - (a(x) \phi_x)_x = 0 \hspace{0.3cm} & \text{в} \hspace{0.3cm} (0,1) \times (0,T)
    \\
    \phi(0,t) = \phi(1,t) = 0 \hspace{0.3cm} & \text{на} \hspace{0.2cm} (0,T)
    \\
    \phi(x,0) = \phi^0(x),\ \phi^\prime(x,0) = \phi^1(x) & \text{в} \hspace{0.2cm} (0,1).
    \end{cases}
\end{equation}

Діючи, як на кроці~1 доведення теореми~\ref{theorem:III.3.1}, спостерігаємо, що~(\ref{eqIII49}) є наслідком
\begin{equation}\label{eqIII51}
    \|\phi^0\|^2_{H^1_0(\Omega)} + \|\phi^1\|^2_{L^2(\Omega)}
    \leq C \int^T_0 \int^{l_2}_{l_1} |\phi^\prime|^2\, dx\, dt.
\end{equation}

Тепер~(\ref{eqIII51}) доводиться у кілька кроків.\\
\smallskip
% p67
\emph{Крок 1: Перша нерівність спостережуваності.}
Метод доведення оцінки спостережуваності~(\ref{eqII73}) з розділу~\ref{chapter02} дозволяє показати, що для будь-якого $\xi \in [0,1]$, якщо
\begin{equation}\label{eqIII52}
    T > \frac{2}{\sqrt{a_0}} \xi,
\end{equation}

\begin{equation}\label{eqIII53}
\begin{split}
    \int^\xi_0 \int^{T + \frac{x-\xi}{\sqrt{a_0}}}_{\frac{\xi-x}{\sqrt{a_0}}}
    \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dt\, dx
    \leq C \int^T_0 \left[|\phi^\prime(\xi,t)|^2 + a(\xi)|\phi_x(\xi,t)|^2\right] dt.
\end{split}
\end{equation}

Аналогічно, якщо
\begin{equation}\label{eqIII54}
    T > \frac{2}{\sqrt{a_0}} (1-\xi),
\end{equation}
маємо
\begin{equation}\label{eqIII55}
\begin{split}
    \int^\xi_0 \int^{T - \frac{x-\xi}{\sqrt{a_0}}}_{\frac{x-\xi}{\sqrt{a_0}}}
    \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dt\, dx
    \leq C \int^T_0 \left[|\phi^\prime(\xi,t)|^2 + a(\xi)|\phi_x(\xi,t)|^2\right] dt.
\end{split}
\end{equation}

Крім того, стала $C$ в обох оцінках~(\ref{eqIII53}) і~(\ref{eqIII55}) неперервно залежить від $\xi$ і $T$.

Оскільки $T>2R$, існує $\varepsilon >0$ таке, що
\begin{equation}\label{eqIII56}
    T > \frac{2}{\sqrt{a_0}} \max(l_1+\varepsilon,\ 1 + \varepsilon -l_2).
\end{equation}

З~(\ref{eqIII56}) виводимо, що~(\ref{eqIII53}) і~(\ref{eqIII55}) виконуються для всіх $\xi \in (l_1, l_1 + \varepsilon)$ і $\xi \in (l_2 - \varepsilon, l_2)$ відповідно. Інтегруючи нерівності~(\ref{eqIII53}) і~(\ref{eqIII55}) по $\xi \in (l_1, l_1 + \varepsilon)$ і $\xi \in (l_2 - \varepsilon, l_2)$ відповідно, безпосередньо отримуємо
\begin{equation}\label{eqIII57}
    \begin{split}
    \int^{l_1}_0 \int^{T + \frac{(x-l_1)}{\sqrt{a_0}}}_{\frac{l_1-x}{\sqrt{a_0}}}
    \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dt\, dx
    \leq C \int^T_0 \int^{l_1+\varepsilon}_{l_1}
    \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dx\, dt,
    \end{split}
\end{equation}
та
\begin{equation}\label{eqIII58}
\begin{split}
    \int^1_{l_2} \int^{T - \frac{(x-l_2)}{\sqrt{a_0}}}_{\frac{x-l_2}{\sqrt{a_0}}}
    \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dt\, dx
    \leq C \int^T_0 \int^{l_2}_{l_2-\varepsilon} \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dx\, dt.
\end{split}
\end{equation}

Тепер, використовуючи~(\ref{eqIII57}) і~(\ref{eqIII58}), маємо
\begin{equation}\label{eqIII59}
    \begin{split}
    & (T-2R) \int^1_0 \left[|\phi^1(x)|^2 + a(x)|\phi^0_x(x)|^2\right] dx
    \\ & = \int^{T-R}_{R} \int^1_0
    \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dx\, dt
    \leq C \int^T_0 \int^{l_2}_{l_1} \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dx\, dt,
    \end{split}
\end{equation}
або, еквівалентно,
\begin{equation}\label{eqIII60}
    \|\phi^0\|^2_{H^1_0(0,1)} + \|\phi^1\|^2_{L^2(\Omega)}  \leq C \int^T_0 \int^{l_2}_{l_1} \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dx\, dt.
\end{equation}
% p69
\label{sec:III.4-d}
\emph{Крок 2: Покращена нерівність спостережуваності.}
Оцінка~(\ref{eqIII60}) справджується для всіх $(l_1, l_2) \subset (0,1)$ при $T > 2R(l_1, l_2)$.

Беручи $\varepsilon >0$ достатньо малим, щоб
$
T - 2\varepsilon > 2R(l_1 - \varepsilon, l_2 - \varepsilon),
$
маємо
\begin{equation}\label{eqIII61}
    \begin{split}
    \|\phi^0\|^2_{H^1_0(0,1)} + \|\phi^1\|^2_{L^2(0,1)}
    \leq C \int^{T-\varepsilon}_{\varepsilon} \int^{l_2-\varepsilon}_{l_1+\varepsilon}
    \left[|\phi^\prime|^2 + a(x)|\phi_x|^2\right] dx\, dt.
    \end{split}
\end{equation}

Аргумент кроку~3 доведення теореми~\ref{theorem:III.3.1} легко дозволяє довести, що
\begin{equation}\label{eqIII62}
    \begin{split}
    \int^{T-\varepsilon}_{\varepsilon} \int^{l_2-\varepsilon}_{l_1+\varepsilon}
    a(x) |\phi_x|^2\, dx\, dt
    \leq C \int^T_0 \int^{l_2}_{l_1} |\phi^\prime|^2\, dx\, dt + C\|\phi\|^{2}_{L^2(Q)}.
    \end{split}
\end{equation}

Тепер, поєднуючи~(\ref{eqIII61}) і~(\ref{eqIII62}), отримуємо
\begin{equation}\label{eqIII63}
    \begin{split}
    \|\phi^0\|^2_{H^1_{0}(0,1)} + \|\phi^1\|^2_{L^2(0,1)}
    \leq C \int^T_0 \int^{l_2}_{l_1} |\phi^\prime|^2\, dx\, dt + C\|\phi\|^{2}_{L^2(Q)}.
    \end{split}
\end{equation}

Отже, достатньо довести нерівність
\begin{equation}\label{eqIII64}
    \|\phi\|^{2}_{L^2(Q)} \leq C \int^T_0 \int^{l_2}_{l_1} |\phi^\prime|^2\, dx\, dt.
\end{equation}
% p70
\label{sec:III.4-e}
\emph{Крок 3: Компактність–єдиність.}
Щоб довести~(\ref{eqIII64}), діємо від супротивного, як у доведенні кроку~4 теореми~\ref{theorem:III.3.1}, використовуючи принцип компактності–єдиності.

Якщо~(\ref{eqIII64}) не виконується, то існує розв'язок $\phi = \phi(x,t)$ системи~(\ref{eqIII50}) такий, що
\begin{equation}\label{eqIII65}
    \|\phi\|_{L^2(Q)} = 1,
\end{equation}

\begin{equation}\label{eqIII66}
    \phi^\prime(x,t) = 0, \hspace{0.5cm} \forall (x,t) \in (l_1, l_2) \times (0,T).
\end{equation}

Тому $\psi = \phi^\prime$ — розв'язок~(\ref{eqIII50}), що задовольняє
\begin{equation}\label{eqIII67}
    \psi \in C([0,T]; L^2(0,1))
\end{equation}
та
\begin{equation}\label{eqIII68}
    \psi = 0 \hspace{0.3cm} \text{в} \hspace{0.3cm} (l_1, l_2) \times (0,T).
\end{equation}

З~(\ref{eqIII68}) випливає, що
$
\psi^\prime = 0$ в $(l_1, l_2) \times (0,T)$.
Це разом з~(\ref{eqIII67}) і~(\ref{eqIII63}) дозволяє виснувати, що
$
\psi(0) \in H^1_0(0,1)$, $\psi^\prime(0) \in L^2(0,1),
$
а отже
\begin{equation}\label{eqIII69}
    \psi \in C([0,T]; H^1_0(0,1)) \cap C^1([0,T]; L^2(0,1)).
\end{equation}

Оскільки $\psi$ належить класу розв'язків зі скінченною енергією у~(\ref{eqIII69}) (факт, який не був відомий заздалегідь, бо $\psi = \phi^\prime$ і, у принципі, на одну похідну менш регулярний за $\phi$), оцінки~(\ref{eqIII53}) і~(\ref{eqIII55}) застосовні до $\psi$, і з~(\ref{eqIII68}) випливає
\begin{equation*}
    \psi(x,t) = 0, \hspace{0.3cm} \forall (x,t) \in (0,1) \times (R,T-R),
\end{equation*}
з $R = R(l_1, l_2)$ як у формулюванні теореми~\ref{theorem:III.3.3}, і завдяки збереженню енергії маємо $\psi \equiv 0$.

Тому $\phi = \phi(x)$ — стаціонарний розв'язок~(\ref{eqIII50}), що задовольняє
\begin{equation*}
    -(a(x) \phi_x)_x = 0 \hspace{0.3cm} \text{в} \hspace{0.3cm} (0,1), \hspace{0.3cm} \phi \in H^1_0(0,1).
\end{equation*}
Але тоді обов'язково $\phi \equiv 0$, що суперечить~(\ref{eqIII65}).
\end{proof}


% p72 III.5
\section{Коментарі}
\label{sec:III.5}
\begin{enumerate}
\item \textbf{Хвильові рівняння зі змінними коефіцієнтами.} Як видно з теорем~\ref{theorem:III.3.1} і~\ref{theorem:III.3.2}, гранична керованість тягне керованість з керуваннями, носіями яких є окіл межі.

Цей тип аргументу легко поширюється на рівняння зі змінними коефіцієнтами виду
\begin{equation}\label{eqIII70}
    \begin{cases}
    y^{\prime\prime} - \operatorname{div}(a(x) \nabla y) = h \chi_\omega \hspace{1.6cm} & \text{в} \hspace{0.3cm} Q
    \\
    y = 0 \hspace{0.5cm} & \text{на} \hspace{0.3cm} \Sigma
    \\
    y(0) = y^0,\ y^\prime(0) = y^1 & \text{в} \hspace{0.3cm} \Omega
    \end{cases}
\end{equation}
з $a \in W^{1, \infty}(\Omega)$ таким, що
\begin{equation}\label{eqIII71}
a(x) \geq a_0 > 0 \hspace{0.5cm} \forall x \in \Omega.
\end{equation}
Припустимо, що коефіцієнт $a=a(x)$ такий, що для деякої відкритої підмножини $\Gamma_0 \subset \Gamma$ межі і $T_0 > 0$ виконується оцінка
\begin{equation}\label{eqIII72}
    \|\phi^0\|^2_{H^1_0(\Omega)} + \|\phi^1\|^2_{L^2(\Omega)}
    \leq C \int_{\Sigma_0} \left|\frac{\partial \phi}{\partial \nu}\right|^2 d\Sigma
\end{equation}
для всіх $T > T_0$, де $\Sigma_0 = \Gamma_0 \times (0,T)$, для будь-якого розв'язку
\begin{equation}\label{eqIII73}
    \begin{cases}
    \phi^{\prime\prime} - \operatorname{div}(a(x) \nabla \phi) = 0 \hspace{1.6cm} & \text{в} \hspace{0.3cm} Q
    \\
    \phi = 0 \hspace{0.5cm} & \text{на} \hspace{0.3cm} \Sigma
    \\
    \phi(0) = \phi^0,\ \phi^\prime(0) = \phi^1 \hspace{0.5cm} & \text{в} \hspace{0.3cm} \Omega.
    \end{cases}
\end{equation}
Нагадаймо, що така нерівність граничної спостережуваності вимагає певних структурних умов на коефіцієнт $a=a(x)$, що забезпечують досягнення всіма променями геометричної оптики межі $\Gamma$ і, конкретніше, $\Gamma_0$.

Виконується такий результат:

% p73
\begin{theorem}
\label{theorem:III.3.4}
Нехай $\Omega$ — обмежена область $\mathbb{R}^n$ з межею $\Gamma$ класу $C^2$. Нехай коефіцієнт $a=a(x)$, $\Gamma_0 \subset \Gamma$ та $T_0 > 0$ такі, що для всіх $T>T_0$ виконується оцінка~(\ref{eqIII72}).

Припустимо, що $\omega$ — окіл $\overline{\Gamma_0}$ в $\Omega$. Тоді, якщо $T>T_0$, то для кожних початкових даних $\{y^0, y^1\} \in H^1_{0}(\Omega) \times L^2(\Omega)$ існує керування $h \in L^2(\omega \times (0,T))$ таке, що розв'язок $y = y(x,t)$ системи~(\ref{eqIII70}) задовольняє~(\ref{eqIII02}).
\end{theorem}

% p.73
\begin{proof}
\label{proof:III.theorem.3.4}
Подамо короткий ескіз.

Метод HUM зводить доведення цієї теореми до отримання оцінки~(\ref{eqIII13}) для розв'язків~(\ref{eqIII73}).
Як на кроці~1 доведення теореми~\ref{theorem:III.3.1}, спостерігаємо, що достатньо довести~(\ref{eqIII14}).
Аргументи кроків~2 і~3 доведення теореми~\ref{theorem:III.3.1} дозволяють легко отримати~(\ref{eqIII36}).
Тому задача знову зводиться до доведення~(\ref{eqIII37}).
Аргумент кроку~4 доведення теореми~\ref{theorem:III.3.1} буквально не застосовний у цьому випадку, оскільки теорема Гольмгрена справджується лише для рівнянь з аналітичними коефіцієнтами.

Однак існують два альтернативні доведення~(\ref{eqIII37}), що застосовні у цьому випадку:
\begin{enumerate}
\item Метод, уведений V.~Komornik~\cite{komornik1988methode}, заснований на використанні негармонічних рядів Фур'є. Цей метод систематично застосовується до еволюційних рівнянь типу
$
\phi^{\prime\prime} + A \phi = 0,
$
де $A$ — самоспряжений еліптичний оператор порядку $\geq 2$, і дозволяє ``поглинути'' члени нижчого порядку в оцінках типу~(\ref{eqIII36}).

\item Аргументи C.~Bardos, G.~Lebeau і J.~Rauch~\cite{bardos1992sharp} зводять~(\ref{eqIII37}) до властивості єдиного продовження для розв'язків еліптичної задачі типу
\begin{equation}\label{eqIII74}
    -\operatorname{div}(a(x) \nabla \phi) + \lambda\phi = 0 \hspace{0.3cm} \text{в} \hspace{0.3cm} \Omega,
\end{equation}
де $\lambda \in \mathbb{C}$ і $\phi = \phi(x)$ — комплекснозначна функція.

Тоді питання зводиться до того, щоб показати, що кожен розв'язок $\phi \in H^2(\Omega)$ рівняння~(\ref{eqIII74}), що задовольняє
$
\phi = 0$ в $\omega$,
обов'язково обнуляється всюди, що є безпосереднім наслідком результатів про єдине продовження для еліптичних рівнянь, отриманих за допомогою нерівностей Карлемана (див.~\cite{hormander2007analysis}). Це — тепер уже класичний результат про єдине продовження власних функцій еліптичних рівнянь, який можна досягти і значно поширити за допомогою нерівностей Карлемана.
\end{enumerate}
\end{proof}

Слід зробити кілька зауважень і геометричних спостережень:
\begin{enumerate}
\item \textbf{Геометричні зауваження.} Теорема~\ref{theorem:III.3.3} показує, що одновимірне хвильове рівняння точно кероване в $H^1_{0}(\Omega) \times L^2(\Omega)$ з керуваннями в $L^2(\omega \times (0,T))$ для довільної непорожньої відкритої підмножини $\omega$ області $\Omega$.

Це суто одновимірний результат, який не узагальнюється на розмірності $n>1$.

Посилаємось на A.~Haraux~\cite{haraux1988controlabilite} та J.~Lagnese~\cite{lagnese1983control} для аналізу внутрішньої керованості хвильового рівняння зі сталим коефіцієнтом за допомогою рядів Фур'є, коли $\Omega$ — квадрат або диск $\mathbb{R}^2$ відповідно.

Наприклад, спектральний аналіз у~\cite{lagnese1983control} показує, що якщо $\Omega = B(0,1) \subset \mathbb{R}^n$ і $\omega = B(0,r)$ з $r<1$, то хвильове рівняння не є точно керованим в $H^1_{0}(\Omega) \times L^2(\Omega)$ з керуваннями $L^2(\omega \times (0,T))$ для будь-якого $T>0$. Це через явище ``шепочучих галерей'' (whispering gallery), тобто існування періодичних променів геометричної оптики, що нескінченно довго захоплені в області $r < |x| < 1$ і ніколи не досягають керуючої області $\omega$. У~\cite{lagnese1983control} показано, що клас керованих початкових даних з $L^2$-керуваннями менший за енергетичний простір $H^1_{0}(\Omega) \times L^2(\Omega)$, але містить, наприклад, усі дані зі скінченною кількістю ненульових коефіцієнтів Фур'є.

Результати в~\cite{haraux1988controlabilite} показують, що коли $\overline{\omega} \subset \Omega$ (а отже GCC не виконано), то для керування початковими даними в $H^1_{0}(\Omega) \times L^2(\Omega)$ необхідно використовувати керування в класі аналітичних функціоналів з носіями в $\omega \times (0,T)$.

Однак, як ми бачили у розділі~\ref{sec:III.1}, завдяки методу HUM і теоремі єдиності Гольмгрена для будь-якої непорожньої відкритої підмножини $\omega \subset \Omega$ існує початковий час $T_0 = T_0(\omega, \Omega) >0$ такий, що якщо $T>T_0$, то точна керованість~(\ref{eqIII01}) з керуваннями в $L^2(\omega \times (0,T))$ виконується у просторі початкових даних таких, що $\{y^1, -y^0\} \in F^\prime$. Характеризація простору $F^\prime$ (або, еквівалентно, $F$), коли $F \neq L^2(\Omega) \times H^{-1}(\Omega)$, значною мірою є відкритою задачею. Посилаємось на~\cite{lebeau1992analytique},~\cite{robbiano1995fonction} та~\cite{allibert1999analytic} для подальших розробок у цьому контексті.

Як зазначено у розділах~\ref{sec:II.6} і~\ref{sec:II.7}, оцінка~(\ref{eqIII72}) виконується для розв'язків~(\ref{eqIII73}), якщо $\Omega$ гладка класу $C^\infty$, $a \in C^\infty(\overline{\Omega})$ і пара $\{\Gamma_0, T\}$ задовольняє GCC, тобто кожен промінь геометричної оптики перетинає $\Gamma_0$ у недифракційній точці за час менший за $T$ (пор.~\cite{bardos1992sharp}). Однак для виконання~(\ref{eqIII72}) коефіцієнт $a(x)$ повинен задовольняти геометричні умови, що забезпечують досягнення всіма променями геометричної оптики зовнішньої межі за рівномірний час. Якщо деякі з цих променів захоплені всередині $\Omega$ і ніколи не досягають її зовнішньої межі, то нерівність граничної спостережуваності~(\ref{eqIII72}) не виконується, і теорема~\ref{theorem:III.3.4} вище марна. Такі патологічні приклади побудовано й обговорено у~\cite{macia2002zuazua}. Зауважимо, що для хвильових рівнянь зі змінними коефіцієнтами промені вже не є прямими лініями, і на практиці може бути важко перевірити, чи всі вони досягають $\omega$ за рівномірний час.



\item \textbf{Керування мінімальної норми.} Аргумент розділу~\ref{sec:II.7} дозволяє довести: якщо точна керованість в $H^1_{0}(\Omega) \times L^2(\Omega)$ з керуваннями в $L^2(\omega \times (0,T))$ виконується, то для кожних $\{y^0, y^1\} \in H^1_{0}(\Omega) \times L^2(\Omega)$ множина допустимих керувань, визначена як
$
U_{ad} = \{h \in L^2(\omega \times (0,T)):\ y$ — розв'язок~(\ref{eqIII70}), що задовольняє~(\ref{eqIII02})$\},
$
містить нескінченно багато елементів.

Керування $h$, що його дає метод HUM, є єдиним оптимальним керуванням мінімальної норми, яке задовольняє
\begin{equation*}
    \|h\|^2_{L^2(\omega \times (0,T))} = \min_{g \in U_{ad}} \|g\|^2_{L^2(\omega \times (0,T))}.
\end{equation*}


\item \textbf{Оптимальний час керування.} Теорема~\ref{theorem:III.3.2} показує, що якщо $\omega$ — окіл $\Gamma(x^0)$ і $T > T(x^0)$, то хвильове рівняння точно кероване в $H^1_{0}(\Omega) \times L^2(\Omega)$ з керуваннями в $L^2(\omega \times (0,T))$.

Цей результат не оптимальний за часом керованості. Дійсно, завдяки~\cite{bardos1992sharp}, ми знаємо, що час керованості зменшується зі збільшенням розміру $\omega$. Оцінки часу керування покращуються при застосуванні аргументів обрізання (cutting-off) і роботи в області $\Omega \setminus \omega$, доповненні $\omega$ в $\Omega$.

Зауважимо, що якщо $\omega = \Omega$, точна керованість виконується за довільно малий час (пор. J.-L.~Lions~\cite{lions1988controlabilite}, розд.~VII.2.2, де доведено існування керувань $h \in L^2(Q)$). Робота на області $\Omega \setminus \omega$ дозволяє довести, що час керованості прямує до нуля, коли $\omega$ збільшено наближається до всієї $\Omega$ (див.~\cite{kim1992exact}).


\item \textbf{Сингулярна границя до граничного керування.} Розглянемо послідовність відкритих множин виду
$
\omega_\varepsilon = \{x \in \Omega:\ d(x, \Gamma(x^0)) < \varepsilon\},
$
де $d(x, \Gamma(x^0))$ — відстань від $x$ у $\mathbb{R}^n$ до множини $\Gamma(x^0)$. Це окіл ``ширини'' $\varepsilon$ навколо $\Gamma(x^0)$.

Для заданого $T>T(x^0)$ можна побудувати керування $h_\varepsilon \in L^2(\omega_\varepsilon \times (0,T))$ так, що система~(\ref{eqIII01}) збігається до хвильового рівняння з граничним керуванням Діріхле (див.~\cite{fabre1990comportement},~\cite{fabre1990equation}).


\item \textbf{Інші моделі.} Метод доведення теореми~\ref{theorem:III.3.1} можна застосовувати у ширшому контексті, зокрема, для хвильових рівнянь з іншими граничними умовами або навіть для моделей пластини (див.~\cite{zuazua1988controlabilite}).

Наприклад, для системи
\begin{equation}\label{eqIII75}
    \begin{cases}
    y^{\prime\prime} + \Delta^2 y = h \chi_\omega \hspace{1.6cm} & \text{в} \hspace{0.3cm} Q
    \\
    y = \partial y / \partial \nu = 0 \hspace{0.5cm} & \text{на} \hspace{0.3cm} \Sigma
    \\
    y(0) = y^0,\ y^\prime(0) = y^1 \hspace{0.5cm} & \text{в} \hspace{0.3cm} \Omega
    \end{cases}
\end{equation}
виконується таке: \emph{``Нехай $\Omega$ — обмежена область $\mathbb{R}^n$ з межею класу $C^3$ і $\omega$ — окіл підмножини межі типу $\overline{\Gamma(x^0)}$. Тоді для всіх $T>0$ і $\{y^0, y^1\} \in H^2_{0}(\Omega) \times L^2(\Omega)$ існує керування $h \in L^2(\omega \times (0,T))$ таке, що розв'язок~(\ref{eqIII75}) задовольняє~(\ref{eqIII02})''.}

Це аналог попереднього результату для хвильового рівняння, але цього разу він справджується за довільно малий час.

У~\cite{machtyngier1990control} техніки цього розділу адаптовано для дослідження внутрішньої керованості рівняння Шредінгера.

\item \textbf{Ряди Фур'є та моделі пластини.} Зауважимо, однак, що цей метод застосовний лише тоді, коли носій керування $\omega$ є околом відповідної підмножини межі. Для розгляду інших керуючих підмножин $\omega$ потрібна суттєва подальша робота. Наприклад, у~\cite{jaffard1988controle} показано методами рядів Фур'є, що якщо $\Omega$ — квадрат $\mathbb{R}^2$, то наведене вище рівняння пластини можна точно керувати в $H^2_{0}(\Omega) \times L^2(\Omega)$ керуваннями в $L^2(\omega \times (0,T))$, де $\omega$ строго міститься в підмножині $\Omega$, тобто $\overline{\omega} \subset \Omega$. Цей результат істотно відрізняється від раніше згаданих щодо хвильового рівняння. Дійсно, у цьому випадку властивість керованості виконується, навіть якщо носій керувань $\omega$ не задовольняє геометричну умову керування. Це показує, що GCC є достатньою, але не необхідною умовою для керування рівнянням пластини.
\end{enumerate}
\end{enumerate}
