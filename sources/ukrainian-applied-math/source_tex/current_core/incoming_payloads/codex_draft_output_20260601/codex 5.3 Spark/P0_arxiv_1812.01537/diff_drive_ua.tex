\section{Самокалібрування диференціального приводу}
\label{sec:diff_drive}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Огляд}

Модель диференціального приводу складається з двох керованих коліс на одній осі, по одному з кожного боку бази робота,
з початковою системою координат у центрі цієї осі.
Робот параметризується
радіусами коліс $(r_l, r_r)$ і довжиною осі $d$.
До кожного з цих параметрів прив'язано коефіцієнт корекції так, що
$\bfc=[c_l~c_r~c_d]\tr$ є вектором параметрів внутрішньої калібровки.
%
Модельний рух зазвичай вимірюють за допомогою енкодерів коліс, які подають кумулятивні кути повороту коліс $
\bfy=[\delta\psi_l, \delta\psi_r]+\bfn$
на кожному часовому кроці $
\delta t$, де $
\bfn$ — адитивний гаусівський шум.


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Визначення станів і дельт}
%
Визначимо стани положення та кута орієнтації
$\bfx=(\bfp,\theta)$, що виступають компактним поданням $SE(2)$.
Відповідно, прирости стану або дельти $\D_{ij}$ і $\delta_k$ також належать до $SE(2)$.


%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Передінтегрування дельти}

Для зручності задаємо тілесні величини
$\bfb=(\delta l,\delta \theta)=f_b(\bfy,\bfc,\bfn)$ як
%
%
\begin{align}
\label{equ:diff_drive_psis}
\begin{split}
\delta l &= \tfrac12(r_r c_r \delta\psi_r + r_l c_l\delta\psi_l)  \\
\delta \theta &= \tfrac{1}{D}(r_r c_r \delta\psi_r - r_l c_l\delta\psi_l) 
~.
\end{split}
\end{align}
%
з $D=d\,c_d$.
Їхні компоненти, відповідно, виділяють спільний (вздовж траєкторії) і диференціальний (поворотний) складники в імпульсах руху коліс.

За припущенням сталого керування протягом інтервалу дискретизації $[t_j,t_k]$ робот
рухається вздовж дуги кола радіуса
%
\begin{align}
R = \frac{\delta l}{\delta \theta} 
~.
\end{align}
%
Приріст руху $\delta_k$ можна записати в $SE(2)$ як
%
\begin{align}
\label{equ:diff_drive_exact_inte}
\delta x &= R\sin(\delta\theta) \notag\\
\delta y &= R(1-\cos(\delta\theta)) \\
\delta\theta &= \delta \theta \notag 
~,
\end{align}
%
що точно дорівнює $\delta=\Exp(\tau)\in SE(2)$ з $\tau=[\delta l,0,\delta \theta]\in\se(2)$, тобто за відсутності бічного ковзання коліс.
%
У разі, коли робот рухається прямолінійно, маємо $\delta \theta\to0$,
і $R\to+\infty$.
Цей випадок обробляється наближенням %using a mid-point integration:
%
\begin{align}
\label{equ:diff_drive_runge_kutta_inte}
%
%
\delta x &= \delta l \cos(\delta\theta/2) \notag \\
%
\delta y &= \delta l \sin(\delta\theta/2) \\
%
\delta\theta &= \delta \theta \notag 
~.
\end{align}


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%\subsection{Інтеграція руху}

Інтеграція дельти — це просто композиція в $SE(2)$,
%
\begin{align}\label{equ:diff_drive_composition}
\begin{split}
\Dp_{ik} &= \Dp_{ij} + \D\bfR_{ij}\delta \bfp_k \\
\D\theta_{ik} &= \D\theta_{ij} + \delta\theta_k
\end{split}
\end{align}
%
де $\D\bfR_{ij}=\Exp(\D\theta_{ij})$ — матриця повороту, відповідна куту повороту $\D\theta_{ij}$ (див.
\eqRef{equ:R_SO2}), а $\delta \bfp_k=(\delta x,\delta y)$ — вектор дельти переносу для $\delta_k$.

\subsection{Матриці Якобі}
\label{sec:diff_drive_jacs}
%
\subsubsection{Якобіан складників руху}
%
З 
\eqRef{equ:diff_drive_psis} маємо
%
\begin{subequations}
\label{equ:diff_drive_psis_jac}
\begin{align}
\jac{\bfb}{\bfy} = \jac{\bfb}{\bfn} &= \begin{bmatrix}
\frac12 r_l c_l & \frac12 r_r c_r  \\
- r_l c_l       & r_r c_r
\end{bmatrix}
&&\in \bbR^{2\tcross2}
\\
\jac{\bfb}{\bfc} &= \begin{bmatrix}
\frac12 \psi_{l} r_l & \frac12 \psi_{r} r_r & 0\\
-\psi_{l} r_l        & \psi_{r} r_r & -\frac{\delta\theta}{c_d}
\end{bmatrix}
&&\in \bbR^{2\tcross3}
\end{align}
\end{subequations}



\subsubsection{Якобіани поточного дельта}

З 
\eqRef{equ:diff_drive_exact_inte},
%
\begin{align}\label{equ:diff_drive_exact_inte_jac}
\jac{\delta_k}{\bfb} &= 
\begin{bmatrix}
\frac{\sin(\delta\theta)}{\delta\theta} 
&
R\big(\cos(\delta\theta)-\frac{\sin(\delta\theta)}{\delta\theta}\big)\\
\frac{1-cos(\delta\theta)}{\delta\theta} 
&
R\big(\sin(\delta\theta)-\frac{1-\cos(\delta\theta)}{\delta\theta}\big)\\
0 & 1
\end{bmatrix}
\in \bbR^{3\tcross2}
\end{align}
%
Аналогічно, з 
\eqRef{equ:diff_drive_runge_kutta_inte},
%
\begin{align}\label{equ:diff_drive_runge_kutta_meas_jac}
\jac{\delta_k}{\bfb} =
\begin{bmatrix}
\cos(\delta\theta/2) &-\frac12\delta l\sin(\delta\theta/2)\\
\sin(\delta\theta/2) &\phantom{-}\frac12\delta l\cos(\delta\theta/2)\\
0 & 1
\end{bmatrix}
\in \bbR^{3\tcross2}
\end{align}
%
\subsubsection{Якобіани композиції дельти}
%
З 
\eqRef{equ:diff_drive_composition}, % is in $SE(2)$
%
\begin{subequations}\label{equ:jac_composition_DD}
\begin{align}
\jac{\D_{ik}}{\D_{ij}} &= \begin{bmatrix}
\bfI & \DR_{ij}\hatx{1}\delta\bfp_k \\
\bf0 & 1
\end{bmatrix}
&&\in \bbR^{3\tcross3}
\\
\jac{\D_{ik}}{\delta_k} &= \begin{bmatrix}
\DR_{ij} & \bf0 \\
\bf0        & 1
\end{bmatrix}
&&\in \bbR^{3\tcross3}
\end{align}
\end{subequations}
%
з $\hatx{1}=\begin{bsmallmatrix}0 & -1 \\ 1 & 0
\end{bsmallmatrix}$, див. 
\appRef{sec:derivatives_SO2}.
%
%where $\DR_{ij}$ is the rotation matrix delta corresponding to the rotation angle of $\D_{ij}$
%and $\delta \bfp_k$ is the translation vector delta corresponding to $\delta_k$.

\subsection{Інтегрування коваріації дельти та Якобіана}

Ми використовуємо загальну процедуру, отримуючи $\bfQ_k$ з 
\eqRef{equ:Q_d}.
Тут $\bfQ_s$ — невелика шумова діагональна коваріація, що враховує ковзання коліс, а $\bfQ_n$ визначається як
%
\begin{align}
\label{equ:diff_drive_meas_cov}
\bfQ_n &=
\begin{bmatrix}
\sigma^2_{\psi_l}+\alpha^2 & 0\\
0 & \sigma^2_{\psi_r}+\alpha^2
\end{bmatrix}, \\
\sigma_{\psi_l}^2&=k_l \delta\psi_l,~~\sigma_{\psi_r}^2=k_r \delta\psi_r,~~\alpha=\tfrac12(\mu_l+\mu_r)  \notag
~,
\end{align}
%
