\par\bigskip\pdfbookmark[0]{On the Connection. Fragment IV}{nexus-fragment-IV}\centerline{[IV.]}\medskip
Let $(n)$ denote the number of values $x$ in the collection
\[
0,1,2,3,4\dotsfour p^n-1
\]
for which $xx-D=xx-ap^\mu$ is divisible by $p^n$.
\[
\begin{array}{c@{\qquad}cc}
1)\ \mu\text{ odd, e.g. }=7.&\multicolumn2c{2)\ \mu\text{ even, e.g. }=6}\\
(1)=1&aNp&aRp\\
(2)=p&(1)=1&(1)=1\\
(3)=p&(2)=p&(2)=p\\
(4)=pp&(3)=p&(3)=p\\
(5)=pp&(4)=pp&(4)=pp
\end{array}
\]

\SourceLeaf{294}
\[
\begin{array}{c@{\qquad}c@{\qquad}c}
(6)=p^3&(5)=pp&(5)=pp\\
(7)=p^3&(6)=p^3&(6)=p^3\\
(8)=0&(7)=0&(7)=2p^3\\
(9)=0&(8)=0&(8)=2p^3\\
\text{etc.}&\text{etc.}&\text{etc.}
\end{array}
\]

\[
\begin{array}{c@{\qquad}c}
\text{Now put}&\text{Then}\\[.4em]
(1)-\dfrac{(2)}p=(1)'&fp=(1)'\\[.4em]
(2)-\dfrac{(3)}p=(2)'&fpp=1+(2)'\\[.4em]
(3)-\dfrac{(4)}p=(3)'&fp^3=(1)'+(3)'\\[.4em]
(4)-\dfrac{(5)}p=(4)'&fp^4=1+(2)'+(4)'\\
\text{etc.}&\text{etc.}
\end{array}
\]
Consequently, putting $\frac{p-1}p(1+\frac{fp}p+\frac{fpp}{pp}+\frac{fp^3}{p^3}+\text{etc.})=T$, we have
\[
\begin{aligned}
\frac{p+1}p T&=1+\frac{(1)'}p+\frac{(2)'}{pp}+\frac{(3)'}{p^3}+\frac{(4)'}{p^4}+\text{etc.}\\
&=1+\frac{(1)}p=1+\frac1p
\end{aligned}
\]
Thus $T=1$
