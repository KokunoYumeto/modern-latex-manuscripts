\AGMArticle{25}
[Aus den Funktionalgleichungen für $P,\allowbreak Q,\allowbreak R,\allowbreak S$, die bei der Verwandlung des zweiten Arguments $\eta$ in seinen Kehrwert, bei dessen Vorzeichenwechsel und bei dessen Multiplikation mit dem ersten Argument $y$ gelten, sowie aus den bekannten Werten der Funktionen für $\eta=1$ oder auch aus der in Artikel 21 für die allgemeinen Funktionen aufgestellten partiellen Differentialgleichung folgt: Wenn sich $P,\allowbreak Q,\allowbreak R,\allowbreak S$ in Reihen nach wachsenden ganzzahligen Potenzen von $y^{\frac14}$ und $\eta^{\frac12}$ entwickeln lassen, müssen diese
\Needspace{30\baselineskip}
\[
\begin{aligned}
P(y,\eta)&=1+y(\eta+\eta^{-1})\\[4pt]&\quad{}+y^4(\eta^2+\eta^{-2})\\[4pt]&\quad{}+y^9(\eta^3+\eta^{-3})\\[4pt]&\quad{}+y^{16}(\eta^4+\eta^{-4})+\,+\,.\\[10pt]
Q(y,\eta)&=1-y(\eta+\eta^{-1})\\[4pt]&\quad{}+y^4(\eta^2+\eta^{-2})\\[4pt]&\quad{}-y^9(\eta^3+\eta^{-3})\\[4pt]&\quad{}+y^{16}(\eta^4+\eta^{-4})-\,+\,.\\[10pt]
R(y,\eta)&=y^{\frac14}(\eta^{\frac12}+\eta^{-\frac12})+y^{\frac94}(\eta^{\frac32}+\eta^{-\frac32})\\[4pt]&\quad{}+y^{\frac{25}4}(\eta^{\frac52}+\eta^{-\frac52})\\[4pt]&\quad{}+y^{\frac{49}4}(\eta^{\frac72}+\eta^{-\frac72})+\,+\,.\\[10pt]
S(y,\eta)&=y^{\frac14}(\eta^{\frac12}-\eta^{-\frac12})-y^{\frac94}(\eta^{\frac32}-\eta^{-\frac32})\\[4pt]&\quad{}+y^{\frac{25}4}(\eta^{\frac52}-\eta^{-\frac52})\\[4pt]&\quad{}-y^{\frac{49}4}(\eta^{\frac72}-\eta^{-\frac72})+\,-\,.
\end{aligned}
\]
sein.
