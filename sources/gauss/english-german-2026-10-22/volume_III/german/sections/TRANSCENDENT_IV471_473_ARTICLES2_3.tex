\fancyhead[L]{\small HUNDERT SÄTZE ÜBER DIE NEUEN TRANSZENDENTEN.}

\Needspace{23\baselineskip}
\par\medskip\Needspace{5\baselineskip}\pdfbookmark[1]{Art. 2}{transscendentes-iv-art-2}\centerline{2.}\nobreak\smallskip
Wenn wir $T$ in folgender Form schreiben
\[
\begin{gathered}
1+\frac{1-a^n}{1-a}\,t
 \\[8pt]\quad{}+\frac{1-a^n}{1-a}.\frac{1-a^{n-1}}{1-aa}.att
 \\[8pt]\quad{}+\frac{1-a^n}{1-a}.\frac{1-a^{n-1}}{1-aa}.
  \frac{1-a^{n-2}}{1-a^3}\,a^3t^3
\\[8pt]
\qquad\\[8pt]\quad{}+\frac{1-a^n}{1-a}.\frac{1-a^{n-1}}{1-aa}.
  \frac{1-a^{n-2}}{1-a^3}.\frac{1-a^{n-3}}{1-a^4}.a^6.t^4
 +\text{etc.}
\end{gathered}
\]
worin die Exponenten von $a$ die Dreieckszahlen sein werden, so ist ersichtlich,
dass das letzte Glied sein wird
\[
a^{\frac12(nn-n)}t^n=y^n
\]

\Needspace{26\baselineskip}
wenn wir $a^{\frac12(n-1)}t=y$ setzen. Die ganze Reihe wird dann,
indem wir das letzte Glied mit dem ersten, das vorletzte mit dem zweiten
usw. zusammenfassen, für ein gerades $n$
\[
\begin{gathered}
(1+y^n)+\frac{1-a^n}{1-a}\,t(1+y^{n-2})
 \\[8pt]\quad{}+\frac{1-a^n}{1-a}.\frac{1-a^{n-1}}{1-aa}\,at(1+y^{n-4})+.\ .
\\[8pt]
\qquad+\frac{1-a^n}{1-a}.\frac{1-a^{n-1}}{1-aa}..\frac{1-a^{\frac12n+3}}{1-a^{\frac12n-1}}
 a^{\frac12(\frac12n-1)(\frac12n-2)}t^{\frac12n-1}(1+yy)
\\[8pt]
\qquad+\frac{1-a^n}{1-a}.\frac{1-a^{n-1}}{1-aa}..\frac{1-a^{\frac12n+1}}{1-a^{\frac12n}}
 a^{\frac12.\frac12n.\frac12n-1}t^{\frac12n}
\end{gathered}
\]

\Needspace{19\baselineskip}
wobei das mittlere Glied isoliert stehen bleibt. Bezeichnen wir es
mit $A$ und setzen $a=xx$, so wird die Reihe
\[
\begin{gathered}
A\bigl\{1+\frac{1-x^n}{1-x^{n+2}}\,x(y+y^{-1})
 \\[8pt]\quad{}+\frac{1-x^n}{1-x^{n+2}}.\frac{1-x^{n-2}}{1-x^{n+4}}\,x^4(yy+y^{-2})
\\[8pt]
+\frac{1-x^n}{1-x^{n+2}}.\frac{1-x^{n-2}}{1-x^{n+4}}.
 \frac{1-x^{n-4}}{1-x^{n+6}}\,x^9(y^3+y^{-3})+\mathinner{.\,.}\bigr\}
\end{gathered}
\]
usw.; diese Reihe besteht aus $\frac12(n+2)$ Gliedern und bricht dann ab.


\Needspace{12\baselineskip}
Unser Produkt $(1+t)(1+at)(1+aat)..(1+a^{n-1}t)$ hingegen geht über in



\SourceLeaf{472}

\[
(1+\frac{y}{x^{n-1}})(1+\frac{y}{x^{n-3}})(1+\frac{y}{x^{n-5}})\mathinner{.\,.\,.}(1+x^{n-1}y)
\]

\Needspace{14\baselineskip}
Das Produkt der ersten Hälfte der Faktoren
\[
\begin{gathered}=\frac{y^{\frac12n}}{x^{\frac14nn}}.(1+\frac{x^{n-1}}y)(1+\frac{x^{n-3}}y)\\[8pt]\quad{}(1+\frac{x^{n-5}}y)\mathinner{.\,.}(1+\frac{x^3}y)(1+\frac{2}{y}\end{gathered}
\]
wozu noch die übrigen hinzukommen
\[
(1+xy)(1+x^3y(1+x^5y)\mathinner{.\,.\,.}(1+x^{n-1}y)
\]

\Needspace{18\baselineskip}
Da nun
\[
\begin{gathered}A=\frac{1-x^{n+2}}{1-xx}.\frac{1-x^{n+4}}{1-x^4}.\frac{1-x^{n+6}}{1-x^6}\mathinner{.\,.}\frac{1-x^{2n}}{1-x^n}\\[8pt]\quad{}.\frac{y^{\frac12n}}{x^{\frac14nn}}\end{gathered}
\]

\Needspace{32\baselineskip}
gilt, so geht der erste \textsc{Satz} in den folgenden \textsc{zweiten} über: \textit{für ein gerades $n$ gilt:}
\[
\begin{gathered}
1+\frac{1-x^n}{1-x^{n+2}}.x(y+\frac1y)
\\[8pt]\quad{}+\frac{1-x^n}{1-x^{n+2}}.\frac{1-x^{n-2}}{1-x^{n+4}}.x^4(yy+\frac1{yy})\\[8pt]
\quad\\[8pt]\quad{}+\frac{1-x^n}{1-x^{n+2}}.\frac{1-x^{n-2}}{1-x^{n+4}}.\frac{1-x^{n-4}}{1-x^{n+6}}.x^9(y^3+y^{-3})+\text{etc.}\\[8pt]
=(1+xy)(1+x^3y(1+x^5y)\mathinner{.\,.\,.}(1+x^{n-1}y)
\\[8pt]\quad{}(1+\frac xy)(1+\frac{x^3}y)(1+\frac{x^5}y)\mathinner{.\,.\,.}(1+\frac{x^{n-1}}y)\\[8pt]
\quad\times\frac{1-xx}{1-x^{n+2}}.\frac{1-x^4}{1-x^{n+4}}.\frac{1-x^6}{1-x^{n+6}}\mathinner{.\,.}\frac{1-x^n}{1-x^{2n}}
\end{gathered}
\]

\Needspace{29\baselineskip}
\par\medskip\Needspace{5\baselineskip}\pdfbookmark[1]{Art. 3}{transscendentes-iv-art-3}\centerline{3.}\nobreak\smallskip
Ist $n$ ungerade, so hat die Reihe folgende Form:
\[
\begin{gathered}
(1+y^n)+\frac{1-a^n}{1-a}t(1+y^{n-2})
\\[8pt]\quad{}+\frac{1-a^n}{1-a}.\frac{1-a^{n-1}}{1-aa}att(1+y^{n-4})+\mathinner{.\,.}\\[8pt]
\quad+\frac{1-a^n}{1-a}.\frac{1-a^{n-1}}{1-aa}\mathinner{.\,.}\frac{1-a^{\frac12(n+3)}}{1-a^{\frac12(n-3)}}\\[8pt]\quad{}a^{\frac18(n-3)(n-5)}t^{\frac12(n-3)}(1+y^3)\\[8pt]
\quad+\frac{1-a^n}{1-a}.\frac{1-a^{n-1}}{1-aa}\mathinner{.\,.}\frac{1-a^{\frac12(n+3)}}{1-a^{\frac12(n-1)}}\\[8pt]\quad{}a^{\frac12.\frac12(n-1)\frac12(n-3)}t^{\frac12(n-1)}(1+y)
\end{gathered}
\]

\Needspace{17\baselineskip}
Setzen wir wie zuvor $a=xx$ und das Glied, das hier das letzte ist, $=Bx^{\frac14}.(y^{\frac12}+y^{-\frac12})$, so wird die Reihe
\[
\begin{gathered}
=B\{x^{\frac14}(y^{\frac12}+y^{-\frac12})
\\[8pt]\quad{}+\frac{1-x^{n-1}}{1-x^{n+3}}x^{\frac94}(y^{\frac32}+y^{-\frac32})\\[8pt]
+\frac{1-x^{n-1}}{1-x^{n+3}}.\frac{1-x^{n-3}}{1-x^{n+5}}x^{\frac{25}4}(y^{\frac52}+y^{-\frac52})+\text{etc.}\}
\end{gathered}
\]

\SourceLeaf{473}



\Needspace{14\baselineskip}
Diese Reihe besteht aus $\frac12(n+1)$ Gliedern und bricht dann ab. Von unserem Produkt stellen wir die ersten $\frac12(n-1)$ Faktoren folgendermaßen dar
\[
\begin{gathered}\frac{y^{\frac12(n-1)}}{x^{\frac14(nn-1)}}.(1+\frac{x^{n-1}}y)(1+\frac{x^{n-3}}y)\\[8pt]\quad{}(1+\frac{x^{n-5}}y)\mathinner{.\,.}(1+\frac{xx}y)\end{gathered}
\]
wozu noch hinzukommt
\[
(1+y)(1+xxy)(1+x^4y)\mathinner{.\,.\,.}(1+x^{n-1}y)
\]

\Needspace{23\baselineskip}
Da nun
\[
\begin{gathered}
B=\frac{1-x^{n+3}}{1-xx}.\frac{1-x^{n+5}}{1-x^4}.\frac{1-x^{n+7}}{1-x^6}\mathinner{.\,.}\frac{1-x^{2n}}{1-x^{n-1}}.
\\[8pt]\quad{}\frac{x^{\frac14nn-n+\frac34}.y^{\frac12(n-1)}y^{\frac12}}{x^{\frac12nn-n+\frac12}.x^{\frac14}}\\[8pt]
=\frac{1-x^{n+3}}{1-xx}\mathinner{.\,.\,.}\frac{1-x^{2n}}{1-x^{n-1}}.\frac{y^{\frac12n}}{x^{\frac14nn}}
\end{gathered}
\]

\Needspace{29\baselineskip}
ergibt der erste \textsc{Satz} den folgenden \textsc{dritten}: \textit{für ein ungerades $n$ gilt}
\[
\begin{gathered}
x^{\frac14}(y^{\frac12}+y^{-\frac12})
\\[8pt]\quad{}+\frac{1-x^{n-1}}{1-x^{n+3}}.x^{\frac94}(y^{\frac32}+y^{-\frac32})
\\[8pt]\quad{}+\frac{1-x^{n-1}}{1-x^{n+3}}.x^{\frac{25}4}(y^{\frac52}+y^{-\frac52})+\text{etc.}\\[8pt]
=x^{\frac14}(y^{\frac12}+y^{-\frac12})(1+xxy)(1+x^4y)(1+x^6y)\mathinner{.\,.\,.}(1+x^{n-1}y)\\[8pt]
\quad\times(1+\frac{xx}y)\quad(1+\frac{x^4}y)\quad(1+\frac{x^6}y)\mathinner{.\,.\,.\,.}(1+\frac{x^{n-1}}y)\\[8pt]
\quad\times\frac{1-xx}{1-x^{n+3}}.\frac{1-x^4}{1-x^{n+5}}.\frac{1-x^6}{1-x^{n+7}}\mathinner{.\,.\,.\,.\,.}\frac{1-x^{n-1}}{1-x^{2n}}
\end{gathered}
\]
