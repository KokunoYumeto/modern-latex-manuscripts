\[
\begin{gathered}
\cos 1=\sin 2\, .\,\sin 5\\[8pt]
\cos 2=\sin 3\, .\,\sin 1\\[8pt]
\cos 3=\sin 4\, .\,\sin 2\\[8pt]
\cos 4=\sin 5\, .\,\sin 3\\[8pt]
\cos 5=\sin 1\, .\,\sin 4
\end{gathered}
\]
\[
\begin{gathered}
1=\cos 1\, .\,\operatorname{tang}3\, .\,\operatorname{tang}4\\[8pt]
 =\cos 2\, .\,\operatorname{tang}4\, .\,\operatorname{tang}5\\[8pt]
 =\cos 3\, .\,\operatorname{tang}5\, .\,\operatorname{tang}1\\[8pt]
 =\cos 4\, .\,\operatorname{tang}1\, .\,\operatorname{tang}2\\[8pt]
 =\cos 5\, .\,\operatorname{tang}2\, .\,\operatorname{tang}3
\end{gathered}
\]

\Needspace{23\baselineskip}
\[
\begin{array}{r@{\qquad}rcl}
8.&-\mathrm{d}1&=&\mathrm{d}2\, .\,\cos 4+\mathrm{d}5\, .\,\cos 3\\[8pt]
9.&-\mathrm{d}2&=&\mathrm{d}3\, .\,\cos 5+\mathrm{d}1\, .\,\cos 4\\[8pt]
10.&-\mathrm{d}3&=&\mathrm{d}4\, .\,\cos 1+\mathrm{d}2\, .\,\cos 5\\[8pt]
11.&-\mathrm{d}4&=&\mathrm{d}5\, .\,\cos 2+\mathrm{d}3\, .\,\cos 1\\[8pt]
12.&-\mathrm{d}5&=&\mathrm{d}1\, .\,\cos 3+\mathrm{d}4\, .\,\cos 2
\end{array}
\]
Eliminating $\mathrm{d}5$ from $8$ and $12$ gives
\[
\mathrm{d}1\, .\,\sin 3^2=-\mathrm{d}2\, .\,\cos 4+\mathrm{d}4\, .\,\cos 2\, .\,\cos 3
\]
Eliminating $\mathrm{d}3$ from $9$ and $10$, on the other hand, gives
\[
\mathrm{d}2\, .\,\sin 5^2=-\mathrm{d}1\, .\,\cos 4+\mathrm{d}4\, .\,\cos 1\, .\,\cos 5
\]
