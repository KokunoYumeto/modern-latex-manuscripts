\pdfbookmark[1]{Art. 1}{transscendentes-v-art-1}\begin{center}[1.]\par\emph{Allgemeines Theorem} (1827 Aug. 6)\end{center}
\[
\begin{gathered}P(x,ty).P\Bigl(x,\frac yt\Bigr)=P(xx,tt).P(xx,yy)\\[8pt]\quad{}+R(xx,tt).R(xx,yy)\end{gathered}
\]

\par\medskip\Needspace{5\baselineskip}\pdfbookmark[1]{Art. 2}{transscendentes-v-art-2}\centerline{[2.]}\nobreak\smallskip
Durch Zerlegung in Factoren bestätigt sich leicht,
\[
(1+x)(1+x^3)(1+x^5)(1+x^7)\ .\ .\ .\ .=M
\]
gesetzt:
\[
\begin{gathered}
P(ix^{\frac32},\quad ix^{\frac12})=\frac{P(x,1)}{M}\\[8pt]
P(ix^{\frac32},-ix^{\frac12})=\frac{P(x^3,1)}{M}
\end{gathered}
\]
also durch Addition
\[
\begin{gathered}
P(x,1)+P(x^3,1)=2M.P(x^6,-x)\\[8pt]
P(x,1)-P(x^3,1)=2Mx.P(x^6,-x^5)
\end{gathered}
\]
also
\[
\begin{gathered}
P(x,1)^2-P(x^3,1)^2
\\[8pt]\quad{}=4MMx.F(x^{12}).(1-x)(1-x^5)\\[8pt]\quad{}(1-x^7)(1-x^{11})(1-x^{13})\ .\ .\ .\\[8pt]
\\[8pt]\quad{}=4MMx.\frac{Fx.F(x^6).F(x^{12})}{Fxx.F(x^3)}
\\[8pt]\quad{}=2\surd\frac{pP}{r}\ R^{\frac32}
\end{gathered}
\]

\[
\begin{gathered}
P(x,xt)=\frac{1}{x^{\frac14}\surd t}R(x,t)\\[8pt]
R(x,xt)=\frac{1}{x^{\frac14}\surd t}P(x,t)
\end{gathered}
\]
\[
\begin{gathered}P(x,1)^2.P(x,y)^2=Q(x,1)^2.Q(x,y)^2\\[8pt]\quad{}+R(x,1)^2.R(x,y)^2\end{gathered}
\]
\[
\begin{gathered}
P(x,y).Q(x,y)=Q(xx,1).Q(xx,yy)\\[8pt]
P(x,y)^2+Q(x,y)^2=2P(xx,1).P(xx,yy)\\[8pt]
P(x,y)^2-Q(x,y)^2=2R(xx,1).R(xx,yy)
\end{gathered}
\]
\[
\begin{gathered}P(x,y).P(xx,yy)=P(x^6,1).P(x^3,y^3)
\\[8pt]\quad{}+\tfrac12\{P(x^{\frac23},1)-P(x^6,1)\}.
\{P(x^{\frac13},y)-P(x^3,y^3)\}\end{gathered}
\]

\end{gathered}

\[

pqr^3.P(x^3,y^3)=pqR.P(x,y)^3\\[8pt]\quad{}+3PQR.P(x,y).Q(x,y)^2\\[8pt]

\end{gathered}
\]
\[
\begin{gathered}
3pPQR=r^3Q-q^3R\\[8pt]
3qPQR=r^3P-p^3R\\[8pt]
3rPQR=p^3Q-q^3P
\end{gathered}
\]
\[
3PPQQ=PPqq+ppQQ+ppqq
\]
\[
\begin{gathered}
3pP=\frac{r^3}{R}-\frac{q^3}{Q}=\frac{3Q^3}{q}-\frac{3R^3}{r}\\[8pt]

3rR=\frac{p^3}{P}-\frac{q^3}{Q}=\frac{3Q^3}{q}-\frac{3P^3}{p}
\end{gathered}

\[

\par\medskip\Needspace{5\baselineskip}\pdfbookmark[1]{Art. 4}{transscendentes-v-art-4}\centerline{[4.]}\nobreak\smallskip
So wie
\[
x^{\frac14}(y^{\frac12}+y^{-\frac12})+x^{\frac94}(y^{\frac32}+y^{-\frac32})+.\ +\ .\ .\ .
\]
durch $R(x,y)$ bezeichnet ist, so wollen wir noch
\[
x^{\frac14}(y^{\frac12}-y^{-\frac12})-x^{\frac94}(y^{\frac32}-y^{-\frac32})+.\ -\ .\ .\ .
\]
durch $S(x,y)$ bezeichnen. Man hat dann
\[
iS(x,y)=R(x,-y)
\]
\[
\begin{gathered}
S(x,y)=\frac{\surd[Q(x,1)^2.P(x,y)^2-P(x,1)^2.Q(x,y)^2]}{R(x,1)}\\[8pt]
=\frac{\surd[P(x,1)^2.R(x,y)^2-R(x,1)^2.P(x,y)^2]}{Q(x,1)}
\end{gathered}
\]
\[
\begin{gathered}
R(x,y).S(x,y)=Q(xx,1).S(xx,yy)\\[8pt]
R(x,y)^2+S(x,y)^2=2P(xx,1).R(xx,yy)\\[8pt]
R(x,y)^2-S(x,y)^2=2R(xx,1).P(xx,yy)
\end{gathered}
\]
\[
\begin{gathered}
R(x,y).R(x,z)+S(x,y).S(x,z)\\[8pt]\quad{}=2R(xx,yz).P\Bigl(xx,\frac yz\Bigr)\\[8pt]
R(x,y).R(x,z)-S(x,y).S(x,z)\\[8pt]\quad{}=2R\Bigl(xx,\frac yz\Bigr).P(xx,yz)\\[8pt]
R(x,y).S(x,z)+S(x,y).R(x,z)\\[8pt]\quad{}=2S(xx,yz).Q\Bigl(xx,\frac yz\Bigr)\\[8pt]
R(x,y).S(x,z)-S(x,y).R(x,z)\\[8pt]\quad{}=2S\Bigl(xx,\frac zy\Bigr).Q(xx,yz)
\end{gathered}
\]
Setzt man
\[
\Sigma e^{-\theta(k+\alpha)^2}=(\theta,\alpha)
\]
wo $k$ alle ganzen Zahlen bedeutet, so sind die Theoreme enthalten in
\[
\begin{gathered}(\theta,2\alpha).(\theta,2\beta)
\\[8pt]\quad{}=(2\theta,\alpha+\beta).(2\theta,\alpha-\beta)
\\[8pt]\quad{}+(2\theta,\alpha+\beta+\tfrac12).(2\theta,\alpha-\beta+\tfrac12)\end{gathered}
\]
