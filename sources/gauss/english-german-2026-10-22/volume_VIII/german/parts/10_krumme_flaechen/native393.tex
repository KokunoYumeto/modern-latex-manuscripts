\noindent\hspace*{1.2em}Die Bestimmenden [eines Punktes $x,y,z$] der Fläche [seien] $t,\theta$ [und die Differentiale der Koordinaten $x,y,z$]:
\begin{equation}\tag*{[1]}
\left\{\begin{aligned}
dx&=a\,dt+\alpha\,d\theta\\
dy&=b\,dt+\beta\,d\theta\\
dz&=c\,dt+\gamma\,d\theta
\end{aligned}\right.
\end{equation}
\begin{equation}\tag*{[2]}
dx^2+dy^2+dz^2=P\,dt^2+2Q\,dt\,d\theta+R\,d\theta^2.
\end{equation}
[Bezeichnen ferner] $X,Y,Z$ [die] Richtungskosinus der Normalen, so gilt]
\begin{equation}\tag*{[3]}
\left\{\begin{aligned}
\sigma X&=b\gamma-c\beta\\
\sigma Y&=c\alpha-a\gamma\\
\sigma Z&=a\beta-b\alpha.
\end{aligned}\right.
\end{equation}
