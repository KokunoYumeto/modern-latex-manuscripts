\noindent Wenn
\[
\boldsymbol{P}\varphi=s,\qquad \boldsymbol{P}(\varphi+2\psi)=c,
\]
\noindent so ist
\[
s\,s\,c\,c=4s+4c,\qquad c=\frac{2(1+\sqrt{1+s^3})}{s\,s},
\]
\[
\boldsymbol{P}(3\psi+\varphi+\varphi')=
\frac{2+s s'(s+s')-\tfrac12(c s s-2)(c' s' s'-2)}{(s-s')^2},
\]
\[
\boldsymbol{P}(3\psi+\varphi-\varphi')=
\frac{2+s s'(s+s')+\tfrac12(c s s-2)(c' s' s'-2)}{(s-s')^2}.
\]
