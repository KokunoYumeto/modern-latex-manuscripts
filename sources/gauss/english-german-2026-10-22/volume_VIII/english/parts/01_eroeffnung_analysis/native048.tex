\frac{1}{2^{2m-1}}\,{}^{2m}\mathfrak{B}^{(m+1)}\frac{1}{2}\cos\varphi

\[
\frac{1}{2^{2m-1}}\frac{{}^{2m}\mathfrak{B}^{(m)}}{2}\cos\varphi,

\[
\frac{1}{2^{2m}}\,{}^{2m+1}\mathfrak{B}^{(m+1)}\cos\varphi.
