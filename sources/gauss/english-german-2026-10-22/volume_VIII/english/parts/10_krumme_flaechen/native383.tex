\noindent and differentiation of
\[
aA+bB+cC=0
\]
gives
\[
Aa''+Bb''+Cc''=-aA''-bB''-cC'',
\]
hence
\[
aA''+bB''+cC''=\alpha A'+\beta B'+\gamma C'
\]
or
\[
0=a(A''+BC'-CB')+b(B''+CA'-AC')+c(C''+AB'-BA'),
\]
which, combined with
\[
0=aA+bB+cC
\]
readily gives the following:
\begin{equation}\tag*{[12]}
\left\{\begin{aligned}
na&=A'+CB''-BC''\\
nb&=B'+AC''-CA''\\
nc&=C'+BA''-AB'',
\end{aligned}\right.
\end{equation}
\begin{equation}\tag*{[13]}
\left\{\begin{aligned}
n\alpha&=A''+BC'-CB'\\
n\beta&=B''+CA'-AC'\\
n\gamma&=C''+AB'-BA',
\end{aligned}\right.
\end{equation}
where
{\small
\begin{equation}\tag*{[14]}
\begin{aligned}
mmnn={}&A'A'+B'B'+C'C'+A''A''+B''B''+C''C''\\
&+2A(B'C''-C'B'')+2B(C'A''-A'C'')+2C(A'B''-B'A'').
\end{aligned}
\end{equation}
}
\indent Moreover, differentiating $a\alpha+b\beta+c\gamma=0$ gives:
\[
a\alpha'+b\beta'+c\gamma'+\alpha a'+\beta b'+\gamma c'=0
\]
or
\[
aa''+bb''+cc''+\alpha a'+\beta b'+\gamma c'=0
\]
or
\[
mm''+\alpha a'+\beta b'+\gamma c'=0.
\]
This, combined with
\[
\begin{aligned}
aa'+bb'+cc'&=mm'\\
Aa'+Bb'+Cc'&=-aA'-bB'-cC'
\end{aligned}
\]
