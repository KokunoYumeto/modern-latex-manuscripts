\noindent [Previously we set:]
\[
\begin{aligned}
aa+bb+cc&=P\\
a\alpha+b\beta+c\gamma&=Q\\
\alpha\alpha+\beta\beta+\gamma\gamma&=R,
\end{aligned}
\]
[hence:]
\[
\frac{\partial Q}{\partial t}=a'\alpha+b'\beta+c'\gamma+\tfrac12\frac{\partial P}{\partial\theta}.
\]

[If we now set once and for all
\[
\begin{aligned}
\frac{\partial P}{\partial t}&=P',&\frac{\partial P}{\partial\theta}&=P'',\\
\frac{\partial Q}{\partial t}&=Q',&\frac{\partial Q}{\partial\theta}&=Q'',\\
\frac{\partial R}{\partial t}&=R',&\frac{\partial R}{\partial\theta}&=R'',
\end{aligned}
\]
we obtain:
\[
\sigma s^3.\frac{\tau}{s}\sin\psi=P(Q'-\tfrac12P'')-\tfrac12QP'.
\]
Now
\[
\frac{\tau}{s}\sin\psi.s\,du=\tau\sin\psi\,du
\]
is the differential of the lateral curvature, so it follows that]
\[
(P(Q'-\tfrac12P'')-\tfrac12QP')dt
\]
equals $\sigma ss$ times the differential of the lateral curvature[. But we had]
\[
\begin{aligned}
\sigma\sigma&=PR-QQ\\
ss&=P;
\end{aligned}
\]
[so that we finally obtain the formula:]
\begin{equation}\tag*{[14]}
\text{Diff. of lat. curv.}=\frac{(PQ'-\tfrac12PP''-\tfrac12QP')dt}{\BareRoot{(PR-QQ)}.P}.
\end{equation}
