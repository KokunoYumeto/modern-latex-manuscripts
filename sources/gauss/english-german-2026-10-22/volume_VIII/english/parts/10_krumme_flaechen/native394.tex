\begin{equation}\tag*{[4]}
\left\{\begin{aligned}
\frac{dx}{du}&=s\xi\\
\frac{dy}{du}&=s\eta\\
\frac{dz}{du}&=s\zeta;
\end{aligned}\right.
\end{equation}
[here $s$ is determined by the formula:
\begin{equation}\tag*{[5]}
ss=P\left(\frac{dt}{du}\right)^2+2Q\frac{dt}{du}\frac{d\theta}{du}+R\left(\frac{d\theta}{du}\right)^2.
\end{equation}
Second, we have:]
\begin{equation}\tag*{[6]}
\left\{\begin{aligned}
\frac{d\xi}{du}&=\tau\xi'\\
\frac{d\eta}{du}&=\tau\eta'\\
\frac{d\zeta}{du}&=\tau\zeta';
\end{aligned}\right.
\end{equation}
[here $\tau$ is determined by the formula:
\begin{equation}\tag*{[7]}
\tau\tau=\left(\frac{d\xi}{du}\right)^2+\left(\frac{d\eta}{du}\right)^2+\left(\frac{d\zeta}{du}\right)^2.
\end{equation}
Third, we have:]
\begin{equation}\tag*{[8]}
\left\{\begin{aligned}
\eta\zeta'-\zeta\eta'&=\xi''\\
\zeta\xi'-\xi\zeta'&=\eta''\\
\xi\eta'-\eta\xi'&=\zeta''.
\end{aligned}\right.
\end{equation}
\par Then:
\begin{equation}\tag*{[9]}
\left\{\begin{aligned}
X\xi+Y\eta+Z\zeta&=0\\
X\xi'+Y\eta'+Z\zeta'&=-\cos\psi\\
X\xi''+Y\eta''+Z\zeta''&=+\sin\psi.
\end{aligned}\right.
\end{equation}

[With this notation we have]
\begin{center}
\begin{tabular}{ll}
$s\,du$&element of the curve\\[3pt]
$\tau\,du$&element of absolute curvature\\[3pt]
$\frac{\tau}{s}$&measure of absolute curvature\\[3pt]
$\frac{\tau}{s}\cos\psi$&measure of normal curvature (convex towards $P$)\\[3pt]
$\frac{\tau}{s}\sin\psi$&measure of lateral curvature.
\end{tabular}
