\pdfbookmark[2]{Handwritten remark on spherical and plane triangles}{carnot-handschriftlich-sphaerisch}
[Handwritten remark on reducing the relations between the elements of a spherical triangle to the relations between the elements of plane triangles:]
\begin{center}\textit{Spherical triangle}\end{center}
\[
\begin{array}{cc}\text{Angles}&\text{Sides}\\ A&a\\B&b\\C&c\end{array}
\]
\Needspace{7\baselineskip}
\begin{center}\textit{Plane triangles}\end{center}
\[
\begin{array}{ll}
A&\sin\tfrac12a\\
90^0-\tfrac12A+\tfrac12B-\tfrac12C&\sin\tfrac12b\cos\tfrac12c\\
90^0-\tfrac12A-\tfrac12B+\tfrac12C&\sin\tfrac12c\cos\tfrac12b
\end{array}
\]
\[
\begin{array}{ll}
180^0-A&\cos\tfrac12a\\
\tfrac12A+\tfrac12B+\tfrac12C-90^0&\sin\tfrac12b\sin\tfrac12c\\
90^0+\tfrac12A-\tfrac12B-\tfrac12C&\cos\tfrac12b\cos\tfrac12c
\end{array}
\]
\[
\begin{array}{ll}
B&\sin\tfrac12b\\
90^0-\tfrac12A-\tfrac12B+\tfrac12C&\sin\tfrac12c\cos\tfrac12a\\
90^0+\tfrac12A-\tfrac12B-\tfrac12C&\sin\tfrac12a\cos\tfrac12c
\end{array}
\]
\[
\begin{array}{ll}
180^0-B&\cos\tfrac12b\\
\tfrac12A+\tfrac12B+\tfrac12C-90^0&\sin\tfrac12a\sin\tfrac12c\\
90^0-\tfrac12A+\tfrac12B-\tfrac12C&\cos\tfrac12a\cos\tfrac12c
\end{array}
\]
\[
\begin{array}{ll}
C&\sin\tfrac12c\\
90^0-\tfrac12A+\tfrac12B-\tfrac12C&\sin\tfrac12b\cos\tfrac12a\\
90^0+\tfrac12A-\tfrac12B-\tfrac12C&\sin\tfrac12a\cos\tfrac12b
\end{array}
\]
\[
\begin{array}{ll}
180^0-C&\cos\tfrac12c\\
\tfrac12A+\tfrac12B+\tfrac12C-90^0&\sin\tfrac12a\sin\tfrac12b\\
90^0-\tfrac12A-\tfrac12B+\tfrac12C&\cos\tfrac12a\cos\tfrac12b
\end{array}
\]
