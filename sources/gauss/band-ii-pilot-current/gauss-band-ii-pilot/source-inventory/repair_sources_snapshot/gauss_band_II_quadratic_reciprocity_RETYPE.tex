\section*{5.}

\textsc{Theorema.} \emph{Sit $x$ quantitas positiva non integra, inter cuius
multipla $x$, $2x$, $3x,\dots$ usque ad $nx$ nullum fiat integer; ponatur
$[nx]=h$, unde facile concluditur, etiam inter multipla quantitatis reciprocae
$\dfrac{1}{x}$, $\dfrac{2}{x}$, $\dfrac{3}{x},\dots$ usque ad $\dfrac{h}{x}$
integrum non reperiri. Tum dico fore}
\[
  \left.
  \begin{aligned}
  &\bigl[x\bigr]+\bigl[2x\bigr]+\bigl[3x\bigr]+\dots+\bigl[nx\bigr]\\
  +\;&\Bigl[\tfrac{1}{x}\Bigr]+\Bigl[\tfrac{2}{x}\Bigr]+\Bigl[\tfrac{3}{x}\Bigr]
       +\dots+\Bigl[\tfrac{h}{x}\Bigr]
  \end{aligned}
  \right\}=nh.
\]

\medskip
\emph{Dem.} Seriei $[x]+[2x]+[3x]+\dots+[nx]$, quam ponemus $=\Omega$, membra
prima usque ad $\bigl[\tfrac{1^{\text{tum}}}{x}\bigr]$ inclus.\ manifesto omnia
erunt $=0$; sequentia usque ad $\bigl[\tfrac{2^{\text{tum}}}{x}\bigr]$ cuncta
$=1$; sequentia usque ad $\bigl[\tfrac{3^{\text{tum}}}{x}\bigr]$ cuncta $=2$
et sic porro. Hinc fit
\[
  \left.
  \begin{aligned}
  \Omega=\quad &\phantom{+\,}0\times\Bigl[\tfrac{1}{x}\Bigr]\\
  &+\,1\times\Bigl\{\Bigl[\tfrac{2}{x}\Bigr]-\Bigl[\tfrac{1}{x}\Bigr]\Bigr\}\\
  &+\,2\times\Bigl\{\Bigl[\tfrac{3}{x}\Bigr]-\Bigl[\tfrac{2}{x}\Bigr]\Bigr\}\\
  &+\,3\times\Bigl\{\Bigl[\tfrac{4}{x}\Bigr]-\Bigl[\tfrac{3}{x}\Bigr]\Bigr\}\\
  &\quad\text{etc.}\\
  &+(h-1)\Bigl\{\Bigl[\tfrac{h}{x}\Bigr]-\Bigl[\tfrac{h-1}{x}\Bigr]\Bigr\}\\
  &+\,h\,\Bigl\{n-\Bigl[\tfrac{h}{x}\Bigr]\Bigr\}
  \end{aligned}
  \right\}
  = hn-\Bigl[\tfrac{1}{x}\Bigr]-\Bigl[\tfrac{2}{x}\Bigr]-\Bigl[\tfrac{3}{x}\Bigr]
        -\dots-\Bigl[\tfrac{h}{x}\Bigr].
\]
\textsc{Q.\,E.\,D.}

\section*{6.}

\textsc{Theorema.} \emph{Designantibus $k,p$ numeros positivos impares inter
se primos quoscunque, erit}
\[
  \left.
  \begin{aligned}
  &\Bigl[\tfrac{k}{p}\Bigr]+\Bigl[\tfrac{2k}{p}\Bigr]+\Bigl[\tfrac{3k}{p}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(p-1)k}{p}\Bigr]\\
  +\;&\Bigl[\tfrac{p}{k}\Bigr]+\Bigl[\tfrac{2p}{k}\Bigr]+\Bigl[\tfrac{3p}{k}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(k-1)p}{k}\Bigr]
  \end{aligned}
  \right\}=\tfrac{1}{4}(k-1)(p-1).
\]

\medskip
\emph{Demonstr.}\ Supponendo, quod licet, $k<p$, erit $\tfrac{\frac{1}{2}(p-1)k}{p}$
minor quam $\tfrac{1}{2}k$, sed maior quam $\tfrac{1}{2}(k-1)$, adeoque
$\bigl[\tfrac{\frac{1}{2}(p-1)k}{p}\bigr]=\tfrac{1}{2}(k-1)$. Hinc patet,
theorema praesens ex praec.\ protinus sequi, statuendo illic $\tfrac{k}{p}=x$,
$\tfrac{1}{2}(p-1)=n$, adeoque $\tfrac{1}{2}(k-1)=h$.

\medskip
Ceterum simili modo demonstrari potest, si $k$ fuerit numerus \emph{par} ad $p$
primus, fore
\[
  \left.
  \begin{aligned}
  &\Bigl[\tfrac{k}{p}\Bigr]+\Bigl[\tfrac{2k}{p}\Bigr]+\Bigl[\tfrac{3k}{p}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(p-1)k}{p}\Bigr]\\
  +\;&\Bigl[\tfrac{p}{k}\Bigr]+\Bigl[\tfrac{2p}{k}\Bigr]+\Bigl[\tfrac{3p}{k}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}kp}{k}\Bigr]
  \end{aligned}
  \right\}=\tfrac{1}{4}k(p-1).
\]
At huic propositioni ad institutum nostrum non necessariae non immoramur.

\section*{7.}

Iam ex combinatione theorematis praec.\ cum propos.\ VIII.\ art.\ 4.\ theorema
fundamentale protinus demanat. Nimirum denotantibus $k$, $p$ numeros primos
positivos inaequales quoscunque, et ponendo
\[
  (k,p)+\Bigl[\tfrac{k}{p}\Bigr]+\Bigl[\tfrac{2k}{p}\Bigr]+\Bigl[\tfrac{3k}{p}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(p-1)k}{p}\Bigr]=L
\]
\[
  (p,k)+\Bigl[\tfrac{p}{k}\Bigr]+\Bigl[\tfrac{2p}{k}\Bigr]+\Bigl[\tfrac{3p}{k}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(k-1)p}{k}\Bigr]=M
\]
per VIII.\ art.\ 4.\ patet, $L$ et $M$ semper fieri numeros pares. At per
theorema art.\ 6.\ erit
\[
  L+M=(k,p)+(p,k)+\tfrac{1}{4}(k-1)(p-1).
\]
Quoties igitur $\tfrac{1}{4}(k-1)(p-1)$ par evadit, quod fit, si vel uterque
$k,p$ vel saltem alterutur est formae $4n+1$, necessario $(k,p)$ et $(p,k)$ vel
ambo pares vel ambo impares esse debent. Quoties autem $\tfrac{1}{4}(k-1)(p-1)$
impar est, quod evenit, si uterque $k,p$ est formae $4n+3$, necessario alter
numerorum $(k,p),(p,k)$ par, alter impar esse debebit. In casu priori itaque
relatio ipsius $k$ ad $p$ et relatio ipsius $p$ ad $k$ (quatenus alter alterius
residuum vel non-residuum est) identicae erunt, in casu posteriori oppositae.
\textsc{Q.\,E.\,D.}
