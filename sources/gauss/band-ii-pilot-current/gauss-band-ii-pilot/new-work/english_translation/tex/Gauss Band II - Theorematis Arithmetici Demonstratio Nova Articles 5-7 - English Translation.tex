\section*{5.}

\textsc{Theorem.} \emph{Let $x$ be a positive non-integral quantity, among whose
multiples $x$, $2x$, $3x,\dots$ up to $nx$ no integer occurs; put $[nx]=h$.
From this it is readily concluded that among the multiples of the reciprocal
quantity
$\dfrac{1}{x}$, $\dfrac{2}{x}$, $\dfrac{3}{x},\dots$ up to $\dfrac{h}{x}$
no integer is found either. Then I say that}
\[
  \left.
  \begin{aligned}
  &\bigl[x\bigr]+\bigl[2x\bigr]+\bigl[3x\bigr]+\dots+\bigl[nx\bigr]\\
  +\;&\Bigl[\tfrac{1}{x}\Bigr]+\Bigl[\tfrac{2}{x}\Bigr]+\Bigl[\tfrac{3}{x}\Bigr]
       +\dots+\Bigl[\tfrac{h}{x}\Bigr]
  \end{aligned}
  \right\}=nh.
\]

\medskip
\emph{Proof.} The members of the series
$[x]+[2x]+[3x]+\dots+[nx]$, which we shall put equal to $\Omega$, are
manifestly all equal to $0$ from the first up to and including
$\bigl[\tfrac{1^{\mathrm{st}}}{x}\bigr]$; all the following members up to
$\bigl[\tfrac{2^{\mathrm{nd}}}{x}\bigr]$ are equal to $1$; all the following
members up to $\bigl[\tfrac{3^{\mathrm{rd}}}{x}\bigr]$ are equal to $2$; and
so on. Hence
\[
  \left.
  \begin{aligned}
  \Omega=\, &\phantom{+\,}0\times\Bigl[\tfrac{1}{x}\Bigr]\\
  &+\,1\times\Bigl\{\Bigl[\tfrac{2}{x}\Bigr]-\Bigl[\tfrac{1}{x}\Bigr]\Bigr\}\\
  &+\,2\times\Bigl\{\Bigl[\tfrac{3}{x}\Bigr]-\Bigl[\tfrac{2}{x}\Bigr]\Bigr\}\\
  &+\,3\times\Bigl\{\Bigl[\tfrac{4}{x}\Bigr]-\Bigl[\tfrac{3}{x}\Bigr]\Bigr\}\\
  &\quad\text{etc.}\\
  &+(h-1)\Bigl\{\Bigl[\tfrac{h}{x}\Bigr]-\Bigl[\tfrac{h-1}{x}\Bigr]\Bigr\}\\
  &+\,h\,\Bigl\{n-\Bigl[\tfrac{h}{x}\Bigr]\Bigr\}
  \end{aligned}
  \right\}
  = hn-\Bigl[\tfrac{1}{x}\Bigr]-\Bigl[\tfrac{2}{x}\Bigr]-\Bigl[\tfrac{3}{x}\Bigr]
        -\dots-\Bigl[\tfrac{h}{x}\Bigr].
\]
\textsc{Q.E.D.}

\section*{6.}

\textsc{Theorem.} \emph{If $k,p$ denote any positive odd numbers prime to one
another, then}
\[
  \left.
  \begin{aligned}
  &\Bigl[\tfrac{k}{p}\Bigr]+\Bigl[\tfrac{2k}{p}\Bigr]+\Bigl[\tfrac{3k}{p}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(p-1)k}{p}\Bigr]\\
  +\;&\Bigl[\tfrac{p}{k}\Bigr]+\Bigl[\tfrac{2p}{k}\Bigr]+\Bigl[\tfrac{3p}{k}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(k-1)p}{k}\Bigr]
  \end{aligned}
  \right\}=\tfrac{1}{4}(k-1)(p-1).
\]

\medskip
\emph{Proof.} Suppose, as is permissible, that $k<p$. Then
$\tfrac{\frac{1}{2}(p-1)k}{p}$ will be less than $\tfrac{1}{2}k$, but greater
than $\tfrac{1}{2}(k-1)$, and therefore
$\bigl[\tfrac{\frac{1}{2}(p-1)k}{p}\bigr]=\tfrac{1}{2}(k-1)$. Hence it is plain
that the present theorem follows at once from the preceding one, by putting
there $\tfrac{k}{p}=x$, $\tfrac{1}{2}(p-1)=n$, and therefore
$\tfrac{1}{2}(k-1)=h$.

\medskip
Moreover, in a similar way it can be proved that, if $k$ is an \emph{even}
number prime to $p$, then
\[
  \left.
  \begin{aligned}
  &\Bigl[\tfrac{k}{p}\Bigr]+\Bigl[\tfrac{2k}{p}\Bigr]+\Bigl[\tfrac{3k}{p}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(p-1)k}{p}\Bigr]\\
  +\;&\Bigl[\tfrac{p}{k}\Bigr]+\Bigl[\tfrac{2p}{k}\Bigr]+\Bigl[\tfrac{3p}{k}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}kp}{k}\Bigr]
  \end{aligned}
  \right\}=\tfrac{1}{4}k(p-1).
\]
But we shall not pause over this proposition, since it is unnecessary for our purpose.

\section*{7.}

Now, by combining the preceding theorem with Proposition VIII of Article 4, the
fundamental theorem follows at once. Namely, if $k$ and $p$ denote any unequal
positive prime numbers, and if one puts
\[
  (k,p)+\Bigl[\tfrac{k}{p}\Bigr]+\Bigl[\tfrac{2k}{p}\Bigr]+\Bigl[\tfrac{3k}{p}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(p-1)k}{p}\Bigr]=L
\]
\[
  (p,k)+\Bigl[\tfrac{p}{k}\Bigr]+\Bigl[\tfrac{2p}{k}\Bigr]+\Bigl[\tfrac{3p}{k}\Bigr]
        +\dots+\Bigl[\tfrac{\frac{1}{2}(k-1)p}{k}\Bigr]=M,
\]
then by Proposition VIII of Article 4 it is clear that $L$ and $M$ always become
even numbers. But by the theorem of Article 6,
\[
  L+M=(k,p)+(p,k)+\tfrac{1}{4}(k-1)(p-1).
\]
Therefore, whenever $\tfrac{1}{4}(k-1)(p-1)$ turns out even, which happens if
both $k,p$, or at least one of them, is of the form $4n+1$, the numbers
$(k,p)$ and $(p,k)$ must necessarily be either both even or both odd. Whenever,
on the other hand, $\tfrac{1}{4}(k-1)(p-1)$ is odd, which occurs if both $k,p$
are of the form $4n+3$, one of the two numbers $(k,p),(p,k)$ must necessarily
be even and the other odd. Thus, in the former case the relation of $k$ to $p$
and the relation of $p$ to $k$ - insofar as one is or is not a residue of the
other - will be identical; in the latter case they will be opposite.
\textsc{Q.E.D.}
