\textbf{[1.]} So folgt leicht, dass
%
\begin{gather*}
r = \frac{\pi}{\omega} \cdot e^{-\frac{\pi\omega'}{2\omega}} \bigl\{1 + 2e^{-\pi\frac{\omega'}{\omega}} \cos 2(\lambda\pi + \mu\omega) + 2e^{-2\pi\frac{\omega'}{\omega}} \cos 4(\lambda\pi + \mu\omega) + \text{etc.}\bigr\} \\
\intertext{In einer andern Form so:}
\frac{\pi r}{\omega} = \sqrt[4]{-1} \cdot e^{-\frac{\pi(\lambda\mu' - \lambda'\mu)}{2}} \cdot \sqrt[4]{e} \cdot \sqrt[4]{\frac{\omega}{2\pi\sqrt{-1}}}
\end{gather*}
%
Oder, $aa' = \frac{1}{2}\pi\sqrt{-1}$ und $-a\omega = (a'\omega)$ gesetzt,
%
\begin{multline*}
2e^{-a(\lambda\mu + \lambda'\mu')} \bigl\{\cos(2\lambda\mu + 2\lambda'\mu')\, \omega\pi + \sqrt{-1}\sin(2\lambda\mu + 2\lambda'\mu')\, \omega\pi\bigr\} \\
= \sqrt[4]{-1} \cdot 2e^{-a\lambda\mu} \bigl\{\cos 2\lambda\omega\pi - \sqrt{-1}\sin 2\lambda\omega\pi\bigr\}
\end{multline*}
%
Man kann den Lehrsatz auch so ausdrücken:
%
\begin{equation*}
\sqrt[4]{e} \cdot \sqrt[\leftroot{2}\uproot{2}4]{\frac{\omega}{2\pi\sqrt{-1}}} \cdot e^{-\frac{\pi(\lambda\mu' - \lambda'\mu)}{2}} \cdot \sqrt{\frac{\pi r}{\omega}}
\end{equation*}
ändert den Werth nicht, wenn $\lambda$ in $-\lambda$ und $\mu$ in $-\mu$ verwandelt wird.

\textbf{[2.]} Man setze
%
\begin{gather*}
P = 1 + x^{\omega} + x^{\omega+2} + x^{\omega+4} + \cdots + x^{\omega^2 - \omega} \\
Q = 1 + x^{\omega+1} + x^{\omega+3} + \cdots + x^{\omega^2 - \omega + 1} \\
R = P - x^{\omega} \cdot Q
\end{gather*}

Man suche zuerst diese Differenz, indem man das erste, zweite, dritte Glied
u.s.w. der Reihe $Q$ von dem ersten, zweiten, dritten Gliede der Reihe $P$ abzieht, so kommt
%
\begin{equation*}
1 - x^{\omega},\; x^{\omega}(1-x^{\omega}),\; x^{\omega+2}(1-x^{\omega}),\; \ldots
\end{equation*}
%
Wir bezeichnen diese Reihe durch $\varphi(x, \omega)$.

Man suche ferner jene Differenz $R$, indem man das erste, zweite, dritte
Glied u.s.w. der Reihe $Q$ von dem zweiten, dritten, vierten u.s.w. der Reihe
$P$ abzieht, so wird
%
\begin{equation*}
R = x^{\omega}\bigl(1 - x^{2(\omega+1)}\bigr) \cdot \varphi(x, \omega+1)
\end{equation*}
%
folglich
%
\begin{equation*}
\frac{R}{x^{\omega}} = \bigl(1 - x^{2(\omega+1)}\bigr) \cdot \varphi(x, \omega+1)
\end{equation*}
%
Dieser Schluss ist allgemein, so lange $\omega > 1$, man hat demnach unter dieser Einschränkung
%
\begin{equation*}
\varphi(x, \omega) = \frac{1 - x^{\omega^2}}{1 - x^{\omega}} \cdot \varphi(x, \omega+1)
\end{equation*}
%
hingegen ist für den Fall $n = 0$ das letzte Glied von $Q$ nicht als verschwindend zu betrachten. Setzt man es $= T$, so wird der erste Werth von $R$, um $T$ kleiner sein als der zweite, also
%
\begin{gather*}
T = 1 - x^{\omega^2} \cdot \varphi(x, 1) = \varphi(x, 0) \\
\varphi(x, 0) = 1 - x \cdot \varphi(x, 1) = 1 - x + x^3 - x^6 + x^{10} - \cdots - T
\intertext{also da}
\varphi(x, 0) = 1 \quad \text{und} \quad T = x^{\omega^2} \cdot \varphi(x, 1)
\intertext{so wird}
\varphi(x, 1) = \frac{1 - x}{1 - x^{\omega^2}},\quad
\varphi(x, 2) = \frac{1 - x^2}{1 - x^{\omega^2}} \cdot \frac{1 - x}{1 - x^{\omega^2}},\quad \text{etc.}
\end{gather*}

\textbf{[3.]} etc. $= 1 - 2x^2 + 2x^4 - 2x^6 + 2x^8 - \text{etc.}$
%

\begin{equation*}

\end{equation*}

\end{equation*}

%
\begin{equation*}
\varphi(x, \omega) = \frac{1 - x^{\omega^2}}{1 - x^{\omega}} \cdot \varphi(x, \omega+1)
\end{equation*}

folglich

\end{equation*}

%
\begin{equation*}
\varphi(x, 0) = 1 + x + x^3 + x^6 + x^{10} + \text{etc.}
\end{equation*}

Die Theoreme in Beziehung auf diejenigen Reihen und unendlichen
Producte, welche zu der Theorie der Arithmetisch-Geometrischen Mittel
gehören, ordnen wir so:

\textbf{1.} $1 - x \cdot 1 - x^2 \cdot 1 - x^3 \cdot 1 - x^4 \cdots = [x]$

\textbf{2.} $1 + x \cdot 1 + x^2 \cdot 1 + x^3 \cdot 1 + x^4 \cdots = [x]$

\textbf{3.} $1 - x \cdot 1 - x^3 \cdot 1 - x^5 \cdot 1 - x^7 \cdots = [x]$

\textbf{5.} $\displaystyle\frac{[xx]^2}{[x]} = 1 + x + x^3 + x^6 + x^{10} + \cdots = \frac{[x^2]^2}{[x]}$

\textbf{6.}
\begin{multline*}
(1 + x^{\frac{1}{2}}y)(1 + x^{\frac{3}{2}}y)(1 + x^{\frac{5}{2}}y) \cdots (1 + x^{\frac{1}{2}}y^{-1})(1 + x^{\frac{3}{2}}y^{-1})(1 + x^{\frac{5}{2}}y^{-1}) \cdots \\
= x^{\frac{1}{8}} \bigl\{(1 + yy^{-1}) + (x + x^{-1})yy^{-1} + (x^2 + x^{-2})y^2y^{-2} + \cdots\bigr\}
\end{multline*}
%
\textbf{7.} $F(x) = (1 - x^2)(1 - x^4)(1 - x^6) \cdots = [x^2]$

\textbf{8.}
\begin{gather*}
Fx = (1 - x^2)(1 - x^4)(1 - x^6) \cdots \quad [xx][x^4]^2 = [x^2]^2 \\
1 - 2x^2 + 2x^8 - 2x^{18} + \cdots = \frac{[xx]}{[x^2]^2}
\end{gather*}

\textbf{9.}
\begin{gather*}
[xx]Fx = [x^2]Fx^2 = [x^4]Fx^4 = [x^8]Fx^8 = \text{etc.} = 1 \\
1 - 2x + 2x^4 - 2x^9 + \cdots = \frac{[xx]}{[x]^2}
\end{gather*}

\textbf{10.}
\begin{equation*}
\frac{1 + x \cdot 1 + x^3 \cdot 1 + x^5 \cdots}{1 - x \cdot 1 - x^3 \cdot 1 - x^5 \cdots} =
\frac{1 + 2x + 2x^4 + 2x^9 + \cdots}{1 - 2x + 2x^4 - 2x^9 + \cdots} =
\frac{[xx]^2}{[x]^2 [x^4]}
\end{equation*}

\textbf{[3.]} Die Relationen zwischen den Seiten des sph\"arischen F\"unfecks so

Quadrate der Gleichungen

Tangenten\quad Secanten\quad Cosecanten
%
\begin{align*}
\alpha && \beta && 1 + \alpha &= \beta\delta \\
\varepsilon && \gamma\varepsilon && \gamma\delta && 1 + \gamma &= \varepsilon\alpha \\
\alpha\delta && \gamma && 1 + \gamma &= \alpha\delta \\
\beta\gamma && 1 + \varepsilon &= \beta\gamma
\end{align*}
%

%
\begin{align*}
(1 + \varepsilon)(1 + \beta - \gamma\delta) - (1 + \varepsilon)(1 + \gamma - \varepsilon\alpha)
&= \beta\delta(1 + \varepsilon)\alpha - (1 + \gamma)\beta \\
&= \beta\delta(\alpha\delta - \delta - 1) - (\gamma\varepsilon - \alpha - 1)\}
\end{align*}

und auf \"ahnliche Weise wird die f\"unfte aus dreien der \"ubrigen abgeleitet.

%
\begin{equation*}
\beta = \frac{1 + \alpha + \gamma}{\alpha\gamma}, \quad
\delta = \frac{1 + \gamma + \varepsilon}{\gamma\varepsilon}, \quad
\varepsilon = \frac{1 + \varepsilon + \alpha}{\varepsilon\alpha}, \quad
\delta = \frac{\alpha\delta - 1}{\alpha\delta - 1}, \quad
\alpha\varepsilon = \frac{\cdots}{\alpha\delta - 1}
\end{equation*}
%
\begin{equation*}
\beta = \frac{1 + \alpha}{\alpha\gamma}, \quad
\delta = \frac{1 + \gamma}{\gamma\varepsilon}, \quad
\varepsilon = \frac{1 + \varepsilon}{\varepsilon\alpha}
\end{equation*}
