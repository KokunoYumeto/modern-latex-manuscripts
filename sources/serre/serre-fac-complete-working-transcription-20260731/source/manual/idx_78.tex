\begin{center}
\textbf{\S6. Fonction caractéristique et genre arithmétique}
\end{center}

\textbf{79. Caractéristique d'Euler-Poincaré.} Soit $V$ une variété projective, et soit $\mathfrak{F}$ un faisceau algébrique cohérent sur $V$. Posons:

\begin{equation*}
h^q(V, \mathfrak{F}) = \dim_K H^q(V, \mathfrak{F}).
\end{equation*}

Nous avons vu (n\textsuperscript{o} 66, Théorème 1) que les $h^q(V, \mathfrak{F})$ sont \textit{finis} pour tout entier $q$, et nuls pour $q > \dim V$. On peut donc définir un entier $\chi(V, \mathfrak{F})$ en posant:

\begin{equation*}
\chi(V, \mathfrak{F}) = \sum_{q=0}^{\infty} (-1)^q h^q(V, \mathfrak{F}).
\end{equation*}

C'est la \textit{caractéristique d'Euler-Poincaré} de $V$, à valeurs dans $\mathfrak{F}$.

\textsc{Lemme} 1. \textit{Soit} $0 \rightarrow L_1 \rightarrow \cdots \rightarrow L_p \rightarrow 0$ \textit{une suite exacte, les $L_i$ étant des espaces vectoriels de dimension finie sur $K$, et les homomorphismes $L_i \rightarrow L_{i+1}$ étant $K$-linéaires. On a alors}:

\begin{equation*}
\sum_{q=1}^{q=p} (-1)^q \dim_K L_q = 0.
\end{equation*}

On raisonne par récurrence sur $p$, le lemme étant évident si $p \leqq 3$; si $L'_{p-1}$ désigne le noyau de $L_{p-1} \rightarrow L_p$ , on a les deux suites exactes:

\begin{equation*}
0 \rightarrow L_1 \rightarrow \cdots \rightarrow L'_{p-1} \rightarrow 0
\end{equation*}
\begin{equation*}
0 \rightarrow L'_{p-1} \rightarrow L_{p-1} \rightarrow L_p \rightarrow 0.
\end{equation*}

En appliquant l'hypothèse de récurrence à chacune de ces suites, on voit que $\sum_{q=1}^{q=p-2} (-1)^q \dim L_q + (-1)^{p-1} \dim L'_{p-1} = 0$, et

\begin{equation*}
\dim L'_{p-1} - \dim L_{p-1} + \dim L_p = 0,
\end{equation*}

d'où aussitôt le Lemme.

\textsc{Proposition} 1. \textit{Soit} $0 \rightarrow \mathfrak{A} \rightarrow \mathfrak{B} \rightarrow \mathfrak{C} \rightarrow 0$ \textit{une suite exacte de faisceaux algébriques cohérents sur une variété projective $V$, les homomorphismes} $\mathfrak{A} \rightarrow \mathfrak{B}$ \textit{et} $\mathfrak{B} \rightarrow \mathfrak{C}$ \textit{étant $K$-linéaires. On a alors}:

\begin{equation*}
\chi(V, \mathfrak{B}) = \chi(V, \mathfrak{A}) + \chi(V, \mathfrak{C}).
\end{equation*}

D'après le Corollaire 2 au Théorème 5 du n\textsuperscript{o} 47, on a une suite exacte de cohomologie:

\begin{equation*}
\cdots \rightarrow H^q(V, \mathfrak{B}) \rightarrow H^q(V, \mathfrak{C}) \rightarrow H^{q+1}(V, \mathfrak{A}) \rightarrow H^{q+1}(V, \mathfrak{B}) \rightarrow \cdots
\end{equation*}

En appliquant le Lemme 1 à cette suite exacte d'espaces vectoriels, on obtient la Proposition.
