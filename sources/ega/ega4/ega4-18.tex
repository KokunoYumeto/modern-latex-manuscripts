\section{Supplement on \'etale morphisms. Henselian local rings and strictly local rings}
\label{section:IV.18}

\oldpage[IV]{109}

In this paragraph, we study various special properties of \'etale morphisms.
Moreover, the notion of \'etale morphism allows us to develop in a natural way the theory of Nagata of Henselian rings, as well as that of strictly local rings.
These rings play an important r\^ole in numerous recent developments, appearing each time one needs a procedure of ``localisation'' finer than that furnished by the Zariski topology (cf.\ for example~\cite{IV-43}, while awaiting the publication of the chapter of our Treatise devoted to the study of the ``\'etale topology'').


\subsection{A remarkable equivalence of categories.}
\label{subsection:IV.18.1}

\begin{proposition}[18.1.1]
\label{IV.18.1.1}
Let $S$ be a prescheme, $S_0$ a closed sub-prescheme of $S$, lisse (resp.\ \'etale) on $S_0$, $x_0$ a point of $X_0$.
There exists then an open neighbourhood $U_0$ of $x_0$ in $X_0$, an S-prescheme $U$ lisse (resp.\ \'etale) on $S$ and an $S_0$-isomorphism $U\times_S S_0\xrightarrow{\sim} U_0$.
\end{proposition}

\begin{proof}
Note that if $X_0$ is \'etale on $S_0$ at the point $x_0$, it is \emph{a fortiori} non ramified on $S_0$ at this point; if one constructs an S-prescheme $U$ lisse on $S$ such that $U\times_S S_0$ is isomorphic to $X_0$ in a neighbourhood of $x_0$ \sref{IV.17.4.1},~$d$)), it will then follow that $U$ is non ramified on $S$ at the point $x_0$ \sref{IV.17.4.1},~$d$)), hence also \'etale in a neighbourhood of $x_0$.
It suffices therefore to prove the proposition when one only supposes $X_0$ lisse on $S_0$.

The question being local on $S$ and $X_0$, one can suppose that $S=\operatorname{Spec}(A)$ and $X_0=\operatorname{Spec}(C_0)$, where $A_0$ is a quotient ring of $A$, $C_0=B_0/\mathfrak{J}_0$, where $B_0=A_0[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ and $\mathfrak{J}_0$ is an ideal of finite type of $B_0$; finally, $C_0$ is an $A_0$-algebra formally lisse for the discrete topologies.
Let $\mathfrak{p}_0$ be the ideal corresponding to $\mathfrak{q}_0$ in $B_0$.
The Jacobian criterion ($\mathbf{0}$, \sref{IV.22.6.4}) applied to ($\mathbf{0}$, \sref{IV.19.1.12}) shows that there exist in $\mathfrak{J}_0$ a family of $r$ polynomials $u_i$ $(1\leq i\leq r)$ and $r$ indices $j_h$ $(1\leq h\leq r)$ such that the images of the $u_i$ in $(\mathfrak{J}_0)_{\mathfrak{q}_0}/(\mathfrak{J}_0^2)_{\mathfrak{q}_0}$ engender this $(\mathrm{B}_0)_{\mathfrak{q}_0}$-module, and that one has
\begin{equation}
\tag{18.1.1.1}
  \det(\partial u_i/\partial\mathrm{T}_{j_h})\notin\mathfrak{q}_0.
\end{equation}
As $(\mathrm{B}_0)_{\mathfrak{q}_0}$ is a local ring, it results from the Nakayama lemma that one can suppose that the images of the $u_i$ in $(\mathfrak{J}_0)_{\mathfrak{q}_0}$ engender this $(\mathrm{B}_0)_{\mathfrak{q}_0}$-module, then, replacing $X_0$ by an affine open neighbourhood of $x_0$ ($\mathbf{0}_I$, \sref{IV.5.2.2}), that the $u_i$ engender $\mathfrak{J}_0$.
Set then $B=A[\mathrm{T}_1,\ldots,\mathrm{T}_n]$; $B_0$ is thus a quotient ring of $B$.
For every $i$, let $v_i\in B$ be an element whose $u_i$ is the image in $B_0$, and let $\mathfrak{J}$ be the ideal of $B$ engendered by the $v_i$.
The proposition will be established by taking $U$ to be an open affine neighbourhood of the point of $\operatorname{Spec}(B/\mathfrak{J})$ corresponding to the ideal prime $\mathfrak{p}=\mathfrak{q}/\mathfrak{J}$, provided that one proves that $\mathrm{B}_{\mathfrak{q}}/\mathfrak{J}_{\mathfrak{q}}$ is an A-algebra formally lisse for the discrete topologies.
Or, by virtue of the Jacobian criterion ($\mathbf{0}$, \sref{IV.22.6.1} and \sref{IV.20.5.12.1}), it suffices to take $\det(\partial v_i/\partial\mathrm{T}_{j_h})\notin\mathfrak{q}$, which is possible since $\mathfrak{J}_0$ is the image of $\mathfrak{J}$ in $B_0$.
\end{proof}

\begin{theorem}[18.1.2]
\label{IV.18.1.2}
Let $S$ be a prescheme, $S_0$ a closed sub-prescheme of $S$ whose underlying space is identical to that of $S$.
Then the functor
\[
X \mapsto X\times_S S_0
\]
from the category of S-preschemes \'etale and finite on $S$, to the category of $S_0$-preschemes \'etale and finite on $S_0$, is an equivalence of categories.
\end{theorem}

\begin{proof}
Let us show first that this functor is \emph{pleinement fid\`ele}.
Let $X$, $Y$ be two S-preschemes \'etale on $S$, and set $X_0=X\times_S S_0$, $Y_0=Y\times_S S_0$.
If $Z=X\times_S Y$, the set $\operatorname{Hom}_S(X,Y)$ is in canonical bijection with the set of X-sections $\Gamma(Z/X)$, and likewise $\operatorname{Hom}_{S_0}(X_0,Y_0)$ is in canonical bijection with $\Gamma(Z_0/X_0)$, where $Z_0=Z\times_S S_0=X_0\times_{S_0} Y_0$.
Or $Z$ is \'etale on $X$, $Z_0$ \'etale on $X_0$ (resp.\ $Z_0$) is a closed sub-prescheme of $X$ (resp.\ $Z$) with the same underlying space.
The open parts of $Z$ on which the restriction of the morphism $Z\to X$ is surjective and radiciel are thus the \emph{same} as the parts of $Z_0$ having the corresponding properties and our assertion follows from \sref{IV.17.9.3}.

To finish the proof of \sref{IV.18.1.2}, it suffices to see that for every $S_0$-prescheme $X_0$ \'etale and finite on $S$, there exists an $\sh{O}_X$-Algebra $\sh{B}$ finite and \'etale and an $S_0$-isomorphism $\sh{B}_0\xrightarrow{\sim}\sh{B}\otimes_{\sh{O}_x}\sh{O}_{X_0}$.
Or, it results from prop.\ \sref{IV.18.1.1} that there is a projective system $(\sh{B}_n)$ such that $\mathrm{B}_n$ is an $\sh{O}_{X_n}$-Algebra finite and \'etale and the homomorphisms $\mathrm{B}_{n+1}\otimes_{A_{n+1}}A_n\to\mathrm{B}_n$ are bijectifs.
It results from \sref{IV.18.3.2.1} and \sref{IV.18.3.1} that the A-algebra $B=\varprojlim B_n$ is a projective A-module of finite type and that $B_0$ is isomorphic to $B\otimes_A A_0$.
To prove that $B$ is an A-algebra \'etale, it suffices, by virtue of \sref{IV.18.2.5}, to show that for every maximal ideal $\mathfrak{m}$ of $A$, $\mathrm{B}_{\mathfrak{m}}/\mathfrak{m}\mathrm{B}_{\mathfrak{m}}$ is a $k(y)$-algebra separable.
But since the morphism structural $f:X\to S$ is \emph{proper}, $f(x)$ is a point belonging to $S_0$, hence $\mathfrak{J}$ is contained in the radical of $A$ ($\mathbf{0}_I$, \sref{IV.7.1.10}); the conclusion results from the fact that $X_0=f^{-1}(S_0)$ and that $\sh{B}_0$ is an $\sh{O}_{X_0}$-algebra finite and \'etale.
\end{proof}

\begin{corollary}[18.1.3]
\label{IV.18.1.3}
Let $S$ be a prescheme, $X$ an S-prescheme \'etale on $S$, $S'$ a sub-prescheme of $S'$ having the same underlying space.
Then the canonical application $X(S')_S\to X(S'_0)_S$ ($\mathbf{I}$, \sref{IV.3.4.3}) is bijective.
\end{corollary}

\begin{proof}
One sets indeed $X'=X\times_S S'$, $X'_0=X\times_S S'_0$, so that $X(S')_S=\Gamma(X'/S')$ and $X(S'_0)_S=\Gamma(X'_0/S'_0)$; the corollary results from the fact that the functor defined in \sref{IV.18.1.2} (with $S$ and $S_0$ replaced by $S'$ and $S'_0$ respectively) is pleinement fid\`ele (or directly from \sref{IV.17.9.3}).
\end{proof}


\subsection{\'Etale coverings.}
\label{subsection:IV.18.2}

\begin{env}[18.2.1]
\label{IV.18.2.1}
Given a ring $A$ and a commutative A-algebra $B$ which is \emph{finite} and is an A-module \emph{libre}, we recall (Bourbaki, \emph{Alg.}, chap.~VIII, \textsection~12, n\textsuperscript{o}~2) that one defines on $B$ an A-\emph{linear form} $\operatorname{Tr}_{B/A}$, the ``trace form''; one deduces from this the definition of a \emph{symmetric A-bilinear form} (also called ``trace form'')
\begin{equation}
\tag{18.2.1.1}
  (x,y)\mapsto\operatorname{Tr}_{B/A}(xy)
\end{equation}
whose data is equivalent to that of the A-linear application associated $\operatorname{astr}_{B/A}:B\to\check{B}$, where $\check{B}$ is the dual of the A-module $B$, equal to its transposed.
When $A$ is a \emph{field}, it is equivalent to say that this bilinear form is \emph{non degenerate} or that $B$ is a \emph{separable} algebra (Bourbaki, \emph{Alg.}, chap.~IX, \textsection~2, prop.~5).
\end{env}

\begin{proposition}[18.2.3]
\label{IV.18.2.3}
Let $f:X\to Y$ be a finite morphism of preschemes and set $\sh{B}=f_*(\sh{O}_X)$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] $f$ is \'etale.
  \item[a')] $f$ is a morphism flat of finite presentation, and for every $x\in X$, if one sets $y=f(x)$, $\sh{O}_{X,x}/\mathfrak{m}_y\sh{O}_{X,x}$ is a field, finite and separable extension of $k(y)$.
  \item[b)] $\sh{B}$ is an $\sh{O}_Y$-Module locally free and for every $y\in Y$, $\sh{B}_y\otimes_{\sh{O}_{Y,y}} k(y)$ is a $k(y)$-algebra finite separable (hence a direct composition of a finite number of fields, separable finite extensions of $k(y)$).
  \item[c)] $\sh{B}$ is an $\sh{O}_Y$-Module locally free, and the homomorphism $\operatorname{astr}_{\sh{B}/\sh{O}_Y}:\sh{B}\to\check{\sh{B}}$ \sref{IV.18.2.2} is bijective.
\end{enumerate}
\end{proposition}

\begin{corollary}[18.2.4]
\label{IV.18.2.4}
Let $f:X\to Y$ be a morphism finite and of finite presentation, and set $\sh{B}=f_*(\sh{O}_X)$.
Let $y$ be a point of $Y$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] There exists an open neighbourhood $U$ of $y$ in $Y$ such that the restriction $f^{-1}(U)\to U$ of $f$ is an \'etale morphism.
  \item[b)] $\sh{B}_y$ is an $\sh{O}_{X,y}$-module libre of finite type and $\sh{B}_y\otimes_{\sh{O}_{Y,y}} k(y)$ is a $k(y)$-algebra separable.
\end{enumerate}
\end{corollary}

\begin{corollary}[18.2.5]
\label{IV.18.2.5}
Let $Y$ be a prescheme quasi-compact or locally Noetherian, $f:X\to Y$ a morphism finite and of finite presentation; set $\sh{B}=f_*(\sh{O}_X)$.
Suppose that for every closed point $y$ of $Y$, $\sh{B}_y$ is an $\sh{O}_{Y,y}$-module libre and $\sh{B}_y\otimes_{\sh{O}_{Y,y}} k(y)$ a $k(y)$-algebra separable.
Then $f$ is \'etale.
\end{corollary}

\begin{corollary}[18.2.6]
\label{IV.18.2.6}
If $f:X\to Y$ is a morphism finite and \'etale, and if one sets $\sh{B}=f_*(\sh{O}_X)$, then the $\sh{O}_Y$-homomorphism $\operatorname{Tr}_{\sh{B}/\sh{O}_Y}:\sh{B}\to\sh{O}_Y$ \sref{IV.18.2.2} (also denoted $\operatorname{Tr}_f$) is surjective.
\end{corollary}

\begin{remarks}[18.2.7]
\label{IV.18.2.7}
(i)
When $f:X\to Y$ is a finite morphism such that $f_*(\sh{O}_X)$ is an $\sh{O}_Y$-Module locally libre (resp.\ locally libre of rank $n$), one says also that $f$ is a morphism \emph{finite locally libre} (resp.\ \emph{locally libre of rank $n$}).
This condition, by virtue of \sref{IV.18.2.3}, is satisfied if $f$ is a finite \'etale morphism, but does not imply by itself that $f$ be \'etale, as is shown by the example where $X=\operatorname{Spec}(K)$ and $Y=\operatorname{Spec}(k)$ are spectra of fields, $K$ being a finite non separable extension of $k$.
When $f:X\to Y$ is a finite \'etale morphism, one says also that $X$ is an \emph{\'etale covering} of $Y$.
In particular $f(X)$ is an open and closed part of $Y$.

One says that a covering is \emph{trivial} if $X$ is the sum of a finite number of preschemes isomorphic to $Y$.
One says that a covering $f:X\to Y$ is \emph{locally trivial} if the morphism $f:X\to Y$ is such that for every point $y\in Y$ there is an open neighbourhood $U$ such that $f^{-1}(U)$ is a trivial covering of $U$.

(ii)
Let $f:X\to Y$ be a finite morphism, locally libre of rank $n$; set $\sh{B}=f_*(\sh{O}_X)$ and let $u=\operatorname{astr}_{\sh{B}/\sh{O}_Y}:\sh{B}\to\check{\sh{B}}$; one deduces from this an exterior $n$-th power homomorphism $\bigwedge^n u:\bigwedge^n\sh{B}\to(\bigwedge^n\sh{B})^\vee$ of $\sh{O}_Y$-Modules \emph{inversibles}, and hence ($\mathbf{0}_I$, \sref{IV.5.4.2}) an element
\begin{equation}
\tag{18.2.7.1}
  d_{X/Y}\in\Gamma(Y,(\bigwedge^n\check{\sh{B}})\otimes_{\sh{O}_Y}(\bigwedge^n\check{\sh{B}}))
\end{equation}
called the \emph{discriminant} of $X$ over $Y$.
Since $(\bigwedge^n\sh{B})\otimes_{\sh{O}_Y}(\bigwedge^n\sh{B})$ is the dual of $(\bigwedge^n\check{\sh{B}})\otimes_{\sh{O}_Y}(\bigwedge^n\check{\sh{B}})$, $d_{X/Y}$ can also be identified with a homomorphism
\begin{equation}
\tag{18.2.7.2}
  (\bigwedge^n\sh{B})\otimes_{\sh{O}_Y}(\bigwedge^n\sh{B})\to\sh{O}_Y
\end{equation}
and one denotes $\sh{D}_{X/Y}$ the quasi-coherent Ideal of finite type of $\sh{O}_Y$, image of the homomorphism \sref{IV.18.2.7.2}, also called the \emph{Ideal discriminant} of $X$ over $Y$.
\end{remarks}

\begin{proposition}[18.2.8]
\label{IV.18.2.8}
Let $f:X\to Y$ be a morphism \'etale, separated and of finite type, and for every $y\in Y$, let $n(y)$ be the geometric number of points of $f^{-1}(y)$.
Then the function $y\mapsto n(y)$ is semi-continuous inferiorly in $Y$.
For it to be continuous in a neighbourhood of $y$ (and hence constant in a neighbourhood of $y$), it is necessary and sufficient that there exist an open neighbourhood $U$ of $y$ such that the restriction $f^{-1}(U)\to U$ of $f$ is a finite (\'etale) morphism.
\end{proposition}

\begin{proof}
As $f$ is quasi-finite \sref{IV.17.6.1} and locally of finite presentation, it amounts to the same to say that $f$ is finite or that $f$ is proper \sref{IV.8.11.1}: moreover each fibre $f^{-1}(y)$ is geometrically reduced on $k(y)$.
The conclusions then result from \sref{IV.15.5.9},~(i) and~(ii)) and from the fact that $f$ is flat.
\end{proof}

\begin{corollary}[18.2.9]
\label{IV.18.2.9}
Let $Y$ be a connected prescheme, $f:X\to Y$ a morphism \'etale separated and of finite type.
For $f$ to be finite (in other words, for $X$ to be an \'etale covering of $Y$), it is necessary and sufficient that all the fibres of $f$ have the same geometric number of points.
\end{corollary}


\subsection{Finite and \'etale algebras.}
\label{subsection:IV.18.3}

\begin{proposition}[18.3.1]
\label{IV.18.3.1}
Let $A$ be a ring, $B$ an A-algebra of finite presentation.
\begin{enumerate}[label=(\roman*)]
  \item For $B$ to be a non ramified A-algebra, it is necessary and sufficient that $B$ be an A-module of finite presentation and that $B$ is a $(\mathrm{B}\otimes_A\mathrm{B})$-module projective.
  \item Suppose moreover that $B$ is a finite A-algebra.
  For $B$ to be an \'etale A-algebra, it is necessary and sufficient that $B$ is an A-module projective and a $(\mathrm{B}\otimes_A\mathrm{B})$-module projective.
\end{enumerate}
\end{proposition}

\begin{proposition}[18.3.2]
\label{IV.18.3.2}
Let $A$ be a ring, $\mathfrak{J}$ an ideal of $A$ such that, for the $\mathfrak{J}$-pre-adic topology, $A$ is separated and complete; set $A_0=A/\mathfrak{J}$.
Then the functor
\[
B\mapsto B\otimes_A A_0
\]
is an equivalence of the category of A-algebras finite and \'etale, to the category of $A_0$-algebras finite and \'etale.
\end{proposition}

\begin{example}[18.3.3]
\label{IV.18.3.3}
Prop.\ \sref{IV.18.3.2} applies in particular when $A$ is a \emph{local ring, separated and complete}, $\mathfrak{J}$ being the maximal ideal of $A$, so that $A_0$ is a field, and the category of $A_0$-algebras finite and \'etale is identical to that of the $A_0$-algebras finite and separable, hence isomorphic to direct compositions of fields, separable finite extensions of $A_0$.
In particular, if the field $A_0$ is \emph{separably closed}, these \'etale coverings are all identical to $A_0$, and hence every \'etale covering of $\operatorname{Spec}(A)$ is \emph{trivial} \sref{IV.18.2.7} by virtue of \sref{IV.18.3.2}.
\end{example}

\begin{theorem}[18.3.4]
\label{IV.18.3.4}
Let $A$ be a Noetherian ring, $\mathfrak{J}$ an ideal of $A$ such that $A$ is separated and complete for the $\mathfrak{J}$-preadic topology, $A_0=A/\mathfrak{J}$.
Set $S=\operatorname{Spec}(A)$, $S_0=\operatorname{Spec}(A_0)$.
Let $X$ be an S-scheme proper on $S$, and set $X_0=X\times_X X_0$.
Then the functor
\[
Z\mapsto Z\times_X X_0
\]
from the category of X-schemes finite and \'etale on $X$, to the category of $X_0$-schemes \'etale and finite on $X_0$, is an equivalence of categories.
\end{theorem}


\subsection{Local structure of non ramified and \'etale morphisms.}
\label{subsection:IV.18.4}

\begin{lemma}[18.4.1]
\label{IV.18.4.1}
Let $A$ be a ring, $B$ an A-algebra finite monogene, $u$ a generator of the A-algebra $B$, $F\in A[\mathrm{T}]$ a polynomial such that $F(u)=0$, $F'$ the derived polynomial; set $u'=F'(u)$.
Then the Ideal of $B$, annihilator of $\Omega^1_{B/A}$, contains $u'B$; it is equal to $u'B$ if the ideal $\mathfrak{J}$ of $A[\mathrm{T}]$ consisting of the polynomials $G$ such that $G(u)=0$, is engendered by $F$, in other words if the surjective canonical homomorphism $\varphi:A[\mathrm{T}]/F.A[\mathrm{T}]\to B$ sending the image of $\mathrm{T}$ to $u$, is bijective.
\end{lemma}

\begin{proposition}[18.4.2]
\label{IV.18.4.2}
With the notations of \sref{IV.18.4.1}, let $\mathfrak{q}$ be a prime ideal of $B$.
Then:
\begin{enumerate}[label=(\roman*)]
  \item If $\mathfrak{q}$ does not contain $u'$, $B_{\mathfrak{q}}$ is an $A_{\mathfrak{p}}$-algebra formally non ramified ($\mathfrak{p}$ being the inverse image of $\mathfrak{q}$ in $A$); in other words, $\operatorname{Spec}(B_u)$ is formally non ramified on $\operatorname{Spec}(A)$.
  \item Suppose moreover that $F$ is unitary and engenders $\mathfrak{J}$.
  Then, for $B$ to be an A-algebra \'etale on $\operatorname{Spec}(A)$ at the point $\mathfrak{q}$, it is necessary and sufficient that $u'\notin\mathfrak{q}$.
\end{enumerate}
\end{proposition}

\begin{corollary}[18.4.3]
\label{IV.18.4.3}
With the notations of \sref{IV.18.4.2}, suppose that $F$ is unitary and engenders $\mathfrak{J}$.
Then, for $B$ to be an A-algebra \'etale, it is necessary and sufficient that $u'$ be inversible in $B$ (or, what amounts to the same, that the ideal of $A[\mathrm{T}]$ engendered by $F$ and $F'$ be equal to $A[\mathrm{T}]$).

One says that a unitary polynomial $F\in A[\mathrm{T}]$ such that the ideal of $A[\mathrm{T}]$ engendered by $F$ and $F'$ is equal to $A[\mathrm{T}]$ is \emph{separable}; it is immediate that this definition coincides with the usual definition (Bourbaki, \emph{Alg.}, chap.~V, \textsection~7, n\textsuperscript{o}~6) when $A$ is a field.
\end{corollary}

\begin{lemma}[18.4.4]
\label{IV.18.4.4}
Let $A$ be a local ring, $B$ an A-algebra finite monogene, $u$ a generator of the A-algebra $B$.
Let $\mathfrak{n}_i$ $(1\leq i\leq r)$ be the maximal ideals of $B$ such that $\operatorname{Spec}(B)$ is formally non ramified on $\operatorname{Spec}(A)$ at the points $\mathfrak{n}_i$.
Then there exists a unitary polynomial $F\in A[\mathrm{T}]$ such that $F(u)=0$ and $F'(u)\notin\mathfrak{n}_i$ for every index $i$ $(1\leq i\leq r)$.
Moreover, if $k$ is the residue field of $A$ and $f\in k[\mathrm{T}]$ the minimal polynomial of the image of $u$ in $B\otimes_A k$ over $k$, there exists $F\in A[\mathrm{T}]$ satisfying all the conditions, whose image canonical is $f$; and $F$ is necessarily of degree $n$.
\end{lemma}

\begin{proposition}[18.4.5]
\label{IV.18.4.5}
Let $A$ be a local ring, $k$ its residue field, $B$ an A-algebra finite (resp.\ of finite presentation).
Suppose moreover, either that the field $k$ is infinite, or that $B$ is a local ring.
Let $n$ be the rank of $L=B\otimes_A k$ over $k$.
For $B$ to be an A-algebra formally non ramified (resp.\ \'etale), it is necessary and sufficient that there exist a unitary separable polynomial $F\in A[\mathrm{T}]$ \sref{IV.18.4.3} such that $B$ is isomorphic to a quotient of $A[\mathrm{T}]/F.A[\mathrm{T}]$ (resp.\ isomorphic to $A[\mathrm{T}]/F.A[\mathrm{T}]$).
Moreover, one can suppose that $F$ is of degree $n$ (resp.\ $F$ is necessarily of degree $n$).
\end{proposition}

\begin{theorem}[18.4.6]
\label{IV.18.4.6}
(Chevalley).
\begin{enumerate}[label=(\roman*)]
  \item Let $f:X\to Y$ be a morphism locally of finite type, $x$ a point of $X$, $y=f(x)$, and set $A=\sh{O}_{Y,y}$.
  For $\sh{O}_{X,x}$ to be an A-algebra formally smooth (resp.\ formally non ramified), it is necessary and sufficient that there exist a unitary polynomial $F\in A[\mathrm{T}]$ and a maximal ideal $\mathfrak{n}$ of $B=A[\mathrm{T}]/F.A[\mathrm{T}]$ (resp.\ an algebra quotient $B$ of $A[\mathrm{T}]/F.A[\mathrm{T}]$) such that $\sh{O}_{X,x}$ is A-isomorphic to $B_{\mathfrak{n}}$ and that, if $u$ is the image of $\mathrm{T}$ in $B$, one has $F'(u)\notin\mathfrak{n}$.
  \item Suppose moreover that $f$ is locally of finite presentation.
  Then, for $f$ to be \'etale at the point $x$, it is necessary and sufficient that one can take $B=A[\mathrm{T}]/F.A[\mathrm{T}]$.
\end{enumerate}
\end{theorem}

\begin{corollary}[18.4.7]
\label{IV.18.4.7}
Let $f:X\to Y$ be a morphism locally of finite type, $x$ a point of $X$.
For $f$ to be formally non ramified at the point $x$, it is necessary and sufficient that there exist an open neighbourhood $U$ of $x$ such that $f|U$ factors as $U\xrightarrow{j}X'\xrightarrow{h}Y$, where $h$ is \'etale and $j$ is a closed immersion.
\end{corollary}
