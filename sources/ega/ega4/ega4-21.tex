\section{Divisors}
\label{section:IV.21}

For the content of the present paragraph, see the comments in the introduction of \textsection20.
For the \emph{global} properties of divisors, the reader will refer to the paragraph dedicated to them in chap.~V.

\subsection{Divisors on a ringed space}
\label{IV.21.1}

\begin{env}[21.1.1]
\label{IV.21.1.1}
\oldpage[IV-4]{255}
Let $(X,\sh{O}_X)$ be a ringed space, $\sh{M}_X$ the sheaf of germs of meromorphic functions on $X$ \sref{IV.20.1.3}, $\sh{M}_X^*$ the sheaf (of multiplicative groups) of germs of regular meromorphic functions on $X$ \sref{IV.20.1.8}.
It is clear that the sheaf (of multiplicative groups) $\sh{O}_X^*$ of germs of \emph{invertible} sections of $\sh{O}_X$ identifies canonically with a sub-sheaf (of multiplicative groups) of $\sh{M}_X^*$.
\end{env}

\begin{definition}[21.1.2]
\label{IV.21.1.2}
We call \emph{sheaf of divisors} on $X$ and denote $\sh{D}iv_X$ the quotient sheaf (of commutative groups) $\sh{M}_X^*/\sh{O}_X^*$;
the sections of this sheaf over $X$ are called the \emph{divisors} on $X$;
they form a commutative group denoted $\operatorname{Div}(X)$.
For every section $f$ of $\sh{M}_X^*$ over $X$ (in other words, every \emph{regular meromorphic function} on $X$ \sref{IV.20.1.8}) we call the \emph{divisor of $f$} and denote $\operatorname{div}(f)$ (or $\operatorname{div}_X(f)$) the divisor on $X$ which is the image of $f$ by the canonical homomorphism $\Gamma(X,\sh{M}_X^*)\to\Gamma(X,\sh{D}iv_X)$.
\end{definition}

\oldpage[IV-4]{256}
The \emph{support} of a divisor $D$ is the closed set of $x\in X$ such that $D_x\neq 0$.
We denote this by $\operatorname{Supp}(D)$.

For every open $U$ of $X$, one has evidently $\sh{M}_X^*|U=\sh{M}_U^*$, $\sh{O}_X^*|U=\sh{O}_U^*$, hence $\sh{D}iv_X|U=\sh{D}iv_U$, and consequently the sheaf $\sh{D}iv_X$ is equal to the presheaf $U\mapsto\operatorname{Div}(U)$.
When $X=\Spec(A)$ is affine, we write $\operatorname{Div}(A)$ instead of $\operatorname{Div}(\Spec(A))$.

\begin{env}[21.1.3]
\label{IV.21.1.3}
We will always write \emph{additively} the group $\operatorname{Div}(X)$ of divisors on $X$.
For two regular meromorphic functions $f$, $g$ on $X$, one thus has
\[
\label{IV.21.1.3.1}
  \operatorname{div}(fg)=\operatorname{div}(f)+\operatorname{div}(g),
  \tag{21.1.3.1}
\]
\[
\label{IV.21.1.3.2}
  \operatorname{div}(f^{-1})=-\operatorname{div}(f).
  \tag{21.1.3.2}
\]

By definition, for every regular meromorphic function $f$ on $X$, one has the equivalence
\[
\label{IV.21.1.3.3}
  \operatorname{div}(f)=0\iff f\in\Gamma(X,\sh{O}_X^*),
  \tag{21.1.3.3}
\]
whence, for two regular meromorphic functions $f$, $g$ on $X$,
\[
\label{IV.21.1.3.4}
  \operatorname{div}(f)=\operatorname{div}(g)\iff fg^{-1}\in\Gamma(X,\sh{O}_X^*).
  \tag{21.1.3.4}
\]
\end{env}

\begin{env}[21.1.4]
\label{IV.21.1.4}
Let now $\sh{L}$ be an $\sh{O}_X$-Module \emph{invertible}, and let $s$ be a \emph{regular} meromorphic section of $\sh{L}$ over $X$ \sref{IV.20.1.8}.
Every $x\in X$ possesses a neighbourhood $U$ such that $\sh{L}|U$ is isomorphic to $\sh{O}_U$, hence $\sh{M}_X(\sh{L})|U$ isomorphic to $\sh{M}_X|U$;
by one of these isomorphisms, $s|U$ corresponds to a section $f\in\Gamma(U,\sh{M}_X^*)$, and since two such isomorphisms differ only by multiplication by an element of $\Gamma(U,\sh{O}_X^*)$ $(\mathbf{0_I}$, 5.4.3), the element $\operatorname{div}_U(f)$ of $\Gamma(U,\sh{D}iv_X)$ is \emph{independent} of the isomorphism chosen;
it is clear that these elements (for $U$ variable) are the restrictions of a section of $\sh{D}iv_X$ over $X$, that we call the \emph{divisor of $s$} and that we denote $\operatorname{div}(s)$ (such a divisor is not necessarily of the form $\operatorname{div}(g)$ for a regular meromorphic function $g$ on $X$;
see \sref{IV.21.2.9}).
For $\sh{L}=\sh{O}_X$, the definition of $\operatorname{div}(s)$ coincides with that of \sref{IV.21.1.2}.
If $\sh{L}$, $\sh{L}'$ are two $\sh{O}_X$-Modules invertibles, $s$ (resp.\ $s'$) a regular meromorphic section of $\sh{L}$ (resp.\ $\sh{L}'$) over $X$, it is immediate that one has
\[
\label{IV.21.1.4.1}
  \operatorname{div}(s\otimes s')=\operatorname{div}(s)+\operatorname{div}(s')
  \tag{21.1.4.1}
\]
\[
\label{IV.21.1.4.2}
  \operatorname{div}(s^{\otimes n})=n\cdot\operatorname{div}(s)\quad\text{for all }n\in\bb{Z}
  \tag{21.1.4.2}
\]
($s^{-1}$ being the regular meromorphic section of $\sh{L}^{-1}$ over $X$ defined in \sref{IV.20.1.10})
and, for two regular meromorphic sections $s$, $s'$ of $\sh{L}$ over $X$, one has the relation
\[
\label{IV.21.1.4.3}
  \operatorname{div}(s)=\operatorname{div}(s')\iff s'=ts\quad\text{with }t\in\Gamma(X,\sh{O}_X^*).
  \tag{21.1.4.3}
\]
\end{env}

\begin{env}[21.1.5]
\label{IV.21.1.5}
The sheaf $\sh{S}(\sh{O}_X)$ \sref{IV.20.1.3} whose sections over an open $U$ of $X$ are the regular elements of $\Gamma(U,\sh{O}_X)$ is a \emph{sub-sheaf of monoids} of $\sh{M}_X^*$;
one can write
\[
\label{IV.21.1.5.1}
  \sh{S}(\sh{O}_X)=\sh{O}_X\cap\sh{M}_X^*.
  \tag{21.1.5.1}
\]

\oldpage[IV-4]{257}
If we denote $\sh{S}(\sh{O}_X)^{-1}$ the sheaf whose sections over $U$ are the \emph{inverses} in $\Gamma(U,\sh{M}_X^*)$ of the elements of $\Gamma(U,\sh{S}(\sh{O}_X))$, it is clear that one has $\Gamma(U,\sh{S}(\sh{O}_X))\cap\Gamma(U,\sh{S}(\sh{O}_X)^{-1})=\Gamma(U,\sh{O}_X^*)$, hence
\[
\label{IV.21.1.5.2}
  \sh{S}(\sh{O}_X)\cap\sh{S}(\sh{O}_X)^{-1}=\sh{O}_X^*.
  \tag{21.1.5.2}
\]
\end{env}

\begin{definition}[21.1.6]
\label{IV.21.1.6}
The sub-sheaf of sets of $\sh{D}iv_X$, canonical image of the sub-sheaf $\sh{S}(\sh{O}_X)$ of $\sh{M}_X^*$, is denoted $\sh{D}iv_X^+$;
its sections over $X$ are called \emph{positive divisors} on $X$ and their set is denoted $\operatorname{Div}^+(X)$.
\end{definition}

Since $\sh{S}(\sh{O}_X)$ is a sheaf of monoids (multiplicative), one has
\[
\label{IV.21.1.6.1}
  \operatorname{Div}^+(X)+\operatorname{Div}^+(X)\subset\operatorname{Div}^+(X)
  \tag{21.1.6.1}
\]
and on the other hand, by virtue of \sref{IV.21.1.5.2} and \sref{IV.21.1.3.3},
\[
\label{IV.21.1.6.2}
  \operatorname{Div}^+(X)\cap(-\operatorname{Div}^+(X))=\{0\}.
  \tag{21.1.6.2}
\]

These two relations show that $\operatorname{Div}^+(X)$ is the set of positive elements for a structure of \emph{order} on the group $\operatorname{Div}(X)$, compatible with this group structure;
we denote this relation of order $D\leq D'$, in other words, one has
\[
\label{IV.21.1.6.3}
  D\geq 0\iff D\in\operatorname{Div}^+(X).
  \tag{21.1.6.3}
\]

We will always assume in what follows that $\operatorname{Div}(X)$ is equipped with this order structure.
The fibre $(\sh{D}iv_X^+)_x$ at a point $x$ of the sheaf $\sh{D}iv_X^+$ is a sub-monoid of the group $(\sh{D}iv_X)_x$, the set of elements $\geq 0$ for an order structure compatible with the group structure;
for a divisor $D$ on $X$, it amounts to the same to say that $D\geq 0$ or that $D_x\geq 0$ for every $x\in X$.

By definition, for every regular meromorphic function $f$ on $X$, one has the relation
\[
\label{IV.21.1.6.4}
  \operatorname{div}(f)\geq 0\iff f\in\Gamma(X,\sh{S}(\sh{O}_X))
  \tag{21.1.6.4}
\]
in other words, $\operatorname{div}(f)\geq 0$ means that $f$ is a \emph{regular} section of $\sh{O}_X$, or equivalently a section of $\sh{O}_X$ invertible in $M(X)$.

More generally, given a divisor $D$ on $X$, the relation $\operatorname{div}(f)\geq D$ is equivalent to the following: for every open $U$ of $X$ such that $D|U=\operatorname{div}(g)$, where $g\in\Gamma(U,\sh{M}_X^*)$, there exists a regular element $t$ of $\Gamma(U,\sh{O}_X)$ such that $f|U=tg$.

\begin{env}[21.1.7]
\label{IV.21.1.7}
Let $\sh{L}$ be an $\sh{O}_X$-Module invertible, $s$ a regular meromorphic section of $\sh{L}$ over $X$;
one has the relation
\[
\label{IV.21.1.7.1}
  \operatorname{div}(s)\geq 0\iff s\in\Gamma(X,\sh{L})\cap\Gamma(X,(\sh{M}_X(\sh{L}))^*)
  \tag{21.1.7.1}
\]
as it results immediately from the definitions \sref{IV.21.1.4} and \sref{IV.21.1.6}.
\end{env}

\begin{proposition}[21.1.8]
\label{IV.21.1.8}
Let $X$ be a locally Noetherian prescheme, $D$ a divisor on $X$.
Suppose that for every $x\in X$ such that $\operatorname{prof}(\sh{O}_{X,x})=1$ one has $D_x\geq 0$ (resp.\ $D_x=0$).
Then one has $D\geq 0$ (resp.\ $D=0$).
\end{proposition}

\begin{proof}
\oldpage[IV-4]{258}
The question being local on $X$, we can suppose that $D=\operatorname{div}(f)$, $f$ being a regular meromorphic function on $X$;
the relation $D_x\geq 0$ is equivalent to $x\in\operatorname{dom}(f)$, hence the hypothesis signifies that if $T=X-\operatorname{dom}(f)$, one has $\operatorname{prof}(\sh{O}_{X,x})\geq 2$ for every $x\in T$ (since $\operatorname{dom}(f)$ contains the maximal points of $X$).
By \sref{IV.5.10.5} the restriction homomorphism $\Gamma(X,\sh{O}_X)\to\Gamma(X-T,\sh{O}_X)$ is \emph{bijective}, which shows that there exists a section $s$ of $\sh{O}_X$ over $X$ such that $f=s|(X-T)$.
But by definition of $T$, this implies $T=\varnothing$, hence $f=s$ and $D\geq 0$.
The assertion relative to $D_x=0$ follows immediately by applying the preceding to $-D$.
\end{proof}

\begin{corollary}[21.1.9]
\label{IV.21.1.9}
Let $X$ be a locally Noetherian prescheme, $D$ a divisor on $X$.
Let $S$ be the support of $D$.
Then, for every maximal point $x$ of $S$, $\operatorname{prof}(\sh{O}_{X,x})=1$.
\end{corollary}

\begin{proof}
Set $X_1=\Spec(\sh{O}_{X,x})$;
taking into account \sref{IV.20.3.6}, the sheaf $\sh{M}_{X_1}^*$ is induced on $X_1$ by $\sh{M}_X^*$, and we can restrict to the case where $X=X_1$, to which case, as $x\in S$ and $x$ is a maximal point of $S$, $S$ is necessarily $S=\{x\}$.
If one had then $\operatorname{prof}(\sh{O}_{X,x})\neq 1$, we would conclude, by virtue of \sref{IV.21.1.8}, that $D=0$, which contradicts the definition of $S$.
\end{proof}

\begin{proposition}[21.1.10]
\label{IV.21.1.10}
Let $A$ be a local Noetherian ring;
for $\operatorname{Div}(A)=0$, it is necessary and sufficient that $\operatorname{prof}(A)=0$ (in other words, that the maximal ideal $\mathfrak{m}$ of $A$ be associated to $A$ $(\mathbf{0}$, 16.4.6)).
\end{proposition}

\begin{proof}
Indeed, to say that $\operatorname{Div}(A)=0$ means that every regular element is invertible, or equivalently that all elements of $\mathfrak{m}$ are zero divisors, which means that $\mathfrak{m}\in\operatorname{Ass}(A)$ (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, n\textsuperscript{o}~1, cor.~3 of prop.~2).
\end{proof}

\subsection{Divisors and invertible fractional ideals}
\label{IV.21.2}

\begin{env}[21.2.1]
\label{IV.21.2.1}
Let $(X,\sh{O}_X)$ be a ringed space.
We call \emph{fractional Ideal} on $X$ a sub-$\sh{O}_X$-Module of the $\sh{O}_X$-Module $\sh{M}_X$ of germs of meromorphic functions on $X$.
A fractional Ideal $\sh{J}$ on $X$ which is an $\sh{O}_X$-Module \emph{invertible} is called a \emph{fractional Ideal invertible}.
\end{env}

\begin{proposition}[21.2.2]
\label{IV.21.2.2}
For a fractional Ideal $\sh{J}$ on $X$ to be invertible, it is necessary and sufficient that for every $x\in X$, there exists a neighbourhood $U$ of $x$ and a section $f\in\Gamma(U,\sh{M}_X^*)$ such that $\sh{J}|U=\sh{O}_U f$.
\end{proposition}

\begin{proof}
The condition is evidently sufficient, the map $s\mapsto s(f|V)$ from $\Gamma(V,\sh{O}_X)$ being evidently bijective for every open $V$ contained in $U$.
To see that it is necessary, note that there exists by hypothesis a neighbourhood $U$ of $x$ and an isomorphism of $\sh{O}_X$-Modules $\sh{O}_U\xrightarrow{\sim}\sh{J}|U$.
If $f$ is the image of the section $1\in\Gamma(U,\sh{O}_X)$ by this isomorphism, we can suppose, restricting $U$, that $f=u/s$, where $u\in\Gamma(U,\sh{O}_X)$ and $s\in\Gamma(U,\sh{S}(\sh{O}_X))$, and the isomorphism considered corresponds, to every section $v\in\Gamma(V,\sh{O}_X)$ (where $V$ is an open contained in $U$) the homomorphism $t(f|V)\mapsto st$ from $\Gamma(V,\sh{J})$ to $\Gamma(V,\sh{O}_X)$;
to say that this map is bijective means that $u|V$ is a regular element of $\Gamma(V,\sh{O}_X)$, hence $f\in\Gamma(U,\sh{M}_X^*)$.

\oldpage[IV-4]{259}
We note that for every open $U$ of $X$ such that $\sh{J}|U=\sh{O}_U f$ with $f\in\Gamma(U,\sh{M}_X^*)$, the section $f$ is determined up to \emph{multiplication by} an element of $\Gamma(U,\sh{O}_X^*)$, since the multiplication by these elements provides all automorphisms of the $\sh{O}_U$-Module $\sh{O}_U$ $(\mathbf{0_I}$, 5.4.3).
\end{proof}

\begin{corollary}[21.2.3]
\label{IV.21.2.3}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item Let $\sh{J}$ be an invertible fractional Ideal;
  then the $\sh{O}_X$-Module invertible $\sh{J}^{-1}$ identifies canonically with the fractional Ideal $\sh{J}'$ (transporter of $\sh{J}$ in $\sh{O}_X$) defined as follows:
  for every open $U$ of $X$ such that $\sh{J}|U=\sh{O}_U f$, where $f\in\Gamma(U,\sh{M}_X^*)$, one has $\sh{J}'|U=\sh{O}_U f^{-1}$.
  \item If $\sh{J}_1$ and $\sh{J}_2$ are two invertible fractional Ideals, the canonical map $\sh{J}_1\otimes\sh{J}_2\to\sh{J}_1\sh{J}_2$ is an isomorphism of $\sh{O}_X$-Modules.
\end{enumerate}
\end{corollary}

\begin{proof}
Assertion~(ii) results immediately from \sref{IV.21.2.2}.
On the other hand, the remark made at the end of \sref{IV.21.2.2} proves that there exists a well-defined fractional Ideal $\sh{J}'$ satisfying the condition of the statement;
the isomorphism canonique of $\sh{J}'$ on $\sh{J}^{-1}=\sh{H}om_{\sh{O}_X}(\sh{J},\sh{O}_X)$ is obtained by making correspond, to every section $s(f^{-1}|V)$ of $\sh{J}^{-1}|V$ (where $V$ is an open contained in $U$ and $s\in\Gamma(V,\sh{O}_X)$) the homomorphism $t(f|V)\mapsto st$ from $\Gamma(V,\sh{J})$ to $\Gamma(V,\sh{O}_X)$.

By virtue of \sref{IV.21.2.3}[(i)], we will henceforth identify generally the $\sh{O}_X$-Modules $\sh{J}$ and $\sh{J}^{-1}$, considering $\sh{J}^{-1}$ as a sub-$\sh{O}_X$-Module of $\sh{M}_X$.
\end{proof}

\begin{env}[21.2.4]
\label{IV.21.2.4}
It results from \sref{IV.21.2.3} that the set $\operatorname{Id.inv}(X)$ of invertible fractional Ideals on $X$ is equipped with a structure of \emph{commutative group} for the law of composition $(\sh{J}_1,\sh{J}_2)\mapsto\sh{J}_1\sh{J}_2$, the neutral element of this group being $\sh{O}_X$.
It is clear that for every open $U$ of $X$, one has $(\sh{J}_1\sh{J}_2)|U=(\sh{J}_1|U)(\sh{J}_2|U)$, hence the map of restriction $\sh{J}\mapsto\sh{J}|U$ is a homomorphism of groups $\operatorname{Id.inv}(X)\to\operatorname{Id.inv}(U)$;
we thus define a \emph{presheaf of commutative groups} $U\mapsto\operatorname{Id.inv}(U)$;
it is immediate that in fact this presheaf is a \emph{sheaf of commutative groups}, that we denote $\sh{J}d.\!\operatorname{inv}_X$.
\end{env}

\begin{env}[21.2.5]
\label{IV.21.2.5}
For every regular meromorphic function $f\in\Gamma(X,\sh{M}_X^*)$, it results from \sref{IV.21.2.2} that $\sh{J}(f)=\sh{O}_X f$ is an invertible fractional Ideal, and one has evidently $\sh{J}(f_1 f_2)=\sh{J}(f_1)\sh{J}(f_2)$, so the map $f\mapsto\sh{J}(f)$ is a homomorphism of the commutative group $\Gamma(X,\sh{M}_X^*)$ in the commutative group $\operatorname{Id.inv}(X)$.
Replacing $X$ by an arbitrary open $U$ of $X$ and noting that the homomorphisms obtained are compatible with the restriction operations, one obtains a canonical homomorphism of sheaves of commutative groups:
\[
\label{IV.21.2.5.1}
  I_0:\sh{M}_X^*\to\sh{J}d.\!\operatorname{inv}_X.
  \tag{21.2.5.1}
\]

Note that if $f\in\Gamma(X,\sh{O}_X^*)$, one has $\sh{J}(f)=\sh{O}_X$;
we deduce immediately that the homomorphism $I_0$ factorises as
\[
\label{IV.21.2.5.2}
  \sh{M}_X^*\to\sh{M}_X^*/\sh{O}_X^*=\sh{D}iv_X\xrightarrow{I}\sh{J}d.\!\operatorname{inv}_X
  \tag{21.2.5.2}
\]
where $I$ is a homomorphism of the (additive) sheaf of groups $\sh{D}iv_X$ in the (multiplicative) sheaf $\sh{J}d.\!\operatorname{inv}_X$;
one has therefore for every open $U$ a homomorphism $\sh{J}_U:\operatorname{Div}(U)\to\operatorname{Id.inv}(U)$ of commutative groups, such that, for every section $f\in\Gamma(U,\sh{M}_X^*)$, one has
\[
\label{IV.21.2.5.3}
  \sh{J}_U(\operatorname{div}_U(f))=\sh{O}_U f.
  \tag{21.2.5.3}
\]

\oldpage[IV-4]{260}
We conclude that $\sh{J}_X(D)$, for every divisor $D\in\operatorname{Div}(X)$, is the invertible fractional Ideal defined in the following way: for every open $U$ of $X$ such that $D|U=\operatorname{div}_U(f)$, where $f\in\Gamma(U,\sh{M}_X^*)$, $\sh{J}_X(D)|U$ is the invertible fractional Ideal $\sh{O}_U\cdot f$.
We thus have, by virtue of \sref{IV.21.1.6}, for every regular meromorphic function $f\in\Gamma(X,\sh{M}_X^*)$, the relation
\[
\label{IV.21.2.5.4}
  f\in\Gamma(X,\sh{J}_X(D))\iff\operatorname{div}(f)\geq D.
  \tag{21.2.5.4}
\]
\end{env}

\begin{proposition}[21.2.6]
\label{IV.21.2.6}
The homomorphism $I:\sh{D}iv_X\to\sh{J}d.\!\operatorname{inv}_X$ is bijective.
\end{proposition}

\begin{proof}
We define indeed a homomorphism $I'_X$ of $\operatorname{Id.inv}(X)$ in $\operatorname{Div}(X)$ by making correspond to every invertible fractional Ideal $\sh{J}$ on $X$ the divisor $I'_X(\sh{J})$ defined in the following way: for every open $U$ of $X$ such that $\sh{J}|U=\sh{O}_U f$, where $f\in\Gamma(U,\sh{M}_X^*)$ \sref{IV.21.2.2}, we take $I'_X(\sh{J})|U=\operatorname{div}_U(f)$;
by virtue of the remark following \sref{IV.21.2.2}, this definition is independent of the generator $f$ chosen in $\sh{J}|U$, and determines well a divisor on $X$.
Furthermore, this definition shows immediately that the homomorphisms $\sh{J}_X$ and $I'_X$ are reciprocal of one another.
Replacing $X$ by an arbitrary open $U$, we deduce the definition of the isomorphism $I':\sh{J}d.\!\operatorname{inv}_X\to\sh{D}iv_X$, reciprocal of $I$, whence the proposition.
We set $I'_X(\sh{J})=\operatorname{div}(\sh{J})$, so that one has, for every regular meromorphic function $f$ on $X$
\[
\label{IV.21.2.6.1}
  \operatorname{div}(\sh{O}_X\cdot f)=\operatorname{div}(f).
  \tag{21.2.6.1}
\]
\end{proof}

\begin{env}[21.2.7]
\label{IV.21.2.7}
We will often identify the sheaves $\sh{D}iv_X$ and $\sh{J}d.\!\operatorname{inv}_X$ (resp.\ the groups $\operatorname{Div}(X)$ and $\operatorname{Id.inv}(X)$) by means of the isomorphisms $I$ and $I'$ (resp.\ $\sh{J}_X$ and $I'_X$) above.
We note that one has the relation
\[
\label{IV.21.2.7.1}
  D\geq 0\iff\sh{J}_X(D)\subset\sh{O}_X\quad\text{for }D\in\operatorname{Div}(X)
  \tag{21.2.7.1}
\]
as it results immediately from the definitions \sref{IV.21.1.6} and from \sref{IV.21.1.5.1};
in other terms, the image $\sh{J}_X(\operatorname{Div}^+(X))$ is the set of Ideals of $\sh{O}_X$ (also sometimes called \emph{integral} Ideals) which are $\sh{O}_X$-Modules invertibles: such an Ideal $\sh{J}$ is further characterised by the fact that for every $x\in X$, there exists an open neighbourhood $U$ of $x$ in $X$ such that $\sh{J}|U=\sh{O}_U f$, where $f$ is a regular element of $\Gamma(U,\sh{O}_X)$.
The set $\sh{J}_X(\operatorname{Div}^+(X))$ of these Ideals is thus a sub-monoid of $\operatorname{Id.inv}(X)$, equal to the set of positive elements for an order relation compatible with the group structure of $\operatorname{Id.inv}(X)$, and it is immediate that this relation is none other than the relation \emph{opposite} to inclusion;
in other words, one has
\[
\label{IV.21.2.7.2}
  \sh{J}_1\subset\sh{J}_2\iff\operatorname{div}(\sh{J}_1)\geq\operatorname{div}(\sh{J}_2)
  \tag{21.2.7.2}
\]
for $\sh{J}_1$, $\sh{J}_2$ in $\operatorname{Id.inv}(X)$.
\end{env}

\begin{env}[21.2.8]
\label{IV.21.2.8}
For every divisor $D$ on $X$, we set
\[
\label{IV.21.2.8.1}
  \sh{O}_X(D)=(\sh{J}_X(D))^{-1};
  \tag{21.2.8.1}
\]
$\sh{O}_X(D)$ is thus an invertible fractional Ideal, defined in the following way: for every open $U$ of $X$ such that $D|U=\operatorname{div}_U(f)$, where $f\in\Gamma(U,\sh{M}_X^*)$, $\sh{O}_X(D)|U$ is the invertible fractional Ideal $\sh{O}_U\cdot f^{-1}$;
by virtue of \sref{IV.21.1.6}, for every regular meromorphic function $f$, one has the relation
\[
\label{IV.21.2.8.2}
  f\in\Gamma(X,\sh{O}_X(D))\iff\operatorname{div}(f)\geq -D.
  \tag{21.2.8.2}
\]

\oldpage[IV-4]{261}
Furthermore, it is clear that one has canonical isomorphisms \sref{IV.21.2.3}
\[
\label{IV.21.2.8.3}
  \begin{cases}
    \sh{O}_X(0)=\sh{O}_X,\quad\sh{O}_X(D+D')=\sh{O}_X(D)\sh{O}_X(D')\simeq\sh{O}_X(D)\otimes\sh{O}_X(D')\\
    \sh{O}_X(nD)=(\sh{O}_X(D))^n\simeq(\sh{O}_X(D))^{\otimes n}
  \end{cases}
  \tag{21.2.8.3}
\]
for every integer $n\in\bb{Z}$, and for any two divisors $D$, $D'$ on $X$.
\end{env}

\begin{env}[21.2.9]
\label{IV.21.2.9}
Let $\sh{J}$ be an invertible fractional Ideal on $X$.
The canonical injection $\sh{J}\to\sh{M}_X$ defines by tensorisation a homomorphism of $\sh{O}_X$-Modules
\[
\label{IV.21.2.9.1}
  \sh{M}_X(\sh{J})=\sh{J}\otimes_{\sh{O}_X}\sh{M}_X\to\sh{M}_X\otimes_{\sh{O}_X}\sh{M}_X=\sh{M}_X
  \tag{21.2.9.1}
\]
which is an \emph{isomorphism}: indeed, if $U$ is an open of $X$ such that $\sh{J}|U=\sh{O}_U\cdot f$, where $f\in\Gamma(U,\sh{M}_X^*)$, the homomorphism \sref{IV.21.2.9.1} restricted to $U$ is none other than the isomorphism which to every section $t$ of $\sh{M}_X(\sh{J})|U=\sh{M}_X|U$ over $V\subset U$, makes correspond the section $t(f|V)^{-1}$ of the same sheaf.
In the isomorphism \sref{IV.21.2.9.1}, to the regular meromorphic sections of $\sh{J}$ over $X$ correspond the regular meromorphic functions on $X$.

Consider in particular the case where $\sh{J}=\sh{O}_X(D)$, where $D$ is a divisor on $X$;
one then has a canonical isomorphism
\[
\label{IV.21.2.9.2}
  \sh{M}_X(\sh{O}_X(D))\simeq\sh{M}_X
  \tag{21.2.9.2}
\]
and one denotes by $s_D$ the regular meromorphic section of $\sh{O}_X(D)$ over $X$ which corresponds by this isomorphism to the section $1$ of $\sh{M}_X$.
If $U$ is an open of $X$ such that $\sh{O}_X(D)|U=\sh{O}_U\cdot f^{-1}$, where $f\in\Gamma(U,\sh{M}_X^*)$, one has $s_D|U=1$ in $\Gamma(U,\sh{M}_X)$;
since then $D|U=\operatorname{div}_U(f)$, we deduce \sref{IV.21.1.4} that one has
\[
\label{IV.21.2.9.3}
  \operatorname{div}(s_D)=D.
  \tag{21.2.9.3}
\]

On the other hand, one deduces immediately from the canonical isomorphisms \sref{IV.21.2.8.3} the formulas
\[
\label{IV.21.2.9.4}
  s_0=1,\quad s_{D+D'}=s_D\otimes s_{D'},\quad s_{nD}=s_D^{\otimes n}\quad(n\in\bb{Z}).
  \tag{21.2.9.4}
\]
\end{env}

\begin{env}[21.2.10]
\label{IV.21.2.10}
Consider, between two couples $(\sh{L},s)$, $(\sh{L}',s')$, where $\sh{L}$ and $\sh{L}'$ are two $\sh{O}_X$-Modules invertibles, $s$ (resp.\ $s'$) a regular meromorphic section of $\sh{L}$ (resp.\ $\sh{L}'$) over $X$, the relation: ``there exists an isomorphism $u:\sh{L}\xrightarrow{\sim}\sh{L}'$ such that $\overline{u}(s)=s'$'', where $\overline{u}:\Gamma(X,\sh{M}_X(\sh{L}))\xrightarrow{\sim}\Gamma(X,\sh{M}_X(\sh{L}'))$ is the isomorphism deduced from $u$ (we note that the isomorphism $u$ satisfying this condition is then determined in a unique way).
It is clear that this is an equivalence relation, and since there exists a set of $\sh{O}_X$-Modules invertibles such that every $\sh{O}_X$-Module invertible is isomorphic to an element of this set $(\mathbf{0_I}$, 5.4.7), one can speak of the \emph{set} $\operatorname{D}(X)$ of \emph{equivalence classes} of couples $(\sh{L},s)$ for the preceding relation.
For every $\sh{O}_X$-Module invertible $\sh{L}$ and every regular meromorphic section $s$ of $\sh{L}$ over $X$, one denotes $\operatorname{cl}(\sh{L},s)$ the element of $\operatorname{D}(X)$ corresponding to the couple $(\sh{L},s)$.
It results from $(\mathbf{0_I}$, 5.4.3) that if $s$, $s'$ are two regular meromorphic sections of $\sh{L}$ over $X$, the relation $\operatorname{cl}(\sh{L},s)=\operatorname{cl}(\sh{L},s')$ is equivalent to the existence of a section $t\in\Gamma(X,\sh{O}_X^*)$ such that $s'=ts$.

\oldpage[IV-4]{262}
It is immediate that if $(\sh{L},s)$ is equivalent to $(\sh{L}_1,s_1)$ and $(\sh{L}',s')$ to $(\sh{L}'_1,s'_1)$, the couples $(\sh{L}\otimes\sh{L}',s\otimes s')$ and $(\sh{L}_1\otimes\sh{L}'_1,s_1\otimes s'_1)$ are equivalent;
one defines thus in $\operatorname{D}(X)$ a law of composition by setting
\[
\label{IV.21.2.10.1}
  \operatorname{cl}(\sh{L},s)\operatorname{cl}(\sh{L}',s')=\operatorname{cl}(\sh{L}\otimes\sh{L}',s\otimes s');
  \tag{21.2.10.1}
\]
it is immediate that this is a law of \emph{commutative group}, whose neutral element is $\operatorname{cl}(\sh{O}_X,1)$ and where the inverse of $\operatorname{cl}(\sh{L},s)$ is $\operatorname{cl}(\sh{L}^{-1},s^{\otimes(-1)})$.
\end{env}

\begin{proposition}[21.2.11]
\label{IV.21.2.11}
The applications
\[
\label{IV.21.2.11.1}
  D\mapsto\operatorname{cl}(\sh{O}_X(D),s_D),\qquad\operatorname{cl}(\sh{L},s)\mapsto\operatorname{div}(s)
  \tag{21.2.11.1}
\]
are reciprocal isomorphisms of $\operatorname{Div}(X)$ on $\operatorname{D}(X)$ and of $\operatorname{D}(X)$ on $\operatorname{Div}(X)$ respectively.
\end{proposition}

\begin{proof}
Taking into account \sref{IV.21.2.8.3}, \sref{IV.21.2.9.4} and \sref{IV.21.2.10.1}, it suffices to see that the composites of these two applications are the identities in $\operatorname{Div}(X)$ and $\operatorname{D}(X)$ respectively.
The first assertion is none other than \sref{IV.21.2.9.3}.
On the other hand, let $D=\operatorname{div}(s)$, where $s$ is a regular meromorphic section of $\sh{L}$ over $X$, and let $U$ be an open of $X$ such that there exists an isomorphism of $\sh{L}|U$ on $\sh{O}_U$, transforming $s|U$ in $f\in\Gamma(U,\sh{M}_X^*)$, so that $D|U=\operatorname{div}_U(f)$, $\sh{O}_X(D)|U=\sh{O}_U\cdot f^{-1}$ and $s_D|U$ the unit element of $\Gamma(U,\sh{M}_X)$.
There is thus an isomorphism $v_U:\sh{L}|U\to\sh{O}_U\cdot f^{-1}=\sh{O}_X(D)|U$ such that $\overline{v}_U$ transforms $s|U$ in $f\cdot f^{-1}=1$;
one sees immediately that these isomorphisms are compatible with the restriction operations, hence define an isomorphism $v:\sh{L}\to\sh{O}_X(D)$ such that $\overline{v}(s)=s_D$.
Q.E.D.

One can transport by the first of the isomorphisms \sref{IV.21.2.11.1} the structure of ordered group of $\operatorname{Div}(X)$ to $\operatorname{D}(X)$;
the elements $\geq 0$ of $\operatorname{D}(X)$ are thus the classes $\operatorname{cl}(\sh{L},s)$ such that $\operatorname{div}(s)\geq 0$, that is to say \sref{IV.21.1.7.1} such that
\[
  s\in\Gamma(X,\sh{L})\cap\Gamma(X,(\sh{M}_X(\sh{L}))^*).
\]
\end{proof}

\begin{env}[21.2.12]
\label{IV.21.2.12}
Let $D$ be a positive divisor on a prescheme $X$;
the Ideal $\sh{J}_X(D)$ is an Ideal of $\sh{O}_X$, which is an $\sh{O}_X$-Module invertible;
let $Y(D)$ be the closed subprescheme of $X$ that it defines.
For every $x\in Y(D)$, there exists by hypothesis an open neighbourhood $U$ of $x$ in $X$ and a regular section $t\in\Gamma(U,\sh{O}_X)$ such that $\sh{J}_X(D)|U=\sh{O}_U\cdot t$ \sref{IV.21.2.7};
in other terms, the canonical immersion $Y(D)\to X$ is regular and of codimension $1$ \sref{IV.19.1.4} at every point of $Y(D)$.
Conversely, if $Y$ is a closed subprescheme of $X$, regularly immersed in $X$ and of codimension $1$ at every point of $Y$, there exists a positive divisor and only one $D$ such that $Y(D)=Y$, since for every $x\in Y$, there exists a neighbourhood $U$ of $x$ in $X$ such that $Y\cap U$ is defined by an Ideal of $\sh{O}_U$ of the form $\sh{O}_U\cdot t$, where $t$ is regular in $\Gamma(U,\sh{O}_X)$.

We note that one then has $\operatorname{Supp}(D)=Y(D)$, since to say that $D_x\neq 0$ means that $t_x$ is not invertible, that is to say that $x\in Y(D)$.
\end{env}

\subsection{Linear equivalence of divisors}
\label{IV.21.3}

\oldpage[IV-4]{263}

\begin{env}[21.3.1]
\label{IV.21.3.1}
We say that a divisor $D$ on $X$ is \emph{principal} if it is of the form $\operatorname{div}(f)$, where $f$ is a regular meromorphic function on $X$;
the regular meromorphic functions $f'$ such that $\operatorname{div}(f')=D$ are then all those of the form $tf$, where $t\in\Gamma(X,\sh{O}_X^*)$ \sref{IV.21.1.3.4}.
The set of principal divisors is a subgroup of $\operatorname{Div}(X)$, denoted $\operatorname{Div.princ}(X)$, isomorphic to $\Gamma(X,\sh{M}_X^*)/\Gamma(X,\sh{O}_X^*)$.
Two divisors $D$, $D'$ are said to be \emph{linearly equivalent} if $D-D'$ is a principal divisor;
the principal divisors are thus the divisors linearly equivalent to $0$.
\end{env}

\begin{env}[21.3.2]
\label{IV.21.3.2}
Recall $(\mathbf{0_I}$, 5.4.7) that one can speak of equivalence classes of $\sh{O}_X$-Modules invertibles for the isomorphism relation;
one denotes this set by $\operatorname{Pic}(X)$, and for every $\sh{O}_X$-Module invertible $\sh{L}$, one denotes $\operatorname{cl}(\sh{L})$ the equivalence class of the $\sh{O}_X$-Modules isomorphic to $\sh{L}$;
furthermore, $\operatorname{Pic}(X)$ is a commutative group for the multiplication defined by $\operatorname{cl}(\sh{L})\operatorname{cl}(\sh{L}')=\operatorname{cl}(\sh{L}\otimes\sh{L}')$.
It is clear that the application
\[
\label{IV.21.3.2.1}
  r:\operatorname{cl}(\sh{L},s)\mapsto\operatorname{cl}(\sh{L})
  \tag{21.3.2.1}
\]
is a homomorphism of the group $\operatorname{D}(X)$ \sref{IV.21.2.10} in the group $\operatorname{Pic}(X)$.
By composition, one deduces thus a homomorphism
\[
\label{IV.21.3.2.2}
  l:\operatorname{Div}(X)\xrightarrow{\sim}\operatorname{D}(X)\to\operatorname{Pic}(X)
  \tag{21.3.2.2}
\]
(also denoted $l_X$) such that, for every divisor $D$, one has
\[
\label{IV.21.3.2.3}
  l(D)=\operatorname{cl}(\sh{O}_X(D)).
  \tag{21.3.2.3}
\]

Note finally that, if $u:X'\to X$ is a morphism of ringed spaces, $\sh{L}_1$, $\sh{L}_2$ two $\sh{O}_X$-Modules invertibles isomorphes, the $\sh{O}_{X'}$-Modules invertibles $u^*(\sh{L}_1)$ and $u^*(\sh{L}_2)$ $(\mathbf{0_I}$, 5.4.5) are isomorphic;
since on the other hand, for any two $\sh{O}_X$-Modules invertibles $\sh{L}_1$, $\sh{L}_2$ one has $u^*(\sh{L}_1\otimes\sh{L}_2)=u^*(\sh{L}_1)\otimes u^*(\sh{L}_2)$ up to canonical isomorphism $(\mathbf{0_I}$, 4.3.3), one sees that the morphism $u$ defines canonically a homomorphism of commutative groups
\[
\label{IV.21.3.2.4}
  \operatorname{Pic}(u):\operatorname{Pic}(X)\to\operatorname{Pic}(X').
  \tag{21.3.2.4}
\]
\end{env}

\begin{proposition}[21.3.3]
\label{IV.21.3.3}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item The kernel of the canonical homomorphism $l:\operatorname{Div}(X)\to\operatorname{Pic}(X)$ is the subgroup $\operatorname{Div.princ}(X)$, in other words, for $\sh{O}_X(D)$ and $\sh{O}_X(D')$ to be isomorphic, it is necessary and sufficient that $D$ and $D'$ be linearly equivalent.
  One thus has a canonical injective homomorphism
  \[
  \label{IV.21.3.3.1}
    \operatorname{Div}(X)/\operatorname{Div.princ}(X)\to\operatorname{Pic}(X)
    \tag{21.3.3.1}
  \]
  deduced from $l$.
  \item For an $\sh{O}_X$-Module invertible $\sh{L}$ to be such that $\operatorname{cl}(\sh{L})$ is of the form $l(D)$, or equivalently for $\sh{L}$ to be isomorphic to an $\sh{O}_X$-Module of the form $\sh{O}_X(D)$, it is necessary and sufficient that there exists a regular meromorphic section of $\sh{L}$.
\end{enumerate}
\end{proposition}

\begin{proof}
The proposition results immediately from the definitions and from \sref{IV.21.2.10}.
\end{proof}

\begin{proposition}[21.3.4]
\label{IV.21.3.4}
\oldpage[IV-4]{264}
Let $X$ be a prescheme satisfying one of the two following hypotheses:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $X$ is locally Noetherian and $\operatorname{Ass}(\sh{O}_X)$ is contained in an affine open of $X$.
  \item $X$ is reduced and the set of its irreducible components is locally finite.
\end{enumerate}
Then the canonical homomorphism $l:\operatorname{Div}(X)\to\operatorname{Pic}(X)$ is surjective, and gives by passage to the quotient an isomorphism
\[
  \operatorname{Div}(X)/\operatorname{Div.princ}(X)\xrightarrow{\sim}\operatorname{Pic}(X).
\]
\end{proposition}

\begin{proof}
It suffices to show that every $\sh{O}_X$-Module invertible $\sh{L}$ admits a regular meromorphic section over $X$ \sref{IV.21.3.3}[(ii)].
In both cases, it will suffice, thanks to \sref{IV.20.2.11}[(ii)], to define a section $s$ of $(\sh{M}_X(\sh{L}))^*$ over a \emph{schematically dense} open $U$ of $X$, or else in case~a), of an open $U$ containing $\operatorname{Ass}(\sh{O}_X)$ \sref{IV.20.2.13}[(iv)].
Indeed, let then $V$ be an arbitrary open of $X$ such that $\sh{L}|V$ is isomorphic to $\sh{O}_V$, so that there exists an isomorphism of $\sh{M}_X(\sh{L})|V$ on $\sh{M}_X|V$, transforming $s|(U\cap V)$ in a section $f_V$ of $\sh{M}_X^*$ over $U\cap V$.
Since $U\cap V$ is schematically dense in $V$, it results from \sref{IV.20.2.11} that there exists a regular meromorphic function and only one $g_V$ in $V$ such that $g_V|(U\cap V)=f_V$, and this section thus corresponds by the considered isomorphism to a regular meromorphic section $u_V$ of $\sh{L}$ over $V$ such that $u_V|(U\cap V)=s|(U\cap V)$.
Furthermore, if $V'$ is a second open of $X$ such that $\sh{L}|V'$ is isomorphic to $\sh{O}_V$, the restrictions of $u_V$ and $u_{V'}$ to $V\cap V'$ are equal, since they correspond by isomorphism to two meromorphic functions which coincide in a schematically dense open $U\cap V\cap V'$, and we conclude again by \sref{IV.20.2.11};
the $u_V$ are thus indeed the restrictions of a section of $(\sh{M}_X(\sh{L}))^*$ over $X$.

This being so, in case~b), we take for each of the maximal points $x_\lambda$ of $X$ an open $U_\lambda$ containing $x_\lambda$, not meeting any irreducible component of $X$ distinct from $\overline{\{x_\lambda\}}$ and such that $\sh{L}|U_\lambda$ is isomorphic to $\sh{O}_{U_\lambda}$;
we take for $s$ the section of $\sh{L}|U_\lambda$ which corresponds by the preceding isomorphism to the unit section of $\sh{O}_{U_\lambda}$.

In case~a), one can take for $U$ by hypothesis an affine open (hence Noetherian), in other words, suppose that $X=\Spec(A)$, where $A$ is Noetherian, and $\sh{L}=\widetilde{P}$, where $P$ is a projective $A$-module of rank $1$.
If $S$ is the set of regular elements of $A$, one has $\Gamma(X,\sh{M}_X)=S^{-1}A$ \sref{IV.20.2.12} and $\Gamma(X,\sh{M}_X(\sh{L}))=S^{-1}P$;
now $S$ is the set of elements not belonging to any of the ideals associated to $A$, hence $S^{-1}A$ is a semi-local ring whose maximal ideals come from the maximal elements of $\operatorname{Ass}(A)$, and $S^{-1}P$ is a projective $S^{-1}A$-module of rank $1$, hence here a \emph{free} module of rank $1$ (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection5, n\textsuperscript{o}~3, prop.~5);
an element forming a base of this $S^{-1}A$-module is thus by \sref{IV.20.1.8} a regular meromorphic section of $\sh{L}$ over $X$.
\end{proof}

\begin{corollary}[21.3.5]
\label{IV.21.3.5}
If $X$ is a Noetherian prescheme such that there exists an ample $\sh{O}_X$-Module invertible $(\mathbf{II}$, 4.5.3) (for example a quasi-projective prescheme over a Noetherian ring spectrum $(\mathbf{II}$, 5.3.1 and 4.6.6)), the canonical homomorphism $l:\operatorname{Div}(X)\to\operatorname{Pic}(X)$ is surjective.
\end{corollary}

\begin{proof}
\oldpage[IV-4]{265}
Indeed $(\mathbf{II}$, 4.5.4), there then exists an affine open neighbourhood of the \emph{finite} set $\operatorname{Ass}(\sh{O}_X)$.
\end{proof}

\begin{remark}[21.3.6]
\label{IV.21.3.6}
Recall $(\mathbf{0_I}$, 5.4.7) that one has a canonical isomorphism $\pi:\mathrm{H}^1(X,\sh{O}_X^*)\xrightarrow{\sim}\operatorname{Pic}(X)$ defined in the following way.
One starts from a $1$-cocycle $(c_{\alpha\beta})$ with values in $\sh{O}_X^*$ corresponding to an open covering $(U_\alpha)$ of $X$, $c_{\alpha\beta}$ being an element of $\Gamma(U_\alpha\cap U_\beta,\sh{O}_X^*)$, and one associates to it the class of the $\sh{O}_X$-Module invertible obtained by gluing the $\sh{O}_{U_\alpha}$ along the isomorphisms $\sh{O}_{U_\alpha}|(U_\alpha\cap U_\beta)\simeq\sh{O}_{U_\beta}|(U_\alpha\cap U_\beta)$ defined by multiplication by $c_{\alpha\beta}$.
On the other hand, one deduces from the short exact sequence of sheaves of commutative groups
\[
  1\to\sh{O}_X^*\to\sh{M}_X^*\to\sh{D}iv_X\to 0
\]
the boundary homomorphism of the exact cohomology sequence
\[
\label{IV.21.3.6.1}
  \partial:\operatorname{Div}(X)\to\mathrm{H}^1(X,\sh{O}_X^*).
  \tag{21.3.6.1}
\]

Let us show that the composed homomorphism
\[
  \operatorname{Div}(X)\xrightarrow{\partial}\mathrm{H}^1(X,\sh{O}_X^*)\xrightarrow{\pi}\operatorname{Pic}(X)
\]
is none other than the homomorphism $l$ defined in \sref{IV.21.3.2.2}.
Indeed one must start from a divisor $D$ and an open covering $(U_\alpha)$ of $X$, where $D|U_\alpha=\operatorname{div}_{U_\alpha}(g_\alpha)$, where $g_\alpha$ is a regular meromorphic function over $U_\alpha$;
$\partial(D)$ is the cohomology class of the cocycle $(c_{\alpha\beta})$, where $c_{\alpha\beta}=g_{\alpha|\beta}g_{\beta|\alpha}^{-1}$, $g_{\alpha|\beta}$ denoting the restriction of $g_\alpha$ to $U_\alpha\cap U_\beta$.
It is clear that the image by $\pi$ of this cohomology class is the class of the invertible fractional Ideal $\sh{J}$ such that for every $\alpha$, $\sh{J}|U_\alpha=\sh{O}_{U_\alpha}\cdot g_\alpha^{-1}$, which is none other by definition than $\sh{O}_X(D)$ \sref{IV.21.2.8}.
\end{remark}

\subsection{Reciprocal images of divisors}
\label{IV.21.4}

\begin{env}[21.4.1]
\label{IV.21.4.1}
Let $f:X'\to X$ be a morphism of ringed spaces;
let us give conditions allowing us to associate to a divisor $D$ on $X$ a divisor $D'$ on $X'$, \emph{reciprocal image} of $D$ by $f$.
Note first for this that for every section $t\in\Gamma(X,\sh{O}_X^*)$, the image of $t$ by the canonical homomorphism $\Gamma(X,\sh{O}_X)\to\Gamma(X',\sh{O}_{X'})$ is still invertible, in other words belongs to $\Gamma(X',\sh{O}_{X'}^*)$.
Consider then $D$ as given by the equivalence class of a couple $(\sh{L},s)$, where $\sh{L}$ is an $\sh{O}_X$-Module invertible and $s$ a regular meromorphic section of $\sh{L}$ \sref{IV.20.1.11}.
Form the $\sh{O}_{X'}$-Module invertible $f^*(\sh{L})=\sh{L}'$;
to say that the reciprocal image $s_0 f$ of $s$ by $f$ exists \sref{IV.20.1.11} and is a regular meromorphic section of $\sh{L}'$ over $X'$ amounts to saying that the reciprocal images by $f$ of $s$ and of $s^{\otimes(-1)}$ exist, in other words that $s\in\Gamma(X,\sh{M}_f(\sh{L}))$ and $s^{\otimes(-1)}\in\Gamma(X,\sh{M}_f(\sh{L}^{-1}))$;
the remark made above shows then that if $(\sh{L}_1,s_1)$ is a couple equivalent to $(\sh{L},s)$ in the sense of \sref{IV.21.2.10}, the reciprocal image $s_1\circ f$ exists and the couples $(\sh{L}'_1,s_1\circ f)$ and $(\sh{L}',s_0\circ f)$ are equivalent.
We can thus give the following definition:
\end{env}

\begin{definition}[21.4.2]
\label{IV.21.4.2}
Given a morphism $f:X'\to X$ of ringed spaces, we say that the \emph{reciprocal image by $f$} of a divisor $D$ on $X$ \emph{exists} if one has $s_D\in\Gamma(X,\sh{M}_f(\sh{O}_X(D)))$ and $s_{-D}\in\Gamma(X,\sh{M}_f(\sh{O}_X(-D)))$ (cf.\ \sref{IV.20.1.11}).
We then call \emph{reciprocal image} of $D$ by $f$ and denote $f^*(D)$ the divisor on $X'$ which corresponds canonically \sref{IV.21.2.11} to the class of the couple $(f^*(\sh{O}_X(D)),s_D\circ f)$.
\end{definition}

It results immediately from this definition that if $D$ and $D'$ have reciprocal images by $f$, the same holds for $-D$ and $D+D'$ (taking into account \sref{IV.21.2.9.4}) and that one has
$f^*(D+D')=f^*(D)+f^*(D')$.

\oldpage[IV-4]{266}
In other words, the set $\operatorname{Div}^f(X)$ of divisors on $X$ whose reciprocal image by $f$ exists is a subgroup of $\operatorname{Div}(X)$, and the application $D\mapsto f^*(D)$ is an \emph{increasing homomorphism} of the ordered subgroup $\operatorname{Div}^f(X)$ in the ordered group $\operatorname{Div}(X')$, making commutative the diagram
\[
\label{IV.21.4.2.1}
  \xymatrix{
    \operatorname{Div}^f(X)\ar[r]^{l_X}\ar[d]_{f^*} & \operatorname{Pic}(X)\ar[d]^{\operatorname{Pic}(f)}\\
    \operatorname{Div}(X')\ar[r]_{l_{X'}} & \operatorname{Pic}(X')
  }
  \tag{21.4.2.1}
\]

\begin{env}[21.4.3]
\label{IV.21.4.3}
The definition \sref{IV.21.4.2} shows immediately that, for $f^*(D)$ to exist, it is necessary and sufficient that for every open $U$ of $X$, the reciprocal image by $f^U:f^{-1}(U)\to U$ (restriction of $f$) of $D|U$ exists.
Now, if $D=\operatorname{div}(g)$, where $g$ is a regular meromorphic function on $X$, to say that $f^*(\sh{O}_X(D))$ is defined means \sref{IV.21.2.9} that the reciprocal image of $g$ by $f$ exists and is a regular meromorphic function on $X'$.
We deduce immediately a second description of $\operatorname{Div}^f(X)$ and of $f^*(D)$:
one considers the sub-sheaf of groups of $\sh{M}_X^*$, denoted $\sh{M}_f^*$, formed of the germs of regular meromorphic functions on an open of $X$ whose reciprocal image by $f$ exists and is regular on the reciprocal image open \sref{IV.20.1.11}.
Then if $f=(\psi,\theta)$, the canonical homomorphism \sref{IV.20.1.11} $\psi^*(\sh{M}_f)\to\sh{M}_{X'}$ gives by restriction a homomorphism of sheaves of groups $\psi^*(\sh{M}_f^*)\to\sh{M}_{X'}^*$.
Setting $\sh{D}iv_X^f=\sh{M}_f^*/\sh{O}_X^*$, one has $\operatorname{Div}^f(X)=\Gamma(X,\sh{D}iv_X^f)$, and the application $D\mapsto f^*(D)$ corresponds to the homomorphism of sheaves of groups $\psi^*(\sh{M}_f^*)/\psi^*(\sh{O}_X^*)\to\sh{M}_{X'}^*/\sh{O}_{X'}^*=\sh{D}iv_{X'}$ deduced from the preceding by passage to quotients.
\end{env}

\begin{env}[21.4.4]
\label{IV.21.4.4}
It results immediately from the preceding definitions that if $f':X''\to X'$ is a second morphism of ringed spaces, $D$ a divisor on $X$ such that the reciprocal images $f^*(D)$ and $f'^*(f^*(D))$ exist, then $(f\circ f')^*(D)$ exists and is equal to $f'^*(f^*(D))$.
\end{env}

\begin{proposition}[21.4.5]
\label{IV.21.4.5}
Let $f:X'\to X$ be a morphism of ringed spaces.
In each of the three following cases, the reciprocal image by $f$ of every divisor on $X$ is defined:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item $f$ is flat.
  \item $X$ and $X'$ are locally Noetherian preschemes and one has $f(\operatorname{Ass}(\sh{O}_{X'}))\subset\operatorname{Ass}(\sh{O}_X)$.
  \item $X$ and $X'$ are preschemes, the set of irreducible components of $X$ is locally finite, $X'$ is reduced and every irreducible component of $X'$ dominates an irreducible component of $X$.
\end{enumerate}
\end{proposition}

\begin{proof}
It suffices indeed to show that in the three cases one has $\sh{M}_f=\sh{M}_X$;
in case~(i), this results from \sref{IV.20.1.12}.
In case~(iii), one can restrict to the case where $X=\Spec(A)$ and $X'=\Spec(A')$ are affine;
if $s\in A$ is regular, it does not belong to any minimal prime ideal of $A$ \sref{IV.20.3.1}, hence the hypothesis implies that its image in $A'$ does not belong to any minimal prime ideal of $A'$, and is thus a regular element of $A'$ \sref{IV.20.1.3.1}.
In case~(ii) the meromorphic functions on $X'$ identify with the pseudo-functions on $X'$ \sref{IV.20.2.11}, and the hypothesis, together with \sref{IV.20.2.13}[(iv)] ensures that the reciprocal image
\oldpage[IV-4]{267}
by $f$ of every schematically dense open in $X$ is a schematically dense open in $X'$;
we thus conclude by \sref{IV.20.3.12}.
\end{proof}

\begin{corollary}[21.4.6]
\label{IV.21.4.6}
Let $X$ be a prescheme having one of the following properties:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item $X$ is locally Noetherian.
  \item $X$ is reduced and the set of its irreducible components is locally finite.
\end{enumerate}
Then, for every $x\in X$, one has a canonical isomorphism
\[
\label{IV.21.4.6.1}
  (\sh{D}iv_X)_x\xrightarrow{\sim}\operatorname{Div}(\sh{O}_{X,x}).
  \tag{21.4.6.1}
\]
\end{corollary}

\begin{proof}
This results indeed from \sref{IV.20.2.11}, \sref{IV.20.3.7} and from the fact that $(\sh{O}_X^*)_x$ identifies with the group of invertible elements of the ring $\sh{O}_{X,x}$.
\end{proof}

\begin{env}[21.4.7]
\label{IV.21.4.7}
Let $X$, $X'$ be two preschemes, $f:X'\to X$ a morphism.
If $D$ is a \emph{positive} divisor on $X$ such that the reciprocal image $f^*(D)$ is defined \sref{IV.21.4.2}, then the closed subprescheme $Y(f^*(D))$ of $X'$ is none other than the reciprocal image $f^{-1}(Y(D))$;
this results immediately from the definitions \sref{IV.21.4.2} and \sref{IV.21.2.12}.
\end{env}

\begin{proposition}[21.4.8]
\label{IV.21.4.8}
Let $X$, $Y$ be two preschemes, $f:X\to Y$ a faithfully flat morphism.
Then, if $D$ is a divisor on $Y$ such that $f^*(D)\geq 0$ (the existence of $f^*(D)$ resulting from \sref{IV.21.4.5}), one has $D\geq 0$.
In particular, the application $D\mapsto f^*(D)$ of $\operatorname{Div}(Y)$ in $\operatorname{Div}(X)$ is injective.
\end{proposition}

\begin{proof}
The question being local on $Y$, one can restrict to the case where $D=\operatorname{div}(w)$, with $w=uv^{-1}$, $u$ and $v$ being two regular sections of $\sh{O}_Y$ over $Y$.
By hypothesis one has $u_y\sh{O}_{X,x}\subset v_y\sh{O}_{X,x}$;
hence, for every $x\in X$, if one sets $y=f(x)$, one has $u_y\sh{O}_{X,x}\subset v_y\sh{O}_{X,x}$;
one thus concludes that $\sh{O}_{X,x}$ is a faithfully flat $\sh{O}_{Y,y}$-module by virtue of the hypothesis and of Bourbaki, \emph{Alg.~comm.}, chap.~I, \textsection3, n\textsuperscript{o}~5, prop.~10, (ii);
whence $u\sh{O}_Y\subset v\sh{O}_Y$ and since $f$ is surjective and therefore $D\geq 0$.
\end{proof}

\subsection{Direct images of divisors}
\label{IV.21.5}

\begin{env}[21.5.1]
\label{IV.21.5.1}
Let $X$, $X'$ be two preschemes, $f:X'\to X$ a morphism.
We will, in this section, give sufficient conditions for one to be able to associate to every divisor $D'$ on $X'$ a divisor $D$ on $X$, \emph{direct image} of $D'$ by $f$.
We will restrict ourselves to the case where $f$ is a \emph{finite} morphism (for more general conditions, see the chapter of this Treatise dedicated to intersection theory).
\end{env}

\begin{lemma}[21.5.2]
\label{IV.21.5.2}
Let $A$ be a ring, $E$ an $A$-module free of finite rank.
For an endomorphism $u$ of $E$ to be injective, it is necessary and sufficient that $\det(u)$ be a regular element of $A$.
\end{lemma}

\begin{proof}
This is proved in Bourbaki, \emph{Alg.}, chap.~III, 3rd ed., \textsection8, n\textsuperscript{o}~2, prop.~3.
\end{proof}

\begin{env}[21.5.3]
\label{IV.21.5.3}
Suppose now that the morphism $f:X'\to X$ is finite, and moreover that $f$ satisfies one of the two following properties:
\begin{enumerate}
  \item[I)] $f$ is a finite locally free morphism, in other words \sref{IV.18.2.7} the $\sh{O}_X$-Module quasi-coherent of finite type $f_*(\sh{O}_{X'})$ is locally free.
  \item[II)] $X$ is a locally Noetherian reduced prescheme, the $\sh{R}(X)$-Module $f_*(\sh{O}_{X'})\otimes_{\sh{O}_X}\sh{R}(X)$ is locally free, and for every section $s'\in\Gamma(f^{-1}(U),\sh{O}_{X'})$ ($U$ open
\end{enumerate}

\oldpage[IV-4]{268}
in $X$), $\mathrm{N}_{f_*(\sh{O}_{X'})/\sh{O}_X}(s')$ is a section of $\sh{O}_X$ over $U$ (cf.\ $(\mathbf{II}$, 6.5.1)).
(Recall that condition~II) is satisfied for every finite morphism $f$ when $X$ is a locally Noetherian \emph{normal} prescheme (\emph{loc.~cit.}).)

One knows then $(\mathbf{II}$, 6.5.5) that for every $\sh{O}_{X'}$-Module invertible $\sh{L}'$ one defines the \emph{norm} $\sh{L}=\mathrm{N}_{X'/X}(\sh{L}')$ (which we will also write $\mathrm{N}(\sh{L}')$), which is an $\sh{O}_X$-Module invertible.
Furthermore, for every open $U$ of $X$ and every regular section $s'\in\Gamma(f^{-1}(U),\sh{O}_{X'})$, the norm $\mathrm{N}_{X'/X}(s')=\mathrm{N}_{f_*(\sh{O}_{X'})/\sh{O}_X}(s')$ (which we will also write $\mathrm{N}(s')$) is a regular element of $\Gamma(U,\sh{O}_X)$;
one is indeed reduced immediately to the case where $U=X$ is affine, and then the conclusion results from \sref{IV.21.5.2} in hypothesis~I);
on the other hand, in hypothesis~II), the fact that $\sh{R}(X)$ is a flat $\sh{O}_X$-Module implies that the section $s'\otimes 1$ of $\Gamma(U,f_*(\sh{O}_{X'})\otimes_{\sh{O}_X}\sh{R}(X))$ is also regular $(\mathbf{0_I}$, 6.1.2), and the conclusion results again from \sref{IV.21.5.2} applied to the ring $\Gamma(U,\sh{R}(X))$, taking into account the definition of the norm of a section $(\mathbf{II}$, 6.5.3).
This being so, let $u'$ be a meromorphic section of $\sh{L}'$ over $X'$;
the morphism $f$ being affine, every point $x\in X$ admits an open neighbourhood $U$ such that $u'|f^{-1}(U)$ is of the form $t'/s'$, where $t'\in\Gamma(f^{-1}(U),\sh{L}')$ and $s'$ is a regular section in $\Gamma(f^{-1}(U),\sh{O}_{X'})$;
the element $\mathrm{N}(t')/\mathrm{N}(s')$ is then a meromorphic section of $\sh{L}=\mathrm{N}(\sh{L}')$ over $U$ by virtue of what precedes, and it is immediately verified that it is independent of the representation $t'/s'$ chosen for $u'|f^{-1}(U)$;
the meromorphic sections of $\sh{L}$ thus defined, for variable $U$, are the restrictions of a single meromorphic section of $\sh{L}$ over $X$, that we call the \emph{norm} of $u'$ and that we denote $\mathrm{N}_{X'/X}(u')$ (or simply $\mathrm{N}(u')$).
If $u'$ is a \emph{regular} meromorphic section of $\sh{L}'$ over $X'$, it results immediately from what precedes that $\mathrm{N}(u')$ is a regular meromorphic section of $\mathrm{N}(\sh{L}')$ over $X$.
The application thus defined
\[
\label{IV.21.5.3.1}
  \mathrm{N}_{X'/X}:\Gamma(X',\sh{M}_{X'}(\sh{L}'))\to\Gamma(X,\sh{M}_X(\mathrm{N}_{X'/X}(\sh{L}')))
  \tag{21.5.3.1}
\]
prolongs the norm defined in $(\mathbf{II}$, 6.5.3);
if $u'$ is a regular meromorphic section of $\sh{L}'$ over $X'$, it results immediately from what precedes that $\mathrm{N}(u')$ is a regular meromorphic section of $\mathrm{N}(\sh{L}')$ over $X$, since (with the same notations) $\mathrm{N}(t')$ is regular if $t'$ is.
Finally, if $\sh{L}'_1$, $\sh{L}'_2$ are two $\sh{O}_{X'}$-Modules invertibles, $s'_1$ (resp.\ $s'_2$) a regular meromorphic section of $\sh{L}'_1$ (resp.\ $\sh{L}'_2$) over $X'$, one has, by virtue of what precedes and of $(\mathbf{II}$, 6.5.3.1)
\[
\label{IV.21.5.3.2}
  \mathrm{N}(s'_1\otimes s'_2)=\mathrm{N}(s'_1)\otimes\mathrm{N}(s'_2).
  \tag{21.5.3.2}
\]
\end{env}

\begin{env}[21.5.4]
\label{IV.21.5.4}
\oldpage[IV-4]{269}
Suppose always that $f$ satisfies one of the hypotheses~I), II) of \sref{IV.21.5.3}.
If $(\sh{L}'_1,s'_1)$, $(\sh{L}'_2,s'_2)$ are two couples each formed of an $\sh{O}_{X'}$-Module invertible and of a regular meromorphic section of $\sh{L}'$ over $X'$, which are moreover \emph{equivalent} in the sense of \sref{IV.21.2.10}, then the couples $(\mathrm{N}(\sh{L}'_1),\mathrm{N}(s'_1))$ and $(\mathrm{N}(\sh{L}'_2),\mathrm{N}(s'_2))$ are also equivalent, since an isomorphism of $\sh{O}_{X'}$-Modules invertibles has for norm an isomorphism of $\sh{O}_X$-Modules invertibles, and one has seen above that $\mathrm{N}_{X'/X}$ transforms the sections of $\Gamma(X',\sh{O}_{X'}^*)$ in sections of $\Gamma(X,\sh{O}_X^*)$.
One can thus give the following definition:
\end{env}

\begin{definition}[21.5.5]
\label{IV.21.5.5}
Given a finite morphism $f:X'\to X$ of preschemes, satisfying one of the conditions~I), II) of \sref{IV.21.5.3}, we call \emph{direct image} (or \emph{norm}) of a divisor $D'$ on $X'$ by $f$, and denote $f_*(D')$ (or $\mathrm{N}_{X'/X}(D')$), the divisor on $X$ which corresponds canonically \sref{IV.21.2.11} to the class of the couple $(\mathrm{N}_{X'/X}(\sh{O}_{X'}(D')),\mathrm{N}_{X'/X}(s_{D'}))$.
\end{definition}

It results from this definition, taking into account \sref{IV.21.2.9.4} and \sref{IV.21.5.3.2}, that
\[
\label{IV.21.5.5.1}
  f_*(D'_1+D'_2)=f_*(D'_1)+f_*(D'_2)
  \tag{21.5.5.1}
\]
and $D'\geq 0$ implies $f_*(D')\geq 0$, in other words $D'\mapsto f_*(D')$ is an \emph{increasing homomorphism} of the ordered group $\operatorname{Div}(X')$ in the ordered group $\operatorname{Div}(X)$.
The definition \sref{IV.21.5.5} shows moreover that for every open $U$ of $X$, one has $f_*^U(D'|f^{-1}(U))=f_*(D')|U$ ($f^U$ being the restriction $f^{-1}(U)\to U$ of $f$), and the homomorphisms $f_*^U$, for variable $U$, define an \emph{homomorphism of sheaves of ordered groups}
\[
\label{IV.21.5.5.2}
  \mathrm{N}_{X'/X}:f_*(\sh{D}iv_{X'})\to\sh{D}iv_X.
  \tag{21.5.5.2}
\]

Furthermore, for every open $U$ of $X$, every $\sh{O}_{X'}$-Module invertible $\sh{L}'$ and every regular meromorphic section $s'$ of $\sh{L}'$ over $f^{-1}(U)$, one has, by the preceding definitions and \sref{IV.21.1.4}
\[
\label{IV.21.5.5.3}
  \operatorname{div}_U(\mathrm{N}(s'))=f_*^U(\operatorname{div}_{f^{-1}(U)}(s')).
  \tag{21.5.5.3}
\]

\begin{proposition}[21.5.6]
\label{IV.21.5.6}
Let $f:X'\to X$ be a finite locally free morphism and suppose that $f_*(\sh{O}_{X'})$ is of constant rank $n$.
Then, for every divisor $D$ on $X$, $f_*(D)$ is defined and one has
\[
\label{IV.21.5.6.1}
  f_*(f^*(D))=nD.
  \tag{21.5.6.1}
\]
\end{proposition}

\begin{proof}
The first assertion results from the fact that $f$ is flat \sref{IV.21.4.5}, and the second is an immediate consequence of the definitions and of $(\mathbf{II}$, 6.5.3.2).
\end{proof}

\begin{proposition}[21.5.7]
\label{IV.21.5.7}
Let $f:X'\to X$ be a finite morphism satisfying one of the hypotheses~I), II) of \sref{IV.21.5.3}, $f':X''\to X'$ a finite locally free morphism of constant rank $n$.
Then $f''=f\circ f':X''\to X$ satisfies the same hypothesis as $f$, and for every divisor $D''$ on $X''$, one has
\[
\label{IV.21.5.7.1}
  f''_*(D'')=f_*(f'_*(D'')).
  \tag{21.5.7.1}
\]
\end{proposition}

Taking into account the definition \sref{IV.21.5.5}, it suffices to prove the following result:

\begin{lemma}[21.5.7.2]
\label{IV.21.5.7.2}
Under the hypotheses of \sref{IV.21.5.7}, one has a functorial isomorphism
\[
\label{IV.21.5.7.3}
  \mathrm{N}_{X''/X}(\sh{L}'')\xrightarrow{\sim}\mathrm{N}_{X'/X}(\mathrm{N}_{X''/X'}(\sh{L}''))
  \tag{21.5.7.3}
\]
in the category of $\sh{O}_X$-Modules invertibles.
\end{lemma}

\oldpage[IV-4]{270}
The question is evidently local on $X$, and one can thus restrict to the case where $X=\Spec(A)$ is affine; one then has $X'=\Spec(A')$ and $X''=\Spec(A'')$, and one can suppose that $A''$ is a projective $A'$-module of rank $n$.
When $f$ is locally free, one can suppose that $A'$ is a free $A$-module of rank $m$, and then $A''$ is a free $A$-module of rank $mn$;
the formula \sref{IV.21.5.7.3} results then from the transitivity of the norm (Bourbaki, \emph{Alg.}, chap.~VIII, \textsection12, n\textsuperscript{o}~2, prop.~7).
When $f$ satisfies hypothesis~II), $A$ is Noetherian reduced, and if $R$ is its total ring of fractions, $A'\otimes_A R$ is a free $R$-module of rank $m$, and then $A''\otimes_A R$ is a free $R$-module of rank $mn$;
and since $R$ is also a semi-local ring, this $R$-module is free;
the proposition results then also from the transitivity of the norms.

\begin{proposition}[21.5.8]
\label{IV.21.5.8}
Let $f:X'\to X$ be a finite morphism, $g:Y\to X$ a morphism;
set $Y'=X'\times_X Y$, $f'=f_{(Y)}:Y'\to Y$, $g'=g_{(X)}:Y'\to X'$.
Suppose satisfied one of the following conditions:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item $f$ is locally free and $g$ is flat.
  \item $f$ is locally free, $X$ and $Y$ are locally Noetherian, $g(\operatorname{Ass}(\sh{O}_Y))\subset\operatorname{Ass}(\sh{O}_X)$ and $g'(\operatorname{Ass}(\sh{O}_{Y'}))\subset\operatorname{Ass}(\sh{O}_{X'})$.
  \item $f$ satisfies hypothesis~\textup{II} of \sref{IV.21.5.3}, $Y$ is locally Noetherian, $Y$ and $Y'$ are reduced and every irreducible component of $Y$ (resp.\ $Y'$) dominates an irreducible component of $X$ (resp.\ $X'$).
\end{enumerate}
Then, for every divisor $D'$ on $X'$, $g'^*(D')$ is defined, $g^*(f_*(D'))$ is defined and one has
\[
\label{IV.21.5.8.1}
  g^*(f_*(D'))=f'_*(g'^*(D')).
  \tag{21.5.8.1}
\]
\end{proposition}

\begin{proof}
Indeed, in all the cases, it results from $(\mathbf{II}$, 6.5.8) that one has a functorial isomorphism
\[
  g^*(\mathrm{N}_{X'/X}(\sh{L}'))\xrightarrow{\sim}\mathrm{N}_{Y'/Y}(g'^*(\sh{L}'))
\]
in the category of $\sh{O}_X$-Modules invertibles;
furthermore $(\mathbf{II}$, 6.5.4), if $s'$ is a section of $\sh{O}_{X'}$ over $f^{-1}(U)$, $s''$ the corresponding section of $\sh{O}_{Y'}$ over $g'^{-1}(f^{-1}(U))$ ($U$ open of $X$), $\mathrm{N}_{Y'/Y}(s'')$ is the section of $\sh{O}_Y$ over $g^{-1}(U)$ which corresponds to the section $\mathrm{N}_{X'/X}(s')$ of $\sh{O}_X$ over $U$.
The formula \sref{IV.21.5.8.1} results then from the definitions if $g'^*(D')$ and $g^*(D)$ are defined, whatever the divisors $D'$ on $X'$ and $D$ on $X$.
As for what concerns $D$, in case~(i) $g'$ is flat, hence in all cases $g'^*(D')$ is defined by virtue of \sref{IV.21.4.5}.
As for what concerns $D'$, in case~(i) $g$ is flat, hence in all cases $g^*(f_*(D'))$ is defined by virtue of \sref{IV.21.4.5}.
\end{proof}

\subsection{The $1$-codimensional cycle associated to a divisor}
\label{IV.21.6}

\oldpage[IV-4]{271}

\begin{env}[21.6.1]
\label{IV.21.6.1}
Let $X$ be a locally Noetherian prescheme, and let $\mathfrak{I}(X)$ be the set of \emph{irreducible closed subsets} of $X$ (which is in bijective correspondence with $X$ by the application $x\mapsto\overline{\{x\}}$).
In the product group $\bb{Z}^X$, consider the subgroup $\mathfrak{Z}(X)$ of the elements $(n_z)_{z\in X}$ such that the set of $\overline{\{x\}}\in\mathfrak{I}(X)$ such that $n_x\neq 0$ is \emph{locally finite}.
It is clear that $\mathfrak{Z}(X)$ is a \emph{subgroup} of $\bb{Z}^X$, which contains the direct sum group $\bb{Z}^{(X)}$ (free group having $\mathfrak{I}(X)$ for basis), and is equal to it when $X$ is Noetherian.
The elements of $\mathfrak{Z}(X)$ are called \emph{cycles} on $X$ and those of $\mathfrak{I}(X)$ \emph{prime cycles} (they do not in general form a basis of $\mathfrak{Z}(X)$ when $X$ is not Noetherian).
One always considers $\mathfrak{Z}(X)$ as \emph{ordered} by the order induced on this subgroup by the product order of $\bb{Z}^X$, and one denotes $\mathfrak{Z}^+(X)$ the set of cycles $\geq 0$.

For every cycle $Z\in\mathfrak{Z}(X)$, equal to $(n_x)_{x\in X}$, one will write by abuse of language,
\[
  Z=\sum_{x\in X}n_x\cdot\overline{\{x\}};
\]
one calls \emph{multiplicity} of $Z$ \emph{at the point} $x$ and one denotes $\operatorname{mult}_x(Z)$ the integer $n_x$ (positive or negative).
To say that $Z\geq 0$ means that $\operatorname{mult}_x(Z)\geq 0$ for every $x\in X$.
One calls \emph{support of $Z$} and one denotes $\operatorname{Supp}(Z)$ the union of the $\overline{\{x\}}$ such that $\operatorname{mult}_x(Z)\neq 0$;
by definition of $\mathfrak{Z}(X)$, this is a closed set of $X$, being the union of a locally finite family of closed subsets.
One calls \emph{dimension} (resp.\ \emph{codimension}) of $Z$ and one denotes $\dim(Z)$ (resp.\ $\operatorname{codim}(Z)$) the dimension (resp.\ codimension in $X$) of $\operatorname{Supp}(Z)$.
\end{env}

\begin{env}[21.6.2]
\label{IV.21.6.2}
One says that a closed subset $Y$ of $X$ is \emph{purely of codimension $p$} (in $X$) if \emph{each} of its irreducible components is of codimension $p$ in $X$.
One says that a cycle is \emph{purely of codimension $p$}, or (by abuse of language) is a \emph{$p$-codimensional cycle} if its support is purely of codimension $p$.
One denotes $X^{(p)}$ the set of $x\in X$ such that $\operatorname{codim}(\overline{\{x\}},X)=p$, or, which amounts to the same \sref{IV.5.1.2.1}, $\dim(\sh{O}_{X,x})=p$:
the purely $p$-codimensional cycles form a subgroup $\mathfrak{Z}^p(X)$ of $\mathfrak{Z}(X)$, isomorphic to the subgroup of $\bb{Z}^{X^{(p)}}$ formed of the $(n_x)_{x\in X^{(p)}}$ such that the set of $\overline{\{x\}}$ (or the set of $x$) for which $n_x\neq 0$ is locally finite;
this subgroup contains the free group $\bb{Z}^{(X^{(p)})}$ and is equal to it when $X$ is Noetherian.
One denotes $\mathfrak{Z}^{p+}(X)$ the set of elements $\geq 0$ of $\mathfrak{Z}^p(X)$.
It is clear that the ordered group $\mathfrak{Z}(X)$ contains the direct sum of the ordered subgroups $\mathfrak{Z}^p(X)$, and is identical to this direct sum when $X$ is Noetherian;
in this last case, one considers $\mathfrak{Z}(X)$ as \emph{graded} by the $\bb{Z}^p(X)$ for $p\in\bb{N}$.
\end{env}

\begin{env}[21.6.3]
\label{IV.21.6.3}
\oldpage[IV-4]{272}
Let $Z=\sum_{x\in X}n_x\cdot\overline{\{x\}}$ be a cycle on $X$, $U$ an open of $X$; one calls \emph{restriction} of $Z$ to $U$ and one denotes $Z|U$ the cycle $\sum_{x\in U}n_x\cdot(U\cap\overline{\{x\}})$ on $U$;
one has $\operatorname{Supp}(Z|U)=\operatorname{Supp}(Z)\cap U$, hence $Z\mapsto Z|U$ is an homomorphism of ordered groups $\mathfrak{Z}(X)\to\mathfrak{Z}(U)$ (resp.\ $\mathfrak{Z}^{p+}(U)\to\mathfrak{Z}^{p+}(U)$).
The sheaf $\mathscr{Z}_X$ (resp.\ $\mathscr{Z}_X^p$) is evidently the \emph{direct sum} of the sheaves $(i_x)_*(\bb{Z}_{\overline{\{x\}}})$, where $x$ ranges over $X$ (resp.\ $X^{(p)}$) and $i_x$ is the canonical injection $\overline{\{x\}}\to X$ and $\bb{Z}_{\overline{\{x\}}}$ is the simple constant sheaf $\bb{Z}$ over the space $\overline{\{x\}}$:
this results immediately from the description of the sections of a direct sum of sheaves of commutative groups (G,~II, 2.7).
One denotes this sheaf $\mathscr{Z}_X$ (resp.\ $\mathscr{Z}_X^p$).
One denotes $\mathscr{Z}_X^+$ (resp.\ $\mathscr{Z}_X^{p+}$) the sub-sheaf of monoids (resp.\ $U\mapsto\mathfrak{Z}^{p+}(U)$) of $\mathscr{Z}_X$ (resp.\ $\mathscr{Z}_X^p$).

Note finally that the sheaves $\mathscr{Z}_X^p$ (and hence also $\mathscr{Z}_X$) are \emph{flasque}.
Suppose in effect given a section $Z$ of $\mathscr{Z}_X^p$ over an open $U$, so that the set $S$ of $x\in U$ such that $\operatorname{mult}_x(Z)\neq 0$ is locally finite \emph{in} $U$;
this set is also locally finite \emph{in} $X$, since for every $z\in X$ and every Noetherian open neighbourhood $V$ of $z$, $U\cap V$ is Noetherian, hence contains only a finite number of points of $S$.
One defines thus a section $Z'$ of $\mathscr{Z}_X^p$ over $X$ by setting $Z'=\sum_{x\in U}\operatorname{mult}_x(Z)\cdot\overline{\{x\}}$ and since the adherence of $x$ in $U$ is $\overline{\{x\}}\cap U$, one has $Z'|U=Z$, whence our assertion.
\end{env}

\begin{env}[21.6.4]
\label{IV.21.6.4}
We now propose to define a canonical homomorphism
\[
\label{IV.21.6.4.1}
  c:\sh{D}iv_X\to\mathscr{Z}_X^1
  \tag{21.6.4.1}
\]
of sheaves of commutative groups.
It suffices evidently to define a homomorphism of $\sh{M}_X^*$ in $\mathscr{Z}_X^1$, which is zero on $\sh{O}_X^*$ \sref{IV.21.1.2};
or, by definition, $\sh{M}_X^*$ is the \emph{symmetrised} of the sheaf of monoids $\sh{S}(\sh{O}_X)=\sh{O}_X\cap\sh{M}_X^*$, and a homomorphism $\sh{M}_X^*\to\mathscr{Z}_X^1$ is determined in a unique way by its restriction $\sh{S}(\sh{O}_X)\to\mathscr{Z}_X^1$, which is a homomorphism of sheaves of \emph{monoids} subject to the sole condition of being zero on $\sh{O}_X^*$;
it amounts to the same to say that to define $c$, it suffices to define a homomorphism of \emph{sheaves of monoids}
\[
\label{IV.21.6.4.2}
  c:\sh{D}iv_X^+\to\mathscr{Z}_X^{1+}.
  \tag{21.6.4.2}
\]
\end{env}

\begin{env}[21.6.5]
\label{IV.21.6.5}
Now, one has seen that every positive divisor $D$ on $X$ determines a closed subprescheme $Y(D)$ of $X$, defined by the Ideal $\sh{J}_X(D)\subset\sh{O}_X$, and which is regularly immersed and of codimension $1$.
In each of the maximal points $x$ of $Y(D)$, one has $((19.1.4)$ and $(5.1.2))$ $\operatorname{codim}(\overline{\{x\}},X)=\dim(\sh{O}_{X,x})=1$, in other words $x\in X^{(1)}$;
moreover the set of these maximal points is locally finite in $X$ \sref{IV.3.1.6}, and $\sh{O}_{Y(D),x}$ is an \emph{Artinian} ring.
At every point $x\in X^{(1)}$ which is not a maximal point of $Y(D)$, one necessarily has $x\notin Y(D)$, hence $\sh{O}_{Y(D),x}=0$.
Set
\[
\label{IV.21.6.5.1}
  \operatorname{cyc}(D)=\sum_{x\in X^{(1)}}\operatorname{long}(\sh{O}_{Y(D),x})\cdot\overline{\{x\}}
  \tag{21.6.5.1}
\]
which is thus an element of $\mathfrak{Z}^{1+}(X)$.
\end{env}

\begin{proposition}[21.6.6]
\label{IV.21.6.6}
The application $D\mapsto\operatorname{cyc}(D)$ is a homomorphism of the monoid $\operatorname{Div}^+(X)$ in $\mathfrak{Z}^{1+}(X)$.
\end{proposition}

\begin{proof}
\oldpage[IV-4]{273}
It amounts to seeing that for two positive divisors $D$, $D'$, one has
\[
  \operatorname{cyc}(D+D')=\operatorname{cyc}(D)+\operatorname{cyc}(D').
\]
Now, one has \sref{IV.21.2.5} $\sh{J}_X(D+D')=\sh{J}_X(D)\sh{J}_X(D')$;
if one sets $A=\sh{O}_{X,x}$, and if $t$ and $t'$ are two regular elements of $A$, one has $\operatorname{long}(A/tt'A)=\operatorname{long}(A/tA)+\operatorname{long}(A/t'A)$;
but since $t$ is regular, $tA/tt'A$ is isomorphic to $A/t'A$, whence the proposition immediately.
\end{proof}

It results immediately from the definitions that for every open $U\subset X$, one has
$Y(D)|U=Y(D)\cap U$, hence $\operatorname{cyc}(D|U)=\operatorname{cyc}(D)|U$, and the homomorphisms $\operatorname{cyc}:\operatorname{Div}^+(U)\to\mathfrak{Z}^{1+}(U)$ define a homomorphism of sheaves of monoids \sref{IV.21.6.4.2}, whence the sought homomorphism \sref{IV.21.6.4.1} of sheaves of groups.

One has $\operatorname{Supp}(\operatorname{cyc}(D))\subset\operatorname{Supp}(D)$ for every divisor $D$ and
\[
\label{IV.21.6.6.2}
  \operatorname{Supp}(\operatorname{cyc}(D))=\operatorname{Supp}(D)\quad\text{when }D\geq 0;
  \tag{21.6.6.2}
\]
one has already seen the second relation \sref{IV.21.2.12};
when $D$ is any divisor, the relation $D_x=0$ implies the existence of an open neighbourhood $U$ of $x$ such that $D|U=0$, whence $\operatorname{cyc}(D)|U=0$, which proves our assertion.

\begin{env}[21.6.7]
\label{IV.21.6.7}
The homomorphism \sref{IV.21.6.4.1} gives, by passage to the groups of sections over $X$, an \emph{increasing homomorphism of ordered groups} $\operatorname{Div}(X)\to\mathfrak{Z}^1(X)$, that we denote again $D\mapsto\operatorname{cyc}(D)$;
the number $\operatorname{mult}_x(\operatorname{cyc}(D))$ for $x\in X^{(1)}$ is also denoted $\operatorname{mult}_x(D)$ or $\operatorname{mult}_{\overline{\{x\}}}(D)$ and is called the \emph{multiplicity of the divisor $D$ at the point $x$}, or \emph{multiplicity of the prime cycle $\overline{\{x\}}$ in $D$};
this is a positive or negative integer, equal as we have seen to $\operatorname{long}(\sh{O}_{Y(D),x})$ when $D$ is a \emph{positive} divisor;
one has therefore by definition
\[
\label{IV.21.6.7.1}
  \operatorname{cyc}(D)=\sum_{x\in X^{(1)}}\operatorname{mult}_x(D)\cdot\overline{\{x\}},
  \tag{21.6.7.1}
\]
and by virtue of the fact that $D\mapsto\operatorname{cyc}(D)$ is a homomorphism, one has
\[
\label{IV.21.6.7.2}
  \operatorname{mult}_x(-D)=-\operatorname{mult}_x(D),\quad\operatorname{mult}_x(D+D')=\operatorname{mult}_x(D)+\operatorname{mult}_x(D')
  \tag{21.6.7.2}
\]
for any two divisors $D$, $D'$.

For every regular meromorphic function $f$ on $X$, one sets in particular, for $x\in X^{(1)}$
\[
\label{IV.21.6.7.3}
  o_x(f)=\operatorname{mult}_x(\operatorname{div}(f))
  \tag{21.6.7.3}
\]
and one says that $o_x(f)$ (a positive or negative integer) is the \emph{order of $f$ at the point} $x\in X^{(1)}$.
If $f_x\in\sh{O}_{X,x}$ (regular element of this local ring by hypothesis), one thus has
\[
\label{IV.21.6.7.4}
  o_x(f)=\operatorname{long}(\sh{O}_{X,x}/f_x\sh{O}_{X,x});
  \tag{21.6.7.4}
\]
for two regular meromorphic functions $f$, $f'$ on $X$, one has
\[
\label{IV.21.6.7.5}
  o_x(ff')=o_x(f)+o_x(f'),\quad o_x(f^{-1})=-o_x(f)
  \tag{21.6.7.5}
\]
for every $x\in X^{(1)}$.
The $1$-codimensional cycles
\[
  Z^+(f)=\sum_{x\in X^{(1)}}(o_x(f))^+\cdot\overline{\{x\}},\qquad Z^-(f)=\sum_{x\in X^{(1)}}(o_x(f))^-\cdot\overline{\{x\}}
\]
are called respectively the \emph{cycle of zeros} (or \emph{polar cycle}) of $f$, and one has evidently $\operatorname{cyc}(\operatorname{div}(f))=Z^+(f)-Z^-(f)$.

\oldpage[IV-4]{274}
The $1$-codimensional cycles of the form $\operatorname{cyc}(\operatorname{div}(f))$ are called \emph{principal} (or \emph{linearly equivalent to $0$}) and form a subgroup $\mathfrak{Z}^1_{\mathrm{princ}}(X)$ of $\mathfrak{Z}^1(X)$.
The sections over $X$ of the image $c(\sh{D}iv_X)\subset\mathscr{Z}_X^1$ are
\end{env}

called \emph{locally principal cycles};
such a cycle $Z$ is thus characterised by the fact that, for every $y\in X$, there exists an open neighbourhood $U$ of $y$ in $X$ and a regular meromorphic function $f$ on $U$ such that $Z|U=\sum_{x\in U\cap X^{(1)}}o_x(f)(\overline{\{x\}}\cap U)$.
If $Z=\sum_{x\in X^{(1)}}n_x\cdot\overline{\{x\}}$, it amounts to the same to say that, for every $y\in X$, if one sets $T_y=\Spec(\sh{O}_{X,x})$ and $Z_y=\sum_{x\in X^{(1)}\cap T_y}n_x(\overline{\{x\}}\cap T_y)$, $Z_y$ is \emph{principal}, in other words there exists a regular meromorphic function $g$ on $T_y$ such that $Z_y=\operatorname{cyc}(\operatorname{div}(g))$.
This results immediately from the preceding definition and from \sref{IV.20.3.7} which establishes a bijective correspondence between regular meromorphic functions on $T_y$ and the germs of regular meromorphic functions on the open neighbourhoods of $y$ in $X$ when $X$ is locally Noetherian.
One expresses also the fact that $Z_y$ is principal by saying that $Z$ is \emph{principal at the point $y$}.
The set of points where $Z$ is principal is evidently \emph{open}, by virtue of what precedes.

One deduces from the canonical homomorphism $\operatorname{cyc}:\operatorname{Div}(X)\to\mathfrak{Z}^1(X)$ a homomorphism $\operatorname{Div}(X)/\operatorname{Div.princ}(X)\to\mathfrak{Z}^1(X)/\mathfrak{Z}^1_{\mathrm{princ}}(X)$, by virtue of the definition of $\mathfrak{Z}^1_{\mathrm{princ}}(X)$.
One says that the group $\mathfrak{Z}^1(X)/\mathfrak{Z}^1_{\mathrm{princ}}(X)$ is the \emph{group of classes of $1$-codimensional cycles} of $X$ and one denotes it $\mathfrak{Cl}(X)$.
Two elements of $\mathfrak{Z}^1(X)$ having the same image in $\mathfrak{Cl}(X)$ are called \emph{linearly equivalent}.

\begin{env}[21.6.8]
\label{IV.21.6.8}
Consider in particular the case where $X=\Spec(A)$, where $A$ is an \emph{integrally closed} Noetherian local ring.
Then $X^{(1)}$ is the set of \emph{prime ideals of height $1$} of $A$, and $\mathfrak{Z}^1(X)$ identifies with the \emph{group of divisors} (in the sense of N.~Bourbaki) of the Krull ring $A$ (Bourbaki, \emph{Alg.~comm.}, chap.~VII, \textsection1, n\textsuperscript{o}~3, cor.\ of th.~2 and n\textsuperscript{o}~6, th.~3).

On the other hand, the regular meromorphic functions on $X$ identify with the elements $\neq 0$ of the field of fractions $K$ of $A$, the application $f\mapsto\operatorname{cyc}(\operatorname{div}(f))$ of $M(X)$ in $\mathfrak{Z}^1(X)$ identifies with the application denoted $f\mapsto\operatorname{div}(f)$ in Bourbaki (\emph{loc.~cit.}, \textsection1, n\textsuperscript{o}~1);
$\mathfrak{Z}^1_{\mathrm{princ}}(X)$ identifies thus with the \emph{group of principal divisors} of $A$ in the sense of Bourbaki, and $\mathfrak{Cl}(X)$ with the \emph{group of classes of divisors} of $A$ in the sense of Bourbaki (\emph{loc.~cit.}, \textsection1, n\textsuperscript{o}~2 and n\textsuperscript{o}~10).
\end{env}

\begin{theorem}[21.6.9]
\label{IV.21.6.9}
Let $X$ be a locally Noetherian normal prescheme.
\begin{enumerate}[label=\emph{(\roman*)}]
  \item The canonical homomorphism $\operatorname{cyc}:\operatorname{Div}(X)\to\mathfrak{Z}^1(X)$ is injective and its image is formed of locally principal cycles.
  \item The following conditions are equivalent:
  \begin{enumerate}[label=\emph{\alph*)}]
    \item The homomorphism $\operatorname{cyc}:\operatorname{Div}(X)\to\mathfrak{Z}^1(X)$ is bijective.
    \item Every $1$-codimensional cycle is locally principal.
    \item For every $x\in X$, the local ring $\sh{O}_{X,x}$ is \emph{factorial}.
  \end{enumerate}
  (One says then that $X$ is a \emph{locally factorial} prescheme.)
\end{enumerate}
\end{theorem}

\begin{proof}
\oldpage[IV-4]{275}
(i) It suffices to show that
\[
\label{IV.21.6.9.1}
  \operatorname{cyc}^{-1}(\mathfrak{Z}^{1+}(X))=\operatorname{Div}^+(X),
  \tag{21.6.9.1}
\]
since one has $\operatorname{Div}^+(X)\cap(-\operatorname{Div}^+(X))=0$ and $\mathfrak{Z}^{1+}(X)\cap(-\mathfrak{Z}^{1+}(X))=0$.
One is thus reduced to proving that if $D$ is a divisor on $X$ such that $\operatorname{mult}_x(D)\geq 0$ for every $x\in X^{(1)}$, one has $D\geq 0$.
Now, for every $x\notin X^{(1)}$, the local ring $\sh{O}_{X,x}$ is integrally closed and of dimension $0$ or $\geq 2$, hence one has $\operatorname{prof}(\sh{O}_{X,x})=0$ or $\operatorname{prof}(\sh{O}_{X,x})\geq 2$ $(\mathbf{0}$, 16.5.1).
On the other hand, at the points $x\in X^{(1)}$, the ring $\sh{O}_{X,x}$ is a discrete valuation ring $(\mathbf{II}$, 7.1.6), hence $\operatorname{prof}(\sh{O}_{X,x})=1$ $(\mathbf{0}$, 16.5.1);
the only points of $X$ such that $\operatorname{prof}(\sh{O}_{X,x})=1$ are thus the points of $X^{(1)}$, and the conclusion results by \sref{IV.21.1.8}.

(ii) The equivalence of~a) and~b) is clear from~(i).
By the characterisation of locally principal cycles given in \sref{IV.21.6.7}, and the relation between $1$-codimensional cycles on $\Spec(A)$ and divisors (in the sense of Bourbaki) of a Noetherian integrally closed ring $A$ \sref{IV.21.6.8}, condition~b) is equivalent to saying that for every $x\in X$, every divisor of the ring $\sh{O}_{X,x}$ is principal, in other words that the ring $\sh{O}_{X,x}$ is factorial (Bourbaki, \emph{Alg.~comm.}, chap.~VII, \textsection3, n\textsuperscript{o}~1), whence the equivalence of~b) and~c).

\begin{env}[21.6.9.2]
\label{IV.21.6.9.2}
When $A$ is a Noetherian \emph{factorial} ring, it is clear that $X=\Spec(A)$ is locally factorial, since every localisation of $A$ is also locally factorial (Bourbaki, \emph{Alg.~comm.}, chap.~VII, \textsection3, n\textsuperscript{o}~4, prop.~3).
If, in this case, one writes an element $f\neq 0$ of the field of fractions $K$ of $A$ in the form $r/s$, where $r$ and $s$ are two \emph{coprime} elements of $A$, then the divisors of $r$ and $s$ are well determined (\emph{loc.~cit.}, \textsection3, n\textsuperscript{o}~3), and these two divisors identify respectively with the \emph{divisor of zeros} and the \emph{divisor of poles} of $f$ \sref{IV.21.6.7}.
\end{env}
\end{proof}

\begin{corollary}[21.6.10]
\label{IV.21.6.10}
Let $X$ be a locally Noetherian normal prescheme.
\begin{enumerate}[label=\emph{(\roman*)}]
  \item There exists a canonical injective homomorphism
  \[
  \label{IV.21.6.10.1}
    \operatorname{Pic}(X)\to\mathfrak{Cl}(X).
    \tag{21.6.10.1}
  \]
  \item If $X$ is locally factorial, the homomorphism \sref{IV.21.6.10.1} is bijective, and reciprocally.
\end{enumerate}
\end{corollary}

\begin{proof}
One has seen \sref{IV.21.6.7} that the image of $\operatorname{Div.princ}(X)$ by the homomorphism $\operatorname{cyc}$ is $\mathfrak{Z}^1_{\mathrm{princ}}(X)$;
one deduces thus from \sref{IV.21.6.9} that the homomorphism $\operatorname{Div}(X)/\operatorname{Div.princ}(X)\to\mathfrak{Z}^1(X)/\mathfrak{Z}^1_{\mathrm{princ}}(X)=\mathfrak{Cl}(X)$ deduced from $\operatorname{cyc}$ is injective, and that it is bijective if and only if $X$ is locally factorial.
The conclusion results from this and from the fact that, when $X$ is locally Noetherian and reduced, the canonical homomorphism $\operatorname{Div}(X)/\operatorname{Div.princ}(X)\to\operatorname{Pic}(X)$ \sref{IV.21.3.3.1} is bijective \sref{IV.21.3.4}[(ii)].
\end{proof}

\begin{corollary}[21.6.11]
\label{IV.21.6.11}
Let $X$ be a locally Noetherian and locally factorial prescheme.
Then the sheaf $\sh{D}iv_X$ is flasque, and for every open $U$ of $X$, the canonical homomorphism $\operatorname{Pic}(X)\to\operatorname{Pic}(U)$ is surjective.
\end{corollary}

\begin{proof}
Taking into account \sref{IV.21.6.9}[(ii)], the first assertion is equivalent to saying that the sheaf $\mathscr{Z}_X^1$ is flasque, which was seen above \sref{IV.21.6.3}.
For every open $U$ of $X$, the homomorphism $\mathfrak{Z}^1(X)\to\mathfrak{Z}^1(U)$ is thus surjective;
since by virtue of \sref{IV.21.6.10}, the homomorphism $\operatorname{Pic}(X)\to\operatorname{Pic}(U)$ identifies canonically with $\mathfrak{Z}^1(X)/\mathfrak{Z}^1_{\mathrm{princ}}(X)\to\mathfrak{Z}^1(U)/\mathfrak{Z}^1_{\mathrm{princ}}(U)$, it is also surjective.
\end{proof}

\begin{proposition}[21.6.12]
\label{IV.21.6.12}
Let $X$ be a Noetherian reduced prescheme.
Let $(U_\lambda)_{\lambda\in L}$ be a decreasing filtered family of opens of $X$ satisfying the following conditions:
\begin{enumerate}
  \item[$1\textsuperscript{o}$] If $Y_\lambda=X-U_\lambda$, one has $\operatorname{codim}(Y_\lambda,X)\geq 2$ for every $\lambda\in L$.
  \item[$2\textsuperscript{o}$] For every $x\in\bigcap_{\lambda\in L}U_\lambda$, the ring $\sh{O}_{X,x}$ is factorial.
\end{enumerate}
\oldpage[IV-4]{276}
Then there exist canonical isomorphisms
\[
\label{IV.21.6.12.1}
  \varinjlim\operatorname{Div}(U_\lambda)\xrightarrow{\sim}\mathfrak{Z}^1(X),\qquad\varinjlim\operatorname{Pic}(U_\lambda)\xrightarrow{\sim}\mathfrak{Cl}(X)
  \tag{21.6.12.1}
\]
such that, for every open $V$ of $X$, the diagrams
\[
\label{IV.21.6.12.2}
  \xymatrix{
    \varinjlim\operatorname{Div}(U_\lambda)\ar[r]^{\sim}\ar[d] & \mathfrak{Z}^1(X)\ar[d]\\
    \varinjlim\operatorname{Div}(U_\lambda\cap V)\ar[r]^{\sim} & \mathfrak{Z}^1(V)
  }\qquad
  \xymatrix{
    \varinjlim\operatorname{Pic}(U_\lambda)\ar[r]^{\sim}\ar[d] & \mathfrak{Cl}(X)\ar[d]\\
    \varinjlim\operatorname{Pic}(U_\lambda\cap V)\ar[r]^{\sim} & \mathfrak{Cl}(V)
  }
  \tag{21.6.12.2}
\]
are commutative.
\end{proposition}

\begin{proof}
Hypothesis~$1\textsuperscript{o}$ on $U_\lambda$ implies that for every $\lambda\in L$, one has $X^{(1)}\subset U_\lambda$ \sref{IV.5.1.3.1} hence the restriction homomorphism $\mathfrak{Z}^1(X)\to\mathfrak{Z}^1(U_\lambda)$ is bijective and therefore one has a canonical isomorphism $\varinjlim\mathfrak{Z}^1(U_\lambda)\xrightarrow{\sim}\mathfrak{Z}^1(X)$.
The canonical homomorphisms $\operatorname{cyc}:\operatorname{Div}(U_\lambda)\to\mathfrak{Z}^1(U_\lambda)$ define thus, by passage to the inductive limit, the first of the canonical homomorphisms \sref{IV.21.6.12.1}, and the second is deduced from it by passage to quotients.
Furthermore, it results from hypothesis~$1\textsuperscript{o}$ that the $U_\lambda$ are dense in $X$, hence schematically dense since $X$ is reduced \sref{IV.11.10.4}, and therefore every meromorphic function on $X$ is entirely determined by its restriction to each $U_\lambda$;
one deduces immediately from this that in the isomorphism $\mathfrak{Z}^1(X)\xrightarrow{\sim}\mathfrak{Z}^1(U_\lambda)$, the image of $\mathfrak{Z}^1_{\mathrm{princ}}(X)$ is $\mathfrak{Z}^1_{\mathrm{princ}}(U_\lambda)$, hence one also has a canonical isomorphism $\varinjlim\mathfrak{Cl}(U_\lambda)\xrightarrow{\sim}\mathfrak{Cl}(X)$.
The second of the canonical homomorphisms \sref{IV.21.6.12.1} is thus also an isomorphism if the first is, and it remains therefore to prove the commutativity of the diagrams \sref{IV.21.6.12.2}, which is trivial.

Let us show first that the homomorphism $\varinjlim\operatorname{Div}(U_\lambda)\to\mathfrak{Z}^1(X)$ is \emph{injective}.
Set $T=\bigcap_\lambda U_\lambda$, and note that the $U_\lambda$ form a \emph{fundamental system of neighbourhoods} of $T$.
Indeed, for every open $V\supset T$, $X-V$ is a closed subset of $X$, hence a Noetherian space whose every closed irreducible subset admits a generic point;
since the sets $(X-V)\cap(X-U_\lambda)$ are closed in $X-V$, form an increasing filtered family and have for union $X-V$, our assertion results from $(\mathbf{0_{III}}$, 9.2.4).
This being so, it suffices to see that if $D\in\operatorname{Div}(U_\lambda)$ is such that $\operatorname{cyc}(D)=0$ in $\mathfrak{Z}^1(U_\lambda)$, then there exists $\mu\geq\lambda$ such that $D|U_\mu=0$.
By virtue of what precedes, it will suffice to see that for every $x\in T$, $D_x=0$, and we are reduced to the case where $X=\Spec(\sh{O}_{X,x})$ with $x\in T$;
but by hypothesis $\sh{O}_{X,x}$ is factorial, hence $\Spec(\sh{O}_{X,x})$ is normal;
hence the conclusion results from \sref{IV.21.6.9}[(i)].

\oldpage[IV-4]{277}
To prove that the homomorphism $\varinjlim\operatorname{Div}(U_\lambda)\to\mathfrak{Z}^1(X)$ is bijective, it will suffice to prove that for every $Z\in\mathfrak{Z}^1(X)$ and every $x\in T$, there exists a neighbourhood $W(x)$ of $x$ in $X$ and a divisor $D^{(x)}$ on $W(x)$ such that $\operatorname{cyc}(D^{(x)})=Z|W(x)$.
Indeed, one can then cover the quasi-compact set $T$ by a finite number of the $W(x_i)$, and by virtue of the first part of the proof, since the couples $(i,j)$ are finite in number, there will exist an index $\lambda$ such that in each of the $W(x_i)\cap W(x_j)\cap U_\lambda$, the restrictions of $D^{(x_i)}$ and $D^{(x_j)}$ coincide;
since one can suppose moreover that $U_\lambda$ is contained in the union of the $W(x_i)$, one sees that the restrictions $D^{(x_i)}|(W(x_i)\cap U_\lambda)$ are the restrictions of a single $D\in\operatorname{Div}(U_\lambda)$ which will be such that $\operatorname{cyc}(D)=Z|U_\lambda$.
We are thus reduced again to the case where $X=\Spec(\sh{O}_{X,x})$ with $x\in T$;
but by hypothesis $\sh{O}_{X,x}$ is factorial, hence the conclusion results from \sref{IV.21.6.9}[(ii)].
\end{proof}

\begin{corollary}[21.6.13]
\label{IV.21.6.13}
Let $A$ be an integral integrally closed Noetherian local ring of dimension $\geq 2$.
Set $X=\Spec(A)$, and let $a$ be the closed point of $X$, $U=X-\{a\}$.
For $A$ to be factorial, it is necessary and sufficient that $U$ be locally factorial and that $\operatorname{Pic}(U)=0$.
\end{corollary}

\begin{proof}
Indeed, to say that $A$ is factorial is equivalent to saying that $\mathfrak{Cl}(X)=0$ (Bourbaki, \emph{Alg.~comm.}, chap.~VII, \textsection1, n\textsuperscript{o}~4, cor.\ of th.~2 and \textsection3, n\textsuperscript{o}~2, th.~1);
it thus suffices to use the existence of the second isomorphism \sref{IV.21.6.12.1}, taking the family $(U_\lambda)$ restricted to the single open $U$.
\end{proof}

\begin{corollary}[21.6.14]
\label{IV.21.6.14}
Let $A$ be a Noetherian local ring of dimension $\geq 2$, $X=\Spec(A)$, $a$ the closed point of $X$, $U=X-\{a\}$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $A$ is factorial.
  \item $\operatorname{Pic}(U)=0$ and $\operatorname{prof}(A)\geq 2$ (conditions that we will express later \sref{IV.21.13} by saying that the ring $A$ is \emph{parafactorial}), and $U$ is locally factorial.
\end{enumerate}
\end{corollary}

\begin{proof}
It is clear that if $A$ is factorial, it is normal, hence $\dim(A)\geq 2$ since $\operatorname{prof}(A)\geq 2$ $(\mathbf{0}$, 16.5.1) and \sref{IV.21.6.13} shows that~a) implies~b).
Conversely, if~b) is satisfied, it suffices to see that $A$ is integrally closed to conclude by \sref{IV.21.6.13} that~b) implies~a).
By virtue of Serre's criterion \sref{IV.5.8.6}, it suffices to verify for $X$ the conditions $(\mathrm{R}_1)$ and $(\mathrm{S}_2)$.
Now, $U$ being locally factorial satisfies these conditions, and the hypothesis $\operatorname{prof}(A)\geq 2$ implies that $X$ also satisfies them.
\end{proof}

\subsection{Interpretation of positive $1$-codimensional cycles in terms of subpreschemes}
\label{IV.21.7}

\oldpage[IV-4]{278}

\begin{env}[21.7.1]
\label{IV.21.7.1}
Let $X$ be a locally Noetherian prescheme, $C=\sum_{x\in X^{(1)}}n_x\cdot\overline{\{x\}}$ a positive $1$-codimensional cycle (so that one has $n_x\geq 0$ for every $x\in X^{(1)}$, and $n_x=0$ except on a locally finite set of points).
Denote by $Y(C)$ the closed subprescheme of $X$, closed image $(\mathbf{I}$, 9.5.3 and 9.5.1) of the sum prescheme $Y'(C)=\coprod_{x\in X^{(1)}}\Spec(\sh{O}_{X,x}/\mathfrak{m}_x^{n_x})$ by the canonical morphism, and by $\sh{J}_X(C)$ (or $\sh{J}(C)$) the Ideal of $\sh{O}_X$ defining $Y(C)$.
\end{env}

\begin{proposition}[21.7.2]
\label{IV.21.7.2}
Let $X$ be a locally Noetherian prescheme satisfying the condition $(\mathrm{R}_1)$ \sref{IV.5.8.2}.
For a closed subprescheme $Y$ of $X$ to be of the form $Y(C)$, where $C$ is a positive $1$-codimensional cycle, it is necessary and sufficient that it satisfy the two following conditions:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item $Y$ is purely of codimension $1$.
  \item $Y$ satisfies the condition $(\mathrm{S}_1)$, in other words \sref{IV.5.7.5} has no associated immersed prime cycle.
\end{enumerate}
One then has
\[
\label{IV.21.7.2.1}
  \operatorname{mult}_x(C)=\operatorname{long}\sh{O}_{X,x}.
  \tag{21.7.2.1}
\]

The application $C\mapsto Y(C)$ is a bijection of $\mathfrak{Z}^{1+}(X)$ on the set of closed subpreschemes of $X$ satisfying conditions~\emph{(i)} and~\emph{(ii)}.
\end{proposition}

\begin{proof}
The conditions are necessary (without assuming that $X$ satisfies $(\mathrm{R}_1)$).
Indeed, the question being local on $X$, one can suppose that $X=\Spec(A)$, where $A$ is a Noetherian ring;
one then has $Y(C)=\Spec(A/\mathfrak{J})$, where $\mathfrak{J}$ is, by definition $(\mathbf{I}$, 9.5.1) the kernel of the canonical homomorphism $A\to\bigoplus A_{\mathfrak{p}_i}/\mathfrak{p}_i^{n_i}A_{\mathfrak{p}_i}$, where the $\mathfrak{p}_i$ are the prime ideals corresponding to $x\in X^{(1)}$ for which $n_x\geq 0$, and where one has set $n_i=n_x$.
The reciprocal image $\mathfrak{q}_i$ of $\mathfrak{p}_i^{n_i}A_{\mathfrak{p}_i}$ in $A$ is a $\mathfrak{p}_i$-primary ideal and one has $\mathfrak{J}=\bigcap_i\mathfrak{q}_i$.
Moreover, since the $x\in X^{(1)}$ are such that $\overline{\{x\}}$ is of codimension $1$, no point of $X^{(1)}$ can be adherent to another point of $X^{(1)}$.
Hence the $\mathfrak{p}_i$ are the minimal prime ideals of $A/\mathfrak{J}$ and the set of the $\mathfrak{p}_i$ is equal to $\operatorname{Ass}(A/\mathfrak{J})$, which proves conditions~(i) and~(ii).

The conditions are sufficient;
denoting by $\sh{J}$ the Ideal of $\sh{O}_X$ defining $Y$, hypothesis~(ii) implies that $\operatorname{Ass}(\sh{O}_X/\sh{J})$ is identical to the set $F$ of maximal points of $Y$, and hypothesis~(i) implies that $F$ is contained in $X^{(1)}$;
hence \sref{IV.3.2.6} $\sh{O}_X/\sh{J}$ identifies with a sub-$\sh{O}_X$-Module of $\bigoplus_{x\in F}\sh{G}(x)$, where for each $x\in F$, $\sh{G}(x)$ is an irreducible $\sh{O}_X$-Module such that $\operatorname{Ass}(\sh{G}(x))=\{x\}$.
Now, since $X$ satisfies $(\mathrm{R}_1)$, for each $x\in F$, $\sh{O}_{X,x}$ is a discrete valuation ring and therefore the primary ideals $\neq 0$ of this ring are the powers $\mathfrak{m}_x^{n_x}$ of the maximal ideal;
supposing further that $X=\Spec(A)$, one concludes that $\sh{J}=\widetilde{\mathfrak{J}}$, where $\mathfrak{J}=\bigcap_i\mathfrak{q}_i$, the $\mathfrak{q}_i$ being the reciprocal images of $\mathfrak{p}_i^{n_i}A_{\mathfrak{p}_i}$ at the corresponding points of $F$.
One sees well that $Y$ is of the form $Y(C)$.
\end{proof}

\begin{corollary}[21.7.3]

\begin{proposition}[21.7.4]
\label{IV.21.7.4}
Let $X$ be a locally Noetherian prescheme, reduced at each of its isolated points.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item The canonical homomorphism $c:\sh{D}iv_X\to\mathscr{Z}_X^1$ \sref{IV.21.6.4} is an isomorphism of sheaves of ordered groups, and on $\operatorname{Div}(X)=\operatorname{Div.princ}(X)$.
\end{enumerate}
\begin{enumerate}[label=\emph{\alph*$'$)}]
  \item The canonical homomorphism $c:\sh{D}iv_X\to\mathscr{Z}_X^1$ is injective, and every positive $1$-codimensional cycle is the image by $\operatorname{cyc}$ of a positive divisor principal, in other words is of the form $\operatorname{cyc}(\operatorname{div}(f))$, where $f$ is a regular section of $\sh{O}_X$ over $X$.
\end{enumerate}
\begin{enumerate}[label=\emph{\alph*$''$)}]
  \item For every closed integral subprescheme $Y$ of $X$, of codimension $1$, there exists a regular section $f$ of $\sh{O}_X$ over $X$ such that $Y$ is defined by the Ideal $f\sh{O}_X$.
\end{enumerate}
\begin{enumerate}
  \item[b)] $X$ is normal and every $1$-codimensional cycle on $X$ is principal.
  \item[c)] $X$ is locally factorial and $\operatorname{Pic}(X)=0$.
  \item[d)] $X$ is normal, and for every open $U$ of $X$, one has $\operatorname{Pic}(U)=0$.
\end{enumerate}
\end{proposition}

\begin{proof}
The equivalence of~a), b) and~c) results immediately from \sref{IV.21.7.4} and from \sref{IV.21.6.11}.
Furthermore, it results immediately from~c) that every non-empty open $U$ of $X$ satisfies also the same conditions, so that~c) implies~d).
It remains to see that~d) implies~b).
Now, denote by $U_\lambda$ the opens that are complementary of finite unions of closed sets of the form $\overline{\{x\}}$, where $\dim(\sh{O}_{X,x})\geq 2$;
it is clear that the $U_\lambda$ form a decreasing filtered family and that for every $x\in\bigcap_\lambda U_\lambda$, one has $\dim(\sh{O}_{X,x})\leq 1$, hence $\sh{O}_{X,x}$ is a principal ring under the hypothesis.
One can thus apply to the family $(U_\lambda)$ the proposition \sref{IV.21.6.12}, which shows that $\mathfrak{Cl}(X)$ is isomorphic to $\varinjlim\operatorname{Pic}(U_\lambda)$, hence $\mathfrak{Cl}(X)=0$ by virtue of the hypothesis, which proves~b) by definition.
\end{proof}

When $X=\Spec(A)$, where $A$ is an integral Noetherian ring, the conditions equivalent of \sref{IV.21.7.6} are equivalent to saying that $A$ is a \emph{factorial} ring.
\end{remark}

\begin{lemma}[21.8.1]
\label{IV.21.8.1}
Let $f:X'\to X$ be an integral morphism of preschemes.
For every $\sh{O}_{X'}$-Module locally free $\sh{E}'$ of constant rank $n$ and every $x\in X$, there exists an open neighbourhood $U$ of $x$ in $X$ such that, if one sets $U'=f^{-1}(U)$, $\sh{E}'|U'$ is isomorphic to $\sh{O}_{U'}^n$.
\end{lemma}

\begin{proof}
The question being local on $X$, one can restrict to the case where $X=\Spec(A)$ is affine, in which case $X'=\Spec(A')$, where $A'$ is an integral $A$-algebra.
Then $A'$ is the inductive limit of its finite sub-$A$-algebras $A'_\lambda$.
Setting $X'_\lambda=\Spec(A'_\lambda)$ and denoting by $\rho_\lambda:X'\to X'_\lambda$ the structural morphism, it results from \sref{IV.8.5.2} and \sref{IV.8.5.5} that there exists an index $\lambda$ and an $\sh{O}_{X'_\lambda}$-Module locally free $\sh{E}'_\lambda$ of rank $n$ such that $\sh{E}'=\rho_\lambda^*(\sh{E}'_\lambda)$, and it will suffice to prove the lemma for $X'_\lambda$ and $\sh{E}'_\lambda$;
in other words, one can restrict to the case where the morphism $f$ is \emph{finite}.
Setting $B=\sh{O}_{X,x}$ and let $B'$ be the ring of the affine scheme $X'_0=X'\times_X\Spec(\sh{O}_{X,x})$;
since $B$ is a local ring and $B'$ is a finite $B$-algebra, $B'$ is a semi-local ring (Bourbaki, \emph{Alg.~comm.}, chap.~V, \textsection2, n\textsuperscript{o}~1, prop.~3);
one concludes that the $\sh{O}_{X'_0}$-Module locally free $\sh{E}'_0=\sh{E}'\otimes_{\sh{O}_{X'}}\sh{O}_{X,x}$ is isomorphic to $\sh{O}_{X'_0}^n$ (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection5, n\textsuperscript{o}~3, prop.~5).
Considering $X'_0$ as the projective limit of preschemes induced by $X'$ on the open affine neighbourhoods of $x$ in $X$, following the method of \sref{IV.8.1.2}[a)], and applying \sref{IV.8.5.2.5}, one obtains the sought conclusion.
\end{proof}

\begin{corollary}[21.8.2]
\label{IV.21.8.2}
Let $f:X'\to X$ be an integral morphism of preschemes. Then:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item $\mathrm{R}^1f_*(\sh{O}_{X'}^*)=0$ $(\mathbf{0_{III}}$, 12.2.1).
  \item The group $\operatorname{Pic}(X')$ is canonically isomorphic to $\mathrm{H}^1(X,f_*(\sh{O}_{X'}^*))$.
\end{enumerate}
\end{corollary}

\begin{proof}
(i) $\mathrm{R}^1f_*(\sh{O}_{X'}^*)$ is the sheaf associated to the presheaf of commutative groups $U\mapsto\mathrm{H}^1(f^{-1}(U),\sh{O}_{X'}^*)$ (\emph{loc.~cit.});
the fibre of this sheaf at a point $x$ can therefore $(\mathbf{0_I}$, 5.4.7) be identified with the commutative group $\varinjlim\operatorname{Pic}(f^{-1}(U))$, where $U$ ranges over the open neighbourhoods of $x$, the transition homomorphisms $\operatorname{Pic}(f^{-1}(U'))\to\operatorname{Pic}(f^{-1}(U))$ for two open $U\subset U'$ being defined by \sref{IV.21.3.2.4}.
Now, for every $x\in X$, every open neighbourhood $U'$ of $x$ in $X$, and every $\zeta\in\operatorname{Pic}(f^{-1}(U'))$, it results from \sref{IV.21.8.1} that there exists an open $U\subset U'$ of $x$ such that the image of $\zeta$ in $\operatorname{Pic}(f^{-1}(U))$ is zero.
Hence the fibre of $\mathrm{R}^1f_*(\sh{O}_{X'}^*)$ at the point $x$ is zero.

\oldpage[IV-4]{281}
(ii) The Leray spectral sequence for the composed functor $\sh{F}'\mapsto\Gamma(X,f_*(\sh{F}'))$ of sheaves of commutative groups on $X'$ (T, 2.4) gives the exact sequence of low degree terms
\[
  0\to\mathrm{H}^1(X,f_*(\sh{O}_{X'}^*))\to\mathrm{H}^1(X',\sh{O}_{X'}^*)\to\mathrm{H}^0(X,\mathrm{R}^1f_*(\sh{O}_{X'}^*))
\]
and the conclusion results from~(i) and from the isomorphism of $\operatorname{Pic}(X')$ and $\mathrm{H}^1(X',\sh{O}_{X'}^*)$ $(\mathbf{0_I}$, 5.4.7).
\end{proof}

\begin{proposition}[21.8.3]
\label{IV.21.8.3}
Let $f=(\psi,\theta):X'\to X$ be an integral morphism of preschemes;
suppose that, for every open $U$ of $X$, the homomorphism $\Gamma(\theta):\Gamma(U,\sh{O}_X)\to\Gamma(U,f_*(\sh{O}_{X'}))$ transforms regular elements in regular elements, which allows to canonically prolong the homomorphism $\theta:\sh{O}_X\to f_*(\sh{O}_{X'})$ in a homomorphism of sheaves of rings $\theta':\sh{M}_X\to f_*(\sh{M}_{X'})$, whence homomorphisms of sheaves of multiplicative groups $\theta^*:\sh{O}_X^*\to f_*(\sh{O}_{X'}^*)$ and $\theta'^*:\sh{M}_X^*\to f_*(\sh{M}_{X'}^*)$, giving by passage to quotients a homomorphism $\theta''':\sh{D}iv_X\to f_*(\sh{D}iv_{X'})$.
One then has a commutative diagram

\label{IV.21.8.3.1}
  \xymatrix{
    1\ar[r] & \sh{O}_X^*\ar[r]\ar[d]^{\theta^*} & \sh{M}_X^*\ar[r]\ar[d]^{\theta'^*} & \sh{D}iv_X\ar[r]\ar[d]^{\theta'''} & 0\\

in which the two rows are exact.
\end{proposition}

\begin{proof}
The only point to verify is the exactness of the second row of the diagram, which results from the application of the exact cohomology sequence for the functor $f_*$ to the short exact sequence of sheaves of commutative groups on $X'$
\[
  1\to\sh{O}_{X'}^*\to\sh{M}_{X'}^*\to\sh{D}iv_{X'}\to 0
\]
and from \sref{IV.21.8.2}.
\end{proof}

\begin{proposition}[21.8.5]
\label{IV.21.8.5}

\begin{enumerate}[label=\emph{(\roman*)}]
  \item The homomorphism $\theta':\sh{M}_X\to f_*(\sh{M}_{X'})$ is defined and bijective, the homomorphism
  \[
    \theta^*:\sh{O}_X^*\to f_*(\sh{O}_{X'}^*)

  is injective, the homomorphism $\theta''':\sh{D}iv_X\to f_*(\sh{D}iv_{X'})$ is surjective, $\operatorname{Ker}(\theta''')$ is isomorphic to $\sh{N}=\operatorname{Coker}(\theta^*)$ and the support of the sheaf of multiplicative groups $\sh{N}$ is contained in $S=X-U$.

    \xymatrix{
      1\ar[r] & \displaystyle\prod_{s\in S}\sh{O}_{X,s}^{*'}/\sh{O}_{X,s}^*\ar[r]^-{j}\ar[d] & \operatorname{Div}(X)\ar[r]\ar[d]^{l_X} & \operatorname{Div}(X')\ar[r]\ar[d]^{l_{X'}} & 0\\

    }
    \tag{21.8.5.1}
  \]
  where the two rows are exact and the left vertical arrow is the canonical homomorphism.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) The hypothesis implies that one has a canonical isomorphism $\sh{M}_X^*\xrightarrow{\sim}f_*(\sh{M}_{X'}^*)$ for the sheaves of germs of pseudo-functions \sref{IV.20.2.2}, whence the assertion relative to $\theta'$, the assertion relative to $\theta'$ following from \sref{IV.20.2.11}.
The rest of assertion~(i) follows from \sref{IV.21.8.4}, except for what concerns the support of $\sh{N}$, which results directly from the hypothesis on $U$.

(ii) Applying the exact cohomology sequence to the two short exact sequences of sheaves of commutative groups
\[
\label{IV.21.8.5.2}
  1\to\sh{N}\to\sh{D}iv_X\to f_*(\sh{D}iv_{X'})\to 0

\[
  1\to\sh{O}_X^*\to f_*(\sh{O}_{X'}^*)\to\sh{N}\to 1
\]
one obtains respectively the exact sequences

  1\to\Gamma(X,\sh{N})\xrightarrow{j}\operatorname{Div}(X)\to\operatorname{Div}(X')\to\mathrm{H}^1(X,\sh{N})
  \tag{21.8.5.3}
\]
\[
  \Gamma(X',\sh{O}_{X'}^*)\to\Gamma(X,\sh{N})\xrightarrow{\beta}\operatorname{Pic}(X)\to\operatorname{Pic}(X')\to\mathrm{H}^1(X,\sh{N})
\]
taking into account the canonical isomorphism $\operatorname{Pic}(X')\xrightarrow{\sim}\mathrm{H}^1(X,f_*(\sh{O}_{X'}^*))$ \sref{IV.21.8.2}.
Now, since the support of $\sh{N}$ is contained in the discrete set $S$, closed in $X$, one has $\mathrm{H}^1(X,\sh{N})=0$, and the conclusion results from~(i) and from the isomorphism of $\operatorname{Pic}(X')$ and $\mathrm{H}^1(X,f_*(\sh{O}_{X'}^*))$ \sref{IV.21.8.2}.

\begin{env}[21.9.1]
\label{IV.21.9.1}
Let $X$ be a topological space, $x$ a point of $X$, $i_x:\{x\}\to X$ the canonical injection.
If $A(x)$ is a commutative group, one can consider it as a sheaf of commutative groups on the space $\{x\}$ reduced to a single point, and take its direct image $(i_x)_*(A(x))$ which is a sheaf of commutative groups on $X$ $(\mathbf{0_I}$, 3.4.1);
it results immediately from the definitions that for every open $U$ of $X$, $\Gamma(U,(i_x)_*(A(x)))=A(x)$ if $x\in U$, and is reduced to $0$ in the contrary case;
hence, for $y\in\overline{\{x\}}$, $((i_x)_*(A(x)))_y=A(x)$, and for $y\notin\overline{\{x\}}$, $((i_x)_*(A(x)))_y=0$.

If now $\sh{F}$ is a sheaf of commutative groups on $X$ and $Y$ a subset of $X$, for every open $U$ of $X$, one has a canonical homomorphism
\[
\label{IV.21.9.1.1}
  \Gamma(U,\sh{F})\to\prod_{x\in U\cap Y}\sh{F}_x=\prod_{x\in Y}\Gamma(U,(i_x)_*(\sh{F}_x))
  \tag{21.9.1.1}
\]
and since these homomorphisms commute with restriction operators, they define a canonical \emph{homomorphism of sheaves}
\[
\label{IV.21.9.1.2}
  j_Y:\sh{F}\to\prod_{x\in Y}(i_x)_*(\sh{F}_x).
  \tag{21.9.1.2}
\]
\end{env}

\begin{lemma}[21.9.2]
\label{IV.21.9.2}
Let $X$ be a locally Noetherian topological space, $X_0$ the set of its closed points, $\sh{F}$ a sheaf of commutative groups on $X$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item The canonical homomorphism $j_{X_0}:\sh{F}\to\prod_{x\in X_0}(i_x)_*(\sh{F}_x)$ is injective and has for image $\bigoplus_{x\in X_0}(i_x)_*(\sh{F}_x)$.
\end{enumerate}
\begin{enumerate}[label=\emph{\alph*$'$)}]
  \item There exists a family of commutative groups $(A(x))_{x\in X_0}$ such that $\sh{F}$ is isomorphic to $\bigoplus_{x\in X_0}(i_x)_*(A(x))$.
\end{enumerate}
\begin{enumerate}
  \item[b)] Every section of $\sh{F}$ over an open $U$ of $X$ has discrete support contained in $X_0$.
\end{enumerate}

When~b) is thus satisfied, for every $x\in X_0$, the group $A(x)$ is determined up to unique isomorphism and is isomorphic to $\sh{F}_x$.
Furthermore, the sheaf $\sh{F}$ is then flasque.
\end{lemma}

\begin{proof}
Since the points of $X_0$ are closed in $X$, one has $((i_x)_*(A(x)))_y=0$ for every $y\neq x\in X_0$;
this remark proves the uniqueness assertion relative to the groups $A(x)$, and it is clear moreover that~a) implies~a$'$).
To see that~a$'$) implies~b), one can restrict to the case where $U$ is Noetherian;
then (G,~II, 3.10) one has
\[
  \Gamma(U,\bigoplus_{x\in X_0}(i_x)_*(A(x)))=\bigoplus_{x\in X_0}\Gamma(U,(i_x)_*(A(x))),
\]
hence every section $s$ of $\sh{F}$ over $U$ is a direct sum of a \emph{finite} number of sections $s_k\in\Gamma(U,(i_{x_k})_*(A(x_k)))$ $(1\leq k\leq n)$ with $x_k\in X_0$.
Since $x_k$ is a closed point of $X$, the support of $s_k$ is reduced to $x_k$, hence the support of $s$ is contained in the finite set of the $x_k$, which is evidently discrete since the points $x_k$ are closed in $X$.
Let us show finally that~b) implies~a) and that $\sh{F}$ is flasque.

Let us show that $\sh{F}$ is flasque.
Consider a section $s$ of $\sh{F}$ over an open $U$ of $X$;
since the support of $s$ is a discrete closed subset of $X$, we prolong $s$ to a section $s'$ of $\sh{F}$ over $X$ by setting $s'_x=0$ in $X-U$, whence our assertion.
\end{proof}

\begin{proposition}[21.9.4]
\label{IV.21.9.4}
Let $X$ be a locally Noetherian prescheme, $X^{(1)}$ the set of points $x\in X$ such that $\dim(\sh{O}_{X,x})=1$.
One then has a canonical isomorphism
\[
\label{IV.21.9.4.1}
  \sh{D}iv_X\xrightarrow{\sim}\bigoplus_{x\in X^{(1)}}(i_x)_*(\operatorname{Div}(\sh{O}_{X,x}))
  \tag{21.9.4.1}
\]
and $\sh{D}iv_X$ is flasque.
\end{proposition}

\begin{proof}
Taking into account the isomorphism \sref{IV.21.4.6.1}, the homomorphism \sref{IV.21.9.4.1} is defined by \sref{IV.21.9.1.1};
let us prove that $c$ is a bijection.
Since $\dim(X)\leq 1$, the points of $X^{(1)}$ are the \emph{non-isolated closed points} of $X$.
One is reduced to proving that:
$1\textsuperscript{o}$ $\sh{D}iv_X$ satisfies condition~b) of \sref{IV.21.9.2};
$2\textsuperscript{o}$ for every isolated point $x\in X$, one has $(\sh{D}iv_X)_x=0$.
The second point results from the fact that $\sh{O}_{X,x}$ is an Artinian ring and that every regular element of $\sh{O}_{X,x}$ is invertible.
For the $1\textsuperscript{o}$, it suffices to note that for every open $U$ of $X$, every divisor $D\in\operatorname{Div}(U)$, the maximal points of the support $S$ of $D$ are such that $\operatorname{prof}(\sh{O}_{X,x})=1$ \sref{IV.21.1.9}, hence \emph{a fortiori} belong to $X^{(1)}\cap U$;
since $\dim(X)\leq 1$, the set of these points is equal to $S$, and $S$ is thus discrete, the set of irreducible components of $S$ being locally finite.
\end{proof}

\begin{proposition}[21.9.11]

Let $X$ be a Noetherian prescheme, $X_0$ a closed subprescheme of $X$ having the following properties:
\begin{enumerate}

\end{enumerate}
Under these conditions:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item Let $Z_0$ be the union of the sets $\overline{\{x\}}\cap X_0$ when $x$ ranges over the part of $\operatorname{Ass}(\sh{O}_X)$ formed of points such that $\overline{\{x\}}\cap X_0$ is finite.
  Then, for every divisor $D_0$ on $X_0$, whose support does not meet $Z_0$, there exists a divisor $D$ on $X$ whose reciprocal image by the canonical injection $X_0\to X$ exists \sref{IV.21.4.5} and is equal to $D_0$;
  if moreover $D_0\geq 0$, one can suppose $D\geq 0$.
  \item Suppose in addition that there exists in $X_0$ an affine open $U_0$ containing $(\operatorname{Ass}(\sh{O}_{X_0}))\cup Z_0$ (condition automatically satisfied when there exists an $\sh{O}_{X_0}$-Module ample $(\mathbf{II}$, 4.5.4)).
  Then the canonical homomorphism \sref{IV.21.3.2.4} $\operatorname{Pic}(X)\to\operatorname{Pic}(X_0)$ is surjective.
\end{enumerate}
\end{proposition}

\begin{corollary}[21.9.12]
\label{IV.21.9.12}
\oldpage[IV-4]{288}
Let $A$ be a henselian local ring \sref{IV.18.5.8}, $S=\Spec(A)$, $s_0$ the closed point of $S$, $f:X\to S$ a separated morphism of finite presentation, and suppose that the set $X_0=f^{-1}(s_0)$ is of dimension $\leq 1$.
Then, for every closed subscheme $X'_0$ of $X$, having $X_0$ as underlying set and of finite presentation over $S$, the canonical homomorphism \sref{IV.21.3.2.4} $\operatorname{Pic}(X)\to\operatorname{Pic}(X'_0)$ is surjective.
\end{corollary}

\subsection{Reciprocal and direct images of $1$-codimensional cycles}
\label{IV.21.10}

\oldpage[IV-4]{289}

In a later chapter dedicated to intersection theory, the notions of inverse and direct image of cycles will be developed systematically.
In the present section, we will content ourselves with defining these notions in certain particular cases useful, and for \emph{$1$-codimensional cycles}, these definitions being chosen in such a way that they are compatible with the corresponding definitions for divisors \sref{IV.21.4} and \sref{IV.21.5}, taking into account the homomorphism $\operatorname{cyc}$ defined in \sref{IV.21.6}.

\begin{env}[21.10.1]
\label{IV.21.10.1}
\oldpage[IV-4]{290}
Let $X$, $X'$ be two locally Noetherian preschemes, $f:X'\to X$ a morphism, $T$ a part of $X$;
suppose that the image by $f$ of every maximal point of $X'$ is either a maximal point of $X$, and that, for every $x'\in X'^{(1)}$ (i.e.\ such that $\dim\sh{O}_{X',x'}=1$), setting $x=f(x')$, one has \emph{one} of the three following conditions:
\begin{enumerate}
  \item[$1\textsuperscript{o}$] $x\notin T$, one takes $n_{x'}=0$;
  \item[$2\textsuperscript{o}$] $x\in X^{(1)}$ and $\sh{O}_{X',x'}$ is a flat $\sh{O}_{X,x}$-module, one knows \sref{IV.6.1.1} that $\sh{O}_{X',x'}/\mathfrak{m}_x\sh{O}_{X',x'}$ is an $\sh{O}_{X',x'}$-module of finite length $\lambda_{x'}$, and one takes $n_{x'}=\lambda_{x'}n_x$;
  \item[$3\textsuperscript{o}$] $\sh{O}_{X,x}$ is factorial and $\mathfrak{m}_x\notin\operatorname{Ass}(\sh{O}_{X',x'})$, one knows \sref{IV.21.6.9} that the canonical homomorphism $\operatorname{cyc}:\operatorname{Div}(\sh{O}_{X,x})\to\mathfrak{Z}^1(\Spec(\sh{O}_{X,x}))$ is bijective, and moreover since $\dim(\sh{O}_{X',x'})=1$ and $\operatorname{Ass}(\sh{O}_{X',x'})$ is composed uniquely of the maximal points of $\Spec(\sh{O}_{X',x'})$, $x'$ being the unique closed point of $\Spec(\sh{O}_{X',x'})$, $\mathfrak{Z}^1(\Spec(\sh{O}_{X',x'}))$ is canonically isomorphic to $\bb{Z}$.
\end{enumerate}
Under these conditions, we propose, for every $1$-codimensional cycle $Z$ on $X$ of support contained in $T$, to define a $1$-codimensional cycle $Z'=f^*(Z)$, so that $Z\mapsto f^*(Z)$ is an \emph{homomorphism} of ordered groups of the subgroup of $\mathfrak{Z}^1(X)$ of support contained in $T$, in the ordered group $\mathfrak{Z}^1(X')$.
For this, let
\[
\label{IV.21.10.1.1}
  Z=\sum_{x\in T\cap X^{(1)}}n_x\cdot\overline{\{x\}}
  \tag{21.10.1.1}
\]
where the family of the $x\in T\cap X^{(1)}$ such that $n_x\neq 0$ is locally finite.
For every $x'\in X'^{(1)}$, define an integer $n_{x'}$ in the following way, setting $x=f(x')$:
in case $1\textsuperscript{o}$ take $n_{x'}=0$;
in case $2\textsuperscript{o}$ take $n_{x'}=\lambda_{x'}n_x$;
in case $3\textsuperscript{o}$, since $\sh{O}_{X,x}$ is factorial and $\mathfrak{m}_x\notin\operatorname{Ass}(\sh{O}_{X',x'})$, the homomorphism $f^*:\operatorname{Div}(\sh{O}_{X,x})\to\operatorname{Div}(\sh{O}_{X',x'})$ is defined, and it results from \sref{IV.21.4.5}[(ii)] that the homomorphism $f^*:\operatorname{Div}(\sh{O}_{X,x})\to\operatorname{Div}(\sh{O}_{X',x'})$ is defined;
finally, $x'$ being the unique closed point of $\Spec(\sh{O}_{X',x'})$, $\mathfrak{Z}^1(\Spec(\sh{O}_{X',x'}))$ is canonically isomorphic to $\bb{Z}$.
\end{env}

One can now set
\[
\label{IV.21.10.3}
  f^*(Z)=\sum_{x'\in X'^{(1)}}n_{x'}\cdot\overline{\{x'\}},
  \tag{21.10.3}
\]
and one says that the $1$-codimensional cycle $f^*(Z)$ is the \emph{reciprocal image} of $Z$ by $f$.

\begin{proposition}[21.10.4]
\label{IV.21.10.4}
Suppose the conditions of \sref{IV.21.10.1} satisfied.
Then, for every divisor $D$ on $X$ such that $\operatorname{Supp}(D)\subset T$ and such that $f^*(D)$ is defined \sref{IV.21.4.2}, one has
\[
\label{IV.21.10.4.1}
  \operatorname{cyc}(f^*(D))=f^*(\operatorname{cyc}(D)).
  \tag{21.10.4.1}
\]
\end{proposition}

\begin{proof}
The question being local on $X$, one can restrict to the case where $X=\Spec(A)$ is affine, where $D=\operatorname{div}(t)$, where $t$ is a regular non-invertible element of $A$, and where the subprescheme $Y(D)$ \sref{IV.21.2.12} has only one maximal point $y$, so that $\operatorname{cyc}(D)=n_y\cdot\overline{\{y\}}$, where $n_y$ is the length of $\sh{O}_{X,y}/t_y\sh{O}_{X,y}$ \sref{IV.21.6.5}[.1].
One has seen in the proof of \sref{IV.21.10.2}[, B)] that the points $x'\in X'$ such that $\operatorname{mult}_{x'}(f^*(\operatorname{cyc}(D)))\neq 0$ are maximal points of $f^{-1}(\overline{\{y\}})$.
If the point $x'$ is in case $3\textsuperscript{o}$ of \sref{IV.21.10.1}, the equality of the multiplicities in $x'$ of the two members of \sref{IV.21.10.4.1} results from the definition of $n_{x'}$ by means of the homomorphism \sref{IV.21.10.1.1}.
Suppose on the contrary that $x'$ is in case $2\textsuperscript{o}$ of \sref{IV.21.10.1}, and let $x=f(x')$;
since $x\in X^{(1)}\cap\overline{\{y\}}$, one has necessarily $x=y$.
Note now that $f^*(D)=\operatorname{div}(t')$, where $t'$ is the image of $t$ in $\Gamma(X',\sh{O}_{X'})$, and $Y(f^*(D))=f^{-1}(Y(D))$;
the multiplicity of $f^*(D)$ at the point $x'$ is therefore the length $n_{x'}$ of the $\sh{O}_{X',x'}$-module $\sh{O}_{X',x'}/t'_{x'}\sh{O}_{X',x'}$;
as $\sh{O}_{X',x'}/t'_{x'}\sh{O}_{X',x'}=(\sh{O}_{X,y}/t_y\sh{O}_{X,y})\otimes_{\sh{O}_{X,y}}\sh{O}_{X',x'}$, it results from \sref{IV.4.7.1} that one has $n_{x'}=\lambda_{x'}n_y$, hence the multiplicities at the point $x'$ of the two members of \sref{IV.21.10.4.1} are still equal, which completes the proof.
\end{proof}

\begin{env}[21.10.5]
\label{IV.21.10.5}

In particular:
\end{env}

\begin{proposition}[21.10.6]
\label{IV.21.10.6}
Let $X$, $X'$ be two locally Noetherian preschemes, $f:X'\to X$ a flat morphism.
Then $f^*(Z)$ is defined for every $1$-codimensional cycle $Z$ on $X$, $f^*(D)$ is defined for every divisor $D$ on $X$ and one has the relation \sref{IV.21.10.4.1}.

\begin{remark}[21.10.7]

The existence of $f^*(Z)$ for every $1$-codimensional cycle $Z$ on $X$ already results from the hypothesis that $f$ is \emph{flat at the points $x'$ of $X'$ of codimension $\leq 1$ in $X'$} (i.e.\ such that $\dim\sh{O}_{X',x'}\leq 1$);

\begin{proposition}[21.10.8]
\label{IV.21.10.8}
Let $X$, $X'$, $X''$ be three locally Noetherian preschemes, $f:X'\to X$, $g:X''\to X'$ two morphisms;
suppose that $f$ (resp.\ $g$) is flat at every point of codimension $\leq 1$ in $X'$ (resp.\ $X''$).
Then $f\circ g$ is flat at every point of codimension $\leq 1$ in $X''$ and for every $1$-codimensional cycle $Z$ on $X$, one has $(f\circ g)^*(Z)=g^*(f^*(Z))$.
\end{proposition}

\begin{proof}
The first assertion results from \sref{IV.6.1.1} and \sref{IV.2.1.6}.
The second results from the fact that $f$ (resp.\ $g$, resp.\ $f\circ g$) transforms the maximal points of $X'$ (resp.\ $X''$, resp.\ $X''$) into maximal points of $X$ (resp.\ $X'$, resp.\ $X$) and from the fact that, if $x''\in X''^{(1)}$ is such that $x'=g(x'')\in X'^{(1)}$ and $x=f(x')\in X^{(1)}$, one has
\[
  \operatorname{long}(\sh{O}_{X'',x''}/\mathfrak{m}_x\sh{O}_{X'',x''})=\operatorname{long}(\sh{O}_{X'',x''}/\mathfrak{m}_{x'}\sh{O}_{X'',x''})\cdot\operatorname{long}(\sh{O}_{X',x'}/\mathfrak{m}_x\sh{O}_{X',x'}).
\]
Indeed, if one sets $A=\sh{O}_{X,x}$, $A'=\sh{O}_{X',x'}$, $A''=\sh{O}_{X'',x''}$, $\mathfrak{m}=\mathfrak{m}_x$, $\mathfrak{m}'=\mathfrak{m}_{x'}$, the formula
\[
  \operatorname{long}_{A''}(A''/\mathfrak{m} A'')=\operatorname{long}_{A''}(A''/\mathfrak{m}' A'')\cdot\operatorname{long}_{A'}(A'/\mathfrak{m} A')
\]
results from \sref{IV.4.7.1} and the hypothesis of flatness of $A''$ over $A'$.
\end{proof}

\begin{env}[21.10.9]

Let $X$ be a locally Noetherian prescheme, $\Lambda$ a commutative group, written additively, and therefore considered as a $\bb{Z}$-module;

\begin{env}[21.10.10]

\begin{lemma}[21.10.13]
\label{IV.21.10.13}
Let $A$ be a local Noetherian ring of dimension $1$, $t$ a regular element of $A$ belonging to the maximal ideal $\mathfrak{m}$ (which implies that $\mathfrak{m}\notin\operatorname{Ass}(A)$).
\begin{enumerate}[label=\emph{(\roman*)}]
  \item For every $A$-module $M$ of finite type, the kernel $\mathrm{N}_t(M)$ and the cokernel $\mathrm{P}_t(M)$ of the homothety $t_M:M\to M$ of ratio $t$ are of finite length. Set $d_t(M)=\operatorname{long}\mathrm{P}_t(M)-\operatorname{long}\mathrm{N}_t(M)$.
  \item If $0\to M'\to M\to M''\to 0$ is an exact sequence of $A$-modules of finite type, one has $d_t(M)=d_t(M')+d_t(M'')$.
  \item One has $d_t(M)\geq 0$ for every $A$-module $M$ of finite type; for $d_t(M)=0$, it is necessary and sufficient that $M$ be of finite length.
  \item If $K$ is the total ring of fractions of $A$ and if $M\otimes_A K$ is a free $K$-module of rank $n$, one has $d_t(M)=n\cdot d_t(A)=n\cdot\operatorname{long}(A/tA)$.
  \item If $M$ satisfies the hypotheses of~\emph{(iv)} and if moreover $t$ is $M$-regular, one has $\operatorname{long}(M/tM)=n\cdot\operatorname{long}(A/tA)$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) $\Spec(A)$ is formed of the point $\mathfrak{m}$ and the minimal prime ideals $\mathfrak{p}_i$;
since by hypothesis $t\notin\mathfrak{p}_i$ for all $i$ (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, n\textsuperscript{o}~1, cor.~3 of prop.~2), the image of $t$ in each $A_{\mathfrak{p}_i}$ is invertible, and the supports of the $A$-modules $\mathrm{N}_t(M)$ and $\mathrm{P}_t(M)$ of finite type are thus empty or reduced to $\mathfrak{m}$;
one concludes $(\mathbf{0}$, 16.1.10) that these modules are of finite length.

\oldpage[IV-4]{295}

(ii) Since $t$ is regular, one has an exact sequence
\[
  0\to A\xrightarrow{t_A} A\to A/tA\to 0
\]
and since $\operatorname{Tor}_i^A(M,A)=0$ for $i\geq 1$, the Tor exact sequence gives on the one hand the exact sequence
\[
  0\to\operatorname{Tor}_1^A(M,A/tA)\to M\xrightarrow{t_M} M\to M/tM\to 0
\]
and on the other hand, for $i\geq 2$,
\[
  0=\operatorname{Tor}_i^A(M,A)\to\operatorname{Tor}_i^A(M,A/tA)\to\operatorname{Tor}_{i-1}^A(M,A)=0
\]
hence $\mathrm{N}_t(M)=\operatorname{Tor}_1^A(M,A/tA)$ and $\operatorname{Tor}_i^A(M,A/tA)=0$ for $i\geq 2$;
the Tor exact sequence applied to $0\to M'\to M\to M''\to 0$ gives the exact sequence
\begin{align*}
  0\to\operatorname{Tor}_1^A(M',A/tA)&\to\operatorname{Tor}_1^A(M,A/tA)\to\operatorname{Tor}_1^A(M'',A/tA)\to\\
  &M'\otimes_A(A/tA)\to M\otimes_A(A/tA)\to M''\otimes_A(A/tA)\to 0
\end{align*}
and this sequence can be written, by what precedes
\[
  0\to\mathrm{N}_t(M')\to\mathrm{N}_t(M)\to\mathrm{N}_t(M'')\to\mathrm{P}_t(M')\to\mathrm{P}_t(M)\to\mathrm{P}_t(M'')\to 0
\]
which proves~(ii).

(iii) To prove~(iii), note that there is a composition series $(M_h)$ of $M$ whose quotients $M_h/M_{h+1}$ are isomorphic to $A/\mathfrak{m}$ or to one of the $A/\mathfrak{p}_i$ (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, n\textsuperscript{o}~4, th.~1), and for $M$ to be of finite length, it suffices that all these quotients be isomorphic to $A/\mathfrak{m}$.
By~(ii), it reduces to proving that $d_t(A/\mathfrak{m})=0$ and $d_t(A/\mathfrak{p}_i)>0$.
Now, the image of $t$ in $A/\mathfrak{m}$ being $0$, one has $\mathrm{N}_t(A/\mathfrak{m})=A/\mathfrak{m}$ and $\mathrm{P}_t(A/\mathfrak{m})=A/\mathfrak{m}$, hence the first assertion;
on the other hand, the image of $t$ in $A/\mathfrak{p}_i$ is regular, hence $\mathrm{N}_t(A/\mathfrak{p}_i)=0$ and $\mathrm{P}_t(A/\mathfrak{p}_i)=A/(tA+\mathfrak{p}_i)$ which is not reduced to $0$, hence the second assertion.

(iv) There is a base of $M\otimes_A K$ of the form $(x_j/s)_{1\leq j\leq n}$, where $s$ is a regular element of $A$ and $x_j\in M$.
Consider the homomorphism $u:A^n\to M$ that transforms $e_j$ (the canonical base element of $A^n$) to $x_j$ and show that $\Ker(u)$ and $\operatorname{Coker}(u)$ are of finite length;
indeed, for each $i$, the image of $s$ in $A_{\mathfrak{p}_i}$ is invertible, and since the images $x_j/s$ in $M_{\mathfrak{p}_i}$ form a base of this $A_{\mathfrak{p}_i}$-module, $u_{\mathfrak{p}_i}:A^n_{\mathfrak{p}_i}\to M_{\mathfrak{p}_i}$ is bijective.
Since~(i), $\Ker(u)$ and $\operatorname{Coker}(u)$ have supports contained in $\{\mathfrak{m}\}$, hence are of finite length.
It then results from~(ii) and~(iii) that $d_t(M)=d_t(A^n)=n\cdot d_t(A)=n\cdot\operatorname{long}(A/tA)$ since $t$ is regular.

(v) It is clear that~(v) follows immediately from~(iv), since then $\mathrm{N}_t(M)=0$.
\end{proof}

This lemma allows to generalise \sref{IV.21.10.4}:

\begin{proposition}[21.10.13]
\label{IV.21.10.13.prop}
Suppose that $f$ satisfies the conditions of \sref{IV.21.10.10}.
Then, for every divisor $D$ on $X$ such that $\operatorname{Supp}(D)\subset T$ and such that $f^*(D)$ is defined, one has
\[
\label{IV.21.10.13.1}
  \operatorname{cyc}(f^*(D))=f^*(\operatorname{cyc}(D)).
  \tag{21.10.13.1}
\]
\end{proposition}

\begin{proof}
Reasoning as in \sref{IV.21.10.4}, everything reduces to seeing (with the same notations) that if $x'$ is in case $4\textsuperscript{o}$ of \sref{IV.21.10.10} and if $n_y$ is the length of $\sh{O}_{X,y}/t_y\sh{O}_{X,y}$, then the length $n_{x'}$ of $\sh{O}_{X',x'}/t'_{x'}\sh{O}_{X',x'}$ is equal to $\mu_{x'}n_y$, $\mu_{x'}$ being the rational number defined in \sref{IV.21.10.10}.
As $\sh{O}_{X',x'}/t'_{x'}\sh{O}_{X',x'}$ is of finite length, it has the same length as its $\mathfrak{m}_{x'}$-preadic completion $A'/t_{x'}' A'$, which one can also write $A'/t_y A'$;
moreover, as $t'_{x'}$ is regular by hypothesis in $\sh{O}_{X',x'}$, it is also regular in $A'$ by flatness $(\mathbf{0_I}$, 6.3.4), and when $A'$ is considered as an $A$-module, one can also say that $t_y$ is $A'$-regular.
As $A$ is of dimension $1$ and $A'$ is an $A$-module of finite type such that $A'\otimes_A K$ is a free $K$-module of rank $r_{x'}$, one can apply \sref{IV.21.10.13}[, (v)] to $M=A'$ and to $t_y$, the length of $A'/t_y A'$ as an $A$-module is therefore $r_{x'}n_y$.
As $k(x')$ is a $k(x)$-vector space of rank $q_{x'}$, the length of $A'/t_y A'$ as an $A'$-module is therefore $r_{x'}n_y/q_{x'}=\mu_{x'}n_y$, which completes the proof.
\end{proof}

\begin{env}[21.10.14]
\label{IV.21.10.14}
Let now $X$, $X'$ be two locally Noetherian preschemes, $f:X'\to X$ a morphism having the two following properties:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $f$ is \emph{finite};
  \item the image by $f$ of every maximal point of $X'$ is a maximal point of $X$.
\end{enumerate}
For every $x\in X^{(1)}$, the points $x'\in f^{-1}(x)$ then all belong to $X'^{(1)}$, since, as it results from hypothesis~b) and from the inequality $(\mathbf{0}$, 16.3.9.1), the fibre $f^{-1}(x)$ is discrete.
Let then
\[
  Z'=\sum_{x'\in X'^{(1)}}n_{x'}\cdot\overline{\{x'\}}
\]
be a $1$-codimensional cycle on $X'$.
For every $x\in X^{(1)}$, let us define an integer $n_x$ by the formula
\[
  n_x=\sum_{x'\in f^{-1}(x)}n_{x'}\cdot[k(x'):k(x)]
\]
which makes sense, the points of $f^{-1}(x)$ being finite in number and $k(x')$ being a field of finite degree over $k(x)$ $(\sref[I]{I.6.4.4})$.
Moreover, the set of $x\in X^{(1)}$ such that $n_x\neq 0$ is \emph{locally finite} in $X$, for it is contained in the image by $f$ of the set of $x'\in X'^{(1)}$ such that $n_{x'}\neq 0$, and the conclusion results from the fact that the morphism $f$ is quasi-compact.
One can therefore define a $1$-codimensional cycle on $X$ by setting
\[
\label{IV.21.10.14.1}
  f_*(Z')=\sum_{x\in X^{(1)}}n_x\cdot\overline{\{x\}}
  \tag{21.10.14.1}
\]
and one says that $f_*(Z')$ is the \emph{direct image} of $Z'$ by $f$.
It is clear that the application $f_*:\mathfrak{Z}^1(X')\to\mathfrak{Z}^1(X)$ is thus defined as a homomorphism of ordered groups.
Moreover, if $U$ is an open of $X$, $f_U:f^{-1}(U)\to U$ the restriction of $f$, it results immediately from the definitions that one has
\[
\label{IV.21.10.14.2}
  (f_U)_*(Z'|f^{-1}(U))=f_*(Z')|U
  \tag{21.10.14.2}
\]
hence, denoting by $\psi$ the continuous map underlying the morphism $f$, the maps $\Gamma(f^{-1}(U),\mathscr{Z}_{X'}^1)\to\Gamma(U,\mathscr{Z}_X^1)$ that one has just defined define a homomorphism of sheaves of ordered commutative groups on $X$
\[
  \psi_*(\mathscr{Z}_{X'}^1)\to\mathscr{Z}_X^1.
\]
\end{env}

\begin{proposition}[21.10.16]
\label{IV.21.10.16}
Let $X$, $X'$, $X_1$ be three locally Noetherian preschemes, $f:X'\to X$ a morphism satisfying conditions~a) and~b) of \sref{IV.21.10.14}, $g:X_1\to X$ a flat morphism.
Set $X'_1=X'\times_X X_1$ (so that $X'_1$ is locally Noetherian) and denote by $f_1:X'_1\to X_1$ and $g_1:X'_1\to X'$ the canonical projections.
Then $f_1$ satisfies conditions~a) and~b) of \sref{IV.21.10.14}, and for every $1$-codimensional cycle $Z'$ on $X'$, one has
\[
\label{IV.21.10.16.1}
  g^*(f_*(Z'))=(f_1)_*(g_1^*(Z')).
  \tag{21.10.16.1}
\]
\end{proposition}

\begin{proof}
It is clear that $f_1$ is finite, and it satisfies condition~b) of \sref{IV.21.10.14} by virtue of \sref{IV.2.3.7}.
To prove \sref{IV.21.10.16.1}, one is immediately reduced to the case where $X$, $X'$ and $X_1$ are spectra of local Noetherian rings of dimension $1$, of closed points $x$, $x'$ and $x_1$ respectively, $A$, $A'$, $A_1$ being the local rings at these points, and $A_1$ a flat $A$-module.
Denoting by $x$, $x'$ and $x_1$ the closed points of $X$, $X'$ and $X_1$ respectively, it is a matter of seeing that one has
\[
\label{IV.21.10.16.2}
  \sum_{x'_1}\lambda_{x'_1}[k(x'_1):k(x_1)]=\lambda_{x_1}[k(x'):k(x)]
  \tag{21.10.16.2}
\]
where $x'_1$ ranges over the set of closed points of $X'_1$ (i.e.\ the set of points above both $x'$ and $x_1$) and where $\lambda_{x'_1}=\operatorname{long}(\sh{O}_{X'_1,x'_1}/\mathfrak{m}_{x'}\sh{O}_{X'_1,x'_1})$ and $\lambda_{x_1}=\operatorname{long}(A_1/\mathfrak{m}_x A_1)$.
As one has
\[
  [k(x'_1):k(x')]\cdot[k(x'):k(x)]=[k(x'_1):k(x)]=[k(x'_1):k(x_1)]\cdot[k(x_1):k(x)]
\]
the first member of \sref{IV.21.10.16.2} can also be written
\[
  \frac{[k(x'):k(x)]}{[k(x_1):k(x)]}\operatorname{long}_{A_1}(A'_1/\mathfrak{m}_{x'}A'_1)
\]
where one has set $A'_1=A'\otimes_A A_1$.
Since $A'_1/\mathfrak{m}_{x'}A'_1=(A'/\mathfrak{m}_{x'}A')\otimes_A A_1$, and as $A_1$ is a flat $A$-module, one has by \sref{IV.4.7.1}
\[
  \operatorname{long}_{A_1}(A'_1/\mathfrak{m}_{x'}A'_1)=\operatorname{long}_A(A'/\mathfrak{m}_{x'}A')\cdot\operatorname{long}_{A_1}(A_1/\mathfrak{m}_x A_1)=[k(x'):k(x)]\cdot\lambda_{x_1}
\]
which gives the result since $[k(x'):k(x)]/[k(x_1):k(x)]\cdot[k(x'):k(x)]\cdot\lambda_{x_1}=\lambda_{x_1}\cdot[k(x'):k(x)]$.
\end{proof}

\begin{proposition}[21.10.17]
\label{IV.21.10.17}
Let $X$, $X'$ be two locally Noetherian preschemes, $f:X'\to X$ a finite locally free morphism.
Then $f$ satisfies condition~b) of \sref{IV.21.10.14}, and for every divisor $D'$ on $X'$, one has
\[
\label{IV.21.10.17.1}
  f_*(\operatorname{cyc}(D'))=\operatorname{cyc}(f_*(D')).
  \tag{21.10.17.1}
\]
\end{proposition}

\begin{proof}
As $f$ is flat and finite, condition~b) of \sref{IV.21.10.14} and the relation $f(X'^{(1)})\subset X^{(1)}$ result from \sref{IV.6.1.1}.
The definition \sref{IV.21.10.14.1} shows that one can reduce to the case where $X=\Spec(A)=\Spec(\sh{O}_{X,x})$ for an $x\in X^{(1)}$;
then $X'=\Spec(A')$, where $A'$ is an $A$-algebra that is a free $A$-module of finite rank;
moreover, one can suppose that $D'=\operatorname{div}(t')$, where $t'$ is a regular element of $A'$;
one has then $f_*(D')=\operatorname{div}(t)$, where $t=\mathrm{N}_{A'/A}(t')$ is a regular element of $A$ \sref{IV.21.5.2}.
One can restrict to the case where $t'$ is not invertible in $A'$, the ring $A/tA$ is then composed of local Artinian rings $A'_i$ $(1\leq i\leq r)$, the residue field of $A'_i$ being $k(x'_i)$, where $x'_i$ $(1\leq i\leq r)$ are the points of $X'$ above $x$.
If $t'_i$ $(1\leq i\leq r)$ is the image of $t'$ in $A'_i$, $A'/t'A'$ is a direct sum of the $A'_i/t'_iA'_i$;
as the product $\operatorname{long}_{A'_i}(A'_i/t'_iA'_i)\cdot[k(x'_i):k(x)]$ is equal to $\operatorname{long}_A(A'_i/t'_iA'_i)$, one sees that the formula to prove reduces to
\[
\label{IV.21.10.17.2}
  \operatorname{long}_A(A'/t'A')=\operatorname{long}_A(A/tA).
  \tag{21.10.17.2}
\]

\oldpage[IV-4]{298}

This relation follows from the following more general lemma:
\end{proof}

\begin{lemma}[21.10.17.3]
\label{IV.21.10.17.3}
Let $A$ be a local Noetherian ring of dimension $1$, $M$ a free $A$-module of finite rank, $u$ an injective endomorphism of $M$. Then one has
\[
\label{IV.21.10.17.4}
  \operatorname{long}_A(\operatorname{Coker} u)=\operatorname{long}_A(A/(\det u)A).
  \tag{21.10.17.4}
\]
\end{lemma}

\begin{proof}
Let us distinguish several cases.

\emph{I) $A$ is a discrete valuation ring.}
Let $\pi$ be a uniformiser of $A$, and note that $\operatorname{long}_A(A/\pi^k A)=k$;
if $n$ is the rank of $M$ and if $\pi^{m_i}$ $(1\leq i\leq n)$ are the invariant factors of $u$, $\operatorname{Coker}(u)$ is a direct sum of the $A$-modules $A/\pi^{m_i}A$, hence has length $m=\sum_{i=1}^n m_i$, and $\det(u)$ is a product of an invertible element and of $\pi^m$, whence the conclusion in this case (Bourbaki, \emph{Alg.}, chap.~VII, \textsection4, n\textsuperscript{o}~5, cor.~1 of prop.~4).

\emph{II) $A$ is a complete integral ring (of dimension $1$).}
One knows then $(\mathbf{0}$, 19.8.8, (ii)) that there is a sub-ring $B$ of $A$, which is a discrete valuation ring, such that $B\to A$ is a local homomorphism making $A$ a $B$-module of finite type;
as this $B$-module is evidently torsion-free, it is \emph{free} (Bourbaki, \emph{Alg.~comm.}, chap.~VI, \textsection3, n\textsuperscript{o}~6, lemme~1).
Denote by $M'$ the set $M$ endowed with its structure of (free) $B$-module, and by $u'$ the endomorphism of $M'$ (the endomorphism $u$ considered as a $B$-endomorphism).
It results from~I) that one has
\[
\label{IV.21.10.17.5}
  \operatorname{long}_B(\operatorname{Coker} u')=\operatorname{long}_B(B/(\det u')B).
  \tag{21.10.17.5}
\]
But one has (Bourbaki, \emph{Alg.}, chap.~VIII, \textsection12, n\textsuperscript{o}~2, prop.~7)
\[
  \det u'=\mathrm{N}_{A/B}(\det u),
\]
hence, applying \sref{IV.21.10.17.4} to the homothety $x\mapsto(\det u)x$ of the $B$-module free $A$, one obtains
\[
  \operatorname{long}_B(B/(\det u')B)=\operatorname{long}_B(A/(\det u)A),
\]
whence, substituting in \sref{IV.21.10.17.5} and dividing by $[k(A):k(B)]$, the length of the $B$-module $k(A)$, one obtains \sref{IV.21.10.17.4} in the case considered.

\emph{III) $A$ is a complete ring.}
Let us note that one can moreover suppose that $\mathfrak{m}\notin\operatorname{Ass}(A)$;
indeed, since $\det(u)$ is a regular element of $A$, if one had $\mathfrak{m}\in\operatorname{Ass}(A)$, one would deduce $\det(u)\notin\mathfrak{m}$, hence $\det(u)$ would be invertible, $u$ an automorphism of $M$, and the formula \sref{IV.21.10.17.4} would become trivial, both members being zero.
In what follows, for an endomorphism $v$ of a module $N$ over a ring $R$, such that $\Ker v$ and $\operatorname{Coker} v$ are of \emph{finite length}, one sets
\[
  \chi(N,v)=\operatorname{long}_R(\Ker(v))-\operatorname{long}_R(\operatorname{Coker}(v)).
\]
One will note that the hypothesis on $v$ amounts to saying that the complex
\[
  K^0:0\to N\xrightarrow{v} N\to 0
\]
has its cohomology modules of finite length and that $\chi(N,v)=\chi(\mathrm{H}^0(K^0))$ $(\mathbf{0_{III}}$, 11.10).
One deduces that if $N'$, $N$, $N''$ are three $R$-modules, if one has a commutative diagram
\[
  \xymatrix{
    0\ar[r] & N'\ar[r]^{r}\ar[d]_{v'} & N\ar[r]^{s}\ar[d]_{v} & N''\ar[r]\ar[d]_{v''} & 0\\
    0\ar[r] & N'\ar[r]_{r} & N\ar[r]_{s} & N''\ar[r] & 0
  }
\]
whose rows are exact, and if two of the three numbers $\chi(N,v)$, $\chi(N',v')$ and $\chi(N'',v'')$ are defined, then so is the third and one has
\[
\label{IV.21.10.17.6}
  \chi(N,v)=\chi(N',v')+\chi(N'',v'').
  \tag{21.10.17.6}
\]
This results indeed immediately from the exact cohomology sequence.
Finally, if $N$ is an $R$-module of \emph{finite length}, one has $\chi(N,v)=0$.

With these notations, one has the following lemma:
\end{proof}

\begin{lemma}[21.10.17.7]
\label{IV.21.10.17.7}
Let $A$ be a local Noetherian ring of dimension $1$ whose maximal ideal $\mathfrak{m}$ is such that $\mathfrak{m}\notin\operatorname{Ass}(A)$;
let $\mathfrak{p}_i$ $(1\leq i\leq n)$ be the minimal prime ideals of $A$.
Let $M$ be a free $A$-module of finite type, $u$ an endomorphism of $M$ such that $\chi(M,u)$ is defined;
for each $i$, set $M_i=M\otimes_A(A/\mathfrak{p}_i)$ and let $u_i$ be the endomorphism $u\otimes 1_{A/\mathfrak{p}_i}$ of $M_i$;
then if $\chi(M_i,u_i)$ is defined for each $i$, one has
\[
\label{IV.21.10.17.8}
  \chi(M,u)=\sum_{i=1}^n\operatorname{long}(A_{\mathfrak{p}_i})\cdot\chi(M_i,u_i).
  \tag{21.10.17.8}
\]
\end{lemma}

\begin{proof}
Since $\mathfrak{m}\notin\operatorname{Ass}(A)$, one has a unique reduced primary decomposition $(0)=\bigcap_{1\leq i\leq n}\mathfrak{q}_i$, where the ideal $\mathfrak{q}_i$ is $\mathfrak{p}_i$-primary for $1\leq i\leq n$.
If one sets $M'_i=M\otimes_A(A/\mathfrak{q}_i)$, one has therefore an exact sequence
\[
  0\to M\to\bigoplus_i M'_i\to M''\to 0
\]
of $A$-modules, where $M''$ is of finite length:
indeed, localising at each of the $\mathfrak{p}_i$, one obtains $M''_{\mathfrak{p}_i}=0$ since $(\mathfrak{q}_i)_{\mathfrak{p}_i}=0$ and $(\mathfrak{q}_j)_{\mathfrak{p}_i}=A_{\mathfrak{p}_i}$ for $j\neq i$ (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection2, n\textsuperscript{o}~4, prop.~6), hence $(M'_i)_{\mathfrak{p}_i}=M_{\mathfrak{p}_i}$ and $(M'_j)_{\mathfrak{p}_i}=0$;
the support of $M''$ is therefore reduced to $\mathfrak{m}$, so $M''$ is of finite length $(\mathbf{0}$, 16.1.10).
Setting $u'_i=u\otimes 1_{A/\mathfrak{q}_i}$, and if $u''$ is the endomorphism of $M''$ deduced from $u$ by restriction to quotients, one deduces from \sref{IV.21.10.17.6} that $\chi(M,u)+\chi(M'',u'')=\sum_i\chi(M'_i,u'_i)$, and since $M''$ is of finite length, $\chi(M'',u'')=0$.
To prove \sref{IV.21.10.17.8}, it therefore suffices to show that, for $n=1$, i.e.\ $A$ integral, if one designates by $\mathfrak{p}$ the unique minimal prime ideal of $A$, and if $A_0=A/\mathfrak{p}$, $M_0=M\otimes_A A_0$, $u_0=u\otimes 1_{A_0}$, one has, assuming $\chi(M_0,u_0)$ is defined,
\[
\label{IV.21.10.17.9}
  \chi(M,u)=\operatorname{long}(A_\mathfrak{p})\cdot\chi(M_0,u_0).
  \tag{21.10.17.9}
\]

\oldpage[IV-4]{300}

Let $\mathfrak{n}_j$ $(0\leq j\leq r)$ be the ``symbolic $j$-th power'' of $\mathfrak{p}$, the preimage in $A$ of the ideal $(\mathfrak{p} A_\mathfrak{p})^j$ of $A_\mathfrak{p}$ $(1\leq j\leq r)$, with $\mathfrak{n}_0=A$ and $\mathfrak{n}_r=0$; set
\[
  M_j=\mathfrak{n}_j M/\mathfrak{n}_{j+1}M\quad(0\leq j\leq r-1),
\]
and denote by $v_j$ the endomorphism of $M_j$ deduced from $u$ by restriction to $\mathfrak{n}_j M$ and passage to quotients.
We shall first show that each of the numbers $\chi(M_j,v_j)$ is defined and that one has
\[
\label{IV.21.10.17.10}
  \chi(M,u)=\sum_j\chi(M_j,v_j).
  \tag{21.10.17.10}
\]
The first assertion will imply the second, by applying \sref{IV.21.10.17.6} to each of the exact sequences
\[
  0\to\mathfrak{n}_j M/\mathfrak{n}_{j+1}M\to M/\mathfrak{n}_{j+1}M\to M/\mathfrak{n}_j M\to 0.
\]

To prove the first assertion, note that if $m$ is the rank of the free $A$-module $M$, $M_j$ is isomorphic to $(\mathfrak{n}_j/\mathfrak{n}_{j+1})^m$, or further (since $\mathfrak{p}$ annihilates each of the quotients $\mathfrak{n}_j/\mathfrak{n}_{j+1}$), $M_j$ is an $A_0$-module isomorphic to $M_0\otimes_{A_0}(\mathfrak{n}_j/\mathfrak{n}_{j+1})^m$.
Denote by $l_j$ the rank of the $A_0$-module $\mathfrak{n}_j/\mathfrak{n}_{j+1}$;
as the field of fractions $K_0$ of $A_0$ is the residue field of $A_\mathfrak{p}$, there is a system of generators of the $A_0$-module $M_j$ which contains a base of $M_j\otimes_{A_0}K_0$; hence there is an $A_0$-homomorphism
\[
  w_j:M_0^{l_j}\to M_j
\]
whose localisation at the ideal $(0)$ of $A_0$ is an isomorphism, so that the supports of $\Ker(w_j)$ and $\operatorname{Coker}(w_j)$ are reduced to the maximal ideal $\mathfrak{m}/\mathfrak{p}$ of $A_0$; $\Ker(w_j)$ and $\operatorname{Coker}(w_j)$ are therefore $A_0$-modules of finite length $(\mathbf{0}$, 16.1.10).
As by hypothesis $\chi(M_0,u_0)$ is defined, so is $\chi(M_0^{l_j},u_0^{l_j})=l_j\chi(M_0,u_0)$ and by virtue of \sref{IV.21.10.17.6} and the fact that $\Ker(w_j)$ and $\operatorname{Coker}(w_j)$ are of finite length, one sees that $\chi(M_j,v_j)$ is defined and equal to $l_j\chi(M_0,u_0)$;
the relation \sref{IV.21.10.17.10} then gives
\[
  \chi(M,u)=\left(\sum_j l_j\right)\chi(M_0,u_0),
\]
and by virtue of a preceding remark, $\sum_j l_j$ is none other than the length of $A_\mathfrak{p}$, which completes the proof of the lemma.
\end{proof}

To apply this lemma when $A$ is a complete ring and $\mathfrak{m}\notin\operatorname{Ass}(A)$, observe that if $u$ is injective, it is the same for each of the $u_i$ (with the notations of the lemma):
indeed, since $\det(u)$ is a regular element of $A$, it does not belong to any of the $\mathfrak{p}_i$, and its image $\det u_i$ in $A/\mathfrak{p}_i$ is by consequence a non-zero element of this integral ring, which proves that $u_i$ is injective.
As $\operatorname{Coker}(u_i)$ is an image of $\operatorname{Coker}(u)$, it is also of finite length and $\chi(M_i,u_i)$ is therefore defined for each $i$;
one has therefore by the formula \sref{IV.21.10.17.8}.
On the other hand, since $\det(u)$ is a regular element of $A$, it is not contained in any $\mathfrak{p}_i$;
the ideal $(\det u)A$ is therefore $\mathfrak{m}$-primary and the quotient $A/(\det u)A$ is of finite length.
Applying the same reasoning as above to the injective homothety $t:\xi\to(\det u)\xi$ of $A$ and to its images $t_i$ in the $A/\mathfrak{p}_i$, one obtains
\[
  \chi(A,t)=\sum_i\operatorname{long}(A_{\mathfrak{p}_i})\cdot\chi(A/\mathfrak{p}_i,t_i).
\]
But as the $A/\mathfrak{p}_i$ are integral rings and complete, and as each $A/\mathfrak{p}_i$ has also $\operatorname{long}(A_{\mathfrak{p}_i})$, one obtains by applying the result of each $A/\mathfrak{p}_i$ the relation \sref{IV.21.10.17.4} for $M$ and $u$ in this case.

\oldpage[IV-4]{301}

\emph{IV) General case.}
Set $A'=\hat{A}$, $M'=M\otimes_A A'$, $u'=u\otimes 1_{A'}$;
one has $\det(u')=\det(u)$, and by flatness, $\operatorname{Coker}(u')=(\operatorname{Coker}(u))\otimes_A A'$ and $A'/(\det u)A'=(A/(\det u)A)\otimes_A A'$;
as the formula \sref{IV.21.10.17.4} is true for $A'$ and $u'$ by virtue of~III), it is also true for $A$ and $u$ by virtue of \sref{IV.4.7.1}.

This completes the proof of \sref{IV.21.10.17}.

\begin{proposition}[21.10.18]
\label{IV.21.10.18}
Let $X$, $X'$ be two locally Noetherian preschemes, $f:X'\to X$ a finite morphism, transforming every maximal point of $X'$ into a maximal point of $X$, and satisfying for every $x'\in X'$ one of the conditions~(ii), (iii), (iv) of \sref{IV.21.10.10}.
Suppose moreover that there exists an integer $n$ such that, for every maximal point $x$ of $X$, $(f_*(\sh{O}_{X'}))_x$ is a free $\sh{O}_{X,x}$-module of rank $n$.
Then, for every $1$-codimensional cycle $Z$ on $X$, one has
\[
\label{IV.21.10.18.1}
  f_*(f^*(Z))=n\cdot Z
  \tag{21.10.18.1}
\]
(``projection formula'').
\end{proposition}

\begin{proof}
By virtue of the definitions, one is immediately reduced to the case where $X$ is the spectrum of a local Noetherian ring $A$ of dimension $1$, of closed point $x$, and where $Z=\overline{\{x\}}$;
set $X'=\Spec(A')$ where $A'$ is a finite $A$-algebra.
Let us make the base change $h:Y\to X$, where $Y=\Spec(B)$, $B=\hat{A}$, and set $Y'=X'\times_X Y=\Spec(B\otimes_A A')$ and $g=f_{(Y)}:Y'\to Y$;
the morphism $g$ satisfies the same conditions as $f$;
if one proves the formula \sref{IV.21.10.18.1} for $g$ and $\overline{\{y\}}$, the same formula will be valid for $f$ and $\overline{\{x\}}$, by virtue of \sref{IV.21.10.16.1}.
One can therefore suppose that $A$ is \emph{complete};
then so is $A'$ which is therefore a direct product of complete local rings;
one can thus restrict to the case where $A'$ is a local ring, and it remains to verify \sref{IV.21.10.18.1} in each of the cases~(ii), (iii), (iv) for the closed point $x'$ of $X'$.

\oldpage[IV-4]{302}

In case~(ii), $A'$ being a flat $A$-module of finite type, is a free $A$-module $(\mathbf{0_{III}}$, 10.1.3), of rank $n$ by virtue of the hypothesis.
Now, by definition \sref{IV.21.10.1} and \sref{IV.21.10.3}, $f^*(Z)=\lambda_{x'}\cdot\overline{\{x'\}}$, where $\lambda_{x'}$ is the length of the $A'$-module $A'/\mathfrak{m}_x A'$, then $f_*(f^*(Z))=\lambda_{x'}\cdot[k(x'):k(x)]\cdot\overline{\{x\}}$;
but $\lambda_{x'}\cdot[k(x'):k(x)]$ is the length of $A'/\mathfrak{m}_x A'$ as an $A$-module, or equivalently its rank as a $k(x)$-vector space, hence is equal to $n$.

In case~(iii), $A$ is a discrete valuation ring, hence regular, and the hypothesis $\mathfrak{m}_{x'}\notin\operatorname{Ass}(\sh{O}_{X',x'})$ implies that $A'$ is a Cohen--Macaulay ring $(\mathbf{0}$, 16.4.6);
as $\dim(A')=1$ and $A'/\mathfrak{m}_x A'$ is an Artinian ring, it results from \sref{IV.6.1.5} that $A'$ is a flat $A$-module, and one is reduced to case~(ii).

In case~(iv), if $K$ is the total ring of fractions of $A$, $K'=A'\otimes_A K$ is by hypothesis a free $K$-module of rank $n$ and by definition \sref{IV.21.10.10}, one has $f^*(Z)=(n/[k(x'):k(x)])\cdot\overline{\{x'\}}$, whence $f_*(f^*(Z))=n\cdot\overline{\{x\}}$.
\end{proof}

One will note that the formula \sref{IV.21.10.18.1} is applicable in particular when the morphism $f$ is \emph{finite} and \emph{flat} and such that for every maximal point $x$ of $X$, $(f_*(\sh{O}_{X'}))_x$ is a free $\sh{O}_{X,x}$-module of rank $n$.

\begin{corollary}[21.10.19]
\label{IV.21.10.19}
Under the hypotheses of \sref{IV.21.10.18}, let $D$ be a divisor on $X$ such that $f^*(D)$ is defined \sref{IV.21.4.5};
then one has
\[
\label{IV.21.10.19.1}
  f_*(\operatorname{cyc}(f^*(D)))=n\cdot\operatorname{cyc}(D).
  \tag{21.10.19.1}
\]
\end{corollary}

\begin{proof}
This results from \sref{IV.21.10.18} and \sref{IV.21.10.13.prop}.
\end{proof}
