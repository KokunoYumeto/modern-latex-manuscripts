\section{Associated prime cycles and primary decompositions}
\label{section:IV.3}

\oldpage[IV-2]{36}

In this paragraph, we mainly give the translation of results on modules presented in Bourbaki, \emph{Alg. comm.}, chap.~IV, which we follow very closely.
The notions that follow seem to be of interest only in the case of \emph{locally Noetherian} preschemes.

\subsection{Prime cycles associated to a Module}
\label{subsection:IV.3.1}

\begin{definition}[3.1.1]
\label{IV.3.1.1}
Let $X$ be a prescheme, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent.
We say that a point $x\in X$ is \emph{associated} to $\sh{F}$ if the maximal ideal $\mathfrak{m}_x$ of $\sh{O}_x$ is associated to the $\sh{O}_x$-Module $\sh{F}_x$ (in other words, if $\mathfrak{m}_x$ is the annihilator of an element of $\sh{F}_x$).
We denote by $\operatorname{Ass}(\sh{F})$ the set of $x\in X$ associated to $\sh{F}$.
We say that an irreducible closed subset $Z$ of $X$ is a \emph{prime cycle associated} to $\sh{F}$ if its generic point is associated to $\sh{F}$.
When $\sh{F}=\sh{O}_X$, we also say that the points (resp.\ \emph{prime cycles}) associated to $\sh{F}$ are associated to the \emph{prescheme} $X$.
\end{definition}

We say that a prime cycle associated to $\sh{F}$ (resp.\ $X$) is \emph{immersed} if it is contained in another prime cycle associated to $\sh{F}$ (resp.\ $X$) (in other words, if it is not \emph{maximal} in the set of these cycles).

If $X$ is locally Noetherian, the \emph{irreducible components} of $X$ are none other than the \emph{maximal} (or \emph{non-immersed}) prime cycles associated to $X$.

It is clear that if $x\in\operatorname{Ass}(\sh{F})$, then $\sh{F}_x\neq 0$, in other words
\[
\label{IV.3.1.1.1}
\operatorname{Ass}(\sh{F})\subset\operatorname{Supp}(\sh{F}).
\tag{3.1.1.1}
\]

If $x\in X$ is associated to $\sh{F}$, it is evidently associated to $\sh{F}|U$ for every open neighbourhood $U$ of $x$, and conversely, if it is associated to $\sh{F}|U$ for one of these neighbourhoods, it is associated to $\sh{F}$.

We note finally that for a quasi-coherent $\sh{O}_X$-Module $\sh{F}$, the existence of immersed associated prime cycles is a \emph{local} question, for if $y$ and $z$ are two points of $\operatorname{Ass}(\sh{F})$ such that $y\in\overline{\{z\}}$, every neighbourhood of $y$ contains $z$.

\begin{proposition}[3.1.2]
\label{IV.3.1.2}
Let $A$ be a Noetherian ring, $M$ an $A$-module, $X=\operatorname{Spec}(A)$, $\sh{F}=\widetilde{M}$; for a point $x\in X$ to be associated to $\sh{F}$, it is necessary and sufficient that the prime ideal $\mathfrak{j}_x$ of $A$ be associated to the module $M$ (in other words, be the annihilator of an element $f\in M$).
\end{proposition}

\begin{proof}
This follows from the definition \sref{IV.3.1.1} and from Bourbaki, \emph{loc.\ cit.}, \textsection~1, no~2, cor.\ of prop.~5, applied to $S=A-\mathfrak{j}_x$.
\end{proof}

\begin{proposition}[3.1.3]
\label{IV.3.1.3}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent, $x$ a point of $X$, $Z$ the closed reduced sub-prescheme of $X$ having $\overline{\{x\}}$ as underlying space \sref[I]{I.5.2.1}.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $x\in\operatorname{Ass}(\sh{F})$.
\oldpage[IV-2]{37}
  \item There exist an open neighbourhood $U$ of $x$ and a section $f\in\Gamma(U,\sh{F})$ such that $U\cap Z$ is equal to $\operatorname{Supp}(\sh{O}_U\cdot f)$.
  \item[\rm b$'$)] There exist an open neighbourhood $U$ of $x$ and a section $f\in\Gamma(U,\sh{F})$ such that $U\cap Z$ is an irreducible component of $\operatorname{Supp}(\sh{O}_U\cdot f)$.
  \item There exist an open neighbourhood $U$ of $x$ and a sub-Module of $\sh{F}|U$ isomorphic to $\sh{O}_Z|U$ ($\sh{O}_Z$ being identified with a quotient of $\sh{O}_X$).
  \item[\rm c$'$)] There exist an open neighbourhood $U$ of $x$ and a coherent sub-Module $\sh{G}$ of $\sh{F}|U$ such that $U\cap Z$ is an irreducible component of $\operatorname{Supp}(\sh{G})$.
\end{enumerate}
\end{proposition}

\begin{proof}
It is clear that \emph{c)} implies \emph{b)}, for it suffices to take for $f$ the element of $\Gamma(U,\sh{F})$ which corresponds to the unit section of $\sh{O}_Z|U$.
As $U\cap Z$ is irreducible \sref[0]{0.2.1.6}, \emph{b)} implies \emph{b$'$)}, and \emph{b$'$)} implies \emph{c$'$)} since $\sh{O}_X$ is coherent \sref[0]{0.5.3.4}.
To see that \emph{c$'$)} implies \emph{a)}, we can reduce to the case where $U=X=\operatorname{Spec}(A)$ is affine, $A$ being Noetherian, and where $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module, and $\sh{G}=\widetilde{N}$, where $N$ is a sub-module of $M$, of finite type.
We then know that the minimal elements of $\operatorname{Supp}(\sh{G})$ are the maximal points of $V(\operatorname{Ann}(N))$ \sref[0]{0.1.7.4}, and these are also the minimal elements of $\operatorname{Ass}(N)$ (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~1, no~3, cor.~1 of prop.~7);
as $\operatorname{Ass}(N)\subset\operatorname{Ass}(M)=\operatorname{Ass}(\sh{F})$, we see that \emph{c$'$)} implies \emph{a)}.
Finally, \emph{a)} implies \emph{c)}, by taking $X$ affine again, $\sh{F}=\widetilde{M}$, and $Z$ defined by the ideal $fA$ (with the notations of \sref{IV.3.1.2}).
\end{proof}

\begin{corollary}[3.1.4]
\label{IV.3.1.4}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent.
Then the maximal prime cycles associated to $\sh{F}$ are the irreducible components of $\operatorname{Supp}(\sh{F})$, and the generic points of these components are the points $x\in X$ such that $\sh{F}_x$ is an $\sh{O}_x$-Module of finite length and $\neq 0$.
\end{corollary}

\begin{proof}
Indeed, if $x$ is the generic point of one of the irreducible components $Z$ of $\operatorname{Supp}(\sh{F})$, it follows from the equivalence of \emph{a)} and \emph{c$'$)} in \sref{IV.3.1.3} that $x$ belongs to $\operatorname{Ass}(\sh{F})$, and $Z$ is a prime cycle associated to $\sh{F}$, necessarily maximal by virtue of \sref{IV.3.1.1.1}; finally, the last assertion follows trivially from \sref{IV.3.1.1.1}; indeed, the last assertion follows from a local result from Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~2, no~5, cor.~2 of prop.~7.
\end{proof}

\begin{corollary}[3.1.5]
\label{IV.3.1.5}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent.
For $\sh{F}=0$, it is necessary and sufficient that $\operatorname{Ass}(\sh{F})=\varnothing$.
\end{corollary}

\begin{proof}
The question being local, we reduce to the case where $X$ is affine, and the conclusion follows immediately from \sref{IV.3.1.2} and from Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~1, no~1, cor.~1 of prop.~2.
\end{proof}

\begin{proposition}[3.1.6]
\label{IV.3.1.6}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent; then $\operatorname{Ass}(\sh{F})$ is \emph{locally finite} (that is to say, every point of $X$ admits a neighbourhood whose intersection with $\operatorname{Ass}(\sh{F})$ is finite).
\end{proposition}

\begin{proof}
It suffices to consider the case where $X$ is affine, hence Noetherian, and then the proposition follows from \sref{IV.3.1.2} and from Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~1, no~4, cor.\ of th.~2.
\end{proof}

\begin{proposition}[3.1.7]
\label{IV.3.1.7}
Let $X$ be a prescheme.
\begin{enumerate}[label=(\roman*)]
  \item Let $0\to\sh{F}'\to\sh{F}\to\sh{F}''\to 0$ be an exact sequence of quasi-coherent $\sh{O}_X$-Modules.
  Then $\operatorname{Ass}(\sh{F}')\subset\operatorname{Ass}(\sh{F})\subset\operatorname{Ass}(\sh{F}')\cup\operatorname{Ass}(\sh{F}'')$.
  \item Let $\sh{F}$ be a quasi-coherent $\sh{O}_X$-Module, $(\sh{F}_\alpha)$ a family of quasi-coherent sub-$\sh{O}_X$-Modules of $\sh{F}$ such that $\sh{F}$ is the union of the $\sh{F}_\alpha$.
  Then $\operatorname{Ass}(\sh{F})=\bigcup_\alpha\operatorname{Ass}(\sh{F}_\alpha)$.
  \item For every family $(\sh{F}_\alpha)$ of quasi-coherent $\sh{O}_X$-Modules, we have $\operatorname{Ass}(\bigoplus_\alpha\sh{F}_\alpha)=\bigcup_\alpha\operatorname{Ass}(\sh{F}_\alpha)$.
\end{enumerate}
\end{proposition}

\begin{proof}
We immediately reduce to the corresponding propositions for modules (Bourbaki, \emph{loc.\ cit.}, \textsection~1, no~1, formula~(1), prop.~3 and cor.~1 of prop.~3).
\end{proof}

\begin{proposition}[3.1.8]
\label{IV.3.1.8}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent, $U$ an open subset of $X$, $\sh{J}$ a coherent Ideal of $\sh{O}_X$ defining a closed sub-prescheme of $X$ having $X-U$ as underlying space.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\operatorname{Ass}(\sh{F})\subset U$.
  \item For every affine open subset $V$, every section of $\sh{F}$ over $V$, whose restriction to $V\cap U$ is zero, is zero.
  \item The canonical homomorphism $\sh{F}\to\shHom_{\sh{O}_X}(\sh{J},\sh{F})$ is injective.
\end{enumerate}
\end{proposition}

\begin{proof}
The question being local, we can suppose $X=\operatorname{Spec}(A)$ affine, $A$ being a Noetherian ring, $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module, and $\sh{J}=\widetilde{\mathfrak{J}}$, where $\mathfrak{J}$ is an ideal of $A$.
The homomorphism $M\to\operatorname{Hom}_A(\mathfrak{J},M)$ sends $m\in M$ to the homomorphism $x\mapsto xm$ of $\mathfrak{J}$ into $M$; to say that it is not injective means that there exists $m\neq 0$ in $M$ such that $\mathfrak{J} m=0$.

Let us first show that \emph{c)} implies \emph{a)}: indeed, if $\operatorname{Ass}(\sh{F})$ met $X-U$, there would be a prime ideal $\mathfrak{p}\in\operatorname{Ass}(M)$ containing $\mathfrak{J}$, hence an element $m\neq 0$ of $M$ such that $\mathfrak{J} m=0$ for a $m\neq 0$ in $M$; indeed, if $l$ had $\mathfrak{J} m=0$ for all $a\in\mathfrak{J}$ not in $\mathfrak{q}$, for every prime ideal $\mathfrak{q}\neq\mathfrak{J}$ there exists $a\in\mathfrak{J}$ not in $\mathfrak{q}$, hence the relation $am=0$ implies that the canonical image of $m$ in $M_\mathfrak{q}$ is $0$, so otherwise said the canonical image of $m$ in $M_\mathfrak{q}$ is $0$ for all $\mathfrak{q}\notin\mathfrak{J}$.
In the second place, \emph{b)} implies \emph{c)}: indeed, if $l$ had $\mathfrak{J} m=0$ for a $m\neq 0$ in $M$, the section of $\sh{F}$ over $U$ whose restriction to $U$ is zero would be a section of $\sh{F}$ whose restriction to $U$ is zero.
Finally, \emph{a)} implies \emph{b)}.
To see this, note that the canonical homomorphism $M\to\prod_{\mathfrak{p}\in\operatorname{Ass}(M)}M_\mathfrak{p}$ is \emph{injective}: indeed, if $N$ is the kernel of this homomorphism, we have $\operatorname{Ass}(N)\subset\operatorname{Ass}(M)$; if there existed $\mathfrak{p}\in\operatorname{Ass}(N)$, there would be an element $n\in N$ with $\mathfrak{p}$ as annihilator; but by definition of $N$, there exists $s\notin\mathfrak{p}$ such that $sn=0$, which is absurd; hence $\operatorname{Ass}(N)=\varnothing$, whence $N=0$.
Now, condition \emph{a)} implies that if $m\in M$ is such that the canonical image of $m$ in $M_\mathfrak{p}$ is zero for all $\mathfrak{p}\in\operatorname{Ass}(M)$, then $m=0$.
\end{proof}

\begin{corollary}[3.1.9]
\label{IV.3.1.9}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent, $f$ a section of $\sh{O}_X$ over $X$.
For $f$ to be $\sh{F}$-regular \sref[0]{0.15.2.2}, it is necessary and sufficient that $\operatorname{Ass}(\sh{F})\subset X_f$.
\end{corollary}

\begin{proof}
Indeed, it is immediate that to say that $f$ is $\sh{F}$-regular means that the canonical homomorphism $\sh{F}\to\shHom_{\sh{O}_X}(f\sh{O}_X,\sh{F})$ is injective, and it suffices to apply \sref{IV.3.1.8} to the Ideal $\sh{J}=f\sh{O}_X$.
\end{proof}

Then, for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, we have $f(\operatorname{Ass}(\sh{F}))=\operatorname{Ass}(f_*(\sh{F}))$.
\end{proposition}

\begin{lemma}[3.1.10.1]
\label{IV.3.1.10.1}
Let $A$, $B$ be two Noetherian rings, $\rho:A\to B$ a ring homomorphism making $B$ a whole $A$-algebra, $M$ a $B$-module.
Then the prime ideals $\mathfrak{p}\in\operatorname{Ass}(M_{[\rho]})$ are the inverse images by $\rho$ of the prime ideals $\mathfrak{q}\in\operatorname{Ass}(M)$.
\end{lemma}

\begin{proof}
Indeed, if $\mathfrak{q}\in\operatorname{Ass}(M)$, $\mathfrak{q}$ is the annihilator in $B$ of an element $x\in M$, hence $\rho^{-1}(\mathfrak{q})$ is the annihilator in $A$ of $x$. Conversely, suppose $\mathfrak{p}\in\operatorname{Ass}(M_{[\rho]})$, so that $\mathfrak{p}$ is the image by $\rho$ of the annihilator $\mathfrak{b}$ in $B$ of an element $x\in M$; it follows from the prime theorem of Cohen--Seidenberg that there exists a prime ideal $\mathfrak{q}$ of $B$ containing $\mathfrak{b}$ and whose inverse image is $\mathfrak{p}$ (Bourbaki, \emph{Alg.\ comm.}, chap.~V, \textsection~2, no~1, cor.~2 of th.~1); by considering a minimal prime ideal among those contained in $\mathfrak{q}$ and containing $\mathfrak{b}$, we can evidently suppose that $\mathfrak{q}$ is itself one of these minimal primes. But as $B.x\subset M$ is isomorphic to $B/\mathfrak{b}$, we know that one then has $\mathfrak{q}\in\operatorname{Ass}(B/\mathfrak{b})\subset\operatorname{Ass}(M)$ (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~1, no~4, th.~2).
\end{proof}

\begin{corollary}[3.1.11]
\label{IV.3.1.11}
Under the hypotheses of \sref{IV.3.1.10}, for $\sh{F}$ to be without associated immersed prime cycle, it suffices that the same be true of $f_*(\sh{F})$.
\end{corollary}

\begin{proof}
Suppose indeed that $f_*(\sh{F})$ has no immersed associated prime cycle.
Note that if $A$ is a whole algebra over a field $k$, all prime ideals of $A$ are maximal (Bourbaki, \emph{Alg.\ comm.}, chap.~V, \textsection~2, no~1, prop.~1); it follows from \sref[I]{I.6.2.2} that the fibres of $f$ are \emph{discrete} spaces; if $x$, $x'$ are two distinct points of $\operatorname{Ass}(\sh{F})$, none of the two can be adherent to the other if $f(x)=f(x')$; and if $f(x)\neq f(x')$, \sref{IV.3.1.10} and the hypothesis imply that neither of the two points $f(x)$, $f(x')$ can be adherent to the other, hence the same is true of $x$ and $x'$.
\end{proof}

\begin{remark}[3.1.12]
\label{IV.3.1.12}
Under the hypotheses of \sref{IV.3.1.10}, it may happen on the contrary that $\sh{F}$ is without associated immersed prime cycle, but not $f_*(\sh{F})$.
Take for example $Y=\operatorname{Spec}(k[\mathrm{T}])$ where $k$ is a field (an ``affine line''), and $X$ the \emph{sum} of $X_1=Y$ and $X_2=\operatorname{Spec}(k)$, the morphism $X_2\to Y$ corresponding to the canonical homomorphism $k[\mathrm{T}]\to k[\mathrm{T}]/\mathfrak{m}$, where $\mathfrak{m}$ is the maximal ideal $(\mathrm{T})$.
It is clear that the morphism $f:X\to Y$ is finite; if one takes $\sh{F}=\sh{O}_X$, $\sh{F}$ is without associated immersed prime cycle, but $f_*(\sh{F})=\widetilde{M}$, where $M$ is the $k[\mathrm{T}]$-module direct sum of $k[\mathrm{T}]$ and $k$, hence $\operatorname{Ass}(M)$ consists of the generic point $(0)$ of $Y$ and of the point $\mathfrak{m}$.
\end{remark}

\begin{proposition}[3.1.13]
\label{IV.3.1.13}
Let $X$ be a prescheme locally Noetherian, $U$ an open subset of $X$, $i:U\to X$ the canonical injection.
For every quasi-coherent $\sh{O}_U$-Module $\sh{F}$, we have $\operatorname{Ass}(i_*(\sh{F}))=\operatorname{Ass}(\sh{F})$.
\end{proposition}

\begin{proof}
Recall that $i_*(\sh{F})$ is a quasi-coherent $\sh{O}_X$-Module \sref[I]{I.2.2} and \sref[I]{I.7.4}; as $i_*(\sh{F})|U=\sh{F}$, on has $({\operatorname{Ass}(i_*(\sh{F}))})\cap U=\operatorname{Ass}(\sh{F})$, and it remains therefore to prove that $\operatorname{Ass}(i_*(\sh{F}))\subset U$.
But by \sref{IV.3.1.8}, this relation means that for every affine open subset $V$ of $X$, every section of $i_*(\sh{F})$ over $V$ which is zero on $U\cap V$, is zero, a condition which is trivially satisfied since $\Gamma(V,i_*(\sh{F}))=\Gamma(U\cap V,\sh{F})=\Gamma(U\cap V,i_*(\sh{F}))$.
\end{proof}

\subsection{Irredundant decompositions}
\label{subsection:IV.3.2}

\begin{proposition}[3.2.1]
\label{IV.3.2.1}
Let $X$ be a prescheme locally Noetherian, $U$ an open dense subset of $X$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $X$ is reduced.
  \item The sub-prescheme induced on $U$ is reduced and $X$ is without associated immersed prime cycle.
  \item $X$ is without associated immersed prime cycle and for every generic point $x$ of an irreducible component of $X$, we have $\operatorname{length}(\sh{O}_x)=1$.
\end{enumerate}
The prime cycles associated to $X$ are then identical to the irreducible components of $X$.
\end{proposition}

\begin{proof}
It is clear that if $X$ is reduced then so is the sub-prescheme induced on $U$.
In addition, the existence of associated immersed prime cycles being a local question, we can restrict to the case where $X=\operatorname{Spec}(A)$ is affine, $A$ being Noetherian.
If $A$ is reduced, we know that the minimal prime ideals of $A$ form a reduced primary decomposition of $(0)$ (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~2, no~5, prop.~10) and these are all the elements of $\operatorname{Ass}(A)$, hence there are no immersed prime ideals associated to $A$, which shows that \emph{a)} implies \emph{b)}.
It is immediate that \emph{b)} implies \emph{c)}, for a generic point $x$ of an irreducible component belongs to $U$, hence $\sh{O}_x$ is a field.
Finally, \emph{c)} implies \emph{a)}: it suffices indeed to note that if $\sh{N}$ is the Nilradical of $\sh{O}_X$, which is a coherent Ideal, $\operatorname{Supp}(\sh{N})$ cannot contain any of the generic points of the irreducible components of $X$ by hypothesis; if $\operatorname{Supp}(\sh{N})$ were non-empty and if $x$ was one of its maximal points, the criterion \sref{IV.3.1.3},~\emph{c$'$)}) would show that $x\in\operatorname{Ass}(\sh{O}_X)$, and $\overline{\{x\}}$ would therefore be an associated immersed prime cycle of $X$, contradicting the hypothesis; hence $\sh{N}=0$.
\end{proof}

\begin{definition}[3.2.2]
\label{IV.3.2.2}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent.
We say that $\sh{F}$ is \emph{reduced} if it satisfies the following two conditions: 1\textsuperscript{o} $\sh{F}$ is without associated immersed prime cycle; 2\textsuperscript{o} for every maximal point $x$ of $\operatorname{Supp}(\sh{F})$, we have $\operatorname{length}(\sh{F}_x)=1$.
\end{definition}

Condition 1\textsuperscript{o} means that the prime cycles associated to $\sh{F}$ are the irreducible components of $\operatorname{Supp}(\sh{F})$ \sref{IV.3.1.4}, and condition 2\textsuperscript{o} means that for every generic point $x$ of such a component, we have $\operatorname{length}(\sh{F}_x)=1$.

For an affine scheme $X$, this definition gives in particular the notion of a \emph{reduced module} over a Noetherian ring $A$; a finitely generated $A$-module $M$ is said to be \emph{reduced} if it has no immersed associated prime ideals and if, for all $\mathfrak{p}\in\operatorname{Ass}(M)$, $\operatorname{length}_{A_\mathfrak{p}}(M_\mathfrak{p})=1$.
Returning to the case of a locally Noetherian prescheme $X$ and a coherent $\sh{O}_X$-Module $\sh{F}$, we say that $\sh{F}$ is \emph{reduced at a point} $x\in X$ if $\sh{F}_x$ is a reduced $\sh{O}_x$-module; this means that $x$ does not belong to any associated immersed prime cycle of $\sh{F}$ and that $\operatorname{length}(\sh{F}_z)=1$ \emph{for all maximal points $z$ of $\operatorname{Supp}(\sh{F})$ such that $x\in\overline{\{z\}}$}.
It is clear that if $\sh{F}$ is a coherent $\sh{O}_X$-Module reduced, it is reduced at every point of $X$; conversely, if $\sh{F}$ is \emph{reduced at a point} $x$, there exists an open neighbourhood $U$ of $x$ such that $\sh{F}|U$ is an $\sh{O}_X$-Module \emph{reduced}: it suffices indeed to take $U$ such that $\sh{F}|U$ does not meet any associated immersed prime cycle of $\sh{F}$ (such a neighbourhood exists since these cycles form a locally finite set of closed subsets of $X$).
To say that $\sh{O}_X$ is reduced at a point $x$ is equivalent to saying that $X$ is \emph{reduced} at the point $x$.

\oldpage[IV-2]{41}

We say that the locally Noetherian prescheme $X$ is \emph{irredundant} if $\sh{O}_X$ is irredundant (which implies that $X$ is \emph{irreducible}); for $X$ to be \emph{integral}, it is necessary and sufficient that the $\sh{O}_X$-Module $\sh{O}_X$ be integral (\sref[I]{I.2.1.8} and \sref{IV.3.2.1}).
If $\sh{O}_X$ is integral at a point $x$, that is to say if the ring $\sh{O}_x$ is integral, we say that $X$ is \emph{integral at the point} $x$.
We say that a closed sub-prescheme $Y$ of $X$ is \emph{primary} in $X$ if the Ideal $\sh{J}$ of $\sh{O}_X$ which defines $Y$ is primary in $\sh{O}_X$.

\begin{proposition}[3.2.3]
\label{IV.3.2.3}
Let $X$ be a prescheme locally Noetherian, $U$ an open subset of $X$, $\sh{F}$ an $\sh{O}_X$-Module coherent such that $U\cap\operatorname{Supp}(\sh{F})$ is dense in $\operatorname{Supp}(\sh{F})$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\sh{F}$ is reduced.
  \item $\sh{F}|U$ is reduced, and $\sh{F}$ is without associated immersed prime cycle.
  \item There exists a closed reduced sub-prescheme $X'$ of $X$ and a coherent $\sh{O}_{X'}$-Module $\sh{F}'$ without torsion and of rank $1$ on every irreducible component of $X'$, such that if $j:X'\to X$ is the canonical injection, we have $j_*(\sh{F}')=\sh{F}$.
\end{enumerate}
Furthermore, when this is so, the sub-prescheme $X'$ is defined by the Ideal $\sh{J}$ of $\sh{O}_X$ annihilator of $\sh{F}$.
\end{proposition}

\begin{proof}
To see that \emph{c)} implies \emph{a)}, we can, by virtue of \sref{IV.3.1.3}, limit ourselves to the case where $X'=X$ is integral, of generic point $x$, and one can in addition suppose $X=\operatorname{Spec}(A)$ affine and $\sh{F}=\widetilde{M}$, where $A$ is integral and $M$ is an $A$-module without torsion of rank $1$; the annihilator of every element of $M$ being then reduced to $0$, we have $\operatorname{Ass}(\sh{F})=\{x\}$ and $\sh{F}_x$ is isomorphic to $\sh{O}_x$, the field of fractions of $A$, hence the conditions of the definition \sref{IV.3.2.2} are satisfied.
As the existence of associated immersed prime cycles is a local question, it is clear that if $\sh{F}$ has no such cycles and if $U\cap\operatorname{Supp}(\sh{F})$ is dense in $\operatorname{Supp}(\sh{F})$, we have $\operatorname{Ass}(\sh{F})=\operatorname{Ass}(\sh{F}|U)$, hence \emph{a)} and \emph{b)} are equivalent.
If \emph{a)} is satisfied, take for $X'$ the closed reduced sub-prescheme of $X$ whose $\operatorname{Supp}(\sh{F})$ is the underlying space \sref[I]{I.5.2.1}, and let $\sh{F}'=j^*(\sh{F})$; a point $x$ of $\operatorname{Ass}(\sh{F})$ is necessarily a maximal point of $X'$ and as $\operatorname{length}(\sh{F}_x)=1$, $\sh{F}'_x$ is isomorphic to $\sh{F}_x$, hence to the residue field $\boldsymbol{k}(x)=\sh{O}_{X',x}$, which proves that $\sh{F}'$ is without torsion and of rank $1$ \sref[I]{I.7.4.6} and \sref[I]{I.7.4.2}.
Finally, the last assertion is trivial, since for all $y\in X'$, the annihilator of the $\sh{O}_{X',y}$-Module $\sh{F}'_y$ is zero.
\end{proof}

\begin{definition}[3.2.4]
\label{IV.3.2.4}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent.
We say that $\sh{F}$ is \emph{irredundant} if $\operatorname{Ass}(\sh{F})$ is reduced to a single element $x$; if $\sh{F}$ is of finite type, we say that $\sh{F}$ is \emph{integral} if in addition $\sh{F}$ is reduced (in other words if $\operatorname{length}(\sh{F}_x)=1$).
We say that a quasi-coherent sub-$\sh{O}_X$-Module $\sh{G}$ of a quasi-coherent $\sh{O}_X$-Module coherent $\sh{F}$ is \emph{primary} in $\sh{F}$ if $\sh{F}/\sh{G}$ is irredundant.
\end{definition}

For an affine scheme $X$, this definition gives in particular the notion of \emph{integral module} over a Noetherian ring $A$; a finitely generated $A$-module $M$ is said to be \emph{integral} if it is of finite type, if $\operatorname{Ass}(M)$ is reduced to a single prime ideal $\mathfrak{p}$) and if in addition $\operatorname{length}_{A_\mathfrak{p}}(M_\mathfrak{p})=1$.
Returning to the case of a locally Noetherian prescheme $X$ and a coherent $\sh{O}_X$-Module $\sh{F}$, we say that $\sh{F}$ is \emph{integral at a point} $x\in X$ if $\sh{F}_x$ is an integral $\sh{O}_x$-module: this means that $x$ \emph{belongs to at most a single associated prime cycle of $\sh{F}$ and that $\operatorname{length}(\sh{F}_z)=1$ at its generic point $z$}.
It is clear that if $\sh{F}$ is a coherent integral $\sh{O}_X$-Module, it is integral at every point of $X$; conversely, if $\sh{F}$ is \emph{integral at a point} $x$, there exists an open neighbourhood $U$ of $x$ such that $\sh{F}|U$ is an $\sh{O}_U$-Module \emph{integral}: it suffices indeed to take $U$ such that $\sh{F}|U$ is an $\sh{O}_X$-Module reduced.

\oldpage[IV-2]{42}

\begin{definition}[3.2.5]
\label{IV.3.2.5}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent.
An \emph{irredundant decomposition} of $\sh{F}$ is called a family $(\sh{F}_\alpha)_{\alpha\in I}$ of $\sh{O}_X$-Module quotients of $\sh{F}$ such that the $\sh{F}_\alpha$ are irredundant, that the family $(\operatorname{Supp}(\sh{F}_\alpha))$ is locally finite, and that the canonical homomorphism $\sh{F}\to\bigoplus_{\alpha\in I}\sh{F}_\alpha$ is injective.
We say that such a decomposition is \emph{reduced} if the sets $\operatorname{Ass}(\sh{F}_\alpha)$ are pairwise distinct, and if there does not exist a proper subset $J\subsetneq I$ such that the subfamily $(\sh{F}_\alpha)_{\alpha\in J}$ is also an irredundant decomposition of $\sh{F}$.
\end{definition}

If $(\sh{F}_\alpha)_{\alpha\in I}$ is an irredundant decomposition (resp.\ irredundant reduced decomposition) of $\sh{F}$, and if one sets $\sh{F}_\alpha=\sh{F}/\sh{G}_\alpha$, we say that the family $(\sh{G}_\alpha)_{\alpha\in I}$ of sub-$\sh{O}_X$-Modules of $\sh{F}$ is a \emph{primary decomposition} (resp.\ \emph{irredundant reduced primary decomposition}) of $\sh{F}$; one observes that the hypothesis of injectivity of the canonical homomorphism $\sh{F}\to\bigoplus_{\alpha\in I}\sh{F}_\alpha$ is equivalent to the condition $\bigcap_{\alpha\in I}\sh{G}_\alpha=0$.

\oldpage[IV-2]{42}

If $(\sh{F}_\alpha)$ is an irredundant decomposition of $\sh{F}$, to say that it is reduced amounts to saying that the sets $\operatorname{Ass}(\sh{F}_\alpha)$ are pairwise distinct and contained in $\operatorname{Ass}(\sh{F})$; if $\operatorname{Ass}(\sh{F}_\alpha)=\{x_\alpha\}$ for all $\alpha\in I$, $\alpha\mapsto x_\alpha$ is then a bijection of $I$ onto $\operatorname{Ass}(\sh{F})$: these properties are local in nature and follow from Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~2, no~3, prop.~4.

\begin{proposition}[3.2.6]
\label{IV.3.2.6}
Let $X$ be a prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent.
Then there exists a reduced irredundant decomposition $(\sh{F}^{(x)})_{x\in\operatorname{Ass}(\sh{F})}$, formed of coherent $\sh{O}_X$-Modules, such that for every $x\in\operatorname{Ass}(\sh{F})$, we have $\operatorname{Ass}(\sh{F}^{(x)})=\{x\}$.
For every $x\in\operatorname{Ass}(\sh{F})$ such that $\overline{\{x\}}$ is not immersed, $\sh{F}^{(x)}$ is uniquely determined as the image of the canonical homomorphism $\sh{F}\to i_*(i^*(\sh{F}))$, where $i$ is the canonical morphism $\operatorname{Spec}(\boldsymbol{k}(x))\to X$.
\end{proposition}

\begin{proof}
For every $x\in\operatorname{Ass}(\sh{F})$, let $U$ be an affine open neighbourhood of $x$, and let $\sh{F}|U=\widetilde{M}$, where $M$ is a finitely generated $A$-module.
We know (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~1, no~1, prop.~4) that there exists a sub-module $N$ of $M$ such that, if one sets $P=M/N$, one has $\operatorname{Ass}(P)=\{x\}$ and $\operatorname{Ass}(N)=\operatorname{Ass}(M)-\{x\}$.
Let $\sh{G}=\widetilde{P}$, which is a quasi-coherent $\sh{O}_U$-Module; let $u:j^*(\sh{F})\to\sh{G}$ be the surjective homomorphism corresponding to the surjection $M\to P$; we obtain by composition a homomorphism $j_*(u):j_*(j^*(\sh{F}))\to j_*(\sh{G})$, whence by composition a homomorphism
\[
v:\sh{F}\xrightarrow{\rho_\sh{F}}j_*(j^*(\sh{F}))\xrightarrow{j_*(u)}j_*(\sh{G})
\]
whose restriction $u$ to $U$; we will denote by $\sh{F}^{(x)}$ the image of $\sh{F}$ by this homomorphism, which is a quasi-coherent $\sh{O}_X$-Module coherent \sref[I]{I.6.1.1}.
We have $\operatorname{Ass}(j_*(\sh{G}))=\{x\}$ by virtue of \sref{IV.3.1.13}, and \emph{a fortiori} \sref{IV.3.1.7} $\operatorname{Ass}(\sh{F}^{(x)})=\{x\}$, since $\sh{F}^{(x)}\neq 0$.
Furthermore, if $\overline{\{x\}}$ is not immersed, the characterisation of $\sh{F}^{(x)}$ when $\overline{\{x\}}$ is not immersed follows from Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~2, no~3, prop.~5, the question being local, and taking into account \sref[I]{I.1.6.7}.

$\sh{N}^{(x)}=\operatorname{Ker}(v)$, hence $x\notin\operatorname{Ass}(\sh{N}^{(x)})$, and $\sh{N}^{(x)}|U=\widetilde{N}$, hence $x\notin\operatorname{Ass}(\sh{N}^{(x)})$.
It follows that the homomorphism $\sh{F}\to\bigoplus_{x\in\operatorname{Ass}(\sh{F})}\sh{F}^{(x)}$ is injective, for its kernel $\sh{H}$ is contained in the intersection of the $\operatorname{Ass}(\sh{N}^{(x)})$, which is empty; by \sref{IV.3.1.5}, $\sh{H}=0$.
Taking into account that $\operatorname{Ass}(\sh{F})$ is locally finite \sref{IV.3.1.6}, it is clear that $(\sh{F}^{(x)})_{x\in\operatorname{Ass}(\sh{F})}$ is a reduced irredundant decomposition of $\sh{F}$ satisfying the conditions of the statement.
\end{proof}

\oldpage[IV-2]{43}

\begin{corollary}[3.2.7]
\label{IV.3.2.7}
Under the hypotheses of \sref{IV.3.2.6}, if $\sh{F}$ has no associated immersed prime cycle, there exists a unique reduced irredundant decomposition of $\sh{F}$.
\end{corollary}

\begin{corollary}[3.2.8]
\label{IV.3.2.8}
Let $X$ be a Noetherian prescheme, $\sh{F}$ an $\sh{O}_X$-Module coherent.
There exists a finite filtration $(\sh{F}_i)_{0\leqslant i\leqslant n}$ of $\sh{F}$ such that $\sh{F}_0=\sh{F}$, $\sh{F}_n=0$, formed of coherent $\sh{O}_X$-Modules, and such that the quotients $\sh{F}_i/\sh{F}_{i+1}$ are either zero or irredundant, and $\operatorname{Ass}(\sh{F}_i/\sh{F}_{i+1})\subset\operatorname{Ass}(\sh{F})$.
\end{corollary}

\begin{proof}
Indeed, $\sh{F}$ is isomorphic to a sub-$\sh{O}_X$-Module direct sum $\bigoplus_{j=1}^n\sh{G}_j$, where the $\sh{G}_j$ are irredundant and coherent \sref{IV.3.2.6}; as every sub-$\sh{O}_X$-Module quasi-coherent of $\sh{G}_j$ is either zero or irredundant \sref{IV.3.1.7}, the $\sh{F}_i=\sh{F}\cap(\bigoplus_{j=1}^{n-i}\sh{G}_j)$ satisfy the requirements, $\sh{F}_i/\sh{F}_{i+1}$ being isomorphic to a sub-$\sh{O}_X$-Module coherent of $\sh{G}_{n-i}$.
\end{proof}

\subsection{Relations with flatness}
\label{subsection:IV.3.3}

\begin{proposition}[3.3.1]
\label{IV.3.3.1}
Let $f:X\to Y$ be a morphism, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent and $f$-flat, $\sh{E}$ an $\sh{O}_Y$-Module quasi-coherent.
If, for every $y\in Y$, we set $\sh{F}_y=\sh{F}\otimes_{\sh{O}_Y}\boldsymbol{k}(y)$, then
\[
\label{IV.3.3.1.1}
\operatorname{Ass}(\sh{E}\otimes_{\sh{O}_Y}\sh{F})\supset\bigcup_{y\in\operatorname{Ass}(\sh{E})}\operatorname{Ass}(\sh{F}_y)
\tag{3.3.1.1}
\]
and the two sides are equal if $Y$ is locally Noetherian.
\end{proposition}

\begin{proof}
(Of course, $\sh{F}_y$ is a sheaf on the fibre $f^{-1}(y)$, and one identifies it with a sub-space of $X$ \sref[I]{I.3.6.1}.)
The question being local on $X$ and on $Y$, we reduce to the case where $X$ and $Y$ are affine, and the proposition is then proved in Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~2, no~6, th.~2.
\end{proof}

\begin{corollary}[3.3.2]
\label{IV.3.3.2}
Let $Y$ be a prescheme locally Noetherian without associated immersed prime cycles, $f:X\to Y$ a morphism, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent and $f$-flat.
Then, for every $x\in\operatorname{Ass}(\sh{F})$, $f(x)$ is a maximal point of $Y$.
\end{corollary}

\begin{proof}
It suffices to apply \sref{IV.3.3.1} with $\sh{E}=\sh{O}_Y$, since $\operatorname{Ass}(\sh{O}_Y)$ is by hypothesis the set of maximal points of $Y$.
\end{proof}

\begin{corollary}[3.3.3]
\label{IV.3.3.3}
Under the hypotheses of \sref{IV.3.3.1}, suppose in addition that $X$ and $Y$ are locally Noetherian, $\sh{E}$ and $\sh{F}$ coherent.
Then the following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\sh{E}\otimes_Y\sh{F}$ is without associated immersed prime cycle.
  \item For every point $y\in\operatorname{Ass}(\sh{E})\cap f(\operatorname{Supp}(\sh{F}))$, $\overline{\{y\}}$ is a non-immersed associated prime cycle of $\sh{E}$ and $\sh{F}_y$ is without associated immersed prime cycle.
\end{enumerate}
\end{corollary}

\begin{proof}
\oldpage[IV-2]{44}
Suppose \emph{a)} is satisfied.
The hypotheses imply that $\sh{E}\otimes_Y\sh{F}$ is a coherent $\sh{O}_X$-Module \sref[I]{I.5.3.11} and \sref[I]{I.5.3.5}; its associated prime cycles are therefore the irreducible components of $\operatorname{Supp}(\sh{E}\otimes_Y\sh{F})$ \sref{IV.3.1.4}, and for every maximal point $x$ of $\operatorname{Supp}(\sh{E}\otimes_Y\sh{F})$, $f(x)=y$ is a maximal point of $\operatorname{Supp}(\sh{E})$ and $x$ is a maximal point of $\operatorname{Supp}(\sh{F}_y)$ \sref{IV.2.5.5}.
As, by virtue of \sref{IV.3.3.1} and the fact that $y\in f(\operatorname{Supp}(\sh{F}))$ implies $\sh{F}_y\neq 0$ \sref[I]{I.9.1.13}, every point of $\operatorname{Ass}(\sh{E})\cap f(\operatorname{Supp}(\sh{F}))$ is the image by $f$ of a maximal point of $\operatorname{Supp}(\sh{E}\otimes_Y\sh{F})$, we see that condition \emph{b)} is satisfied.

Conversely, suppose \emph{b)} is satisfied, and let us show that if $z$, $z'$ are two distinct points of $\operatorname{Ass}(\sh{E}\otimes_Y\sh{F})$, neither of the two can be adherent to the other.
In the first place, if $f(z)=f(z')=y$, we have $y\in\operatorname{Ass}(\sh{E})$ and $z$ and $z'$ belong to $\operatorname{Ass}(\sh{F}_y)$ by \sref{IV.3.3.1}, whence $y\notin f(\operatorname{Supp}(\sh{F}))$; as by hypothesis neither of the two points $z$, $z'$ is adherent to the other \emph{within} $f^{-1}(y)$, neither can be adherent to the other \emph{within} $X$.
If $y=f(z)$ and $y'=f(z')$ are distinct, they belong to $\operatorname{Ass}(\sh{E})\cap f(\operatorname{Supp}(\sh{F}))$, hence neither of the two can be adherent to the other in $Y$; it follows from the continuity of $f$ that neither of the points $z$, $z'$ can be adherent to the other in $X$.
\end{proof}

\begin{proposition}[3.3.4]
\label{IV.3.3.4}
Let $X$, $Y$ be two preschemes locally Noetherian, $f:X\to Y$ a morphism, $\sh{E}$ an $\sh{O}_Y$-Module coherent, $\sh{F}$ an $\sh{O}_X$-Module coherent and $f$-flat.
Then the following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\sh{E}\otimes_Y\sh{F}$ is reduced \sref{IV.3.2.2}.
  \item For every point $y\in\operatorname{Ass}(\sh{E})\cap f(\operatorname{Supp}(\sh{F}))$, $\overline{\{y\}}$ is a non-immersed associated prime cycle of $\sh{E}$ and $\sh{F}_y$ is reduced.
\end{enumerate}
\end{proposition}

\begin{proof}
Suppose \emph{a)} is satisfied.
We already know \sref{IV.3.3.3} that for every $y\in\operatorname{Ass}(\sh{E})\cap f(\operatorname{Supp}(\sh{F}))$, $\overline{\{y\}}$ is a non-immersed associated prime cycle of $\sh{E}$ and $\sh{F}_y$ is without associated immersed prime cycle.
Furthermore \sref{IV.2.5.5}, for every $x\in\operatorname{Ass}(\sh{E}\otimes_Y\sh{F})\cap f^{-1}(y)$, we have $1=\operatorname{length}((\sh{E}\otimes_Y\sh{F})_x)=\operatorname{length}(\sh{E}_y)\cdot\operatorname{length}((\sh{F}_y)_x)$, hence $\operatorname{length}(\sh{E}_y)=1$, which proves \emph{b)}.

Conversely, suppose \emph{b)} is satisfied; we already know that every point $x\in\operatorname{Ass}(\sh{E}\otimes_Y\sh{F})$ is a maximal point of $\operatorname{Supp}(\sh{E}\otimes_Y\sh{F})$, that $y=f(x)$ is a maximal point of $\operatorname{Supp}(\sh{E})$ and $x$ a maximal point of $\operatorname{Supp}(\sh{F}_y)$ \sref{IV.3.3.1} and \sref{IV.3.3.3}; in addition it follows from the hypothesis and from \sref{IV.2.5.5} that $\operatorname{length}((\sh{E}\otimes_Y\sh{F})_x)=1$, which proves \emph{a)}.
\end{proof}

\begin{corollary}[3.3.5]
\label{IV.3.3.5}
Let $X$, $Y$ be two preschemes locally Noetherian, $f:X\to Y$ a flat morphism; if $Y$ is reduced at the points of $f(X)$ and if $f^{-1}(y)$ is a $\boldsymbol{k}(y)$-reduced prescheme for every $y\in f(X)$, then $X$ is reduced.
\end{corollary}

\begin{proof}
As the Nilradical $\sh{N}_Y$ is coherent, the set of points where $Y$ is reduced is open \sref[0]{0.5.2.2}, and one can restrict to the case where $Y$ is reduced.
It suffices then to apply \sref{IV.3.3.4} with $\sh{E}=\sh{O}_Y$ and $\sh{F}=\sh{O}_X$.
\end{proof}

\begin{proposition}[3.3.6]
\label{IV.3.3.6}
Let $f:X\to S$, $g:Y\to S$ be two morphisms, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent, $\sh{G}$ an $\sh{O}_Y$-Module quasi-coherent.
Suppose that: 1\textsuperscript{o} $\sh{G}$ is $g$-flat; 2\textsuperscript{o} $X$ is locally Noetherian, and for every $s\in S$, $g^{-1}(s)$ is locally Noetherian (which will hold if $Y$ is also locally Noetherian).
Let $Z=X\times_S Y$; for every pair $(x,y)$ such that $x\in X$, $y\in Y$,
\oldpage[IV-2]{45}
and $f(x)=g(y)=s$, let $\mathrm{T}_{x,y}$ be the prescheme $\operatorname{Spec}(\boldsymbol{k}(x)\otimes_{\boldsymbol{k}(s)}\boldsymbol{k}(y))$, and let $\mathrm{I}_{x,y}$ be the image of $\operatorname{Ass}(\sh{O}_{\mathrm{T}_{x,y}})$ by the canonical monomorphism $\mathrm{T}_{x,y}\to Z$ \sref[I]{I.3.4.9}.
Then
\[
\label{IV.3.3.6.1}
\operatorname{Ass}(\sh{F}\otimes_S\sh{G})=\bigcup_{x\in\operatorname{Ass}(\sh{F})}\bigg(\bigcup_{y\in\operatorname{Ass}(\sh{G}_{f(x)})}\mathrm{I}_{x,y}\bigg)
\tag{3.3.6.1}
\]
where for all $s\in S$, $\sh{G}_s=\sh{G}\otimes_{\sh{O}_S}\boldsymbol{k}(s)$.
\end{proposition}

\begin{proof}
Let $p:Z\to X$, $q:Z\to Y$ be the canonical projections, so that we have the commutative diagram
\[
\xymatrix{
X & Z\ar[l]_-{p}\ar[r]^-{q} & Y\\
& S\ar[ul]^-{f}\ar[ur]_-{g} &
}
\]
Set $\sh{G}'=q^*(\sh{G})$, so that $\sh{F}\otimes_S\sh{G}=\sh{F}\otimes_X\sh{G}'$; as $\sh{G}'$ is $p$-flat \sref{IV.2.1.4}, it follows from \sref{IV.3.3.1} that
\[
\label{IV.3.3.6.2}
\operatorname{Ass}(\sh{F}\otimes_S\sh{G})=\bigcup_{x\in\operatorname{Ass}(\sh{F})}\operatorname{Ass}(\sh{G}'_x)
\tag{3.3.6.2}
\]
with $\sh{G}'_x=\sh{G}'\otimes_{\sh{O}_X}\boldsymbol{k}(x)$.
If $s=f(x)$, we have $\sh{G}'_x=\sh{G}_s\otimes_{\boldsymbol{k}(s)}\boldsymbol{k}(x)$, and $p^{-1}(x)=g^{-1}(s)\otimes_{\boldsymbol{k}(s)}\boldsymbol{k}(x)$; furthermore, as the field $\boldsymbol{k}(x)$ is a flat $\boldsymbol{k}(s)$-module, the morphism $p^{-1}(x)\to g^{-1}(s)$ is flat; applying \sref{IV.3.3.1} to this morphism, we get
\[
\label{IV.3.3.6.3}
\operatorname{Ass}(\sh{G}'_x)=\bigcup_{y\in\operatorname{Ass}(\sh{G}_s)}\operatorname{Ass}(\sh{O}_{\mathrm{T}_{x,y}})
\tag{3.3.6.3}
\]
whence the proposition.
\end{proof}

We note that if, in the statement, we suppress hypothesis 2\textsuperscript{o}, we can still conclude, by virtue of \sref{IV.3.3.1}, the relation
\[
\label{IV.3.3.6.4}
\operatorname{Ass}(\sh{F}\otimes_S\sh{G})\supset\bigcup_{x\in\operatorname{Ass}(\sh{F})}\bigg(\bigcup_{y\in\operatorname{Ass}(\sh{G}_{f(x)})}\mathrm{I}_{x,y}\bigg).
\tag{3.3.6.4}
\]

\begin{corollary}[3.3.7]
\label{IV.3.3.7}
Under the hypotheses of \sref{IV.3.3.6}, suppose in addition that $S$ is also locally Noetherian and that $f(\operatorname{Ass}(\sh{F}))\subset\operatorname{Ass}(\sh{O}_S)$.
Then
\[
\label{IV.3.3.7.1}
\operatorname{Ass}(\sh{F}\otimes_S\sh{G})=\bigcup_{(x,y)\in C}\mathrm{I}_{x,y}
\tag{3.3.7.1}
\]
where $C$ is the set of pairs $(x,y)$ such that $x\in\operatorname{Ass}(\sh{F})$, $y\in\operatorname{Ass}(\sh{G})$ and $f(x)=g(y)$.
\end{corollary}

\begin{proof}
As $\sh{G}$ is $g$-flat, it follows indeed from \sref{IV.3.3.1} that the relation ``$s\in\operatorname{Ass}(\sh{G}_s)$'' is equivalent to ``$y\in\operatorname{Ass}(\sh{G})$'': the conclusion follows from \sref{IV.3.3.6.1}.
\end{proof}

\begin{remark}[3.3.8]
\label{IV.3.3.8}
We shall see later \sref{IV.4.2.2} that under the hypotheses of \sref{IV.3.3.6}, $\mathrm{T}_{x,y}$ is a prescheme \emph{without associated immersed prime cycle}; it will follow that if $\sh{F}$ and the $\sh{G}_s$ are without associated immersed prime cycle, the same is true of $\sh{F}\otimes_S\sh{G}$.
\end{remark}

\begin{corollary}[3.3.9]
\label{IV.3.3.9}
Under the conditions of \sref{IV.3.3.7}, we have
\[
\label{IV.3.3.9.1}
q(\operatorname{Ass}(\sh{F}\otimes_S\sh{G}))\subset\operatorname{Ass}(\sh{G})
\tag{3.3.9.1}
\]
(where $q:X\times_S Y\to Y$ is the canonical projection).
\end{corollary}

\begin{proof}
Indeed, if $(x,y)\in Z$, we have $q(\mathrm{I}_{x,y})=\{y\}\subset\operatorname{Ass}(\sh{G})$.
\end{proof}

\subsection{Properties of the sheaves \texorpdfstring{$\sh{F}/t\sh{F}$}{F/tF}}
\label{subsection:IV.3.4}

\begin{proposition}[3.4.1]
\label{IV.3.4.1}
Let $X$ be a prescheme locally Noetherian, $t$ a section of $\sh{O}_X$ over $X$, $Y$ the closed sub-prescheme of $X$ defined by the Ideal $t\sh{O}_X$ of $\sh{O}_X$.
Let $\sh{F}$ be an $\sh{O}_X$-Module coherent, $S$ the closed reduced sub-prescheme of $X$ having as underlying space $\operatorname{Supp}(\sh{F})$, $(S_i)$ the family of closed reduced sub-preschemes of $X$ having as underlying spaces the irreducible components of $S$; we denote by $s_i$ the generic point of $S_i$.
Finally, let $Z$ be an irreducible component of $\operatorname{Supp}(\sh{F}/t\sh{F})=S\cap Y$, $z$ its generic point.
\begin{enumerate}[label=(\roman*)]
  \item For every $i$ such that $Z\subset S_i$, $Z$ is an irreducible component of $S_i\cap Y$.
  \item If $Z$ is not equal to any of the $S_i$, then
\end{enumerate}
\[
\label{IV.3.4.1.1}
\operatorname{length}((\sh{F}/t\sh{F})_z)\geqslant\sum_i\operatorname{length}(\sh{F}_{s_i})
\tag{3.4.1.1}
\]
where the sum on the right-hand side is extended to all $i$ such that $Z\subset S_i$.
\begin{enumerate}[resume,label=(\roman*)]
  \item Suppose that $Z$ is not equal to any of the $S_i$.
  For the two sides of \sref{IV.3.4.1.1} to be equal, it is necessary and sufficient that the following two conditions be satisfied:
  \begin{enumerate}[label=$\alpha$)]
    \item[\rm$\alpha$)] $t_z$ is $\sh{F}_z$-regular \sref[0]{0.15.1.4}.
    \item[\rm$\beta$)] For every $i$ such that $Z\subset S_i$, there is a unique $S_i$ containing $Z$, and for this value of $i$, we have $\operatorname{length}(\sh{F}_{s_i})=1$; furthermore $\sh{O}_{S_i,z}$ is a discrete valuation ring whose uniformiser is $t_z$, and $t_z$ is $\sh{F}_z$-regular.
  \end{enumerate}
\end{enumerate}
\end{proposition}

\begin{proof}
(i) As in the proof of \sref{IV.3.4.1}, we can reduce to the case where $X=\operatorname{Spec}(\sh{O}_z)$; the proposition then follows (taking into account \sref{IV.3.1.2}) from \sref[0]{0.16.4.6.3}.

(ii) and (iii) Suppose that $Z$ is not equal to any of the $S_i$.
For the two sides of \sref{IV.3.4.1.1} to be equal, it is necessary and sufficient that the following two conditions be satisfied:
\begin{enumerate}[label=\rm$\alpha$)]
  \item $t_z$ is $\sh{F}_z$-regular \sref[0]{0.15.2.4}.
  \item For every $i$ such that $Z\subset S_i$, the canonical image of the germ $t_z$ in $\sh{O}_{S_i,z}$ generates the maximal ideal of this ring (which implies that $\sh{O}_{S_i,z}$ is a discrete valuation ring and the image of $t_z$ is a uniformiser of this ring).
\end{enumerate}
\end{proof}

\begin{corollary}[3.4.2]
\label{IV.3.4.2}
Under the general hypotheses of \sref{IV.3.4.1}, suppose that $Z$ is not equal to any of the $S_i$ and that $\operatorname{length}((\sh{F}/t\sh{F})_z)=1$.
Then there is a unique $S_i$ containing $Z$, and for this value of $i$, $\operatorname{length}(\sh{F}_{s_i})=1$; furthermore $\sh{O}_{S_i,z}$ is a discrete valuation ring whose uniformiser is $t_z$, and $t_z$ is $\sh{F}_z$-regular.
\end{corollary}

\begin{proof}
This follows from \sref{IV.3.4.1}, the two sides of \sref{IV.3.4.1.1} being then equal.
\end{proof}

\begin{proposition}[3.4.3]
\label{IV.3.4.3}
Let $X$ be a prescheme locally Noetherian, $t$ a section of $\sh{O}_X$ over $X$, $Y$ the closed sub-prescheme of $X$ defined by the Ideal $t\sh{O}_X$ of $\sh{O}_X$.
Let $\sh{F}$ be an $\sh{O}_X$-Module coherent, $T$ a prime cycle associated to $\sh{F}$, $T'$ an irreducible component of $T\cap Y$, $x$ the generic point of $T'$.
Suppose that $t_x$ is $\sh{F}_x$-regular; then $x\in\operatorname{Ass}(\sh{F}/t\sh{F})$.
\end{proposition}

\begin{proof}
As in the proof of \sref{IV.3.4.1}, we can reduce to the case where $X=\operatorname{Spec}(\sh{O}_x)$; the proposition then follows from \sref{IV.3.1.2}, as a consequence of \sref[0]{0.16.4.6.3}.
\end{proof}

\begin{proposition}[3.4.4]
\label{IV.3.4.4}
Let $X$ be a prescheme locally Noetherian, $t$ a section of $\sh{O}_X$ over $X$, $Y$ the closed sub-prescheme of $X$ defined by the Ideal $t\sh{O}_X$ of $\sh{O}_X$.
Let $\sh{F}$ be an $\sh{O}_X$-Module coherent, $(S_i)$ the family of irreducible components of $\operatorname{Supp}(\sh{F})$.
Let $y$ be a point of $Y$ such that $t_y$ is $\sh{F}_y$-regular and such that none of the associated immersed prime cycles of $\sh{F}/t\sh{F}$ contains $y$.
Then the irreducible components of $\operatorname{Supp}(\sh{F}/t\sh{F})$ that contain $y$ are exactly the irreducible components of the $S_i\cap Y$ that contain $y$, and the prime cycles associated to $\sh{F}$ that contain $y$ are non-immersed.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{49}
Let us first prove the last assertion.
Let $T\supset T_1$ be two prime cycles associated to $\sh{F}$ containing $y$; if $x$ is the generic point of an irreducible component of $T\cap Y$ containing $y$, $x$ is the generic point of an irreducible component of $T_1\cap Y$ containing $y$, and the hypothesis that $t_y$ is $\sh{F}_y$-regular implies that $t_x$ is $\sh{F}_x$-regular \sref[0]{0.15.2.4}, hence, by virtue of \sref{IV.3.4.3}, $x\in\operatorname{Ass}(\sh{F}/t\sh{F})$.
Let $x_1$ be the generic point of an irreducible component of $T_1\cap Y$ containing $y$, and let $x$ be the generic point of an irreducible component of $T\cap Y$ containing $x_1$; it follows from the preceding that $x_1$ and $x$ both belong to $\operatorname{Ass}(\sh{F}/t\sh{F})$, and as $x_1\in\overline{\{x\}}$, the hypothesis implies that $x_1=x$.
Let us still denote by $T$ and $T_1$ the integral closed sub-preschemes of $X$ having $T$ and $T_1$ as underlying spaces respectively, and set $A=\sh{O}_{T,x}$, $A_1=\sh{O}_{T_1,x}$; we thus have $A_1=A/\mathfrak{p}$, where $\mathfrak{p}$ is a prime ideal of $A$.
By definition of $x$ and $x_1$, $A/tA$ and $A_1/tA_1$ are Artinian rings; on the other hand, since $x$ cannot belong to $\operatorname{Ass}(\sh{F})$, and since $A_1$ is not Artinian, $x$ cannot belong to $\operatorname{Ass}(\sh{F})$.
We thus have $\dim A=\dim A_1$; but this implies $\mathfrak{p}=0$ and $A=A_1$; as $\operatorname{Spec}(A)$ and $\operatorname{Spec}(A_1)$ are respectively dense in $T$ and $T_1$, we have indeed $T=T_1$.

The $S_i$ that contain $y$ are therefore all the prime cycles associated to $\sh{F}$ containing $y$; if $x$ is the generic point of an irreducible component of $S_i\cap Y$ containing $y$, we deduce again from \sref[0]{0.15.2.4} that $t_x$ is $\sh{F}_x$-regular, hence, by \sref{IV.3.4.3}, $x\in\operatorname{Ass}(\sh{F}/t\sh{F})$; this proves the first assertion \sref{IV.3.4.4}.
\end{proof}

\begin{proposition}[3.4.5]
\label{IV.3.4.5}
Let $X$ be a prescheme locally Noetherian, $t$ a section of $\sh{O}_X$ over $X$, $Y$ the closed sub-prescheme of $X$ defined by the Ideal $t\sh{O}_X$ of $\sh{O}_X$.
Let $\sh{F}$ be an $\sh{O}_X$-Module coherent, $y$ a point of $Y$; suppose that $t_y$ is $\sh{F}_y$-regular and that $\sh{F}/t\sh{F}$ is integral at the point $y$ \sref{IV.3.2.4}.
Then $\sh{F}$ is integral at the point $y$.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{50}
Taking into account \sref{IV.3.4.4}, it suffices to prove that $y$ is contained in a \emph{unique} irreducible component of $\operatorname{Supp}(\sh{F})$, and that if $s$ is the generic point of this component, $\operatorname{length}(\sh{F}_s)=1$.
Now, by hypothesis, $y$ belongs to at most a unique irreducible component of $\operatorname{Supp}(\sh{F}/t\sh{F})$, and if $z$ is the generic point of this component, $\operatorname{length}((\sh{F}/t\sh{F})_z)=1$; the conclusion therefore follows from \sref{IV.3.4.2}.
\end{proof}

\begin{proposition}[3.4.6]
\label{IV.3.4.6}
The hypotheses being those of \sref{IV.3.4.1}, let $x$ be a point of $Y$.
Suppose that $Y$ does not contain any of the $S_i$ containing $x$, and that $\sh{F}/t\sh{F}$ is reduced at the point $x$ \sref{IV.3.2.2}.
Then $t_x$ is $\sh{F}_x$-regular and $\sh{F}$ is reduced at the point $x$.
Furthermore, if $z_j$ is the generic point of an irreducible component of $\operatorname{Supp}(\sh{F}/t\sh{F})$ containing $x$, $z_j$ is contained in a unique $S_i$, and $\sh{O}_{S_i,z_j}$ is a discrete valuation ring whose uniformiser is $t_{z_j}$.
\end{proposition}

\begin{proof}
The fact that $t_x$ is $\sh{F}_x$-regular follows from the lemma below applied to the ring $\sh{O}_x$:
\end{proof}

\begin{lemma}[3.4.6.1]
\label{IV.3.4.6.1}
Let $A$ be a Noetherian ring, $M$ a finitely generated $A$-module, $\mathfrak{p}_i$ the minimal elements of $\operatorname{Supp}(M)$, $t$ an element of $A$.
Suppose that $t$ does not belong to any of the $\mathfrak{p}_i$ and that $M/tM$ is a reduced $A$-module \sref{IV.3.2.2}.
Then $t$ is $M$-regular.
\end{lemma}

\begin{proof}
Every prime ideal $\mathfrak{p}\in\operatorname{Supp}(M)$ contains one of the $\mathfrak{p}_i$; as $t$ does not belong to any of the $\mathfrak{p}_i$, the homothethy of ratio $t$ in $M_\mathfrak{p}$ is not nilpotent (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \textsection~1, no~4, cor.\ of prop.~9).
Let $N$ be the sub-module of $M$ consisting of elements annihilated by a power of $t$, and set $P=M/N$; we are going to show that $N=0$.
As $t$ is $P$-regular, we have an exact sequence \sref{IV.3.4.1.4}
\[
0\to N/tN\to M/tM\to P/tP\to 0.
\]
As $N$ is of finite type, it is annihilated by a power of $t$, and it suffices therefore to show that $N/tN=0$.
As $N/tN$ is a sub-module of $M/tM$, it suffices to prove that $(N/tN)_\mathfrak{p}=0$ for all $\mathfrak{p}\in\operatorname{Ass}(M/tM)$, or equivalently that $u_\mathfrak{p}:(M/tM)_\mathfrak{p}\to(P/tP)_\mathfrak{p}$ is bijective for all $\mathfrak{p}\in\operatorname{Ass}(M/tM)$.
Now, we have $(P/tP)_\mathfrak{p}\neq 0$; indeed, as $\mathfrak{p}\in\operatorname{Supp}(M/tM)=\operatorname{Supp}(M)\cap V(t)$ \sref[0]{0.1.7.5}, the image of $t$ in $A_\mathfrak{p}$ is contained in the maximal ideal $\mathfrak{p} A_\mathfrak{p}$, hence the hypothesis $P_\mathfrak{p}=tP_\mathfrak{p}$ would imply $P_\mathfrak{p}=0$ by the lemma of Nakayama; we would then have $M_\mathfrak{p}=N_\mathfrak{p}$ and the homothethy of ratio $t$ would be nilpotent, contradicting the remark made at the beginning, since $\mathfrak{p}\in\operatorname{Supp}(M)$.
This being so, the hypothesis that $M/tM$ is reduced implies $\operatorname{length}((M/tM)_\mathfrak{p})=1$, and as $(P/tP)_\mathfrak{p}\neq 0$, $u_\mathfrak{p}$ is necessarily bijective, which proves the lemma.
\end{proof}

By hypothesis, none of the associated immersed prime cycles of $\sh{F}/t\sh{F}$ contains $x$, hence none of the associated immersed prime cycles of $\sh{F}$ contains $x$, by virtue of \sref{IV.3.4.4}.
On the other hand, applying \sref{IV.3.4.2} to an irreducible component of $\operatorname{Supp}(\sh{F}/t\sh{F})$ containing $x$, we see that $\operatorname{length}(\sh{F}_{s_i})=1$ for every $S_i$ containing $x$, which shows that $\sh{F}_x$ is reduced; finally, the last assertions are also consequences of \sref{IV.3.4.2}.

\begin{corollary}[3.4.7]
\label{IV.3.4.7}
Let $A$ be a local Noetherian ring, $\mathfrak{m}$ its maximal ideal, $M$ a finitely generated $A$-module, $(x_i)_{1\leqslant i\leqslant k}$ a family of elements of $\mathfrak{m}$ forming part of a system of parameters \sref[0]{0.16.3.6}.
\oldpage[IV-2]{51}
If the $A$-module $N=M/(\sum_{i=1}^k x_i M)$ is integral \sref{IV.3.2.4}, then $M$ is integral and the sequence $(x_i)_{1\leqslant i\leqslant k}$ is $M$-regular.
\end{corollary}

\begin{proof}
By induction on $k$, we are immediately reduced to the case $k=1$; we write $x$ instead of $x_1$; the hypothesis that $x$ forms part of a system of parameters for $M$ implies that $\dim(N)=\dim(M)-1$ \sref[0]{0.16.3.7}.
Set $n=\dim(N)$; there is thus a unique minimal element $\mathfrak{p}$ of $\operatorname{Supp}(M)$ such that $\dim(M/\mathfrak{p} M)=n+1$ \sref[0]{0.16.3.4}, and for every integer $j>0$, also $\dim(M/\mathfrak{p}^j M)=n+1$ \sref[0]{0.16.3.5}; furthermore, $x$ also forms part of a system of parameters for $M/\mathfrak{p}^j M$ \sref[0]{0.16.3.5}, hence, if one sets $M'=M/\mathfrak{p}^j M$ and $N'=M'/xM'$, we have $\dim(N')=n$.
It is clear that there exists a surjective homomorphism $v:N\to N'$; let us show that $v$ is \emph{bijective}.
Indeed, if $P=\operatorname{Ker}(v)$, we have $\operatorname{Ass}(P)\subset\operatorname{Ass}(N)$, and since $N$ is integral, $\operatorname{Ass}(P)$ and $\operatorname{Ass}(N)$ would both be reduced to the unique point $\mathfrak{q}$ of $\operatorname{Ass}(N)=\operatorname{Supp}(N)$; but as $\dim(N')=\dim(N)$, we have $N'_\mathfrak{q}\neq 0$, and the hypothesis $\operatorname{length}(N'_\mathfrak{q})=1$ implies that $N_\mathfrak{q}\to N'_\mathfrak{q}$ is surjective.
We would then have $P_\mathfrak{q}=0$, contradicting our assumption; hence our assertion.
But then $N'$, being isomorphic to $N$, is integral; furthermore, the support of $N'$ (equal to the intersection of $\operatorname{Supp}(M)$ and $V(x)$) cannot contain $\operatorname{Supp}(M')$, and this last set is irreducible by construction.
The hypothesis that $M'/xM'$ is integral (hence reduced) implies then that $x$ is $M'$-regular by virtue of \sref{IV.3.4.6}.
We conclude that the kernel of the homothethy $z\mapsto xz$ in $M$ is contained in $\mathfrak{p}^j M\subset\mathfrak{m}^j M$ for every integer $j$, and hence this kernel is reduced to $0$ \sref[0]{0.7.3.5}, which proves that $x$ is $M$-regular.
We can then apply \sref{IV.3.4.5}, which proves that $M$ is integral.
\end{proof}

\begin{remark}[3.4.8]
\label{IV.3.4.8}
The analogous proposition to \sref{IV.3.4.7}, where one replaces ``integral'' by ``reduced'', is no longer necessarily exact.
Consider for example the polynomial ring $C=K[X,Y,Z]$ over a field $K$, the quotient ring $B=C/\mathfrak{p}\mathfrak{q}$, where $\mathfrak{p}=CZ$, $\mathfrak{q}=CX^2+CY$; let $A$ be the local ring of $B$ corresponding to the maximal image of $CX+CY+CZ$ in $B$.
If $x$, $z$ are the canonical images of $X$, $Z$ in $B$, it is clear that $xz\neq 0$ but $x^2z^2=0$; on the other hand, as $A/xA$ is isomorphic to $K[Y,Z]/(YZ)$, we have $\dim(A/xA)=1$ while $\dim(A)=2$, hence $x$ belongs to a system of parameters of $A$ \sref[0]{0.16.3.4}, $A/xA$ is reduced, but $A$ is not.
\end{remark}

\begin{proposition}[3.4.9]
\label{IV.3.4.9}
Let $A$ be a Noetherian ring, $M$ a finitely generated $A$-module, $f$ an $M$-regular element such that $M/fM$ has no immersed associated prime ideals.
If the $\mathfrak{p}_i$ $(1\leqslant i\leqslant m)$ are the prime ideals associated to $M/fM$, then, for every integer $n>0$, $f^n M$ is the intersection of the inverse images of the $f^n M_{\mathfrak{p}_i}$ by the canonical maps $M\to M_{\mathfrak{p}_i}$ $(1\leqslant i\leqslant m)$.
\end{proposition}

\begin{proof}
It amounts to the same to show that the $\mathfrak{p}_i$ are the prime ideals associated to $M/f^n M$, for then the $\mathfrak{p}_i$ are the sub-modules of the unique reduced primary decomposition of $f^n M$ in $M$.
Now, we have
\[
\operatorname{Ass}(f^{n-1}M/f^n M)\subset\operatorname{Ass}(M/f^n M)\subset\operatorname{Ass}(M/f^{n-1}M)\cup\operatorname{Ass}(f^{n-1}M/f^n M)
\]
by \sref{IV.3.1.7}, and as $f$ is $M$-regular, $f^{n-1}M/f^n M$ is isomorphic to $M/fM$; it suffices therefore to reason by induction on $n$.
\end{proof}
