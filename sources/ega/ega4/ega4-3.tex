\section{Associated prime cycles and primary decomposition}
\label{section:IV.3}

\oldpage[IV-2]{36}

We give in this section primarily the translation of results on modules presented in Bourbaki, \emph{Alg.~comm.}, chap.~IV, which we follow very closely.
The notions that follow seem scarcely to be of interest except in the case of locally Noetherian preschemes.

\subsection{Associated prime cycles of a Module}
\label{subsection:IV.3.1}

\begin{definition}[3.1.1]
\label{IV.3.1.1}
Let $X$ be a prescheme, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
We say that a point $x\in X$ is \emph{associated} to $\sh{F}$ if the maximal ideal $\mathfrak{m}_x$ of $\sh{O}_x$ is associated to the $\sh{O}_x$-Module $\sh{F}_x$ (in other words, if $\mathfrak{m}_x$ is the annihilator of an element of $\sh{F}_x$).
We denote by $\operatorname{Ass}(\sh{F})$ the set of $x\in X$ associated to $\sh{F}$.
We say that an irreducible closed subset $Z$ of $X$ is an \emph{associated prime cycle} of $\sh{F}$ if its generic point is associated to $\sh{F}$.
When $\sh{F}=\sh{O}_X$, we also say that the points (resp.\ prime cycles) associated to $\sh{F}$ are \emph{associated to the prescheme} $X$.

We say that an associated prime cycle of $\sh{F}$ (resp.\ $X$) is \emph{immersed} if it is contained in another associated prime cycle of $\sh{F}$ (resp.\ $X$) (in other words, if it is not \emph{maximal} among these cycles).

If $X$ is locally Noetherian, the \emph{irreducible components} of $X$ are none other than the maximal (or \emph{non-immersed}) prime cycles associated to $X$.

It is clear that if $x\in\operatorname{Ass}(\sh{F})$, then $\sh{F}_x\neq 0$, in other words
\end{definition}

\begin{equation}
\label{IV.3.1.1.1}
  \operatorname{Ass}(\sh{F})\subset\Supp(\sh{F}).\tag{3.1.1.1}
\end{equation}

If $x\in X$ is associated to $\sh{F}$, it is obviously associated to $\sh{F}|U$ for every open neighbourhood $U$ of $x$, and conversely, if it is associated to $\sh{F}|U$ for one of these neighbourhoods, it is associated to $\sh{F}$.

We note finally that for a quasi-coherent $\sh{O}_X$-Module $\sh{F}$, the existence of immersed associated prime cycles is a \emph{local} question, since if $y$ and $z$ are two points of $\operatorname{Ass}(\sh{F})$ such that $y\in\overline{\{z\}}$, every neighbourhood of $y$ contains $z$.

\begin{proposition}[3.1.2]
\label{IV.3.1.2}
Let $A$ be a Noetherian ring, $M$ an $A$-module, $X=\Spec(A)$, $\sh{F}=\widetilde{M}$; for a point $x\in X$ to be associated to $\sh{F}$, it is necessary and sufficient that the prime ideal $\mathfrak{j}_x$ of $A$ be associated to the module $M$ \emph{(}in other words, be the annihilator of an element $f\in M$\emph{)}.
\end{proposition}

\begin{proof}
This follows from the definition \sref{IV.3.1.1} and from Bourbaki, \emph{loc.~cit.}, \textsection1, no.~2, cor.\ of prop.~5, applied to $S=A-\mathfrak{j}_x$.
\end{proof}

\begin{proposition}[3.1.3]
\label{IV.3.1.3}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module, $x$ a point of $X$, $Z$ the reduced closed subprescheme of $X$ having $\{x\}$ as underlying space \sref[I]{I.5.2.1}.
The following conditions are equivalent:
\begin{enumerate}[a)]
  \item $x\in\operatorname{Ass}(\sh{F})$.
  \item There exists an open neighbourhood $U$ of $x$ and a section $f\in\Gamma(U,\sh{F})$ such that $U\cap Z$ is equal to $\Supp(\sh{O}_U\cdot f)$.
  \item[\emph{b')}] There exists an open neighbourhood $U$ of $x$ and a section $f\in\Gamma(U,\sh{F})$ such that $U\cap Z$ is an irreducible component of $\Supp(\sh{O}_U\cdot f)$.
  \item There exists an open neighbourhood $U$ of $x$ and a sub-Module of $\sh{F}|U$ isomorphic to $\sh{O}_Z|U$ \emph{(}$\sh{O}_Z$ being identified with a quotient of $\sh{O}_X$\emph{)}.
  \item[\emph{c')}] There exists an open neighbourhood $U$ of $x$ and a coherent sub-Module $\sh{G}$ of $\sh{F}|U$ such that $U\cap Z$ is an irreducible component of $\Supp(\sh{G})$.
\end{enumerate}
\end{proposition}

\begin{proof}
It is clear that c) implies b), for it suffices to take for $f$ the element of $\Gamma(U,\sh{F})$ corresponding to the unit section of $\sh{O}_Z|U$.
Since $U\cap Z$ is irreducible $(\mathbf{0_I}$, 2.1.6), b) implies b') and b') implies c') since $\sh{O}_X$ is coherent $(\mathbf{0_I}$, 5.3.4).
To see that c') implies a), we can restrict to the case $U=X=\Spec(A)$ is affine, $A$ being therefore Noetherian, and where $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module, and $\sh{G}=\widetilde{N}$, where $N$ is a submodule of $M$, of finite type.
We know then that the minimal elements of $\Supp(\sh{G})$ are the maximal points of $V(\operatorname{Ann}(N))$ $(\mathbf{0_I}$, 1.7.4), and these are also the minimal elements of $\operatorname{Ass}(N)$ (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, no.~3, cor.~1 of prop.~7); since $\operatorname{Ass}(N)\subset\operatorname{Ass}(M)=\operatorname{Ass}(\sh{F})$, we see that c') implies a).
Finally a) implies c) by virtue of \sref{IV.3.1.2}, taking again $X$ affine, $\sh{F}=\widetilde{M}$, and $Z$ defined by the ideal $fA$ (with the notations of \sref{IV.3.1.2}).
\end{proof}

\oldpage[IV-2]{37}

\begin{corollary}[3.1.4]
\label{IV.3.1.4}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module.
Then the maximal associated prime cycles of $\sh{F}$ are the irreducible components of $\Supp(\sh{F})$, and the generic points of these components are the points $x\in X$ such that $\sh{F}_x$ is an $\sh{O}_x$-Module of finite length and $\neq 0$.
\end{corollary}

\begin{proof}
Indeed, if $x$ is the generic point of one of the irreducible components $Z$ of $\Supp(\sh{F})$, it follows from the equivalence of a) and c') in \sref{IV.3.1.3} that $x$ belongs to $\operatorname{Ass}(\sh{F})$, and $Z$ is an associated prime cycle of $\sh{F}$, necessarily maximal by virtue of \sref{IV.3.1.1.1}; the converse follows trivially from \sref{IV.3.1.1.1}; finally, the last assertion being obviously local follows from Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection2, no.~5, cor.~2 of prop.~7.
\end{proof}

\begin{corollary}[3.1.5]
\label{IV.3.1.5}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
For $\sh{F}=0$, it is necessary and sufficient that $\operatorname{Ass}(\sh{F})=\emp$.
\end{corollary}

\begin{proof}
The question being local, we are reduced to the case where $X$ is affine, and the conclusion follows immediately from \sref{IV.3.1.2} and from Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, no.~1, cor.~1 of prop.~2.
\end{proof}

\begin{proposition}[3.1.6]
\label{IV.3.1.6}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module; then $\operatorname{Ass}(\sh{F})$ is locally finite \emph{(}that is to say, every point of $X$ admits a neighbourhood whose intersection with $\operatorname{Ass}(\sh{F})$ is finite\emph{)}.
\end{proposition}

\begin{proof}
It suffices to consider the case where $X$ is affine, hence Noetherian, and then the proposition follows from \sref{IV.3.1.2} and from Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, no.~4, cor.\ of th.~2.
\end{proof}

\begin{proposition}[3.1.7]
\label{IV.3.1.7}
Let $X$ be a prescheme.
\begin{enumerate}[\rm(i)]
  \item Let $0\to\sh{F}'\to\sh{F}\to\sh{F}''\to 0$ be an exact sequence of quasi-coherent $\sh{O}_X$-Modules.
  Then $\operatorname{Ass}(\sh{F}')\subset\operatorname{Ass}(\sh{F})\subset\operatorname{Ass}(\sh{F}')\cup\operatorname{Ass}(\sh{F}'')$.
  \item Let $\sh{F}$ be a quasi-coherent $\sh{O}_X$-Module, $(\sh{F}_\alpha)$ a family of quasi-coherent sub-$\sh{O}_X$-Modules of $\sh{F}$ such that $\sh{F}$ is the union of the $\sh{F}_\alpha$.
  Then $\operatorname{Ass}(\sh{F})=\bigcup_\alpha\operatorname{Ass}(\sh{F}_\alpha)$.
  \item For every family $(\sh{F}_\alpha)$ of quasi-coherent $\sh{O}_X$-Modules, we have $\operatorname{Ass}(\bigoplus\sh{F}_\alpha)=\bigcup\operatorname{Ass}(\sh{F}_\alpha)$.
\end{enumerate}
\end{proposition}

\begin{proof}
\oldpage[IV-2]{38}
We are immediately reduced to the corresponding propositions for modules (Bourbaki, \emph{loc.~cit.}, \textsection1, no.~1, formula~(1), prop.~3 and cor.~1 of prop.~3).
\end{proof}

\begin{proposition}[3.1.8]
\label{IV.3.1.8}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module, $U$ an open subset of $X$, $\sh{J}$ a coherent Ideal of $\sh{O}_X$ defining a closed subprescheme of $X$ having $X-U$ as underlying space.
The following conditions are equivalent:
\begin{enumerate}[a)]
  \item $\operatorname{Ass}(\sh{F})\subset U$.
  \item For every affine open $V$ of $X$, every section of $\sh{F}$ over $V$ whose restriction to $V\cap U$ is zero, is equal to $0$.
  \item The canonical homomorphism $\sh{F}\to\shHom_{\sh{O}_X}(\sh{J},\sh{F})$ is injective.
\end{enumerate}
\end{proposition}

\begin{proof}
The question being local, we can suppose $X=\Spec(A)$ affine, $A$ being a Noetherian ring, $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module, and $\sh{J}=\widetilde{\mathfrak{J}}$, where $\mathfrak{J}$ is an ideal of $A$.
The homomorphism $M\to\Hom_A(\mathfrak{J},M)$ sends $m\in M$ to the homomorphism $x\mapsto xm$ from $\mathfrak{J}$ to $M$; to say that it is not injective means that there exists an $m\neq 0$ in $M$ such that $\mathfrak{J}m=0$.

Let us first show that c) implies a): indeed, if $\operatorname{Ass}(\sh{F})$ met $X-U$, there would exist a prime ideal $\mathfrak{p}\in\operatorname{Ass}(M)$ containing $\mathfrak{J}$, hence an element $m\neq 0$ of $M$ such that $\mathfrak{J}m=0$.
Secondly, b) implies c): indeed, if $\mathfrak{J}m=0$ for some $m\neq 0$ in $M$, for every prime ideal $\mathfrak{q}\not\supset\mathfrak{J}$, there exists an $a\in\mathfrak{J}$ not in $\mathfrak{q}$, hence the relation $am=0$ implies that the canonical image of $m$ in $M_\mathfrak{q}$ is $0$, in other words $m$ would be a section $\neq 0$ of $\sh{F}$ over $X$ whose restriction to $U$ would be zero.
Finally a) implies b).
To see this, note that the canonical homomorphism $M\to\prod_{\mathfrak{p}\in\operatorname{Ass}(M)}M_\mathfrak{p}$ is \emph{injective}: indeed, if $N$ is the kernel of this homomorphism, we have $\operatorname{Ass}(N)\subset\operatorname{Ass}(M)$; if there existed a $\mathfrak{p}\in\operatorname{Ass}(N)$, there would be an element $n\in N$ whose $\mathfrak{p}$ would be the annihilator; but by definition of $N$, there is an element $s\notin\mathfrak{p}$ such that $sn=0$, which is absurd; we conclude that $\operatorname{Ass}(N)=\emp$, whence $N=0$.
Now condition a) implies that if $m\in M$ is a section of $\sh{F}$ whose restriction to $U$ is zero, the canonical image of $m$ in $M_\mathfrak{p}$ is zero for every $\mathfrak{p}\in\operatorname{Ass}(M)$, hence $m=0$.
\end{proof}

\begin{corollary}[3.1.9]
\label{IV.3.1.9}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module, $f$ a section of $\sh{O}_X$ over $X$.
For $f$ to be $\sh{F}$-regular $(\mathbf{0_{IV}}$, 15.2.2), it is necessary and sufficient that $\operatorname{Ass}(\sh{F})\subset X_f$.
\end{corollary}

\begin{proof}
Indeed, it is immediate that to say that $f$ is $\sh{F}$-regular means that the canonical homomorphism $\sh{F}\to\shHom_{\sh{O}_X}(f\sh{O}_X,\sh{F})$ is injective, and it suffices to apply \sref{IV.3.1.8} to the Ideal $\sh{J}=f\sh{O}_X$.
\end{proof}

\begin{proposition}[3.1.10]
\label{IV.3.1.10}
\oldpage[IV-2]{39}
Let $X$, $Y$ be locally Noetherian preschemes, $f:X\to Y$ an integral morphism.
Then, for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, we have $f(\operatorname{Ass}(\sh{F}))=\operatorname{Ass}(f_*(\sh{F}))$.
\end{proposition}

\begin{proof}
The question being local on $Y$ and the morphism $f$ being affine, we are immediately reduced to the case where $Y$ is affine, in other words to the following:
\end{proof}

\begin{lemma}[3.1.10.1]
\label{IV.3.1.10.1}
Let $A$, $B$ be two Noetherian rings, $\rho:A\to B$ a ring homomorphism making $B$ an integral $A$-algebra, $M$ a $B$-module.
Then the prime ideals $\mathfrak{p}\in\operatorname{Ass}(M_{[\rho]})$ are the inverse images by $\rho$ of the prime ideals $\mathfrak{q}\in\operatorname{Ass}(M)$.
\end{lemma}

\begin{proof}
Indeed, if $\mathfrak{q}\in\operatorname{Ass}(M)$, $\mathfrak{q}$ is the annihilator in $B$ of an element $x\in M$, hence $\rho^{-1}(\mathfrak{q})$ is the annihilator in $A$ of $x$.
Conversely, let $\mathfrak{p}\in\operatorname{Ass}(M_{[\rho]})$, so that $\mathfrak{p}$ is the inverse image by $\rho$ of the annihilator $\mathfrak{b}$ in $B$ of an element $x\in M$; it follows from the first theorem of Cohen-Seidenberg that there exists a prime ideal $\mathfrak{q}$ of $B$ containing $\mathfrak{b}$ and whose inverse image is $\mathfrak{p}$ (Bourbaki, \emph{Alg.~comm.}, chap.~V, \textsection2, no.~1, cor.~2 of th.~1); by considering one of the minimal prime ideals among those contained in $\mathfrak{q}$ and containing $\mathfrak{b}$, we can obviously suppose that $\mathfrak{q}$ itself is one of these minimal ideals.
But since $B\cdot x\subset M$ is isomorphic to $B/\mathfrak{b}$, we know that $\mathfrak{q}\in\operatorname{Ass}(B/\mathfrak{b})\subset\operatorname{Ass}(M)$ (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, no.~4, th.~2).
\end{proof}

\begin{corollary}[3.1.11]
\label{IV.3.1.11}
Under the hypotheses of \sref{IV.3.1.10}, for $\sh{F}$ to be without immersed associated prime cycle, it suffices that the same be true of $f_*(\sh{F})$.
\end{corollary}

\begin{proof}
Suppose indeed that $f_*(\sh{F})$ has no immersed associated prime cycle.
Note that if $A$ is an integral algebra over a field $k$, all the prime ideals of $A$ are maximal (Bourbaki, \emph{Alg.~comm.}, chap.~V, \textsection2, no.~1, prop.~1); it therefore follows from \sref[I]{I.6.2.2} that the fibres of $f$ are \emph{discrete} spaces; if $x$, $x'$ are two distinct points of $\operatorname{Ass}(\sh{F})$, neither of them can be adherent to the other if $f(x)=f(x')$; and if $f(x)\neq f(x')$, \sref{IV.3.1.10} and the hypothesis imply that neither of the two points $f(x)$, $f(x')$ can be adherent to the other, hence the same is true of $x$ and $x'$.
\end{proof}

\begin{remark}[3.1.12]
\label{IV.3.1.12}
Under the hypotheses of \sref{IV.3.1.10}, it can on the contrary happen that $\sh{F}$ be without immersed associated prime cycle, but not $f_*(\sh{F})$.
Take for example $Y=\Spec(k[\mathrm{T}])$ where $k$ is a field (``affine line''), and $X$ the sum of $X_1=Y$ and $X_2=\Spec(k)$, the morphism $X_2\to Y$ corresponding to the canonical homomorphism $k[\mathrm{T}]\to k[\mathrm{T}]/\mathfrak{m}$, where $\mathfrak{m}$ is the maximal ideal $(\mathrm{T})$.
It is clear that the morphism $f:X\to Y$ is finite; if one takes $\sh{F}=\sh{O}_X$, $\sh{F}$ is without immersed associated prime cycle, but $f_*(\sh{F})=\widetilde{M}$, where $M$ is the $k[\mathrm{T}]$-module direct sum of $k[\mathrm{T}]$ and $k$, so $\operatorname{Ass}(M)$ is formed by the generic point $(0)$ of $Y$ and the point $\mathfrak{m}$.
\end{remark}

\begin{proposition}[3.1.13]
\label{IV.3.1.13}
Let $X$ be a locally Noetherian prescheme, $U$ an open subset of $X$, $i:U\to X$ the canonical injection.
For every quasi-coherent $\sh{O}_U$-Module $\sh{F}$, we have $\operatorname{Ass}(i_*(\sh{F}))=\operatorname{Ass}(\sh{F})$.
\end{proposition}

\begin{proof}
Recall that $i_*(\sh{F})$ is a quasi-coherent $\sh{O}_X$-Module \sref{IV.1.2.2} and \sref{IV.1.7.4}; since $i_*(\sh{F})|U=\sh{F}$, we have $(\operatorname{Ass}(i_*(\sh{F})))\cap U=\operatorname{Ass}(\sh{F})$, and it therefore remains to prove that $\operatorname{Ass}(i_*(\sh{F}))\subset U$.
But by \sref{IV.3.1.8}, this relation means that for every affine open $V$ of $X$, every section of $i_*(\sh{F})$ over $V$ which is zero in $U\cap V$, is zero, a condition trivially satisfied since $\Gamma(V,i_*(\sh{F}))=\Gamma(U\cap V,\sh{F})=\Gamma(U\cap V,i_*(\sh{F}))$.
\end{proof}

\subsection{Irredundant decompositions}
\label{subsection:IV.3.2}

\oldpage[IV-2]{40}

\begin{proposition}[3.2.1]
\label{IV.3.2.1}
Let $X$ be a locally Noetherian prescheme, $U$ an open dense subset of $X$.
The following conditions are equivalent:
\begin{enumerate}[a)]
  \item $X$ is reduced.
  \item The sub-prescheme induced on $U$ is reduced and $X$ is without immersed prime cycle.
  \item $X$ is without immersed prime cycle and for every generic point $x$ of an irreducible component of $X$, we have $\operatorname{long}(\sh{O}_x)=1$.
\end{enumerate}
The prime cycles associated to $X$ are then identical to the irreducible components of $X$.
\end{proposition}

\begin{proof}
It is clear that if $X$ is reduced it is the same for the sub-prescheme induced on $U$.
Moreover, the existence of immersed prime cycles being a local question, we can restrict to the case $X=\Spec(A)$ is affine, $A$ being Noetherian.
If $A$ is reduced, we know that the minimal prime ideals of $A$ form a reduced primary decomposition of $(0)$ (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection2, no.~5, prop.~10) and are the elements of $\operatorname{Ass}(A)$, so there are no immersed associated prime ideals of $A$, which shows that $a)$ implies $b)$.
It is immediate that $b)$ implies $c)$, since a generic point $x$ of an irreducible component of $X$ belongs to $U$, hence $\sh{O}_x$ is a field.
Finally, $c)$ implies $a)$: it suffices to remark that if $\sh{N}$ is the Nilradical of $\sh{O}_X$, which is a coherent Ideal, $\Supp(\sh{N})$ cannot contain any of the generic points of the irreducible components of $X$ by hypothesis; if $\Supp(\sh{N})$ were not empty and $x$ was one of the maximal points of this closed set, the criterion \sref{IV.3.1.3}, c') would show that $x\in\operatorname{Ass}(\sh{O}_X)$, and $\{x\}$ would be an immersed prime cycle of $X$, contradicting the hypothesis; hence $\sh{N}=0$.
\end{proof}

We say that a locally Noetherian prescheme $X$ is \emph{irredundant} if $\sh{O}_X$ is irredundant (which implies that $X$ is irreducible); for $X$ to be \emph{integral}, it is necessary and sufficient that the $\sh{O}_X$-Module $\sh{O}_X$ be integral \sref[I]{I.2.1.8} and \sref{IV.3.2.1}.
If $\sh{O}_X$ is integral, $\sh{O}_X$ is integral at every point $x$, that is to say $\sh{O}_x$ is an integral ring.
We say that a closed subprescheme $Y$ of $X$ is \emph{primary} in $X$ if the Ideal $\sh{J}$ of $\sh{O}_X$ defining $Y$ is primary in $\sh{O}_X$.

\begin{definition}[3.2.2]
\label{IV.3.2.2}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module.
We say that $\sh{F}$ is \emph{reduced} if it satisfies the two following conditions: $1^\circ$ $\sh{F}$ is without immersed associated prime cycle; $2^\circ$ for every maximal point $x$ of $\Supp(\sh{F})$, we have $\operatorname{long}(\sh{F}_x)=1$.

Condition $1^\circ$ means that the associated prime cycles of $\sh{F}$ are the irreducible components of $\Supp(\sh{F})$ \sref{IV.3.1.4}, and condition $2^\circ$ means that for every generic point $x$ of such a component we have $\operatorname{long}(\sh{F}_x)=1$.

For an affine scheme $X$, this definition gives in particular the notion of \emph{reduced module} over a Noetherian ring $A$; an $A$-module $M$ of finite type is said \emph{reduced} if it has no immersed associated prime ideals and if, for every $\mathfrak{p}\in\operatorname{Ass}(M)$, $\operatorname{long}_{A_\mathfrak{p}}(M_\mathfrak{p})=1$.
Returning to the case of a locally Noetherian prescheme $X$ and a coherent $\sh{O}_X$-Module $\sh{F}$, we say that $\sh{F}$ is \emph{reduced at a point} $x\in X$ if $\sh{F}_x$ is a reduced $\sh{O}_x$-module; that means that, on the local scheme $\Spec(\sh{O}_x)$, $\widetilde{\sh{F}_x}$ is reduced; it amounts therefore to saying that $x$ does not belong to any immersed associated prime cycle of $\sh{F}$ and that $\operatorname{long}(\sh{F}_z)=1$ for all maximal points $z$ of $\Supp(\sh{F})$ such that $x\in\overline{\{z\}}$.
It is clear that if $\sh{F}$ is a coherent $\sh{O}_X$-Module that is reduced, it is reduced at every point of $X$; conversely, if $\sh{F}$ is \emph{reduced at a point} $x$, there exists an open neighbourhood $U$ of $x$ such that $\sh{F}|U$ is a reduced $\sh{O}_U$-Module: it suffices in fact to take $U$ not meeting any immersed associated prime cycle of $\sh{F}$ (such a neighbourhood exists since these cycles form a locally finite set of closed subsets of $X$).
To say that $\sh{O}_X$ is reduced at a point $x$ is equivalent to saying that $X$ is \emph{reduced} at the point $x$.
\end{definition}

\oldpage[IV-2]{41}

\begin{proposition}[3.2.3]
\label{IV.3.2.3}
Let $X$ be a locally Noetherian prescheme, $U$ an open subset of $X$, $\sh{F}$ a coherent $\sh{O}_X$-Module such that $U\cap\Supp(\sh{F})$ is dense in $\Supp(\sh{F})$.
The following conditions are equivalent:
\begin{enumerate}[a)]
  \item $\sh{F}$ is reduced.
  \item $\sh{F}|U$ is reduced, and $\sh{F}$ is without immersed associated prime cycle.
  \item There exists a reduced closed subprescheme $X'$ of $X$ and a coherent $\sh{O}_{X'}$-Module $\sh{F}'$ without torsion and of rank~$1$ on every irreducible component of $X'$, such that if $j:X'\to X$ is the canonical injection, we have $j_*(\sh{F}')=\sh{F}$.
\end{enumerate}
Moreover, when this holds, the subprescheme $X'$ is defined by the Ideal $\sh{J}$ of $\sh{O}_X$ which is the annihilator of $\sh{F}$.
\end{proposition}

\begin{proof}
To see that $c)$ implies $a)$, we can, by virtue of \sref{IV.3.1.3}, limit to the case where $X'=X$ is integral, with generic point $x$ and one can further suppose $X=\Spec(A)$ affine and $\sh{F}=\widetilde{M}$, where $A$ is integral and $M$ is an $A$-module without torsion of rank~$1$; the annihilator of every element of $M$ being then reduced to $0$, we have $\operatorname{Ass}(\sh{F})=\{x\}$ and $\sh{F}_x$ is isomorphic to the field of fractions of $A$, so the conditions of the definition \sref{IV.3.2.2} are satisfied.
It is clear that if $\sh{F}$ has no immersed associated prime cycles and if $U\cap\Supp(\sh{F})$ is dense in $\Supp(\sh{F})$, then $\operatorname{Ass}(\sh{F})=\operatorname{Ass}(\sh{F}|U)$, hence $a)$ and $b)$ are equivalent.
If $a)$ is satisfied, take for $X'$ the reduced closed subprescheme of $X$ whose $\Supp(\sh{F})$ is the underlying space \sref[I]{I.5.2.1}, and let $\sh{F}'=j^*(\sh{F})$; a point $x$ of $\operatorname{Ass}(\sh{F})$ is necessarily a maximal point of $X'$, and as $\operatorname{long}(\sh{F}_x)=1$, $\sh{F}'_x$ is isomorphic to $\sh{F}_x$, hence to the field $k(x)=\sh{O}_{X',x}$, which proves that $\sh{F}'$ is without torsion and of rank~$1$ \sref[I]{I.7.4.6} and \sref[I]{I.7.4.2}.
Finally, the last assertion is trivial, since for every $y\in X'$, the annihilator of the $\sh{O}_{X',y}$-Module $\sh{F}'_y$ is zero.
\end{proof}

\begin{definition}[3.2.4]
\label{IV.3.2.4}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module.
We say that $\sh{F}$ is \emph{irredundant} if $\operatorname{Ass}(\sh{F})$ is reduced to a single element; if $\sh{F}$ is of finite type, we say that $\sh{F}$ is \emph{integral} if moreover $\sh{F}$ is reduced (in other words if $\operatorname{long}(\sh{F}_x)=1$).
We say that a quasi-coherent sub-$\sh{O}_X$-Module $\sh{G}$ of a quasi-coherent $\sh{O}_X$-Module $\sh{F}$ is \emph{primary} in $\sh{F}$ if $\sh{F}/\sh{G}$ is irredundant.

For an affine scheme $X$, this definition gives in particular the notion of \emph{integral module} over a Noetherian ring $A$; an $A$-module $M$ of finite type is said \emph{integral} if $\operatorname{Ass}(M)$ is reduced to a single prime ideal $\mathfrak{p}$, and if moreover $\operatorname{long}_{A_\mathfrak{p}}(M_\mathfrak{p})=1$.
Returning to the case of a locally Noetherian prescheme $X$ and a coherent $\sh{O}_X$-Module $\sh{F}$, we say that $\sh{F}$ is \emph{integral at a point} $x\in X$ if $\sh{F}_x$ is an \emph{integral $\sh{O}_x$-module}: that means that $x$ belongs to only \emph{a single} associated prime cycle (necessarily non immersed) of $\sh{F}$ and that $\operatorname{long}(\sh{F}_z)=1$ at its generic point $z$.
It is clear that if $\sh{F}$ is a coherent $\sh{O}_X$-Module that is integral, it is integral at every point of $X$; conversely, if $\sh{F}$ is \emph{integral at a point} $x$, there exists an open neighbourhood $U$ of $x$ such that $\sh{F}|U$ is an integral $\sh{O}_U$-Module: it suffices in fact to take $U$ meeting no immersed associated prime cycle of $\sh{F}$ (such a neighbourhood exists since these cycles form a locally finite set of closed subsets of $X$).
\end{definition}

\oldpage[IV-2]{42}

\begin{definition}[3.2.5]
\label{IV.3.2.5}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module.
A \emph{decomposition} of $\sh{F}$ is called \emph{irredundant} if it is a family $(\sh{F}_\alpha)_{\alpha\in I}$ of quotient $\sh{O}_X$-Modules of $\sh{F}$ such that the $\sh{F}_\alpha$ are irredundant, the family $(\Supp(\sh{F}_\alpha))$ is locally finite, and the canonical homomorphism $\sh{F}\to\bigoplus_{\alpha\in I}\sh{F}_\alpha$ is injective.
Such a decomposition is said to be \emph{reduced} if the sets $\operatorname{Ass}(\sh{F}_\alpha)$ are pairwise distinct, and if there exists no subset $J\neq I$ such that the subfamily $(\sh{F}_\alpha)_{\alpha\in J}$ is an irredundant decomposition of $\sh{F}$.

If $(\sh{F}_\alpha)_{\alpha\in I}$ is an irredundant decomposition (resp.\ reduced irredundant decomposition) of $\sh{F}$, and if one sets $\sh{F}_\alpha=\sh{F}/\sh{G}_\alpha$, we say also that the family $(\sh{G}_\alpha)_{\alpha\in I}$ of sub-$\sh{O}_X$-Modules of $\sh{F}$ is a \emph{primary decomposition} (resp.\ a \emph{reduced primary decomposition}) of $0$ in $\sh{F}$; one observes that the injectivity of the canonical homomorphism $\sh{F}\to\bigoplus_{\alpha\in I}\sh{F}_\alpha$ is equivalent to the condition $\bigcap_{\alpha\in I}\sh{G}_\alpha=0$.
\end{definition}

\begin{proposition}[3.2.6]
\label{IV.3.2.6}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module.
Then there exists a reduced irredundant decomposition $(\sh{F}^{(x)})_{x\in\operatorname{Ass}(\sh{F})}$, formed of coherent $\sh{O}_X$-Modules such that for every $x\in\operatorname{Ass}(\sh{F})$, we have $\operatorname{Ass}(\sh{F}^{(x)})=\{x\}$.
For every $x\in\operatorname{Ass}(\sh{F})$ such that $\{x\}$ is not immersed, $\sh{F}^{(x)}$ is uniquely determined as the image of the canonical homomorphism $\sh{F}\to i_*(i^*(\sh{F}))$, where $i$ is the canonical morphism $\Spec(k(x))\to X$.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{43}
For every $x\in\operatorname{Ass}(\sh{F})$, let $U$ be an affine open neighbourhood of $x$, with ring $A$, and let $\sh{F}|U=\widetilde{M}$, where $M$ is an $A$-module of finite type.
We know (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection2, no.~3, prop.~4) that there exists a submodule $N$ of $M$ such that, if we set $P=M/N$, we have $\operatorname{Ass}(P)=\{x\}$ and $\operatorname{Ass}(N)=\operatorname{Ass}(M)-\{x\}$.
Let $\sh{G}=\widetilde{P}$, which is a quasi-coherent $\sh{O}_U$-Module; let $u:j^*(\sh{F})\to\sh{G}$ be the surjective homomorphism corresponding to $M\to P$; from this we derive by composition a homomorphism $j_*(u):j_*(j^*(\sh{F}))\to j_*(\sh{G})$, from which by composition a homomorphism
\[
  v:\sh{F}\xrightarrow{\rho_\sh{F}} j_*(j^*(\sh{F}))\xrightarrow{j_*(u)} j_*(\sh{G})
\]
whose $u$ is the restriction to $U$; we denote by $\sh{F}^{(x)}$ the image of $\sh{F}$ by this homomorphism, which is a coherent $\sh{O}_X$-Module \sref[I]{I.6.1.1}.
We have $\operatorname{Ass}(j_*(\sh{G}))=\{x\}$ by virtue of \sref{IV.3.1.13}, and \emph{a fortiori} \sref{IV.3.1.7} $\operatorname{Ass}(\sh{F}^{(x)})=\{x\}$, since $\sh{F}^{(x)}\neq 0$.
Moreover, if $\sh{H}$ is the kernel of $v$, $\operatorname{Ass}(\sh{H})$ is contained in the intersection of the $\operatorname{Ass}(\sh{N}^{(x)})$, which is empty; by \sref{IV.3.1.5}, $\sh{H}=0$.
In view of the fact that $\operatorname{Ass}(\sh{F})$ is locally finite \sref{IV.3.1.6}, it is clear that $(\sh{F}^{(x)})_{x\in\operatorname{Ass}(\sh{F})}$ is a reduced irredundant decomposition of $\sh{F}$ satisfying the conditions of the statement.
The characterisation of $\sh{F}^{(x)}$ when $\{x\}$ is not immersed follows from Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection2, no.~3, prop.~5, the question being local, and taking into account \sref[I]{I.1.6.7}.
\end{proof}

\begin{corollary}[3.2.7]
\label{IV.3.2.7}
Under the hypotheses of \sref{IV.3.2.6}, if $\sh{F}$ has no immersed associated prime cycle, there exists only one reduced irredundant decomposition of $\sh{F}$.
\end{corollary}

\begin{corollary}[3.2.8]
\label{IV.3.2.8}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module.
There exists a finite filtration $(\sh{F}_i)_{0\leq i\leq n}$ of $\sh{F}$ such that $\sh{F}_0=\sh{F}$, $\sh{F}_n=0$, formed of coherent $\sh{O}_X$-Modules and such that the quotients $\sh{F}_i/\sh{F}_{i+1}$ are zero or irredundant, and $\operatorname{Ass}(\sh{F}_i/\sh{F}_{i+1})\subset\operatorname{Ass}(\sh{F})$.
\end{corollary}

\begin{proof}
Indeed, $\sh{F}$ is isomorphic to a sub-$\sh{O}_X$-Module of a finite direct sum $\bigoplus_{j=1}^{n}\sh{G}_j$, where the $\sh{G}_j$ are irredundant and coherent \sref{IV.3.2.6}; since every quasi-coherent sub-$\sh{O}_X$-Module of $\sh{G}_j$ is either zero or irredundant \sref{IV.3.1.7}, the $\sh{F}_i=\sh{F}\cap(\bigoplus_{j=1}^{n-i}\sh{G}_j)$ answer the question, $\sh{F}_i/\sh{F}_{i+1}$ being isomorphic to a sub-$\sh{O}_X$-Module coherent of $\sh{G}_{n-i}$.
\end{proof}

\subsection{Relations with flatness}
\label{subsection:IV.3.3}

\begin{proposition}[3.3.1]
\label{IV.3.3.1}
Let $f:X\to Y$ be a morphism, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module which is $f$-flat, $\sh{E}$ a quasi-coherent $\sh{O}_Y$-Module.
If, for every $y\in Y$, we set $\sh{F}_y=\sh{F}\otimes_{\sh{O}_Y}k(y)$, then
\begin{equation}
\label{IV.3.3.1.1}
  \operatorname{Ass}(\sh{E}\otimes_{\sh{O}_Y}\sh{F})\supset\bigcup_{y\in\operatorname{Ass}(\sh{E})}\operatorname{Ass}(\sh{F}_y)\tag{3.3.1.1}
\end{equation}
and the two members are equal if $Y$ is locally Noetherian.
\end{proposition}

\begin{proof}
(It is understood that $\sh{F}_y$ is a sheaf on the fibre $f^{-1}(y)$, and we identify this fibre with a subspace of $X$ \sref[I]{I.3.6.1}.)
The question being local on $X$ and on $Y$, we are reduced to the case where $X$ and $Y$ are affine, and the proposition is then proved in Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection2, no.~6, th.~2.
\end{proof}

\begin{corollary}[3.3.2]
\label{IV.3.3.2}
Let $Y$ be a locally Noetherian prescheme without immersed associated prime cycles, $f:X\to Y$ a morphism, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module which is $f$-flat.
Then, for every $x\in\operatorname{Ass}(\sh{F})$, $f(x)$ is a maximal point of $Y$.
\end{corollary}

\begin{proof}
It suffices to apply \sref{IV.3.3.1} with $\sh{E}=\sh{O}_Y$, since $\operatorname{Ass}(\sh{O}_Y)$ is by hypothesis the set of maximal points of $Y$.
\end{proof}

\begin{corollary}[3.3.3]
\label{IV.3.3.3}
Under the hypotheses of \sref{IV.3.3.1}, suppose moreover that $X$ and $Y$ are locally Noetherian, $\sh{E}$ and $\sh{F}$ coherent.
Then the following conditions are equivalent:
\begin{enumerate}[a)]
  \item $\sh{E}\otimes_{\sh{O}_Y}\sh{F}$ is without immersed associated prime cycle.
  \item For every point $y\in\operatorname{Ass}(\sh{E})\cap f(\Supp(\sh{F}))$, $\overline{\{y\}}$ is a non-immersed associated prime cycle of $\sh{E}$ and $\sh{F}_y$ is without immersed associated prime cycle.
\end{enumerate}
\end{corollary}

\oldpage[IV-2]{44}

\begin{proof}
Suppose $a)$ verified.
The hypotheses imply that $\sh{E}\otimes_{\sh{O}_Y}\sh{F}$ is a coherent $\sh{O}_X$-Module $(\mathbf{0_I}$, 5.3.11 and \sref[I]{I.5.3.5}); its associated prime cycles are therefore the irreducible components of $\Supp(\sh{E}\otimes_{\sh{O}_Y}\sh{F})$ \sref{IV.3.1.4} and for every maximal point $x$ of $\Supp(\sh{E}\otimes_{\sh{O}_Y}\sh{F})$, $f(x)=y$ is a maximal point of $\Supp(\sh{E})$ and $x$ is a maximal point of $\Supp(\sh{F}_y)$ \sref{IV.2.5.5}.
Since, by virtue of \sref{IV.3.3.1} and of the hypothesis, $y\in f(\Supp(\sh{F}))$, we see that $\sh{F}_y\neq 0$ \sref[I]{I.9.1.13}, every point of $\operatorname{Ass}(\sh{E})\cap f(\Supp(\sh{F}))$ is the image by $f$ of a maximal point of $\Supp(\sh{E}\otimes_{\sh{O}_Y}\sh{F})$, whence condition $b)$ is verified.

Conversely, suppose $b)$ verified, and show that if $z$, $z'$ are two distinct points of $\operatorname{Ass}(\sh{E}\otimes_{\sh{O}_Y}\sh{F})$, neither of them can be adherent to the other.
Firstly, if $f(z)=f(z')=y$, we have $y\in\operatorname{Ass}(\sh{E})$ and $z$ and $z'$ belong to $\operatorname{Ass}(\sh{F}_y)$ by \sref{IV.3.3.1}, hence by hypothesis none of the two points $z$, $z'$ is adherent to the other \emph{in} $f^{-1}(y)$, so none of them can be adherent to the other \emph{in} $X$.
If $y=f(z)$ and $y'=f(z')$ are distinct, they belong to $\operatorname{Ass}(\sh{E})\cap f(\Supp(\sh{F}))$, hence neither of them can be adherent to the other in $Y$; it follows from the continuity of $f$ that neither of the two points $z$, $z'$ can be adherent to the other in $X$.
\end{proof}

\begin{proposition}[3.3.4]
\label{IV.3.3.4}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism, $\sh{E}$ a coherent $\sh{O}_Y$-Module, $\sh{F}$ a coherent $\sh{O}_X$-Module which is $f$-flat.
Then the following conditions are equivalent:
\begin{enumerate}[a)]
  \item $\sh{E}\otimes_{\sh{O}_Y}\sh{F}$ is reduced \sref{IV.3.2.2}.
  \item For every point $y\in\operatorname{Ass}(\sh{E})\cap f(\Supp(\sh{F}))$, $\overline{\{y\}}$ is a non-immersed associated prime cycle of $\sh{E}$, $\operatorname{long}(\sh{E}_y)=1$ and $\sh{F}_y$ is reduced.
\end{enumerate}
\end{proposition}

\begin{proof}
Suppose $a)$ verified.
On already knows from \sref{IV.3.3.3} that for every $y\in\operatorname{Ass}(\sh{E})\cap f(\Supp(\sh{F}))$, $\overline{\{y\}}$ is a non-immersed associated prime cycle of $\sh{E}$ and $\sh{F}_y$ is without immersed associated prime cycle.
Moreover \sref{IV.2.5.5}, for every $x\in\operatorname{Ass}(\sh{E}\otimes_{\sh{O}_Y}\sh{F})\cap f^{-1}(y)$, we have $1=\operatorname{long}((\sh{E}\otimes_{\sh{O}_Y}\sh{F})_x)=\operatorname{long}(\sh{E}_y)\cdot\operatorname{long}((\sh{F}_y)_x)$, hence $\operatorname{long}(\sh{E}_y)=1$, which proves $b)$.

Conversely, suppose $b)$ verified; one already knows that every $x\in\operatorname{Ass}(\sh{E}\otimes_{\sh{O}_Y}\sh{F})$ is a maximal point of $\Supp(\sh{E}\otimes_{\sh{O}_Y}\sh{F})$, that $y=f(x)$ is a maximal point of $\Supp(\sh{E})$ and $x$ a maximal point of $\Supp(\sh{F}_y)$ \sref{IV.3.3.1} and \sref{IV.3.3.3}; moreover it follows from the hypothesis and \sref{IV.2.5.5} that $\operatorname{long}((\sh{E}\otimes_{\sh{O}_Y}\sh{F})_x)=1$, which proves $a)$.
\end{proof}

\begin{corollary}[3.3.5]
\label{IV.3.3.5}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a \emph{flat} morphism; if $Y$ is reduced at the points of $f(X)$ and if $f^{-1}(y)$ is a reduced $k(y)$-prescheme for every $y\in f(X)$, then $X$ is reduced.
\end{corollary}

\begin{proof}
Since the Nilradical $\sh{N}_Y$ is coherent, the set of points where $Y$ is reduced is open $(\mathbf{0_I}$, 5.2.2), and one can restrict to the case where $Y$ is reduced.
It suffices then to apply \sref{IV.3.3.4} with $\sh{E}=\sh{O}_Y$ and $\sh{F}=\sh{O}_X$.
\end{proof}

\begin{proposition}[3.3.6]
\label{IV.3.3.6}
\oldpage[IV-2]{45}
Let $f:X\to\mathrm{S}$, $g:Y\to\mathrm{S}$ be two morphisms, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module, $\sh{G}$ a quasi-coherent $\sh{O}_Y$-Module.
Suppose that: $1^\circ$ $\sh{G}$ is $g$-flat; $2^\circ$ $X$ is locally Noetherian, and for every $s\in\mathrm{S}$, $g^{-1}(s)$ is locally Noetherian \emph{(}which will hold if $Y$ is also locally Noetherian\emph{)}.
Let $Z=X\times_\mathrm{S} Y$; for every couple $(x,y)$ such that $x\in X$, $y\in Y$ and $f(x)=g(y)=s$, let $\mathrm{T}_{x,y}$ be the prescheme $\Spec(k(x)\otimes_{k(s)}k(y))$, and let $\mathrm{I}_{x,y}$ be the image of $\operatorname{Ass}(\sh{O}_{\mathrm{T}_{x,y}})$ by the canonical monomorphism $\mathrm{T}_{x,y}\to Z$ \sref[I]{I.3.4.9}.
Then
\begin{equation}
\label{IV.3.3.6.1}
  \operatorname{Ass}(\sh{F}\otimes_\mathrm{S}\sh{G})=\bigcup_{x\in\operatorname{Ass}(\sh{F})}\Big(\bigcup_{y\in\operatorname{Ass}(\sh{G}_{f(x)})}\mathrm{I}_{x,y}\Big)\tag{3.3.6.1}
\end{equation}
where for every $s\in\mathrm{S}$, $\sh{G}_s=\sh{G}\otimes_{\sh{O}_\mathrm{S}}k(s)$.
\end{proposition}

\begin{proof}
Let $p:Z\to X$, $q:Z\to Y$ be the canonical projections, so that we have the commutative diagram
\[
  \xymatrix{
    X & Z\ar[l]_p \\
    \mathrm{S} \ar[u]^f & Y\ar[l]^g\ar[u]_q
  }
\]
Setting $\sh{G}'=q^*(\sh{G})$, so that $\sh{F}\otimes_\mathrm{S}\sh{G}=\sh{F}\otimes_{\sh{O}_X}\sh{G}'$; since $\sh{G}'$ is $p$-flat \sref{IV.2.1.4}, it follows from \sref{IV.3.3.1} that
\begin{equation}
\label{IV.3.3.6.2}
  \operatorname{Ass}(\sh{F}\otimes_\mathrm{S}\sh{G})=\bigcup_{x\in\operatorname{Ass}(\sh{F})}\operatorname{Ass}(\sh{G}'_x)\tag{3.3.6.2}
\end{equation}
with $\sh{G}'_x=\sh{G}'\otimes_{\sh{O}_X}k(x)$.
If $s=f(x)$, we have $\sh{G}'_x=\sh{G}_s\otimes_{k(s)}k(x)$, and $p^{-1}(x)=g^{-1}(s)\otimes_{k(s)}k(x)$; moreover, since the field $k(x)$ is a flat $k(s)$-module, the morphism $p^{-1}(x)\to g^{-1}(s)$ is flat \sref{IV.2.1.4}; applying \sref{IV.3.3.1} to this morphism, it follows that
\begin{equation}
\label{IV.3.3.6.3}
  \operatorname{Ass}(\sh{G}'_x)=\bigcup_{y\in\operatorname{Ass}(\sh{G}_s)}\operatorname{Ass}(\sh{O}_{\mathrm{T}_{x,y}})\tag{3.3.6.3}
\end{equation}
from which the proposition follows.
\end{proof}

We note that if, in the statement, we drop hypothesis $2^\circ$, one can still deduce, by virtue of \sref{IV.3.3.1}, the relation
\begin{equation}
\label{IV.3.3.6.4}
  \operatorname{Ass}(\sh{F}\otimes_\mathrm{S}\sh{G})\supset\bigcup_{x\in\operatorname{Ass}(\sh{F})}\Big(\bigcup_{y\in\operatorname{Ass}(\sh{G}_{f(x)})}\mathrm{I}_{x,y}\Big).\tag{3.3.6.4}
\end{equation}

\begin{corollary}[3.3.7]
\label{IV.3.3.7}
Under the hypotheses of \sref{IV.3.3.6}, suppose moreover that $\mathrm{S}$ is locally Noetherian and that $f(\operatorname{Ass}(\sh{F}))\subset\operatorname{Ass}(\sh{O}_\mathrm{S})$.
Then
\begin{equation}
\label{IV.3.3.7.1}
  \operatorname{Ass}(\sh{F}\otimes_\mathrm{S}\sh{G})=\bigcup_{(x,y)\in\mathrm{C}}\mathrm{I}_{x,y}\tag{3.3.7.1}
\end{equation}
where $\mathrm{C}$ is the set of couples $(x,y)$ such that $x\in\operatorname{Ass}(\sh{F})$, $y\in\operatorname{Ass}(\sh{G})$ and $f(x)=g(y)$.
\end{corollary}

\begin{proof}
Since $\sh{G}$ is $g$-flat, it follows from \sref{IV.3.3.1} that the relation ``$s\in\operatorname{Ass}(\sh{G}_s)$'' is equivalent to $y\in\operatorname{Ass}(\sh{G})$: the conclusion follows from \sref{IV.3.3.6.1}.
\end{proof}

\begin{remarks}[3.3.8]
\label{IV.3.3.8}
We will see later \sref{IV.4.2.2} that under the hypotheses of \sref{IV.3.3.6}, $\mathrm{T}_{x,y}$ is a prescheme \emph{without immersed associated prime cycle}; it will follow that if $\sh{F}$ and the $\sh{G}_s$ are without immersed associated prime cycle, the same is true of $\sh{F}\otimes_\mathrm{S}\sh{G}$.
\end{remarks}

\begin{corollary}[3.3.9]
\label{IV.3.3.9}
Under the conditions of \sref{IV.3.3.7}, we have
\begin{equation}
\label{IV.3.3.9.1}
  q(\operatorname{Ass}(\sh{F}\otimes_\mathrm{S}\sh{G}))\subset\operatorname{Ass}(\sh{G})\tag{3.3.9.1}
\end{equation}
\emph{(}where $q:X\times_\mathrm{S} Y\to Y$ is the canonical projection\emph{)}.
\end{corollary}

\begin{proof}
Indeed, if $(x,y)\in Z$, we have $q(\mathrm{I}_{x,y})=\{y\}\subset\operatorname{Ass}(\sh{G})$.
\end{proof}

\subsection{Properties of the sheaves \texorpdfstring{$\sh{F}/t\sh{F}$}{F/tF}}
\label{subsection:IV.3.4}

\oldpage[IV-2]{46}

\begin{proposition}[3.4.1]
\label{IV.3.4.1}
Let $X$ be a locally Noetherian prescheme, $t$ a section of $\sh{O}_X$ over $X$, $Y$ the closed subprescheme of $X$ defined by the Ideal $t\sh{O}_X$ of $\sh{O}_X$.
Let $\sh{F}$ be a coherent $\sh{O}_X$-Module, $\mathrm{S}$ the closed subprescheme having $\Supp(\sh{F})$ as underlying space, $(\mathrm{S}_i)$ the family of the reduced closed subpreschemes of $X$ having as underlying spaces the irreducible components of $\mathrm{S}$; we denote by $s_i$ the generic point of $\mathrm{S}_i$.
Finally, let $Z$ be an irreducible component of $\Supp(\sh{F}/t\sh{F})=\mathrm{S}\cap Y$, $z$ its generic point.
\begin{enumerate}[\rm(i)]
  \item For every $i$ such that $Z\subset\mathrm{S}_i$, $Z$ is an irreducible component of $\mathrm{S}_i\cap Y$.
  \item If $Z$ is not equal to any of the $\mathrm{S}_i$, we have
\end{enumerate}
\begin{equation}
\label{IV.3.4.1.1}
  \operatorname{long}((\sh{F}/t\sh{F})_z)\geq\sum_i\operatorname{long}(\sh{F}_{s_i})\tag{3.4.1.1}
\end{equation}
where the sum of the second member is extended to all $i$ such that $Z\subset\mathrm{S}_i$.
\begin{enumerate}
  \item[\rm(iii)] Suppose that $Z$ is not equal to any $\mathrm{S}_i$.
  For the two members of \sref{IV.3.4.1.1} to be equal, it is necessary and sufficient that the two following conditions be satisfied:
  \begin{enumerate}[$\alpha)$]
    \item $t_z$ is $\sh{F}_z$-regular $(\mathbf{0_{IV}}$, 15.1.4).
    \item For every $i$ such that $M_{\mathfrak{p}_i}\neq 0$, the image of the germ $t_z$ in $\sh{O}_{\mathrm{S}_i,z}$ generates the maximal ideal of this ring \emph{(}which implies that $\sh{O}_{\mathrm{S}_i,z}$ is a discrete valuation ring whose $t_z$ is a uniformiser\emph{)}.
  \end{enumerate}
\end{enumerate}
\end{proposition}

\begin{proof}
Suppose that $Z$ is not equal to any $\mathrm{S}_i$.
As in the demonstration of \sref{IV.3.4.1}, we can reduce to the case where $X=\Spec(\sh{O}_z)$; the proposition is then (taking into account \sref{IV.3.1.2}) a consequence of $(\mathbf{0_{IV}}$, 16.4.6.3).

\emph{(i)} If $j:Y\to X$ is the canonical injection, we have $\sh{F}/t\sh{F}=\sh{F}\otimes_{\sh{O}_X}\sh{O}_Y=j^*(\sh{F})$, hence $\Supp(\sh{F}/t\sh{F})=j^{-1}(\mathrm{S})=\mathrm{S}\cap Y$ \sref[I]{I.9.1.13}, whence the assertion.

\emph{(ii) and (iii)}: Suppose that $Z$ is not equal to any of the $\mathrm{S}_i$.
We can, to prove \emph{(ii)} and \emph{(iii)}, replace $X$ by $\Spec(\sh{O}_z)$; and if $M=\sh{F}_z$, we can suppose $\sh{F}=\widetilde{M}$, denoting by $\mathfrak{p}_i$ the minimal ideals of $\sh{O}_z$.
Moreover, since $M$ is an $\sh{O}_z$-module of finite type, on a $\mathrm{S}=\mathrm{S}'_{\mathrm{red}}$, with $\mathrm{S}'=\Spec(\sh{O}_z/\mathfrak{a})$, where $\mathfrak{a}$ is the annihilator of $M$ in $\sh{O}_z$ $(\mathbf{0_I}$, 1.7.4), and the two members of \sref{IV.3.4.1.1} have the same values whether one considers $M$ as an $\sh{O}_z$-module or as an $(\sh{O}_z/\mathfrak{a})$-module; one can therefore finally replace $X$ by $\mathrm{S}'=\Spec(A)$, where $A$ is a local Noetherian ring, $M$ being a faithful $A$-module; since $Z$ is closed in $\mathrm{S}$, the hypothesis that $Z\neq\mathrm{S}_i$ for all $i$ means that $s_i\notin Z$, hence $\dim(A)>0$; finally, to say that $z$ is the generic point of $Z$, an irreducible component of $\mathrm{S}_i\cap V(t)$, means that $A/tA$ is an Artinian local ring.
We are therefore reduced to proving the following statement:
\end{proof}

\begin{lemma}[3.4.1.2]
\label{IV.3.4.1.2}
Let $A$ be a local Noetherian ring of dimension $>0$, $\mathfrak{p}_i$ the minimal prime ideals of $A$, $\mathfrak{m}$ its maximal ideal, $t$ an element of $\mathfrak{m}$ such that $A/tA$ is Artinian.
Then, for every $A$-module $M$ of finite type, we have
\begin{equation}
\label{IV.3.4.1.3}
  \operatorname{long}(M/tM)\geq\sum_i\operatorname{long}(M_{\mathfrak{p}_i})\,;\tag{3.4.1.3}
\end{equation}
moreover, for the two members of \sref{IV.3.4.1.3} to be equal, it is necessary and sufficient that the following conditions be satisfied:
\begin{enumerate}[$\alpha)$]
  \item $t$ is $M$-regular;
  \item for every $i$ such that $M_{\mathfrak{p}_i}\neq 0$, the image of $t$ in $A/\mathfrak{p}_i$ generates the maximal ideal of this ring \emph{(}which implies that $A/\mathfrak{p}_i$ is a discrete valuation ring\emph{)}.
\end{enumerate}
\end{lemma}

\oldpage[IV-2]{47}

\begin{proof}
Since $A$ is not of dimension $0$ and $A/tA$ is Artinian, we necessarily have $\dim(A)=1$ $(\mathbf{0_{IV}}$, 16.3.4) and $t\notin\mathfrak{p}_i$ for every $i$: the principal ideal $(t)$ is therefore an ideal of definition, and contains by consequence a power of its maximal ideal $\mathfrak{m}$.
Let $N$ be the submodule of elements of $M$ annihilated by a power of $t$ (or by a power of $\mathfrak{m}$, which amounts to the same since one has just seen this); if one sets $P=M/N$, $t$ is $P$-regular, since for $x\in M$ the relation $tx\in N$ implies $t^k(tx)=0$ for an integer $k$, hence $x\in N$.
This being so, we have the

\begin{lemma}[3.4.1.4]
\label{IV.3.4.1.4}
Let $A$ be a ring,
\[
  0\to M'\to M\to M''\to 0
\]
an exact sequence of $A$-modules.
If $t\in A$ is $M''$-regular, the sequence
\[
  0\to M'/tM'\to M/tM\to M''/tM''\to 0
\]
is exact.
\end{lemma}

\begin{proof}
Since $M/tM=M\otimes_A(A/tA)$, it suffices to prove the exactness at $M'/tM'$; now, if the image $x'$ of an element $x'$ of $M'$ is such that $x'=ty$ with $y\in M$, we deduce, for the images $x''$, $y''$ in $M''$, $x''=ty''$; but as $x''=0$, the hypothesis implies $y''=0$, hence $y$ is the image of an element $y'\in M'$, and the relation $x'=ty'$ proves that $M'\to M$ is injective.
\end{proof}

This lemma being established, we derive the relation
\begin{equation}
\label{IV.3.4.1.5}
  \operatorname{long}(M/tM)=\operatorname{long}(N/tN)+\operatorname{long}(P/tP).\tag{3.4.1.5}
\end{equation}

On the other hand, for every $i$, we have $N_{\mathfrak{p}_i}=0$ since $t\notin\mathfrak{p}_i$; hence $M_{\mathfrak{p}_i}=P_{\mathfrak{p}_i}$; to prove \sref{IV.3.4.1.3}, it suffices to do so replacing $M$ by $P$; moreover, if the two members of \sref{IV.3.4.1.3} are equal, it follows from the same inequality for $P$ and from \sref{IV.3.4.1.5} that necessarily $\operatorname{long}(N/tN)=0$, hence $N/tN=0$ and finally $N=0$, by the lemma of Nakayama, $N$ being of finite type; now, $N=0$ means that $t$ is $M$-regular.
$\operatorname{Ass}(M)\subset\bigcup\{\mathfrak{p}_i\}$.

We proceed then by induction on $n=\sum\operatorname{long}(M_{\mathfrak{p}_i})$.
If $n=0$, we have $M_{\mathfrak{p}_i}=0$ for every $i$, hence $M=0$ since none of the $\mathfrak{p}_i$ belongs to $\operatorname{Ass}(M)$; the two members of \sref{IV.3.4.1.3} are then zero, and the assertion $\beta)$ of \sref{IV.3.4.1.2} is trivially satisfied.
If $n>0$, the reasoning from the beginning of the proof of \sref{IV.3.4.1} allows us moreover to suppose that the $A$-module $M$ is \emph{faithful}: this implies $M_{\mathfrak{p}_i}\neq 0$ for every $i$ (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection2, no.~2, cor.~2 of prop.~4), and therefore $\operatorname{Ass}(M)=\bigcup\{\mathfrak{p}_i\}$.

\oldpage[IV-2]{48}

Suppose first $n=1$; there is then only a single minimal prime ideal $\mathfrak{p}$ of $A$, and to say that $M_\mathfrak{p}$ is of length~$1$ means that $M_\mathfrak{p}$ is isomorphic to the residue field $k=A_\mathfrak{p}/\mathfrak{p} A_\mathfrak{p}$, hence $\mathfrak{p}$ is the annihilator of $M$ as an $A_\mathfrak{p}$-module.
By consequence $M_\mathfrak{p}$ is annihilated by $\mathfrak{p} A_\mathfrak{p}$, hence $\mathfrak{p}$ is the annihilator of $M$ since $M$ is supposed faithful.
This being so, the hypothesis $M\neq 0$ implies $M/tM\neq 0$ by the lemma of Nakayama, hence $\operatorname{long}(M/tM)\geq 1$, which proves \sref{IV.3.4.1.3} in this case.
Moreover, if $\operatorname{long}(M/tM)=1$, $M$ is necessarily \emph{monogenic} (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection3, no.~2, cor.~2 of prop.~4) hence isomorphic to a quotient $A/\mathfrak{b}$; since it is faithful, we necessarily have $\mathfrak{b}=0$ and $M$ is isomorphic to $A$; since $\operatorname{long}(A/tA)=1$, $tA$ is necessarily equal to the maximal ideal $\mathfrak{m}$, and since $A$ is a local integral Noetherian ring, this proves that $A$ is a discrete valuation ring (Bourbaki, \emph{Alg.~comm.}, chap.~VI, \textsection3, no.~6, prop.~9), whose $t$ is the uniformiser.
Conversely, if $A$ is a discrete valuation ring, $t$ its uniformiser, $M$ is without torsion, hence isomorphic to a submodule of $A$ ($M$ being of finite type), and therefore isomorphic to $A$ itself, whence $\operatorname{long}(M/tA)=\operatorname{long}(A/tA)=1$.

Suppose now $n\geq 2$; there exists then an exact sequence
\[
  0\to M'\to M\to M''\to 0
\]
with $M'\neq 0$, $M''\neq 0$ and $\operatorname{Ass}(M)=\operatorname{Ass}(M')\cup\operatorname{Ass}(M'')$; indeed, if $\operatorname{Ass}(M)$ is not reduced to a single element, this follows from Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, no.~1, prop.~4; if on the contrary $\operatorname{Ass}(M)$ is reduced to a single prime ideal, the latter is necessarily the unique minimal prime ideal $\mathfrak{p}$ of $A$; the hypothesis implies $\operatorname{long}(M_\mathfrak{p})\geq 2$ and it suffices to take for $M'$ the inverse image of a submodule of $M_\mathfrak{p}$ distinct from $0$ and from $M_\mathfrak{p}$.
Since $t$ is $M$-regular, $t$ does not belong to any of the prime ideals of $\operatorname{Ass}(M)$ (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, no.~1, cor.~2 of prop.~2), hence, for the same reason, $t$ is $M'$-regular and $M''$-regular.
This last property implies by \sref{IV.3.4.1.4} that the sequence
\[
  0\to M'/tM'\to M/tM\to M''/tM''\to 0
\]
is exact; as moreover it is the same for the sequence
\[
  0\to M'_{\mathfrak{p}_i}\to M_{\mathfrak{p}_i}\to M''_{\mathfrak{p}_i}\to 0
\]
for every $i$, we have
\begin{align*}
  \operatorname{long}(M/tM) &= \operatorname{long}(M'/tM') + \operatorname{long}(M''/tM'') \\
  \operatorname{long}(M_{\mathfrak{p}_i}) &= \operatorname{long}(M'_{\mathfrak{p}_i}) + \operatorname{long}(M''_{\mathfrak{p}_i})
\end{align*}
and the induction hypothesis implies the inequality \sref{IV.3.4.1.3}.
Moreover, the two members can only be equal if the analogous inequalities for $M'$ and $M''$ are also equalities.
By virtue of the induction hypothesis, this is equivalent to the property $\beta)$ for the $\mathfrak{p}_i$ such that $M'_{\mathfrak{p}_i}\neq 0$ or $M''_{\mathfrak{p}_i}\neq 0$; but these ideals are precisely those for which $M_{\mathfrak{p}_i}\neq 0$.
\end{proof}

\begin{corollary}[3.4.2]
\label{IV.3.4.2}
Under the general hypotheses of \sref{IV.3.4.1}, suppose that $Z$ is not equal to any of the $\mathrm{S}_i$ and that $\operatorname{long}((\sh{F}/t\sh{F})_z)=1$.
Then there exists a single $\mathrm{S}_i$ containing $Z$, and for this value of $i$, we have $\operatorname{long}(\sh{F}_{s_i})=1$; moreover $\sh{O}_{\mathrm{S}_i,z}$ is a discrete valuation ring whose $t_z$ is a uniformiser, and $t_z$ is $\sh{F}_z$-regular.
\end{corollary}

\begin{proof}
This follows from \sref{IV.3.4.1}, the two members of \sref{IV.3.4.1.1} being then equal.
\end{proof}

\begin{proposition}[3.4.3]
\label{IV.3.4.3}
\oldpage[IV-2]{49}
Let $X$ be a locally Noetherian prescheme, $t$ a section of $\sh{O}_X$ over $X$, $Y$ the closed subprescheme of $X$ defined by the Ideal $t\sh{O}_X$ of $\sh{O}_X$.
Let $\sh{F}$ be a coherent $\sh{O}_X$-Module, $\mathrm{T}$ an associated prime cycle of $\sh{F}$, $\mathrm{T}'$ an irreducible component of $\mathrm{T}\cap Y$, $x$ the generic point of $\mathrm{T}'$.
Suppose that $t_x$ is $\sh{F}_x$-regular; then $x\in\operatorname{Ass}(\sh{F}/t\sh{F})$.
\end{proposition}

\begin{proof}
As in the proof of \sref{IV.3.4.1}, we can reduce to the case where $X=\Spec(\sh{O}_x)$; the proposition is then (taking into account \sref{IV.3.1.2}) a consequence of $(\mathbf{0_{IV}}$, 16.4.6.3).
\end{proof}

\begin{proposition}[3.4.4]
\label{IV.3.4.4}
Let $X$ be a locally Noetherian prescheme, $t$ a section of $\sh{O}_X$ over $X$, $Y$ the closed subprescheme of $X$ defined by the Ideal $t\sh{O}_X$ of $\sh{O}_X$.
Let $\sh{F}$ be a coherent $\sh{O}_X$-Module, $(\mathrm{S}_i)$ the family of the irreducible components of $\Supp(\sh{F})$.
Let $y$ be a point of $Y$ such that $t_y$ is $\sh{F}_y$-regular and such that none of the associated prime cycles immersed of $\sh{F}/t\sh{F}$ contains $y$.
Then the irreducible components of $\Supp(\sh{F}/t\sh{F})$ containing $y$ are exactly the irreducible components of the $\mathrm{S}_i\cap Y$ containing $y$, and the prime cycles associated to $\sh{F}$ containing $y$ are non-immersed.
\end{proposition}

\begin{proof}
Let us first prove the last assertion.
Let $\mathrm{T}\supset\mathrm{T}_1$ be two associated prime cycles of $\sh{F}$ containing $y$; if $x$ is the generic point of an irreducible component of $\mathrm{T}\cap Y$ containing $y$, $x$ is a generalisation of $y$, hence contained in every neighbourhood of $y$, and the hypothesis that $t_y$ is $\sh{F}_y$-regular implies that $t_x$ is $\sh{F}_x$-regular $(\mathbf{0_{IV}}$, 15.2.4), hence, by virtue of \sref{IV.3.4.3}, $x\in\operatorname{Ass}(\sh{F}/t\sh{F})$.
Let $x_1$ be the generic point of an irreducible component of $\mathrm{T}_1\cap Y$ containing $y$, and let $x$ be the generic point of an irreducible component of $\mathrm{T}\cap Y$ containing $x_1$; it follows from what precedes that $x_1$ and $x$ both belong to $\operatorname{Ass}(\sh{F}/t\sh{F})$, and since $x_1\in\overline{\{x\}}$, the hypothesis implies that $x_1=x$.
Let us still designate by $\mathrm{T}$ and $\mathrm{T}_1$ the integral closed subpreschemes of $X$ having $\mathrm{T}$ and $\mathrm{T}_1$ as underlying spaces respectively, and set $A=\sh{O}_{\mathrm{T},x}$, $A_1=\sh{O}_{\mathrm{T}_1,x}$; then $A_1=A/\mathfrak{p}$, where $\mathfrak{p}$ is a prime ideal of $A$.
By definition of $x$ and $x_1$, $A/tA$ and $A_1/tA_1$ are Artinian rings; on the other hand, it was seen above that $t_x$ is $\sh{F}_x$-regular, hence $x$ cannot belong to $\operatorname{Ass}(\sh{F})$, and consequently $A_1$ is not Artinian.
We therefore have $\dim A=\dim A_1=1$; but this implies $\mathfrak{p}=0$ and $A=A_1$ $(\mathbf{0_{IV}}$, 16.1.2.2); since $\Spec(A)$ and $\Spec(A_1)$ are respectively dense in $\mathrm{T}$ and $\mathrm{T}_1$, we have indeed $\mathrm{T}=\mathrm{T}_1$.

The $\mathrm{S}_i$ containing $y$ are therefore all the associated prime cycles of $\sh{F}$ containing $y$; if $x$ is the generic point of an irreducible component of $\mathrm{S}_i\cap Y$ containing $y$, we deduce also from $(\mathbf{0_{IV}}$, 15.2.4) that $t_z$ is $\sh{F}_z$-regular, hence, by \sref{IV.3.4.3}, $x\in\operatorname{Ass}(\sh{F}/t\sh{F})$; this demonstrates the first assertion of \sref{IV.3.4.4}.
\end{proof}

\begin{proposition}[3.4.5]
\label{IV.3.4.5}
Let $X$ be a locally Noetherian prescheme, $t$ a section of $\sh{O}_X$ over $X$, $Y$ the closed subprescheme of $X$ defined by the Ideal $t\sh{O}_X$ of $\sh{O}_X$.
Let $\sh{F}$ be a coherent $\sh{O}_X$-Module, $y$ a point of $Y$; suppose that $t_y$ is $\sh{F}_y$-regular and that $\sh{F}/t\sh{F}$ is integral at the point $y$ \sref{IV.3.2.4}.
Then $\sh{F}$ is integral at the point $y$.
\end{proposition}

\oldpage[IV-2]{50}

\begin{proof}
In view of \sref{IV.3.4.4}, it suffices to prove that $y$ is contained in a \emph{single} irreducible component of $\Supp(\sh{F})$, and that if $s$ is the generic point of this component, we have $\operatorname{long}(\sh{F}_s)=1$; the conclusion then follows from \sref{IV.3.4.2}.
\end{proof}

\begin{proposition}[3.4.6]
\label{IV.3.4.6}
The hypotheses being those of \sref{IV.3.4.1}, let $x$ be a point of $Y$.
Suppose that $Y$ contains none of the $\mathrm{S}_i$ containing $x$, and that $\sh{F}/t\sh{F}$ is reduced at the point $x$ \sref{IV.3.2.2}.
Then $t_z$ is $\sh{F}_z$-regular and $\sh{F}$ is reduced at the point $x$.
Moreover, if $z_j$ is the generic point of an irreducible component of $\Supp(\sh{F}/t\sh{F})$ containing $x$, $z_j$ is contained in only one of the $\mathrm{S}_i$, and $\sh{O}_{\mathrm{S}_i,z_j}$ is a discrete valuation ring whose $t_{z_j}$ is a uniformiser.
\end{proposition}

\begin{proof}
The fact that $t_z$ is $\sh{F}_z$-regular follows from the following lemma applied to the ring $\sh{O}_z$:

\begin{lemma}[3.4.6.1]
\label{IV.3.4.6.1}
Let $A$ be a Noetherian ring, $M$ an $A$-module of finite type, $\mathfrak{p}_i$ the minimal elements of $\Supp(M)$, $t$ an element of $A$.
Suppose that $t$ does not belong to any of the $\mathfrak{p}_i$ and that $M/tM$ is a reduced $A$-module \sref{IV.3.2.2}.
Then $t$ is $M$-regular.
\end{lemma}

Every prime ideal $\mathfrak{p}\in\Supp(M)$ contains one of the $\mathfrak{p}_i$; since $t$ does not belong to any of the $\mathfrak{p}_i$, the homothety of ratio $t$ in $M_\mathfrak{p}$ is not nilpotent (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, no.~4, cor.\ of prop.~9).
Denote by $N$ the submodule of $M$ formed by the elements annihilated by a power of $t$, and set $P=M/N$; we shall show that $N=0$.
Since $t$ is $P$-regular, on a an exact sequence \sref{IV.3.4.1.4}
\[
  0\to N/tN\to M/tM\to P/tP\to 0.
\]
Since $N$ is of finite type, it is annihilated by a power of $t$, and it suffices therefore to prove that $N/tN=0$ for every $\mathfrak{p}\in\operatorname{Ass}(M/tM)$, or equivalently that the homomorphism $u_\mathfrak{p}:(M/tM)_\mathfrak{p}\to(P/tP)_\mathfrak{p}$ is bijective for every $\mathfrak{p}\in\operatorname{Ass}(M/tM)$.
Now, we have $(P/tP)_\mathfrak{p}\neq 0$; in effect, since $\mathfrak{p}\in\Supp(M/tM)=\Supp(M)\cap V(t)$ $(\mathbf{0_I}$, 1.7.5), the image of $t$ in $A_\mathfrak{p}$ is contained in the maximal ideal $\mathfrak{p} A_\mathfrak{p}$, hence the hypothesis $P_\mathfrak{p}=tP_\mathfrak{p}$ would imply $P_\mathfrak{p}=0$ by the lemma of Nakayama; one would then have $M_\mathfrak{p}=N_\mathfrak{p}$ and the homothety of ratio $t$ in $M_\mathfrak{p}$ would be nilpotent, since $\mathfrak{p}\in\Supp(M)$.
This being so, the hypothesis that $M/tM$ is reduced implies $\operatorname{long}((M/tM)_\mathfrak{p})=1$, and since $(P/tP)_\mathfrak{p}\neq 0$, $u_\mathfrak{p}$ is necessarily bijective, which proves the lemma.

By hypothesis, none of the associated prime cycles immersed of $\sh{F}/t\sh{F}$ contains $x$, hence none of the associated prime cycles immersed of $\sh{F}$ contains $x$, by virtue of \sref{IV.3.4.4}.
On the other hand, applying \sref{IV.3.4.2} to an irreducible component of $\Supp(\sh{F}/t\sh{F})$ containing $x$, we see that $\operatorname{long}(\sh{F}_{s_i})=1$ for every $\mathrm{S}_i$ containing $x$, which completes the proof that $\sh{F}_x$ is reduced; the last assertions are also consequences of \sref{IV.3.4.2}.
\end{proof}

\begin{corollary}[3.4.7]
\label{IV.3.4.7}
\oldpage[IV-2]{51}
Let $A$ be a local Noetherian ring, $\mathfrak{m}$ its maximal ideal, $M$ an $A$-module of finite type, $(x_i)_{1\leq i\leq k}$ a family of elements of $\mathfrak{m}$ forming part of a system of parameters for $M$ $(\mathbf{0_{IV}}$, 16.3.6).
If the $A$-module $N=M/(\sum_{i=1}^{k}x_i M)$ is integral \sref{IV.3.2.4}, then $M$ is integral and the sequence $(x_i)_{1\leq i\leq k}$ is $M$-regular.
\end{corollary}

\begin{proof}
By induction on $k$, one is immediately reduced to the case $k=1$; we write $x$ instead of $x_1$; the hypothesis that $x$ forms part of a system of parameters for $M$ implies $\dim(M)=\dim(M)-1$ $(\mathbf{0_{IV}}$, 16.3.7).
Setting $n=\dim(N)$; there is therefore an element $\mathfrak{p}$ of $\Supp(M)$ such that $\dim(M/\mathfrak{p} M)=n+1$ $(\mathbf{0_{IV}}$, 16.3.5), and moreover $x$ forms part of a system of parameters for $M/\mathfrak{p} M$ $(\mathbf{0_{IV}}$, 16.3.5); in addition, the support of $N'$, being isomorphic to $N$, is integral; moreover, the support of $N'$ (equal to the intersection of $\Supp(M)$ and $V(x)$) cannot contain $\Supp(M')$, and this last set is irreducible by construction.
The hypothesis implies then that $x$ is $M'$-regular by virtue of \sref{IV.3.4.6}.
One concludes that the kernel of the homothety $z\mapsto xz$ in $M$ is contained in $\mathfrak{p}^j M\subset\mathfrak{m}^j M$ for every integer $j$, and this kernel is therefore reduced to $0$ $(\mathbf{0_{IV}}$, 7.3.5), which proves that $x$ is $M$-regular.
One can then apply \sref{IV.3.4.5}.
\end{proof}

\begin{remark}[3.4.8]
\label{IV.3.4.8}
An analogous proposition to \sref{IV.3.4.7}, where one replaces ``integral'' by ``reduced'', is no longer necessarily exact.
Consider for example the ring of polynomials $C=K[X,Y,Z]$ over a field $K$, the quotient ring $B=C/\mathfrak{p}\mathfrak{q}$, where $\mathfrak{p}=CZ$, $\mathfrak{q}=CX^2+CY$; let $A$ be the local ring of $B$ corresponding to the maximal image of $CX+CY+CZ$ in $B$.
If $x$, $z$ are the canonical images of $X$, $Z$ in $A$, it is clear that $xz\neq 0$ but $x^2z^2=0$; on the other hand, since $A/xA$ is isomorphic to $K[Y,Z]/(YZ)$, we have $\dim(A/xA)=1$ while $\dim(A)=2$, hence $x$ forms part of a system of parameters of $A$ $(\mathbf{0_{IV}}$, 16.3.4), $A/xA$ is reduced, but $A$ is not.
\end{remark}

\begin{proposition}[3.4.9]
\label{IV.3.4.9}
Let $A$ be a Noetherian ring, $M$ an $A$-module of finite type, $f$ an $M$-regular element such that $M/fM$ has no immersed associated prime ideal.
If $\mathfrak{p}_i$ ($1\leq i\leq m$) are the prime ideals associated to $M/fM$, then, for every integer $n>0$, $f^nM$ is the intersection of the inverse images of $f^nM_{\mathfrak{p}_i}$ by the canonical homomorphisms $M\to M_{\mathfrak{p}_i}$ ($1\leq i\leq m$).
\end{proposition}

\begin{proof}
It amounts to showing that the $\mathfrak{p}_i$ are also the prime ideals associated to $M/f^nM$, for then the saturations of $f^nM$ for the $\mathfrak{p}_i$ are the submodules of the reduced primary decomposition (necessarily unique) of $f^nM$ in $M$.
Now, we have
\[
  \operatorname{Ass}(f^{n-1}M/f^nM)\subset\operatorname{Ass}(M/f^nM)\subset\operatorname{Ass}(M/f^{n-1}M)\cup\operatorname{Ass}(f^{n-1}M/f^nM)
\]
by \sref{IV.3.1.7}, and since $f$ is $M$-regular, $f^{n-1}M/f^nM$ is isomorphic to $M/fM$; it therefore suffices to reason by induction on $n$.
\end{proof}
