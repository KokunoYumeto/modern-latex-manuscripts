\section{Constructible properties}
\label{section:IV.9}

\subsection{The principle of finite extension}
\label{subsection:IV.9.1}

\begin{proposition}[9.1.1]
\label{IV.9.1.1}
\oldpage[IV-3]{54}
(Principle of finite extension).
Let $k$ be a field, $\mathfrak{C}$ a set of extensions of $k$.
Suppose the following conditions are satisfied:
\begin{enumerate}[label=(\roman*)]
  \item If $K\in\mathfrak{C}$ and if there exists a $k$-homomorphism $K\to K'$ (where $K'$ belongs to the universe under consideration), then $K'\in\mathfrak{C}$.
  \item If $K\in\mathfrak{C}$, there exists a sub-extension $K'$ of $K$, of finite type over $k$, such that $K'\in\mathfrak{C}$.
  \item If $K\in\mathfrak{C}$ is the field of fractions of an integral $k$-algebra $A$ of finite type, there exists $f\in A-\{0\}$ such that for every maximal ideal $\mathfrak{m}$ of $A$ we have $A_f/\mathfrak{m}\in\mathfrak{C}$.
\end{enumerate}
\oldpage[IV-3]{55}
Let $\Omega$ be an algebraically closed extension of $k$ (in the universe under consideration).
The following conditions are equivalent:
\begin{enumerate}[label=\emph{(\alph*)}]
  \item $\mathfrak{C}$ is non-empty.
  \item There exists $K'\in\mathfrak{C}$ which is a finite extension of $k$.
  \item We have $\Omega\in\mathfrak{C}$.
\end{enumerate}
\end{proposition}

\begin{proof}
The condition (i) implies that \emph{b)} entails \emph{c)}, and \emph{c)} trivially entails \emph{a)}; let us prove that \emph{a)} entails \emph{b)}.
By virtue of (ii) and (iii) there exists an extension $K'\in\mathfrak{C}$ which is of the form $A/\mathfrak{m}$, where $A$ is a $k$-algebra of finite type on $k$ and $\mathfrak{m}$ a maximal ideal of $A$.
By the Hilbert Nullstellensatz (Bourbaki, \emph{Alg.~comm.}, chap.~V, \textsection\,3, n\textsuperscript{o}\,1, cor.~2 of th.~1), $K'$ is a finite extension of $k$.
\end{proof}

\begin{corollary}[9.1.2]
\label{IV.9.1.2}
Under the hypotheses \emph{(i)}, \emph{(ii)}, \emph{(iii)} of \sref{IV.9.1.1}, if $k$ is algebraically closed and if $\mathfrak{C}$ is non-empty, $\mathfrak{C}$ contains all the extensions of $k$ in the universe under consideration.
\end{corollary}

\begin{proof}
Indeed, as \emph{a)} entails \emph{c)}, we have $k\in\mathfrak{C}$, and the conclusion follows from hypothesis~(i).
\end{proof}

\begin{remark}[9.1.3]
\label{IV.9.1.3}
In practice, one will verify condition (ii) of \sref{IV.9.1.1} by noting that $K$ is the inductive limit of its sub-extensions of finite type $K_\alpha$, and by applying the results of \textsection\,8, taking into account the fact that for $K_\alpha\subset K_\beta$, $K_\beta$ is faithfully flat over $K_\alpha$.
Frequently the set $\mathfrak{C}$ is formed by \emph{fields} belonging to a set $\mathfrak{C}'$ of $k$-algebras satisfying the following condition:
\begin{enumerate}
  \item[\emph{(i~bis)}] If $A\in\mathfrak{C}'$ and if there exists a homomorphism of $k$-algebras $A\to A'$ (where $A'$ belongs to the universe under consideration), then $A'\in\mathfrak{C}'$.
\end{enumerate}
When this is the case, condition (i) is trivially satisfied, and one will verify generally condition (iii) of \sref{IV.9.1.1} by noting that the field of fractions $K$ of $A$ is the inductive limit of the algebras $A_f$ (for the relation $\mathrm{D}(f)\supset\mathrm{D}(g)$), and by applying the results of \textsection\,8, taking into account eventually the fact that the morphisms $\mathrm{D}(g)\to\mathrm{D}(f)$ are open immersions.
\end{remark}

\begin{proposition}[9.1.4]
\label{IV.9.1.4}
Let $k$ be a field, $\Omega$ an algebraically closed extension of $k$, $X$ and $Y$ two preschemes of finite type over $k$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{(\alph*)}]
  \item There exists an $\Omega$-morphism $X_{(\Omega)}\to Y_{(\Omega)}$ (resp.\ an $\Omega$-morphism having one (determined) of the properties \emph{(i)} to \emph{(xiv)} of \sref{IV.8.10.5}).
  \item There exists a finite extension $k'$ of $k$ and a $k'$-morphism $X_{(k')}\to Y_{(k')}$ (resp.\ a $k'$-morphism having the considered property).
  \item There exists an extension $K$ of $k$ and a $K$-morphism $X_{(K)}\to Y_{(K)}$ (resp.\ a $K$-morphism having the considered property).
\end{enumerate}
\end{proposition}

\begin{proof}
We apply the remark \sref{IV.9.1.3}, taking for $\mathfrak{C}'$ the set of all $k$-algebras $A$ (of the universe under consideration) such that there exists an $A$-morphism $X\otimes_k A\to Y\otimes_k A$ (resp.\ a morphism having one of the properties of \sref{IV.8.10.5}).
The condition~(i~bis) of \sref{IV.9.1.3} is then verified because the
\oldpage[IV-3]{56}
properties considered in \sref{IV.8.10.5} are all stable under base change.
The condition (iii) of \sref{IV.9.1.1} is therefore satisfied by virtue of \sref{IV.8.8.2}[, (i)] (resp.\ \sref{IV.8.10.5}), since $\operatorname{Spec}(k)$ is quasi-compact and quasi-separated.
It remains to verify condition (ii) of \sref{IV.9.1.1} which also follows from \sref{IV.8.8.2}[, (i)] and \sref{IV.8.10.5}, by considering $K$ as an inductive limit of its sub-extensions of finite type.
We conclude therefore by \sref{IV.9.1.1}.
In particular, if there exists an extension $K$ of $k$ and a $K$-isomorphism $X_{(K)}\xrightarrow{\sim}Y_{(K)}$, we say that $X$ and $Y$ are \emph{geometrically isomorphic}.
\end{proof}

The following corollary generalizes \sref[II]{II.6.6.5}:

\begin{corollary}[9.1.5]
\label{IV.9.1.5}
Let $k$ be a field, $X$ a $k$-prescheme.
If there exists an extension $K$ of $k$ such that $X_{(K)}$ is projective (resp.\ quasi-projective) over $K$, then $X$ is projective (resp.\ quasi-projective) over $k$.
\end{corollary}

\begin{proof}
The morphism $\operatorname{Spec}(K)\to\operatorname{Spec}(k)$ being faithfully flat and quasi-compact, and $X$ being of finite type over $k$ already by \sref[II]{II.2.7.1}[, (v)], the hypothesis means that there exists a closed immersion (resp.\ an immersion) $X_{(K)}\to\bb{P}^r_K=\bb{P}^r_k\otimes_k K$ (\sref[II]{II.5.4.2}, (ii) and \sref[II]{II.5.5.3}); applying property (iv) (resp.\ (ii)) of \sref{IV.8.10.5}, we deduce that there exists a finite extension $k'$ of $k$ and a closed immersion (resp.\ an immersion) $X_{(k')}\to\bb{P}^r_{k'}$, in other words $X_{(k')}$ is projective (resp.\ quasi-projective) over $k'$.
We conclude by \sref[II]{II.6.6.5}.
\end{proof}

\begin{proposition}[9.1.6]
\label{IV.9.1.6}
Let $k$ be a field, $\Omega$ an algebraically closed extension of $k$, $X$ a prescheme of finite type over $k$, $\sh{F}$, $\sh{G}$ two coherent $\sh{O}_X$-Modules.
Suppose that there exists an isomorphism $\sh{F}\otimes_k\Omega\xrightarrow{\sim}\sh{G}\otimes_k\Omega$.
Then there exists a finite extension $k'$ of $k$ and an isomorphism $\sh{F}\otimes_k k'\xrightarrow{\sim}\sh{G}\otimes_k k'$.
\end{proposition}

\begin{proof}
The reasoning is the same as in \sref{IV.9.1.4}, applying \sref{IV.8.5.2}[, (i)] (one uses in the proof the property (iii) of \sref{IV.9.1.1}), the fact that the morphisms $\mathrm{D}(g)\to\mathrm{D}(f)$ (with the notations of \sref{IV.9.1.3}) are open immersions, and $a$ $fortiori$ the morphisms are flat.
\end{proof}

\subsection{Constructible and ind-constructible properties}
\label{subsection:IV.9.2}

\begin{definition}[9.2.1]
\label{IV.9.2.1}
\oldpage[IV-3]{56}
Let $\mathbf{P}(X,\sh{F},k)$ be a relation.
We say that $\mathbf{P}$ is a \emph{constructible} (resp.\ \emph{ind-constructible}) \emph{property of algebraic preschemes} if the two following conditions are satisfied:
\begin{enumerate}[label=(\roman*)]
  \item If $k$ is a field, $X$ an algebraic prescheme over $k$, $\sh{F}$ a coherent $\sh{O}_X$-Module, $k'$ an extension of $k$, for $\mathbf{P}(X,\sh{F},k)$ to be true, it is necessary and sufficient that $\mathbf{P}(X_{(k')},\sh{F}\otimes_k k',k')$ be true.
  \item Let $S$ be an integral noetherian prescheme, of generic point $\eta$, $u:X\to S$ a morphism of finite type, $\sh{F}$ a coherent $\sh{O}_X$-Module.
  For every $s\in S$, set $X_s=u^{-1}(s)=X\times_S\operatorname{Spec}(k(s))$, $\sh{F}_s=\sh{F}\otimes_{\sh{O}_S}k(s)$.
  Let $E$ be the set of $s\in S$ such that $\mathbf{P}(X_s,\sh{F}_s,k(s))$ is true.
  Then one of the two sets $E$, $S-E$ contains an open non-empty set (and consequently is a neighborhood of $\eta$, or $E$ is a neighborhood of $\eta$ if it contains $\eta$).
\end{enumerate}
\end{definition}

\begin{remarks}[9.2.2]
\label{IV.9.2.2}
\oldpage[IV-3]{57}
\emph{(i)}
It is understood here a convention of language and not a mathematical definition in the strict sense.
We have analogous ``definitions'' for relations between $k$, one or several $k$-preschemes, coherent Modules on these preschemes, or homomorphisms between these Modules.

\emph{(ii)}
We shall also have to consider relations involving \emph{constructible parts} of preschemes.
For example, let $\mathbf{P}(X,Z,k)$ be a relation; we say (by abuse of language) that $\mathbf{P}$ is a constructible (resp.\ ind-constructible) property of the constructible part $Z$ of $X$ if the two following conditions are satisfied:
\begin{enumerate}
  \item[1\textsuperscript{o}] If $k$ is a field, $X$ an algebraic prescheme over $k$, $Z$ a \emph{constructible part} of $X$, $k'$ an extension of $k$, then, for $\mathbf{P}(X,Z,k)$ to be true, it is necessary and sufficient that $\mathbf{P}(X_{(k')},p^{-1}(Z),k')$ be true ($p:X_{(k')}\to X$ being the projection).
  \item[2\textsuperscript{o}] Let $S$ be an integral noetherian prescheme, of generic point $\eta$, $u:X\to S$ a morphism of finite type, $Z$ a constructible part of $X$.
  For every $s\in S$, set $X_s=u^{-1}(s)$, $Z_s=Z\cap X_s$.
  Let $E$ be the set of $s\in S$ such that $\mathbf{P}(X_s,Z_s,k(s))$ is true.
  Then one of the two sets $E$, $S-E$ contains an open non-empty set (resp.\ contains an open non-empty set if it contains $\eta$).
\end{enumerate}

We note that in condition 2\textsuperscript{o} one must assume that $Z$ is a constructible part of $X$, and \emph{not merely} a constructible part of $X_s$ for every $s$; the conjunction of these two properties entails the second \sref[0]{0.1.8.2}, but not conversely.

\emph{(iii)}
If $\mathbf{P}$ is a constructible property, it is clear that the same is true of ``non $\mathbf{P}$''.
If $\mathbf{P}$, $\mathbf{Q}$ are two constructible (resp.\ ind-constructible) properties, then so are the properties ``$\mathbf{P}$ or $\mathbf{Q}$'' and ``$\mathbf{P}$ and $\mathbf{Q}$''; indeed, if, under the hypotheses of \sref{IV.9.2.1}[, (ii)], $E$, $E'$ are two parts of $S$ and if $E$ contains an open non-empty set, then so does $E\cup E'$, and if $S-E$ and $S-E'$ each contain an open non-empty set, then so does $(S-E)\cap(S-E')=(S-(E\cup E'))$.

\emph{(iv)}
Let $\mathbf{P}(X,\sh{F},k)$ be a relation satisfying condition \sref{IV.9.2.1}[, (i)]; let $S$ be a prescheme, $u:X\to S$ a morphism of type fini, with the notations of \sref{IV.9.2.1}[, (ii)], and let $E$ be the set of $s\in S$ such that $\mathbf{P}(X_s,\sh{F}_s,k(s))$ is true.
Let moreover $g:S'\to S$ be any morphism, and set $X'=X_{(S')}$, $\sh{F}'=\sh{F}\otimes_{\sh{O}_S}\sh{O}_{S'}$; it follows from the transitivity of the fibres \sref[I]{I.3.6.4} and condition \sref{IV.9.2.1}[, (i)] that the set of $s'\in S'$ such that $\mathbf{P}(X'_{s'},\sh{F}'_{s'},k(s'))$ is true is equal to $g^{-1}(E)$.
This extends immediately to the case where $\mathbf{P}$ involves several preschemes, Modules, morphisms of preschemes or homomorphisms of Modules.

\emph{(v)}
As we shall see in the sequel of this paragraph and in the rest of chap.~IV, most of the properties $\mathbf{P}$ satisfying condition \sref{IV.9.2.1}[, (i)] also satisfy \sref{IV.9.2.1}[, (ii)].
As possible exceptions, note the property of being \emph{projective}, or \emph{affine}, or \emph{quasi-affine} on the base field (for an algebraic prescheme); we shall see \sref{IV.9.6.2} that these properties are \emph{ind-constructible}, but we shall give later an example where $S$ is an open non-empty part of $\operatorname{Spec}(\bb{Z})$ (or an open part of an elliptic curve over a finite field) and where all the fibres $X_s$
\oldpage[IV-3]{58}
except the generic fibre $X_\eta$ are projective over $k(s)$ (all the preschemes $X_s$ being of dimension $2$).
\end{remarks}

\begin{proposition}[9.2.3]
\label{IV.9.2.3}
Let $\mathbf{P}$ be a constructible (resp.\ ind-constructible) property of algebraic preschemes, $S$ a prescheme, $X$ a prescheme of finite presentation over $S$, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module of finite presentation.
Then the set $E$ of $s\in S$ such that $\mathbf{P}(X_s,\sh{F}_s,k(s))$ is true is locally constructible (resp.\ ind-constructible).
Moreover, if $S$ is irreducible of generic point $\eta$, one of the two sets $E$, $S-E$ is a neighborhood of $\eta$ in $S$ (resp.\ $E$ is a neighborhood of $\eta$ if it contains this point).
\end{proposition}

\begin{proof}
To prove these assertions, one can reduce to the case where $S=\operatorname{Spec}(A)$ is affine.
One knows then that there exists a sub-ring $A_0$ of $A$ which is a $\bb{Z}$-algebra of finite type, an $A_0$-prescheme of finite type $X_0$ and a coherent $\sh{O}_{X_0}$-Module $\sh{F}_0$ such that $X$ is isomorphic to $X_0\otimes_{A_0}A$ and $\sh{F}$ to $\sh{F}_0\otimes_{A_0}A$ \sref{IV.8.9.1}.
Let $p:S\to S_0=\operatorname{Spec}(A_0)$ be the morphism corresponding to the injection $A_0\to A$, and let $E_0$ be the set of $s_0\in S_0$ such that
\[
  \mathbf{P}((X_0)_{s_0},(\sh{F}_0)_{s_0},k(s_0))
\]
is true; then, by virtue of \sref{IV.9.2.2}[, (iv)], we have $E=p^{-1}(E_0)$; one can therefore reduce to the case where $S$ is the spectrum of a $\bb{Z}$-algebra of finite type, hence a noetherian scheme.
We use the criterion of constructibility ($\mathbf{0}_{\mathrm{III}}$, 9.2.3) (resp.\ the criterion of ind-constructibility \sref[I]{I.1.9.10}); we are then led, using as in \sref{IV.9.2.2}[, (iv)] above and replacing $S$ by a closed integral \emph{subscheme} of $S$, to the case where $S$ is noetherian and integral, and where it suffices to prove that $E$ is rare in $S$ or contains an open non-empty set of $S$ (resp.\ that $E$ contains an open non-empty set if it contains the generic point); but this is guaranteed by virtue of condition \sref{IV.9.2.1}[, (ii)].
\end{proof}

\begin{corollary}[9.2.4]
\label{IV.9.2.4}
Let $\mathbf{P}$ be a constructible (resp.\ ind-constructible) property of algebraic preschemes, $X$, $Y$ two $S$-preschemes of finite presentation, $f:X\to Y$ an $S$-morphism.
For every $s\in S$, set $X_s=X\times_S\operatorname{Spec}(k(s))$, $Y_s=Y\times_S\operatorname{Spec}(k(s))$, $f_s=f\times 1:X_s\to Y_s$.
Then the set $E$ of $s\in S$ such that the property $\mathbf{P}(f_s^{-1}(y),k(y))$ is true for every $y\in Y_s$ is locally constructible (resp.\ ind-constructible).
\end{corollary}

\begin{proof}
Indeed, let $Z$ be the set of $y\in Y$ such that $\mathbf{P}(f^{-1}(y),k(y))$ is true.
As the fibres $f^{-1}(y)$ and $f_s^{-1}(y)$ are isomorphic, we see that $E$ is the set of $s\in S$ such that $Y_s\subset Z$; if $g:Y\to S$ is the structural morphism, we therefore have $E=S-g(Y-Z)$.
\oldpage[IV-3]{59}
Since $f$ is of finite presentation \sref[I]{I.1.6.2}[, (v)], it follows from \sref{IV.9.2.3} that $Z$ is locally constructible (resp.\ ind-constructible) in $Y$.
As $g$ is of finite presentation, $g(Y-Z)$ is locally constructible (resp.\ pro-constructible) in $S$, by virtue of the theorem of Chevalley \sref[I]{I.1.8.4} (resp.\ of \sref[I]{I.1.9.5}[, (vii)]); hence $E$ is locally constructible (resp.\ ind-constructible) in $S$.
\end{proof}

\begin{remark}[9.2.5]
\label{IV.9.2.5}
We note that if $\mathbf{P}$ is a property of algebraic preschemes for which Proposition~\sref{IV.9.2.3} is true, then $\mathbf{P}$ also satisfies the condition \sref{IV.9.2.1}[, (ii)]: this follows from the fact that in an irreducible noetherian space, a constructible set is either rare or contains an open non-empty set ($\mathbf{0}_{\mathrm{III}}$, 9.2.3).
\end{remark}

\begin{proposition}[9.2.6]
\label{IV.9.2.6}
Denote by $\mathbf{P}$ one of the following properties of a $k$-prescheme algebraic $X$:
\begin{enumerate}[label=(\roman*)]
  \item $X$ is empty.
  \item $X$ is finite over $k$.
  \item $X$ is radiciel over $k$.
  \item $\dim(X)$ belongs to a given part $\Phi$ of the set $\overline{\bb{Z}}=\bb{Z}\cup\{-\infty\}$.
\end{enumerate}
Then $\mathbf{P}$ is constructible.
\end{proposition}

\begin{proof}
It is clear that (i) and (ii) are particular cases of (iv), by taking $\Phi$ to be the set $\{-\infty\}$ and the set $\{-\infty,0\}$ respectively.
It thus suffices to prove (iii) and (iv).
In each of the two cases, condition (i) of \sref{IV.9.2.1} is satisfied by virtue of \sref{IV.2.7.1}[, (xv)] and \sref{IV.4.1.4}.
On the other hand, in case (iii), property $\mathbf{P}$ satisfies the conclusion of \sref{IV.9.2.3} by virtue of \sref{IV.1.8.7}; it remains to see that the same holds in case (iv).
This will follow from the following more precise proposition:
\end{proof}

\begin{proposition}[9.2.6.1]
\label{IV.9.2.6.1}
If $f:X\to S$ is a morphism of finite presentation, the function $s\mapsto\dim(f^{-1}(s))$ is locally constructible.
\end{proposition}

\begin{proof}
The question is local on $S$, so one can assume that $S=\operatorname{Spec}(A)$ is affine and prove that for every $n$, the set $E$ of $s\in S$ such that $\dim(X_s)=n$ is constructible.
The same reasoning as in \sref{IV.9.2.3} reduces us to the case where $A$ is noetherian and integral.
\end{proof}

\begin{corollary}[9.2.6.2]
\label{IV.9.2.6.2}
Let $S$ be an integral noetherian prescheme of generic point $\eta$, $f:X\to S$ a morphism of type fini.
Then there exists a neighborhood $U$ of $\eta$ in $S$ such that the function $s\mapsto\dim(X_s)$ is constant in this neighborhood.
\end{corollary}

\subsection{Constructible properties of morphisms of algebraic preschemes}
\label{subsection:IV.9.3}

\begin{proposition}[9.3.1]
\label{IV.9.3.1}
\oldpage[IV-3]{60}
Let $\mathbf{P}(X,k)$ be a constructible (resp.\ ind-constructible) property of algebraic preschemes.
Denote by $\mathbf{P}'(f,X,Y,k)$ the following relation: $f:X\to Y$ is a $k$-morphism of $k$-preschemes algebraic such that for every $y\in Y$, we have the property $\mathbf{P}(f^{-1}(y),k(y))$.
Then $\mathbf{P}'$ is constructible (resp.\ ind-constructible).
\end{proposition}

\begin{proof}
Indeed, as $\mathbf{P}$ satisfies condition \sref{IV.9.2.1}[, (i)], so does $\mathbf{P}'$ by virtue of the transitivity of the fibres \sref[I]{I.3.6.4}; on the other hand, condition \sref{IV.9.2.1}[, (ii)] follows from \sref{IV.9.2.4}, by virtue of the remark \sref{IV.9.2.5}.
\end{proof}

\begin{proposition}[9.3.2]
\label{IV.9.3.2}
Denote by $\mathbf{P}$ one of the following properties of a $k$-morphism of $k$-preschemes algebraic $f:X\to Y$:
\begin{enumerate}[label=(\roman*)]
  \item $f$ is surjective.
  \item $f$ is quasi-finite.
  \item $f$ is radiciel.
  \item For every $y\in Y$, $\dim(f^{-1}(y))$ belongs to $\Phi$ (notation of \sref{IV.9.2.6}).
\end{enumerate}
Then $\mathbf{P}$ is constructible.
\end{proposition}

\begin{proof}
This follows immediately from \sref{IV.9.3.1} and \sref{IV.9.2.6} if one takes into account that $f$ is of finite type \sref[I]{I.1.5.4}[, (v)], the characterization of radiciel morphisms \sref[I]{I.3.5.8}, and that of quasi-finite morphisms \sref[II]{II.6.2.2}.
\end{proof}

\begin{proposition}[9.3.3]
\label{IV.9.3.3}
Suppose verified the hypotheses of \sref{IV.8.8.1}, and let us moreover retain the notations; suppose moreover that $S_\alpha$ is quasi-compact, $X_\alpha$ and $Y_\alpha$ are of finite presentation over $S_\alpha$, and let $f_\alpha:X_\alpha\to Y_\alpha$ be an $S_\alpha$-morphism.
Let $\mathbf{P}$ be an ind-constructible property of morphisms of algebraic preschemes.
For every $s\in S$ (resp.\ $s_\alpha\in S_\alpha$) set $X_s=X\times_S\operatorname{Spec}(k(s))$, $Y_s=Y\times_S\operatorname{Spec}(k(s))$, $f_s=f_\alpha\times 1:X_s\to Y_s$ (resp.\ $X_{\lambda,s_\alpha}=X_\lambda\times_{S_\lambda}\operatorname{Spec}(k(s_\lambda))$, $Y_{\lambda,s_\alpha}=Y_\lambda\times_{S_\lambda}\operatorname{Spec}(k(s_\lambda))$, $f_{\lambda,s_\alpha}=f_\lambda\times 1:X_{\lambda,s_\alpha}\to Y_{\lambda,s_\alpha}$).
Then, in order that for every $s\in S$ we have $\mathbf{P}(f_s)$, it is necessary and sufficient that there exist $\lambda\geq\alpha$ such that for every $s_\lambda\in S_\lambda$ we have $\mathbf{P}(f_{\lambda,s_\lambda})$.
\end{proposition}

\begin{proof}
Indeed, let $E$ (resp.\ $E_\lambda$) be the set of $s\in S$ (resp.\ $s_\lambda\in S_\lambda$) such that property $\mathbf{P}(f_s)$ (resp.\ $\mathbf{P}(f_{\lambda,s_\lambda})$) is true; it follows from \sref{IV.9.2.2}[, (iv)] that $E_\mu=u_{\lambda\mu}^{-1}(E_\lambda)$ for $\lambda\leq\mu$, and $E=u_\lambda^{-1}(E_\lambda)$; by virtue of \sref{IV.9.2.3}, $E$ (resp.\ $E_\lambda$) is ind-constructible in $S$ (resp.\ $S_\lambda$); the proposition then follows from \sref{IV.8.3.4} applied to the ind-constructible part $E$ of $S$.
\end{proof}

\begin{remark}[9.3.4]
\label{IV.9.3.4}
The combination of \sref{IV.9.3.3} and \sref{IV.9.3.2}[, (ii)] proves the assertion \sref{IV.8.10.5}[, (xi)].
\end{remark}

\begin{proposition}[9.3.5]
\label{IV.9.3.5}
\oldpage[IV-3]{61}
Let $\mathbf{P}(X,Y,k)$ be the property: ``$X$ and $Y$ are two preschemes of finite type over the field $k$, and there exists an extension $k'$ of $k$ and a $k'$-morphism $g:X_{(k')}\to Y_{(k')}$ satisfying $\mathbf{Q}(g)$'', where $\mathbf{Q}$ is one of the properties \emph{(i)} to \emph{(xiv)} of \sref{IV.8.10.5}.
Then $\mathbf{P}$ is an ind-constructible property.
\end{proposition}

\begin{proof}
The definition of $\mathbf{P}$ shows that condition \sref{IV.9.2.1}[, (i)] is satisfied, taking into account that the property $\mathbf{Q}(g)$ is stable under change of the base field, and that two extensions of $k$ can always be considered as sub-extensions of a third extension.
To verify \sref{IV.9.2.1}[, (ii)], one can reduce to the case where $S=\operatorname{Spec}(A)$ is affine; if $K=k(\eta)$, the field of fractions of $A$, and a finite extension $K'$ of $K$ and a $K'$-morphism $g':(X_\eta)_{(K')}\to(Y_\eta)_{(K')}$, satisfying $\mathbf{Q}(g')$, and $K'$ is evidently the field of fractions of an integral $A$-algebra finite $A'$.
If one sets $S'=\operatorname{Spec}(A')$, it follows from \sref{IV.8.10.5} that there is a neighborhood $U'$ of the generic point $\eta'$ of $S'$ such that, if one sets $X'=X\otimes_A A'$, $Y'=Y\otimes_A A'$, there exists for every $s'\in U'$, a morphism $X'_{s'}\to Y'_{s'}$ having property $\mathbf{Q}$.
But the morphism $h:S'\to S$ is finite, hence closed, and since $h^{-1}(\eta)=\{\eta'\}$, $h(U')$ contains a neighborhood $U$ of $\eta$ in $S$; for every $s\in U$, there is therefore an $s'\in U'$ such that $h(s')=s$, and as $X'_{s'}=X_s\otimes_{k(s)}k(s')$, $Y'_{s'}=Y_s\otimes_{k(s)}k(s')$, property $\mathbf{P}(X_s,Y_s,k(s))$ is true for every $s\in U$.
\end{proof}

\begin{example}[9.3.6]
\label{IV.9.3.6}
Take for example $\mathbf{Q}$ the property of being an isomorphism.
Then, combining \sref{IV.9.3.5} and \sref{IV.9.3.3}, one obtains the following property: with the notations and hypotheses being those of \sref{IV.8.8.1}, $S_\alpha$ being quasi-compact, $X_\alpha$ and $Y_\alpha$ of finite presentation on $S_\alpha$, in order that, for every $s\in S$, $X_s$ and $Y_s$ be \emph{geometrically isomorphic} \sref{IV.9.1.4}, it is necessary and sufficient that there exist $\lambda\geq\alpha$ such that, for every $s_\lambda\in S_\lambda$, $X_{\lambda,s_\lambda}$ and $Y_{\lambda,s_\lambda}$ be geometrically isomorphic.

An analogous result holds when the preschemes considered are endowed with ``composition laws'' of a certain kind, for example ``preschemes in groups'', ``preschemes in rings'', etc.\ ($\mathbf{0}_{\mathrm{III}}$, 8.2.1).
Then the preceding statement is also valid when one replaces ``isomorphism'' by ``isomorphisms'' which are also \emph{homomorphisms} for the considered composition laws ($\mathbf{0}_{\mathrm{III}}$, 8.2.2); it suffices here to use not only \sref{IV.8.10.5} but also \sref{IV.8.8.2}[, (i)] by noting that the notion of homomorphism for a composition law is expressed by writing that diagrams of morphisms of preschemes are commutative.

One can also, instead of considering morphisms of preschemes as in \sref{IV.9.3.5}, consider homomorphisms of Modules, using \sref{IV.9.1.6} instead of \sref{IV.9.1.5}.
\end{example}

\subsection{Constructibility of certain properties of modules}
\label{subsection:IV.9.4}

\begin{env}[9.4.1]
\label{IV.9.4.1}
\oldpage[IV-3]{62}
\textbf{Notations.}
In this number and the following ones until the end of \textsection\,9, we shall systematically use the following notations: given a morphism $f:X\to S$, we set, for every $s\in S$, $X_s=f^{-1}(s)=X\times_S\operatorname{Spec}(k(s))$; for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, $\sh{F}_s$ will denote the $\sh{O}_{X_s}$-Module $\sh{F}\otimes_{\sh{O}_S}k(s)$, and for every homomorphism $u:\sh{F}\to\sh{G}$ of $\sh{O}_X$-Modules quasi-coherent, $u_s:\sh{F}_s\to\sh{G}_s$ will be the morphism $p^*(u)$, $p$ denoting the canonical projection $X_s\to X$.
For every section $g$ of $\sh{F}$ over $X$, we will denote by $g_s$ the image of $g$ by the canonical homomorphism $\Gamma(X,\sh{F})\to\Gamma(X_s,\sh{F}_s)$.
For every closed part $Z$ of $X$, we will denote by $Z_s$ the preimage $p^{-1}(Z)=Z\cap X_s$ \sref[I]{I.3.6.1}.
Finally, if $Y$ is a second $S$-prescheme and $h:X\to Y$ an $S$-morphism, we will denote $h_s$ the morphism $h\times 1:X_s\to Y_s$.
\end{env}

\begin{proposition}[9.4.2]
\label{IV.9.4.2}
Let $S$ be an integral noetherian prescheme of generic point $\eta$, $f:X\to S$ a morphism of finite presentation, $\sh{F}$, $\sh{G}$, $\sh{H}$ three quasi-coherent $\sh{O}_X$-Modules of finite presentation.
Let $u:\sh{F}\to\sh{G}$, $v:\sh{G}\to\sh{H}$ be two homomorphisms of $\sh{O}_X$-Modules, and suppose that the sequence
\[
  \sh{F}_\eta\xrightarrow{u_\eta}\sh{G}_\eta\xrightarrow{v_\eta}\sh{H}_\eta
\]
is exact.
Then there exists a neighborhood $U$ of $\eta$ in $S$ such that, for every $s\in U$, the sequence $\sh{F}_s\xrightarrow{u_s}\sh{G}_s\xrightarrow{v_s}\sh{H}_s$ is exact.
\end{proposition}

\begin{proof}
With the general notations of \sref{IV.9.2.1}, it is here the relation $\mathbf{P}(X,\sh{F},\sh{G},\sh{H},u,v,k)$: ``$X$ is an algebraic prescheme over the field $k$, $\sh{F}\xrightarrow{u}\sh{G}\xrightarrow{v}\sh{H}$ is an exact sequence of coherent $\sh{O}_X$-Modules''.
As for every extension $k'$ of $k$, the projection $X_{(k')}\to X$ is a faithfully flat morphism \sref{IV.2.2.13}, the condition \sref{IV.9.2.1}[, (i)] is satisfied \sref{IV.2.2.7}.
By virtue of \sref{IV.9.2.3}, it suffices to reduce to the case where $S$ is affine, noetherian, and integral, and where $\sh{F}$, $\sh{G}$, $\sh{H}$ are coherent $\sh{O}_X$-Modules, and the hypothesis implies that there exists a neighborhood $U$ of $\eta$ in $S$ such that the sequence is exact on $f^{-1}(U)$, applying the general method of \sref{IV.8.1.2}[, $a$)] and the fact that it suffices to prove that $\operatorname{Ker}(v_s)=(\operatorname{Ker} v)_s$ and $\operatorname{Im}(u_s)=(\operatorname{Im} u)_s$ for $s$ close to $\eta$ in $S$; by the flatness results ($\mathbf{0}_{\mathrm{I}}$, 5.3.4), one is led to prove the proposition in the particular case where the sequence $\sh{F}\xrightarrow{u}\sh{G}\xrightarrow{v}\sh{H}\to 0$ is exact.
But then there is an open $U$ in $S$ containing $\eta$ and such that $\sh{H}|f^{-1}(U)$ is flat on $U$ \sref{IV.6.9.1}; it follows from \sref{IV.2.1.8} that for every $s\in U$, the sequence $0\to\sh{F}_s\to\sh{G}_s\to\sh{H}_s\to 0$ is exact, which completes the proof.
\end{proof}

\begin{corollary}[9.4.3]
\label{IV.9.4.3}
Let $S$ be an integral noetherian prescheme, of generic point $\eta$, $f:X\to S$ a morphism of finite presentation.
Let $\sh{L}^\bullet=(\sh{L}^i)_{i\in\bb{Z}}$ be a complex of $\sh{O}_X$-Modules quasi-coherent of finite presentation.
For every $i$, there exists a neighborhood $U$ of $\eta$ in $S$ such that the canonical homomorphisms
\[
  (\sh{H}^i(\sh{L}^\bullet))_s\to\sh{H}^i(\sh{L}^\bullet_s)
\]
are bijective for every $s\in U$.
\end{corollary}

\begin{proof}
\oldpage[IV-3]{63}
One can evidently reduce to a complex of three terms of degrees $-1$, $0$, $+1$: $\sh{M}\xrightarrow{u}\sh{N}\xrightarrow{v}\sh{P}$, and to $i=0$; the homomorphism to consider is then the canonical homomorphism $(\operatorname{Ker} v/\operatorname{Im} u)_s\to\operatorname{Ker}(v_s)/\operatorname{Im}(u_s)$.
Using \sref{IV.8.9.1} and \sref{IV.8.5.2}[, (i)], one sees that one can reduce to the case where $S$ (hence also $X$) is noetherian, and hence $\sh{M}$, $\sh{N}$, $\sh{P}$ are $\sh{O}_X$-Modules \emph{coherent}; then $\operatorname{Im}(u)$ and $\operatorname{Ker}(v)$ are also coherent ($\mathbf{0}_{\mathrm{I}}$, 5.3.4) and there exists moreover a neighborhood $U$ of $\eta$ such that for $s\in U$, we have $\operatorname{Ker}(v_s)=(\operatorname{Ker}(v))_s$ and $\operatorname{Im}(u_s)=(\operatorname{Im}(u))_s$; the conclusion then follows from \sref{IV.9.4.2} applied to the suite exact $0\to(\operatorname{Im} u)_\eta\to(\operatorname{Ker} v)_\eta\to(\operatorname{Ker} v/\operatorname{Im} u)_\eta\to 0$, taking into account the fact that $\sh{O}_\eta=k(\eta)$ (since $S$ is integral) and by the flatness of the suite exact on the right.
\end{proof}

\begin{proposition}[9.4.4]
\label{IV.9.4.4}
Let $f:X\to S$ be a morphism of finite presentation, $\sh{F}$, $\sh{G}$, $\sh{H}$ three $\sh{O}_X$-Modules quasi-coherent of finite presentation.
Let $u:\sh{F}\to\sh{G}$, $v:\sh{G}\to\sh{H}$ be two homomorphisms of $\sh{O}_X$-Modules.
Then the set $E$ of $s\in S$ such that the sequence $\sh{F}_s\xrightarrow{u_s}\sh{G}_s\xrightarrow{v_s}\sh{H}_s$ is exact is locally constructible.
\end{proposition}

\begin{proof}
Taking into account \sref{IV.9.2.3}, it suffices to establish that the property $\mathbf{P}$ considered in \sref{IV.9.4.2} is locally constructible.
We have already remarked in the proof of \sref{IV.9.4.2} that condition \sref{IV.9.2.1}[, (i)] is satisfied, and it remains to verify condition \sref{IV.9.2.1}[, (ii)].
Suppose therefore $S$ integral noetherian of generic point $\eta$, and let us prove that $E$ or $S-E$ is a neighborhood of $\eta$.
If $\eta\in E$, our assertion follows from \sref{IV.9.4.2}, and one can therefore assume $\eta\notin E$, that is, the sequence $\sh{F}_\eta\xrightarrow{u_\eta}\sh{G}_\eta\xrightarrow{v_\eta}\sh{H}_\eta$ is not exact.
We distinguish two cases:

1\textsuperscript{o} Set $w=v\circ u$, and suppose first that $w_\eta=v_\eta\circ u_\eta\neq 0$.
As $\sh{F}$, $\sh{G}$, $\sh{H}$ are coherent, so is $\sh{N}=\operatorname{Ker}(w)$ ($\mathbf{0}_{\mathrm{I}}$, 5.3.4); it follows from \sref{IV.9.4.2} that there is a neighborhood $U$ of $\eta$ in $S$ such that, for $s\in S$, $\operatorname{Ker}(w_s)=\sh{N}_s$; restricting $S$, one can therefore assume this relation is verified for every $s\in S$.
Let $j$ the canonical injection $\sh{N}\to\sh{F}$, and set $\sh{M}=\operatorname{Coker}(j)$; the exactness on the right of the functor $\sh{F}_s$ entails that $\operatorname{M}_s=\operatorname{Coker}(j_s)$ for every $s\in S$; as $M$ is coherent ($\mathbf{0}_{\mathrm{I}}$, 5.3.4), the hypothesis $w_\eta\neq 0$ means that $M_\eta\neq 0$; it follows from \sref{IV.1.8.6} that there is a neighborhood $U$ of $\eta$ in $S$ such that $M_s\neq 0$ for $s\in U$, hence $w_s\neq 0$ for $s\in U$, and therefore $\sh{F}_s$ is not $g_s$-flat, i.e., $S-E$ is a neighborhood of $\eta$.

2\textsuperscript{o} Suppose that $w_\eta=0$; by virtue of \sref{IV.8.5.2}[, (i)], applied following the general method of \sref{IV.8.1.2}[, $a$]), there exists a neighborhood $U$ of $\eta$ such that $w|f^{-1}(U)=0$; replacing $S$ by $U$, one can already suppose $w=0$.
Then $\sh{F}\xrightarrow{u}\sh{G}\xrightarrow{v}\sh{H}$ is a complex of three terms $\sh{L}^\bullet$, to which one can apply \sref{IV.9.4.3}; by hypothesis we have $(\sh{H}^0(\sh{L}^\bullet))_\eta\neq 0$, and $\sh{H}^0(\sh{L}^\bullet)$ is coherent ($\mathbf{0}_{\mathrm{I}}$, 5.3.3 and 5.3.4), so it follows from \sref{IV.1.8.6} that there is a neighborhood $U$ of $\eta$ such that $(\sh{H}^0(\sh{L}^\bullet))_s\neq 0$ for every $s\in U$; but as one can assume that $\sh{H}^0(\sh{L}^\bullet_s)=(\sh{H}^0(\sh{L}^\bullet))_s$ for $s\in U$ by \sref{IV.9.4.3}, one sees that $S-E$ is again a neighborhood of $\eta$ in $S$.
\end{proof}

\begin{corollary}[9.4.5]
\label{IV.9.4.5}
\oldpage[IV-3]{64}
Let $f:X\to S$ be a morphism of finite presentation, $\sh{F}$, $\sh{G}$ two $\sh{O}_X$-Modules quasi-coherent of finite presentation, $u:\sh{F}\to\sh{G}$ a homomorphism of $\sh{O}_X$-Modules.
Then the set of $s\in S$ such that $u_s$ is injective (resp.\ surjective, bijective, zero) is locally constructible.
\end{corollary}

\begin{proof}
It suffices to apply \sref{IV.9.4.4} to the sequences $0\to\sh{F}\xrightarrow{u}\sh{G}$, $\sh{F}\xrightarrow{u}\sh{G}\to 0$, $0\to\sh{F}\xrightarrow{u}\sh{G}\to 0$, $\sh{F}\xrightarrow{u}\sh{G}\xrightarrow{1_\sh{G}}\sh{G}$.
\end{proof}

\begin{corollary}[9.4.6]
\label{IV.9.4.6}
Let $f:X\to S$ be a morphism of finite presentation, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent of finite presentation.
Let $h$ be a section of $\sh{F}$ over $X$; for every $s\in S$, let $h_s$ be the corresponding section of $\sh{F}_s$ over $X_s$ (for the projection morphism $X_s\to X$).
Then, the set of $s\in S$ such that $h_s=0$ is locally constructible.
\end{corollary}

\begin{proof}
It suffices to remark that $h$ corresponds to a homomorphism $u:\sh{O}_X\to\sh{F}$ ($\mathbf{0}_{\mathrm{I}}$, 5.1.1) and $h_s$ to the homomorphism $u_s$.
\end{proof}

\begin{proposition}[9.4.7]
\label{IV.9.4.7}
Let $f:X\to S$ be a morphism of finite presentation, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent of finite presentation.
The set $E$ (resp.\ $E'$) of $s\in S$ such that $\sh{F}_s$ is an $\sh{O}_{X_s}$-Module locally free (resp.\ locally free of rank $n$) is locally constructible.
\end{proposition}

\begin{proof}
If $X$ is an algebraic prescheme over a field $k$, $\sh{F}$ a coherent $\sh{O}_X$-Module, $k'$ an extension of $k$, then, for $\sh{F}$ to be locally free (resp.\ locally free of rank $n$), it is necessary and sufficient that $\sh{F}\otimes_k k'$ also be, since the projection $X_{(k')}\to X$ is a faithfully flat morphism \sref{IV.2.2.7}.
In other words, the condition \sref{IV.9.2.1}[, (i)] is verified for the properties considered.
The constructibility then follows, with the structure of the proof considering four separate cases depending on the generic fiber.
\end{proof}

\begin{lemma}[9.4.7.1]
\label{IV.9.4.7.1}
Let $S$ be an integral noetherian prescheme of generic point $\eta$, $X$, $Y$ two $S$-preschemes of type fini, $g:X\to Y$ an $S$-morphism, $\sh{F}$ a coherent $\sh{O}_X$-Module.
If $\sh{F}_\eta$ is not $g_\eta$-flat, then there exists a neighborhood $U$ of $\eta$ in $S$ such that for every $s\in U$, $\sh{F}_s$ is not $g_s$-flat.
\end{lemma}

\begin{proof}
Taking into account \sref{IV.2.1.2} and Bourbaki, \emph{Alg.~comm.}, chap.~I\textsuperscript{er}, \textsection\,2, n\textsuperscript{o}\,3, Remark~1, the hypothesis means that there exists an open non-empty $V$ of $Y_\eta$, and an $\sh{O}_\eta$-Module coherent $\sh{G}$ and an $\sh{O}_Z$-homomorphism $u:\sh{G}\to\sh{H}$ such that $(1\otimes u)_s$ is not injective for $s\in U$.
Using \sref{IV.8.5.2}[, (i) and (ii)] and the method of \sref{IV.8.1.2}[, $a$)], one can assume that these data are defined over a neighborhood of $\eta$; the result follows from \sref{IV.9.4.5}.
\end{proof}

\begin{remark}[9.4.7.2]
\label{IV.9.4.7.2}
The lemma \sref{IV.9.4.7.1} will later be generalized and freed from noetherian hypotheses \sref{IV.11.2.8}.
\end{remark}

\begin{proposition}[9.4.8]
\label{IV.9.4.8}
\oldpage[IV-3]{65}
Let $S$ be a prescheme locally noetherian, $f:X\to S$ a morphism of type fini, $\sh{F}$ a coherent $\sh{O}_X$-Module.
Suppose that for every $s\in S$, $X_s$ is a locally integral prescheme.
Then the set $E$ (resp.\ $E'$) of $s\in S$ such that $\sh{F}_s$ is an $\sh{O}_{X_s}$-Module of torsion (resp.\ a torsion-free $\sh{O}_{X_s}$-Module) is locally constructible.
\end{proposition}

\begin{proof}
One can evidently assume $S=\operatorname{Spec}(A)$ affine and noetherian and prove $E$ (resp.\ $E'$) is constructible; replacing $S$ by the reduced closed subprescheme having as underlying space a closed part of $S$, and noting \sref[I]{I.3.6.4} that for $s\in Y$, the fibre $(X_{(Y)})_s$ is canonically identified to $X_s$ and the sheaf $(\sh{F}\otimes_{\sh{O}_S}\sh{O}_Y)_s$ to $\sh{F}_s$, one sees that one is brought to the case where $S$ is integral of generic point $\eta$, and has to prove that $E$ or $S-E$ (resp.\ $E'$ or $S-E'$) is a neighborhood of $\eta$.
Note moreover that $X$ is a finite union of affine opens $U_i$, each of the $(U_i)_s$ is induced on an open of $X_s$, hence locally integral; moreover, if $(U_i)_\eta$ is empty (resp.\ non-empty), one knows \sref{IV.9.2.6} that it is also true (resp.\ non-empty) in a neighborhood of $\eta$.
One can therefore assume that all the $(U_i)_\eta$ are non-empty and integral; saying that $\sh{F}_s$ is of torsion (resp.\ torsion-free) is equivalent to saying that each $\sh{F}_s|(U_i)_s$ is of torsion (resp.\ torsion-free).
One is therefore brought to the case where $X=\operatorname{Spec}(B)$ is affine, and $\sh{F}=\widetilde{M}$, where $M$ is a $B$-module of finite type; one will set $B_s=B\otimes_A k(s)$, $M_s=M\otimes_A k(s)$ and one can assume $B_s$ is integral.
We have four cases to consider:

1\textsuperscript{o} $\eta\in E$; $M_\eta$ is then a $B_\eta$-module of torsion of finite type, and there is therefore an element $h\neq 0$ in $B_\eta$ such that $hM_\eta=0$; by virtue of \sref{IV.8.5.2}[, (i)], applied following the method of \sref{IV.8.1.2}[, $a$)], one can (replacing $S$ by a neighborhood of $\eta$) assume that $h\in\Gamma(X,\sh{O}_X)$ is of the form $g_\eta$, where $g\in\Gamma(X,\sh{O}_X)$; let $u:\sh{F}\to\sh{F}$ the endomorphism of $\sh{F}$ defined by $g$; by hypothesis, $u_\eta=0$, hence \sref{IV.9.4.5} the endomorphism $u_s:\sh{F}_s\to\sh{F}_s$, which is the multiplication by $g_s$, is zero in a neighborhood of $\eta$.
On the other hand, let $v:\sh{O}_X\to\sh{O}_X$ the endomorphism defined by the multiplication by $g$; as $B_\eta$ is integral and $h=g_\eta\neq 0$, $v_\eta$ is injective, and it follows hence from \sref{IV.9.4.5} that $v_s$ is an injective endomorphism of $\sh{O}_{X_s}$ for $s$ close to $\eta$, i.e., $g_s$ is an $\sh{O}_{X_s}$-regular element for these values of $s$; hence $\sh{F}_s$ is of torsion in a neighborhood of $\eta$.

2\textsuperscript{o} $\eta\in S-E$.
Say that a $B_\eta$-module of finite type $M_\eta$ is not a module of torsion means that its quotient $M_\eta/T$ by its torsion submodule is $\neq 0$, and as it is a torsion-free module of finite type, it is isomorphic to a submodule of $B_\eta^n$; there exists therefore an injective homomorphism $w:M_\eta\to B_\eta^n$ which is $\neq 0$.
Applying \sref{IV.8.5.2}[, (i)] and \sref{IV.9.4.5} as in 1\textsuperscript{o}, there exists a homomorphism $v:\sh{F}\to\sh{O}_X^n$ such that $v_\eta=\widetilde{w}$, and such that for $s$ close to $\eta$, $v_s:\sh{F}_s\to\sh{O}_{X_s}^n$ is injective; for these values of $s$, $\sh{F}_s$ is therefore torsion-free.

3\textsuperscript{o} As $M_\eta$ is a $B_\eta$-module of finite type with torsion, i.e., it is not locally free, following from the analysis above there exists a homomorphism $v:\sh{F}\to\sh{O}_X^n$ such that $v_\eta$ is injective, and such that for $s$ in a neighborhood of $\eta$, $v_s:\sh{F}_s\to\sh{O}_{X_s}^n$ is injective; for these values of $s$, $\sh{F}_s$ is torsion-free.

4\textsuperscript{o} $\eta\in S-E'$.
Let $T$ the torsion submodule of $M_\eta$; by hypothesis $T\neq 0$, and $T$ is of finite type since $M_\eta$ is noetherian.
Using this time \sref{IV.8.5.2}[, (i) and (ii)] a coherent $\sh{O}_X$-Module $\sh{G}$ and an injective homomorphism $u:\sh{G}\to\sh{F}$ such that $\sh{G}_\eta=\widetilde{T}$ and that $u_\eta$ is the canonical injection $\widetilde{T}\to\sh{F}_\eta$.
It follows from 1\textsuperscript{o} and from \sref{IV.1.8.6} that in a neighborhood of $\eta$, $\sh{G}_s$ is a torsion $\sh{O}_{X_s}$-Module $\neq 0$, and on the other hand it follows from \sref{IV.9.4.5} that in a neighborhood of $\eta$, $u_s$ is injective.
We conclude that in a neighborhood of $\eta$, the torsion submodule of $\sh{F}_s$ is not zero.
\end{proof}

\begin{remark}[9.4.9]
\label{IV.9.4.9}
\oldpage[IV-3]{66}
The property ``$X$ is a $k$-prescheme algebraic locally integral'' does not satisfy condition \sref{IV.9.2.1}[, (i)], and it is therefore not certain that the statement \sref{IV.9.4.8} remains valid when one makes no hypothesis on $S$ and one only assumes that $f$ is a morphism of finite presentation and $\sh{F}$ an $\sh{O}_X$-Module of finite presentation.
Let us consider however the following particular case: $S_0$ being a locally noetherian prescheme, let $f_0:X_0\to S_0$ a morphism of type fini, such that the fibres $(X_0)_{s_0}$ are locally integral (for every $s_0\in S_0$), and $\sh{F}_0$ a coherent $\sh{O}_{X_0}$-Module; let $g:S\to S_0$ be any morphism, set $X=X_0\times_{S_0}S$, $F=\sh{F}_0\otimes_{\sh{O}_{S_0}}\sh{O}_S$, $f=f_0\times 1_S:X\to S$, and \emph{suppose} that for every $s\in S$, the fibre $X_s$ is again locally integral.
Then the set $E$ (resp.\ $E'$) of $s\in S$ such that $\sh{F}_s$ is of torsion (resp.\ torsion-free) is again \emph{locally constructible}.
\end{remark}

\subsection{Constructibility of topological properties}
\label{subsection:IV.9.5}

\begin{proposition}[9.5.1]
\label{IV.9.5.1}
\oldpage[IV-3]{67}
Let $f:X\to S$ be a morphism of finite presentation, $Z$ a locally constructible part of $X$.
Then the set $E$ of $s\in S$ such that $Z_s\neq\emp$ is locally constructible.
\end{proposition}

\begin{proof}
Indeed, $E=f(Z)$, and it suffices to apply the theorem of Chevalley \sref[I]{I.1.8.4}.
\end{proof}

\begin{corollary}[9.5.2]
\label{IV.9.5.2}
If $Z'$, $Z''$ are two locally constructible parts of $X$, the set of $s\in S$ such that $Z'_s\subset Z''_s$ (resp.\ $Z'_s=Z''_s$) is locally constructible.
\end{corollary}

\begin{proof}
Indeed, the relation $Z'_s\subset Z''_s$ is equivalent to $(Z'\cap(\complement Z''))_s=\emp$ and $Z'\cap\complement Z''$ is locally constructible.
\end{proof}

\begin{proposition}[9.5.3]
\label{IV.9.5.3}
Let $f:X\to S$ be a morphism of finite presentation, $Z$, $Z'$ two locally constructible parts of $X$ such that $Z\subset Z'$.
Then the set $E$ of $s\in S$ such that $Z_s$ is dense in $Z'_s$ is locally constructible in $S$.
\end{proposition}

\begin{proof}
We must verify the two conditions of \sref{IV.9.2.2}[, (ii)].
For the first condition, concerning an algebraic prescheme $X$ over a field $k$, two constructible parts $Z\subset Z'$, and an extension $k'$ of $k$, the projection $p:X_{(k')}\to X$ is faithfully flat and quasi-compact, hence $p^{-1}(\overline{Z})=\overline{p^{-1}(Z)}$ and $p^{-1}(\overline{Z'})=\overline{p^{-1}(Z')}$ by virtue of \sref{IV.2.3.10}; as $p$ is surjective, the relation $\overline{Z}=\overline{Z'}$ is equivalent to $\overline{p^{-1}(Z)}=\overline{p^{-1}(Z')}$.

Let us now verify the second condition, supposing $S$ affine, noetherian, integral, and of generic point $\eta$.
We distinguish two cases:

1\textsuperscript{o} $\eta\in S-E$, hence $Z_\eta$ is not dense in $Z'_\eta$.
Montrons first that one can assume $Z'$ closed.
We have $\overline{Z'_\eta}=\overline{Z_\eta}$ (closure in $X_\eta$); set $V_\eta=X_\eta-\overline{Z_\eta}$, which is open in $X_\eta$ and does not meet $Z'_\eta$; one can assume $V_\eta$ of the form $V\cap X_\eta$, where $V$ is open (hence constructible) in $X$, and the hypothesis $V_\eta\cap Z'_\eta=\emp$ entails then by \sref{IV.9.5.1} that $V_s\cap Z'_s=\emp$ for $s\in U$; hence $Z_s$ is not dense in $Z'_s$ for $s\in U$, otherwise $U\subset S-E$ is a neighborhood of $\eta$.

2\textsuperscript{o} $\eta\in E$, hence $Z_\eta$ is dense in $Z'_\eta$.
It suffices to show that one can assume $Z'=Z$, and this follows from the method of reducing to irreducible closed parts.
\end{proof}

\begin{corollary}[9.5.4]
\label{IV.9.5.4}
Let $f:X\to S$ be a morphism of finite presentation, $Z$ a locally constructible part of $X$.
The set $E$ of $s\in S$ such that $Z_s$ is open (resp.\ closed, resp.\ locally closed) in $X_s$ is locally constructible in $S$.
\end{corollary}

\begin{proof}
Saying that $Z_s$ is open in $X_s$ means that $(X-Z)_s=X_s-Z_s$ is closed in $X_s$, and as $X-Z$ is locally constructible, one can reduce to considering the set of $s\in S$ such that $Z_s$ is closed, and the set of $s$ such that $Z_s$ is locally closed.
We verify in each case the two conditions of \sref{IV.9.2.2}[, (ii)].
The first condition is a consequence of the fact that $p:X_{(k')}\to X$ is faithfully flat and quasi-compact, and of \sref{IV.2.3.12}.
We then verify the second condition, assuming $S$ affine, noetherian, and integral.
\end{proof}

\begin{proposition}[9.5.5]
\label{IV.9.5.5}
\oldpage[IV-3]{69}
Let $f:X\to S$ be a morphism of finite presentation, $Z$ a locally constructible part of $X$.
For every $s\in S$, let $\mathrm{D}(s)\subset\bb{Z}\cup\{-\infty\}$ be the set of dimensions of the irreducible components of $Z_s$.
Then the function $s\mapsto\mathrm{D}(s)$ is locally constructible in $S$.
\end{proposition}

\begin{corollary}[9.5.6]
\label{IV.9.5.6}
Let $S$ be a prescheme noetherian and irreducible of generic point $\eta$, $X$ a prescheme irreducible, $f:X\to S$ a morphism dominant of type fini.
Then there exists a neighborhood $U$ of $\eta$ in $S$ such that, for $s\in U$, all the irreducible components of $X_s$ are of dimension $n=\dim(X_\eta)$.
\end{corollary}

\begin{remark}[9.5.7]
\label{IV.9.5.7}

One must be careful that under the hypotheses of \sref{IV.9.5.6}, it can happen that $X_s$ is \emph{not irreducible for any $s\neq\eta$} in a neighborhood of $\eta$; in other words, the property ``$X$ is an irreducible $k$-prescheme algebraic'' is \emph{not constructible}.
\end{remark}

\subsection{Constructibility of certain properties of morphisms}
\label{subsection:IV.9.6}

\begin{proposition}[9.6.1]
\label{IV.9.6.1}
\oldpage[IV-3]{71}
Let $X$, $Y$ be two $S$-preschemes of finite presentation, $f:X\to Y$ an $S$-morphism.
Let $E$ be the set of $s\in S$ for which $f_s$ has one of the following properties:
\begin{enumerate}[label=(\roman*)]
  \item being surjective;
  \item being dominant;
  \item being separated;
  \item being proper;
  \item being radiciel;
  \item being finite;
  \item being quasi-finite;
  \item being an immersion;
  \item being a closed immersion;
  \item being an open immersion;
  \item being an isomorphism;
  \item being a monomorphism.
\end{enumerate}
Then $E$ is locally constructible in $S$.
\end{proposition}

\begin{proof}
The assertions for (i), (v) and (vii) are inserted here only as a reminder, having already been established in \sref{IV.9.3.2}.

(ii): Note first that $f$ is of finite presentation \sref[I]{I.1.6.2}[, (v)], hence, by the theorem of Chevalley \sref[I]{I.1.8.4}, $f(X)$ is locally constructible in $Y$.
On the other hand, $f_s(X_s)=(f(X))_s$ \sref[I]{I.3.6.1}; the set of $s$ such that $f_s$ is dominant is the set of $s$ such that $(f(X))_s$ is dense in $Y_s$; the conclusion follows from \sref{IV.9.5.3}.

(iii): As $f:X\to Y$ is of finite presentation, the diagonal immersion $\Delta_f:X\to X\times_Y X$ is of finite presentation \sref[I]{I.1.6.2}[, (iv) and (v)]; it follows hence from \sref{IV.1.8.4.1} that $\Delta_f(X)=\Delta_X$ is a locally constructible part of $X\times_Y X$.
Saying that $f_s$ is separated means (taking into account \sref[I]{I.5.3.4}) that $(\Delta_X)_s$ is closed in $X_s\times_{Y_s}X_s=(X\times_Y X)_s$; the conclusion follows from \sref{IV.9.5.4}.

(iv): We show that the property ``$X$ and $Y$ are algebraic preschemes over a field $k$, and $f:X\to Y$ a $k$-morphism proper'' is constructible.
Condition \sref{IV.9.2.1}[, (i)] is verified by virtue of \sref{IV.2.7.1}[, (vii)].
It thus remains to verify \sref{IV.9.2.1}[, (ii)], hence one can assume $S$ is affine, noetherian, and integral of generic point $\eta$, and one must prove that $E$ or $S-E$ is a neighborhood of $\eta$.
The proof considers separately the cases where $\eta\in E$ and $\eta\notin E$.

(vi): The property is the conjunction of the properties (iv) and (vii) \sref{IV.8.11.1}, hence the proposition follows from what has already been proved.

(ix): Verification of condition \sref{IV.9.2.1}[, (i)] follows from \sref{IV.2.7.1}[, (xi)].
One can reduce to the case where $S$ is affine, noetherian, and integral of generic point $\eta$, and prove that $E$ or $S-E$ is a neighborhood of $\eta$.
\oldpage[IV-3]{72}

(viii): The verification of condition \sref{IV.9.2.1}[, (i)] is done as in (ix), using \sref{IV.2.7.1}[, (x)] and the fact that every immersion of one prescheme noetherian into another is quasi-compact.

(x): Using this time \sref{IV.2.7.1}[, (ix)] and \sref{IV.8.10.5}[, (iii)], we are brought to the case where $S$ is affine, noetherian, integral of generic point $\eta$, and where $\eta\in S-E$.
We distinguish three cases:
1\textsuperscript{o} $f_\eta(X_\eta)$ is not a locally closed part of $Y_\eta$.
2\textsuperscript{o} $f_\eta(X_\eta)$ is locally closed in $Y_\eta$ but $f_\eta$ is not an immersion.
3\textsuperscript{o} $f_\eta(X_\eta)$ is open in $Y_\eta$ and $f_\eta$ is an immersion.

(xi): The property is the conjunction of (i) and (x) and the conclusion follows from what has been proved.

(xii): By virtue of \sref[I]{I.5.3.8}, saying that $f_s$ is a monomorphism means that $\Delta_{f_s}=(\Delta_f)_s:X_s\to X_s\times_{Y_s}X_s=(X\times_Y X)_s$ is an isomorphism, and as $X\times_Y X$ is an $S$-prescheme of finite presentation \sref[I]{I.1.6.2}[, (iv)], the conclusion follows from (xi).
\end{proof}

\begin{proposition}[9.6.2]
\label{IV.9.6.2}
\oldpage[IV-3]{73}
Let $X$, $Y$ be two $S$-preschemes of finite presentation, $f:X\to Y$ an $S$-morphism.
\begin{enumerate}[label=\textup{I)}]
  \item Let $E$ be the set of $s\in S$ for which $f_s$ has one of the following properties:
  \begin{enumerate}[label=(\roman*)]
    \item being affine;
    \item being quasi-affine;
    \item being projective;
    \item being quasi-projective.
  \end{enumerate}
  Then $E$ is ind-constructible in $S$.
\end{enumerate}
\begin{enumerate}[label=\textup{II)}]
  \item Let $\sh{L}$ be an $\sh{O}_X$-Module inversible.
  Then the set $E'$ of $s\in S$ such that $\sh{L}_s$ is an $\sh{O}_{X_s}$-Module ample (resp.\ very ample) relatively to $f_s$ is ind-constructible in $S$.
\end{enumerate}
\end{proposition}

\begin{proof}
I) We verify conditions (i) and (ii) of \sref{IV.9.2.1}.
For the condition \sref{IV.9.2.1}[, (i)], it follows, for properties (i) and (ii), from \sref{IV.2.7.1}[, (xiii) and (xiv)]; for the properties (iii) and (iv), it follows from \sref{IV.9.1.5}.
We then verify \sref{IV.9.2.1}[, (ii)], assuming $S$ noetherian, integral of generic point $\eta$, and that $f_\eta:X_\eta\to Y_\eta$ has one of the properties (i) to (iv) of the statement.
Applying \sref{IV.8.10.5}[, (viii), (ix), (xiii) and (xiv)], one sees that there exists a neighborhood $U$ of $\eta$ such that, if $V$ and $W$ are the preimages of $U$ in $X$ and $Y$ respectively by the structural morphisms, the restriction $V\to W$ of $f$ has the same property.
The conclusion follows from the stability of these properties under base change \sref[II]{II.4.6.13} and \sref{IV.4.4.10}.

II) One proceeds similarly.
For condition \sref{IV.9.2.1}[, (i)] it follows this time from \sref{IV.2.7.2}.
For the condition \sref{IV.9.2.1}[, (ii)], with the same notations as in I), it follows from \sref{IV.8.10.5}[, (vi) and (x)], applied following the method of \sref{IV.8.1.2}[, $a$)]), that there exists a neighborhood $U$ of $\eta$ in $S$ such that the restriction $\sh{L}|V$ is ample (resp.\ very ample) relatively to the restriction $V\to W$ of $f$.
The conclusion follows from the stability of the properties considered under base change \sref[II]{II.4.6.13} and the stability of amplitude by base change \sref[III]{III.4.7.1} and \sref[II]{II.4.4.10}.
\end{proof}

\subsection{Constructibility of properties of separability, geometric irreducibility, and geometric connectedness}
\label{subsection:IV.9.7}

\begin{lemma}[9.7.1]
\label{IV.9.7.1}
\oldpage[IV-3]{76}
Let $S$ be an irreducible prescheme of generic point $\eta$, $f:X\to S$ a morphism of finite presentation.
If $X_\eta$ has $n$ irreducible components (resp.\ $n'$ connected components), there exists a neighborhood $U$ of $\eta$ in $S$ such that for every $s\in U$, $X_s$ has at least $n$ irreducible components (resp.\ at least $n'$ connected components).
\end{lemma}

\begin{proof}
One can reduce to the case where $S$ is affine, hence $X$ quasi-compact and quasi-separated.
Let $V_i$ ($1\leq i\leq n$) be the interiors of the irreducible components of $X_\eta$, which are two by two disjoint and quasi-compact, and whose reunion is dense in $X_\eta$; by virtue of \sref{IV.8.2.11}, applied following the method of \sref{IV.8.1.2}[, $a$)]), there exist for each $i$ open quasi-compact $W_i$ in $X$ such that $W_i\cap X_\eta=V_i$; as $X$ is quasi-separated, the intersections $W_i\cap W_j$ are quasi-compact \sref[I]{I.1.2.7}, hence, replacing $S$ by a neighborhood of $\eta$, one can assume $W_i\cap W_j=\emp$ for $i\neq j$ by virtue of \sref{IV.8.3.3}.
Moreover, as the $W_i$ are constructible, there is a neighborhood $U$ of $\eta$ in $S$ such that for $s\in U$ the $(W_i)_s$ are non-empty \sref{IV.9.5.1} and \sref{IV.9.2.3} and that the reunion of the $(W_i)_s$ is dense \sref{IV.9.5.3} and \sref{IV.9.2.3}.
The rest of the proof follows by considering the components of $X_s$.

The case of connected components is treated similarly, replacing the $V_i$ by the connected components $C_j$ ($1\leq j\leq n'$) of $X_\eta$.
\end{proof}

\begin{lemma}[9.7.2]
\label{IV.9.7.2}
\oldpage[IV-3]{77}
Let $S$ be an irreducible prescheme of generic point $\eta$, $f:X\to S$ a morphism of finite presentation.
If $X_\eta$ is not reduced, there exists a neighborhood $U$ of $\eta$ in $S$ such that, for every $s\in U$, $X_s$ is not reduced.
\end{lemma}

\begin{proof}
Indeed, let $\sh{N}$ the Nilradical of $\sh{O}_X$; it follows from \sref{IV.8.2.13}, applied following the method of \sref{IV.8.1.2}[, $a$)]) that $\sh{N}_\eta$ is the nilradical of $\sh{O}_{X_\eta}$, and the hypothesis is that $\sh{N}_\eta\neq 0$.
We conclude from \sref{IV.9.4.5} and \sref{IV.9.2.3} that there is a neighborhood $U$ of $\eta$ such that, for every $s\in U$, $\sh{N}_s\neq 0$; and as $\sh{N}_s$ is evidently contained in the Nilradical of $\sh{O}_{X_s}$, one sees that $X_s$ is not reduced for $s\in U$.
\end{proof}

\begin{lemma}[9.7.4]
\label{IV.9.7.4}
Let $k$ be a field, $\Omega$ an algebraically closed extension of $k$, $F$ a non-constant polynomial of $k[T_1,\ldots,T_n]$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{(\alph*)}]
  \item $X=\operatorname{Spec}(k[T_1,\ldots,T_n]/(F))$ is geometrically integral.
  \item $F_{(K)}$ is irreducible for every extension $K$ of $k$.
  \item $F_{(\Omega)}$ is irreducible.
\end{enumerate}
We say in this case that $F$ is \emph{geometrically irreducible}.
\end{lemma}

\begin{lemma}[9.7.5]
\label{IV.9.7.5}
\oldpage[IV-3]{78}
Let $A$ be an integral ring, $K$ its field of fractions, $F$ a non-constant polynomial of $A[T_1,\ldots,T_n]$ of degree $d$ such that $F_{(K)}$ is geometrically irreducible.
Then there exists $f\neq 0$ in $A$ such that for every $x\in\mathrm{D}(f)$, $F_{(k(x))}$ is geometrically irreducible.
\end{lemma}

\begin{lemma}[9.7.5.1]
\label{IV.9.7.5.1}
Let $A$ be an integral ring, $S=\operatorname{Spec}(A)$, $\eta$ the generic point of $S$, $B$ the polynomial ring $A[T_1,\ldots,T_m]$, $(P_i)_{i\in I}$ a finite family of elements of $B$, $\mathfrak{a}$ the ideal of $B$ generated by the $P_i$.
For every $s\in S$, let $\mathfrak{a}_s$ the ideal of $k(s)[T_1,\ldots,T_m]$ generated by the polynomials $(P_i)_{(k(s))}$ ($i\in I$).
Then, if $V(\mathfrak{a}_\eta)=\emp$, there exists a neighborhood $U$ of $\eta$ in $S$ such that $V(\mathfrak{a}_s)=\emp$ for every $s\in U$.
\end{lemma}

\begin{proof}
Indeed, let $Y=\operatorname{Spec}(B)$, and let $Z$ the closed part $V(\mathfrak{a})$ of $Y$; as $\mathfrak{a}$ is an ideal of finite type, $Z$ is constructible in $Y$ ($\mathbf{0}_{\mathrm{III}}$, 9.1.5) and with the notations introduced in \sref{IV.9.4.1}, $V(\mathfrak{a}_s)=Z_s$ for every $s\in S$; as the structural morphism $Y\to S$ is of finite presentation, the conclusion follows from \sref{IV.9.5.1} and \sref{IV.9.2.3}.
\end{proof}

\begin{proposition}[9.7.6]
\label{IV.9.7.6}
Let $S$ be a noetherian prescheme integral of generic point $\eta$, $f:X\to S$ a morphism of type fini, $\sh{F}$ a coherent $\sh{O}_X$-Module.
If $\sh{F}_\eta$ is without first associated immersed cycle, then there exists a neighborhood $U$ of $\eta$ in $S$ such that, for every $s\in U$, $\sh{F}_s$ is without first associated immersed cycle.
\end{proposition}

\begin{theorem}[9.7.7]
\label{IV.9.7.7}
\oldpage[IV-3]{79}
Let $f:X\to S$ be a morphism of finite presentation, and let $E$ be the set of $s\in S$ for which $X_s$ has one of the following properties:
\begin{enumerate}[label=(\roman*)]
  \item being geometrically irreducible;
  \item being geometrically connected;
  \item being geometrically reduced;
  \item being geometrically integral.
\end{enumerate}
Then $E$ is locally constructible in $S$.
\end{theorem}

\begin{proof}
To see that the properties considered are constructible, remark first that they trivially verify condition \sref{IV.9.2.1}[, (i)], by virtue of \sref{IV.4.5.6}[, (i)] and \sref{IV.4.6.5}[, (i)].
It thus remains to verify \sref{IV.9.2.1}[, (ii)], hence one can assume that $S=\operatorname{Spec}(A)$ is affine noetherian and integral of generic point $\eta$.
By virtue of \sref{IV.4.6.8}, there exists a finite extension $K'$ of $K=k(\eta)$ such that $(X_\eta)_{(K')}$ has its connected components (resp.\ irreducible components) geometrically irreducible (resp.\ geometrically connected) and that $((X_\eta)_{(K')})_{\mathrm{red}}$ is geometrically reduced.

\oldpage[IV-3]{80}
Suppose first that $\eta\in S-E$ for one of the properties (i) to (iv).
The proof proceeds by showing that $X_s$ is not geometrically irreducible (resp.\ connected, reduced, integral) in a neighborhood of $\eta$, using the lemmas \sref{IV.9.7.1} and \sref{IV.9.7.2}.

Now suppose that $\eta\in E$ and examine each of the four properties separately.

1\textsuperscript{o} Suppose $X_\eta$ geometrically integral.
The proof uses the function field and the lemma on geometrically irreducible polynomials \sref{IV.9.7.4} and \sref{IV.9.7.5}.

2\textsuperscript{o} Suppose $X_\eta$ geometrically irreducible.
One replaces $X$ by $X_{\mathrm{red}}$ \sref[I]{I.1.5.8} and concludes from 1\textsuperscript{o}.

\oldpage[IV-3]{81}
3\textsuperscript{o} Suppose $X_\eta$ geometrically connected.
The proof uses the surjection \sref{IV.5.10.8.1} applied to $\mathfrak{F}=\{\emp\}$ and the characterization of geometric connectedness \sref{IV.4.5.13.1}.

4\textsuperscript{o} Suppose $X_\eta$ geometrically reduced.
The proof uses the results on geometrically irreducible components and \sref{IV.9.7.6}.
\end{proof}

\begin{proposition}[9.7.8]
\label{IV.9.7.8}
\oldpage[IV-3]{82}
Let $S$ be an irreducible prescheme of generic point $\eta$, and $f:X\to S$ a morphism of finite presentation.
Let $n$ (resp.\ $n'$) be the geometric number of irreducible components (resp.\ connected components) of $X_\eta$ \sref{IV.4.5.2}.
Then there exists a neighborhood $U$ of $\eta$ in $S$ such that for every $s\in U$, the geometric number of irreducible components (resp.\ connected components) of $X_s$ is equal to $n$ (resp.\ $n'$).

Let $f:X\to S$ be a morphism of finite presentation.

\begin{corollary}[9.7.10]
\label{IV.9.7.10}
Let $X$ be a locally noetherian prescheme such that, if $X'$ is the normalization of $X_{\mathrm{red}}$, the canonical morphism $f:X'\to X$ is finite.
Then the set of points $x\in X$ such that $X$ is geometrically unibranch at the point $x$ is locally constructible in $X$.
\end{corollary}

\begin{proof}
Indeed, this set is by definition the set of points $x\in X$ such that the number of geometric points of $f^{-1}(x)$ is equal to $1$.
But as $f$ is finite, this number is also the number of geometric irreducible components of the discrete space $f^{-1}(x)$ \sref[II]{II.6.3.8} and of \sref{IV.4.5.11}; the conclusion follows from \sref{IV.9.7.9}.

\begin{proposition}[9.7.12]
\label{IV.9.7.12}
Let $f:X\to S$ be a morphism of finite presentation, and $g:S\to X$ a $S$-section of $X$ \sref[I]{I.2.5.5}.
For every $s\in S$, let $X^0_s$ be the connected component of $f^{-1}(s)$ which contains $g(s)$; then, the reunion $X^0$ of the $X^0_s$ for $s\in S$ is a locally constructible part of $X$.
\end{proposition}

\begin{proof}
We first show that one can always reduce to the case where $S=\operatorname{Spec}(A)$ is affine; with the notations of \sref{IV.9.2.3}, we have $f=(f_0)_{(S)}$, where $f_0:X_0\to S_0$ is a morphism of type fini, and one can moreover assume that there exists a $S_0$-section $g_0:S_0\to X_0$ such that $g=(g_0)_{(S)}$ \sref{IV.8.9.1}.
\oldpage[IV-3]{83}
Let $x$ be a point of $X$, $Y$ the sub-prescheme reduced of $S$ having $\overline{\{f(x)\}}$ as underlying space, and $Y'$ the sub-prescheme reduced of $S$ having $\overline{f(Z)}$ as underlying space; as the restriction of $f$ to $Z$ factorizes in $Z\to Y\to S$ \sref[I]{I.5.2.2}, one can assume, replacing $S$ by $Y$, that $f(x)=\eta$, the generic point of the integral prescheme $S$.
By hypothesis, $X_\eta$ is the sum of two sub-preschemes $X^0_\eta$, $X^1_\eta$, induced on open complementary sets of $X_\eta$.
As $g:S\to X$ is continuous and injective, one of these two opens is empty, and as $g(\eta)\in X^0_\eta$ by definition, we have $g^{-1}(X^1)=\emp$.
In other words, $g$ is a $S$-section of $X^0$; on the other hand, as $(X^0)_\eta$ is geometrically connected \sref{IV.4.5.13}, it follows that for $s$ close to $\eta$ by \sref{IV.9.7.7}, $(X^0)_s=X^0_s$.
\end{proof}

\subsection{Primary decomposition in a neighborhood of a generic fibre}
\label{subsection:IV.9.8}

\begin{proposition}[9.8.1]
\label{IV.9.8.1}
\oldpage[IV-3]{83}
Let $f:X\to S$ be a morphism of finite presentation, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent of finite presentation.
Then the set $E$ of $s\in S$ such that $\sh{F}_s$ is an $\sh{O}_{X_s}$-Module without first associated immersed cycle is locally constructible.
\end{proposition}

\begin{proof}
One knows that for quasi-coherent Modules on algebraic preschemes, the property of being without first associated immersed cycle is invariant by extension of the base field \sref{IV.4.2.7}; one can therefore reduce to the case where $S$ is affine, noetherian, and integral of generic point $\eta$, and has to prove that $E$ or $S-E$ is a neighborhood of $\eta$.
\oldpage[IV-3]{84}
One has seen in \sref{IV.9.7.6} that if $\eta\in E$, $E$ is a neighborhood of $\eta$; it remains to consider the case where $\sh{F}_\eta$ has associated immersed cycles.
One can reduce to the case where $X$ is affine, and where there is a coherent sub-$\sh{O}_{X_\eta}$-Module $\sh{H}$ of $\sh{F}_\eta$ whose support $Z'$ is non-empty and \emph{rare} with respect to the support $Z$ of $\sh{F}$ \sref{IV.3.1.3};
then, by virtue of (\sref{IV.8.5.2}, (i) and (ii)) and \sref{IV.8.5.8}, applied following the method of (\sref{IV.8.1.2}, $a$)), there exists a coherent sub-$\sh{O}_X$-Module $\sh{G}$ of $\sh{F}$ such that $\sh{G}_\eta=\sh{H}$;
if $Y=\Supp(\sh{F})$, $Y'=\Supp(\sh{G})$, $Y'_s=\Supp(\sh{G}_s)$ for all $s\in S$ \sref[I]{I.9.1.13} and in particular $Z=Y_\eta$;
since $Z-Z'$ is dense in $Z$ and $Z'\neq\emp$, there is a neighborhood $U$ of $\eta$ such that for $s\in U$, $Y_s-Y'_s$ is dense in $Y_s$, and $Y'_s\neq\emp$ (\sref{IV.9.5.1} and \sref{IV.9.5.3});
by considering a neighborhood of an irreducible component of $Y'_s$ and a sufficiently small neighborhood of this point in $X_s$, we immediately deduce from \sref{IV.3.1.3} that $\sh{F}_s$ admits first associated immersed cycles.
\end{proof}

\begin{env}{9.8.2}
\label{IV.9.8.2}
Let $S$ be a noetherian integral prescheme, with generic point $\eta$, $f:X\to S$ a morphism of finite type, $\sh{F}$ a coherent $\sh{O}_X$-Module.
Consider an irredundant reduced decomposition $(\sh{G}_i)_{i\in I}$ of $\sh{F}_\eta$ \sref{IV.3.2.6};
the $\sh{G}_i$ are thus quotients of $\sh{F}_\eta$ and there is an injective homomorphism $h:\sh{F}_\eta\to\bigoplus\sh{G}_i$;
moreover $\operatorname{Ass}(\sh{G}_i)$ is reduced to a single point $x_i$.
Let $Z_i$ be the closure of $\{x_i\}$ in $X$, so that $(Z_i)_\eta=\Supp(\sh{G}_i)$.
Denote by $\sh{J}_i$ the coherent Ideal of $\sh{O}_X$ defining the closed reduced sub-prescheme of $X$ with underlying space $Z_i$ (sub-prescheme also denoted $Z_i$).
By virtue of \sref{IV.8.5.2} and \sref{IV.8.5.8}, applied following the method of (\sref{IV.8.1.2}, $a$)), we can (restricting $S$ to a neighborhood of $\eta$ if necessary) suppose that there exist quotients $\sh{F}_i$ of $\sh{F}$ ($i\in I$) such that $\sh{G}_i=(\sh{F}_i)_\eta$ for every $i$, and a homomorphism $g:\sh{F}\to\bigoplus\sh{F}_i$ such that $h=g_\eta$.
Moreover \sref[I]{I.9.3.5}, there exists an integer $m$ such that $(\sh{J}_i^m\sh{F}_i)_\eta=(\sh{J}_i)_\eta^m\sh{G}_i=0$ and, restricting $S$ again, we can therefore assume that $\sh{J}_i^m\sh{F}_i=0$ \sref{IV.8.5.2.5}, hence that the support of $\sh{F}_i$ is contained in $Z_i$.
\end{env}

\begin{proposition}[9.8.3]
\label{IV.9.8.3}
Under the conditions of \sref{IV.9.8.2}, for every $s\in S$ and every $i\in I$, denote by $x_{i_{s\alpha}}$ ($\alpha\in J_{s,i}$) the maximal points of $(Z_i)_s$.
There exists a neighborhood $U$ of $\eta$ in $S$ such that, for every $s\in U$, the $x_{i_{s\alpha}}$ (for $i\in I$ and $\alpha\in J_{s,i}$) are two by two distinct and that $\operatorname{Ass}(\sh{F}_s)$ is the set $\{x_{i_{s\alpha}}\}$ (in other words, the first associated cycles of $\sh{F}_s$ are the irreducible components of $(Z_i)_s$).
Moreover, one can shrink $U$ such that the closure of $\{x_{i_{s\alpha}}\}$ in $X_s$ is a first associated cycle of $\sh{F}_s$, and it is necessary and sufficient that $(Z_i)_\eta$ be a maximal first associated cycle of $\sh{F}_\eta$.
\end{proposition}

\begin{corollary}[9.8.4]
\label{IV.9.8.4}
With the notations and hypotheses of \sref{IV.9.8.2}, there exists a neighborhood $U$ of $\eta$ in $S$ such that, for every $s\in U$, each of the $(\sh{F}_i)_s$ is without first associated immersed cycle, and the family $(\sh{F}_{i_{s\alpha}})_{\alpha\in J_{s,i}}$ is the unique irredundant reduced decomposition of $(\sh{F}_i)_s$, while the family $(\sh{F}_{i_{s\alpha}})_{i\in I,\alpha\in J_{s,i}}$ is an irredundant reduced decomposition of $\sh{F}_s$.
\end{corollary}

\begin{remark}[9.8.9]
\label{IV.9.8.9}
One can make more precise the preceding propositions in various ways; let us limit ourselves to one statement, given as an example.
Say that a finite part $P$ of a $k$-prescheme algebraic $X$ is \emph{saturated} if, for every couple of points $x$, $y$ of $P$, the generic points of the irreducible components of $\overline{\{x\}}\cap\overline{\{y\}}$ also belong to $P$; for every finite part $Q$ of $X$, there exists a smallest finite part $P$ of $X$ containing $Q$ and saturated; we say that $P$ is the \emph{saturate} of $Q$.
For every coherent $\sh{O}_X$-Module $\sh{F}$, we call \emph{primary skeleton} of $\sh{F}$ the system $(P,Q,\omega,d,m)$ where $Q=\operatorname{Ass}(\sh{F})$, $P$ is the saturate of $Q$, $\omega$ is the order relation $\{x\}\subset\overline{\{y\}}$ on $P$, $d$ the function $x\mapsto\dim\overline{\{x\}}$ on $P$, $m$ the application of the set of maximal elements of $Q$ for the relation $\omega$.
We define in an evident manner the notion of \emph{isomorphism} of two virtual skeletons, and finally, with the previous notations, we call \emph{geometric primary type} of $\sh{F}$ the class (for the relation of isomorphism of virtual skeletons) of the primary skeleton of $\sh{F}$.
\end{remark}

\subsection{Constructibility of local properties of fibres}
\label{subsection:IV.9.9}

\begin{proposition}[9.9.1]
\label{IV.9.9.1}
\oldpage[IV-3]{88}
Let $f:X\to S$ be a morphism locally of finite presentation, $Z$ a locally constructible part of $X$ such that for every $s\in S$, $Z_s$ is closed in $X_s$, $\Phi$ a finite part of $\bb{Z}\cup\{\pm\infty\}$.
The following parts of $X$ are locally constructible:
\begin{enumerate}[label=(\roman*)]
  \item The set of $x\in X$ such that $\dim_x(Z_{f(x)})$ belongs to $\Phi$.
  \item The set of $x\in X$ such that $\operatorname{codim}_x(Z_{f(x)},X_{f(x)})$ belongs to $\Phi$.
  \item The set of $x\in X$ such that the local ring $\sh{O}_{X_{f(x),x}}$ is equidimensional.
\end{enumerate}
We note that properties (i) and (ii) express respectively that the functions $x\mapsto\dim_x(Z_{f(x)})$ and $x\mapsto\operatorname{codim}_x(Z_{f(x)},X_{f(x)})$ are locally constructible in $X$ ($\mathbf{0}_{\mathrm{III}}$, 9.3.1).
\end{proposition}

\begin{proposition}[9.9.2]
\label{IV.9.9.2}
\oldpage[IV-3]{90}
Let $f:X\to S$ be a morphism locally of finite presentation, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent of finite presentation, $\Phi$ a finite part of $\bb{Z}\cup\{\pm\infty\}$.
The following parts of $X$ are locally constructible:
\begin{enumerate}[label=(\roman*)]
  \item The set of points $x\in X$ such that $\sh{F}_{f(x)}$ is equidimensional at the point $x$ \sref{IV.5.1.12}.
  \item The set of points $x\in X$ such that the geometric lengths of $\sh{F}_{f(x)}$, relative to $k(f(x))$, of the generic points of the irreducible components of $\operatorname{Supp}(\sh{F}_{f(x)})$ containing $x$ \sref{IV.4.7.5}, belong to $\Phi$.
  \item The set of points $x\in X$ such that the dimensions of the first associated cycles of $\sh{F}_{f(x)}$ and containing $x$ belong to $\Phi$.
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is geometrically reduced at the point $x$ \sref{IV.4.6.17}.
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is geometrically integral at the point $x$ \sref{IV.4.6.22}.
  \item The set of $x\in X$ such that $\operatorname{dim.proj}((\sh{F}_{f(x)})_x)\in\Phi$.
  \item The set of $x\in X$ such that $\operatorname{coprof}((\sh{F}_{f(x)})_x)\in\Phi$.
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ possesses the property $(S_k)$ at the point $x$ \sref{IV.5.7.2}.
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is a Cohen--Macaulay $\sh{O}_{X_{f(x)}}$-Module at the point $x$ \sref{IV.5.7.1}.
\end{enumerate}
\end{proposition}

\begin{corollary}[9.9.3]
\label{IV.9.9.3}
\oldpage[IV-3]{93}
Let $f:X\to S$ be a morphism of finite presentation, $\sh{F}$ an $\sh{O}_X$-Module of finite presentation, $\mathbf{P}$ one of the properties \emph{(i)} to \emph{(ix)} of \sref{IV.9.9.2}.
Then the set of $s\in S$ such that the property $\mathbf{P}$ is true for all points $x\in X_s$ is locally constructible in $S$.
\end{corollary}

\begin{proof}
Indeed, its complement is the image by $f$ of the complement of $E$, the set of points where $\mathbf{P}$ holds.
As $E$ is locally constructible in $X$, $X-E$ is also, hence $f(X-E)$ is locally constructible in $S$, by virtue of the theorem of Chevalley \sref[I]{I.1.8.4}.
\end{proof}

\begin{proposition}[9.9.4]
\label{IV.9.9.4}
Let $f:X\to S$ be a morphism locally of finite presentation.
The set of $x\in X$ such that $X_{f(x)}$ has at the point $x$ one (determined) of the following properties:
\begin{enumerate}[label=(\roman*)]
  \item being geometrically regular;
  \item possessing the geometric property $(\mathrm{R}_k)$;
  \item being geometrically normal;
  \item being geometrically reduced (i.e.\ separable);
  \item being geometrically punctually integral;
\end{enumerate}
is a locally constructible part of $X$.
\end{proposition}

\begin{proof}
For properties (iv) and (v), this follows from \sref{IV.9.9.2}[, (iv) and (v)] applied to $\sh{F}=\sh{O}_X$.
For the other properties, taking into account \sref{IV.6.7.8}, one reduces, as at the beginning of \sref{IV.9.9.2}, to the case where $S$ is noetherian and integral of generic point $\eta=f(x)$.
Moreover, by the theorem of generic flatness \sref{IV.6.9.1}, one can, by replacing $S$ by a neighborhood of $\eta$, assume that $f$ is a \emph{flat} morphism.
Then, saying that $X_{f(x)}$ is geometrically regular at the point $y$ means that $f$ is \emph{regular} at the point $y$ \sref{IV.6.8.1}, and one knows that the set $L$ of these points is \emph{open} in $X$ \sref{IV.6.8.7}, which proves the proposition in case (i).
\oldpage[IV-3]{94}
\end{proof}

\begin{corollary}[9.9.5]
\label{IV.9.9.5}
Let $f:X\to S$ be a morphism of finite presentation, and denote by $\mathbf{P}$ one of the properties \emph{(i)} to \emph{(v)} of \sref{IV.9.9.4}.
Then the set of $s\in S$ such that the property $\mathbf{P}$ is true for all points $x\in X_s$ is locally constructible in $S$.
\end{corollary}

\begin{proof}
The proof is the same as that of \sref{IV.9.9.3} starting from \sref{IV.9.9.4}.
\end{proof}

\begin{proposition}[9.9.6]
\label{IV.9.9.6}
Let $f:X\to S$ be a morphism locally of finite presentation, $\sh{L}_\bullet$ a complex formed of $\sh{O}_X$-Modules quasi-coherent of finite presentation; for every $s\in S$, let $(\sh{L}_\bullet)_s$ the complex $\sh{L}_\bullet\otimes_{\sh{O}_S}k(s)$ of $\sh{O}_{X_s}$-Modules coherent of finite type.
For a given integer $n$, the set of $x\in X$ such that $(\sh{H}_n((\sh{L}_\bullet)_{f(x)}))_x=0$ is locally constructible in $X$.
\end{proposition}

\begin{proof}
One can reduce to the case where $\sh{L}_i=0$ except for $i=0$, $1$ or $2$, and where $n=1$.
Moreover, the question being local on $X$, one can reduce to the case where $S=\operatorname{Spec}(A)$ and $X=\operatorname{Spec}(B)$ are affine, $B$ being a finitely presented $A$-algebra, and the complex $\sh{L}_\bullet$ is a complex of $\sh{O}_X$-Modules of finite presentation, zero except in degrees $0$, $1$ and $2$.
There exists then a sub-ring $A_0$ of $A$ which is a $\bb{Z}$-algebra of finite type, an $A_0$-prescheme of finite type $X_0$ and a complex $\sh{L}_\bullet^{(0)}$ of $\sh{O}_{X_0}$-Modules coherent, zero except in degrees $0$, $1$ and $2$, such that $X$ is isomorphic to $X_0\otimes_{A_0}A$ and $\sh{L}_\bullet\simeq\sh{L}_\bullet^{(0)}\otimes_{A_0}A$.
For every $s\in S$, if $s_0$ is the projection of $s$ in $S_0=\operatorname{Spec}(A_0)$, the fibres of $X$ and $X_0$ at $s$ and $s_0$ are the same, and $(\sh{H}_n((\sh{L}_\bullet)_{f(x)}))_x=(\sh{H}_n((\sh{L}_\bullet^{(0)})_{f_0(x)}))_x$, hence one can reduce to the case where $A$ is noetherian.
Then, by \sref{IV.1.8.2}, one is led to show that, for every $x\in X$, there is a neighborhood $V$ of $x$ in $\overline{\{x\}}$ such that, for every $x'\in V$, the set of dimensions of the first cycles associated to $\sh{F}_{f(x')}$ containing $x'$ is the same.
But this follows from \sref{IV.9.5.1} applied to the support $Z$ of $\sh{H}_n(\sh{L}_\bullet)$, and $Z\cap\overline{\{x\}}$ is closed in the noetherian space $X$; hence $Z_s\cap\overline{\{x\}}=\emp$ (resp.\ $\neq\emp$) for $s$ in a neighborhood of $\eta$ by \sref{IV.9.5.1}.
The set $V=f^{-1}(U)\cap\overline{\{x\}}$ is then the desired neighborhood.
\end{proof}
