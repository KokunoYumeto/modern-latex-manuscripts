\section{Regular immersions and normal flatness}
\label{section:IV.19}


\subsection{Properties of regular immersions}
\label{subsection:IV.19.1}

\oldpage[IV-4]{185}

\begin{env}[19.1.1]
\label{IV.19.1.1}
Let $\sh{J}$ be a quasi-coherent Ideal of $\sh{O}_X$ of finite type, $Y$ the closed sub-prescheme of $X$ that it defines.
We say that $\sh{J}$ is \emph{quasi-regular} (or \emph{quasi-regulier}) at a point $z$ of $Y$ if the $(\sh{O}_X/\sh{J})_z$-module $\sh{J}_z/\sh{J}_z^2$ is free and if the canonical homomorphism from the symmetric algebra $\operatorname{S}((\sh{O}_X/\sh{J})_z;\sh{J}_z/\sh{J}_z^2)$ to the graded ring $\operatorname{gr}(\sh{O}_{X,z};\sh{J}_z)=\bigoplus_{n\geq 0}\sh{J}_z^n/\sh{J}_z^{n+1}$ is bijective (\sref[0]{0.15.1.7}).
We say that $\sh{J}$ is \emph{regular} at the point $z$ if, in addition, $\sh{O}_{X,z}$ is a flat $\sh{O}_{Y,z}$-module (i.e., $\sh{O}_{X,z}/\sh{J}_z$ is a flat $\sh{O}_{X,z}$-module).
We say that $\sh{J}$ is a \emph{quasi-regular} (resp.\ \emph{regular}) \emph{Ideal} of $\sh{O}_X$ if it is quasi-regular (resp.\ regular) at every point of $Y$.
\end{env}

\begin{env}[19.1.2]
\label{IV.19.1.2}
An immersion $u:Y\to X$ of preschemes is said to be \emph{quasi-regular} (resp.\ \emph{regular}) at a point $y\in Y$ if $u$ is a \emph{closed} immersion in a neighbourhood of $y$ and the Ideal $\sh{J}$ of $\sh{O}_X$ which defines $u(Y)$ in such a neighbourhood is quasi-regular (resp.\ regular) at the point $u(y)$.
We say that $u$ is a \emph{quasi-regular immersion} (resp.\ \emph{regular immersion}) if it is quasi-regular (resp.\ regular) at every point of $Y$.
The \emph{codimension} of $u$ at a point $y\in Y$ is, by definition, equal to the rank of the free $(\sh{O}_X/\sh{J})_{u(y)}$-module $\sh{J}_{u(y)}/\sh{J}_{u(y)}^2$.
\end{env}

\begin{env}[19.1.3]
\label{IV.19.1.3}
When $\sh{O}_{X,z}$ is a Noetherian local ring, to say that $\sh{J}$ is quasi-regular at $z$ means (by \sref[0]{0.15.1.9}) that $\sh{J}_z$ is generated by a quasi-regular sequence of elements (i.e., by a sequence $(f_i)_{1\leq i\leq n}$ of elements of the maximal ideal of $\sh{O}_{X,z}$ which is quasi-regular for $\sh{O}_{X,z}$);
to say that $\sh{J}$ is regular at $z$ means (\sref[0]{0.15.1.10}) that $\sh{J}_z$ is generated by a regular sequence of elements.
In addition, when $X$ is locally Noetherian, the set of $z\in Y$ where $\sh{J}$ is quasi-regular (resp.\ regular) is \emph{open} in $Y$ (\sref[0]{0.15.1.11}).
\end{env}

\begin{env}[19.1.4]
\label{IV.19.1.4}
Suppose that $X$ is locally Noetherian, and let $u:Y\to X$ be a quasi-regular closed immersion (resp.\ regular closed immersion) defined by an Ideal $\sh{J}$ of $\sh{O}_X$.
Then for every point $z\in Y$:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\operatorname{codim}_z^*(Y,X)=\operatorname{rk}_z(\sh{J}/\sh{J}^2)$ (\sref[0]{0.16.9.5});
  \item $\operatorname{codim}_z^*(Y,X)=\operatorname{codim}(\sh{O}_{X,z}/\sh{J}_z,\sh{O}_{X,z})$, i.e., is equal to the height of $\sh{J}_z$ in the local ring $\sh{O}_{X,z}$ (\sref[0]{0.16.9.5}), (\sref[0]{0.16.3.3}).
\end{enumerate}
In particular, the function $z\mapsto\operatorname{codim}_z^*(Y,X)$ is \emph{locally constant} on $Y$.
When $u$ is a regular closed immersion, $\operatorname{codim}_z^*(Y,X)=\operatorname{prof}(\sh{J}_z,\sh{O}_{X,z})$ (\sref[0]{0.15.1.10}).
In addition, $\operatorname{codim}_z^*(Y,X)$ is also equal to the length of a maximal chain of prime ideals of $\sh{O}_{X,z}$ containing $\sh{J}_z$ (\sref[0]{0.16.3.6}).
Finally, let us mention a few supplementary properties of $\sh{J}$:
\begin{enumerate}[label=\emph{\alph*)}]
  \item the module $\sh{J}^n/\sh{J}^{n+1}$ is a locally free $(\sh{O}_X/\sh{J})$-Module of rank $\binom{m+n-1}{n}$ if $m=\operatorname{rk}(\sh{J}/\sh{J}^2)$ (\sref[0]{0.16.9.5});
  \item for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, the canonical homomorphism $\sh{F}/\sh{J}\sh{F}\otimes_{\sh{O}_X/\sh{J}}\sh{J}^n/\sh{J}^{n+1}\to\sh{J}^n\sh{F}/\sh{J}^{n+1}\sh{F}$ is an isomorphism (\sref[0]{0.15.1.7});
  \item if $u$ is a regular closed immersion, for every locally free $\sh{O}_X$-Module $\sh{F}$ of finite rank, the canonical map $\sh{F}/\sh{J}\sh{F}\otimes_{\sh{O}_X/\sh{J}}\sh{J}^n/\sh{J}^{n+1}\to\sh{J}^n\sh{F}/\sh{J}^{n+1}\sh{F}$ is again an isomorphism.
\end{enumerate}

We also note the following property: if $n=\operatorname{codim}_z^*(Y,X)$, then for every $y$ in
\oldpage[IV-4]{186}
a neighbourhood of $z$ in $Y$, we have $\operatorname{codim}_y^*(Y,X)\leq n$; since any such point $z$ is contained in every neighbourhood of $y$ in $X$, $\sh{J}_z/\sh{J}_z^2$ is a free A-module of rank $n=\operatorname{codim}_y^*(Y,X)$; it amounts to saying that $\dim(\mathrm{A})=n$.
Now, since $z$ is a maximal point of the sub-prescheme $Y$ defined by $\sh{J}$, we have $\dim(\sh{O}_{X,z}/\sh{J}_z)=0$, so $\sh{J}_z$ is an \emph{ideal of definition} of the Noetherian local ring A; in addition, the hypothesis that $\sh{J}$ is quasi-regular implies that $\sh{J}_z^m/\sh{J}_z^{m+1}$ is an $(A/\sh{J}_z)$-module free of rank $\binom{m+n-1}{n-1}$; if $r$ is the length of the Artinian ring $A/\sh{J}_z$, the length of $\sh{J}_z^m/\sh{J}_z^{m+1}$ is therefore $r\binom{m+n-1}{n-1}$; the Hilbert--Samuel polynomial of $A$ for the $\sh{J}_z$-preadic filtration is thus indeed of degree $n$, which proves the proposition (\sref[0]{0.16.2.3}).

By virtue of \sref{IV.19.1.4}, we will say henceforth ``codimension'' instead of ``transverse codimension'' when it concerns a sub-prescheme $Y$ regularly immersed in $X$ (even when $X$ is not locally Noetherian).
\end{env}

\begin{proposition}[19.1.5]
\label{IV.19.1.5}
\medskip\noindent
\begin{enumerate}[label=(\roman*)]
  \item For an immersion $j:Y\to X$ to be open, it is necessary and sufficient that it be regular and of codimension $0$ at every point.
  \item Let $f:Y\to X$ be a regular (resp.\ quasi-regular) immersion, $g:X'\to X$ a flat morphism, $Y'=Y\times_X X'$; then the morphism $f'=f_{(X')}:Y'\to X'$ is a regular (resp.\ quasi-regular) immersion, and for every $y'\in Y'$, if $y$ is the projection of $y'$ in $Y$, the codimension of $f'$ at the point $y'$ is equal to the codimension of $f$ at the point $y$.
Conversely, if $f$ is an immersion and $g:X'\to X$ a faithfully flat and quasi-compact morphism, then, if $f'$ is a quasi-regular immersion, so is $f$.
  \item If $f:Y\to X$ and $g:Z\to Y$ are regular immersions, then $f\circ g:Z\to X$ is a regular immersion, and for every $z\in Z$, the codimension of $f\circ g$ at the point $z$ is the sum of the codimension of $g$ at the point $z$ and of the codimension of $f$ at the point $g(z)$.
Moreover, the sequence of canonical homomorphisms \sref{IV.16.2.7.1}
\[
  0\to g^*(\sh{N}_{Y/X})\to\sh{N}_{Z/X}\to\sh{N}_{Z/Y}\to 0
\]
\hyperref[IV.19.1.5.1]{(\textbf{19.1.5.1})}
\label{IV.19.1.5.1}
is exact, and for every $z\in Z$, there exists a neighbourhood of $z$ in $Z$ in which the restrictions of the homomorphisms of this sequence form a split exact sequence.
  \item Let $X$ be a locally Noetherian prescheme, $f:Y\to X$ and $g:Z\to Y$ two immersions, $z$ a point of $Z$.
The following conditions are equivalent:
  \begin{enumerate}[label=\emph{\alph*)}]
    \item $g$ is a regular immersion at the point $z$ and $f$ is a regular immersion at the point $g(z)$.
    \item $g$ and $f\circ g$ are regular immersions at the point $z$.
    \item $f\circ g$ is a regular immersion at the point $z$, $f$ is a regular immersion at the point $g(z)$, and the sequence of canonical homomorphisms \sref{IV.19.1.5.1}, restricted to a sufficiently small open neighbourhood of $z$ in $Z$, is exact and split.
  \end{enumerate}
  \item Let $X$ be a locally Noetherian prescheme, $f:Y\to X$, $g:Z\to Y$ two immersions, $z$ a point of $Z$; suppose that $y=g(z)$ is a regular point of $Y$ and $x=f(g(z))$ a regular point
\oldpage[IV-4]{187}
of $X$ (so that, by \sref{IV.19.1.1}, $f$ is a regular immersion at the point $g(z)$); then,
for $f\circ g$ to be a regular immersion at the point $z$, it is necessary and sufficient that $g$ be a regular immersion at the point $z$.
\end{enumerate}
\end{proposition}

\begin{proof}
Assertion~(i) is another way of expressing \sref{IV.16.1.10}.
To prove the first assertion of~(ii), we can restrict ourselves to the case where $Y$ is a closed sub-prescheme of $X$ defined by a regular (resp.\ quasi-regular) Ideal $\sh{J}$; then $Y'$ is the closed sub-prescheme of $X'$ defined by the Ideal $g^*(\sh{J})\sh{O}_{X'}$ which is identified here with $g^*(\sh{J})$ since $g$ is flat; by replacing $X$ by a sufficiently small affine open, we can suppose that $\sh{J}$ admits a (resp.\ quasi-regular) regular sequence of generators $f_i$ (sections of $\sh{O}_X$ over $X$).
If the sequence $(f_i)$ is regular, so is the sequence $(f_i\otimes 1)$ (which generates $g^*(\sh{J})\sh{O}_X$) by virtue of \sref[0]{0.15.2.5}; the analogous result for quasi-regular sequences follows immediately from the flatness of $g$ and from the criterion \sref{IV.16.9.4}.
Similarly, if $g$ is faithfully flat and quasi-compact, and if $g^*(\sh{J})$ is quasi-regular, $\sh{J}$ is of finite type in virtue of \sref{IV.2.5.2}; by flatness, we have $g^*(\sh{J})/g^*(\sh{J})^2=g^*(\sh{J}/\sh{J}^2)$, hence, since $g^*(\sh{J}/\sh{J}^2)$ is locally free by hypothesis \sref{IV.16.9.4}, it follows that $\sh{J}/\sh{J}^2$ \sref{IV.2.5.2}.
Finally, condition~(iii) of \sref{IV.16.9.4} relative to $g^*(\sh{J})$ implies the same condition for $\sh{J}$ by virtue of the faithful flatness of $g$ \sref{IV.2.2.7}.
We conclude from \sref{IV.16.9.4} that $\sh{J}$ is quasi-regular.

To prove~(iii), we can likewise suppose that $Y$ and $Z$ are closed sub-preschemes of $X$, defined respectively by Ideals $\sh{J}$, $\sh{K}$ of $\sh{O}_X$ such that $\sh{J}\subset\sh{K}$; moreover, we can suppose that there exist sections $f_i$ ($1\leq i\leq m$) of $\sh{J}$ over $X$ generating $\sh{J}$ and such that for $1\leq i\leq m$, the endomorphism of $\sh{O}_X/(\sum_{j=1}^{i-1}f_j\sh{O}_X)$ defined by $f_i$ is injective, and sections $\bar{f}_i$ ($m+1\leq i\leq n$) of $\sh{K}/\sh{J}$ over $X$ generating $\sh{K}/\sh{J}$ and such that for $m+1\leq i\leq n$, the endomorphism of $(\sh{O}_X/\sh{J})/(\sum_{j=m+1}^{i-1}\bar{f}_i(\sh{O}_X/\sh{J}))$ defined by $\bar{f}_i$ is injective.
For every $z\in Z$, let $U$ be an open neighbourhood of $z$ in $X$ such that there exists a section $f_i$ of $\sh{K}$ over $U$, whose class in $\Gamma(U,\sh{K}/\sh{J})$ is $\bar{f}_i|U$ for $m+1\leq i\leq n$; then $(\sh{O}_X/\sh{J})/(\sum_{j=m+1}^{i-1}\bar{f}_j\sh{O}_U)$, restricted to $U$ is isomorphic to $\sh{O}_U/((\sh{J}|U)+\sum_{j=m+1}^{i-1}f_j\sh{O}_U)=\sh{O}_U/(\sum_{j=1}^{i-1}f_j\sh{O}_U)$, and we see therefore that the sequence $(f_i)_{1\leq i\leq n}$ is regular in $\sh{O}_U$; as it generates $\sh{K}|U$, this proves the first assertion of~(iii).
To prove the last assertion of~(iii), we can restrict ourselves (with the same notations) to the case where the images $t_i$ of the $f_i$ in $\sh{K}/\sh{K}^2$ (for $1\leq i\leq n$) form a basis of this $\sh{O}_X/\sh{K}$-Module \sref{IV.16.9.5}; for the same reason, we can suppose that the $t_i$ for $1\leq i\leq m$ are the canonical images of elements of a basis of the $\sh{O}_X/\sh{K}$-Module $g^*(\sh{J}/\sh{J}^2)$, and that the canonical images $\bar{t}_i$ of the $t_i$ in $(\sh{K}/\sh{J})/(\sh{K}/\sh{J})^2$ for $m+1\leq i\leq n$ form a basis of this $\sh{O}_X$-$\sh{K}$-Module.
This completes the proof of~(iii).

Let us pass to the proof of~(iv).
The fact that \emph{a)} implies \emph{c)} follows from~(iii).
Let us prove that \emph{c)} implies \emph{a)}.
Set again $\mathrm{A}=\sh{O}_{X,f(g(z))}$, so that $\sh{O}_{Y,g(z)}=\mathrm{A}/\mathfrak{J}$,
\oldpage[IV-4]{188}
$\sh{O}_{Z,z}=\mathrm{A}/\mathfrak{R}$ with $\mathfrak{J}\subset\mathfrak{R}$.
By hypothesis, $\mathfrak{J}$ and $\mathfrak{R}$ are regular ideals of $\mathrm{A}$ and we have a split exact sequence
\[
  0\to\mathfrak{J}/\mathfrak{J}\mathfrak{R}\to\mathfrak{R}/\mathfrak{R}^2\to(\mathfrak{R}/\mathfrak{J})/(\mathfrak{R}/\mathfrak{J})^2\to 0
\]
(cf.\ \sref{IV.16.2.7}).
This implies that there exist elements $f_j$ ($1\leq j\leq r+s$) of $\mathfrak{R}$ whose images $f_j'$ in $\mathfrak{R}/\mathfrak{R}^2$ form a basis of this $(\mathrm{A}/\mathfrak{R})$-module, and such that, in addition, the $f_j$ of index $j\leq r$ belong to $\mathfrak{J}$ and are such that their images in $\mathfrak{J}/\mathfrak{J}\mathfrak{R}$ form a basis of this $(\mathrm{A}/\mathfrak{R})$-module.
The $f_j$ for $1\leq j\leq r+s$ generate $\mathfrak{R}$, and the $f_j$ for $1\leq j\leq r$ generate $\mathfrak{J}$ since $\mathrm{A}$ is Noetherian (Bourbaki, \emph{Alg.\ comm.}, chap.~II, \S~3, n\textsuperscript{o}~2, prop.~5); as $\mathfrak{R}$ is a regular ideal, it follows from \sref{IV.16.9.5} that the sequence $(f_j)_{1\leq j\leq r+s}$ is regular.
By definition, the images of $f_{r+1},\ldots,f_{r+s}$ in $\mathfrak{R}/\mathfrak{J}$ form a regular sequence in this $(\mathrm{A}/\mathfrak{J})$-module, which completes the proof that \emph{c)} implies \emph{a)} in~(iv).

Finally, taking into account \sref{IV.16.9.10}, it is immediate that~(v), as well as the equivalence of \emph{a)} and \emph{b)} in~(iv), follow from the
\end{proof}

\begin{corollary}[19.1.6]
\label{IV.19.1.6}
Let $\mathrm{A}$ be a local Noetherian ring, $\mathfrak{J}$, $\mathfrak{R}$ two ideals of $\mathrm{A}$ contained in its maximal ideal $\mathfrak{m}$ and such that $\mathfrak{J}\subset\mathfrak{R}$, $\mathrm{A}'=\mathrm{A}/\mathfrak{J}$, $\mathfrak{R}'=\mathfrak{R}/\mathfrak{J}$.
Then:
\begin{enumerate}[label=(\roman*)]
  \item If $\mathfrak{J}$ and $\mathfrak{R}'$ are regular ideals, so is $\mathfrak{R}$.
  \item If $\mathfrak{R}$ and $\mathfrak{R}'$ are regular ideals, so is $\mathfrak{J}$.
  \item Suppose the rings $\mathrm{A}$ and $\mathrm{A}'=\mathrm{A}/\mathfrak{J}$ are regular.
For the ideal $\mathfrak{R}$ to be regular, it is necessary and sufficient that $\mathfrak{R}'$ be so.
\end{enumerate}
\end{corollary}

\begin{proof}
Assertion~(i) is a particular case of \sref{IV.19.1.5},~(iii).
To prove~(ii), consider the canonical surjective homomorphism $u:\mathfrak{R}/\mathfrak{R}^2\to\mathfrak{R}'/\mathfrak{R}'^2=\mathfrak{R}/(\mathfrak{R}^2+\mathfrak{J})$.
Since by hypothesis $\mathfrak{R}/\mathfrak{R}^2$ and $\mathfrak{R}'/\mathfrak{R}'^2$ are free $(\mathrm{A}/\mathfrak{R})$-modules, whose ranks we denote respectively by $p+q$ and $q$, the kernel of $u$ is an $(\mathrm{A}/\mathfrak{R})$-module \emph{projective} (Bourbaki, \emph{Alg.}, chap.~II, 3\textsuperscript{e} ed., \S~2, n\textsuperscript{o}~2, prop.~4), hence \emph{free} of rank $p$ since $\mathrm{A}/\mathfrak{R}$ is a local Noetherian ring (\sref[0]{0.10.1.3}); in other words, there exists a basis of $\mathfrak{R}/\mathfrak{R}^2$ of which the first $p$ elements form a basis of $(\mathfrak{R}^2+\mathfrak{J})/\mathfrak{R}^2$.
This means (by virtue of Nakayama's lemma) that there exists a system of generators $(f_i)_{1\leq i\leq p+q}$ of $\mathfrak{R}$, such that the sequence $(f_i)_{1\leq i\leq p}$ is formed of elements of $\mathfrak{J}$; it remains to show that this last sequence generates $\mathfrak{J}$.
For this, denote by $\mathfrak{J}'\subset\mathfrak{J}$ the ideal it generates; considering the ring $\mathrm{A}/\mathfrak{J}'$, we see that we are reduced to proving the following lemma:
\end{proof}

\begin{lemma}[19.1.6.1]
\label{IV.19.1.6.1}
Let $\mathrm{B}$ be a Noetherian local ring with maximal ideal $\mathfrak{n}$, $(g_j)_{1\leq j\leq q}$ a regular sequence in $\mathrm{B}$ of $\mathfrak{n}$ elements, $\mathfrak{b}$ the ideal it generates, $\mathfrak{c}\subset\mathfrak{b}$ an ideal such that the canonical images of the $g_j$ in $\mathrm{B}/\mathfrak{c}$ form a regular sequence in this ring.
Then $\mathfrak{c}=0$.
\end{lemma}

\begin{proof}
The lemma being trivial for $q=0$, we reason by induction on $q$.
The hypothesis, applied to the ring $\mathrm{B}/g_1\mathrm{B}$ and to the canonical images of the $g_j$ ($2\leq j\leq q$)
\oldpage[IV-4]{189}
and of $\mathfrak{c}$ in this quotient ring, shows that $\mathfrak{c}\subset g_1\mathrm{B}$.
Let $\mathfrak{b}$ be the ideal of $\mathrm{B}$ consisting of the $x\in\mathrm{B}$ such that $g_1 x\in\mathfrak{c}$; we thus have $\mathfrak{c}=g_1\mathfrak{b}$; but since by hypothesis $g_1$ is regular in $\mathrm{B}/\mathfrak{c}$, we necessarily have $\mathfrak{b}=\mathfrak{c}$, hence $\mathfrak{c}=g_1\mathfrak{c}$.
Since $\mathfrak{c}$ is of finite type and $g_1$ is contained in the radical of $\mathrm{B}$, Nakayama's lemma shows that $\mathfrak{c}=0$.

Let us now prove assertion~(iii) of \sref{IV.19.1.6}.
By virtue of~(i) and of \sref{IV.19.1.2}, it suffices to show that if $\mathfrak{R}$ is regular, so is $\mathfrak{J}$.
Note that $\mathfrak{J}$ is regular \sref{IV.19.1.2}, and reason by induction on the rank $q$ of the $(\mathrm{A}/\mathfrak{J})$-module $\mathfrak{J}/\mathfrak{J}^2$.
If $q=0$, we have $\mathfrak{J}=0$ by Nakayama's lemma and the assertion is trivial.
Since $\mathrm{A}$ and $\mathrm{A}/\mathfrak{J}$ are regular, we know (\sref[0]{0.17.1.9}) that $\mathfrak{J}$ is generated by a subset of $q$ elements of a regular system of parameters of $\mathrm{A}$, hence, if $f$ is one of the elements of this system of generators of $\mathfrak{J}$, $\mathrm{A}_1=\mathrm{A}/f\mathrm{A}$ is regular and $f\notin\mathfrak{m}^2$ (\sref[0]{0.17.1.8}).
Let $\mathfrak{J}_1,\mathfrak{R}_1$ be the images of $\mathfrak{J},\mathfrak{R}$ in $\mathrm{A}_1$; we have $\mathfrak{J}_1\subset\mathfrak{R}_1$, $\mathrm{A}_1/\mathfrak{J}_1=\mathrm{A}/\mathfrak{J}$, hence $\mathrm{A}_1/\mathfrak{J}_1$ is regular and by the sequence $\mathfrak{J}_1$ is a regular ideal \sref{IV.19.1.2}.
Let us show that $\mathfrak{R}_1$ is a regular ideal in $\mathrm{A}_1$; since $\mathfrak{R}$ is a regular ideal in $\mathrm{A}$, it suffices to show that $f$ is part of a regular system of generators of $\mathfrak{R}$, and for this \sref{IV.16.9.5} it suffices to show that the image of $f$ in $\mathfrak{R}/(\mathfrak{R}^2+\mathfrak{m}\mathfrak{R})=\mathfrak{R}/\mathfrak{m}\mathfrak{R}$ is part of a basis of this $(\mathrm{A}/\mathfrak{m})$-vector space, in other words that $f\notin\mathfrak{m}\mathfrak{R}$, which indeed follows from $f\notin\mathfrak{m}^2$.
Then, as $\mathfrak{J}_1/\mathfrak{J}_1^2$ is a free $(\mathrm{A}_1/\mathfrak{J}_1)$-module of rank $q-1$, the induction hypothesis shows that $\mathfrak{R}_1'=\mathfrak{R}_1/\mathfrak{J}_1$ is a regular ideal of $\mathrm{A}_1/\mathfrak{J}_1$, and since $\mathfrak{R}'$ is identified with $\mathfrak{R}_1'$ in the canonical isomorphism between $\mathrm{A}/\mathfrak{J}$ and $\mathrm{A}_1/\mathfrak{J}_1$, $\mathfrak{R}'$ is regular.
\end{proof}

\begin{remarks}[19.1.7]
\label{IV.19.1.7}
\medskip\noindent
\begin{enumerate}[label=(\roman*)]
  \item We do not know whether the composite of two quasi-regular immersions is quasi-regular.
  \item In a ringed space with local rings $X$, for every Ideal $\sh{J}$ of $\sh{O}_X$, $\sh{O}_X/\sh{J}$ is of finite type and therefore has a closed support in $X$ (\sref[0]{0.5.2.2}).
We can therefore define, as in (\sref[I]{I.4.1.3} and \sref{IV.4.2.1}), the notions of closed ringed subspace (locally defined by an Ideal $\sh{J}$ of a sheaf $\sh{O}_U$, where $U$ is an open subset of $X$, and the notion of immersion.
The definitions of quasi-regular and regular immersions, of codimension transverse, and the results of \sref{IV.19.1.4},~(i) to~(iv) are still valid, assuming in the second assertion of~(i) and in~(iv) that the sheaf $\sh{O}_X$ is coherent and the local rings $\sh{O}_{X,x}$ are Noetherian; as for~(ii), it is necessary to \emph{define} $Y'$ as the ringed subspace defined by the $\sh{O}_X$-Ideal $g^*(\sh{J})$; the proofs are unchanged.
\end{enumerate}
\end{remarks}

\begin{env}[19.1.8]
\label{IV.19.1.8}
We have already noted (in particular \sref{IV.16.9.6},~(ii)) that in non-Noetherian rings, the notion of ``regular sequence'' of elements of the ring does not possess the properties that would make it usable.
Recent research seems to show that in this case the notion that should be substituted for that of regular sequence is that of the sequence $\mathbf{f}=(f_1,\ldots,f_n)$ having the property that the complex of the exterior algebra $\mathrm{K}_0(\mathbf{f},\mathrm{M})$ (\sref[III]{III.1.1.3}) has \emph{zero homology in positive degrees} (cf.\ \sref[III]{III.1.1.4}); see in particular A.~Grothendieck, S\'eminaire de g\'eom\'etrie alg\'ebrique, 1966, to appear.
\end{env}

\subsection{Transversally regular immersions}
\label{subsection:IV.19.2}

\oldpage[IV-4]{190}

\begin{definition}[19.2.1]
\label{IV.19.2.1}
Let $u:X\to S$ be a morphism of preschemes, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent.
We say that a sequence $(f_i)_{1\leq i\leq n}$ of sections of $\sh{F}$ over $X$ is \emph{transversally $\sh{F}$-regular relatively to $S$} (or relatively to $u$) if the sequence $(f_i)$ is $\sh{F}$-regular and such that the $\sh{O}_X$-Modules $\sh{F}/(\sum_{1\leq i}f_i\sh{F})$ are S-flat for $0\leq i\leq n$.
We say that an Ideal $\sh{J}$ quasi-coherent of $\sh{O}_X$ is \emph{transversally regular relatively to $S$} (or relatively to $u$) at a point $x\in\operatorname{Supp}(\sh{O}_X/\sh{J})$ if there exists an open neighbourhood $U$ of $x$ and a finite sequence $(f_i)$ of sections of $\sh{J}$ over $U$ that is $\sh{O}_U$-regular relatively to $S$ and that generates $\sh{J}|U$.

In a $A$-algebra $B$, we say that an ideal $\mathfrak{J}$ is \emph{transversally regular relatively} to $A$ if $\sh{J}=\widetilde{\mathfrak{J}}$ is transversally regular relatively to $S=\operatorname{Spec}(A)$ at every point of $V(\mathfrak{J})$ in $\operatorname{Spec}(B)$.
\end{definition}

\begin{definition}[19.2.2]
\label{IV.19.2.2}
Let $S$ be a prescheme, $X$, $Y$ two S-preschemes, $u:Y\to X$ an S-morphism which is an immersion.
We say that $u$ is a \emph{transversally regular immersion relatively to $S$} at a point $y\in Y$ if there exists a neighbourhood $V$ of $u(y)$ in $X$ such that the sub-prescheme of $X$ associated to $u(Y)\cap V$ is defined by an Ideal of $\sh{O}_U$ transversally regular relatively to $S$ at the point $u(y)$.

It is clear that the set of points of $Y$ where an immersion is transversally regular relatively to $S$ is \emph{open} in $Y$; if it is equal to $Y$, we say simply that $u$ is a \emph{transversally regular immersion relatively to $S$}.

If $S\to T$ is a flat morphism, then, if $\sh{J}$ is an Ideal of $\sh{O}_X$ transversally regular relatively to $S$ at $x\in X$, it is also relatively to $T$ (\sref[0]{0.6.2.1}).
A S-immersion $Y\to X$ transversally regular relatively to $S$ at a point $y\in Y$ is therefore also a T-immersion transversally regular relatively to $T$ at this point.
\end{definition}

\begin{proposition}[19.2.3]
\label{IV.19.2.3}
Let $S$ be a prescheme, $X$, $Y$ two S-preschemes, $u:Y\to X$ an S-morphism which is a transversally regular immersion relatively to $S$ at a point $y\in Y$.
Then, for every change of base $S'\to S$, if we set $X'=X\times_S S'$, $Y'=Y\times_S S'$, the immersion $u'=u_{(S')}:Y'\to X'$ is transversally regular relatively to $S'$ at every point $y'\in Y'$ above $y$.
\end{proposition}

\begin{proof}
This follows immediately from the definition and from \sref[0]{0.15.1.15}.
\end{proof}

\begin{proposition}[19.2.4]
\label{IV.19.2.4}
Let $S$ be a prescheme, $X$, $Y$ two S-preschemes, $u:Y\to X$ an S-morphism which is an immersion.
Suppose, either that $S$ and $X$ are locally Noetherian, or that the structural morphisms $g:X\to S$, $h:Y\to S$ are locally of finite presentation.
Let $y$ be a point of $Y$, $s$ its image in $S$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $u$ is transversally regular relatively to $S$ at the point $y$.
  \item The morphisms $g$ and $h$ are flat in a neighbourhood of the points $u(y)$ and $y$ respectively, and if we set $X_s=g^{-1}(s)$, $Y_s=h^{-1}(s)$, the immersion $u_s:Y_s\to X_s$ is regular at the point $y$.
  \item[\rm b$'$)] The morphisms $g$ and $h$ are flat in a neighbourhood of the points $u(y)$ and $y$ respectively, and, for every change of base $S'\to S$, if we set $X'=X\times_S S'$, $Y'=Y\times_S S'$, the immersion $u':Y'\to X'$ is regular at every point of $Y'$ above $y$.
  \item The morphism $h$ is flat in a neighbourhood of the point $y$, and the immersion $u$ is regular in a neighbourhood of the point $y$.
\end{enumerate}
\end{proposition}

\begin{proof}
\oldpage[IV-4]{191}
\emph{c$'$) The morphism $h$ is flat in a neighbourhood of the point $y$, and the immersion $u$ is quasi-regular in a neighbourhood of the point $y$.}

We have already proved in \sref{IV.19.2.3} that \emph{a)} implies \emph{b$'$)} without the hypothesis of finiteness, and it is trivial that \emph{b$'$)} implies \emph{b)}.
Let us show that \emph{b)} implies \emph{c)}.
We can for this (the question being local on $X$ and on $S$) suppose that $Y$ is a closed sub-prescheme of $X$ defined by a quasi-coherent Ideal $\sh{J}$ of $\sh{O}_X$, and $u$ the canonical injection.
The hypothesis that $\sh{O}_X/\sh{J}$ is $g$-flat implies that the sequence
\[
  0\to\sh{J}\otimes_{\sh{O}_g}\mathbf{k}(s)\to\sh{O}_X\otimes_{\sh{O}_g}\mathbf{k}(s)\to(\sh{O}_X/\sh{J})\otimes_{\sh{O}_g}\mathbf{k}(s)\to 0
\]
is exact (\sref[0]{0.6.1.2}), hence $Y_s$ is the closed sub-prescheme of $X_s$ defined by the Ideal $\sh{J}_s$ of $\sh{O}_{X_s}=\sh{O}_X\otimes_{\sh{O}_g}\mathbf{k}(s)$ which is identified with $\sh{J}\otimes_{\sh{O}_g}\mathbf{k}(s)=\sh{J}/\mathfrak{m}_s\sh{J}$.
Consider a finite sequence $(f_i)$ of sections of $\sh{J}$ over a neighbourhood of $y$ in $X$, whose images in $\sh{J}_y/(\mathfrak{m}_s\sh{J}_y+\sh{J}_y^2)$ form a basis of this $(\sh{O}_{X,y}/(\mathfrak{m}_s\sh{O}_{X,y}+\sh{J}_y))$-module (which is free by the hypothesis \emph{b)})).
Since $X_s$ is locally Noetherian, it follows from \sref{IV.16.9.11},
\sref{IV.16.9.5} and the hypothesis \emph{b)}) that the images $(f_i)_s=f_i\otimes 1$, sections of $\sh{J}_s=\sh{J}\otimes_{\sh{O}_g}\mathbf{k}(s)$ over a neighbourhood of $y$ in $X_s$, form a regular sequence generating the restriction of $\sh{J}_s$ at this neighbourhood.
Since $\mathfrak{m}_s\sh{O}_{X,y}$ is contained in the radical of $\sh{O}_{X,y}$, and since in both cases envisaged $\sh{J}_y$ is an Ideal of finite type of $\sh{O}_{X,y}$, it follows from Nakayama's lemma that the $(f_i)_y$ also generate $\sh{J}_y$; since $\sh{J}$ is an Ideal of finite type of $\sh{O}_X$, there is a neighbourhood $U$ of $y$ in $X$ such that the $f_i|U$ generate $\sh{J}|U$.
The fact that \emph{b)} implies \emph{c)} is then a consequence of \sref[0]{0.15.1.16} when $S$ and $X$ are locally Noetherian, and of \sref{IV.11.3.8} when $g$ and $h$ are locally of finite presentation; in this last case, \sref{IV.11.3.8} proves also the equivalence of \emph{c)} and \emph{c$'$)} (which is trivial when $X$ and $S$ are locally Noetherian (\sref[0]{0.15.1.11})).
Finally, if \emph{c)} is satisfied, to prove \emph{a)}, we can restrict ourselves to the case where $Y$ is a closed sub-prescheme of $X$ defined by $\sh{J}$, and by hypothesis there is a regular sequence $(f_i)$ of sections of $\sh{J}$ over a neighbourhood of $y$ in $X$ such that the $f_i$ generate $\sh{J}|U$ and that $h$ is flat in $U$; to see that \emph{c)} implies \emph{a)}, it suffices to apply (\sref[0]{0.15.1.6}) when $S$ and $X$ are locally Noetherian, and \sref{IV.11.3.8} when $g$ and $h$ are locally of finite presentation.
\end{proof}

\begin{proposition}[19.2.7]
\label{IV.19.2.7}
\medskip\noindent
\begin{enumerate}[label=(\roman*)]
  \item Every open S-immersion is transversally regular relatively to $S$.
  \item Let $f:Y\to X$ be an S-immersion, $g:S'\to S$ a morphism and set $X'=X_{(S')}$, $Y'=Y_{(S')}$, $f'=f_{(S')}:Y'\to X'$.
If $f$ is a transversally regular immersion relatively to $S$, the converse is true if $g$ is faithfully flat, and if moreover $X$ and $Y$ are locally of finite presentation over $S$.
  \item If $f:Y\to X$ and $g:Z\to Y$ are transversally regular immersions relatively to $S$, so is $f\circ g:Z\to X$.
  \item Let $X$ be an S-prescheme, $f:Y\to X$, $g:Z\to Y$ two S-immersions.
Suppose that $S$ and $X$ are locally Noetherian, that $X$, $Y$ and $Z$ are locally of finite presentation over $S$.
Then, if $g$ and $f\circ g$ are transversally regular at the point $z\in Z$ relatively to $S$, $f$ is transversally regular at the point $g(z)$ relatively to $S$.
  \item Under the general conditions of~(iv), suppose that $X$ and $Y$ are S-flat, $Y_s$ regular at the point $g(z)$ and $X_s$ regular at the point $g(z)$; then, if $f\circ g$ is transversally regular at the point $z$, so is $g$.
\end{enumerate}
\end{proposition}

\begin{proof}
Assertion~(i) is immediate, and the first assertion of~(ii) follows immediately from \sref{IV.19.2.3}.
To prove the second assertion of~(ii), we apply \sref{IV.19.2.4}: $X'$ and $Y'$ being locally of finite presentation over $S'$, it follows first from this criterion that $X'$ and $Y'$ are flat over $S'$, hence $X$ and $Y$ are flat over $S$ \sref{IV.2.5.1}; on the other hand, for every $s'\in S'$, if $s=g(s')$, it is a point above $S$, and we deduce from \sref{IV.19.1.5},~(ii) that the same is true of $Y_s\to X_s$, since these preschemes are locally Noetherian and it thus amounts to saying that the immersion $Y_s\to X_s$ is regular or quasi-regular.
To prove~(iii), we reduce, as when $Z$ is a closed sub-prescheme of $Y$ and $Y$ a closed sub-prescheme of $X$, and we form, as in \sref{IV.19.1.5},~(iii), a system of generators of the Ideal of $\sh{O}_X$ defining $Z$.
To prove~(iv), we note that the hypothesis together with \sref{IV.19.2.4} shows first of all that $X$, $Y$ and $Z$ are S-flat in a neighbourhood of $z$, and it suffices moreover, if $s$ is the image of $z$ in $S$, to prove that the immersion $f_s:Y_s\to X_s$ is transversally regular at the point $z$, knowing that the same is true of $g_s:Z_s\to Y_s$ and $f_s\circ g_s$; but this follows from \sref{IV.19.1.5},~(iv).
Finally, for~(v), we reason in the same way as for~(iv); the hypothesis on $f\circ g$ implies already by \sref{IV.19.2.4},~\emph{b)}) that $Z$ is S-flat in a neighbourhood of $z$, and it is the same for $X$ and $Y$ by hypothesis; we are then reduced to proving that $g_s$ is regular, knowing that $f_s\circ g_s$ is, which follows from \sref{IV.19.1.5},~(v), given the hypotheses made on $X_s$ and $Y_s$.
\end{proof}

\begin{proposition}[19.2.8]
\label{IV.19.2.8}
Let $X$ be an S-prescheme, $Y$ a closed sub-prescheme of $X$ defined by a quasi-coherent Ideal $\sh{J}$ of $\sh{O}_X$, $u:Y\to X$ the canonical injection.
If $u$ is transversally regular relatively to $S$, the $\sh{O}_X$-Modules $\sh{J}^n$ and $\sh{J}^n/\sh{J}^{n+1}$ ($0\leq n<m$) are S-flat at the points of $Y$.

\begin{proposition}[19.2.9]
\label{IV.19.2.9}
Let $f:X\to S$ be a morphism, $x$ a point of $X$, $s=f(x)$.
Suppose that $f$ is of finite presentation at the point $x$, and is a Cohen-Macaulay morphism at the point $x$ (\sref{IV.6.8.1}).
Then there exists a closed sub-prescheme $Y$ of $X$ containing $x$ and having the following properties:
\begin{enumerate}[label=(\roman*)]
  \item The canonical injection $i:Y\to X$ is transversally regular relatively to $S$.
  \item The morphism $g=f\circ j:Y\to S$ is de Cohen-Macaulay (which amounts to saying, taking into account~(i), that all the fibres $g^{-1}(g(y))$ of $g$ are preschemes de Cohen-Macaulay).
  \item The point $x$ is a maximal point of $Y_s=g^{-1}(s)$.
\end{enumerate}
\end{proposition}

\begin{proof}
Set $X_s=f^{-1}(s)$; by hypothesis, the ring $\sh{O}_{X_s,x}$ is de Cohen-Macaulay; let $(t_i)_{1\leq i\leq n}$ be a system of parameters of this ring, which is therefore a regular sequence (\sref[0]{0.16.5.7}).
There is a neighbourhood open $U$ of $x$ in $X$ and sections $h_i$ ($1\leq i\leq n$) of $\sh{O}_X$ over $U$ such that the $t_i$ are the canonical images of the $(h_i)_x$ in the quotient ring $\sh{O}_{X_s,x}$ of $\sh{O}_{X,x}$.
Let $\sh{J}$ be the Ideal of $\sh{O}_U$ generated by the $h_i$, and take for $Y$ the closed sub-prescheme of $U$ defined by $\sh{J}$.
We can suppose that $f|U$ is locally of finite presentation.
It follows from the definition of $Y$; it follows from the equivalence of \emph{c)} and \emph{a)} in \sref{IV.11.3.8} that the sequence $((h_i)_x)$ is regular in $\sh{O}_{X_s,x}$ and that $\sh{O}_{Y,x}$ is flat over $\sh{O}_{S,s}$; by virtue of \sref{IV.11.3.1} $f$ and $g$ are flat in a neighbourhood of $x$ and $y$ respectively.
It follows from \sref{IV.19.2.4} that $i$ is a transversally regular immersion relatively to $S$ (in replacing $Y$ by $Y\cap V$ for a neighbourhood open $V$ of $x$ in $X$).
On thus proved~(i) and we see that we can suppose that $g:Y\to S$ is a flat morphism.
Since $\sh{O}_{Y_s,x}$, quotient of $\sh{O}_{X_s,x}$ by the ideal generated by the $t_i$, is Artinian since by construction (\sref[0]{0.16.3.6}), $x$ is a maximal point of $Y_s$.
Finally, the ring $\sh{O}_{Y_s,x}$, being Artinian, is a ring de Cohen-Macaulay (\sref[0]{0.16.5.1}), hence, in replacing $Y$ by its intersection with a neighbourhood open of $x$ in $X$, we can suppose that $g$ is a Cohen-Macaulay morphism, by virtue of (\sref{IV.12.1.7},~(iii)).

In particular, we find the existence of ``quasi-sections'' \emph{flat} $Y$ of a morphism $f:X\to S$ at a point $x\in X_s$ which is \emph{closed} in $X_s$ and such that $f$ is flat at the point $x$ and $\sh{O}_{X_s,x}$ is a Cohen-Macaulay ring (\sref{IV.17.16.1}) and we see moreover that the canonical immersion $Y\to X$ is \emph{transversally regular} relatively to $S$.
This last complement is also valid for the quasi-section defined in (\sref{IV.17.16.3},~(i)).
\end{proof}

\subsection{Relative complete intersections (flat case)}
\label{subsection:IV.19.3}

\oldpage[IV-4]{194}

\begin{definition}[19.3.1]
\label{IV.19.3.1}
Let $\mathrm{A}$ be a local Noetherian ring.
We say that $\mathrm{A}$ is a \emph{complete intersection ring} (or an \emph{absolute complete intersection ring}, to avoid any risk of confusion) if its completion $\widehat{\mathrm{A}}$ is isomorphic to the quotient of a complete Noetherian regular local ring $\mathrm{B}$ by an ideal (i.e.\ \sref{IV.16.9.7}) \emph{generated by a regular sequence of elements} of $\mathrm{B}$.

Every regular local ring is a complete intersection ring.

We say that a locally Noetherian prescheme $X$ is a \emph{complete intersection} (absolute) at a point $x\in X$ if the ring $\sh{O}_{X,x}$ is a complete intersection ring.
\end{definition}

\begin{proposition}[19.3.2]
\label{IV.19.3.2}
Let $\mathrm{B}$ be a local Noetherian regular ring, $\mathfrak{J}$ an ideal of $\mathrm{B}$.
For $\mathrm{A}=\mathrm{B}/\mathfrak{J}$ to be a complete intersection ring, it is necessary and sufficient that the ideal $\mathfrak{J}$ be regular (i.e.\ \sref{IV.16.9.7} generated by a regular sequence of elements of $\mathrm{B}$).
\end{proposition}

\begin{proof}
We can write $\widehat{\mathrm{A}}=\widehat{\mathrm{B}}/\mathfrak{J}\widehat{\mathrm{B}}$ and since $\widehat{\mathrm{B}}$ is a $\mathrm{B}$-module faithfully flat and $\mathrm{B}$ Noetherian, $\mathfrak{J}\widehat{\mathrm{B}}$ is regular if and only if $\mathfrak{J}$ is \sref{IV.19.1.5},~(ii).
This shows (by virtue of \sref[0]{0.17.1.5},~(ii)) that one part of the condition is sufficient, and on the other hand that to prove that the condition is necessary, we can restrict ourselves to the case where $\mathrm{A}$ and $\mathrm{B}$ are \emph{complete}.
By hypothesis, there exists then a complete Noetherian regular local ring $\mathrm{B}'$ such that $\mathrm{A}$ is isomorphic to a quotient $\mathrm{B}'/\mathfrak{J}'$, where $\mathfrak{J}'$ is a regular ideal of $\mathrm{B}'$.
Let us consider then the fibre product $\mathrm{B}\times_{\mathrm{A}}\mathrm{B}'=\mathrm{B}''$ for the surjective homomorphisms $\mathrm{B}\to\mathrm{A}$, $\mathrm{B}'\to\mathrm{A}$ (\sref[0]{0.18.1.2}); we will first prove the following lemma:
\end{proof}

\begin{lemma}[19.3.2.1]
\label{IV.19.3.2.1}
Let $\mathrm{A}$, $\mathrm{E}$, $\mathrm{F}$ be three rings, $f:\mathrm{E}\to\mathrm{A}$, $g:\mathrm{F}\to\mathrm{A}$ two homomorphisms, and set $\mathrm{G}=\mathrm{E}\times_{\mathrm{A}}\mathrm{F}$ (\sref[0]{0.18.1.2}).
\begin{enumerate}[label=(\roman*)]
  \item If $f$ is surjective and if $\mathrm{E}$ and $\mathrm{F}$ are Noetherian rings, $\mathrm{G}$ is Noetherian.
  \item If $\mathrm{A}$, $\mathrm{E}$ and $\mathrm{F}$ are local rings and if $f$ and $g$ are local homomorphisms, $\mathrm{G}$ is local.
  \item If the conditions of~(i) and~(ii) are satisfied, if $g$ is surjective, if $\mathrm{E}$ and $\mathrm{F}$ are complete Noetherian local rings, $\mathrm{G}$ is complete.
\end{enumerate}
\end{lemma}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item Taking into account (\sref[0]{0.18.1.3} and \sref{IV.18.1.5}), $\mathrm{G}$ is an A-augmented ring over $\mathrm{F}$, whose ideal of augmentation $\mathfrak{J}'$ is canonically identified with that of the E-module structure on $\mathfrak{J}$ of $f$, via the canonical homomorphism $\mathrm{G}\to\mathrm{E}$.
If $\mathfrak{a}$ is any ideal of $\mathrm{G}$, $\mathfrak{a}/(\mathfrak{a}\cap\mathfrak{J}')$ is isomorphic to $(\mathfrak{a}+\mathfrak{J}')/\mathfrak{J}'$, hence to an ideal of $\mathrm{F}$, and is therefore of finite type; on the other hand, $\mathfrak{a}\cap\mathfrak{J}'$ is isomorphic to an ideal of $\mathrm{E}$ contained in $\mathfrak{J}$, hence also of finite type, which proves that $\mathfrak{a}$ is of finite type, hence $\mathrm{G}$ is Noetherian.
  \item Let $\mathfrak{r}$, $\mathfrak{s}$ be the maximal ideals of $\mathrm{E}$ and $\mathrm{F}$ respectively; the set $\mathfrak{m}=\mathfrak{r}\times_{\mathrm{A}}\mathfrak{s}$ is evidently an ideal of $\mathrm{G}$, inverse image of $\mathfrak{r}$ by the projection $\mathrm{G}\to\mathrm{E}$ and of $\mathfrak{s}$ by the projection $\mathrm{G}\to\mathrm{F}$.
On the other hand, if $(x,y)\notin\mathfrak{m}$, and if for example $x\notin\mathfrak{r}$, $f(x)$ does not belong to the maximal ideal of $\mathrm{A}$, hence, since $g(y)=f(x)$, and $g$ is local, we also have $y\notin\mathfrak{s}$; we
\oldpage[IV-4]{195}
conclude that $x$ and $y$ are invertible, hence $(x,y)$ is invertible, which completes the proof of~(ii).
  \item The image of $\mathfrak{m}$ by the surjective homomorphism $\mathrm{G}\to\mathrm{E}$ is the maximal ideal $\mathfrak{r}$ of $\mathrm{E}$, hence the image of $\mathfrak{m}^n\cap\mathfrak{J}'$ is $\mathfrak{r}^n\cap\mathfrak{J}'$, and since $\mathfrak{J}$ is closed (hence complete) for the topology induced by the $\mathfrak{r}$-adic topology of $\mathrm{E}$ (\sref[0]{0.7.3.5}), $\mathfrak{J}'$ is complete for the topology induced by the $\mathfrak{m}$-preadic topology of $\mathrm{G}$.
On the other hand, $\mathrm{G}/\mathfrak{J}'$, equipped with the topology induced, is isomorphic to $\mathrm{F}$ equipped with the $\mathfrak{s}$-adic topology, hence is complete for the quotient topology of the $\mathfrak{m}$-preadic topology of $\mathrm{G}$.
We deduce that $\mathrm{G}$ is complete for the $\mathfrak{m}$-preadic topology.
\end{enumerate}

This lemma being established, it follows from the Cohen theorem (\sref[0]{0.19.8.8}) that $\mathrm{B}''$ is isomorphic to a quotient of a complete Noetherian local \emph{regular} ring $\mathrm{C}$.
It then follows from \sref{IV.19.1.2} that the immersion $\operatorname{Spec}(\mathrm{B}')\to\operatorname{Spec}(\mathrm{C})$ is regular; since by hypothesis $\operatorname{Spec}(\mathrm{A})\to\operatorname{Spec}(\mathrm{B}')$, we conclude by \sref{IV.19.1.5},~(iii) that the immersion $\operatorname{Spec}(\mathrm{A})\to\operatorname{Spec}(\mathrm{C})$ is regular; but this immersion is written also as the composite $\operatorname{Spec}(\mathrm{A})\to\operatorname{Spec}(\mathrm{B})\to\operatorname{Spec}(\mathrm{C})$.
Now as $\mathrm{B}$ is regular by hypothesis, the immersion $\operatorname{Spec}(\mathrm{B})\to\operatorname{Spec}(\mathrm{C})$ is regular \sref{IV.19.1.2}; hence it is the same for $\operatorname{Spec}(\mathrm{A})\to\operatorname{Spec}(\mathrm{B})$ by \sref{IV.19.1.5},~(iv).
\end{proof}

\begin{corollary}[19.3.3]
\label{IV.19.3.3}
Let $X$ be a locally Noetherian prescheme locally immersible in a regular prescheme.
Then the set of $x\in X$ where $X$ is a complete intersection is open in $X$ and contains the set of regular points of $X$.
\end{corollary}

\begin{proof}
We can indeed restrict ourselves to the case where $X=\operatorname{Spec}(\mathrm{B}/\mathfrak{J})$, where $\mathrm{B}$ is a regular ring and $\mathfrak{J}$ an ideal of $\mathrm{B}$.
The set of $\mathfrak{p}\supset\mathfrak{J}$ in $\operatorname{Spec}(\mathrm{B})$ such that $\mathfrak{J}_\mathfrak{p}$ is regular in $\mathrm{B}_\mathfrak{p}$ is then open (\sref{IV.16.9.5}), and this is the set of points of $\operatorname{Spec}(\mathrm{B})$ where $X$ is a complete intersection by virtue of \sref{IV.19.3.2}.
\end{proof}

\begin{corollary}[19.3.4]
\label{IV.19.3.4}
Let $k$ be a field, $X$ a prescheme locally of finite type over $k$, $k'$ an extension of $k$, $X'=X\otimes_k k'$.
Let $x'$ be a point of $X'$, $x$ its projection in $X$.
For $X$ to be a complete intersection at $x$, it is necessary and sufficient that $X'$ be a complete intersection at $x'$.
\end{corollary}

\begin{proof}
The question being local on $X$, we can suppose that $X=\operatorname{Spec}(\mathrm{A})$, where $\mathrm{A}$ is a $k$-algebra of finite type, hence a quotient of a polynomial algebra $k[\mathrm{T}_1,\ldots,\mathrm{T}_n]$, so $X$ is a closed sub-prescheme of the regular scheme $Y=\operatorname{Spec}(k[\mathrm{T}_1,\ldots,\mathrm{T}_n])$ (\sref[0]{0.17.3.7}).
If we set $Y'=Y\otimes_k k'=\operatorname{Spec}(k'[\mathrm{T}_1,\ldots,\mathrm{T}_n])$, $Y'$ is also a regular scheme.
Now, since $\operatorname{Spec}(k')$ is faithfully flat over $\operatorname{Spec}(k)$, it follows from \sref{IV.19.1.5},~(ii) and from the fact that we are dealing with Noetherian preschemes that the immersion $Y\to X$ is regular at the point $x$ if and only if the immersion $Y'\to X'$ is so at the point $x'$.
The conclusion then follows from the criterion \sref{IV.19.3.2}.
\end{proof}

\begin{corollary}[19.3.5]
\label{IV.19.3.5}
Let $\mathrm{A}$, $\mathrm{C}$ be two Noetherian local rings, $\rho:\mathrm{A}\to\mathrm{C}$ a local homomorphism, $\mathrm{B}=\mathrm{C}/\mathfrak{J}$ a quotient of $\mathrm{C}$, $k$ the residue field of $\mathrm{A}$.
Suppose that $\rho$ makes $\mathrm{C}$ a flat $\mathrm{A}$-module, and that the ring $\mathrm{C}\otimes_{\mathrm{A}} k$ is regular.
Then the following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item The ideal $\mathfrak{J}$ is transversally regular relatively to $\mathrm{A}$.
  \item $\mathrm{B}$ is a flat $\mathrm{A}$-module, and $\mathrm{B}\otimes_{\mathrm{A}} k$ is a complete intersection ring.
\end{enumerate}
\end{corollary}

\begin{proof}
\oldpage[IV-4]{196}
By virtue of \sref{IV.19.2.4}, condition \emph{a)} amounts to saying that $\mathrm{B}$ is a flat $\mathrm{A}$-module and that $\mathfrak{J}\otimes_{\mathrm{A}} k$ (which is identified with an ideal of $\mathrm{C}\otimes_{\mathrm{A}} k$ by virtue of the flatness of $\mathrm{B}$ over $\mathrm{A}$ (\sref[0]{0.6.1.2})) is regular in this ring.
Since the ring $\mathrm{C}\otimes_{\mathrm{A}} k$ is regular, it amounts to the same, by virtue of \sref{IV.19.3.2}, to say (when $\mathrm{B}$ is a flat $\mathrm{A}$-module) that $\mathfrak{J}\otimes_{\mathrm{A}} k$ is a regular ideal of $\mathrm{C}\otimes_{\mathrm{A}} k$, or that $\mathrm{B}\otimes_{\mathrm{A}} k=(\mathrm{C}\otimes_{\mathrm{A}} k)/(\mathfrak{J}\otimes_{\mathrm{A}} k)$ is a complete intersection ring; whence the corollary.
\end{proof}

\begin{definition}[19.3.6]
\label{IV.19.3.6}
Let $f:X\to S$ be a flat morphism locally of finite presentation.
We say that $X$ is a \emph{complete intersection relatively to $S$} at a point $x$, if the fibre $f^{-1}(f(x))$ is a complete intersection (absolute) at the point $x$.
We say that $X$ is a \emph{complete intersection relatively to $S$}, or that the morphism $f$ is a \emph{morphism of complete intersection}, if $X$ is a complete intersection relatively to $S$ at each of its points.
\end{definition}

\begin{proposition}[19.3.7]
\label{IV.19.3.7}
Let $g:X\to S$, $h:Y\to S$ be two flat morphisms locally of finite presentation, $f:Y\to X$ an S-immersion, $y$ a point of $Y$, $x=f(y)$, $s=g(x)=h(y)$.
Suppose that the fibre $X_s=g^{-1}(s)$ is regular at the point $x$.
Then, for $f$ to be a transversally regular immersion at the point $y$, it is necessary and sufficient that $Y$ be a complete intersection relatively to $S$ at the point $y$.
\end{proposition}

\begin{proof}
To say that $f$ is transversally regular at the point $y$ amounts, by virtue of \sref{IV.19.2.4}, to saying that $f_s:Y_s\to X_s$ is a regular immersion at the point $y$.
But since $X_s$ is regular at the point $x=f(y)$, this is equivalent, by virtue of \sref{IV.19.3.2}, to saying that $Y_s$ is a complete intersection (absolute) at the point $y$, whence the proposition.
\end{proof}

\begin{corollary}[19.3.8]
\label{IV.19.3.8}
Let $f:X\to S$ be a flat morphism locally of finite presentation.
The set $U$ of $x\in X$ for which $X$ is a complete intersection relatively to $S$ is open in $X$.
If moreover $f$ is proper, the set $E$ of $s\in S$ such that $X_s=f^{-1}(s)$ is a complete intersection at each of its points is an open subset of $S$.
\end{corollary}

\begin{proof}
Since $E=Y-f(X-U)$, the second assertion follows from the first and from the fact that $f$ is a closed map.
The first assertion is local on $X$, hence we can suppose that $X$ is a closed sub-prescheme of a prescheme of the form $Z=S[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ ($\mathrm{T}_i$ indeterminate).
This last being smooth over $S$ (\sref{IV.17.3.8}), its fibres $Z_s$ are regular, and it amounts to the same, by virtue of \sref{IV.19.3.7}, to say that $x\in U$ or that the immersion $X\to Z$ is transversally regular at the point $x$; we conclude by the corollary by \sref{IV.19.2.2}.
\end{proof}

\begin{proposition}[19.3.9]
\label{IV.19.3.9}
\medskip\noindent
\begin{enumerate}[label=(\roman*)]
  \item Every open immersion is a complete intersection.
  \item Let $f:X\to S$ be a flat morphism and locally of finite presentation which is a morphism of complete intersection.
For every change of base $g:S'\to S$, $f'=f_{(S')}:X_{(S')}\to S'$ is a morphism of complete intersection.
The converse is true if $g$ is faithfully flat and quasi-compact.
  \item If $f:X\to Y$ and $g:Y\to Z$ are flat morphisms locally of finite presentation which are complete intersection morphisms, so is $g\circ f:X\to Z$.
\end{enumerate}
\end{proposition}

\begin{proof}
\oldpage[IV-4]{197}
Assertion~(i) follows immediately from the definition \sref{IV.19.3.1}.
To prove~(ii), note first that if $f$ is flat and locally of finite presentation, the same is true of $f'$ (\sref{IV.2.5.1} and \sref{IV.2.7.1}); moreover, for every $s'\in S'$, if $s=g(s')$, it is equivalent to say that $f^{-1}(s)$ is a complete intersection or that $f'^{-1}(s')$ is a complete intersection \sref{IV.19.3.4}; whence~(ii), taking into account that $g$ is surjective when it is faithfully flat.

Finally, under the hypotheses of~(iii), $g\circ f$ is flat and locally of finite presentation.
By definition \sref{IV.19.3.6}, we are reduced to the case where $Z$ is the spectrum of a field, and we have to prove then that the local rings $\sh{O}_{X,x}$ are complete intersection rings, which follows from the more general result:
\end{proof}

\begin{corollary}[19.3.10]
\label{IV.19.3.10}
Let $f:X\to S$ be a flat morphism locally of finite presentation which is a complete intersection morphism, $x$ a point of $X$, $s=f(x)$.
If $\sh{O}_{S,s}$ is a complete Noetherian intersection ring, so is $\sh{O}_{X,x}$.
\end{corollary}

\begin{proof}
We can evidently reduce to the case where $S=\operatorname{Spec}(\mathrm{A})$ with $\mathrm{A}=\sh{O}_{S,s}$.
Let us show moreover that we can restrict ourselves to the case where the local ring $\mathrm{A}$ is \emph{complete}.
Indeed, set $\mathrm{A}'=\widehat{\mathrm{A}}$, $X'=X\otimes_{\mathrm{A}}\mathrm{A}'$, and let $s'$ be the unique closed point of $S'=\operatorname{Spec}(\mathrm{A}')$, which is the unique point above $S$; if $f'=f_{(S')}:X'\to S'$, the fibre $f'^{-1}(s')$ is canonically isomorphic to $f^{-1}(s)$, since $\mathrm{A}$ and $\mathrm{A}'$ have the same residue field (\sref[I]{I.3.6.4}); there is thus a single point $x'\in X'$ above $x$ and above $s'$, and we have $\sh{O}_{X',x'}=\sh{O}_{X,x}\otimes_{\mathrm{A}}\mathrm{A}'$.
We deduce that the local rings $\sh{O}_{X,x}$ and $\sh{O}_{X',x'}$ have isomorphic \emph{completions}, since in general, if $\mathrm{E}$ is a local ring which is a local $\mathrm{A}$-algebra (the local homomorphism $\mathrm{A}\to\mathrm{E}$ being local), the separated completion $(\mathrm{E}\otimes_{\mathrm{A}}\widehat{\mathrm{A}})^\wedge$ of the ring $\mathrm{E}\otimes_{\mathrm{A}}\widehat{\mathrm{A}}$ equipped with the product topology is isomorphic to the separated completion $\widehat{\mathrm{E}}\otimes_{\mathrm{A}}\widehat{\mathrm{A}}\cong\widehat{\mathrm{E}}$, hence to the separated completed $\widehat{\mathrm{E}}$ of $\mathrm{E}$ (\sref[0]{0.7.7.1}), so it amounts to the same to say that $\sh{O}_{X,x}$ is a complete intersection ring, or that $\sh{O}_{X',x'}$ is one.

Suppose therefore that $\mathrm{A}$ is \emph{complete}; we can suppose moreover that $X=\operatorname{Spec}(\mathrm{B})$, $\mathrm{B}$ being a quotient of a polynomial algebra $\mathrm{C}=\mathrm{A}[\mathrm{T}_1,\ldots,\mathrm{T}_r]$.
Since the polynomial algebras over a field are regular rings (\sref[0]{0.17.3.7}), it follows from \sref{IV.19.3.7} that the immersion $X\to Z=\operatorname{Spec}(\mathrm{C})$ can be supposed transversally regular relatively to $S$; hence we can suppose that $\mathrm{B}=\mathrm{C}/\mathfrak{J}$, where $\mathfrak{J}$ is generated by a regular sequence $(g_i)_{1\leq i\leq n}$ of elements of $\mathrm{C}$.
On the other hand, since $\mathrm{A}$ is complete, we can write by hypothesis $\mathrm{A}'\to\mathrm{A}'/\mathfrak{R}$, where $\mathrm{A}'$ is a regular local ring and $\mathfrak{R}$ an ideal of $\mathrm{A}'$ (\sref[0]{0.19.8.8}), and the hypothesis that $\mathrm{A}$ is a complete intersection ring implies that $\mathfrak{R}$ is generated by a regular sequence $(f_j)_{1\leq j\leq m}$ \sref{IV.19.3.2}.
In the polynomial algebra $\mathrm{C}'=\mathrm{A}'[\mathrm{T}_1,\ldots,\mathrm{T}_r]$, the elements $f_j$ ($1\leq j\leq m$) form a regular sequence generating the ideal $\mathfrak{R}'=\mathfrak{R}[\mathrm{T}_1,\ldots,\mathrm{T}_r]$ (\sref[0]{0.15.1.4}); since $\mathrm{C}'/\mathfrak{R}'=\mathrm{C}$, we see that since, for each $i$, $g_i'$ is an element of $\mathrm{C}'$ with image $g_i$ in $\mathrm{C}$, the sequence formed of the $f_j$ ($1\leq j\leq m$) and the $g_i'$ ($1\leq i\leq n$) is a regular sequence in $\mathrm{C}'$; if $\mathfrak{J}'$ is the ideal of $\mathrm{C}'$ that it generates, we have $\mathrm{B}=\mathrm{C}'/\mathfrak{J}'$, whence the conclusion by virtue of \sref{IV.19.3.2}, since $\mathrm{C}'$ is a regular ring (\sref[0]{0.17.3.7}).
\end{proof}

\subsection{Application: criteria of regularity and smoothness for blown-up preschemes}
\label{subsection:IV.19.4}

\oldpage[IV-4]{198}

\begin{env}[19.4.1]
\label{IV.19.4.1}
Let $X$ be a prescheme, $Y$ a closed sub-prescheme of $X$, defined by a quasi-coherent Ideal $\sh{J}$ of $\sh{O}_X$.
Recall (\sref[II]{II.8.1.3}) that the X-scheme $X'$ obtained by blowing up $Y$ is the prescheme
\[
  X'=\operatorname{Proj}(\sh{S}), \quad \text{where}\quad \sh{S}=\bigoplus_{n\geq 0}\sh{J}^n.
\]
If $\sh{J}$ is of finite type, the structural morphism $f:X'\to X$ is projective.
Without hypothesis on $\sh{J}$, the closed sub-prescheme
\[
  Y'=f^{-1}(Y)
\]
of $X'$ is defined by the quasi-coherent Ideal $\sh{J}\sh{O}_{X'}$ of $\sh{O}_{X'}$, canonically isomorphic to $\sh{O}_{X'}(1)$; more precisely, we have an exact sequence
\[
  0\to\sh{O}_{X'}(1)\xrightarrow{\widetilde{u}_0}\sh{O}_{X'}\to\sh{O}_{Y'}\to 0
\]
\hyperref[IV.19.4.1.1]{(\textbf{19.4.1.1})}
\label{IV.19.4.1.1}
where $\sh{J}\sh{O}_{X'}$ is the image of $\widetilde{u}_0$ (\sref[II]{II.8.1.7} and \sref{IV.8.1.8}).
Since in a neighbourhood of a point of $X'$, the $\sh{O}_{X'}$-Module $\sh{O}_{X'}(1)$ is free of rank $1$, $\sh{J}\sh{O}_{X'}$ is generated in such a neighbourhood by an invertible section of $\sh{O}_{X'}$ and $\sh{J}\sh{O}_{X'}/\sh{J}^2\sh{O}_{X'}$ is therefore an $\sh{O}_{X'}$-Module free of rank $1$.
This proves the first assertion of the following lemma:
\end{env}

\begin{lemma}[19.4.2]
\label{IV.19.4.2}
The canonical immersion $Y'\to X'$ is regular and of codimension $1$, and $Y'$ is canonically isomorphic to the Y-scheme $\operatorname{Proj}(\operatorname{gr}_\sh{J}^*(\sh{O}_X))$.
\end{lemma}

\begin{proof}
The second assertion follows from (\sref[II]{II.3.5.3}) applied to the canonical injection $Y\to X$, taking into account that $\sh{O}_Y$ is isomorphic to $\sh{O}_X/\sh{J}$ and $\sh{J}^n\otimes_{\sh{O}_X}\sh{O}_Y=\operatorname{gr}_\sh{J}^*(\sh{O}_X)$ by definition, since $\sh{O}_Y$ is isomorphic to $\sh{O}_X/\sh{J}$ and $\sh{J}^n\otimes_{\sh{O}_X}(\sh{O}_X/\sh{J})$ to $\sh{J}^n/\sh{J}^{n+1}$.
\end{proof}

\begin{corollary}[19.4.3]
\label{IV.19.4.3}
If the canonical immersion $Y\to X$ is quasi-regular, the restriction $g:Y'\to Y$ of the morphism $f$ is smooth.
\end{corollary}

\begin{proof}
Indeed, $\sh{N}_{Y/X}=\sh{J}/\sh{J}^2$ is then an $\sh{O}_Y$-Module locally free of finite rank and $\operatorname{gr}_\sh{J}^*(\sh{O}_X)$ is isomorphic to $\mathbf{S}_{\sh{O}_Y}^*(\sh{N}_{Y/X})$ \sref{IV.16.9.8}, hence $Y'$ is Y-isomorphic to $\mathbf{P}(\sh{N}_{Y/X})$ and is therefore smooth over $Y$ \sref{IV.17.3.9}.
\end{proof}

We shall see later (chap.~V) that $g$ is smooth also when $Y\to X$ ``has only ordinary singularities''.

\begin{proposition}[19.4.4]
\label{IV.19.4.4}
With the notations of \sref{IV.19.4.1}, suppose that $X$ is locally Noetherian and that the morphism $g:Y'\to Y$, restriction of $f$, is smooth.
Let $x'$ be a point of $Y'$, $x=g(x')$.
For $Y$ to be regular at the point $x$, it is necessary and sufficient that $Y'$ be regular at the point $x'$; when this is so, $X'$ is then regular at the point $x'$.
\end{proposition}

\begin{proof}
The first assertion follows from \sref{IV.17.5.8} and the second from \sref{IV.19.1.1} and \sref{IV.19.4.2}.
\end{proof}

\begin{corollary}[19.4.5]
\label{IV.19.4.5}
Under the general hypotheses of \sref{IV.19.4.4}, suppose that $Y$ is regular at the point $x$.
Then, for every generization $x_1$ of $x$ in $X$ not belonging to $Y$, $X$ is regular at the point $x_1$.
If $\operatorname{Reg}(X)$ is open ($(\sref{IV.6.12})$, for example if $X$ is excellent (\sref{IV.7.8.6},~(iii)))
\oldpage[IV-4]{199}
and if $Y$ is regular, then there exists an open neighbourhood $U$ of $Y$ in $X$ such that $X$ is regular at the points of $U-Y$.
\end{corollary}

\begin{proof}
To prove the first assertion, we can restrict ourselves to the case where $X$ is a scheme local of closed point $x$; the hypothesis then implies that $Y$ is regular (\sref[0]{0.17.3.2}), hence it is the same for $Y'$ by \sref{IV.19.4.4}.
Let us now note that the morphism $f:X'\to X$ is projective, hence proper (\sref[II]{II.7.2.1}), if we consider the unique point $x_1'$ of $X'$ above $x_1$ belonging to $Y'$; applying \sref{IV.19.4.4} and using (\sref[0]{0.17.3.2}) we conclude that $X'$ is regular at the point $x_1$, hence $X$ is regular at the point $x_1'$ since $X'-Y'$ is isomorphic to $X-Y$ (\sref[II]{II.8.1.3}).
If $Y$ is regular, every point of $X-Y$ which is a generization of a point of $Y$ belongs to $\operatorname{Reg}(X)$; if $\operatorname{Reg}(X)$ is open, $U=Y\cup\operatorname{Reg}(X)$ is then locally constructible and stable under generization, hence open (\sref[0]{0.9.2.5}), and responds to the question.
\end{proof}

\begin{proposition}[19.4.6]
\label{IV.19.4.6}
Let $g:X\to S$ be a morphism, $Y$ a closed sub-prescheme of $X$ defined by a quasi-coherent Ideal $\sh{J}$, $X'$ the X-scheme obtained by blowing up $Y$, $Y'$ the inverse image of $Y$ in $X'$.
\begin{enumerate}[label=(\roman*)]
  \item The following conditions are equivalent:
  \begin{enumerate}[label=\emph{\alph*)}]
    \item $\operatorname{gr}_\sh{J}^*(\sh{O}_X)$ is a flat $\sh{O}_X$-Algebra.
    \item For every $n\geq 0$, the $\sh{O}_X$-Module $\sh{O}_X/\sh{J}^{n+1}$ is $g$-flat (in other words, the $n$-th infinitesimal neighbourhood $Y^{(n)}$ of $Y$ in $X$ \sref{IV.16.1.2} is S-flat).
    \item For every change of base $S_1\to S$ and every integer $n\geq 1$, if we set $X_1=X\times_S S_1$, the canonical homomorphism $\sh{J}^n\otimes_{\sh{O}_g}\sh{O}_{S_1}\to\sh{J}^n\sh{O}_{X_1}$ is bijective.
  \end{enumerate}
When this is so, $Y'$ is S-flat, and for every change of base $S_1\to S$, if $Y_1$ is the inverse image of $Y$ in $X_1$, the prescheme $X_1'$ obtained by blowing up $Y_1$ in $X_1$ is canonically isomorphic to $X'\times_S S_1$.
  \item Suppose that the equivalent conditions of~(i) are satisfied and moreover that the morphisms $X\to S$ and $Y\to S$ are locally of finite presentation.
Then $\sh{S}=\bigoplus_{n\geq 0}\sh{J}^n$ is a $\sh{O}_X$-Algebra of finite presentation, the morphism $X'\to X$ is of finite presentation, the canonical immersion $Y'\to X'$ is transversally regular of codimension $1$ relatively to $S$, and the set of points of $X$ (resp.\ $X'$) where $X$ (resp.\ $X'$) is S-flat, is a neighbourhood of $Y$ (resp.\ $Y'$).
\end{enumerate}
\end{proposition}

\begin{proof}
\oldpage[IV-4]{200}
\begin{enumerate}[label=(\roman*)]
  \item The equivalence of \emph{a)} and \emph{b)} follows immediately from \sref[0]{0.6.1.2}; to show the equivalence of \emph{b)} and \emph{c)}, we can restrict ourselves to the case where $S=\operatorname{Spec}(\mathrm{A})$ and $X=\operatorname{Spec}(\mathrm{B})$ are affine, with $\sh{J}=\widetilde{\mathfrak{J}}$, where $\mathfrak{J}$ is an ideal of $\mathrm{B}$; the Tor exact sequence shows that condition \emph{c)} implies that $\operatorname{Tor}_1^{\mathrm{A}}(\mathrm{B}/\mathfrak{J}^{n+1},\mathrm{A}')=0$ for every $n\geq 0$ and every $\mathrm{A}$-algebra direct sum $\mathrm{A}\oplus\mathrm{M}$, where $\mathrm{M}$ is an $\mathrm{A}$-module, where the multiplication is defined by $(a,m)(a',m')=(aa',am'+a'm)$; we see that the preceding condition is equivalent to $\operatorname{Tor}_1^{\mathrm{A}}(\mathrm{B}/\mathfrak{J}^{n+1},\mathrm{M})=0$, hence to the fact that $\mathrm{B}/\mathfrak{J}^{n+1}$ is a flat $\mathrm{A}$-module.
If we set $\sh{J}_1=\sh{J}\otimes_{\sh{O}_g}\sh{O}_{S_1}$, we then have $\bigoplus_{n\geq 0}\sh{J}_1^n=(\bigoplus_{n\geq 0}\sh{J}^n)\otimes_{\sh{O}_g}\sh{O}_{S_1}$ and the assertion relative to the blown-up prescheme $X_1'$ follows immediately.
Finally, to prove that $Y'$ is S-flat, we can further restrict ourselves to the case where $S$ and $X$ are affine; if we set $\mathrm{C}=\operatorname{gr}_\mathfrak{J}^*(\mathrm{B})$, then every point of $Y'$ admits, by virtue of \sref{IV.19.4.2} and (\sref[II]{II.2.3.6}), a neighbourhood whose affine ring is of the form $\mathrm{C}_{(t)}$, where $t$ is a homogeneous element of degree $1$ of $\mathrm{C}$; since by hypothesis $\mathrm{C}$ is a flat $\mathrm{A}$-module, so is its ring of fractions $\mathrm{C}_{(t)}$, and of the component of degree $0$, $\mathrm{C}_{(t)}$, of this $\mathrm{A}$-module graded, whence our assertion.

  \item We can again restrict ourselves to the case where $S$ and $X$ are affine, $\mathrm{B}$ being therefore a finitely presented $\mathrm{A}$-algebra, $\mathfrak{J}$ an ideal of finite type of $\mathrm{B}$.
By virtue of (\sref{IV.8.9.1}), (\sref{IV.8.6.3}) and \sref{IV.11.2.9}, there exists a Noetherian subring $\mathrm{A}_0$ of $\mathrm{A}$, an $\mathrm{A}_0$-algebra of finite type $\mathrm{B}_0$, an ideal $\mathfrak{J}_0$ of type finite of $\mathrm{B}_0$ such that $\mathrm{B}=\mathrm{B}_0\otimes_{\mathrm{A}_0}\mathrm{A}$, $\mathfrak{J}=\mathfrak{J}_0\mathrm{B}$, that $\operatorname{gr}_{\mathfrak{J}_0}^*(\mathrm{B}_0)$ is an $\mathrm{A}_0$-module \emph{flat} and that $\bigoplus_{n\geq 0}\mathfrak{J}_0^n$ is a finitely presented $\mathrm{B}_0$-algebra of finite presentation.
Likewise, the morphism $\operatorname{Proj}(\bigoplus\mathfrak{J}^n)\to\operatorname{Spec}(\mathrm{B}_0)$ is of type finite (\sref[II]{II.2.7.1}), and by base change, is of finite presentation; hence the morphism $X'\to X$, which follows by base change, is of finite presentation.
We already knew (\sref{IV.19.4.2}) that the canonical immersion $Y'\to X'$ is regular; since $Y'$ is S-flat, as we have seen in~(i), we deduce from \sref{IV.19.2.4} that the immersion $Y'\to X'$ is transversally regular relatively to $S$ and that $X'$ is S-flat at the points of $Y'$, hence \sref{IV.11.3.1}, $X$ is S-flat at the points of $Y$, hence (\sref{IV.11.3.4}), $X$ is S-flat at the points of $Y$ in a neighbourhood of $Y$.
On the other hand, by the $\sh{O}_X$-Algebra S-flat, is a flat $\sh{O}_X$-Algebra of finite presentation; hence the morphism $X'\to X$ is of finite presentation, hence (\sref{IV.11.3.1}), the fact that $\operatorname{gr}_\sh{J}^*(\sh{O}_X)$ is a flat $\sh{O}_X$-Algebra S-plate implies \sref{IV.11.3.4}, that $X$ is S-flat in a neighbourhood of $Y$.
\end{enumerate}
\end{proof}

\begin{corollary}[19.4.7]
\label{IV.19.4.7}
Let $g:X\to S$ be a morphism locally of finite presentation, $Y$ a closed sub-prescheme of $X$ such that the composite morphism $Y\to X\to S$ is locally of finite presentation; suppose moreover that $Y$ is S-flat and that $\sh{O}_X$ is normally flat along $Y$ (\sref{IV.11.3.4}).
Then (with the notations of \sref{IV.19.4.1}), the morphism $X'\to X$ is of finite presentation, the canonical immersion $Y'\to X'$ is transversally regular of codimension $1$ relatively to $S$, and the set of points of $X$ (resp.\ $X'$) where $X$ (resp.\ $X'$) is S-flat is a neighbourhood of $Y$ (resp.\ $Y'$).
\end{corollary}

\begin{proof}
Indeed, by hypothesis $\operatorname{gr}_\sh{J}^*(\sh{O}_X)$ is an $\sh{O}_Y$-Module flat, hence is S-flat.
We can therefore apply the results of \sref{IV.19.4.6},~(i) and~(ii).
Comme $Y'$ is Y-isomorphic to $\operatorname{Proj}(\operatorname{gr}_\sh{J}^*(\sh{O}_X))$ \sref{IV.19.4.2}, the same reasoning as in \sref{IV.19.4.6} shows that $Y'$ is Y-flat.
\end{proof}

\begin{proposition}[19.4.8]
\label{IV.19.4.8}
Suppose the hypotheses of \sref{IV.19.4.7} satisfied and moreover that the morphism $Y'\to Y$ is smooth.
Let $x'$ be a point of $Y'$, $x$ its image in $Y$.
For $Y$ to be smooth over $S$ at the point $x$, it is necessary and sufficient that $Y'$ be smooth over $S$ at the point $x'$; when this is so, $X'$ is then smooth over $S$ at the point $x'$.
\end{proposition}

\begin{proof}
The preceding conclusions are in particular satisfied when the morphism $X\to S$ is locally of finite presentation and the immersion $Y\to X$ is transversally regular relatively to $S$.

Taking into account the compatibility of the formation of a blown-up prescheme with changes of base in the case envisaged \sref{IV.19.4.6},~(i) and of (\sref{IV.17.5.1}) and \sref{IV.17.7.1},
\oldpage[IV-4]{201}
it suffices to consider the case where $S$ is the spectrum of an algebraically closed field.
But then it amounts to the same to say that an S-prescheme locally of finite type is smooth or that it is regular at this point (\sref{IV.17.15.1}), hence the conclusions follow from \sref{IV.19.4.4}.

When $X\to S$ is locally of finite presentation, and the canonical immersion $Y'\to X$ is quasi-regular, this immersion is by definition of finite presentation since the Ideal that defines it is of finite type, hence the composite morphism $Y\to X\to S$ is locally of finite presentation.
We have already seen \sref{IV.19.4.3} that in these conditions the morphism $Y'\to Y$ is smooth; moreover, we know then that $\sh{N}_{Y/X}$ is an $\sh{O}_Y$-Module locally free and that $\operatorname{gr}_\sh{J}^*(\sh{O}_X)$ is isomorphic to $\mathbf{S}^*_{\sh{O}_Y}(\sh{N}_{Y/X})$, hence is a flat $\sh{O}_X$-Module.
The hypotheses of \sref{IV.19.4.7} are thus all satisfied and we can apply the conclusions of \sref{IV.19.4.8}.
\end{proof}

\begin{corollary}[19.4.9]
\label{IV.19.4.9}
Under the general hypotheses of \sref{IV.19.4.8}, suppose moreover that $Y$ is smooth over $S$ at the point $x$.
Then, for every generization $x_1$ of $x$ in $X$ not belonging to $Y$, $X$ is smooth over $S$ at the point $x_1$.
If $Y$ is smooth over $S$, then there exists an open neighbourhood $U$ of $Y$ in $X$ such that $X$ is smooth over $S$ at the points of $U-Y$.
\end{corollary}

\begin{proof}
To prove the first assertion, we can restrict ourselves to the case where $X$ is a scheme local of closed point $x$, and the hypothesis implies then that $Y$ is smooth over $S$, every closed neighbourhood of the point being necessarily $Y$ itself; we conclude then by the same reasoning as in \sref{IV.19.4.5}, using the fact that the set of points where $X'$ is smooth over $S$ is open in $X'$.
To prove the second assertion, we can reduce to the case where $S$ is Noetherian, and $X$ of finite type over $S$, using (\sref{IV.8.9.1}), (\sref{IV.8.6.3}), (\sref{IV.11.2.9}) and \sref{IV.17.7.6}; we then conclude as in \sref{IV.19.4.5}, using the fact that the set of points where $X$ is smooth over $S$ is open in $X$.
\end{proof}

\begin{remark}[19.4.10]
\label{IV.19.4.10}
In \sref{IV.19.4.6},~(ii), replace the hypothesis that $X$ and $Y$ are locally of finite presentation over $S$ by the hypothesis that $S$ and $X$ (hence also $Y$) are \emph{locally Noetherian}.
Then it is clear that $\bigoplus_{n\geq 0}\sh{J}^n$ is an $\sh{O}_X$-Algebra of finite type, that the morphism $X'\to X$ is of finite type, and it follows again from \sref{IV.19.2.4} applied to locally Noetherian preschemes, that the immersion $Y'\to X'$ is transversally regular relatively to $S$ and that $X'$ is S-flat at the points of $Y'$.
\end{remark}

\begin{proposition}[19.4.11]
\label{IV.19.4.11}
Let $X$ be a prescheme, $\sh{E}$ an $\sh{O}_X$-Module quasi-coherent, $u:\sh{E}\to\sh{O}_X$ a homomorphism of $\sh{O}_X$-Modules, $\sh{J}=u(\sh{E})$ the quasi-coherent Ideal of $\sh{O}_X$ which is the image of $u$, $Y$ the closed sub-prescheme of $X$ defined by $\sh{J}$.
On the other hand, let $\mathrm{P}=\mathbf{P}(\sh{E})$ be the projective bundle over $X$ defined by $\sh{E}$ (\sref[II]{II.4.1.1}), $p:\mathrm{P}\to X$ the structural morphism, $\alpha_1^{\#}:p^*(\sh{E})\to\sh{O}_{\mathrm{P}}(1)$ the canonical homomorphism (\sref[II]{II.4.1.5.1}) and $\sh{H}$ its kernel, so that we have an exact sequence
\[
  0\to\sh{H}\to p^*(\sh{E})\xrightarrow{\alpha_1^{\#}}\sh{O}_{\mathrm{P}}(1)\to 0.
\]
Finally, let $\sh{K}$ be the quasi-coherent Ideal of $\sh{O}_{\mathrm{P}}$ which is the image of the restriction $v:\sh{H}\to\sh{O}_{\mathrm{P}}$ of $p^*(u)$, and let $Z$ be the closed sub-prescheme of $\mathrm{P}$ defined by $\sh{K}$.
\oldpage[IV-4]{202}
\begin{enumerate}[label=(\roman*)]
  \item The image of $X'$ by the closed immersion $j:X'\to\mathbf{P}(\sh{E})$ corresponding to the surjective homomorphism of graded $\sh{O}_X$-Algebras $\mathbf{S}_{\sh{O}_X}(\sh{J})\to\bigoplus_{n\geq 0}\sh{J}^n$ (\sref[II]{II.3.6.2}) is a closed sub-prescheme of $Z$.
  \item If $\sh{E}$ is an $\sh{O}_X$-Module locally free of rank $m$, $\sh{H}$ is an $\sh{O}_{\mathrm{P}}$-Module locally free of rank $m-1$.
  \item Suppose $X$ is locally Noetherian, that $\sh{E}$ is locally free of finite type, that at every point $x\in Y$, the canonical immersion $i:Y\to X$ is regular and of codimension equal to $\operatorname{rg}_{\sh{O}_x}(\sh{E}_x)$ (which we shall express by saying that the homomorphism $u$ is regular at the point $x$ (cf.\ chap.~V)).
Then the image of $X'$ by the closed immersion $j:X'\to\mathrm{P}$ is the sub-prescheme $Z$; at all points of $X'-Y'$ ($Y'$ being the inverse image of $Y$ in $X'$) the immersion $j$ is transversally regular relatively to $X$; and finally, at all points $x'$ of $Y'$, $X'$ is regular, the immersion $j$ is regular and of codimension $\operatorname{rg}_{\sh{O}_{P,z}}(\sh{H}_z)$ where $z=j(x')$ (in other words the homomorphism $v:\sh{H}\to\sh{O}_{\mathrm{P}}$ is regular in $z$).
\end{enumerate}
\end{proposition}

\begin{proof}
All the questions being local on $X$, we can restrict ourselves to the case where $X=\operatorname{Spec}(\mathrm{A})$ is affine and $\sh{E}=\widetilde{\mathrm{E}}$, where $\mathrm{E}$ is an $\mathrm{A}$-module.
To verify~(i), we can in addition examine what happens in an affine open of $\mathrm{P}$ of the form $\mathrm{D}_+(t)$, where $t\in\mathrm{E}=\mathbf{S}_\mathrm{A}^1(\mathrm{E})$ (\sref[II]{II.2.3.14}).
If we set $S=\mathbf{S}_\mathrm{A}(\mathrm{E})$, we have $\Gamma(\mathrm{D}_+(t),\sh{O}_{\mathrm{P}})=\mathrm{S}_{(t)}$ and $p^*(\sh{E})|D_+(t)=(E\otimes_\mathrm{A}\mathrm{S}_{(t)})^\sim$.
If we look back at the definition of $\alpha_1^{\#}$ (\sref[II]{II.2.6.2.2}), we see that for every element $a=\sum_i x^{(i)}(x_1^{(i)}\cdots x_n^{(i)})/t^n$ of $\mathrm{E}\otimes_\mathrm{A}\mathrm{S}_{(t)}$ (where the $x^{(i)}$ and $x_k^{(i)}$ are in $\mathrm{E}$), $\alpha_1^{\#}$ makes it correspond to the element $\sum_i(x^{(i)}x_1^{(i)}\cdots x_n^{(i)})/t^{n+1}$ of $(\mathrm{S}(1))_{(t)}$.
It is then a matter (to prove~(i)) of showing that if this last element is zero, i.e., $a'=v(a)=\sum u(x^{(i)})u(x_1^{(i)})\cdots u(x_n^{(i)})/(u(t))^n$ by the canonical homomorphism $\mathrm{S}_{(t)}\to(\mathbf{S}_\mathrm{A}(\mathrm{E}))_{(t)}\to(\mathbf{S}_\mathrm{A}(\mathrm{A}))_{(t)}$.
Now, this image is nothing other than the product of the element $u(t)$ and the canonical extension to $\mathrm{S}_{(t)}$ of the homomorphism of $\mathrm{A}$-modules $u:\mathrm{E}\to\mathrm{A}$.
This proves~(i).

To establish~(ii), let us note that we can suppose $\mathrm{E}$ is an $\mathrm{A}$-module free of rank $m$, and that $\alpha_1^{\#}:\mathrm{E}\otimes_\mathrm{A}\mathrm{S}_{(t)}\to(\mathrm{S}(1))_{(t)}$ is surjective and that $(\mathrm{S}(1))_{(t)}$ is an $\mathrm{S}_{(t)}$-module free of rank $1$, the exact sequence
\[
  0\to\mathrm{H}_{(t)}\to\mathrm{E}\otimes_\mathrm{A}\mathrm{S}_{(t)}\xrightarrow{\alpha_1^{\#}}(\mathrm{S}(1))_{(t)}\to 0
\]
is split, and $\mathrm{H}_{(t)}$ is therefore an $\mathrm{S}_{(t)}$-module projective of rank $m-1$, whence~(ii).

To prove~(iii), let us examine first what happens above a point $x\in X-Y$.
The hypothesis of ``regularity'' of $u$ at this point made in~(iii) amounts to (by virtue of \sref{IV.19.1.1} and of (\sref[0]{0.17.1.7})) saying that $X$ is regular at the point $x$ and that the homomorphism $u_X\otimes 1:\sh{E}_x\otimes_{\sh{O}_{X,x}}\mathbf{k}(x)\to\mathfrak{m}_x/\mathfrak{m}_x^2$ is \emph{injective}.
This being the case, since the morphism $p:\mathrm{P}\to X$ is smooth (\sref{IV.17.3.9}), for every $z\in\mathrm{P}$ above $x$, $\mathrm{P}$ is regular at the point $z$ (\sref{IV.17.5.8}); moreover (\sref[0]{0.17.3.3}) the canonical homomorphism
\[
  (\mathfrak{m}_z/\mathfrak{m}_z^2)\otimes_{\mathbf{k}(z)}\mathbf{k}(z)\to\mathfrak{m}_z/\mathfrak{m}_z^2
\]
\hyperref[IV.19.4.11.1]{(\textbf{19.4.11.1})}
\label{IV.19.4.11.1}
is injective.
We deduce that the homomorphism
\[
  (\sh{E}_z\otimes_{\sh{O}_{X,x}}\sh{O}_{\mathrm{P},z})\otimes_{\sh{O}_{\mathrm{P},z}}\mathbf{k}(z)\to\mathfrak{m}_z/\mathfrak{m}_z^2
\]
deduced from $p^*(u)$ is also injective, since it is written as the composite
\[
  (\sh{E}_z\otimes_{\sh{O}_{X,x}}\mathbf{k}(x))\otimes_{\mathbf{k}(z)}\mathbf{k}(z)\to(\mathfrak{m}_z/\mathfrak{m}_z^2)\otimes_{\mathbf{k}(z)}\mathbf{k}(z)\to\mathfrak{m}_z/\mathfrak{m}_z^2
\]
\oldpage[IV-4]{203}
where the first arrow is injective by flatness and the second is the injective homomorphism \sref{IV.19.4.11.1}.
Since $\sh{H}$ is (in a neighbourhood of $z$ in $\mathrm{P}$) a direct factor of $p^*(\sh{E})$, the homomorphism $v_z\otimes 1:\sh{H}_z\otimes_{\sh{O}_{\mathrm{P},z}}\mathbf{k}(z)\to\mathfrak{m}_z/\mathfrak{m}_z^2$ is also injective, which establishes the ``regularity'' of the homomorphism $v$ at the point $z$.
It remains to see that the image by $j$ of $X'$ is identical to the closed sub-prescheme $Z$.
It will suffice for this to show that the image by $j$ of the points of $q^{-1}(x)$ (where $q:X'\to X$ is the structural morphism) are exactly those of $Z\cap p^{-1}(x)$ and that on the other hand at each of these points $z=j(x')$, the surjective homomorphism $\sh{O}_{Z,z}\to\sh{O}_{X',x'}$ is bijective.
But, taking into account the hypothesis and of \sref{IV.19.4.3}, the restriction $Y'\to Y$ of $q$ is smooth, and since $q$ is supposed regular, $\sh{O}_{Y,x'}$ is therefore also regular (\sref{IV.19.1.1}).
To prove that the homomorphism $\sh{O}_{Z,z}\to\sh{O}_{X',x'}$ is bijective, it therefore suffices to show that the dimensions
\oldpage[IV-4]{204}
of these two regular rings are equal (\sref[0]{0.17.1.9}).
But by virtue of (\sref[0]{0.17.3.3}), this is also equivalent to the equality of dimensions of the two rings $\sh{O}_{Z,z}\otimes_{\sh{O}_{X,x}}\mathbf{k}(x)$ and $\sh{O}_{X',z}\otimes_{\sh{O}_{X,x}}\mathbf{k}(x)$, so that we can finally replace $X'$ and $Z$ by the fibres $q^{-1}(x)$ and $Z\cap p^{-1}(x)$ respectively, and we are reduced to proving that the morphism $q^{-1}(x)\to Z\cap p^{-1}(x)$ deduced from $j$ is an \emph{isomorphism}; this we can here restrict ourselves to the case where $X=\operatorname{Spec}(\mathrm{B})$, $\mathrm{B}=\sh{O}_{X,x}$ and $\mathrm{E}=\mathrm{B}^m$, whence $\mathrm{S}=\mathrm{B}[\mathrm{T}_1,\ldots,\mathrm{T}_{m-1}]$; moreover, the hypothesis implies that there exists a regular system of parameters $(t_i)_{1\leq i\leq n}$ of $\mathrm{B}$ such that $u(\mathrm{T}_i)=t_i$ for $1\leq i\leq m$ (\sref[0]{0.17.1.7}); let $\mathfrak{J}$ be the image of $u$; if $\mathfrak{m}$ is the maximal ideal of $\mathrm{B}$, we have $\mathfrak{J}^n\otimes_{\mathrm{B}}(\mathrm{B}/\mathfrak{m})=(\mathfrak{J}^n/\mathfrak{m}\mathfrak{J}^n)\otimes_\mathrm{B}(\mathrm{B}/\mathfrak{m})$ and by \sref{IV.16.9.4.1}, $(\bigoplus_{n\geq 0}\mathfrak{J}^n)\otimes_{\mathrm{B}}\mathbf{k}(x)$ is identified with $\mathbf{k}(x)[\mathrm{T}_1,\ldots,\mathrm{T}_m]$, whence it follows that the immersion $q^{-1}(x)\to p^{-1}(x)$ is an \emph{isomorphism}; this also implies \emph{a fortiori} that the closed sub-prescheme $Z\cap p^{-1}(x)$ is identical to the fibre $p^{-1}(x)$, and completes the proof of \sref{IV.19.4.11}.
\end{proof}

\subsection{Criteria for M-regularity}
\label{subsection:IV.19.5}

\oldpage[IV-4]{205}

The relation $\mathrm{H}_1(\mathbf{f},\mathrm{M})=0$ implies therefore $\mathrm{H}_0(f_n,\mathrm{H}_1(\mathbf{f}',\mathrm{M}))=0$ and $\mathrm{H}_1(f_n,\mathrm{H}_0(\mathbf{f}',\mathrm{M}))=0$.
The first of these two relations means that we have $f_n\mathrm{H}_1(\mathbf{f}',\mathrm{M})=\mathrm{H}_1(\mathbf{f}',\mathrm{M})$ \quad (\textbf{III},
$1.1.3.5$). Now, by definition (\textbf{III}, $1.1.1$), $\mathrm{H}_1(\mathbf{f}',\mathrm{M})$ is isomorphic to a quotient of a submodule $\mathrm{N}$ of $\mathrm{M}^{n-1}$; taking on $\mathrm{M}^{n-1}$ the filtration of the $\mathrm{M}^j$ for $j\leqslant n-1$, on $\mathrm{N}$ the induced filtration and on $\mathrm{H}_1(\mathbf{f}',\mathrm{M})$ the quotient filtration from that of $\mathrm{N}$, we obtain on $\mathrm{H}_1(\mathbf{f}',\mathrm{M})$ a finite filtration whose quotients are isomorphic to quotients of submodules of $\mathrm{M}$, hence are separated for the $\mathfrak{J}$-preadic topology by hypothesis.
As the relation $f_n\mathrm{H}_1(\mathbf{f}',\mathrm{M})=\mathrm{H}_1(\mathbf{f}',\mathrm{M})$ implies \emph{a fortiori} $\mathfrak{J}\mathrm{H}_1(\mathbf{f}',\mathrm{M})=\mathrm{H}_1(\mathbf{f}',\mathrm{M})$,
we deduce from the hypothesis and the preceding remarks that $\mathrm{H}_1(\mathbf{f}',\mathrm{M})=0$. The induction hypothesis proves then that the sequence $\mathbf{f}'$ is M-regular. On the other hand, the relation
$\mathrm{H}_1(f_n,\mathrm{H}_0(\mathbf{f}',\mathrm{M}))=0$ means that $f_n$ is $(\mathrm{M}/(\sum_{i=1}^{n-1}f_i\mathrm{M}))$-regular, hence the sequence $\mathbf{f}$ is M-regular.

\begin{corollary}[19.5.2]
\label{IV.19.5.2}
Suppose that $\mathrm{A}$ is a Noetherian ring, that the $f_i$ $(1\leqslant i\leqslant n)$ belong to the radical of $\mathrm{A}$, and that $\mathrm{M}$ is a finitely generated $\mathrm{A}$-module. Then the four conditions \emph{a)}, \emph{b')}, \emph{c)} of \emph{(\sref{IV.19.5.1})} are equivalent.
\end{corollary}

\begin{proof}
We know indeed from $(\mathbf{0}_\mathrm{I},7.3.5)$ that every finitely generated $\mathrm{A}$-module is separated for the $\mathfrak{J}$-preadic topology.
\end{proof}

\begin{proposition}[19.5.3]
\label{IV.19.5.3}
Let
\[
  0\to\mathrm{M}'\to\mathrm{M}\to\mathrm{M}''\to 0
\]
be an exact sequence of $\mathrm{A}$-modules, $\mathbf{f}=(f_i)_{1\leqslant i\leqslant n}$ an M-regular sequence, $\mathfrak{J}=\sum_{i=1}^{n}f_i\mathrm{A}$. Consider the following properties:
\begin{enumerate}
\item[\rm a)] The sequence $\mathbf{f}$ is $\mathrm{M}''$-regular.
\item[\rm b)] The $\mathfrak{J}$-preadic filtration of $\mathrm{M}'$ is induced on $\mathrm{M}'$ by the $\mathfrak{J}$-preadic filtration of $\mathrm{M}$ (in other words, the canonical homomorphism $\operatorname{gr}_\mathfrak{J}^*(\mathrm{M}')\to\operatorname{gr}_\mathfrak{J}^*(\mathrm{M})$ is injective).
\item[\rm c)] The canonical homomorphism $\mathrm{M}'/(\sum_{i=1}^{n}f_i\mathrm{M}')\to\mathrm{M}/(\sum_{i=1}^{n}f_i\mathrm{M})$ is injective.
\end{enumerate}
Then we have the implications \emph{a)} $\Rightarrow$ \emph{b)} $\Rightarrow$ \emph{c)}, and \emph{a)} implies moreover that the sequence $\mathbf{f}$ is $\mathrm{M}'$-regular.
If every quotient of a submodule of $\mathrm{M}''$ is separated for the $\mathfrak{J}$-preadic topology, the conditions \emph{a)}, \emph{b)} and \emph{c)} are equivalent.
\end{proposition}

\begin{proof}
It is clear that $c)$ is a consequence of $b)$. Let us prove that under all the hypotheses of the last assertion, $c)$ implies $a)$: since $\mathbf{f}$ is M-regular, by \sref{IV.19.5.1}, $\mathrm{H}_1(\mathbf{f},\mathrm{M})=0$, whence an exact sequence, portion of the exact homology sequence
\[
  0\to\mathrm{H}_1(\mathbf{f},\mathrm{M}'')\to\mathrm{H}_0(\mathbf{f},\mathrm{M}')\to\mathrm{H}_0(\mathbf{f},\mathrm{M}).
\]
Now, condition $c)$ expresses that the homomorphism $\mathrm{H}_0(\mathbf{f},\mathrm{M}')\to\mathrm{H}_0(\mathbf{f},\mathrm{M})$ is injective (\textbf{III}, $1.1.3.5$); it is therefore equivalent to $\mathrm{H}_1(\mathbf{f},\mathrm{M}'')=0$, whence, by virtue of the separation hypothesis and \sref{IV.19.5.1}, the fact that $\mathbf{f}$ is $\mathrm{M}''$-regular.

Let us show then that $a)$ implies $b)$, and also that the sequence $\mathbf{f}$ is $\mathrm{M}'$-regular, proceeding by induction on $n$. For $n=1$, $f_1$ being M-regular is also $\mathrm{M}'$-regular and the property $c)$ is none other than the lemma $(3.4.1.4)$. For $n>1$, the induction hypothesis shows that
\oldpage[IV-4]{206}
if we set $\mathbf{f}'=(f_1,\ldots,f_{n-1})$, the sequence $\mathbf{f}'$ is $\mathrm{M}'$-regular and, by virtue of $c)$, we have an exact sequence
\[
  0\to\mathrm{M}'/(\textstyle\sum_{i=1}^{n-1}f_i\mathrm{M}')\to\mathrm{M}/(\textstyle\sum_{i=1}^{n-1}f_i\mathrm{M})\to\mathrm{M}''/(\textstyle\sum_{i=1}^{n-1}f_i\mathrm{M}'')\to 0.
\]
By hypothesis, $f_n$ is $(\mathrm{M}/(\sum_{i=1}^{n-1}f_i\mathrm{M}))$-regular and $(\mathrm{M}''/(\sum_{i=1}^{n-1}f_i\mathrm{M}''))$-regular, hence by the same reasoning, by virtue of $(3.4.1.4)$, $f_n$ is also $(\mathrm{M}'/(\sum_{i=1}^{n-1}f_i\mathrm{M}'))$-regular, and on the other hand, the sequence
\[
  0\to\mathrm{M}'/(\textstyle\sum_{i=1}^{n}f_i\mathrm{M}')\to\mathrm{M}/(\textstyle\sum_{i=1}^{n}f_i\mathrm{M})\to\mathrm{M}''/(\textstyle\sum_{i=1}^{n}f_i\mathrm{M}'')\to 0
\]
is exact, whence $c)$.

Let us show finally that $a)$ implies $b)$. As the sequence $\mathbf{f}$ is then M-regular, M-quasi-regular, and $\mathrm{M}'$-quasi-regular, we have canonical isomorphisms
\[
  \operatorname{gr}_\mathfrak{J}^*(\mathrm{M}')\xrightarrow{\sim}\operatorname{gr}_\mathfrak{J}^0(\mathrm{M}')[\mathrm{T}_1,\ldots,\mathrm{T}_n],\qquad
  \operatorname{gr}_\mathfrak{J}^*(\mathrm{M})\xrightarrow{\sim}\operatorname{gr}_\mathfrak{J}^0(\mathrm{M})[\mathrm{T}_1,\ldots,\mathrm{T}_n]
\]
$(\mathbf{0},15.1.7)$, and as we saw above that $\operatorname{gr}_\mathfrak{J}^0(\mathrm{M}')\to\operatorname{gr}_\mathfrak{J}^0(\mathrm{M})$ is injective, the same holds for $\operatorname{gr}_\mathfrak{J}^*(\mathrm{M}')\to\operatorname{gr}_\mathfrak{J}^*(\mathrm{M})$.

We note moreover that the equivalence of conditions $a)$, $b)$, $c)$ of \sref{IV.19.5.3} holds in particular when $\mathrm{A}$ is Noetherian, the $f_i$ belong to the radical of $\mathrm{A}$, and the $\operatorname{gr}_k(\mathrm{M})$ are finitely generated $\mathrm{A}$-modules, \emph{then all conditions} \emph{a)} to \emph{h)} \emph{are equivalent}.
\end{proof}

In the rest of this number, we preserve the notations $\mathrm{A}$, $\mathbf{f}$, $\mathrm{M}$, $\mathfrak{J}$, and we suppose moreover $\mathrm{M}$ endowed with a decreasing filtration $(\mathrm{M}_k)$ formed of sub-$\mathrm{A}$-modules, such that $\mathrm{M}_0=\mathrm{M}$. Recall $(\mathbf{0},15.1.5)$ that we then define on $\mathrm{M}$ a second decreasing filtration formed of submodules

\]
\end{env}

\begin{theorem}[19.5.5]
\label{IV.19.5.5}
With the notations of \emph{(\sref{IV.19.5.4})}, consider the following properties:
\begin{enumerate}
\item[\rm a)] The sequence $\mathbf{f}$ is $\operatorname{gr}_\bullet(\mathrm{M})$-regular.
\item[\rm b)] The canonical homomorphism $\psi_\mathbf{M}$ is bijective.
\item[\rm c)] The canonical homomorphism $\varphi_{\operatorname{gr}_\bullet(\mathrm{M})}$ is bijective (in other words $(\mathbf{0},15.1.7)$, the sequence $\mathbf{f}$ is $\operatorname{gr}_\bullet(\mathrm{M})$-quasi-regular).
\end{enumerate}
\oldpage[IV-4]{207}
Then we have the implications\footnote{The implication $b)\Rightarrow c)$, without the separation hypothesis on the $\operatorname{gr}_k(\mathrm{M})$, is due to P.~Deligne, whose demonstration we reproduce below.}
\[
  \text{a)}\Rightarrow\text{b)}\Rightarrow\text{c)}.
\]
Moreover, if for every integer $k\geqslant 0$, the quotient modules of submodules of $\operatorname{gr}_k(\mathrm{M})$ are separated for the $\mathfrak{J}$-preadic topology (which is the case when $\mathrm{A}$ is Noetherian, the $f_i$ belong to the radical of $\mathrm{A}$, and the $\operatorname{gr}_k(\mathrm{M})$ are finitely generated $\mathrm{A}$-modules), \emph{then all conditions} \emph{a)}, \emph{b)} and \emph{c)} \emph{are equivalent}.

The implication $a)\Rightarrow b)$ has been demonstrated in $(\mathbf{0},15.1.8)$; the implication $c)\Rightarrow a)$ under the separation hypothesis is a particular case of $(\mathbf{0},15.1.9)$. It remains to prove that $b)$ implies $c)$, which will be done in several steps.

We denote by $\mathfrak{J}$ the ideal $\sum_{i=1}^{n}f_i\mathrm{A}$ generated by the $f_i$; for every sequence $\mathbf{q}=(q_1,\ldots,q_n)$ of $n$ integers $\geqslant 0$, we set $|\mathbf{q}|=\sum_{i=1}^{n}q_i$ and $\mathbf{f}^\mathbf{q}=f_1^{q_1}f_2^{q_2}\cdots f_n^{q_n}$.
\end{theorem}

\begin{env}[19.5.5.1]
\label{IV.19.5.5.1}
For the map $\psi_\mathbf{M}$ to be injective (and hence bijective), it is necessary and sufficient that the following condition be satisfied:

$(\mathrm{I}_{q,p})$ \emph{For every family $(x_\mathbf{q}^p)_{|\mathbf{q}|=q}$ of elements of $\mathrm{M}_p$, the relation}
\[
  \textstyle\sum_{|\mathbf{q}|=q}\mathbf{f}^\mathbf{q} x_\mathbf{q}^p\in\textstyle\sum_{i=1}^{q}\mathfrak{J}^{q-i}\mathrm{M}_{p+i}+\mathrm{M}_{p+q+1}'
\]
\emph{implies} $x_\mathbf{q}^p\in\mathrm{M}_{p+1}+\mathfrak{J}\mathrm{M}_p$ \emph{for every $\mathbf{q}$ such that $|\mathbf{q}|=q$.}

We proceed as in $(\mathbf{0},15.1.6)$ by considering the sub-$\mathrm{A}$-module $\mathrm{Q}_k$ (resp. $\mathrm{Q}_k'$) as a term of degree $k$ in the first (resp. second) member of $(\mathbf{0},15.1.5.2)$. We endow $\mathrm{Q}_k$ with the filtration
\[
  (\mathrm{Q}_k)_i=\textstyle\sum_{j\leqslant k-i}\bigl(\textstyle\sum_{|\mathbf{l}|=j}(\operatorname{gr}_{k-j}(\mathrm{M})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))\bigr)\mathrm{T}^\mathbf{l}
\]
and $\mathrm{Q}_k'$ the image filtration formed by $\psi_\mathbf{M}((\mathrm{Q}_k)_i)=(\mathrm{Q}_k')_i$; it suffices then to prove that the homomorphisms $(\psi_\mathbf{M})_{ki}:\operatorname{gr}_i(\mathrm{Q}_k)\to\operatorname{gr}_i(\mathrm{Q}_k')$ are injective. Now, we have
\[
  \operatorname{gr}_i(\mathrm{Q}_k)=\textstyle\sum_{|\mathbf{l}|=k-i}\bigl((\mathrm{M}_i/\mathrm{M}_{i+1})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J})\bigr)\mathrm{T}^\mathbf{l}
\]
\oldpage[IV-4]{208}
on the other hand, $(\mathrm{Q}_k')_{i+1}$ is the image of $\sum_{h=0}^{k-i-1}\mathfrak{J}^{k-i-h-1}\mathrm{M}_{i+h+1}$ in $\mathrm{M}_k'/\mathrm{M}_{k+1}'$. To write that $(\psi_\mathbf{M})_{ki}$ is injective is therefore equivalent to writing the condition $(\mathrm{I}_{k-i,i})$; whence our assertion.
\end{env}

\begin{env}[19.5.5.2]
\label{IV.19.5.5.2}
\emph{For the sequence $\mathbf{f}$ to be $\operatorname{gr}_\bullet(\mathrm{M})$-quasi-regular, it suffices that, for $p\geqslant 0$ and $q\geqslant 0$, the following condition be satisfied:}

$(\mathrm{S}_{q,p})$ \emph{For every family $(x_\mathbf{q}^p)_{|\mathbf{q}|=q}$ of elements of $\mathrm{M}_p$, the relation $\sum_{|\mathbf{q}|=q}\mathbf{f}^\mathbf{q} x_\mathbf{q}^p\in\mathrm{M}_{p+1}+\mathfrak{J}\mathrm{M}_p$ \emph{for every $\mathbf{q}$ such that $|\mathbf{q}|=q$.}}

By definition, saying that $\mathbf{f}$ is $\operatorname{gr}_\bullet(\mathrm{M})$-quasi-regular means that, for every $p\geqslant 0$ and every $q\geqslant 0$, the relation
\[
  \textstyle\sum_{|\mathbf{q}|=q}\mathbf{f}^\mathbf{q} x_\mathbf{q}^p\in\mathrm{M}_{p+1}+\mathfrak{J}^{q+1}\mathrm{M}_p
\]
for a family $(x_\mathbf{q}^p)_{|\mathbf{q}|=q}$ of elements of $\mathrm{M}_p$ implies $x_\mathbf{q}^p\in\mathrm{M}_{p+1}+\mathfrak{J}\mathrm{M}_p$ for every $\mathbf{q}$ such that $|\mathbf{q}|=q$. The condition $(\mathrm{S}_{q,p})$ means that there exists a family $(y_\mathbf{q}^p)_{|\mathbf{q}|=q}$ of elements of $\mathfrak{J}\mathrm{M}_p$ such that $\sum_{|\mathbf{q}|=q}\mathbf{f}^\mathbf{q}(x_\mathbf{q}^p-y_\mathbf{q}^p)\in\mathrm{M}_{p+1}$; it then implies $x_\mathbf{q}^p-y_\mathbf{q}^p\in\mathrm{M}_{p+1}+\mathfrak{J}\mathrm{M}_p$, whence $x_\mathbf{q}^p\in\mathrm{M}_{p+1}+\mathfrak{J}\mathrm{M}_p$; hence one sees that the conditions $(\mathrm{S}_{q,p})$ imply that the sequence $\mathbf{f}$ is $\operatorname{gr}_\bullet(\mathrm{M})$-quasi-regular.
\end{env}

It suffices therefore to prove the following proposition:

\begin{env}[19.5.5.3]
\label{IV.19.5.5.3}
\emph{For every $r\geqslant 0$, if the conditions $(\mathrm{I}_{q,p})$ are satisfied for $p+q<r$, then the conditions $(\mathrm{S}_{q,p})$ are satisfied for $p+q<r$.}

Let us introduce the conditions

$(\mathrm{A}_{q,p})$ \emph{For every family $(x_\mathbf{q}^p)_{|\mathbf{q}|=q}$ of elements of $\mathrm{M}_p$, the relation $\sum_{|\mathbf{q}|=q}\mathbf{f}^\mathbf{q} x_\mathbf{q}^p\in\mathrm{M}_{p+1}$ implies $x_\mathbf{q}^p\in\mathrm{M}_{p+1}+\mathfrak{J}\mathrm{M}_p$.}

It is clear that the conjunction of $(\mathrm{A}_{q,p})$ and of $(\mathrm{I}_{q,p})$ implies $(\mathrm{S}_{q,p})$. On the other hand, the conditions $(\mathrm{A}_{1,p})$ and $(\mathrm{S}_{0,p})$ are trivial. By \sref{IV.19.5.3}, the following proposition will result:
\end{env}

\begin{env}[19.5.5.4]
\label{IV.19.5.5.4}
\emph{For all $p\geqslant 0$ and $q\geqslant 1$, the condition $(\mathrm{A}_{q+1,p})$ is implied by the conjunction of the conditions $(\mathrm{A}_{q,p})$, $(\mathrm{I}_{q-1,p+1})$, $(\mathrm{I}_{q-2,p+2})$, $\ldots$, $(\mathrm{I}_{0,p+q})$.}

Indeed, once this proposition is proved, the conditions $(\mathrm{I}_{q,p})$ supposed satisfied for $p+q<r$ will imply, for each $p\leqslant r$, the condition $(\mathrm{A}_{q,p})$ for $1\leqslant q<r-p$, by induction on $q$.

Let us prove \sref{IV.19.5.5.4}. Let us note that the condition $(\mathrm{A}_{q,p})$ is also equivalent to
\end{env}

\begin{env}[19.5.5.5]
\label{IV.19.5.5.5}
\[
  \mathfrak{J}^q\mathrm{M}_p\cap\mathrm{M}_{p+1}\subset\textstyle\sum_{i=1}^{q}\mathfrak{J}^{q-i}\mathrm{M}_{p+i}.
\]

Let therefore $m\in\mathfrak{J}^{q+1}\mathrm{M}_p\cap\mathrm{M}_{p+1}\subset\mathfrak{J}^q\mathrm{M}_p\cap\mathrm{M}_{p+1}$. Applying condition $(\mathrm{A}_{q,p})$, we see that for each $i$ such that $1\leqslant i\leqslant q$, there exists a family $(y_\mathbf{q}^{p+i})_{|\mathbf{q}|=q-i}$ of elements of $\mathrm{M}_{p+i}$ such that
\[
  m=\textstyle\sum_{i=1}^{q}\bigl(\textstyle\sum_{|\mathbf{q}|=q-i}\mathbf{f}^\mathbf{q} y_\mathbf{q}^{p+i}\bigr).
\]
Suppose that for $1\leqslant i<j$ ($j\geqslant 1$), we have proved that
\[
  y_\mathbf{q}^{p+i}\in\mathfrak{J}\mathrm{M}_{p+i}+\mathrm{M}_{p+i+1}\qquad(\text{for }|\mathbf{q}|=q-i).
\]
We deduce by definition \sref{IV.19.5.4}
\[
  m-\textstyle\sum_{i<j}\bigl(\textstyle\sum_{|\mathbf{q}|=q-i}\mathbf{f}^\mathbf{q} y_\mathbf{q}^{p+i}\bigr)\in\mathrm{M}_{p+q+1}',
\]
Since by hypothesis $(\mathrm{I}_{q-j,p+j})$ is satisfied, we deduce
\[
  y_\mathbf{q}^{p+j}\in\mathfrak{J}\mathrm{M}_{p+j}+\mathrm{M}_{p+j+1}
\]
\oldpage[IV-4]{209}
and, by induction on $j$, one sees therefore that this condition is satisfied for \emph{every} $j$ such that $1\leqslant j\leqslant q$. We have therefore
\[
  m\in\textstyle\sum_{i=1}^{q}\mathfrak{J}^{q-i}(\mathfrak{J}\mathrm{M}_{p+i}+\mathrm{M}_{p+i+1})=\textstyle\sum_{i=1}^{q+1}\mathfrak{J}^{q-i+1}\mathrm{M}_{p+i}
\]
and we have thus proved $(\mathrm{A}_{q+1,p})$ and completed the demonstration of \sref{IV.19.5.5}.
\end{env}

\begin{remarks}[19.5.6]
\label{IV.19.5.6}
\medskip\noindent
\begin{enumerate}
\item[\rm(i)] The results of this number remain valid when, instead of an $\mathrm{A}$-module $\mathrm{M}$ and elements $f_i\in\mathrm{A}$, one takes an object $\mathrm{M}$ of any abelian category, and $n$ endomorphisms $f_i$ of $\mathrm{M}$ that mutually commute; the demonstrations adapt without difficulty to this more general case. The hypothesis of separation for an $\mathrm{A}$-module $\mathrm{N}$ relative to the $\mathfrak{J}$-preadic topology must here be replaced by the hypothesis that every sub-object of $\mathrm{N}$ contained in all the $\sum f_i^r(\mathrm{N})$ ($r$ an arbitrary integer) is necessarily zero. The same remark applies to the results of \sref{IV.19.7}.
\item[\rm(ii)] Suppose that $\mathrm{A}$ is a \emph{graded} ring with degrees bounded below. If $\mathrm{M}$ (resp. each $\operatorname{gr}_p(\mathrm{M})$) is a graded $\mathrm{A}$-module with degrees bounded below, and if the $f_i$ contain no homogeneous component of degree $0$, the separation hypothesis of $(\mathbf{0},15.1.9)$ is \emph{ipso facto} satisfied for $\mathrm{M}$ (resp. each $\operatorname{gr}_p(\mathrm{M})$), and one sees that in this case we have the implication $c)\Rightarrow a)$ in \sref{IV.19.5.1} (resp. \sref{IV.19.5.5}).
\item[\rm(iii)] Recall that, without the separation hypothesis, the implication $c)\Rightarrow b)$ is no longer valid even for $n=1$ $(\mathbf{0},15.1.12$, (iii)).
\end{enumerate}
\end{remarks}

\subsection{Regular sequences relative to a filtered quotient module}
\label{subsection:IV.19.6}

\begin{env}[19.6.1]
\label{IV.19.6.1}
Let $\mathrm{A}$ be a ring, $\mathbf{f}=(f_1,\ldots,f_n)$ a finite sequence of elements of $\mathrm{A}$, $\mathfrak{J}=f_1\mathrm{A}+\ldots+f_n\mathrm{A}$ the ideal it generates, and consider an exact sequence of $\mathrm{A}$-modules
\[
  0\to\mathrm{R}\xrightarrow{j}\mathrm{N}\xrightarrow{p}\mathrm{M}\to 0
\]
where one supposes that $\mathrm{N}$ is endowed with a decreasing filtration $(\mathrm{N}_k)$ of sub-$\mathrm{A}$-modules, with $\mathrm{N}_0=\mathrm{N}$. We set $\mathrm{R}_k=\mathrm{R}\cap\mathrm{N}_k$ (filtration induced by $(\mathrm{N}_k)$), $\mathrm{M}_k=p(\mathrm{N}_k)$ (quotient filtration of $(\mathrm{N}_k)$) and we denote by $\operatorname{gr}_\bullet(\mathrm{N})$, $\operatorname{gr}_\bullet(\mathrm{R})$ and $\operatorname{gr}_\bullet(\mathrm{M})$ the graded $\mathrm{A}$-modules associated with these three filtrations. We set on the other hand for all $k$,
\[
  \mathrm{N}_k'=\mathrm{N}_k+\mathfrak{J}\mathrm{N}_{k-1}+\ldots+\mathfrak{J}^{k-1}\mathrm{N}_1+\mathfrak{J}^k\mathrm{N}_0
\]
so that $\mathrm{M}_k'=p(\mathrm{N}_k')=\mathrm{M}_k+\mathfrak{J}\mathrm{M}_{k-1}+\ldots+\mathfrak{J}^{k-1}\mathrm{M}_1+\mathfrak{J}^k\mathrm{M}_0$; we set on the other hand $\mathrm{R}_k'=\mathrm{R}\cap\mathrm{N}_k'$ (filtration \emph{induced} by $(\mathrm{N}_k')$), and we denote by $\operatorname{gr}_\bullet'(\mathrm{N})$, $\operatorname{gr}_\bullet'(\mathrm{M})$ and $\operatorname{gr}_\bullet'(\mathrm{R})$ the graded $\mathrm{A}$-modules associated with these three filtrations; we thus have a commutative diagram of exact sequences
\[
\xymatrix{
  0 \ar[r] & \operatorname{gr}_\bullet(\mathrm{R}) \ar[r]^-{\operatorname{gr}(j)} \ar[d] & \operatorname{gr}_\bullet(\mathrm{N}) \ar[r]^-{\operatorname{gr}(p)} \ar[d] & \operatorname{gr}_\bullet(\mathrm{M}) \ar[r] \ar[d] & 0 \\
  0 \ar[r] & \operatorname{gr}_\bullet'(\mathrm{R}) \ar[r]^-{\operatorname{gr}'(j)} & \operatorname{gr}_\bullet'(\mathrm{N}) \ar[r]^-{\operatorname{gr}'(p)} & \operatorname{gr}_\bullet'(\mathrm{M}) \ar[r] & 0
}
\tag{19.6.1.1}
\]
\oldpage[IV-4]{210}
where the vertical arrows are the canonical homomorphisms from a graded module associated with a filtration to a graded module associated with a coarser filtration.

We note that if one sets
\[
  \mathrm{R}_k''=\mathrm{R}_k+\mathfrak{J}\mathrm{R}_{k-1}+\ldots+\mathfrak{J}^{k-1}\mathrm{R}_1+\mathfrak{J}^k\mathrm{R}_0
\]
we evidently have $\mathrm{R}_k''\subset\mathrm{R}_k'$, but the filtrations $(\mathrm{R}_k'')$ and $(\mathrm{R}_k')$ are in general distinct; we denote by $\operatorname{gr}_\bullet''(\mathrm{R})$ the graded $\mathrm{A}$-module associated with the filtration $(\mathrm{R}_k'')$.
\end{env}

\begin{env}[19.6.2]
\label{IV.19.6.2}
We defined in $(\mathbf{0},15.1.5.2)$ the surjective graded homomorphisms of degree $0$
\begin{align*}
  \psi_\mathrm{N} &: (\operatorname{gr}_\bullet(\mathrm{N})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \to \operatorname{gr}_\bullet'(\mathrm{N}) \\
  \psi_\mathrm{M} &: (\operatorname{gr}_\bullet(\mathrm{M})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \to \operatorname{gr}_\bullet'(\mathrm{M}) \\
  \psi_\mathrm{R} &: (\operatorname{gr}_\bullet(\mathrm{R})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \to \operatorname{gr}_\bullet''(\mathrm{R})
\end{align*}
the first two of which are deduced from the last two vertical arrows of \sref{IV.19.6.1.1}.

As the filtration $(\mathrm{R}_k'')$ is finer than $(\mathrm{R}_k')$, on $\mathrm{R}$ we have a canonical homomorphism $\gamma:\operatorname{gr}_\bullet''(\mathrm{R})\to\operatorname{gr}_\bullet'(\mathrm{R})$; we will denote by $\psi_\mathrm{R}'$ the composite homomorphism $\gamma\circ\psi_\mathrm{R}$. It then follows immediately from the definitions that we have a commutative diagram
\[
\xymatrix{
  & 0 \ar[d] \\
  (\operatorname{gr}_\bullet(\mathrm{R})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \ar[r]^-{\psi_\mathrm{R}'} \ar[d]^{j'} & \operatorname{gr}_\bullet'(\mathrm{R}) \ar[d]^{\operatorname{gr}'(j)} \\
  (\operatorname{gr}_\bullet(\mathrm{N})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \ar[r]^-{\psi_\mathrm{N}} \ar[d]^{p'} & \operatorname{gr}_\bullet'(\mathrm{N}) \ar[d]^{\operatorname{gr}'(p)} \\
  (\operatorname{gr}_\bullet(\mathrm{M})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \ar[r]^-{\psi_\mathrm{M}} \ar[d] & \operatorname{gr}_\bullet'(\mathrm{M}) \ar[d] \\
  0 & 0
}
\tag{19.6.2.1}
\]
where the columns are exact.
\end{env}

\begin{proposition}[19.6.3]
\label{IV.19.6.3}
With the preceding notations, suppose that the sequence $\mathbf{f}$ is $\operatorname{gr}_\bullet(\mathrm{N})$-regular. Consider the following properties:
\begin{enumerate}
\item[\rm a)] $\mathbf{f}$ is $\operatorname{gr}_\bullet(\mathrm{M})$-regular.
\item[\rm b)] $\psi_\mathrm{M}$ is bijective.
\item[\rm c)] $\psi_\mathrm{R}'$ is surjective (in other words $\operatorname{gr}_\bullet'(\mathrm{R})$ is generated as $\operatorname{gr}_\mathfrak{J}^*(\mathrm{A})$-module by $\operatorname{Im}(\operatorname{gr}_\bullet(\mathrm{R})\to\operatorname{gr}_\bullet'(\mathrm{R})))$.
\item[\rm d)] $\psi_\mathrm{R}'$ is injective.
\item[\rm d$'$)] The homomorphism $(\psi_\mathrm{R}')_0:\operatorname{gr}_\bullet(\mathrm{R})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J})\to\operatorname{gr}_\bullet'(\mathrm{R})$ obtained by restricting $\psi_\mathrm{R}'$ to polynomials of degree $0$, is injective.
\item[\rm e)] $\psi_\mathrm{R}'$ is bijective.
\item[\rm f)] $j'$ is injective.
\oldpage[IV-4]{211}
\item[\rm f$'$)] The homomorphism $\operatorname{gr}_\bullet(\mathrm{R})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J})\to\operatorname{gr}_\bullet(\mathrm{N})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J})$ obtained by restricting $j'$ to polynomials of degree $0$, is injective.
\item[\rm g)] The $\mathfrak{J}$-preadic filtration of $\operatorname{gr}_\bullet(\mathrm{R})$ is induced by that of $\operatorname{gr}_\bullet(\mathrm{N})$.
\item[\rm h)] The filtrations $(\mathrm{R}_k')$ and $(\mathrm{R}_k'')$ on $\mathrm{R}$ are identical.
\end{enumerate}
Then we have the following implications:
\[
\begin{array}{ccccccccc}
  & & \text{a)} & \Rightarrow & \text{e)} & \Rightarrow & \text{b)} & \Leftrightarrow & \text{c)} \Leftrightarrow \text{h)} \\
  & & & & & \searrow & & & \\
  & & & & & & \text{d)} & & \\
  & & \text{g)} & \Rightarrow & \text{d)} & \Leftrightarrow & \text{d$'$)} & \Leftrightarrow & \text{f)} \Leftrightarrow \text{f$'$)}
\end{array}
\]

Moreover, when every quotient module of a submodule of $\operatorname{gr}_k(\mathrm{M})$ is separated for the $\mathfrak{J}$-preadic topology (which is the case when $\mathrm{A}$ is Noetherian, the $f_i$ belong to the radical of $\mathrm{A}$, and the $\operatorname{gr}_k(\mathrm{M})$ are finitely generated $\mathrm{A}$-modules), \emph{then all conditions} \emph{a)} to \emph{h)} \emph{are equivalent}.
\end{proposition}

\begin{proof}
Let us note that, from \sref{IV.19.5.5}, the hypothesis that $\mathbf{f}$ is $\operatorname{gr}_\bullet(\mathrm{N})$-regular implies that $\psi_\mathrm{N}$ is \emph{bijective}; as $\operatorname{gr}'(j)$ is injective, the equivalence of $d)$ and $f)$ results from the diagram \sref{IV.19.6.2.1}; as moreover $f)$ and $f')$ are trivially equivalent, and also $d)$ and $d')$, by the five lemma this proves the equivalence of $d)$, $d')$, $f)$ and $f')$.

As one already knows that $\psi_\mathrm{M}$ is surjective and $\psi_\mathrm{N}$ bijective, the equivalence of $b)$ and $c)$ also follows from the diagram \sref{IV.19.6.2.1} by a particular case of the five lemma.

The relation $\psi_\mathrm{R}'=\gamma\circ\psi_\mathrm{R}$ \sref{IV.19.6.2} and the fact that $\psi_\mathrm{R}$ is surjective show that condition $c)$ is equivalent to saying that $\gamma$ is surjective, which is also equivalent to condition $h)$; hence we have proved the equivalence of $b)$, $c)$ and $h)$.

It is trivial that $e)$ is equivalent to the conjunction of $c)$ and $d)$, and hence also of $b)$ and $d)$. Let us note now that $a)$ implies $b)$ and that $b)$ implies $a)$ under the separation hypothesis \sref{IV.19.5.5}. On the other hand, the implications $a)\Rightarrow g)\Rightarrow f')$, and the implication $f')\Rightarrow a)$ under the separation hypothesis, result from \sref{IV.19.5.3}. Hence we have proved $a)$ implies $e)$ (equivalent to the conjunction of $b)$ and $f')$), and also that all conditions are equivalent under the separation hypothesis.
\end{proof}

\begin{corollary}[19.6.4]
\label{IV.19.6.4}
Let $\mathfrak{R}$ be an ideal of $\mathrm{A}$, $\mathfrak{L}=\mathfrak{J}+\mathfrak{R}$. Suppose that the filtration (hence also $(\mathrm{M}_k)$) is the $\mathfrak{R}$-preadic filtration, so that the filtrations $(\mathrm{M}_k')$ and $(\mathrm{N}_k')$ are $\mathfrak{L}$-preadic filtrations. Consider $\operatorname{gr}_\bullet(\mathrm{R})$ (resp. $\operatorname{gr}_\bullet'(\mathrm{R})$) as a graded module over $\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})$ (resp. $\operatorname{gr}_\mathfrak{L}^*(\mathrm{A})$).
\begin{enumerate}
\item[\rm(i)] Let $\mathrm{S}$ be a part of $\operatorname{gr}_\bullet(\mathrm{R})$ that generates $\operatorname{gr}_\bullet(\mathrm{R})$ as a $\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})$-module. Then condition \emph{c)} of \emph{(\sref{IV.19.6.3})} is equivalent to the following condition:
\item[\rm c$'$)] The image of $\mathrm{S}$ in $\operatorname{gr}_\bullet'(\mathrm{R})$ by $\psi_\mathrm{R}'$ generates $\operatorname{gr}_\bullet'(\mathrm{R})$ as a $\operatorname{gr}_\mathfrak{L}^*(\mathrm{A})$-module.
\item[\rm(ii)] Suppose condition \emph{e)} of \emph{(\sref{IV.19.6.3})} is satisfied, and also that the quotients of $\operatorname{gr}_k(\mathrm{R})$ are separated for the $\mathfrak{J}$-preadic topology (which is the case when $\mathrm{A}$ is Noetherian, the $f_i$ belong to the radical of $\mathrm{A}$, and the $\operatorname{gr}_k(\mathrm{M})$ are finitely generated $\mathrm{A}$-modules). Then, \emph{for a part $\mathrm{S}$ of $\operatorname{gr}_\bullet(\mathrm{R})$ formed of homogeneous elements to generate $\operatorname{gr}_\bullet(\mathrm{R})$ as a $\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})$-module, it is necessary and sufficient that its image by $\psi_\mathrm{R}'$ generates $\operatorname{gr}_\bullet'(\mathrm{R})$ as a $\operatorname{gr}_\mathfrak{L}^*(\mathrm{A})$-module.}
\end{enumerate}
\end{corollary}

\begin{proof}
\oldpage[IV-4]{212}
\begin{enumerate}
\item[\rm(i)] Consider the homomorphism of degree $0$
\[
  \psi_\mathrm{A}:(\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]\to\operatorname{gr}_\bullet'(\mathrm{A})=\operatorname{gr}_\mathfrak{L}^*(\mathrm{A})
\]
$(\mathbf{0},15.1.5.2)$. The fact that this homomorphism is \emph{surjective} implies that the sub-$\operatorname{gr}_\mathfrak{L}^*(\mathrm{A})$-algebra of $\operatorname{gr}_\bullet'(\mathrm{R})$ generated by $\mathrm{S}'=\psi_\mathrm{R}'(\mathrm{S})$ is none other than $\operatorname{Im}(\psi_\mathrm{R}')$, whence the equivalence of $c)$ and $c')$.

\item[\rm(ii)] Since condition $e)$ is satisfied, it follows by (i) that it remains to prove the sufficiency of the condition in the statement. Now, as $\psi_\mathrm{R}'$ is bijective, to say that $\mathrm{S}'$ generates $\operatorname{gr}_\bullet'(\mathrm{R})$ considered as a $\operatorname{gr}_\mathfrak{L}^*(\mathrm{A})$-module amounts to saying that $\mathrm{S}$ generates $(\operatorname{gr}_\bullet(\mathrm{R})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ as a $\operatorname{gr}_\mathfrak{L}^*(\mathrm{A})$-module, and by the surjectivity of $\psi_\mathrm{A}$, we can in this statement replace the ring $\operatorname{gr}_\mathfrak{L}^*(\mathrm{A})$ by $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]$. It amounts to saying that $\mathrm{S}$ generates $\operatorname{gr}_\bullet(\mathrm{R})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J})$ as a module over the ring $\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J})$, or also as a $\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})$-module. Now, if $\mathrm{T}$ is the sub-$\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})$-module graded of $\operatorname{gr}_\bullet(\mathrm{R})$ generated by $\mathrm{S}$, and if we set $\mathrm{T}'=\operatorname{gr}_\bullet(\mathrm{R})/\mathrm{T}$, on the one hand by hypothesis $\mathfrak{J}\mathrm{T}'=0$, or $\mathfrak{J}\mathrm{T}'=\mathrm{T}'$; but $\mathrm{T}'$ is a graded module, and each $\mathrm{T}_p'$ is a quotient of $\operatorname{gr}_p(\mathrm{R})$, hence separated for the $\mathfrak{J}$-preadic topology; the hypothesis therefore implies $\mathrm{T}'=0$, which completes the proof of the corollary.
\end{enumerate}
\end{proof}

\subsection{Hironaka's normal flatness criterion}
\label{subsection:IV.19.7}

\begin{theorem}[19.7.1]
\label{IV.19.7.1}
\textup{(Hironaka).}
Let $\mathrm{A}$ be a ring, $\mathfrak{R}$ an ideal of $\mathrm{A}$, $\mathbf{f}=(f_1,\ldots,f_n)$ a sequence of elements of $\mathrm{A}$ that is $(\mathrm{A}/\mathfrak{R})$-regular, $\mathfrak{J}=f_1\mathrm{A}+\ldots+f_n\mathrm{A}$ the ideal generated by $\mathbf{f}$, $\mathfrak{L}=\mathfrak{J}+\mathfrak{R}$, $\mathrm{M}$ an $\mathrm{A}$-module.
In this case $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J})=(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{L})=(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$, since $(\mathfrak{R}^n\mathrm{M}/\mathfrak{R}^{n+1}\mathrm{M})\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{J})=\mathfrak{R}^n\mathrm{M}/(\mathfrak{R}^{n+1}+\mathfrak{J}\mathfrak{R}^n)\mathrm{M}=\mathfrak{R}^n\mathrm{M}/(\mathfrak{R}^{n+1}+\mathfrak{L}\mathfrak{R}^n)\mathrm{M}$, and the last equality is a result of Bourbaki, \emph{Alg.}, chap.~II, $3^e$~ed., \textsection~3, n\textsuperscript{o}~7, cor.~2 of prop.~6.

Consider the following conditions:
\begin{enumerate}
\item[\rm a)] $\operatorname{gr}_\mathfrak{R}^*(\mathrm{M})$ is a flat $(\mathrm{A}/\mathfrak{R})$-module.
\item[\rm b)] $\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})$ is a flat $(\mathrm{A}/\mathfrak{L})$-module and the canonical homomorphism $(\mathbf{0},15.1.5.2)$
\[
  \psi_\mathrm{M}:((\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))\otimes_\mathrm{A}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]\to\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})
\]
is bijective.
\item[\rm c)] $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$ is a flat $(\mathrm{A}/\mathfrak{L})$-module and the sequence $\mathbf{f}$ is $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))$-regular.
\item[\rm d)] $\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})$ is a flat $(\mathrm{A}/\mathfrak{L})$-module, and for every $\mathfrak{p}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$, $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))_\mathfrak{p}$ is a flat $(\mathrm{A}/\mathfrak{R})_\mathfrak{p}$-module.
\end{enumerate}
Then we have the implications
\[
  \text{a)}\Rightarrow\text{c)}\Rightarrow\text{b)}
  \qquad\searrow\quad\text{d)}
\]

When every quotient of a submodule of $\operatorname{gr}_\mathfrak{R}^k(\mathrm{M})$ $(k\geqslant 0)$ is ideally separated for $\mathfrak{J}$ (Bourbaki, \emph{Alg.~comm.}, chap.~III, \textsection~5, n\textsuperscript{o}~1), the conditions \emph{a)}, \emph{b)}, \emph{c)} are equivalent.

\oldpage[IV-4]{213}
Finally, when $\mathrm{A}$ is Noetherian, the $f_i$ belong to the radical of $\mathrm{A}$, the sequence $\mathbf{f}$ is $\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})$-regular, and $\mathrm{M}$ is a finitely generated $\mathrm{A}$-module, the conditions \emph{a)}, \emph{b)}, \emph{c)}, \emph{d)} are equivalent.
\end{theorem}

\begin{proof}
The implication $a)\Rightarrow c)$ is immediate $(\mathbf{0},15.1.13)$. If $c)$ is satisfied, it results from \sref{IV.19.5.5} that $\psi_\mathrm{M}$ is bijective; moreover, as $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$, isomorphic to $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]$, is a flat $(\mathrm{A}/\mathfrak{L})$-module, hence so is it for $\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})$ since $\psi_\mathrm{M}$ is an $(\mathrm{A}/\mathfrak{L})$-isomorphism; hence $c)$ implies $b)$. The fact that $a)$ implies also $d)$ follows immediately.

When every quotient of a submodule of $\operatorname{gr}_\mathfrak{R}^k(\mathrm{M})$ is ideally separated for $\mathfrak{J}$, it results from \sref{IV.19.5.5} that condition $b)$ implies that $\mathbf{f}$ is $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M}))$-regular; moreover, as $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]$, isomorphic to $\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})$, is a flat $(\mathrm{A}/\mathfrak{L})$-module, it is a direct factor of this same $\operatorname{gr}_\mathfrak{R}^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$; this proves therefore that $b)$ implies $c)$. On the other hand, under the same hypothesis, $c)$ implies $a)$ by virtue of $(\mathbf{0},15.1.21)$ (where we can replace the Noetherian hypothesis by the hypothesis that $\mathrm{M}$ is ideally separated for $\mathfrak{J}$, by virtue of $(\mathbf{0}_\mathrm{III},10.2.1)$).

It remains therefore to prove that when $\mathrm{A}$ is Noetherian, $\mathrm{M}$ of finite type, and the $f_i$ belong to the radical of $\mathrm{A}$, $d)$ implies $a)$.

There exists a free $\mathrm{A}$-module $\mathrm{N}$ of \emph{finite type} and an exact sequence $0\to\mathrm{R}\to\mathrm{N}\to\mathrm{M}\to 0$. It suffices to prove that $d)$ implies $b)$, since every quotient of a sub-$\operatorname{gr}_\mathfrak{R}^k(\mathrm{M})$ is then ideally separated for $\mathfrak{J}$, by virtue of the fact that $\operatorname{gr}_\mathfrak{R}^k(\mathrm{M})$ is a finitely generated $\mathrm{A}$-module, that $\mathfrak{J}$ is contained in the radical of $\mathrm{A}$, and that $\mathrm{A}$ is Noetherian (Bourbaki, \emph{Alg.~comm.}, chap.~III, \textsection~5, n\textsuperscript{o}~1); in other words, it suffices to show that $\psi_\mathrm{M}$ is \emph{bijective}.

As by hypothesis the sequence $\mathbf{f}$ is also $\operatorname{gr}_\mathfrak{R}^*(\mathrm{N})$-regular, one can therefore apply \sref{IV.19.6.3}, since every quotient module of a sub-$\operatorname{gr}_k(\mathrm{M})$ is separated for the $\mathfrak{J}$-preadic topology since the $f_i$ belong to the radical of $\mathrm{A}$ by hypothesis. The diagram \sref{IV.19.6.2.1} here reads
\[
\xymatrix{
  & 0 \ar[d] \\
  (\operatorname{gr}_\mathfrak{R}^*(\mathrm{R},\mathrm{N})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \ar[r]^-{\psi_\mathrm{R}'} \ar[d] & \operatorname{gr}_\mathfrak{L}^*(\mathrm{R},\mathrm{N}) \ar[d] \\
  (\operatorname{gr}_\mathfrak{R}^*(\mathrm{N})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \ar[r]^-{\psi_\mathrm{N}} \ar[d] & \operatorname{gr}_\mathfrak{L}^*(\mathrm{N}) \ar[d] \\
  (\operatorname{gr}_\mathfrak{R}^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L}))[\mathrm{T}_1,\ldots,\mathrm{T}_n] \ar[r]^-{\psi_\mathrm{M}} \ar[d] & \operatorname{gr}_\mathfrak{L}^*(\mathrm{M}) \ar[d] \\
  0 & 0
}
\tag{19.7.1.1}

where $\operatorname{gr}_\mathfrak{R}^*(\mathrm{R},\mathrm{N})$ (resp. $\operatorname{gr}_\mathfrak{L}^*(\mathrm{R},\mathrm{N})$) is the graded associated to $\mathrm{R}$ for the filtration $\mathrm{R}\cap\mathfrak{R}^m\mathrm{N}$ (resp. $\mathrm{R}\cap\mathfrak{L}^m\mathrm{N}$); recall that in this diagram the columns are exact and that $\psi_\mathrm{N}$ is \emph{bijective}.
\oldpage[IV-4]{214}

\begin{env}[19.7.1.2]
\label{IV.19.7.1.2}
Set $\mathrm{B}=\operatorname{gr}_\mathfrak{R}^*\mathrm{A}\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$, $\mathrm{P}=\operatorname{gr}_\mathfrak{R}^*(\mathrm{N})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$, $\mathrm{Q}=\operatorname{gr}_\mathfrak{L}^*(\mathrm{R},\mathrm{N})$; since $\psi_\mathrm{N}$ is bijective, $\mathrm{P}[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ identifies with $\operatorname{gr}_\mathfrak{L}^*(\mathrm{N})$, and $\mathrm{Q}$ is a sub-$\mathrm{B}[\mathrm{T}_1,\ldots,\mathrm{T}_n]$-module of $\mathrm{P}[\mathrm{T}_1,\ldots,\mathrm{T}_n]$; we will first show that we have
\end{env}
\begin{env}[19.7.1.3]
\label{IV.19.7.1.3}
\[
  \mathrm{Q}=(\mathrm{Q}\cap\mathrm{P})[\mathrm{T}_1,\ldots,\mathrm{T}_n].
\]
\end{env}

For this, set $\mathrm{Z}=\mathrm{Q}/((\mathrm{Q}\cap\mathrm{P})[\mathrm{T}_1,\ldots,\mathrm{T}_n])$; this is a sub-$\mathrm{B}[\mathrm{T}_1,\ldots,\mathrm{T}_n]$-module of $\mathrm{P}[\mathrm{T}_1,\ldots,\mathrm{T}_n]/((\mathrm{Q}\cap\mathrm{P})[\mathrm{T}_1,\ldots,\mathrm{T}_n])=(\mathrm{P}/(\mathrm{Q}\cap\mathrm{P}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]$. As an $(\mathrm{A}/\mathfrak{L})$-module, it is therefore isomorphic to a submodule of a direct sum of $(\mathrm{A}/\mathfrak{L})$-modules isomorphic to $\mathrm{P}/(Q\cap\mathrm{P})=(P+Q)/Q$. But $(P+Q)/Q$ is a $(A/\mathfrak{L})$-module isomorphic to a submodule of $\mathrm{P}[\mathrm{T}_1,\ldots,\mathrm{T}_n]/\mathrm{Q}$, which is none other than $\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})$ by the diagram \sref{IV.19.7.1.1}.

But each $\operatorname{gr}_\mathfrak{L}^m(\mathrm{M})$ is by hypothesis a \emph{flat} $(\mathrm{A}/\mathfrak{L})$-module of finite type, hence projective, and we have the inclusion $\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\operatorname{gr}_\mathfrak{L}^*(\mathrm{M}))\subset\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{A}/\mathfrak{L})$. To see that $\mathrm{Z}=0$, it suffices by (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection~1, n\textsuperscript{o}~1, cor.~1 of prop.~2 and n\textsuperscript{o}~3, prop.~7) to see that for every $\mathfrak{q}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{A}/\mathfrak{L})$, $\mathrm{Z}_\mathfrak{q}=0$. But the ideals of $\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{A}/\mathfrak{L})$ are of the form $\mathfrak{p}/(\mathfrak{L}/\mathfrak{R})$, where $\mathfrak{p}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$, and it comes down to the same to see that $\mathrm{Z}_\mathfrak{p}=0$ for every $\mathfrak{p}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$. Now, if $\mathfrak{p}=\mathfrak{r}/\mathfrak{R}$, where $\mathfrak{r}$ is a prime ideal of $\mathrm{A}$, and $(\mathrm{A}/\mathrm{K})_\mathfrak{p}=\mathrm{A}_\mathfrak{r}/\mathrm{K}_\mathfrak{r}$, and the images of the $f_i$ in $\mathrm{A}_\mathfrak{r}$ form a $\operatorname{gr}_{\mathfrak{R}_\mathfrak{r}}^*(\mathrm{M}_\mathfrak{r})$-regular sequence $(\mathbf{0},15.1.14)$; applying to $\mathrm{A}_\mathfrak{r}$, $\mathfrak{R}_\mathfrak{r}$ and $\mathrm{M}_\mathfrak{r}$ the implication $a)\Rightarrow b)$ of the statement, we see from the hypothesis $d)$ that $\psi_{\mathrm{M}_\mathfrak{r}}$ is bijective, hence also $\psi_{\mathrm{R}_\mathfrak{r}}'$ by virtue of \sref{IV.19.6.3}; in other words, $\mathrm{Q}_\mathfrak{r}=\mathrm{Q}_\mathfrak{r}'[\mathrm{T}_1,\ldots,\mathrm{T}_n]$, where we have set $\mathrm{Q}'=\operatorname{gr}_\mathfrak{R}^*(\mathrm{R},\mathrm{N})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$, which is here a sub-$\mathrm{B}$-module of $\mathrm{P}$; in particular $\mathrm{Q}_\mathfrak{r}'\subset\mathrm{Q}_\mathfrak{r}\cap\mathrm{P}_\mathfrak{r}$; hence finally $\mathrm{Z}_\mathfrak{r}=\mathrm{Z}_\mathfrak{p}=0$, which completes the demonstration of \sref{IV.19.7.1.3}.

\begin{env}[19.7.1.5]
\label{IV.19.7.1.5}
By virtue of \sref{IV.19.7.1.3}, to prove that $\psi_\mathrm{R}'$ is surjective, it suffices to show that every homogeneous element of degree $m$ of $\mathrm{Q}\cap\mathrm{P}$ is the image of an element of the same degree of $\mathrm{P}$. Now, the elements of degree $m$ of $\mathrm{P}$ identify (by $\psi_\mathrm{N}$) with elements of $\operatorname{gr}_\mathfrak{L}^m(\mathrm{N})$; those which belong to elements of $\operatorname{gr}_\mathfrak{L}^m(\mathrm{R},\mathrm{N})$ identify with elements of $((\mathrm{R}\cap\mathfrak{R}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N})/\mathfrak{L}^{m+1}\mathrm{N}$; those of $\operatorname{gr}_\mathfrak{R}^m(\mathrm{R},\mathrm{N})$ identify with elements of $((\mathrm{R}\cap\mathfrak{R}^m\mathrm{N}+\mathfrak{L}^{m+1}\mathrm{N})/\mathfrak{L}^{m+1}\mathrm{N}$. It will therefore suffice to prove, for $m>0$, the inclusion
\[
  ((\mathrm{R}\cap\mathfrak{L}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N})\cap(\mathfrak{R}^m\mathrm{N}+\mathfrak{L}^{m+1}\mathrm{N})\subset(\mathrm{R}\cap\mathfrak{R}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N}.
\]
As $\mathfrak{R}\subset\mathfrak{L}$, we see immediately that the first member of this relation is equal to $(\mathrm{R}\cap(\mathfrak{R}^m\mathrm{N}+\mathfrak{L}^{m+1}\mathrm{N}))+\mathfrak{L}^{m+1}\mathrm{N}$, so that everything reduces to proving
\end{env}
\begin{env}[19.7.1.6]
\label{IV.19.7.1.6}
\[
  \mathrm{R}\cap(\mathfrak{R}^m\mathrm{N}+\mathfrak{L}^{m+1}\mathrm{N})\subset(\mathrm{R}\cap\mathfrak{R}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N}.
\]
\end{env}

\oldpage[IV-4]{215}
We will proceed by induction on $m$, assuming therefore \sref{IV.19.7.1.6} verified when $m$ is replaced by an integer $m'<m$. We will consider on the other hand, for $m$ fixed and for an integer $d\geqslant 0$, the relation
\begin{equation}
  \mathrm{R}\cap(\mathfrak{R}^m\mathrm{N}+\mathfrak{L}^{m+1}\mathrm{N})\subset(\mathrm{R}\cap\mathfrak{R}^d\mathrm{N}\cap\mathfrak{L}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N}.
  \tag{$*_d$}
\end{equation}

It is clear that it suffices to demonstrate $(*_d)$ for $d=m$, and on the other hand $(*_0)$ is trivially true. We will prove $(*_d)$ by \emph{induction on $d$}; in other words, we suppose, for a $d\geqslant 0$ fixed (with $d<m$), that $(*_d)$ is true, and we wish to prove
\begin{equation}
  \mathrm{R}\cap(\mathfrak{R}^m\mathrm{N}+\mathfrak{L}^{m+1}\mathrm{N})\subset(\mathrm{R}\cap\mathfrak{R}^{d+1}\mathrm{N}\cap\mathfrak{L}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N}.
  \tag{$*_{d+1}$}
\end{equation}

Consider for this, for an $h\geqslant 0$, the relation
\begin{equation}
  \mathrm{R}\cap(\mathfrak{R}^m\mathrm{N}+\mathfrak{R}^d\mathfrak{L}^h\mathrm{N})\subset(\mathrm{R}\cap(\mathfrak{R}^{d+1}\mathrm{N}+\mathfrak{R}^d\mathfrak{L}^h\mathrm{N})\cap\mathfrak{L}^m\mathrm{N})+\mathfrak{L}^{m+1}\mathrm{N}.
  \tag{$**_{d+1,h}$}
\end{equation}

The hypothesis $(*_d)$ implies that $(**_{d+1,0})$ is true. We will prove by \emph{induction on $h$} that $(**_{d+1,h})$ is true for every $h\geqslant 0$. But we have

\begin{lemma}[19.7.1.7]
\label{IV.19.7.1.7}
Let $\mathrm{A}$ be a Noetherian ring, $\mathrm{N}$ a finitely generated $\mathrm{A}$-module, $\mathfrak{L}$ an ideal of $\mathrm{A}$, $\mathrm{E}$, $\mathrm{F}$ two submodules of $\mathrm{N}$. For every $k>0$ there exists $h>0$ such that $\mathrm{E}\cap(\mathrm{F}+\mathfrak{L}^h\mathrm{N})\subset\mathrm{E}\cap(\mathrm{E}\cap\mathrm{F})+\mathfrak{L}^k\mathrm{N}$.
\end{lemma}

\begin{proof}
Indeed, let $\varphi:\mathrm{N}\to\mathrm{N}/(\mathrm{E}\cap\mathrm{F})=\mathrm{N}_1$ be the canonical homomorphism, and set $\mathrm{E}_1=\varphi(\mathrm{E})$, $\mathrm{F}_1=\varphi(\mathrm{F})$, so that $\mathrm{E}_1\cap\mathrm{F}_1=0$ and $\varphi(\mathrm{E}\cap(\mathrm{F}+\mathfrak{L}^h\mathrm{N}))=\mathrm{E}_1\cap(\mathrm{F}_1+\mathfrak{L}^h\mathrm{N}_1)$.
We know $(\mathbf{0}_\mathrm{I},7.3.2)$ that there exists $h$ such that $(\mathrm{E}_1+\mathrm{F}_1)\cap\mathfrak{L}^h\mathrm{N}_1\subset\mathfrak{L}^k(\mathrm{E}_1+\mathrm{F}_1)$; if $h$ is thus chosen and if $x_1\in\mathrm{E}_1$ is such that there exists $y_1\in\mathrm{F}_1$ for which $x_1-y_1\in\mathfrak{L}^h\mathrm{N}_1$, we deduce $x_1-y_1\in\mathfrak{L}^k\mathrm{N}_1$, hence $\mathrm{E}_1\cap(\mathrm{F}_1+\mathfrak{L}^h\mathrm{N}_1)\subset\mathfrak{L}^k\mathrm{N}_1$; therefore $\mathrm{E}\cap(\mathrm{F}+\mathfrak{L}^h\mathrm{N})\subset\mathrm{E}\cap\mathrm{F}+\mathfrak{L}^k\mathrm{N}$, which proves the lemma.
\end{proof}

It suffices then to apply this lemma with $\mathrm{E}=\mathrm{R}\cap\mathfrak{L}^m\mathrm{N}$, $\mathrm{F}=\mathfrak{R}^{d+1}\mathrm{N}$ and to take $k=m+1$ to deduce that $(**_{d+1,h})$ implies $(*_{d+1})$.

We suppose therefore in all that follows that $(**_{d+1,h})$ \emph{is satisfied.}

\begin{env}[19.7.1.8]
\label{IV.19.7.1.8}
\emph{Demonstration of $(**_{d+1,h+1})$; First case: $h\leqslant m-d$.}

We will first prove that, for every $h\geqslant 0$, we have
\end{env}
\begin{env}[19.7.1.9]
\label{IV.19.7.1.9}
\[
  \mathrm{R}\cap(\mathfrak{R}^{d+1}\mathrm{N}+\mathfrak{R}^d\mathfrak{L}^h\mathrm{N})\subset\mathfrak{L}^h(\mathrm{R}\cap\mathfrak{R}^d\mathrm{N})+\mathfrak{L}^{h+1}\mathfrak{R}^d\mathrm{N}+\mathfrak{R}^{d+1}\mathrm{N}.
\]
\end{env}

Consider for this an element $z$ of the first member of \sref{IV.19.7.1.9}. We note that $\mathfrak{R}^{d+1}\mathrm{N}+\mathfrak{R}^d\mathfrak{L}^h\mathrm{N}=\mathfrak{R}^{d+1}\mathrm{N}+\mathfrak{R}^d\mathfrak{J}^h\mathrm{N}$ since $\mathfrak{L}=\mathfrak{J}+\mathfrak{R}$; we will consider the class $\bar{z}$ of $z$ in $\operatorname{gr}_\mathfrak{J}^{h+d}(\operatorname{gr}_\mathfrak{R}^*(\mathrm{N}))$, which, by virtue of \sref{IV.19.5.5} and the hypothesis on $\mathbf{f}$, identifies with a sub-$(\mathrm{A}/\mathfrak{L})$-module of $\operatorname{gr}_\mathfrak{L}^{h+d}(\mathrm{N})$. Since $z\in\mathrm{R}$, we will show that the image of $\bar{z}$ in $\operatorname{gr}_\mathfrak{L}^*(\mathrm{N})$ is zero.
\oldpage[IV-4]{216}

\begin{env}[19.7.1.12]
\label{IV.19.7.1.12}
Indeed, in the demonstration of \sref{IV.19.7.1.3} that, for every prime ideal $\mathfrak{r}$ of $\mathrm{A}$ such that $\mathfrak{p}=\mathfrak{r}/\mathfrak{R}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$, the map $\psi_{\mathrm{R}_\mathfrak{r}}'$ is bijective, and therefore all the modules when localised at their primes, the image $z/1$ in $\mathrm{R}_\mathfrak{r}$ belongs to $(\mathrm{R}_\mathfrak{r}\cap\mathfrak{R}_\mathfrak{r}^d\mathrm{N}_\mathfrak{r})+\mathfrak{L}_\mathfrak{r}^{h+1}\mathrm{N}_\mathfrak{r}$, hence the image $\bar{z}/1$ of $\bar{z}$ in $\operatorname{gr}_{\mathfrak{L}_\mathfrak{r}}(\mathrm{R}_\mathfrak{r},\mathrm{N}_\mathfrak{r})=(\operatorname{gr}_\mathfrak{L}^*(\mathrm{R},\mathrm{N}))_\mathfrak{r}$, and hence $\bar{z}$ belongs to $\operatorname{gr}_\mathfrak{L}^{h+d}(\mathrm{R},\mathrm{N})$. In other words, the image of $\bar{z}$ by the canonical map $(\operatorname{gr}_\mathfrak{L}^*(\mathrm{N}))_\mathfrak{q}\to(\operatorname{gr}_\mathfrak{L}^*(\mathrm{M}))_\mathfrak{q}$ is zero for every $\mathfrak{q}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{A}/\mathfrak{L})$. \end{env}

\begin{env}[19.7.1.13]
\label{IV.19.7.1.13}
It suffices to prove that, in general, for $d<m$, we have
\[
  \mathfrak{L}^{m+1}\cap\mathfrak{R}^d\subset\mathfrak{L}^{m-d+1}\mathfrak{R}^d.

\end{env}

Indeed, as $\mathfrak{R}\subset\mathfrak{L}$, hence $\mathfrak{L}^{m-d+1}\mathfrak{R}^d\subset\mathfrak{L}^{m+1}$, the preceding relation implies
\[
  \mathfrak{L}^{m+1}\cap\mathfrak{R}^d\subset\mathfrak{L}^{m-d+1}\mathfrak{R}^d+\mathfrak{L}^{m+1}\cap\mathfrak{R}^{d+1}
\]
and by induction on $d<m$, one concludes
\[
  \mathfrak{L}^{m+1}\cap\mathfrak{R}^d\subset\mathfrak{L}^{m-d+1}\mathfrak{R}^d.

\begin{env}[19.7.1.14]
\label{IV.19.7.1.14}
Indeed, as $\mathfrak{R}\subset\mathfrak{L}$, hence $\mathfrak{L}^{m-d+1}\mathfrak{R}^d\subset\mathfrak{L}^{m+1}$, we have

  \mathfrak{L}^{m+1}\cap\mathfrak{R}^d\subset\mathfrak{L}^{m-d+1}\mathfrak{R}^d+\mathfrak{R}^{d+1}.

\end{env}

Since $\mathfrak{R}\subset\mathfrak{L}$, $\mathfrak{L}^{m-d+1}\mathfrak{R}^d\subset\mathfrak{L}^{m+1}$, the preceding relation implies
\[
  \mathfrak{L}^{m+1}\cap\mathfrak{R}^d\subset\mathfrak{L}^{m-d+1}\mathfrak{R}^d+\mathfrak{L}^{m+1}\cap\mathfrak{R}^{d+1}
\]
and by induction on $d<m$, one concludes that $\mathfrak{R}^{m+1}\subset\mathfrak{L}^{m-d+1}\mathfrak{R}^d$.

\begin{env}[19.7.1.15]
\label{IV.19.7.1.15}
As $\mathfrak{L}=\mathfrak{J}+\mathfrak{R}$, the relation \sref{IV.19.7.1.14} also reads
\[
  (\mathfrak{J}^{m+1}+\mathfrak{J}^m\mathfrak{R}+\ldots+\mathfrak{J}\mathfrak{R}^m+\mathfrak{R}^{m+1})\cap\mathfrak{R}^d\subset\mathfrak{J}^{m-d+1}\mathfrak{R}^d+\mathfrak{J}^{m-d}\mathfrak{R}^{d+1}+\ldots+\mathfrak{J}\mathfrak{R}^m+\mathfrak{R}^{m+1}+\mathfrak{R}^{d+1}=\mathfrak{J}^{m-d+1}\mathfrak{R}^d+\mathfrak{R}^{d+1}.

We will show that this inclusion is itself a consequence of
\end{env}
\begin{env}[19.7.1.16]
\label{IV.19.7.1.16}
\[
  \mathfrak{R}^a\mathfrak{J}^b\cap\mathfrak{R}^{a+1}\subset\mathfrak{R}^{a+1}\mathfrak{J}^{b-1}
\]
\end{env}
valid for $a\geqslant 0$, $b\geqslant 1$. Indeed, it suffices, to prove \sref{IV.19.7.1.15}, to show that for $0<q\leqslant d$,
\begin{env}[19.7.1.17]
\label{IV.19.7.1.17}
\[
  (\mathfrak{J}^{m+1}+\mathfrak{J}^m\mathfrak{R}+\ldots+\mathfrak{J}\mathfrak{R}^m+\mathfrak{R}^{m+1})\cap\mathfrak{R}^d\subset(\mathfrak{J}^{m+1-q}\mathfrak{R}^q+\mathfrak{J}^{m-q}\mathfrak{R}^{q+1}+\ldots+\mathfrak{R}^{m+1})\cap\mathfrak{R}^d

since for $q=d$, this will imply \sref{IV.19.7.1.15}. Now, to prove \sref{IV.19.7.1.17}, it suffices to proceed by induction on $q$ assuming it true for $q<d$; an element of the first member is written by hypothesis $y+z$ with $y\in\mathfrak{J}^{m+1-q}\mathfrak{R}^q$, $z\in\mathfrak{R}^d\subset\mathfrak{R}^{q+1}$; as $y+z\in\mathfrak{R}^d\subset\mathfrak{R}^{q+1}$, we deduce $y\in\mathfrak{R}^{q+1}$, since $z\in\mathfrak{R}^{q+1}$; hence

  y+z\in\mathfrak{J}^{m-q}\mathfrak{R}^{q+1}+\ldots+\mathfrak{R}^{m+1},\quad\text{and}\quad y+z\in\mathfrak{R}^d
\]
by hypothesis, which proves \sref{IV.19.7.1.17} where $q$ has been replaced by $q+1$.

It remains therefore to establish \sref{IV.19.7.1.16}. An element of the first member is written $a_1f_1+\ldots+a_nf_n$, where the $a_i$ belong to $\mathfrak{R}^a\mathfrak{J}^{b-1}$; taking into account that the sequence $\mathbf{f}$ is $(\mathfrak{R}^a/\mathfrak{R}^{a+1})$-regular, hence also $(\mathfrak{R}^a/\mathfrak{R}^{a+1})$-quasi-regular $(\mathbf{0},15.1.9)$, it follows from the definition $(\mathbf{0},15.1.7)$ that the relation $a_1f_1+\ldots+a_nf_n\in\mathfrak{R}^{a+1}$, where the $a_i\in\mathfrak{R}^a$, necessarily implies $a_i\in\mathfrak{R}^a\mathfrak{J}^{b-1}\cap\mathfrak{R}^{a+1}$; it suffices to argue by

Since $z\in\mathrm{R}$, we will show that the image of $\bar{z}$ is zero.

\begin{env}[19.7.1.20]

  \bar{z}\in\operatorname{gr}_\mathfrak{L}^{h+d}(\mathrm{R},\mathrm{N}).

Indeed, we have noted, in the demonstration of \sref{IV.19.7.1.3} that, for every prime ideal $\mathfrak{r}$ of $\mathrm{A}$ such that $\mathfrak{p}=\mathfrak{r}/\mathfrak{R}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{L})$, the map $\psi_{\mathrm{R}_\mathfrak{r}}'$ is bijective, and therefore all the modules, localised at $\mathfrak{r}$, the image $z/1$ in $\mathrm{R}_\mathfrak{r}$ belongs to $(\mathrm{R}_\mathfrak{r}\cap\mathfrak{R}_\mathfrak{r}^d\mathrm{N}_\mathfrak{r})+\mathfrak{L}_\mathfrak{r}^{h+1}\mathrm{N}_\mathfrak{r}$; hence $\operatorname{gr}_{\mathfrak{L}_\mathfrak{r}}(\mathrm{R}_\mathfrak{r},\mathrm{N}_\mathfrak{r})=(\operatorname{gr}_\mathfrak{L}^*(\mathrm{R},\mathrm{N}))_\mathfrak{r}$, and hence the image of $\bar{z}$ by the canonical map $(\operatorname{gr}_\mathfrak{L}^*(\mathrm{N}))_\mathfrak{q}\to(\operatorname{gr}_\mathfrak{L}^*(\mathrm{M}))_\mathfrak{q}$ is zero for every $\mathfrak{q}\in\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{A}/\mathfrak{L})$. As we saw in the demonstration of \sref{IV.19.7.1.3} that $\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\operatorname{gr}_\mathfrak{L}^*(\mathrm{M}))\subset\operatorname{Ass}_{\mathrm{A}/\mathfrak{L}}(\mathrm{A}/\mathfrak{L})$, we deduce (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection~1, n\textsuperscript{o}~1, cor.~1 of prop.~2 and n\textsuperscript{o}~3, prop.~7) that the image of $\bar{z}$ by the canonical map $\operatorname{gr}_\mathfrak{L}^*(\mathrm{N})\to\operatorname{gr}_\mathfrak{L}^*(\mathrm{M})$ is zero, that is to say we have the relation \sref{IV.19.7.1.19}.

This being so, we can write by definition

  z\equiv\textstyle\sum_{|\mathbf{p}|=h}c_\mathbf{p}\mathbf{f}^\mathbf{p}\bmod{\mathfrak{R}^{d+1}\mathrm{N}}

where one sets as usual $\mathbf{p}=(p_1,\ldots,p_n)$, $\mathbf{f}^\mathbf{p}=f_1^{p_1}f_2^{p_2}\cdots f_n^{p_n}$, $|\mathbf{p}|=p_1+\ldots+p_n$ and $c_\mathbf{p}\in\mathfrak{R}^d\mathrm{N}$.   \bar{z}=\textstyle\sum_{|\mathbf{p}|=h}\bar{c}_\mathbf{p}\mathrm{T}^\mathbf{p}.

Using \sref{IV.19.7.1.3}, we deduce that necessarily, for every $\mathbf{p}$ such that $|\mathbf{p}|=h$ ($\bar{c}_\mathbf{p}$ being this time identified by $\psi_\mathrm{N}$ with an element of $\operatorname{gr}_\mathfrak{L}^*(\mathrm{N})$),
\begin{env}[19.7.1.23]

  \bar{c}_\mathbf{p}\in\operatorname{gr}_\mathfrak{L}^d(\mathrm{R},\mathrm{N}).

But since $d<m$, the hypothesis that $(*_d)$ is true implies that $\bar{c}_\mathbf{p}$ belongs to the image of $\psi_\mathrm{R}'$, hence we can suppose that $c_\mathbf{p}\in(\mathrm{R}\cap\mathfrak{R}^d\mathrm{N})+\mathfrak{R}^d\mathfrak{L}\mathrm{N}$. The relation \sref{IV.19.7.1.21} then gives
\[
  z\in\mathfrak{J}^h(\mathrm{R}\cap\mathfrak{R}^d\mathrm{N})+\mathfrak{J}^h\mathfrak{R}^d\mathfrak{L}\mathrm{N}+\mathfrak{R}^{d+1}\mathrm{N}

and as $\mathfrak{J}\subset\mathfrak{L}$, this proves \sref{IV.19.7.1.19}.

In particular, we can therefore write
\[
  g=t+g_2
\]
with $t\in\mathfrak{L}^h(\mathrm{R}\cap\mathfrak{R}^d\mathrm{N})$ and $g_2\in\mathfrak{R}^{d+1}\mathrm{N}+\mathfrak{R}^d\mathfrak{L}^{h+1}\mathrm{N}$; moreover, as $t\in\mathrm{R}$ and $g\in\mathrm{R}$, we have $g_2\in\mathrm{R}$. On the other hand, using the hypothesis $h\geqslant m-d+1$, one sees that $t\in\mathfrak{L}^{m+1}\mathrm{N}$. The element $g_2$ hence satisfies all the conditions stated at the start of \sref{IV.19.7.1.8}, which completes the demonstration of \sref{IV.19.7.1}.
\end{proof}

\begin{remarks}[19.7.2]
\label{IV.19.7.2}
\medskip\noindent
\begin{enumerate}
\item[\rm(i)] When one supposes that $\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})$ is an $(\mathrm{A}/\mathfrak{R})$-module \emph{flat}, the hypothesis that the sequence $\mathbf{f}$ is $(\mathrm{A}/\mathfrak{R})$-regular implies that it is also $\operatorname{gr}_\mathfrak{R}^*(\mathrm{A})$-regular $(\mathbf{0},15.1.14)$. \item[\rm(ii)] It would be interesting to be able to prove the last conclusion of \sref{IV.19.7.1} under the sole hypothesis that there exists an $\mathrm{A}$-algebra $\mathrm{B}$ which is a Noetherian ring, for which $\mathfrak{J}\mathrm{B}$ is contained in the radical of $\mathrm{B}$, and that $\mathrm{M}$ is a finitely generated $\mathrm{B}$-module.
\end{enumerate}
\end{remarks}

\begin{corollary}[19.7.3]
\label{IV.19.7.3}
Let $\mathrm{A}$ be a local Noetherian ring, $\mathfrak{m}$ its maximal ideal, $\mathfrak{R}$ an ideal of $\mathrm{A}$ such that the ring $\mathrm{A}/\mathfrak{R}$ is regular and of dimension $n$, $\mathrm{M}$ a finitely generated $\mathrm{A}$-module. The following conditions are equivalent:
\begin{enumerate}
\item[\rm a)] $\operatorname{gr}_\mathfrak{R}^*(\mathrm{M})$ is a flat $(\mathrm{A}/\mathfrak{R})$-module.
\item[\rm b)] If $\mathrm{P}(\mathrm{T})$ and $\mathrm{Q}(\mathrm{T})$ are the Poincar\'e series of the $(\mathrm{A}/\mathfrak{m})$-modules graded $\operatorname{gr}_\mathfrak{m}^*(\mathrm{M})$ and $\operatorname{gr}_\mathfrak{R}^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{m})$, we have
\end{enumerate}
\begin{equation}
  \mathrm{Q}(\mathrm{T})=\mathrm{P}(\mathrm{T})(1-\mathrm{T})^n.
  \tag{19.7.3.1}
\end{equation}

Let $\mathbf{f}=(f_1,\ldots,f_n)$ be a sequence of elements of $\mathrm{A}$ such that the images of the $f_i$ in $\mathrm{A}/\mathfrak{R}$ form a regular system of parameters of $\mathrm{A}/\mathfrak{R}$ $(\mathbf{0},17.1.6)$, so that $\mathbf{f}$ is an $(\mathrm{A}/\mathfrak{R})$-regular sequence; moreover, if $\mathfrak{J}$ is the ideal generated by $\mathbf{f}$, $\mathfrak{J}+\mathfrak{R}=\mathfrak{m}$, and as $\mathrm{A}/\mathfrak{m}$ is a field, $\operatorname{gr}_\mathfrak{m}^*(\mathrm{M})$ is a flat $(\mathrm{A}/\mathfrak{m})$-module. As $\mathrm{A}$ is Noetherian and $\mathrm{M}$ a finitely generated $\mathrm{A}$-module, one can apply the equivalence of $a)$ and $b)$ in \sref{IV.19.7.1}. Moreover, $\mathrm{A}/\mathfrak{m}$ being a field, the fact that $\psi_\mathrm{M}$ is bijective is equivalent to saying that for every $h\geqslant 0$, $\operatorname{gr}_\mathfrak{m}^h(\mathrm{M})$ and the submodule of elements of degree $h$ of $(\operatorname{gr}_\mathfrak{R}^*(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{m}))[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ have the same rank over $\mathrm{A}/\mathfrak{m}$; but for the second of these modules the rank in question is clearly equal to
\[
  \textstyle\sum_{r=0}^{h}\binom{r+n-1}{n-1}\operatorname{rg}(\operatorname{gr}_\mathfrak{R}^{h-r}(\mathrm{M})\otimes_{\mathrm{A}/\mathfrak{R}}(\mathrm{A}/\mathfrak{m})).\quad\text{As}\quad(1-\mathrm{T})^{-n}=\textstyle\sum_{r=0}^{\infty}\binom{r+n-1}{n-1}\mathrm{T}^r,\quad\text{this}
\]
proves that the relation \sref{IV.19.7.3.1} is necessary and sufficient for $\psi_\mathrm{M}$ to be bijective.
\end{corollary}

\begin{corollary}[19.7.4]
\label{IV.19.7.4}
Let $\mathrm{X}$ be a locally Noetherian prescheme, $\mathrm{Y}$ a closed sub-prescheme of $\mathrm{X}$, $\mathrm{Y}$ regular and connected, $\sh{F}$ a coherent $\sh{O}_\mathrm{X}$-Module normally flat $(\sref{IV.6.10.1})$ along $\mathrm{Y}$. Then there exists a formal power series $\mathrm{R}\in\bb{Q}[[\mathrm{T}]]$ (independent of $x\in\mathrm{Y}$) such that, for every $x\in\mathrm{Y}$, the Poincar\'e series of $\operatorname{gr}_{\mathfrak{m}_x}^*(\sh{F}_x)$ is given by the formula
\begin{equation}
  \mathrm{P}_x(\sh{F})(\mathrm{T})=\mathrm{R}(\mathrm{T})(1-\mathrm{T})^{-n}\qquad\text{where }n=\dim(\sh{O}_{\mathrm{Y},x}).
  \tag{19.7.4.1}
\end{equation}
\oldpage[IV-4]{219}

Let $\sh{J}$ be the coherent Ideal of $\sh{O}_\mathrm{X}$ defining $\mathrm{Y}$; the hypothesis on $\sh{F}$ means that $\operatorname{gr}_\sh{J}^*(\sh{F})$ is an $\sh{O}_\mathrm{Y}$-Module flat. We can therefore, for every $x\in\mathrm{Y}$, apply \sref{IV.19.7.3} taking $\mathrm{A}=\sh{O}_{\mathrm{X},x}$, $\mathfrak{R}=\sh{J}_x$, $\mathrm{M}=\sh{F}_x$; the hypothesis $a)$ being satisfied by hypothesis, the Poincar\'e series $\mathrm{P}_x(\sh{F})(\mathrm{T})$ is therefore equal to $\mathrm{R}_x(\mathrm{T})(1-\mathrm{T})^{-n}$, where $\mathrm{R}_x(\mathrm{T})$ is the Poincar\'e series of $\operatorname{gr}_{\sh{J}_x}^*(\sh{F}_x)\otimes_{\sh{O}_{\mathrm{Y},x}}k(x)$. Now, each of the $\sh{O}_\mathrm{Y}$-Modules $\sh{J}^h\sh{F}/\sh{J}^{h+1}\sh{F}$ is flat and coherent by hypothesis, and since $\mathrm{Y}$ is supposed connected, hence each of $\sh{J}^h\sh{F}/\sh{J}^{h+1}\sh{F}$ is of constant rank, locally free $(2.1.12)$, and since $\mathrm{Y}$ is supposed connected, each of $\sh{J}^h\sh{F}/\sh{J}^{h+1}\sh{F}$ is of constant rank independent of $x\in\mathrm{Y}$, which proves that the Poincar\'e series $\mathrm{R}_x(\mathrm{T})$ is independent of $x\in\mathrm{Y}$.
\end{corollary}

\begin{corollary}[19.7.5]
\label{IV.19.7.5}
Let $\mathrm{X}$ be a locally Noetherian prescheme, $\mathrm{Y}$ a closed sub-prescheme of $\mathrm{X}$, $\mathrm{Z}$ a closed sub-prescheme of $\mathrm{Y}$; one supposes $\mathrm{Y}$ and $\mathrm{Z}$ regular. Let $\sh{F}$ be a coherent $\sh{O}_\mathrm{X}$-Module.
\begin{enumerate}
\item[\rm(i)] If $\sh{F}$ is normally flat along $\mathrm{Y}$ at the points of $\mathrm{Z}$ (cf. \sref{IV.11.3.4} for the definition of normal flatness at a point), then $\sh{F}$ is normally flat along $\mathrm{Z}$.
\item[\rm(ii)] Suppose moreover $\mathrm{Z}$ connected and that $\sh{O}_\mathrm{X}$ is normally flat along $\mathrm{Y}$ at the regular points of $\mathrm{Z}$. If $\sh{F}$ is normally flat along $\mathrm{Y}$ and if there exists a point $z\in\mathrm{Z}$ such that $\sh{F}$ is normally flat along $\mathrm{Y}$, then $\sh{F}$ is normally flat along $\mathrm{Y}$ at all the points of $\mathrm{Z}$.
\end{enumerate}

Let $\sh{X}_z$ and $\sh{L}_z$ be the coherent ideals of $\sh{O}_\mathrm{X}$ defining respectively $\mathrm{Y}$ and $\mathrm{Z}$.
\begin{enumerate}
\item[\rm(i)] The hypothesis means that $\operatorname{gr}_{\sh{X}_z}^*(\sh{F}_z)$ is an $\sh{O}_{\mathrm{Y},z}$-module flat; it implies therefore that $\operatorname{gr}_{\sh{X}_z}^*(\sh{F}_z)$ is an $\sh{O}_{\mathrm{Z},z}$-module flat, by virtue of \sref{IV.19.7.1}, $a)$ implies $b)$.
\item[\rm(ii)] The supplementary hypothesis on $\mathrm{X}$ implies that the sequence $(f_i)$ is regular at every point $z\in\mathrm{Z}$ \sref{IV.19.7.2}. On the other hand, $\mathrm{Z}$ being regular and connected, is integral, and if $\operatorname{gr}_{\sh{X}_z}^*(\sh{F}_z)$ is $\sh{O}_{\mathrm{Y},z}$-flat at a point $z\in\mathrm{Z}$, then, as the point generic $\zeta$ of $\mathrm{Z}$ is a generization of $z$, one concludes that $\operatorname{gr}_{\sh{X}_\zeta}^*(\sh{F}_\zeta)$ is $\sh{O}_{\mathrm{Y},\zeta}$-flat $(\mathbf{0}_\mathrm{I},6.3.1)$. We then deduce that if in every point $z'$ of $\mathrm{Z}$, $\operatorname{gr}_{\sh{X}_{z'}}^*(\sh{F}_{z'})$ is an $\sh{O}_{\mathrm{Z},z'}$-module flat, then $\operatorname{gr}_{\sh{X}_{z'}}^*(\sh{F}_{z'})$ is an $\sh{O}_{\mathrm{Y},z'}$-module flat, since we can apply the equivalence of $a)$ and $d)$ in \sref{IV.19.7.1}.
\end{enumerate}
\end{corollary}

\subsection{Properties of passage to the projective limit}
\label{subsection:IV.19.8}

\oldpage[IV-4]{219}
In this number, the notations and conventions on projective limits are those of $(8.5.1)$ and $(8.8.1)$.

\begin{proposition}[19.8.1]
\label{IV.19.8.1}
Suppose that the transition morphisms $\mathrm{S}_\mu\to\mathrm{S}_\lambda$ $(\lambda\leqslant\mu)$ are flat, and moreover that one of the two following hypotheses is satisfied:
\begin{enumerate}
\item[$1^\circ$] The preschemes $\mathrm{S}_\lambda$ are locally Noetherian.
\item[$2^\circ$] The transition morphisms $\mathrm{S}_\mu\to\mathrm{S}_\lambda$ are surjective (and hence faithfully flat).
\end{enumerate}
\oldpage[IV-4]{220}

\emph{Under these conditions:}
\begin{enumerate}
\item[\rm(i)] Suppose $\mathrm{S}_\alpha$ quasi-compact. Let $\sh{F}_\alpha$ be a quasi-coherent $\sh{O}_{\mathrm{S}_\alpha}$-Module, which one supposes moreover of finite type when hypothesis $1^\circ$ is satisfied. Let $(f_{i\alpha})_{1\leqslant i\leqslant n}$ be a finite sequence of sections of $\sh{F}_\alpha$ above $\mathrm{S}_\alpha$, the sequence of sections $f_i$ of the $\sh{O}_\mathrm{S}$-Module $\sh{F}$ above $\mathrm{S}$, corresponding to the $f_{i\alpha}$. For the sequence of sections $f_i$ $(1\leqslant i\leqslant n)$ to be $\sh{F}$-regular, it is necessary and sufficient that there exist $\lambda\geqslant\alpha$ such that the sequence of sections $f_{i\lambda}$ of $\sh{F}_\lambda$ above $\mathrm{S}_\lambda$ is $\sh{F}_\lambda$-regular.
\item[\rm(ii)] Suppose $\mathrm{X}_\alpha$ quasi-compact, and let $j_\alpha:\mathrm{Y}_\alpha\to\mathrm{X}_\alpha$ be an immersion, which one supposes locally of finite presentation when hypothesis $2^\circ$ is satisfied. Then, for the corresponding immersion $j:\mathrm{X}\to\mathrm{S}$ to be regular, it is necessary and sufficient that there exist $\lambda\geqslant\alpha$ such that $j_\lambda:\mathrm{X}_\lambda\to\mathrm{S}_\lambda$ is regular.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}
\item[\rm(i)] If $p_\lambda:\mathrm{S}\to\mathrm{S}_\lambda$ is the canonical projection, we have $\sh{F}/(\sum_{j=1}^{i-1}f_j\sh{F})=p_\lambda^*(\sh{F}_\lambda/(\sum_{j=1}^{i-1}f_{j\lambda}\sh{F}_\lambda))$, hence we are reduced to the case $n=1$, in which case on omits the index $i$. The fact that the condition is sufficient results from the fact that $p_\lambda$ is flat $(8.3.8)$ and from $(\mathbf{0},15.1.5)$. In case $2^\circ$, $p_\lambda$ is faithfully flat $(8.3.8)$ and the necessity of the condition results from $(\mathbf{0},15.2.5)$, with $\lambda=\alpha$. In case $1^\circ$, we denote by $\sh{N}_\lambda$ (resp. $\sh{N}$) the kernel of the homomorphism $\sh{F}_\lambda\to\sh{F}_\lambda$ (resp. $\sh{F}\to\sh{F}$), multiplication by $f_\lambda$ (resp. by $f$); since $\sh{F}_\lambda$ is coherent by hypothesis, the hypothesis $\sh{N}=0$ implies $\sh{N}_\lambda=0$ for $\lambda\geqslant\alpha$ by virtue of $(8.5.8$, (ii)).
\item[\rm(ii)] As $p_\lambda$ is flat, the sufficiency of the condition results from $(\sref{IV.19.1.5}$, (ii)). To prove its necessity, let us note that, since $j_\alpha(\mathrm{X}_\alpha)$ is quasi-compact and the transition morphisms $(\mathrm{X}_\mu)_{s_\mu}\to(\mathrm{X}_\lambda)_{s_\lambda}$ are flat (since the structural morphisms $\mathrm{X}_\lambda\to\mathrm{S}_\lambda$ are flat $(\sref{IV.11.2.6})$ and $(8.2.11)$), we can apply the argument $(19.8.1$, (i)) to the images of a finite generating set of sections of $\sh{J}_{\mathrm{Y}_\alpha}$ above each of a finite cover, to deduce that there exists $\lambda\geqslant\alpha$ such that $j_\lambda$ is regular. In case $1^\circ$ and in particular when hypothesis $2^\circ$ is satisfied, the image of $\mathrm{X}_\alpha$ in $\mathrm{S}_\alpha$ is locally Noetherian and the transition morphisms are flat; by the argument of $(19.2.4)$, there exists an open neighbourhood $\mathrm{V}_\lambda$ of the image $x_\lambda$ of $x$ in $\mathrm{X}_\lambda$ such that the restrictions to $\mathrm{V}_\lambda$ of $f_{i\lambda}$ form a $(\sh{F}_\lambda|\mathrm{U}_\lambda)$-transversally regular sequence relative to $\mathrm{S}_\lambda$; the conclusion then follows.
\end{enumerate}
\end{proof}

\begin{proposition}[19.8.2]
\label{IV.19.8.2}
Let $\mathrm{X}_\alpha$ be an $\mathrm{S}_\alpha$-prescheme quasi-compact and locally of finite presentation.
\begin{enumerate}
\item[\rm(i)] Let $\sh{F}_\alpha$ be a quasi-coherent $\sh{O}_{\mathrm{X}_\alpha}$-Module of finite presentation, $(f_{i\alpha})_{1\leqslant i\leqslant n}$ a finite sequence of sections of $\sh{O}_\mathrm{X}$ above $\mathrm{X}_\alpha$, corresponding to sections $f_i$ of $\sh{O}_\mathrm{X}$ above $\mathrm{X}$. For the sequence of sections $f_{i\alpha}$ to be $\sh{F}$-transversally regular relative to $\mathrm{S}$ \emph{(\sref{IV.19.2.1})}, it is necessary and sufficient that there exist $\lambda\geqslant\alpha$ such that the sequence of sections $f_{i\lambda}$ of $\sh{O}_{\mathrm{X}_\lambda}$ above $\mathrm{X}_\lambda$ is $\sh{F}_\lambda$-transversally regular relative to $\mathrm{S}_\lambda$.
\item[\rm(ii)] Let $\mathrm{Y}_\alpha$ be an $\mathrm{S}_\alpha$-prescheme quasi-compact, $j_\alpha:\mathrm{Y}_\alpha\to\mathrm{X}_\alpha$ an $\mathrm{S}_\alpha$-immersion locally of finite presentation. For the corresponding $\mathrm{S}$-immersion $j:\mathrm{Y}\to\mathrm{X}$ to be transversally regular relative to $\mathrm{S}$, it is necessary and sufficient that there exist $\lambda\geqslant\alpha$ such that $j_\lambda:\mathrm{Y}_\lambda\to\mathrm{X}_\lambda$ is transversally regular relative to $\mathrm{S}_\lambda$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}
\item[\rm(i)]
\oldpage[IV-4]{221}
The sufficiency of the condition results from $(\mathbf{0},15.1.15)$. To prove its necessity, let us note that, since $j_\alpha(\mathrm{X}_\alpha)$ is quasi-compact, we may limit ourselves to the case where $\mathrm{S}_\alpha$ is also quasi-compact, and $j_\alpha$ a closed immersion, so that the image of $\mathrm{X}_\alpha$ is defined by a quasi-coherent Ideal $\sh{J}_\alpha$ of $\sh{O}_{\mathrm{S}_\alpha}$ of finite type (in cases $1^\circ$ and $2^\circ$); the image of $\mathrm{X}_\lambda$ (resp. $\mathrm{X}$) is then defined by the Ideal $\sh{J}_\lambda=\sh{J}_\alpha\sh{O}_{\mathrm{S}_\lambda}$ (resp. $\sh{J}=\sh{J}_\alpha\sh{O}_\mathrm{S}$) which is of finite type. To see that for $\lambda\geqslant\alpha$ sufficiently large, the sections $f_{i\lambda}$ of $\sh{J}_\lambda$ above $\mathrm{X}_\lambda$ form an $(\sh{F}_\lambda|\mathrm{U}_\lambda)$-transversally regular sequence relative to $\mathrm{S}_\lambda$, it suffices to apply \sref{IV.19.8.1}, (i) to the restrictions of $f_{i\alpha}$ to the elements $\mathrm{U}_\alpha$ of a finite affine open cover of $\mathrm{X}_\alpha$, after having verified that the corresponding covers have the required properties. As $\mathrm{X}_\alpha$ is quasi-compact and of finite presentation over $\mathrm{S}_\alpha$, the transition morphisms $(\mathrm{X}_\mu)_{s_\mu}\to(\mathrm{X}_\lambda)_{s_\lambda}$ are flat (since the structural morphisms $\mathrm{X}_\lambda\to\mathrm{S}_\lambda$ and $\mathrm{S}_\mu\to\mathrm{S}_\lambda$ are flat $(\sref{IV.11.2.6})$ and $(8.2.11)$), and we may apply \sref{IV.19.8.1} to deduce that for $\lambda\geqslant\alpha$ sufficiently large, one can suppose $\sh{F}_\lambda$ is $\mathrm{S}_\lambda$-flat at the point $x_\lambda$. It results then from $(\sref{IV.11.3.8})$ that there exists an open neighbourhood $\mathrm{V}_\lambda$ of $x_\lambda$ in $\mathrm{X}_\lambda$ satisfying the required conditions.

Moreover, using \sref{IV.19.2.7}, (ii), the sufficiency of the condition results from $(\sref{IV.19.2.7}$, (ii)). To prove its necessity, it suffices to show that for every point $x\in j(\mathrm{Y})$ there exists an open neighbourhood $\mathrm{V}(x)$ of $x$ and an index $\lambda=\lambda(x)\geqslant\alpha$ and an open neighbourhood $\mathrm{V}_\lambda$ of the image $x_\lambda$ of $x$ in $\mathrm{X}_\lambda$, such that the sequence of restrictions of $f_{i\lambda}$ to $\mathrm{V}_\lambda$ be $(\sh{F}_\lambda|\mathrm{V}_\lambda)$-transversally regular relative to $\mathrm{S}_\lambda$: indeed, by the condition of being quasi-compact, one deduces from this the existence of an open neighbourhood $\mathrm{V}_\lambda$ of $x_\lambda$ satisfying the conditions.

Let $s$ be the image of $x$ in $\mathrm{S}$, and for every $\lambda\geqslant\alpha$, let $s_\lambda$ be the image of $s$ in $\mathrm{S}_\lambda$. As $\mathrm{X}_s$ (resp. $\mathrm{Y}_s$) is the projective limit of the fibres $(\mathrm{X}_\lambda)_{s_\lambda}$ (resp. $(\mathrm{Y}_\lambda)_{s_\lambda}$), and we can apply to these fibres the conditions of \sref{IV.19.8.1}, (ii) (since by hypothesis the immersion $\mathrm{Y}_s\to\mathrm{X}_s$ is regular), it follows that there exists an index $\lambda(s)\geqslant\alpha$ such that for $\lambda\geqslant\lambda(s)$ the immersion $(\mathrm{Y}_\lambda)_{s_\lambda}\to(\mathrm{X}_\lambda)_{s_\lambda}$ is regular. Moreover $(\sref{IV.11.2.6})$, one can suppose that $\sh{F}_\lambda$ is $\mathrm{S}_\lambda$-flat at $x_\lambda$. It results then from $(\sref{IV.11.3.8})$ that there exists an open neighbourhood $\mathrm{V}_\lambda$ of $x_\lambda$ in $\mathrm{X}_\lambda$ satisfying the required conditions, together with open sets $\mathrm{W}(x_\lambda)$ in $\mathrm{X}$ such that the structural morphisms $\mathrm{W}(x_\lambda)\to\mathrm{S}$ and $\mathrm{Y}\cap\mathrm{W}(x_\lambda)\to\mathrm{S}$ are flat; applying $(\sref{IV.11.2.6})$ and $(8.2.11)$, there exists an open neighbourhood $\mathrm{V}_\lambda$ of $x_\lambda$ in $\mathrm{X}_\lambda$ such that the restrictions to $\mathrm{V}_\lambda$ and to $\mathrm{Y}_\lambda\cap\mathrm{V}_\lambda$ of the structural morphisms $\mathrm{X}_\lambda\to\mathrm{S}_\lambda$ and $\mathrm{Y}_\lambda\to\mathrm{S}_\lambda$ are flat. We conclude therefore from \sref{IV.19.2.4} that there exists an open neighbourhood $\mathrm{V}_\lambda$ of $x_\lambda$ satisfying the required conditions.
\end{enumerate}
\end{proof}

\subsection{\texorpdfstring{$\sh{F}$}{F}-regular sequences and depth}
\label{subsection:IV.19.9}

\oldpage[IV-4]{222}

\begin{env}[19.9.1]
\label{IV.19.9.1}
Recall that if $\mathrm{X}$ is a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_\mathrm{X}$-Module, $\mathrm{T}$ a part of $\mathrm{X}$, one sets $(5.10.1)$
\begin{equation}
  \operatorname{prof}_\mathrm{T}(\sh{F})=\inf_{x\in\mathrm{T}}\operatorname{prof}(\sh{F}_x).
  \tag{19.9.1.1}
\end{equation}
We set on the other hand, for every point $t\in\mathrm{T}$,
\begin{equation}
  \operatorname{prof}_{\mathrm{T},t}(\sh{F})=\inf_{x\in\mathrm{T}\cap\operatorname{Spec}(\sh{O}_{\mathrm{X},t})}\operatorname{prof}(\sh{F}_x),
  \tag{19.9.1.2}
\end{equation}
and we say that $\operatorname{prof}_{\mathrm{T},t}(\sh{F})$ is the \emph{T-depth of $\sh{F}$ at the point $t$}; it is clear that we have
\begin{equation}
  \operatorname{prof}_\mathrm{T}(\sh{F})=\inf_{t\in\mathrm{T}}\operatorname{prof}_{\mathrm{T},t}(\sh{F}).
  \tag{19.9.1.3}
\end{equation}
\end{env}

\begin{lemma}[19.9.2]
\label{IV.19.9.2}
Let $\mathrm{A}$ be a Noetherian ring, $\mathrm{M}$ a finitely generated $\mathrm{A}$-module, $\mathfrak{J}$ an ideal of $\mathrm{A}$. In order that, for every prime ideal $\mathfrak{p}\supset\mathfrak{J}$, we have $\operatorname{prof}_{\mathrm{A}_\mathfrak{p}}(\mathrm{M}_\mathfrak{p})\geqslant r$, it is necessary and sufficient that there exist an M-regular sequence of $r$ elements belonging to $\mathfrak{J}$.
\end{lemma}

\begin{proof}
The sufficiency of the condition resulting immediately from $(\mathbf{0},16.4.5)$, let us prove its necessity. We argue by induction on $r$ (there is nothing to prove if $r=0$): since $\operatorname{prof}_{\mathrm{A}_\mathfrak{p}}(\mathrm{M}_\mathfrak{p})\geqslant r-1$ for every $\mathfrak{p}\supset\mathfrak{J}$, there exists by hypothesis an M-regular sequence $(g_i)_{1\leqslant i\leqslant r-1}$ of elements of $\mathfrak{J}$. Set $\mathrm{N}=\mathrm{M}/(g_1\mathrm{M}+\ldots+g_{r-1}\mathrm{M})$; for every $\mathfrak{p}\supset\mathfrak{J}$, the images of the $g_i$ in $\mathrm{A}_\mathfrak{p}$ belong to the maximal ideal $\mathfrak{p}\mathrm{A}_\mathfrak{p}$, and form an $\mathrm{M}_\mathfrak{p}$-regular sequence $(\mathbf{0},15.1.14)$; we conclude therefore from $(\mathbf{0},16.4.6)$ that $\operatorname{prof}_{\mathrm{A}_\mathfrak{p}}(\mathrm{N}_\mathfrak{p})=\operatorname{prof}_{\mathrm{A}_\mathfrak{p}}(\mathrm{M}_\mathfrak{p})-(r-1)\geqslant 1$, and one sees that this reduces to proving the lemma for $r=1$. Now the hypothesis means then that, for every $\mathfrak{p}\supset\mathfrak{J}$, $\mathfrak{p}\mathrm{A}_\mathfrak{p}$ is not associated to $\mathrm{M}$ $(\mathbf{0},16.4.6)$, hence (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection~1, n\textsuperscript{o}~2, prop.~5), $\mathfrak{p}$ is not contained in any of the prime ideals of $\operatorname{Ass}(\mathrm{M})$, hence it is not contained in their union (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection~1, n\textsuperscript{o}~1, prop.~2). But as this union is the set of elements that are not M-regular (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection~1, cor.~2 of prop.~2), this proves the lemma.
\end{proof}

\begin{proposition}[19.9.3]
\label{IV.19.9.3}
Let $\mathrm{X}$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_\mathrm{X}$-Module, $\mathrm{Y}$ a closed subset of $\mathrm{X}$, $y$ a point of $\mathrm{Y}$, $n$ an integer $>0$. The following conditions are equivalent:
\begin{enumerate}
\item[\rm a)] There exists an open neighbourhood $\mathrm{U}$ of $y$ in $\mathrm{X}$ such that $\operatorname{prof}_{\mathrm{Y}\cap\mathrm{U}}(\sh{F}|\mathrm{U})\geqslant n$.
\item[\rm b)] There exists an open neighbourhood $\mathrm{U}$ of $y$ in $\mathrm{X}$ and a suite $(\sh{F}|\mathrm{U})$-regular sequence of $n$ sections $f_i$ of $\sh{O}_\mathrm{U}$ above $\mathrm{U}$, such that $f_i(x)=0$ $(\mathbf{0}_\mathrm{I},5.5.1)$ for $1\leqslant i\leqslant n$ at every point $x\in\mathrm{U}\cap\mathrm{Y}$.
\item[\rm c)] $\operatorname{prof}_{\mathrm{Y},y}(\sh{F})\geqslant n$.
\end{enumerate}

Moreover, the set of $y\in\mathrm{Y}$ satisfying these conditions is open in $\mathrm{Y}$.
\end{proposition}

\begin{proof}
The last assertion results trivially from $a)$. The equivalence of $a)$ and $b)$ follows from \sref{IV.19.9.2} applied to an affine open neighbourhood of $y$ in $\mathrm{X}$ and from $(\mathbf{0},15.1.14)$. It is clear that $a)$ implies $c)$, since every open neighbourhood of $y$ in $\mathrm{X}$ containing $\operatorname{Spec}(\sh{O}_{\mathrm{X},y})$; conversely that $c)$ implies $b)$. It results from $c)$ and the definition \sref{IV.19.9.1} that

we can apply \sref{IV.19.9.2} to the $\sh{O}_{\mathrm{X},y}$-module $\sh{F}_y$ and the ideal $\sh{J}_y$ of $\sh{O}_{\mathrm{X},y}$, hence there exists a $\sh{F}_y$-regular sequence $(t_i)_{1\leqslant i\leqslant n}$ of $n$ elements of $\sh{J}_y$. \begin{corollary}[19.9.4]
\label{IV.19.9.4}
With the notations of \emph{(\sref{IV.19.9.3})}, the function $y\mapsto\operatorname{prof}_{\mathrm{Y},y}(\sh{F})$ is lower semi-continuous on $\mathrm{Y}$.
\end{corollary}

\begin{proposition}[19.9.5]
\label{IV.19.9.5}
With the notations of \emph{(\sref{IV.19.9.3})}, let $\mathrm{X}'$ be a locally Noetherian prescheme, $g:\mathrm{X}'\to\mathrm{X}$ a flat morphism, $\mathrm{Y}'=g^{-1}(\mathrm{Y})$, $\sh{F}'=g^*(\sh{F})$. Then, for every point $y'\in\mathrm{Y}'$ above $y$, we have $\operatorname{prof}_{\mathrm{Y}',y'}(\sh{F}')=\operatorname{prof}_{\mathrm{Y},y}(\sh{F})$.
\end{proposition}

\begin{proof}
Indeed, for every $z'\in\operatorname{Spec}(\sh{O}_{\mathrm{X}',y'})$, it results from $(6.3.1)$ that $\operatorname{prof}(\sh{F}_{z'})\geqslant\operatorname{prof}(\sh{F}_{g(z')})$, and if $z''$ is a maximal point of the fibre $g^{-1}(g(z'))$, generization of $z'$ (hence belonging also to $\operatorname{Spec}(\sh{O}_{\mathrm{X}',y'})$), we have (\emph{loc.~cit.}) $\operatorname{prof}(\sh{F}_{z''})=\operatorname{prof}(\sh{F}_{g(z')})$. The proposition then results from the fact that $g(\operatorname{Spec}(\sh{O}_{\mathrm{X}',y'}))=\operatorname{Spec}(\sh{O}_{\mathrm{X},g(y')})$ since $g$ is flat $(2.3.4)$.
\end{proof}

\begin{proposition}[19.9.6]
\label{IV.19.9.6}
Let $f:\mathrm{X}\to\mathrm{S}$ be a morphism locally of finite presentation, $\mathrm{Y}$ a closed subset of $\mathrm{X}$, $\sh{F}$ a coherent $\sh{O}_\mathrm{X}$-Module of finite presentation and $f$-flat, $n$ an integer $>0$. For every point $y\in\mathrm{Y}$, denoting by $\mathrm{X}_{f(y)}$ the fibre of $f$ at the point $f(y)$, by $\sh{F}_{f(y)}$ the inverse image of $\sh{F}$ in $\mathrm{X}_{f(y)}$, and setting $\mathrm{Y}_{f(y)}=\mathrm{X}_{f(y)}\cap\mathrm{Y}$:
\begin{enumerate}
\item[\rm a)] $\operatorname{prof}_{f(y),y}(\sh{F}_{f(y)})\geqslant n$.
\item[\rm b)] There exists an open neighbourhood $\mathrm{U}$ of $y$ in $\mathrm{X}$ and a sequence $(g_i)_{1\leqslant i\leqslant n}$ of $n$ sections of $\sh{O}_\mathrm{X}$ above $\mathrm{U}$, that is transversally $\sh{F}$-regular relative to $\mathrm{S}$ (cf. \sref{IV.19.2.1} for the terminology) and such that $g_i(x)=0$ for $1\leqslant i\leqslant n$ at every point $x\in\mathrm{U}\cap\mathrm{Y}$.
\end{enumerate}

The set $\mathrm{V}_n$ of $y\in\mathrm{Y}$ satisfying these conditions is open and constructible in $\mathrm{X}$ $(\mathbf{0}_\mathrm{III},9.1.1)$; if moreover $\mathrm{Y}$ is locally constructible in $\mathrm{X}$, $\mathrm{V}_n$ is also locally constructible.
\end{proposition}

\begin{proof}
By virtue of \sref{IV.19.9.3}, condition $a)$ is equivalent to the existence of an open neighbourhood $\mathrm{U}$ of $y$ in $\mathrm{X}$ and of a sequence $(\sh{F}_{f(y)}|(\mathrm{U}\cap\mathrm{X}_{f(y)}))$-regular of $n$ sections $h_i$ of $\sh{O}_{\mathrm{X}_{f(y)}}$ above $\mathrm{U}\cap\mathrm{X}_{f(y)}$, such that $h_i(z)=0$ for $1\leqslant i\leqslant n$ at every point $z\in\mathrm{U}\cap\mathrm{Y}_{f(y)}$. If we denote by $\mathrm{Y}$ a closed sub-prescheme of $\mathrm{X}$ having $\mathrm{Y}$ as underlying space and by $\sh{J}$ the quasi-coherent Ideal of $\sh{O}_\mathrm{X}$ defining $\mathrm{Y}$, one can further say that $h_i\in\sh{J}_{f(y)}$ for $1\leqslant i\leqslant n$. As one can replace $\mathrm{U}$ by an affine open neighbourhood smaller than $y$, one can suppose that the $h_i$ are the images of a sequence $(f_i)_{1\leqslant i\leqslant n}$ of sections of $\sh{F}$ above $\mathrm{U}$, so that the germs $(g_i)_y$ belong to the maximal ideal $\mathfrak{m}_y$ of $\sh{O}_{\mathrm{X},y}$. The equivalence of $a)$ and $b)$ results then from \sref{IV.11.3.8}, which proves also that the set $\mathrm{V}_n$ is open in $\mathrm{Y}$.

It remains to prove the last assertion; as this involves local properties on $\mathrm{X}$, we can suppose $\mathrm{S}=\operatorname{Spec}(\mathrm{A})$ affine and $\mathrm{X}$ of finite presentation over $\mathrm{S}$; there exists then a Noetherian sub-ring $\mathrm{A}_0$ of $\mathrm{A}$, a prescheme $\mathrm{X}_0$ of finite type over $\mathrm{S}_0=\operatorname{Spec}(\mathrm{A}_0)$ and a coherent $\sh{O}_{\mathrm{X}_0}$-Module $\sh{F}_0$ such that $\mathrm{X}=\mathrm{X}_0\times_{\mathrm{S}_0}\mathrm{S}$ and $\sh{F}=\sh{F}_0\otimes_{\sh{O}_{\mathrm{X}_0}}\sh{O}_\mathrm{X}$ $(8.9.1)$; one can moreover suppose that, if $f_0:\mathrm{X}_0\to\mathrm{S}_0$ is the structural morphism, $\sh{F}_0$ is $f_0$-flat $(\sref{IV.11.2.6})$. Since $\mathrm{Y}$ is constructible, we can moreover $(8.3.11)$ suppose that $\mathrm{Y}$ is locally constructible in $\mathrm{X}_0$, so $\mathrm{Y}_0$ is the inverse image of a locally constructible $\mathrm{Y}_0$ in $\mathrm{X}_0$; we use the argument of \sref{IV.19.9.5} and the fact that $\operatorname{prof}$ is invariant under flat base change. The assertion on $\mathrm{V}_n$ being constructible follows from the criterion $(\mathbf{0}_\mathrm{III},9.1.1)$.
\end{proof}
