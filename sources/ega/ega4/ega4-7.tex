\section{Relations between a local Noetherian ring and its completion. Excellent rings}
\label{section:IV.7}

\subsection{Formal equidimensionality and formally catenary rings}
\label{subsection:IV.7.1}

\oldpage[IV-2]{183}

\begin{definition}[7.1.1]
\label{IV.7.1.1}
We say that a local Noetherian ring $A$ is \emph{formally equidimensional} if its completion $\widehat{A}$ is equidimensional $(\mathbf{0}$, 16.1.4).
\end{definition}

\begin{proposition}[7.1.2]
\label{IV.7.1.2}
Let $A$ be a local Noetherian ring, $\mathfrak{p}_i$ $(1\leq i\leq n)$ its minimal prime ideals.
For $A$ to be formally equidimensional, it is necessary and sufficient that the $A/\mathfrak{p}_i$ be equidimensional (in other words that the $A/\mathfrak{p}_i$ have the same dimension).
\end{proposition}

\begin{proof}
Indeed, this follows from the fact that for each of the $\mathfrak{p}_i$, $\mathfrak{p}'\cap A$ contains one of the $\mathfrak{p}_i$, hence $\mathfrak{p}'$ contains one of the $\mathfrak{p}_i\widehat{A}$, and $\widehat{A}/\mathfrak{p}_i\widehat{A}=(A/\mathfrak{p}_i)\otimes_A\widehat{A}$ is the completion of the local ring $A/\mathfrak{p}_i$ $(\mathbf{0_I}$, 7.3.3), hence a maximal chain of prime ideals of $\widehat{A}$ identifies therefore canonically to a chain of maximal prime ideals of one of the $\widehat{A}/\mathfrak{p}_i\widehat{A}$, and reciprocally, hence the conclusion.
\end{proof}

\begin{proposition}[7.1.3]
\label{IV.7.1.3}
Let $A$, $A'$ be two local Noetherian rings, $\mathfrak{m}$ the maximal ideal of $A$, $\varphi:A\to A'$ a local homomorphism; suppose that $A'$ is a flat $A$-module, and that $A'$ is equidimensional and catenary.
Then:
\begin{enumerate}
  \item[\rm(i)] $A$ is equidimensional and catenary.
  \item[\rm(ii)] Suppose moreover that $\mathfrak{m} A'$ is an ideal of definition of $A'$.
  Then, if $\mathfrak{a}$ is an ideal of $A$, for $A'/\mathfrak{a} A'$ to be equidimensional, it is necessary and sufficient that $A/\mathfrak{a}$ be so.
  In particular, for every prime ideal $\mathfrak{p}$ of $A$, $A'/\mathfrak{p} A'$ is formally equidimensional.
\end{enumerate}
\end{proposition}

\begin{proof}
Let us first prove the second assertion of (ii).
Set $X=\Spec(A)$, $X'=\Spec(A')$, $Y=\mathrm{V}(\mathfrak{p})=\Spec(A/\mathfrak{p})$, $Y'=\mathrm{V}(\mathfrak{p} A')=\Spec(A'/\mathfrak{p} A')$;
if $f:X'\to X$ is the morphism corresponding to $\varphi$, $Y'$ is also the closed sub-prescheme $f^{-1}(Y)$ of $X'$.
As $\mathfrak{m} A'$ is an ideal of definition of $A'$, one has
\begin{equation}
\label{IV.7.1.3.1}
  \dim(X)=\dim(X')\tag{7.1.3.1}
\end{equation}
by \sref{IV.6.1.3}; similarly $\mathfrak{m} A'/\mathfrak{p} A'$ is an ideal of definition of $A'/\mathfrak{p} A'$ and $Y'$ is flat over $Y$ \sref[I]{2.1.4}, hence \sref{IV.6.1.3}
\begin{equation}
\label{IV.7.1.3.2}
  \dim(Y)=\dim(Y').\tag{7.1.3.2}
\end{equation}
Let $y$ be the generic point of $Y$, $Y'_i$ $(1\leq i\leq r)$ the irreducible components of $Y'$, $y'_i$ the generic point of $Y'_i$ $(1\leq i\leq r)$; one has $f(y'_i)=y$ for all $i$ \sref[I]{2.3.4}.
On the other hand $y'_i$ is a maximal point of the fibre $f^{-1}(y)$ $(\mathbf{0_I}$, 2.1.8), hence $\sh{O}_{X',y'_i}\otimes_{\sh{O}_{X,y}}k(y)=\sh{O}_{X',y'_i}/\mathfrak{m}_y\sh{O}_{X',y'_i}$ is of dimension $0$, in other words $\mathfrak{m}_y\sh{O}_{X',y'_i}$ is an ideal of definition of $\sh{O}_{X',y'_i}$; one can again apply \sref{IV.6.1.3}, which gives
\begin{equation}
\label{IV.7.1.3.3}
  \dim(\sh{O}_{X',y'_i})=\dim(\sh{O}_{X,y})\tag{7.1.3.3}
\end{equation}
that is to say \sref[0]{5.1.2}
\begin{equation}
\label{IV.7.1.3.4}
  \codim(Y'_i,X')=\codim(Y,X)\tag{7.1.3.4}
\end{equation}
\oldpage[IV-2]{184}
for every $i$; by \sref{IV.7.1.3.1}, \sref{IV.7.1.3.2} and \sref{IV.7.1.3.4} and the inequality $\dim(Y'_i)\leq\dim(Y')$, one deduces
\begin{equation}
\label{IV.7.1.3.6}
  \dim(X)\leq\dim(Y)+\codim(Y,X)\tag{7.1.3.6}
\end{equation}
hence the two sides of this inequality are equal by $(\mathbf{0}$, 14.2.2.2); moreover
\begin{equation}
\label{IV.7.1.3.7}
  \dim(Y'_i)=\dim(Y'_i)=\dim(Y)\tag{7.1.3.7}
\end{equation}
for all $i$, which proves the second assertion of (ii).

Let us now demonstrate the first assertion of (ii).
Let $\mathfrak{p}_j$ $(1\leq j\leq n)$ be the minimal prime ideals of $A$ among those containing $\mathfrak{a}$, and let $\mathfrak{p}'_{jh}$ $(1\leq h\leq m_j)$ be the minimal prime ideals of $A'$ among those containing $\mathfrak{p}_j A'$; we know then \sref[I]{2.3.4} that the $\mathfrak{p}'_{jh}$ $(1\leq j\leq n$, $1\leq h\leq m_j$ for every $j)$ are also the minimal prime ideals of $A'$ among those containing $\mathfrak{a} A'$.
From what we have seen above, for every $j$, the dimensions of the rings $A'/\mathfrak{p}'_{jh}$ $(1\leq h\leq m_j)$ are all equal and are equal to $\dim(A'/\mathfrak{p}_j A')$, which itself is equal to $\dim(A/\mathfrak{p}_j)$ by \sref{IV.7.1.3.2}.
To say that $A'/\mathfrak{a} A'$ is equidimensional means therefore that the dimensions of the $A/\mathfrak{p}_j$ are all equal, that is to say that $A/\mathfrak{a}$ is equidimensional.

Let us now prove (i).
Let $\mathfrak{r}'$ be a minimal prime ideal of $A'$ among those containing $\mathfrak{m} A'$, and let $A''=A'_{\mathfrak{r}'}$; as $A''$ is a flat $A'$-module, it is also a flat $A$-module;
moreover $A''$ is equidimensional and catenary $(\mathbf{0}$, 16.1.4), and $\mathfrak{m} A''$ is an ideal of definition of $A''$.
One can hence restrict to the case where $\mathfrak{m} A'$ is an ideal of definition of $A'$.
If then $\mathfrak{p}$ and $\mathfrak{q}\subset\mathfrak{p}$ are two prime ideals of $A$, one can apply (ii) to $A/\mathfrak{q}$ and to $A'/\mathfrak{q} A'$, which is equidimensional and flat over $A/\mathfrak{q}$; if $Z=\Spec(A/\mathfrak{q})$ and $Y=\Spec(A/\mathfrak{p})$, one can apply \sref{IV.7.1.3.6}, hence $\dim(Z)=\dim(Y)+\codim(Y,Z)$; moreover one can apply \sref{IV.7.1.3.6}; as $X$ is equicodimensional, these relations show that it is biequidimensional by $(\mathbf{0}$, 14.3.3).
\end{proof}

\begin{corollary}[7.1.4]
\label{IV.7.1.4}
Let $A$ be a local Noetherian ring formally equidimensional.
Then:
\begin{enumerate}
  \item[\rm(i)] $A$ is equidimensional and catenary (in other words, \emph{biequidimensional}).
  \item[\rm(ii)] For every ideal $\mathfrak{a}$ of $A$, for $A/\mathfrak{a}$ to be equidimensional, it is necessary and sufficient that $A/\mathfrak{a}$ be so.
  In particular, for every prime ideal $\mathfrak{p}$ of $A$, $A/\mathfrak{p}$ is formally equidimensional.
\end{enumerate}
\end{corollary}

\begin{proof}
If $\mathfrak{m}$ is the maximal ideal of $A$, $\mathfrak{m}\widehat{A}$ is an ideal of definition of $\widehat{A}$.
By hypothesis $\widehat{A}$ is equidimensional, and we know that it is catenary \sref[0]{5.6.4}; it therefore suffices to apply \sref{IV.7.1.3} to $A'=\widehat{A}$.
\end{proof}

\begin{corollary}[7.1.5]
\label{IV.7.1.5}
Let $A$ be a local Noetherian ring such that there exists an $A$-module of finite type $M$ which is an $A$-module of Cohen-Macaulay and for which $\Supp(M)=\Spec(A)$ (this is the case for example if $A$ is a Cohen-Macaulay ring).
Then $A$ is formally equidimensional, hence \sref{IV.7.1.4} for every quotient ring $B$ of $A$ to be formally equidimensional, it is necessary and sufficient that it be equidimensional.
\end{corollary}

\begin{proof}
\oldpage[IV-2]{185}
Indeed, $\widehat{M}=M\otimes_A\widehat{A}$ is a $\widehat{A}$-module of Cohen-Macaulay $(\mathbf{0}$, 16.5.2), of support equal to $\Spec(\widehat{A})$; hence $(\mathbf{0}$, 16.5.4), $\widehat{A}$ is equidimensional.
\end{proof}

\begin{remark}[7.1.6]
\label{IV.7.1.6}
We have seen \sref{IV.6.3.8} that under the hypotheses of \sref{IV.7.1.5}, for every quotient ring $B$ of $A$, the fibres of the canonical morphism $\Spec(\widehat{B})\to\Spec(B)$ are Cohen-Macaulay schemes; hence (\sref{IV.6.4.3} and \sref[0]{5.7.5}), \emph{for $B$ to have no immersed associated prime ideals, it is necessary and sufficient that it be so for $\widehat{B}$}.
\end{remark}

\begin{lemma}[7.1.7]
\label{IV.7.1.7}
Let $A$ be a local Noetherian ring, $\mathfrak{p}_i$ $(1\leq i\leq r)$ its minimal prime ideals.
Suppose that each of the rings $A/\mathfrak{p}_i$ is formally equidimensional.
Then, for every prime ideal $\mathfrak{q}$ of $A$ and every $i$ such that $\mathfrak{p}_i\subset\mathfrak{q}$, the ring $A_\mathfrak{q}/\mathfrak{p}_i A_\mathfrak{q}$ is formally equidimensional.
\end{lemma}

\begin{proof}
As $A_\mathfrak{q}/\mathfrak{p}_i A_\mathfrak{q}$ is the local ring of $A/\mathfrak{p}_i$ at the prime ideal $\mathfrak{q}/\mathfrak{p}_i$, we can restrict to the case where $A$ is integral and formally equidimensional.
Set $A'=\widehat{A}$, and let $\mathfrak{q}'$ be one of the minimal prime ideals of $A'$ among those containing $\mathfrak{q} A'$; set $B=A_\mathfrak{q}$, $B'=A'_{\mathfrak{q}'}$.
Set $C=\widehat{B}$, $C'=\widehat{B}'$; as $B'$ is a $B$-module flat $(\mathbf{0_I}$, 6.3.2), we know that $C'$ is a $C$-module flat (Bourbaki, \emph{Alg.\ comm.}, chap.~III, \textsection~5, n\textsuperscript{o}~4, prop.~4); as $B'$ is catenary \sref[0]{5.6.4} and equidimensional by hypothesis (by \sref{IV.7.1.3}, (i)), it is the same for $B'=A'_{\mathfrak{q}'}$ $(\mathbf{0}$, 16.1.4); moreover, as $A'$ is isomorphic to a quotient of a regular ring by the theorem of Cohen $(\mathbf{0}$, 19.8.8), it is the same of $B'$ $(\mathbf{0}$, 17.3.9); hence we conclude from \sref{IV.7.1.5} that $C'$ is equidimensional.
On the other hand $C'$ is catenary \sref[0]{5.6.4}, hence $C$ is equidimensional by \sref{IV.7.1.3}, (i)).
\end{proof}

\begin{theorem}[7.1.8]
\label{IV.7.1.8}
Let $A$ be a local Noetherian ring.
The following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] Every integral quotient ring of $A$ is formally equidimensional.
  \item[\rm b)] The quotient rings of $A$ by its minimal prime ideals are formally equidimensional.
  \item[\rm c)] Every local ring $B$ which is an $A$-algebra essentially of finite type \sref[0]{1.3.8} and which is equidimensional, is formally equidimensional.
\end{enumerate}
\end{theorem}

\begin{proof}
It is trivial that $c)$ implies $a)$ and that $a)$ implies $b)$.
On the other hand, as every integral prime ideal of $A$ contains a minimal prime ideal of $A$, $b)$ implies $a)$ by \sref{IV.7.1.4}, (ii).
It remains to see that $a)$ implies $c)$.

It suffices to prove that the quotients of $B$ by its minimal prime ideals are formally equidimensional \sref{IV.7.1.2}; as every quotient ring of $B$ is still an $A$-algebra essentially of finite type \sref[0]{1.3.9}, one can in the first instance suppose $B$ \emph{integral}.
If $\mathfrak{q}$ is the inverse image in $A$ of the maximal ideal of $B$, we know that $B$ is also an $A_\mathfrak{q}$-algebra essentially of finite type \sref[0]{1.3.10}, and it follows from $a)$ and from \sref{IV.7.1.7} that every integral quotient of $A_\mathfrak{q}$ is formally equidimensional; one can hence suppose that the homomorphism $A\to B$ is a \emph{local} homomorphism.
The kernel $\mathfrak{r}$ of this homomorphism is then a prime ideal of $A$, and by $a)$, every integral quotient of $A/\mathfrak{r}$ is formally equidimensional; as $B$ is an $(A/\mathfrak{r})$-algebra essentially of finite type, one sees that one can also suppose that $A$ is \emph{integral} and is a sub-ring of $B$.

\oldpage[IV-2]{186}
Set $A'=\widehat{A}$, and $C'=A'[T_1,\ldots,T_n]=C\otimes_A A'$; there is a unique prime ideal $\mathfrak{p}'$ of $C'$ above the maximal ideal $\mathfrak{m} A'$ of $A'$, hence above $\mathfrak{p}$; set $B'=C'_{\mathfrak{p}'}$;
the homomorphism $B\to B'$ is local and makes $B'$ a $B$-module flat.
We know then that $\widehat{B}'$ is \emph{equidimensional}, from which we will deduce that $\widehat{B}$ is the same, which will complete the proof.

Indeed, $A'$ is a quotient of a regular ring by the theorem of Cohen $(\mathbf{0}$, 19.8.8), hence it is the same for $B'$ $(\mathbf{0}$, 17.3.9); by \sref{IV.7.1.5}, in order to show that $B'$ is \emph{equidimensional}, it suffices to show that $C'$ is \emph{equidimensional}, since $C'$ is a quotient of a regular ring.
But the minimal prime ideals of $C'$ are the ideals $\mathfrak{q}' C'$, where $\mathfrak{q}'$ ranges over the minimal prime ideals of $A'$ \sref[0]{5.5.3}, and one has $C'/\mathfrak{q}' C'=(A'/\mathfrak{q}')[T_1,\ldots,T_n]$.
As $A'$ is equidimensional by hypothesis (since $A$ is integral), it is the same of $C'$ by \sref[0]{5.5.4}. \textsc{q.e.d.}
\end{proof}

\begin{definition}[7.1.9]
\label{IV.7.1.9}
When the equivalent conditions of \sref{IV.7.1.8} are satisfied, we say that $A$ is a \emph{formally catenary} ring.
For a local Noetherian ring, to be formally equidimensional or formally catenary and equidimensional is therefore the same, by \sref{IV.7.1.2}.
\end{definition}

\begin{proposition}[7.1.10]
\label{IV.7.1.10}
Every quotient local ring of a local Cohen-Macaulay ring is formally catenary.
\end{proposition}

\begin{proof}
This follows immediately from \sref{IV.7.1.8} and from \sref{IV.7.1.5} applied to the integral quotient rings of $A$.
\end{proof}

\begin{proposition}[7.1.11]
\label{IV.7.1.11}
\begin{enumerate}
  \item[\rm(i)] A local Noetherian ring that is formally catenary is universally catenary.
  \item[\rm(ii)] If $A$ is a local Noetherian ring formally catenary, every local ring $B$ which is an $A$-algebra essentially of finite type is formally catenary.
\end{enumerate}
\end{proposition}

\begin{proof}
Assertion (ii) follows immediately from condition $c)$ of \sref{IV.7.1.8} and from the fact that if $B$ is a local ring which is an $A$-algebra essentially of finite type, $C$ is also an $A$-algebra essentially of finite type \sref[0]{1.3.9}.
To prove (i), note first that, by \sref{IV.7.1.8}, $b)$, a local Noetherian ring formally catenary $A$ is catenary, since the quotients of $A$ by its minimal prime ideals are catenary \sref{IV.7.1.4}.
If now $E$ is an $A$-algebra of finite type, it follows from what precedes and from (ii) that every local ring of $E$ at a prime ideal is catenary, hence $E$ is catenary.
Hence $A$ is universally catenary.
\end{proof}

\begin{remarks}[7.1.12]
\label{IV.7.1.12}
\begin{enumerate}
  \item[\rm(i)] It is not known whether, conversely, a local Noetherian ring which is universally catenary is always formally catenary.
  \item[\rm(ii)] A local Noetherian ring $A$ of \emph{dimension} $1$ is necessarily formally equidimensional, its maximal ideal not being able to be minimal (for in this case $A$ would be of dimension $0$).
  As $\dim(\widehat{A})=1$, this shows that $A$ is even \emph{formally equidimensional}, and \emph{a fortiori} formally catenary (hence \emph{universally catenary}).
  On the other hand, the local ring of dimension $2$ defined in \sref[0]{5.6.11}, which is catenary but non-universally catenary, is \emph{a fortiori} not formally catenary.
\end{enumerate}
\end{remarks}

\begin{corollary}[7.1.13]
\label{IV.7.1.13}
Let $Y$ be a locally Noetherian prescheme irreducible of dimension $1$, $X$ an irreducible prescheme, $f:X\to Y$ a dominant morphism of finite type, $\xi$, $\eta$ the generic points of $X$ and $Y$ respectively.
Then, for every $y\in f(X)$, the dimensions of the irreducible components of $f^{-1}(y)$ are all equal to $\deg.\mathrm{tr}_{k(\eta)}k(\xi)$.
\end{corollary}

\begin{proof}
By hypothesis, $f^{-1}(\eta)$ is irreducible of generic point $\xi$ and of dimension $\deg.\mathrm{tr}_{k(\eta)}k(\xi)$ \sref[0]{5.2.1}.
As $Y$ is irreducible and of dimension $1$, one has $\dim(\sh{O}_y)=1$ for every $y\neq\eta$, and $\sh{O}_y$ is universally catenary by \sref{IV.7.1.12}, (ii).
If $y\in f(X)$ and if $z$ is the generic point of an irreducible component $Z$ of $f^{-1}(y)$, one has, by \sref[0]{5.6.5}
\[
  \dim(Z)=1+\dim(f^{-1}(\eta))-\dim(\sh{O}_{X,z}).
\]
But one has $\dim(\sh{O}_{X,z})\leq\dim(\sh{O}_y)$ $(\mathbf{0}$, 16.3.9), and on the other hand, as $z$ is not the generic point of $X$, $\dim(\sh{O}_{X,z})>0$; hence $\dim(\sh{O}_{X,z})=1$ and $\dim(Z)=\dim(f^{-1}(\eta))$.
\end{proof}

\subsection{Strictly formally catenary rings}
\label{subsection:IV.7.2}

\begin{env}[7.2.1]
\label{IV.7.2.1}
\emph{Notations} \sref{IV.7.2.1}. ---
For an \emph{integral} Noetherian ring $A$, we denote by $A^{(1)}$ the intersection of the local rings $A_\mathfrak{p}$, where $\mathfrak{p}$ ranges over the set of prime ideals of $A$ \emph{of height} $1$ \sref[0]{5.10.17}.
If $A$ is a local Noetherian integral ring, we denote by $A^{(\omega)}$ the intersection of the $A_\mathfrak{p}$ where this time $\mathfrak{p}$ ranges over the set of all prime ideals of $A$ \emph{distinct from the maximal ideal}.

We say that a local Noetherian ring $A$ is \emph{strictly equidimensional} if, for every associated prime ideal $\mathfrak{p}\in\operatorname{Ass}(A)$, one has $\dim(A/\mathfrak{p})=\dim(A)$; this amounts to saying that $A$ is \emph{equidimensional} and \emph{without immersed associated prime ideals}.
\end{env}

\begin{env}[7.2.1.1]
\label{IV.7.2.1.1}
\emph{Example} \sref{IV.7.2.1.1}. ---
Let $A$ be a local Noetherian ring of \emph{dimension} $1$.
Then the following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] $A$ is without immersed associated prime ideals.
  \item[\rm a')] $\widehat{A}$ is without immersed associated prime ideals.
  \item[\rm b)] $A$ is strictly equidimensional.
  \item[\rm b')] $\widehat{A}$ is strictly equidimensional.
  \item[\rm c)] $A$ is a Cohen-Macaulay ring.
\end{enumerate}
Indeed $a)$ signifies that the maximal ideal $\mathfrak{m}$ of $A$ is not associated to $A$.
Since $a)$ implies $b)$.
Inversely, $b)$ implies that every associated prime ideal $\mathfrak{p}\in\operatorname{Ass}(A)$ is distinct from $\mathfrak{m}$, hence minimal, and hence $b)$ implies $a)$.
We already know that $a)$ and $c)$ are equivalent for a local ring of dimension $1$ \sref[0]{5.7.8}, and that $a')$ and $b')$ are equivalent by the same reasoning.
As $\dim(\widehat{A})=1$, one sees also that $a')$ and $b')$ are equivalent to $c)$.
\end{env}

\begin{proposition}[7.2.2]
\label{IV.7.2.2}
Let $A$ be a local Noetherian integral ring; set $A'=\widehat{A}$, $X=\Spec(A)$, $X'=\Spec(A')$ and denote by $f:X'\to X$ the canonical morphism.
Let $a$ be the closed point of $X$, $j:X-\{a\}\to X$ the canonical injection.
Let $\sh{F}$ be a coherent $\sh{O}_X$-Module, $\sh{F}'=f^*(\sh{F})=\sh{F}\otimes_{\sh{O}_X}\sh{O}_{X'}$; then the following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] The $\sh{O}_X$-Module $j_*(\sh{F}|X-\{a\})$ is coherent.
  \item[\rm b)] For every $x'\in\operatorname{Ass}(\sh{F}')$, one has $\dim(\overline{\{x'\}})\geq 2$.
\end{enumerate}
\end{proposition}

\begin{proof}
Let $a'$ be the unique point of the fibre $f^{-1}(a)$, and let $j':X'-\{a'\}\to X'$ be the canonical injection.
As the morphism $f$ is faithfully flat and quasi-compact, it is equivalent to say that $j_*(\sh{F}|X-\{a\})$ is coherent and that $j'_*(\sh{F}'|X'-\{a'\})$ is coherent \sref{IV.5.9.5}.
On the other hand, as $A'$ is isomorphic to a quotient of a regular local ring by the theorem of Cohen $(\mathbf{0}$, 19.8.8), it follows from \sref{IV.5.11.4} that condition $b)$ of the statement is equivalent to the fact that $j'_*(\sh{F}'|X'-\{a'\})$ is coherent.
Hence the proposition.
\end{proof}

\begin{proposition}[7.2.3]
\label{IV.7.2.3}
Let $A$ be a local Noetherian integral ring.
The notations being those of \sref{IV.7.2.2}, the following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] $A^{(1)}$ is a finite $A$-algebra.
  \item[\rm b)] For every closed part $T$ of $X$ of codimension $\geq 2$, if one denotes by $i:X-T\to X$ the canonical injection, $i_*(\sh{O}_U)$ is a coherent $\sh{O}_X$-Module.
  \item[\rm c)] For every $x'\in\operatorname{Ass}(\sh{O}_{X'})$ and every closed part $T$ of $X$ of codimension $\geq 2$, one has $\codim(f^{-1}(T)\cap\overline{\{x'\}},\{x'\})\geq 2$.
\end{enumerate}
\end{proposition}

\begin{proof}
Set $Z=Z^{(2)}(X)$ \sref[0]{5.10.13} and $Z'=f^{-1}(Z)$, which are stable by specialisation; the conditions $a)$ and $b)$ are equivalent respectively to the following properties:
$a_1)$ $\mathscr{H}^0_{X/Z}(\sh{O}_X)$ is coherent; $b_1)$ $\mathscr{H}^0_{X/T}(\sh{O}_X)$ is coherent for every closed part $T$ of $X$ of codimension $\geq 2$.
By \sref{IV.5.9.5}, these two last properties are equivalent respectively to: $a'_1)$ $\mathscr{H}^0_{X'/Z'}(\sh{O}_{X'})$ is coherent; $b'_1)$ posing $T'=f^{-1}(T)$, $\mathscr{H}^0_{X'/T'}(\sh{O}_{X'})$ is coherent for every closed part $T$ of $X$ of codimension $\geq 2$.
Since every point of $\operatorname{Ass}(\sh{O}_{X'})$ projects to the generic point of $X$ since $f$ is a flat morphism \sref[I]{3.3.2}; since by definition this point does not belong to $Z$, we see that $\operatorname{Ass}(\sh{O}_{X'})$ does not meet $Z'$; the equivalence of $a'_1)$ and $b'_1)$ follows from \sref{IV.5.11.4}, and this gives the equivalence of $a)$, $b)$ and $c)$.
\end{proof}

\begin{proposition}[7.2.4]
\label{IV.7.2.4}
Let $A$ be a local Noetherian ring and set $X=\Spec(A)$.
The following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] For every integral quotient ring $B$ of $A$, the ring $B^{(1)}$ is a finite $B$-algebra.
  \item[\rm b)] For every coherent $\sh{O}_X$-Module $\sh{F}$ and every part $Z$, stable by specialisation, and such that for every $x\in\operatorname{Ass}(\sh{F})\cap(X-Z)$, one has $\codim(\overline{\{x\}}\cap Z,\{x\})\geq 2$, the $\sh{O}_X$-Module $\mathscr{H}^0_{X/Z}(\sh{F})$ is coherent.
  \item[\rm c)] For every coherent $\sh{O}_U$-Module $\sh{G}$ (where $U=X-T$) such that, for every $x\in\operatorname{Ass}(\sh{G})$, one has $\codim(\overline{\{x\}}\cap T,\{x\})\geq 2$, $i_*(\sh{G})$ (where $i:U\to X$ is the canonical injection) is a coherent $\sh{O}_X$-Module.
  \item[\rm d)] For every integral quotient ring $B$ of $A$ and every ideal $\mathfrak{J}$ of height $\geq 2$ in $B$, the ring $\bigcap_{\mathfrak{p}\not\supset\mathfrak{J}} B_\mathfrak{p}$ is a finite $B$-algebra.
  \item[\rm e)] For every integral quotient ring $B$ of $A$ and every local integral ring $C=B_\mathfrak{q}$ and every prime ideal $\mathfrak{q}$ of $B$ such that $\dim(C)\geq 2$, the ring $C^{(\omega)}$ is a finite $C$-algebra (or, what amounts to the same, if $Y=\Spec(C)$, and $j:U\to Y$ is the complement in $Y$ of the closed point of $C$ and $j:U\to Y$ the canonical injection, $j_*(\sh{O}_U)$ is a coherent $\sh{O}_Y$-Module).
  \item[\rm f)] For every integral quotient ring $B$ of $A$ and every local ring $C=B_\mathfrak{q}$ and every prime ideal $\mathfrak{q}$ of $B$ such that $\dim(C)\geq 2$, one has $\dim(\widehat{C}/\mathfrak{r})\geq 2$ for every $\mathfrak{r}\in\operatorname{Ass}(\widehat{C})$.
\end{enumerate}
\end{proposition}

\begin{theorem}[7.2.5]
\label{IV.7.2.5}
Let $A$ be a local Noetherian ring.
The following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] For every integral quotient ring $B$ of $A$, the completion $\widehat{B}$ is strictly equidimensional \sref{IV.7.2.1}.
  \item[\rm b)] $A$ is formally catenary \sref{IV.7.1.9} and the fibres of the morphism $\Spec(\widehat{A})\to\Spec(A)$ satisfy $(\mathrm{S}_1)$ (in other words, \emph{have no immersed associated prime cycle}).
  \item[\rm c)] $A$ is universally catenary \sref[0]{5.6.2} and for every integral quotient ring $B$ of $A$, the ring $B^{(1)}$ is a finite $B$-algebra (cf.\ \sref{IV.7.2.4}).
  \item[\rm d)] $A$ is universally catenary and for every integral quotient ring $B$ of $A$ and every local integral ring $C=B_\mathfrak{q}$ and every prime ideal $\mathfrak{q}$ of $B$ such that $\dim(C)\geq 2$, the ring $C^{(\omega)}$ is a finite $C$-algebra.
  \item[\rm e)] $A$ is universally catenary and for every integral quotient ring $B$ of $A$ and every local integral ring $C=B_\mathfrak{q}$ and every prime ideal $\mathfrak{q}$ of $B$ such that $\dim(C)\geq 2$, the completion $\widehat{C}$ has no associated prime cycle of dimension $1$.
\end{enumerate}
Moreover, when these conditions are satisfied, then, for every quotient ring $B$ of $A$ which is strictly equidimensional, the completion $\widehat{B}$ is strictly equidimensional.
\end{theorem}

\begin{definition}[7.2.6]
\label{IV.7.2.6}
When a local Noetherian ring $A$ satisfies the equivalent conditions of \sref{IV.7.2.5}, we say that it is \emph{strictly formally catenary}.
\end{definition}

\begin{corollary}[7.2.7]
\label{IV.7.2.7}
Let $A$ be a local Noetherian ring such that there exists an $A$-module of finite type $M$ which is a Cohen-Macaulay $A$-module and for which $\Supp(M)=\Spec(A)$; then $A$ is strictly formally catenary.
\end{corollary}

\begin{proof}
Indeed, $A$ is formally catenary \sref{IV.7.1.5} and \sref{IV.7.1.9}), and on the other hand the fibres of $\Spec(\widehat{A})\to\Spec(A)$ are Cohen-Macaulay preschemes \sref{IV.6.3.8}, hence \emph{a fortiori} satisfy $(\mathrm{S}_1)$.
\end{proof}

\begin{corollary}[7.2.8]
\label{IV.7.2.8}
If $A$ is a local Noetherian ring strictly formally catenary, every local ring $B$ which is an $A$-algebra essentially of finite type is strictly formally catenary.
\end{corollary}

\begin{proof}
Indeed, $B$ is formally catenary \sref{IV.7.1.11} and it follows from \sref{IV.7.4.4}\footnote{The corollary \sref{IV.7.2.8} is not used to demonstrate \sref{IV.7.4.4}.} that the fibres of $\Spec(\widehat{B})\to\Spec(B)$ satisfy $(\mathrm{S}_1)$, hence the conclusion.
\end{proof}

\begin{corollary}[7.2.9]
\label{IV.7.2.9}
Every local Noetherian ring of dimension $1$ is strictly formally catenary.
\end{corollary}

\begin{proof}
Indeed, every integral quotient ring of such a ring $A$ is of dimension $0$ or $1$, hence its completion is strictly equidimensional \sref{IV.7.2.1.1}.
\end{proof}

\begin{remarks}[7.2.10]
\label{IV.7.2.10}
\begin{enumerate}
  \item[\rm(i)] It follows from \sref{IV.7.2.7} and \sref{IV.7.2.8} that every quotient of a Cohen-Macaulay local ring is strictly formally catenary.
  A local Noetherian ring of dimension $2$ satisfying $(\mathrm{S}_2)$ is strictly formally catenary, since it is then a Cohen-Macaulay ring.
  Recall however that there are local Noetherian integral rings of dimension $2$ which are not universally catenary, nor \emph{a fortiori} strictly formally catenary \sref[0]{5.6.11}.
  \item[\rm(ii)] It is not known whether a formally catenary ring \sref{IV.7.1.9} is strictly formally catenary; this is related to whether one knows whether for a local Noetherian integral ring, $\widehat{B}$ has no immersed associated prime ideals \sref{IV.6.4.3}.
  We also do not know whether, in the equivalent conditions $c)$, $d)$, $e)$ of \sref{IV.7.2.5}, one can replace the hypothesis that $A$ is universally catenary by the hypothesis that $A$ is catenary; one can show that this is so when $A$ is henselian (18.9.6).
\end{enumerate}
\end{remarks}

\subsection{Formal fibres of local Noetherian rings}
\label{subsection:IV.7.3}

\begin{env}[7.3.1]
\label{IV.7.3.1}
We shall consider in this number and the next two properties $\mathbf{P}(Z,k)$ of the following form:
\begin{quote}
``$Z$ is a locally Noetherian prescheme on a field $k$,
and for every $z\in Z$, one has $Q(\sh{O}_z,k)$''
\end{quote}
where $Q(A,k)$ is a property of a local Noetherian ring $A$ which is a $k$-algebra; one supposes that if $k'$ is a field isomorphic to $k$, and if $A'$ is a local Noetherian ring which is a $k'$-algebra di-isomorphic to $A$, the properties $Q(A,k)$ and $Q(A',k')$ are equivalent.
\end{env}

If $X$, $Y$ are two locally Noetherian preschemes, we say that a morphism $f:X\to Y$ is a $\mathbf{P}$\emph{-morphism} if:
\begin{enumerate}
  \item[1\textsuperscript{o}] $f$ is \emph{flat};
  \item[2\textsuperscript{o}] for every $y\in Y$, the property $\mathbf{P}(f^{-1}(y),k(y))$ is true.
\end{enumerate}

\begin{lemma}[7.3.2]
\label{IV.7.3.2}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism.
The following properties are equivalent:
\begin{enumerate}
  \item[\rm a)] $f$ is a $\mathbf{P}$-morphism.
  \item[\rm b)] For every $x\in X$, if one sets $y=f(x)$, then $\sh{O}_x$ is an $\sh{O}_y$-module flat and the property $Q(\sh{O}_x\otimes_{\sh{O}_y}k(y),k(y))$ is true.
  \item[\rm c)] For every $x\in X$, if one sets $y=f(x)$, the morphism $\Spec(\sh{O}_x)\to\Spec(\sh{O}_y)$ corresponding to $\sh{O}_y\to\sh{O}_x$ is a $\mathbf{P}$-morphism.
  \item[\rm c')] For every closed point $x\in X$, the morphism $\Spec(\sh{O}_x)\to\Spec(\sh{O}_y)$ is a $\mathbf{P}$-morphism.
\end{enumerate}
\end{lemma}

\begin{proof}
The equivalence of a) and b) follows from the definitions; it is the same for c), by $(\textbf{I}$, 2.4.2); finally the equivalence of b) and c') follows from the fact that for every $x\in X$, $\overline{\{x\}}$ contains a closed point \sref[0]{5.1.11}.
\end{proof}

\begin{corollary}[7.3.3]
\label{IV.7.3.3}
Let $A$, $B$ be two Noetherian rings, $\varphi:A\to B$ a homomorphism such that the corresponding morphism $\Spec(B)\to\Spec(A)$ is a $\mathbf{P}$-morphism.
Then, for every multiplicative subset $S$ of $A$, the morphism $\Spec(S^{-1}B)\to\Spec(S^{-1}A)$ is a $\mathbf{P}$-morphism.
\end{corollary}

\begin{proof}
This follows immediately from \sref{IV.7.3.2} and from $(\textbf{I}$, 1.6.2).
\end{proof}

\begin{env}[7.3.4]
\label{IV.7.3.4}
We shall always suppose in what follows that the property $Q$ is such that the three following conditions are satisfied:
\begin{itemize}
  \item[$(\mathrm{P_I})$] \emph{(transitivity).} --- If $f:X\to Y$ is a regular morphism \sref{IV.6.8.1} and $g:Y\to Z$ a $\mathbf{P}$-morphism, then $g\circ f$ is a $\mathbf{P}$-morphism.
  \item[$(\mathrm{P_{II}})$] \emph{(descent).} --- If $f:X\to Y$ and $g:Y\to Z$ are morphisms of locally Noetherian preschemes such that $f$ is faithfully flat and that $g\circ f$ is a $\mathbf{P}$-morphism, then $g$ is a $\mathbf{P}$-morphism.
  \item[$(\mathrm{P_{III}})$] --- For every field $k$, the property $\mathbf{P}(\Spec(k),k)$ is true.
\end{itemize}
\end{env}

\begin{remarks}[7.3.5]
\label{IV.7.3.5}
\begin{enumerate}
  \item[\rm(i)] Conditions $(\mathrm{P_I})$ and $(\mathrm{P_{III}})$ imply that every \emph{regular morphism} is a $\mathbf{P}$-morphism.
  \item[\rm(ii)] Note that the hypotheses of $(\mathrm{P_I})$ (resp.\ $(\mathrm{P_{II}})$) imply that $h=g\circ f$ (resp.\ $g$) is flat \sref[I]{2.2.13}; the hypotheses of $(\mathrm{P_I})$ or of $(\mathrm{P_{II}})$ also imply that for every $z\in Z$, $f_z:h^{-1}(z)\to g^{-1}(z)$ is flat \sref[I]{2.1.4}; as for every $y\in g^{-1}(z)$, $f_z^{-1}(y)$ is isomorphic to $f^{-1}(y)$ $(\textbf{I}$, 3.6.4), this shows that it suffices, to verify conditions $(\mathrm{P_I})$ and $(\mathrm{P_{II}})$, to do so only when $Z$ is the \emph{spectrum of a field}.
  \item[\rm(iii)] In certain cases, the property $Q$ will be such that the following condition is also satisfied:
  \begin{itemize}
    \item[$(\mathrm{P'_I})$] --- If $f:X\to Y$ and $g:Y\to Z$ are two $\mathbf{P}$-morphisms, then $g\circ f$ is a $\mathbf{P}$-morphism.
  \end{itemize}
\end{enumerate}
\end{remarks}

\begin{env}[7.3.6]
\label{IV.7.3.6}
We shall say that the property $\mathbf{P}$ is \emph{geometric} if it satisfies (in addition to $(\mathrm{P_I})$, $(\mathrm{P_{II}})$ and $(\mathrm{P_{III}})$) the condition:
\begin{itemize}
  \item[$(\mathrm{P_{IV}})$] \emph{(finite extension of base field).} --- If $\mathbf{P}(Z,k)$ is true, then for every extension $k'$ of finite type of $k$, $\mathbf{P}(Z\otimes_k k',k')$ is true.
\end{itemize}
\end{env}

\begin{lemma}[7.3.7]
\label{IV.7.3.7}
Let $f:X\to Y$ be a $\mathbf{P}$-morphism of locally Noetherian preschemes, $g:Y'\to Y$ a morphism locally of finite type.
Then, if $\mathbf{P}$ satisfies condition $(\mathrm{P_{IV}})$, the morphism $f'=f_{(Y')}:X\times_Y Y'\to Y'$ is a $\mathbf{P}$-morphism.
\end{lemma}

\begin{proof}
Indeed, for every $y'\in Y'$, if one sets $y=g(y')$, $k(y')$ is an extension of finite type of $k(y)$ $(\textbf{I}$, 6.4.11) and $f'^{-1}(y')=f^{-1}(y)\otimes_{k(y)}k(y')$ $(\textbf{I}$, 3.6.4); it therefore suffices to apply $(\mathrm{P_{IV}})$.
\end{proof}

\begin{env}[7.3.8]
\label{IV.7.3.8}
\emph{Examples} \sref{IV.7.3.8}. ---
The following properties $\mathbf{P}(Z,k)$ satisfy conditions $(\mathrm{P_I})$, $(\mathrm{P_{II}})$ and $(\mathrm{P_{III}})$:
\begin{enumerate}
  \item[\rm(i)] \quad (also denoted $(\mathrm{i}_n)$) $Z$ is of coprofondeur $\leq n$.
  \item[\rm(ii)] \quad $Z$ is a Cohen-Macaulay prescheme.
  \item[\rm(iii)] (also denoted $(\mathrm{iii}_n)$) $Z$ satisfies $(\mathrm{S}_n)$.
  \item[\rm(iv)] \quad $Z$ is regular.
  \item[\rm(v)] \quad (also denoted $(\mathrm{v}_n)$) $Z$ satisfies $(\mathrm{R}_n)$.
  \item[\rm(vi)] \quad $Z$ is reduced.
  \item[\rm(vii)] \quad $Z$ is normal.
\end{enumerate}
For properties (ii) to (vii), this follows from \sref{IV.6.6.1} which, in fact, demonstrates the stronger condition $(\mathrm{P'_I})$.
For (i), property $(\mathrm{P_I})$ follows from \sref{IV.6.6.2}; property $(\mathrm{P_{II}})$ follows from \sref{IV.6.3.2} and from the fact that a regular prescheme is of coprofondeur $0$.

Moreover, it follows from \sref{IV.6.7.1} that properties (i), (ii) and (iii) are \emph{geometric}; and by \sref{IV.6.7.8}, it is the same for the following:
\begin{enumerate}
  \item[\rm(iv')] \quad $Z$ is geometrically regular.
  \item[\rm(v')] \quad (also denoted $(\mathrm{v}'_n)$) $Z$ possesses the geometric property $(\mathrm{R}_n)$.
  \item[\rm(vi')] \quad $Z$ is geometrically reduced.
  \item[\rm(vii')] \quad $Z$ is geometrically normal.
\end{enumerate}
\end{env}

\begin{remarks}[7.3.9]
\label{IV.7.3.9}
\begin{enumerate}
  \item[\rm(i)] Each of properties (iv'), $(\mathrm{v}'_n)$, (vi'), (vii') implies the corresponding property (iv), $(\mathrm{v}_n)$, (vi), (vii).
  Property (iv) implies all of (i) through (vii), and property (iv') implies all the properties enumerated in \sref{IV.7.3.8}.
  Let us also recall that $(\mathrm{i}_0)$ is equivalent to (ii); the conjunction of $(\mathrm{iii}_1)$ and of $(\mathrm{v}_0)$ (resp.\ of $(\mathrm{iii}_1)$ and $(\mathrm{v}'_0)$) is equivalent to (vi) (resp.\ (vi')) \sref[0]{5.8.5}; finally the conjunction of $(\mathrm{iii}_2)$ and $(\mathrm{v}_1)$ (resp.\ $(\mathrm{iii}_2)$ and $(\mathrm{v}'_1)$) is equivalent to (vii) (resp.\ (vii')) \sref[0]{5.8.6}.
  \item[\rm(ii)] In all the examples of \sref{IV.7.3.8}, the property $Q(\sh{O}_z,k)$ which serves to define $\mathbf{P}$ is such that, for every generisation $z'$ of $z$ in $Z$, $Q(\sh{O}_z,k)$ \emph{implies} $Q(\sh{O}_{z'},k)$: by \sref[I]{2.3.4}, it suffices to verify this for properties (i) to (vii) (and even (i) to (v) by the remark \sref{IV.7.3.9}, (i)); this follows from \sref{IV.6.3.9} for (i), and finally from $(\mathbf{0}$, 16.5.10 and 17.3.2) for (ii) and (iv).
\end{enumerate}
\end{remarks}

\begin{env}[7.3.10]
\label{IV.7.3.10}
We shall say that a property $\mathbf{P}(Z,k)$ is \emph{of the first type} if the property $Q(A,k)$ which serves to define $\mathbf{P}$ is a property $\mathbf{R}(A)$ \emph{not involving $k$} and is true when $A$ is a field.
It is clear that in the examples of \sref{IV.7.3.8}, properties (i) to (vii) are of the first type, properties (iv') to (vii') of the second type.

We shall say that $\mathbf{P}(Z,k)$ is \emph{of the second type} if the property $Q(A,k)$ is of the form
\begin{quote}
``For every extension $k'$ of finite type of $k$ and every closed point $z'$ of $\Spec(A\otimes_k k')$ above the closed point of $\Spec(A)$, one has $\mathbf{R}(\sh{O}_{z'})$''
\end{quote}
$\mathbf{R}(A)$ being assumed true when $A$ is a field.
\end{env}

\begin{env}[7.3.13]
\label{IV.7.3.13}
Given a semi-local Noetherian ring $A$, we call \emph{formal fibres} of $A$ the fibres of the canonical morphism $f:\Spec(\widehat{A})\to\Spec(A)$; for every prime ideal $\mathfrak{p}$ of $A$, the formal fibre at $\mathfrak{p}$ is then the prescheme $\Spec(\widehat{A}\otimes_A k(\mathfrak{p}))$; as the completion of the local ring $A/\mathfrak{p}$ is $\widehat{A}/\mathfrak{p}\widehat{A}=(A/\mathfrak{p})\otimes_A\widehat{A}$, one sees that the formal fibre of $A/\mathfrak{p}$ at the generic point (o) of $\Spec(A/\mathfrak{p})$ is also the formal fibre of $A$ at $\mathfrak{p}$.
\end{env}

The property $\mathbf{P}$ being defined as in \sref{IV.7.3.1}, we shall say that \emph{the formal fibres of $A$ have property $\mathbf{P}$}, or that $A$ is a $\mathbf{P}$\emph{-ring}, if, for every $x\in\Spec(A)$, $\mathbf{P}(f^{-1}(x),k(x))$ is true.
As $f$ is \emph{flat}, this amounts to saying that $f$ is a $\mathbf{P}$-morphism.

\begin{proposition}[7.3.14]
\label{IV.7.3.14}
Let $A$ be a semi-local Noetherian ring, $\mathfrak{m}_i$ $(1\leq i\leq r)$ its maximal ideals, and set $A_i=A_{\mathfrak{m}_i}$; for $A$ to be a $\mathbf{P}$-ring, it is necessary and sufficient that each of the $A_i$ be so.
\end{proposition}

\begin{proof}
Indeed, $\widehat{A}$ is the direct product of the $\widehat{A}_i$, hence the formal fibre of $A$ at a point $x\in\Spec(A)$ is the sum of the formal fibres at $x$ of those $A_i$ such that $x\in\Spec(A_i)$.
\end{proof}

\begin{proposition}[7.3.15]
\label{IV.7.3.15}
Let $A$ be a semi-local Noetherian ring.
\begin{enumerate}
  \item[\rm(i)] If $A$ is a $\mathbf{P}$-ring, every quotient ring of $A$ is a $\mathbf{P}$-ring.
  \item[\rm(ii)] If $\mathbf{P}$ satisfies moreover condition $(\mathrm{P_{IV}})$ \sref{IV.7.3.6}, then every finite $A$-algebra is a $\mathbf{P}$-ring.
\end{enumerate}
\end{proposition}

\begin{proof}
For every ideal $\mathfrak{a}$ of $A$, $\widehat{A}\otimes_A(A/\mathfrak{a})$ is the completion of $A/\mathfrak{a}$, and the formal fibres of $A/\mathfrak{a}$ are the formal fibres of $A$ at the points of $\mathrm{V}(\mathfrak{a})$, hence (i).
On the other hand, if $B$ is a finite $A$-algebra, $\widehat{B}=B\otimes_A\widehat{A}$, and (ii) follows from \sref{IV.7.3.7}.
\end{proof}

\begin{proposition}[7.3.16]
\label{IV.7.3.16}
Suppose that the property $\mathbf{P}$ is of the first (resp.\ second) type, defined from a property $\mathbf{R}$ \sref{IV.7.3.10}.
When $\mathbf{P}$ is of the second type, suppose in addition that the relation
\begin{quote}
``for every finite extension $k'$ of $k$, and every $z'\in Z\otimes_k k'$, $\mathbf{R}(\sh{O}_{z'})$ is true''
\end{quote}
implies $\mathbf{P}(Z,k)$ for the examples (iv') to (vii') of \sref{IV.7.3.8}, by \sref{IV.6.7.7} and \sref{IV.4.6.1}).

Let $A$ be a semi-local Noetherian ring.
If the property $\mathbf{R}$ satisfies conditions $(\mathrm{R_I})$ and $(\mathrm{R_{II}})$, the following properties are equivalent:
\begin{enumerate}
  \item[\rm a)] $A$ is a $\mathbf{P}$-ring.
  \item[\rm b)] For every integral quotient ring $B$ of $A$ (resp.\ every finite integral $A$-algebra $B$) and every prime ideal $\mathfrak{q}$ of $\widehat{B}$ whose inverse image in $B$ is $0$, $\mathbf{R}(\widehat{B}_\mathfrak{q})$ is true.
\end{enumerate}
If moreover $\mathbf{R}$ satisfies condition $(\mathrm{R'_I})$, properties $a)$ and $b)$ are also equivalent to:
\begin{enumerate}
  \item[\rm c)] For every integral quotient ring $B$ of $A$ (resp.\ every finite $A$-algebra $B$) if one sets $Y=\Spec(B)$, $Y'=\Spec(\widehat{B})$, and if $g:Y'\to Y$ is the canonical morphism, one has (with the notations of \sref{IV.7.3.12})
\begin{equation}
\label{IV.7.3.16.1}
  \mathrm{U}_\mathbf{R}(Y')=g^{-1}(\mathrm{U}_\mathbf{R}(Y)).\tag{7.3.16.1}
\end{equation}
\end{enumerate}
\end{proposition}

\begin{proposition}[7.3.17]
\label{IV.7.3.17}
Suppose that the property $\mathbf{R}$ satisfies conditions $(\mathrm{R'_I})$ and $(\mathrm{R_{II}})$ and that $\mathbf{P}$ is a property of the first (resp.\ second) type defined from $\mathbf{R}$ \sref{IV.7.3.10}.
Then, if $A$ is a $\mathbf{P}$-ring, $\mathbf{R}(A)$ and $\mathbf{R}(\widehat{A})$ are equivalent.
\end{proposition}

\begin{proof}
This follows from \sref{IV.7.3.12.1} applied to $Y=\Spec(A)$ and $X=\Spec(\widehat{A})$.
\end{proof}

\begin{proposition}[7.3.18]
\label{IV.7.3.18}
Suppose that $\mathbf{P}$ is a property of the first (resp.\ second) type defined from a property $\mathbf{R}$ satisfying conditions $(\mathrm{R'_I})$ and $(\mathrm{R_{II}})$ as well as the following condition:
\begin{itemize}
  \item[$(\mathrm{R_{III}})$] For every local Noetherian complete ring $C$, if one sets $Z=\Spec(C)$, the set $\mathrm{U}_\mathbf{R}(Z)$ \sref{IV.7.3.12} is open in $Z$.
\end{itemize}
Then, if $A$ is a $\mathbf{P}$-ring and if one sets $X=\Spec(A)$, the set $\mathrm{U}_\mathbf{R}(X)$ is open in $X$.
\end{proposition}

\begin{proof}
Indeed, if $X'=\Spec(\widehat{A})$ and if $f:X'\to X$ is the canonical morphism, on a $\mathrm{U}_\mathbf{R}(X')=f^{-1}(\mathrm{U}_\mathbf{R}(X))$ \sref{IV.7.3.12}.
As $f$ is faithfully flat and quasi-compact and by hypothesis $\mathrm{U}_\mathbf{R}(X')$ is open in $X'$, the conclusion follows from \sref[I]{2.3.12}.
\end{proof}

\begin{remarks}[7.3.19]
\label{IV.7.3.19}
\begin{enumerate}
  \item[\rm(i)] The property $\mathbf{R}$ which serves to define $\mathbf{P}$ satisfies condition $(\mathrm{R_{III}})$ in all the examples enumerated in \sref{IV.7.3.8}.
  For (i), (ii) and (iii), this follows from \sref{IV.6.11.2} and from the theorem of Cohen $(\mathbf{0}$, 19.8.8);
  when $\mathbf{P}$ is one of the properties (iv), (iv'), (v), (v'), this follows from \sref{IV.6.12.7} and \sref{IV.6.12.9},
  and when $\mathbf{P}$ is one of the properties (vii), (vii'), from \sref{IV.6.12.7} and \sref{IV.6.13.4};
  finally, for (vi) and (vi') the assertion of \sref{IV.7.3.18} is trivial, being true for every locally Noetherian prescheme.
  \item[\rm(ii)] We have already noted \sref{IV.6.4.3} that when $\mathbf{P}$ is one of the properties (ii) or $(\mathrm{iii}_n)$ of \sref{IV.7.3.8}, we do not know whether every local Noetherian ring is a $\mathbf{P}$-ring;
  recall however that when $A$ is a \emph{quotient of a Cohen-Macaulay ring}, the formal fibres of $A$ are Cohen-Macaulay preschemes \sref{IV.6.3.8}.
  \item[\rm(iii)] The most interesting case of the notion of $\mathbf{P}$-ring is that which corresponds to the strongest property (iv') of \sref{IV.7.3.8}, that is to say the rings whose \emph{formal fibres are geometrically regular}.
  Fields, and more generally local Noetherian complete rings, trivially satisfy this property.
  \item[\rm(iv)] Let $A$ be a local Noetherian ring of \emph{dimension} $1$; $\Spec(A)$ is then formed of the closed point $a$, corresponding to the maximal ideal $\mathfrak{m}$, and of the maximal points $b_i$ $(1\leq i\leq r)$ corresponding to the minimal prime ideals of $A$.
  One has $\dim(\widehat{A})=1$ and the maximal ideal of $\widehat{A}$ is $\mathfrak{m}\widehat{A}$; the formal fibre at $a$ is then $\Spec(k)$, where $k=A/\mathfrak{m}$ is the residue field; the formal fibre at each $b_i$ is the spectrum of an Artinian ring whose residue fields are the residue fields $L_{ij}$ of $\Spec(\widehat{A})$ at the maximal points $b_{ij}$ above $b_i$.
  As an Artinian ring is a Cohen-Macaulay ring, one sees that $A$ is a $\mathbf{P}$-ring when $\mathbf{P}$ is property (ii) of \sref{IV.7.3.8}.
  Moreover, as a reduced Artinian ring is a direct sum of fields, the following properties are equivalent \sref{IV.6.7.7}:
  \begin{enumerate}
    \item[\rm a)] the formal fibres of $A$ are geometrically reduced;
    \item[\rm b)] the formal fibres of $A$ are geometrically normal;
    \item[\rm c)] the formal fibres of $A$ are geometrically regular;
  \end{enumerate}
  moreover, when $A$ is reduced, they are also equivalent to:
  \begin{enumerate}
    \item[\rm d)] $\widehat{A}$ is reduced and $L_{ij}$ is a separable extension of $K_i$ for every couple $(i,j)$ \sref{IV.4.6.1}.
  \end{enumerate}
  In particular, if $A$ is a \emph{discrete valuation ring}, $K$ its field of fractions, and if $\widehat{K}$ is the completion of $K$ for a valuation corresponding to $A$ (field of fractions of $\widehat{A}$), for the formal fibres of $A$ to be geometrically regular, it is necessary and sufficient that $\widehat{K}$ be a \emph{separable extension} of $K$.
  This is always the case when $K$ is of \emph{characteristic} $0$.
\end{enumerate}
\end{remarks}

\subsection{Permanence of properties of formal fibres}
\label{subsection:IV.7.4}

\begin{env}[7.4.1]
\label{IV.7.4.1}
In the remainder of this number, we suppose that the property $\mathbf{P}$ is of the form defined in \sref{IV.7.3.1}, and \emph{satisfies conditions} $(\mathrm{P_I})$, $(\mathrm{P_{II}})$ and $(\mathrm{P_{III}})$ of \sref{IV.7.3.4}.
Moreover, we suppose that the property $Q$ which serves to define $\mathbf{P}$ is such that, for every generisation $z'$ of $z$ in $Z$, $Q(\sh{O}_z,k)$ \emph{implies} $Q(\sh{O}_{z'},k)$.
\end{env}

\begin{lemma}[7.4.2]
\label{IV.7.4.2}
Let $A$, $A'$ be local Noetherian rings, $\varphi:A\to A'$ a local homomorphism such that $f={}^a\varphi:\Spec(A')\to\Spec(A)$ is a $\mathbf{P}$-morphism.
Then, if the formal fibres of $A'$ are geometrically regular, $A$ is a $\mathbf{P}$-ring.
\end{lemma}

\begin{proof}
Consider the completed homomorphism $\widehat{\varphi}:\widehat{A}\to\widehat{A}'$ and the corresponding morphism $\widehat{f}={}^a\widehat{\varphi}$; one has the commutative diagram
\[
\begin{tikzcd}
  \Spec(\widehat{A}) \arrow[leftarrow]{r}{\widehat{f}} & \Spec(\widehat{A}') \\
  \Spec(A) \arrow{u}{g} \arrow[leftarrow]{r}{f} & \Spec(A') \arrow{u}{g'}
\end{tikzcd}
\]
where $g$ and $g'$ are the canonical morphisms.
As by hypothesis $f$ is a $\mathbf{P}$-morphism and $g'$ a regular morphism, it follows from $(\mathrm{P_I})$ that $f\circ g'=g\circ\widehat{f}$ is a $\mathbf{P}$-morphism.
On the other hand, the hypothesis that $f$ is a $\mathbf{P}$-morphism implies that $f$ is flat, hence it is the same for $\widehat{f}$ (Bourbaki, \emph{Alg.\ comm.}, chap.~III, \textsection~5, n\textsuperscript{o}~4, cor.\ de la prop.~3), which is moreover a local homomorphism, hence faithfully flat $(\mathbf{0_I}$, 6.6.2); it follows then from $(\mathrm{P_{II}})$ that $g$ is a $\mathbf{P}$-morphism.
\end{proof}

\begin{corollary}[7.4.3]
\label{IV.7.4.3}
\begin{enumerate}
  \item[\rm(i)] Let $A$ be a $\mathbf{P}$-ring local Noetherian, $A'=\widehat{A}$ its completion.
  Suppose that for every prime ideal $\mathfrak{p}'$ of $A'$, the formal fibres of $A'_{\mathfrak{p}'}$ are geometrically regular; then, for every prime ideal $\mathfrak{p}$ of $A$, $A_\mathfrak{p}$ is a $\mathbf{P}$-ring.
  \item[\rm(ii)] Suppose that $\mathbf{P}$ satisfies condition $(\mathrm{P_{IV}})$ \sref{IV.7.3.6}.
  Let $A$ be a $\mathbf{P}$-ring local Noetherian, $B$ a local $A$-algebra essentially of finite type, and such that the homomorphism $A\to B$ is local.
  Set $A'=\widehat{A}$, and let $\mathfrak{n}'$ be the unique prime ideal of $B'=B\otimes_A A'$ above the maximal ideal of $B$ and the maximal ideal of $A'$.
  If the formal fibres of $B'_{\mathfrak{n}'}$ are geometrically regular, then $B$ is a $\mathbf{P}$-ring.
\end{enumerate}
\end{corollary}

\begin{theorem}[7.4.4]
\label{IV.7.4.4}
The hypotheses on $\mathbf{P}$ being those of \sref{IV.7.4.1}, let $A$ be a $\mathbf{P}$-ring local Noetherian.
\begin{enumerate}
  \item[\rm(i)] For every prime ideal $\mathfrak{p}$ of $A$, $A_\mathfrak{p}$ is a $\mathbf{P}$-ring.
  \item[\rm(ii)] Suppose moreover that $\mathbf{P}$ satisfies condition $(\mathrm{P_{IV}})$.
  Then every local ring which is an $A$-algebra essentially of finite type is a $\mathbf{P}$-ring.
\end{enumerate}
\end{theorem}

\begin{corollary}[7.4.5]
\label{IV.7.4.5}
Suppose that the property $\mathbf{P}$ satisfies conditions $(\mathrm{P_I})$, $(\mathrm{P_{II}})$, $(\mathrm{P_{III}})$.
The following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] For every prime ideal $\mathfrak{p}$ of $A$, $A_\mathfrak{p}$ is a $\mathbf{P}$-ring.
  \item[\rm b)] For every maximal ideal $\mathfrak{m}$ of $A$, $A_\mathfrak{m}$ is a $\mathbf{P}$-ring.
\end{enumerate}
If moreover $\mathbf{P}$ satisfies $(\mathrm{P_{IV}})$, properties $a)$ and $b)$ are also equivalent to:
\begin{enumerate}
  \item[\rm c)] For every $A$-algebra of finite type $B$ and every prime ideal $\mathfrak{q}$ of $B$, $B_\mathfrak{q}$ is a $\mathbf{P}$-ring.
\end{enumerate}
The equivalence of $a)$ and $b)$ follows from \sref{IV.7.4.4}, (i)), and that of $a)$ and $c)$ from \sref{IV.7.4.4}, (ii).
When condition $a)$ of \sref{IV.7.4.5} is satisfied, for a semi-local Noetherian ring, this definition coincides with that of \sref{IV.7.3.13}.
The ring $\bb{Z}$ is a $\mathbf{P}$-ring when conditions $(\mathrm{P_I})$, $(\mathrm{P_{II}})$ and $(\mathrm{P_{III}})$ are satisfied.
The ring $\bb{Z}$ is a $\mathbf{P}$-ring \sref{IV.7.3.19}, (iv)); every local Noetherian complete ring is a $\mathbf{P}$-ring.
\end{corollary}

\begin{proposition}[7.4.6]
\label{IV.7.4.6}
Suppose that the property $\mathbf{P}$ satisfies conditions $(\mathrm{P_I})$, $(\mathrm{P_{II}})$ and $(\mathrm{P_{III}})$.
Let $A$ be a Noetherian ring, $\mathfrak{J}$ an ideal of $A$, $\widehat{A}$ the separated completion of $A$ for the $\mathfrak{J}$-preadic topology.
Then, if $A$ is a $\mathbf{P}$-ring \sref{IV.7.4.5}, the canonical morphism $\Spec(\widehat{A})\to\Spec(A)$ is a $\mathbf{P}$-morphism.
\end{proposition}

\begin{corollary}[7.4.7]
\label{IV.7.4.7}
Suppose that $\mathbf{P}$ satisfies conditions $(\mathrm{P_I})$, $(\mathrm{P'_I})$, $(\mathrm{P_{II}})$, $(\mathrm{P_{III}})$ and $(\mathrm{P_{IV}})$.
Then, if $A$ is a $\mathbf{P}$-ring \sref{IV.7.4.5}, the canonical morphism $\Spec(A[[T_1,\ldots,T_r]])\to\Spec(A)$ is a $\mathbf{P}$-morphism.
In particular, if moreover $A$ is integral, $K$ its field of fractions, and if $\mathfrak{p}$ is a prime ideal of $B=A[[T_1,\ldots,T_r]]$ such that $\mathfrak{p}\cap A=0$, then the property $\mathbf{P}(B_\mathfrak{p},K)$ is true.
\end{corollary}

\begin{proof}
Indeed, the canonical morphism $\Spec(B)\to\Spec(A)$ factorises as
\[
  \Spec(A[[T_1,\ldots,T_r]])\xrightarrow{f}\Spec(A[T_1,\ldots,T_r])\xrightarrow{g}\Spec(A).
\]
It is clear that the morphism $g$ is regular $(\mathbf{0}$, 17.3.7); by \sref{IV.7.4.5}, $A[T_1,\ldots,T_r]$ is a $\mathbf{P}$-ring, hence it follows from \sref{IV.7.4.6} that $f$ is a $\mathbf{P}$-morphism; as $g$ is regular, hence also a $\mathbf{P}$-morphism \sref{IV.7.3.5}, (i)), it is the same for $g\circ f$ by $(\mathrm{P'_I})$.
\end{proof}

\begin{remark}[7.4.8]
\label{IV.7.4.8}
The preceding results pose the following problems:
\begin{enumerate}
  \item[\rm A)] Let $A$ be a Zariski ring \emph{complete}, and let $\mathfrak{J}$ be an ideal of definition of $A$; if $A/\mathfrak{J}$ is a $\mathbf{P}$-ring, is it the same for $A$?
  \item[\rm B)] Let $k$ be a complete non-discrete valued field; we call the ring of \emph{restricted formal power series} $k\{T_1,\ldots,T_n\}$ the sub-ring of $k[[T_1,\ldots,T_n]]$ formed by the series whose coefficients \emph{tend to} $0$.
  Is such a ring a $\mathbf{P}$-ring?
  \item[\rm C)] If $A$ is a linearly topologised $\mathbf{P}$-ring, $S$ a multiplicative subset of $A$, are the rings $A\{S^{-1}\}$ $(\mathbf{0_I}$, 7.6.1) and $A_{\{S\}}$ $(\mathbf{0_I}$, 7.6.15) also $\mathbf{P}$-rings?
\end{enumerate}
\end{remark}

\subsection{A criterion for $\mathbf{P}$-morphisms}
\label{subsection:IV.7.5}

\begin{env}[7.5.0]
\label{IV.7.5.0}
\oldpage[IV-2]{203}
This number will not be used in the sequel of chapter IV, and can therefore be omitted on a first reading.
We shall see later (7.9.8) that the results of this number can be considerably improved when one has the ``resolution of singularities'' at one's disposal.

In the remainder of this number, we shall consider a property $\mathbf{R}(A)$, and we denote by $\mathbf{P}(Z,k)$ the property
\begin{quote}
``$Z$ is a locally Noetherian prescheme on a field $k$, and, for every \emph{finite} extension $k'$ of $k$, if one sets $Z'=Z\otimes_k k'$, the property $\mathbf{R}(\sh{O}_{z'})$ is true for every $z'\in Z'$.''
\end{quote}
\end{env}

\begin{theorem}[7.5.1]
\label{IV.7.5.1}
Let $A$, $B$ be two local Noetherian complete rings, $\mathfrak{m}$ the maximal ideal of $A$, $k=A/\mathfrak{m}$ its residue field, $\varphi:A\to B$ a local homomorphism such that:
\begin{enumerate}
  \item[\rm(i)] The residue field of $B$ is a finite extension of $k$.
  \item[\rm(ii)] $B$ is a flat $A$-module.
\end{enumerate}
Let moreover $\mathbf{R}(C)$ be a property satisfying condition $(\mathrm{R_{III}})$ \sref{IV.7.3.18} and the following condition:
\begin{itemize}
  \item[$(\mathrm{R_{IV}})$] For every local ring $C$ in a local Noetherian complete ring and every $C$-regular element $t$ above the maximal ideal of $C$, the property $\mathbf{R}(C/tC)$ implies $\mathbf{R}(C)$.
\end{itemize}
Then, for every ideal $\mathfrak{p}$ of $A$, $\mathbf{R}(\widehat{B}_\mathfrak{q})$ is true for every prime ideal $\mathfrak{q}$ of $\widehat{B}$ above $\mathfrak{p}$, if and only if $\mathbf{R}(\widehat{A}_{\mathfrak{p}'})$ is true for every prime ideal $\mathfrak{p}'$ of $\widehat{A}$ above $\mathfrak{p}$ and the fibre of the morphism $\Spec(B)\to\Spec(A)$ over $\mathfrak{p}$ satisfies $\mathbf{P}$.
\end{theorem}

\subsection{Two applications to the finiteness of integral closure}
\label{subsection:IV.7.6}

\begin{env}[7.6.1]
\label{IV.7.6.1}
Let $A$ be a local Noetherian integral ring.
We shall say (following Nagata) that $A$ is a $\mathrm{N}$\emph{-1 ring} (or a $(\mathrm{N}_1)$\emph{-ring}) if its integral closure $A'$ is a \emph{finite} $A$-module.
We shall say that $A$ is a $\mathrm{N}$\emph{-2 ring} (or a $(\mathrm{N}_2)$\emph{-ring}, or a \emph{Japanese ring}) if, for every extension $K'$ of finite degree of the field of fractions $K$ of $A$, the integral closure $A'$ of $A$ in $K'$ is a finite $A$-module.
\end{env}

\begin{proposition}[7.6.2]
\label{IV.7.6.2}
Let $A$ be a local Noetherian integral ring whose formal fibres are geometrically reduced.
Then $A$ is a $(\mathrm{N}_1)$-ring.
\end{proposition}

\begin{proposition}[7.6.3]
\label{IV.7.6.3}
Let $A$ be a local Noetherian integral ring whose formal fibres are geometrically normal.
Then $A$ is an $(\mathrm{N}_2)$-ring.
\end{proposition}

\begin{corollary}[7.6.4]
\label{IV.7.6.4}
Every local Noetherian \emph{complete} integral ring is an $(\mathrm{N}_2)$-ring (i.e.\ \emph{Japanese}).
\end{corollary}

\begin{proof}
Indeed, the formal fibres of a complete local ring are geometrically regular \sref{IV.7.3.19}, (iii)), hence \emph{a fortiori} geometrically normal.
\end{proof}

\begin{corollary}[7.6.5]
\label{IV.7.6.5}
Let $A$ be a Noetherian ring whose formal fibres are geometrically normal (in the sense of \sref{IV.7.4.5}).
Then $A$ is \emph{universally Japanese} $(\textbf{0}$, 23.1.1).
\end{corollary}

\begin{proposition}[7.6.6]
\label{IV.7.6.6}
Let $A$ be a reduced local Noetherian ring, $A'$ its normalisation, $f:X'=\Spec(A')\to X=\Spec(A)$ the corresponding morphism.
Let $\sh{F}$ be a coherent $\sh{O}_X$-Module, $U$ the complement of the closed point $a$ in $X$, $U'=f^{-1}(U)$, $\sh{F}'=f^*(\sh{F})$; suppose that, for every point $x\in U'$, the $\sh{O}_{x}$-module $\sh{F}'_x=\sh{F}_x\otimes A'_x$ is \emph{reflexive} (i.e.\ that $\sh{F}'$ is reflexive on $U'$).
Then the canonical homomorphism $\sh{F}\to f_*(f^*(\sh{F}))$ is bijective.
\end{proposition}

\begin{corollary}[7.6.7]
\label{IV.7.6.7}
Under the hypotheses of \sref{IV.7.6.6}, the canonical homomorphism $A\to\Gamma(U',\sh{O}_{X'})$ is bijective.
Moreover, if $A$ is integral and $A'$ is finite over $A$, the normalisation $A'$ of $A$ identifies to $\Gamma(U,\sh{O}_X)$.
\end{corollary}

\subsection{Nagata rings}
\label{subsection:IV.7.7}

\begin{definition}[7.7.1]
\label{IV.7.7.1}
We say that a local Noetherian ring $A$ is a \emph{Nagata ring} if the following two conditions are satisfied:
\begin{enumerate}
  \item[\rm(i)] $A$ is universally Japanese $(\mathbf{0}$, 23.1.1).
  \item[\rm(ii)] For every prime ideal $\mathfrak{p}$ of $A$ and every finite extension of the field of fractions of $A/\mathfrak{p}$, the integral closure of $A/\mathfrak{p}$ in this extension has an open regular locus.
\end{enumerate}
A Noetherian ring $A$ is said to be a \emph{Nagata ring} if, for every prime ideal $\mathfrak{p}$ of $A$, $A_\mathfrak{p}$ is a Nagata ring in the above sense.
\end{definition}

\begin{proposition}[7.7.2]
\label{IV.7.7.2}
A local Noetherian ring $A$ whose formal fibres are geometrically regular is a Nagata ring.
In particular, every complete local Noetherian ring is a Nagata ring.
\end{proposition}

\begin{proposition}[7.7.3]
\label{IV.7.7.3}
Let $A$ be a Nagata ring.
Then every $A$-algebra of finite type is a Nagata ring.
\end{proposition}

\subsection{Excellent rings}
\label{subsection:IV.7.8}

\begin{definition}[7.8.1]
\label{IV.7.8.1}
A Noetherian ring $A$ is said to be \emph{quasi-excellent} if it satisfies the following conditions:
\begin{enumerate}
  \item[\rm(i)] For every prime ideal $\mathfrak{p}$ of $A$, the formal fibres of $A_\mathfrak{p}$ are geometrically regular.
  \item[\rm(ii)] For every $A$-algebra of finite type $B$, $\mathrm{Reg}(B)$ is open in $\Spec(B)$.
\end{enumerate}
A Noetherian ring $A$ is said to be \emph{excellent} if it is quasi-excellent and universally catenary.
\end{definition}

\begin{proposition}[7.8.2]
\label{IV.7.8.2}
A Noetherian ring $A$ is quasi-excellent if and only if it satisfies the condition (i) of \sref{IV.7.8.1} and the following condition:
\begin{enumerate}
  \item[\rm(ii')] For every integral quotient ring $B$ of $A$, there exists a non-empty open subset of $\Spec(B)$ which is regular.
\end{enumerate}
\end{proposition}

\begin{theorem}[7.8.3]
\label{IV.7.8.3}
\begin{enumerate}
  \item[\rm(i)] Every quotient ring of a quasi-excellent (resp.\ excellent) ring is quasi-excellent (resp.\ excellent).
  \item[\rm(ii)] Every algebra of finite type over a quasi-excellent (resp.\ excellent) ring is quasi-excellent (resp.\ excellent).
  \item[\rm(iii)] Every ring of fractions of a quasi-excellent (resp.\ excellent) ring is quasi-excellent (resp.\ excellent).
  \item[\rm(iv)] If $A$ is a quasi-excellent (resp.\ excellent) local ring, its completion $\widehat{A}$ is quasi-excellent (resp.\ excellent).
\end{enumerate}
\end{theorem}

\begin{corollary}[7.8.4]
\label{IV.7.8.4}
Every complete Noetherian local ring is excellent.
The ring $\bb{Z}$ is excellent.
Every field is excellent.
Every Dedekind ring whose field of fractions is of characteristic $0$ is excellent.
\end{corollary}

\begin{corollary}[7.8.5]
\label{IV.7.8.5}
Every algebra of finite type over an excellent ring is excellent.
In particular, every algebra of finite type over $\bb{Z}$ or over a field is excellent.
\end{corollary}

\begin{corollary}[7.8.6]
\label{IV.7.8.6}
Let $A$ be an excellent ring.
Then $A$ is a Nagata ring \sref{IV.7.7.1}.
In particular, $A$ is universally Japanese.
\end{corollary}

\begin{proposition}[7.8.7]
\label{IV.7.8.7}
Let $A$ be a quasi-excellent ring.
Then, for every ideal $\mathfrak{J}$ of $A$, the $\mathfrak{J}$-adic completion $\widehat{A}$ is quasi-excellent.
\end{proposition}

\begin{remark}[7.8.8]
\label{IV.7.8.8}
Let $A$ be a quasi-excellent local ring, $\widehat{A}$ its completion.
Then the canonical morphism $\Spec(\widehat{A})\to\Spec(A)$ is a \emph{regular} morphism \sref{IV.6.8.1}.
This is one of the key properties of excellent rings.
\end{remark}

\subsection{Excellent rings and resolution of singularities}
\label{subsection:IV.7.9}

\oldpage[IV-2]{218}

\begin{env}[7.9.1]
\label{IV.7.9.1}
Given a reduced locally Noetherian prescheme $X$, we call a morphism $f:X'\to X$ a \emph{resolution morphism} for $X$ if $X'$ is regular and if $f$ is proper and birational.
When such a morphism exists, we say that one can \emph{resolve the singularities} of $X$ (or more simply that one can \emph{resolve} $X$).
For example, if $A$ is a Japanese ring of dimension $1$, one can resolve $X=\Spec(A)$, since the morphism $X'\to X$ (where $X'$ is the normalisation of $X$) is finite (hence proper) and birational, and the local rings of $X'$ are integrally closed rings of dimension $1$, hence discrete valuation rings, and therefore regular.
\end{env}

\begin{env}[7.9.2]
\label{IV.7.9.2}
It is clear that if one can resolve $X$, then one can also resolve every prescheme induced on an open subset of $X$ and every local prescheme $\Spec(\sh{O}_x)$ of $X$ \sref[II]{II.5.4.2}.
On the other hand, if one can resolve $X$, it is clear that there exists in $X$ an everywhere dense open subset contained in $\mathrm{Reg}(X)$.
These remarks show immediately that, if for every integral closed sub-prescheme $Y$ of $X$ and every integral $Y$-prescheme $Y'$ finite and radicial over $Y$ one can resolve $Y'$, then the affine open subsets of $X$ satisfy condition (iii) of \sref{IV.7.8.2}.
\end{env}

\begin{proposition}[7.9.3]
\label{IV.7.9.3}
Let $A$ be a reduced local Noetherian ring, and suppose that one can resolve $\Spec(A)$.
Then the fibres of the canonical morphism $\Spec(\widehat{A})\to\Spec(A)$ at the maximal points of $\Spec(A)$ are regular.
\end{proposition}

\begin{proof}
Set $X=\Spec(A)$, $X'=\Spec(\widehat{A})$, and let $g:X'\to X$ be the canonical morphism.
Let $v:Y\to X$ be a resolution morphism; set $Y'=X'\times_X Y$, and let $f:Y'\to X'$ and $g':Y'\to Y$ be the canonical projections, so that we have a commutative diagram
\[
\begin{tikzcd}
  X' & Y' \arrow[l, "f"'] \arrow[d, "g'"] \\
  X \arrow[u, "g"] & Y \arrow[l, "v"] .
\end{tikzcd}
\]
It will suffice to prove that the prescheme $Y'$ is regular.
Indeed, as $v$ is birational and of finite type, there is an everywhere dense open subset $U$ of $X$ such that the restriction $v^{-1}(U)\to U$ of $v$ is an isomorphism and such that $v^{-1}(U)$ is everywhere dense in $Y$ \sref[I]{I.6.5.5};
the preschemes induced respectively on the open subset $g^{-1}(U)$ of $X'$ and on the open subset ${g'}^{-1}(v^{-1}(U))$ of $Y'$ are therefore isomorphic.
This proves that $g^{-1}(U)$ is regular, and \emph{a fortiori} so are the fibres of $g$ at the maximal points of $X$, which are contained in $U$.

\oldpage[IV-2]{219}

Note that the morphism $f$ is proper \sref[II]{II.5.4.2}, and in particular of finite type, hence $Y'$ is Noetherian, since $X'$ is.
Let $a'$ be the closed point of $X'$.
To see that $Y'$ is regular, it will suffice to prove that, for every $y'\in f^{-1}(a')$, the local ring $\sh{O}_{Y',y'}$ is regular.
Indeed, then $f^{-1}(a')\subset\mathrm{Reg}(Y')$.
But since $X'$ is the spectrum of a complete local ring and $f$ is of finite type, $\mathrm{Reg}(Y')$ is open in $Y'$ \sref{IV.6.12.8};
as $f$ is closed and $f^{-1}(a')\subset\mathrm{Reg}(Y')$, there exists a neighbourhood $V'$ of $a'$ in $X'$ such that $f^{-1}(V')\subset\mathrm{Reg}(Y')$.
Since $A$ is a local ring, necessarily $V'=X'$, hence $\mathrm{Reg}(Y')=Y'$, and the proposition will be proved.

Now if $a$ is the closed point of $X$, then $g^{-1}(a)=\{a'\}$ and the residue fields $\kres(a)$ and $\kres(a')$ are isomorphic;
therefore ${g'}^{-1}(v^{-1}(a))=f^{-1}(a')$, and the restriction $f^{-1}(a')\to v^{-1}(a)$ of $g'$ is an isomorphism of these two fibres.
Let $y=g'(y')$ and set $B=\sh{O}_{Y,y}$; if $\mathfrak{n}$ is the maximal ideal of $B$, what precedes shows that in the Noetherian ring $B'=B\otimes_A\widehat{A}$ there is a unique maximal ideal $\mathfrak{n}'$ above $\mathfrak{n}$ and that $\sh{O}_{Y',y'}=B'_{\mathfrak{n}'}$.
To prove that the local ring $B'_{\mathfrak{n}'}$ is regular, it suffices to show that its completion \sref[0]{0.17.1.5} is regular;
but, by hypothesis, $B$ is regular, hence so is its completion $\widehat{B}$.
The conclusion therefore follows from the following lemma.
\end{proof}

\begin{lemma}[7.9.3.1]
\label{IV.7.9.3.1}
Let $A$, $B$ be two local Noetherian rings, and let $\rho:A\to B$ be a local homomorphism such that the ring $B'=B\otimes_A\widehat{A}$ is Noetherian.
Then, for every maximal ideal $\mathfrak{n}'$ of $B'$ above the maximal ideal $\mathfrak{n}$ of $B$ and the maximal ideal of $\widehat{A}$, the completions of $B$ and of $B'_{\mathfrak{n}'}$ are isomorphic.
\end{lemma}

\begin{proof}
Let $\mathfrak{m}$ be the maximal ideal of $A$ and set $B''=B'_{\mathfrak{n}'}$.
For every integer $h>0$, one has $B'/\mathfrak{m}^hB'=B\otimes_A(\widehat{A}/\mathfrak{m}^h\widehat{A})$;
but $\widehat{A}/\mathfrak{m}^h\widehat{A}$ is isomorphic to $A/\mathfrak{m}^h$, so $B'/\mathfrak{m}^hB'$ is isomorphic to $B/\mathfrak{m}^hB$, and in particular is a local ring whose maximal ideal is $\mathfrak{n}'/\mathfrak{m}^hB'$.
It follows that $B''/\mathfrak{m}^hB''$, isomorphic to $(B'/\mathfrak{m}^hB')_{\mathfrak{n}'}$, is $B$-isomorphic to $B'/\mathfrak{m}^hB'$, and finally to $B/\mathfrak{m}^hB$.
In particular, the maximal ideal of $B''$ is equal to $\mathfrak{n}B''$, and $B''/\mathfrak{n}^hB''$ is isomorphic to $B/\mathfrak{n}^h$ by the preceding argument.
Since $\widehat{B''}=\varprojlim(B''/\mathfrak{n}^hB'')$, the lemma is proved, and so is \sref{IV.7.9.3}.
\end{proof}

\begin{corollary}[7.9.4]
\label{IV.7.9.4}
Let $A$ be a local Noetherian ring.
\begin{enumerate}
  \item[\rm(i)] If, for every integral quotient ring $B$ of $A$, one can resolve $\Spec(B)$, then the formal fibres of $A$ are regular.
  \item[\rm(ii)] Suppose that, for every integral quotient ring $B$ of $A$, and every integral ring $B'$ containing $B$, finite as a $B$-algebra and whose field of fractions is radicial over that of $B$, one can resolve $\Spec(B')$; then the formal fibres of $A$ are geometrically regular.
\end{enumerate}
\end{corollary}

\begin{proof}
Every formal fibre of $A$ being a formal fibre of an integral quotient ring of $A$ at the generic point of its spectrum, it is clear that (i) is an immediate consequence of \sref{IV.7.9.3}, and (ii) follows from (i) and from \sref{IV.6.7.7}.
\end{proof}

\begin{proposition}[7.9.5]
\label{IV.7.9.5}
Let $X$ be a locally Noetherian prescheme such that, for every integral prescheme $Y$ finite over $X$, one can resolve $Y$.
Then the rings of the affine open subsets of $X$ satisfy conditions (ii) and (iii) of \sref{IV.7.8.2};
if moreover $X$ is universally catenary \sref{IV.5.6.3} (and in particular if $X$ is locally immersible in a regular prescheme \sref{IV.5.6.4}), then $X$ is an excellent prescheme \sref{IV.7.8.5}.
\end{proposition}

\begin{proof}
This follows immediately from \sref{IV.7.9.4} and \sref{IV.7.9.2}.
\end{proof}

\begin{remark}[7.9.6]
\label{IV.7.9.6}
It is possible that the converse of \sref{IV.7.9.5} is true, that is, that the hypothesis that $X$ is reduced and that the rings of the affine open subsets of $X$ satisfy conditions (ii) and (iii) of \sref{IV.7.8.2} implies the possibility of resolving $X$ (and hence also the possibility of resolving every reduced prescheme locally of finite type over $X$, by \sref{IV.7.4.4} and \sref{IV.6.12.4}).
This is true, at least, if one restricts to reduced Noetherian preschemes whose residue fields have characteristic $0$, as shown by recent results of Hironaka [35].
By contrast, in the general case, one so far knows how to resolve only algebraic schemes of dimension $2$ over a perfect field.
In view of the importance that resolution of singularities is coming to have in the topological study of schemes (notably as regards their homological and homotopical properties), this gives a special interest to the category of excellent rings or excellent preschemes.

One may also note in this connection that the less delicate part of Hironaka's arguments shows that, independently of any hypothesis on the characteristics of the residue fields, and using only properties (ii) and (iii) of \sref{IV.7.8.2} for the local rings of a prescheme $X$, the problem of resolving the singularities of $X$ reduces to the case where $X=\Spec(A)$, $A$ being a complete local Noetherian integral ring.
Consequently, if later results were to disprove the conjecture advanced above, and were therefore to lead to the formulation of more restrictive conditions on the local rings of $X$, this could hardly happen except through restrictive conditions imposed on complete local rings (probably conditions concerning their residue fields, perhaps the condition $[k:k^p]+0$ when $p$ is the characteristic exponent of $k$).
\end{remark}

\begin{env}[7.9.7]
\label{IV.7.9.7}
Consider, on the one hand, a full subcategory $\mathcal{C}$ of the category of locally Noetherian preschemes and, on the other hand, a property $\mathbf{R}(A)$, subject to the following conditions:
\begin{enumerate}
  \item[1\textsuperscript{o}] For every $X\in\mathcal{C}$, every prescheme locally of finite type over $X$ belongs to $\mathcal{C}$.
  For every Noetherian ring $A$ such that $\Spec(A)\in\mathcal{C}$ and every multiplicative subset $S$ of $A$, one has $\Spec(S^{-1}A)\in\mathcal{C}$.
  \item[2\textsuperscript{o}] For every $X\in\mathcal{C}$, the set $\mathrm{U}_{\mathbf{R}}(X)$ of $x\in X$ such that $\mathbf{R}(\sh{O}_{X,x})$ is true is open in $X$.
  \item[3\textsuperscript{o}] For every local Noetherian ring $A$ such that $\Spec(A)\in\mathcal{C}$ and every regular element $t$ in the maximal ideal of $A$, $\mathbf{R}(A/tA)$ implies $\mathbf{R}(A)$.
\end{enumerate}
As in \sref{IV.7.5.0}, denote by $\mathbf{P}(Z,k)$, for a field $k$ and a $k$-prescheme $Z\in\mathcal{C}$, the following property:
\begin{quote}
``For every finite extension $k'$ of $k$, all the local rings of $Z\otimes_k k'$ satisfy the property $\mathbf{R}$.'' 
\end{quote}
We shall further suppose that $\mathbf{R}$ satisfies the following condition:
\begin{enumerate}
  \item[4\textsuperscript{o}] If $\mathbf{P}(Z,k)$ is true, then, for every extension $k''$ of finite type of $k$, all the local rings of $Z\otimes_k k''$ satisfy the property $\mathbf{R}$.
\end{enumerate}
Note that these conditions are satisfied when one takes for $\mathcal{C}$ the category of excellent preschemes and for $\mathbf{R}$ one of the properties (i) to (viii) of \sref{IV.7.5.3}.
For condition 1\textsuperscript{o}, this follows from \sref{IV.7.8.3}, (ii), and for conditions 2\textsuperscript{o} and 3\textsuperscript{o}, from the arguments of \sref{IV.7.5.3}, taking into account that every excellent ring is catenary.
Finally, condition 4\textsuperscript{o} follows, for (i), (ii) and (iii), from \sref{IV.6.7.1}; for (iv), (v), (vi), from \sref{IV.6.7.7}; and for (viii), from \sref{IV.4.6.1}.
As regards property (vii) of \sref{IV.7.5.3}, the corresponding property $\mathbf{P}(Z,k)$ implies that $Z$ is locally integral (being locally Noetherian) and that each of the integral sub-preschemes of which $Z$ is the sum is geometrically integral, by \sref{IV.4.5.9} and \sref{IV.4.6.1};
one again concludes condition 4\textsuperscript{o} above in this case.
\end{env}

\begin{proposition}[7.9.8]
\label{IV.7.9.8}
Let $A$ be a local Noetherian ring, $k$ its residue field, $B$ a local Noetherian ring such that $\Spec(B)\in\mathcal{C}$, and $\varphi:A\to B$ a local homomorphism making $B$ a flat $A$-module.
Set $Y=\Spec(A)$, $X=\Spec(B)$, and let $f:X\to Y$ be the morphism corresponding to $\varphi$.
Suppose that:
\begin{enumerate}
  \item[1\textsuperscript{o}] the property $\mathbf{P}(\Spec(B\otimes_A k),k)$ is true;
  \item[2\textsuperscript{o}] for every finite morphism $Y_1\to Y$, one can resolve $(Y_1)_{\mathrm{red}}$.
\end{enumerate}
Then, for every $y\in Y$, the property $\mathbf{P}(f^{-1}(y),k(y))$ is true.
\end{proposition}

\begin{proof}
Note that condition 2\textsuperscript{o} of \sref{IV.7.9.8} is still satisfied when one replaces $Y$ by the spectrum of a local ring at a maximal ideal of a finite $A$-algebra $A'$ \sref{IV.7.9.2};
on the other hand, by condition 1\textsuperscript{o} of \sref{IV.7.9.7}, the spectrum of a ring of fractions of $B\otimes_A A'$ also belongs to $\mathcal{C}$.
The lemma \sref{IV.7.3.16.2} then shows (as in part I) of the proof of \sref{IV.7.5.2}) that it suffices to prove that, if $Y$ is integral and if $y$ is the generic point of $Y$, the local rings of the fibre $f^{-1}(y)$ satisfy $\mathbf{R}$.

This being so, by hypothesis there exists a regular prescheme $Y'$ and a proper birational morphism $g:Y'\to Y$.
As $Y'$ is Noetherian and locally integral, the hypothesis that $g$ is birational implies that $Y'$ is integral.
Let $X'=X\times_Y Y'$, so that we have a commutative diagram
\[
\label{IV.7.9.8.1}
\begin{tikzcd}
  X & X' \arrow[l, "g'"'] \arrow[d, "f'"] \\
  Y & Y' \arrow[l, "g"] .
\end{tikzcd}
\tag{7.9.8.1}
\]
where $f'$ and $g'$ are the canonical projections.
Taking condition 2\textsuperscript{o} of \sref{IV.7.9.7} into account, the same argument as at the beginning of \sref{IV.7.9.3} shows that it is enough to prove that $\mathrm{U}_{\mathbf{R}}(X')=X'$, then, denoting by $a$ the closed point of $X$, that ${g'}^{-1}(a)\subset\mathrm{U}_{\mathbf{R}}(X')$.

\oldpage[IV-2]{222}

Let $x'\in {g'}^{-1}(a)$, and set $y'=f'(x')$, so that $b=g(y')$ is the closed point of $Y$.
Then ${f'}^{-1}(y')=f^{-1}(b)\otimes_k k(y')$, and as $k(y')$ is an extension of finite type of $k$ (since $g$ is proper, hence of finite type), the hypothesis that $\mathbf{P}(f^{-1}(b),k)$ is true implies, by condition 4\textsuperscript{o} of \sref{IV.7.9.7}, that the same is true of $\mathbf{P}({f'}^{-1}(y'),k(y'))$.
But since $\sh{O}_{Y',y'}$ is a regular ring and $\Spec(\sh{O}_{X',x'})\in\mathcal{C}$ by condition 1\textsuperscript{o} of \sref{IV.7.9.7}, the lemma \sref{IV.7.5.1.1} proves that $\mathbf{R}(\sh{O}_{X',x'})$ is true, which finishes the proof.
\end{proof}

\begin{corollary}[7.9.9]
\label{IV.7.9.9}
Let $A$, $B$ be two local Noetherian rings, and let $\varphi:A\to B$ be a local homomorphism making $B$ a flat $A$-module.
Suppose that $A$ is integral and geometrically unibranch, and that for every finite integral $A$-algebra $A'$ one can resolve $\Spec(A')$ (which will be the case, for example, if the formal fibres of $A$ are geometrically regular and the residue field of $A$ has characteristic $0$, by Hironaka's results).
Let $k$ be the residue field of $A$, and suppose that $\Spec(B\otimes_A k)$ is geometrically pointwise integral \sref{IV.4.6.9}.
Then $B$ is integral.
\end{corollary}

\begin{proof}
We apply \sref{IV.7.9.8}, taking for $\mathcal{C}$ the category of all locally Noetherian preschemes and for $\mathbf{R}$ the property of being integral.
The arguments of \sref{IV.7.5.3} and \sref{IV.7.9.7} show that conditions 1\textsuperscript{o}, 2\textsuperscript{o}, 3\textsuperscript{o} and 4\textsuperscript{o} of \sref{IV.7.9.7} are satisfied in this case.
The hypothesis on $B\otimes_A k$ and the proposition \sref{IV.7.9.8} show (with the notation of \sref{IV.7.9.8}) that the fibres $f^{-1}(y)$ are geometrically pointwise integral for $y\in Y$.
\emph{A fortiori} the $f^{-1}(y)$ for $y\in Y$ are reduced preschemes, and since $A$ is integral (hence \emph{a fortiori} reduced), we already see that $B$ is reduced \sref{IV.3.3.5}.
It remains to prove that $X=\Spec(B)$ is irreducible.
But the proof of \sref{IV.7.9.8} proves that $X'$ is locally integral (being locally Noetherian).
As the morphism $g':X'\to X$ is surjective, it is enough to see that $X'$ is irreducible, or (since $X'$ is locally integral) that $X'$ is connected.
For this it suffices to prove that the fibre ${g'}^{-1}(a')$ is connected.
Indeed, if this point is established, $X'$ cannot be the sum of two non-empty open preschemes $X'_1$, $X'_2$, for then one would have, say, ${g'}^{-1}(a)\cap X'_1=\varnothing$;
but the restriction $X'_1\to X$ of $g'$ is proper (since $X'_1$ is a closed sub-prescheme of $X'$), and $g'(X'_1)$ would be a non-empty closed subset of $X$ not containing the closed point $a$, which is absurd.
Now ${g'}^{-1}(a)=g^{-1}(b)\otimes_k k(a)$;
but $g:Y'\to Y$ is proper and birational, so, in the Stein factorisation $Y'\to Y''\to Y$ of $g$ \sref[III]{III.4.3.3}, the finite morphism $Y''\to Y$ is also birational and therefore $Y''$ is integral.
Since $A$ is assumed geometrically unibranch, it follows from \sref[III]{III.4.3.4} that $g^{-1}(b)$ is geometrically connected, which finishes the proof.
\end{proof}

\begin{remarks}[7.9.10]
\label{IV.7.9.10}
\begin{enumerate}
  \item[\rm(i)] We shall show later \sref{IV.18.8.11} that, for a reduced local ring $B$ to be geometrically unibranch, it is necessary and sufficient that, for every \'etale morphism $h:X'\to X=\Spec(B)$ and every point $x'\in X'$ above the closed point of $X$, $\sh{O}_{X',x'}$ be integral.
  It follows that if, in \sref{IV.7.9.9}, one supposes not only that $\Spec(B\otimes_A k)$ is geometrically pointwise integral, but geometrically unibranch, then one can conclude that $B$ is geometrically unibranch.
  \item[\rm(ii)] Let $X$, $Y$ be two excellent preschemes, let $f:X\to Y$ be a flat morphism, and suppose that, for every finite morphism $Y_1\to Y$, one can resolve $(Y_1)_{\mathrm{red}}$.
  It follows from \sref{IV.7.9.8}, taking for $\mathbf{R}$ the property of being regular, that the set $U$ of points $x\in X$ such that the fibre at the point $f(x)$ of the morphism
  \[
    \Spec(\sh{O}_{X,x}\otimes_{\sh{O}_{Y,f(x)}}k(f(x)))\to\Spec(k(f(x)))
  \]
  is geometrically regular is stable under generisation.
  We do not know whether this set is open (or, equivalently \sref[0]{0.9.2.5}, whether it is constructible), even in the particular case where $Y$ is the spectrum of $\bb{Z}$ or of a polynomial ring in one indeterminate $k[T]$ over a field $k$, and where consequently the ``resolvability'' condition imposed on $Y$ is trivially satisfied.
\end{enumerate}
\end{remarks}
