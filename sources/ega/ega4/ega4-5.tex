\section{Dimension, depth, and regularity in locally Noetherian preschemes}
\label{section:IV.5}

\oldpage[IV-2]{86}

This section merely restates, in geometric language, and with various complements of a technical nature, the notions and results of commutative Algebra exposed in chapter 0, \textsection\textsection16 and 17.

\subsection{Dimension of preschemes}
\label{subsection:IV.5.1}

\begin{env}[5.1.1]
\label{IV.5.1.1}
Let $A$ be a ring, $\mathfrak{J}$ an ideal of $A$.
Recall $(\mathbf{0}$, 16.1) that the definitions of the dimension of a ring $(\mathbf{0}$, 16.1.1) and those of the theory of combinatorial dimension of topological spaces $(\mathbf{0}$, 14.1.2) give the following relations:
\begin{align}
  \dim(\Spec(A))&=\dim(A)\label{IV.5.1.1.1}\tag{5.1.1.1}\\
  \dim(\mathrm{V}(\mathfrak{J}))&=\dim(A/\mathfrak{J})\label{IV.5.1.1.2}\tag{5.1.1.2}\\
  \codim(\mathrm{V}(\mathfrak{J}),\Spec(A))&=\operatorname{ht}(\mathfrak{J})\label{IV.5.1.1.3}\tag{5.1.1.3}
\end{align}
\begin{itemize}
  \item[\rm(5.1.1.4)] $\Spec(A)$ is a catenary space $\Leftrightarrow$ $A$ is a catenary ring.
  \item[\rm(5.1.1.5)] $\Spec(A)$ is equidimensional $\Leftrightarrow$ $A$ is equidimensional $\Leftrightarrow$ the minimal prime ideals $\mathfrak{p}_\alpha$ of $A$ are such that the dimensions of the rings $A/\mathfrak{p}_\alpha$ are all equal.
  \item[\rm(5.1.1.6)] $\Spec(A)$ is equicodimensional $\Leftrightarrow$ $A$ is equicodimensional $\Leftrightarrow$ all the maximal ideals have the same height.
\end{itemize}
Recall that a Noetherian ring $A$ is called \emph{biequidimensional} if $\Spec(A)$ is biequidimensional, that is if $\Spec(A)$ is Noetherian and $A$ is equidimensional, catenary and of finite dimension.
\end{env}

\begin{proposition}[5.1.2]
\label{IV.5.1.2}
Let $X$ be a prescheme, $Y$ a closed irreducible part of $X$, $y$ the generic point of $Y$.
We have
\begin{equation}
\label{IV.5.1.2.1}
  \codim(Y,X)=\dim(\sh{O}_{X,y}).\tag{5.1.2.1}
\end{equation}
\end{proposition}

\begin{proof}
Indeed \sref[I]{I.2.4.2} the closed irreducible parts of $X$ containing $y$ are canonically in bijective correspondence with the closed irreducible parts of $\Spec(\sh{O}_{X,y})$, hence in bijective correspondence with the prime ideals of $\sh{O}_{X,y}$, and \sref{IV.5.1.2.1} results from the definitions.
\end{proof}

In particular, one recovers the fact that the generic points of the irreducible components of $X$ are the \emph{only} points $x\in X$ such that $\dim(\sh{O}_x)=0$ \sref[I]{I.1.1.14}.

\oldpage[IV-2]{87}

\begin{corollary}[5.1.3]
\label{IV.5.1.3}
For every closed part $Y$ of a prescheme $X$, we have
\begin{equation}
\label{IV.5.1.3.1}
  \codim(Y,X)=\inf_{y\in Y}\dim(\sh{O}_{X,y}).\tag{5.1.3.1}
\end{equation}
Moreover, if $X$ is locally Noetherian, then for every $x\in Y$
\begin{equation}
\label{IV.5.1.3.2}
  \codim_x(Y,X)=\inf_{\substack{y\in Y,\, x\in\overline{\{y\}}}}\dim(\sh{O}_{X,y}).\tag{5.1.3.2}
\end{equation}
\end{corollary}

\begin{proof}
The relation \sref{IV.5.1.3.2} follows from \sref{IV.5.1.3.1} and $(\mathbf{0}$, 14.2.6).
\end{proof}

This corollary permits us to \emph{define}, for an \emph{arbitrary} part $Y$ of a prescheme $X$, the \emph{codimension} $\codim(Y,X)$ of $Y$ in $X$ as equal to the second member of \sref{IV.5.1.3.1}.

\begin{proposition}[5.1.4]
\label{IV.5.1.4}
For every prescheme $X$, we have
\begin{equation}
\label{IV.5.1.4.1}
  \dim(X)=\sup_{x\in X}(\dim(\sh{O}_x)).\tag{5.1.4.1}
\end{equation}
If every closed irreducible part of $X$ contains a closed point, we also have
\begin{equation}
\label{IV.5.1.4.2}
  \dim(X)=\sup_{x\in\mathrm{F}}(\dim(\sh{O}_x))\tag{5.1.4.2}
\end{equation}
where $\mathrm{F}$ is the set of \emph{closed} points of $X$.
\end{proposition}

\begin{proof}
It suffices to remark (by virtue of \sref[I]{I.2.4.2}) that the chains of closed irreducible parts of $X$ correspond bijectively to the chains of closed irreducible parts of a local scheme of $X$; when every closed irreducible part of $X$ contains a closed point, one can obviously restrict to the local schemes at the closed points of $X$.
\end{proof}

One will note that every closed irreducible part of $X$ contains a closed point when $X$ is \emph{quasi-compact} $(\mathbf{0_I}$, 2.1.3); we shall see a little further \sref{IV.5.1.11} that this also holds when $X$ is \emph{locally Noetherian}.

\begin{corollary}[5.1.5]
\label{IV.5.1.5}
For a prescheme $X$ to be catenary, it is necessary and sufficient that, for every $x\in X$, the local ring $\sh{O}_x$ be catenary.
If moreover every closed irreducible part of $X$ contains a closed point, for $X$ to be catenary, it suffices, for every \emph{closed} point $x$ of $X$, that $\sh{O}_x$ be catenary.
\end{corollary}

\begin{proof}
The reasoning is the same as in \sref{IV.5.1.4}, since it involves comparing chains of closed irreducible parts having the same endpoints.
\end{proof}

\begin{env}[5.1.6]
\label{IV.5.1.6}
We shall now examine properties more special linked to Noetherian hypotheses.

Recall $(\mathbf{0}$, 16.2.3) that a \emph{local Noetherian} ring $A\neq 0$ is of \emph{finite} dimension, equal to the minimum number of generators of an ideal of definition of $A$; for every prime ideal $\mathfrak{p}$ of $A$, the height of $\mathfrak{p}$, equal to $\dim(A_\mathfrak{p})$, is therefore also finite.
These properties, combined with \sref{IV.5.1.2}, \sref{IV.5.1.3} and \sref{IV.5.1.4}, show that:
\end{env}

\begin{proposition}[5.1.7]
\label{IV.5.1.7}
For every non-empty closed part $Y$ of a locally Noetherian prescheme $X$, $\codim(Y,X)$ is finite.
If $X$ is Noetherian and affine and $Y$ a closed irreducible part of $X$, $\codim(Y,X)$ is equal to the minimum number of sections $s_i$ of $\sh{O}_X$ above $X$ such that $Y$ is an irreducible component of the set of $x\in X$ such that $s_i(x)=0$ for every $i$.
\end{proposition}

\begin{corollary}[5.1.8]
\label{IV.5.1.8}
Let $X$ be a locally Noetherian prescheme, $\sh{L}$ an $\sh{O}_X$-Module invertible, $f$ a section of $\sh{L}$ above $X$.
Then every irreducible component of the set $Z$ of $x\in X$ such that $f(x)=0$ is of codimension $\leq 1$ in $X$; it is of codimension $1$ if $Z$ contains no irreducible component of $X$.
\end{corollary}

\oldpage[IV-2]{88}

\begin{proof}
One can restrict to the case where $\sh{L}=\sh{O}_X$.
If $y$ is the generic point of an irreducible component of $Z$, the ideal $(f_y)$ of $\sh{O}_{X,y}$ must be such that $\sh{O}_{X,y}/(f_y)$ has only a single prime ideal, which means that $f_y$ generates an ideal of definition of the local Noetherian ring $\sh{O}_{X,y}$; we therefore have $\codim(Y,X)\leq 1$ \sref{IV.5.1.7}; if $Z$ contains no irreducible component of $X$, we cannot have $\codim(Y,X)=0$ by virtue of $(\mathbf{0}$, 14.2.1).
\end{proof}

\begin{proposition}[5.1.9]
\label{IV.5.1.9}
Let $X$ be a locally Noetherian prescheme, $Y$ a closed part of $X$.
If $x\in Y$ is such that $\sh{O}_{X,x}$ is a catenary ring, we have
\begin{equation}
\label{IV.5.1.9.1}
  \codim_x(Y,X)=\dim(\sh{O}_{X,x})-\codim(\overline{\{x\}},Y)=\dim(\sh{O}_{X,x})-\dim(\sh{O}_{Y,x}).\tag{5.1.9.1}
\end{equation}
\end{proposition}

\begin{proof}
Let $Y_i$ ($1\leq i\leq n$) be the irreducible components of $Y$ containing $x$ (which are finite in number since $Y$ is locally Noetherian), and let $y_i$ be the generic point of $Y_i$.
Setting $A=\sh{O}_{X,x}$, and $\mathfrak{p}_i$ being the prime ideal of $A$ corresponding to $Y_i\cap\Spec(A)$, the closed irreducible parts of $Y$ containing $x$ correspond bijectively to the prime ideals of $A$ containing one of the $\mathfrak{p}_i$, hence $\dim(\sh{O}_{Y,x})=\sup_i\dim(A/\mathfrak{p}_i)$; on the other hand, $\sh{O}_{X,y_i}=A_{\mathfrak{p}_i}$, and the hypothesis on $A$ implies $\dim(A)=\dim(A/\mathfrak{p}_i)+\dim(A_{\mathfrak{p}_i})$ $(\mathbf{0}$, 16.1.4); the conclusion results therefore from the relation $\codim_x(Y,X)=\inf_i(\dim(\sh{O}_{X,y_i}))$ (\sref{IV.5.1.2} and $(\mathbf{0}$, 14.2.6)).
\end{proof}

\begin{proposition}[5.1.10]
\label{IV.5.1.10}
\begin{enumerate}[\rm(i)]
  \item In every prescheme $X$, every locally constructible non-empty part $E$ contains a point $x$ such that $\{x\}$ is a locally closed part of $X$ \emph{(}or, which amounts to the same, such that $x$ is isolated in $\overline{\{x\}}$\emph{)}.
  \item Let $X$ be a locally Noetherian prescheme, $x$ a point of $X$ such that $\{x\}$ is locally closed in $X$; then $\dim(\overline{\{x\}})\leq 1$; every point $y\neq x$ of $\overline{\{x\}}$ is therefore a closed point in $X$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) The result is a particular case of the following lemma:
\end{proof}

\begin{lemma}[5.1.10.1]
\label{IV.5.1.10.1}
Let $X$ be a topological space having the following property: for every locally closed non-empty part $Z\neq\emp$ of $X$, there exists a part $Z'$ of $Z$, locally closed in $X$ \emph{(}or in $Z$, which amounts to the same\emph{)}, and a point $x\in Z'$, closed in $Z'$.
Then every locally closed non-empty part $Z\neq\emp$ of $X$ contains a point $x$ isolated in $\overline{\{x\}}$.
\end{lemma}

\begin{proof}
Indeed, let $Z'\subset Z$ be a locally closed part containing a point $x$ such that $x$ be closed \emph{in} $Z'$.
There is a neighbourhood $U$ open of $x$ in $X$ such that $Z'\cap U$ is closed in $U$, hence $x$ is also closed in $U$; this means that $U\cap\overline{\{x\}}=\{x\}$, in other words $x$ is isolated in $\overline{\{x\}}$.
\end{proof}

The lemma applies to every space underlying a prescheme $X$, for $Z$ is then also an underlying space of a prescheme \sref[I]{I.5.2.1} and it suffices to take for $Z'$ an open affine \emph{in} $Z$, which is a Kolmogorov space quasi-compact, hence contains a closed point $(\mathbf{0_I}$, 2.1.3).

\oldpage[IV-2]{89}

(ii) Let $Z$ be the reduced subprescheme of $X$ having $\overline{\{x\}}$ as underlying space; the hypothesis implies that $\{x\}$ is open \emph{in} $Z$; hence, for every $z\in Z$, the generic point $x$ of $\Spec(\sh{O}_{Z,z})$ is \emph{isolated} in $\Spec(\sh{O}_{Z,z})$; but $A=\sh{O}_{Z,z}$ is a local Noetherian integral ring, and the hypothesis implies that there exists an $f\in A$ such that $A_f$ is a field; one knows $(\mathbf{0}$, 16.3.3) that this implies $\dim(A)\leq 1$.
Since $\dim(\sh{O}_{Z,z})\leq 1$ for every $z\in Z$, we have $\dim(Z)\leq 1$.

\begin{corollary}[5.1.11]
\label{IV.5.1.11}
If $X$ is a locally Noetherian prescheme non-empty and containing a closed point, every closed part of $X$ contains a closed point.
\end{corollary}

\begin{proof}
Indeed, every closed part of $X$ is constructible $(\mathbf{0_{III}}$, 9.1.1 and 9.1.5) and it suffices to apply \sref{IV.5.1.10}.
\end{proof}

\begin{env}[5.1.12]
\label{IV.5.1.12}
Let $X$ be a prescheme, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent \emph{of finite type}, $\mathrm{S}=\Supp(\sh{F})$ its support, which is \emph{closed} in $X$ $(\mathbf{0_I}$, 5.2.2).
If, for every $x\in X$, one considers $\Supp(\sh{F}_x)$ as a closed part of the local scheme $\Spec(\sh{O}_x)$, by definition $(\mathbf{0}$, 16.1.7) $\dim(\sh{F}_x)=\dim(\Supp(\sh{F}_x))$; but we have
\[
  \Supp(\sh{F}_x)=\mathrm{S}\cap\Spec(\sh{O}_{X,x})=\Spec(\sh{O}_{\mathrm{S},x})
\]
where, in the last term, $\mathrm{S}$ denotes any closed subprescheme of $X$ having $\mathrm{S}$ as underlying space.
Remarking that the irreducible components of $\Spec(\sh{O}_{X,x})$ and the irreducible components of $\Spec(\sh{O}_{\mathrm{S},x})$ are the intersections of the irreducible components of $\mathrm{S}$ containing $x$, and correspond bijectively to these latter, one sees that
\begin{equation}
\label{IV.5.1.12.1}
  \dim(\sh{F}_x)=\dim(\sh{O}_{\mathrm{S},x})\tag{5.1.12.1}
\end{equation}
by virtue of \sref{IV.5.1.1.1}; it also results from \sref{IV.5.1.2} that
\begin{equation}
\label{IV.5.1.12.2}
  \dim(\sh{F}_x)=\codim(\overline{\{x\}},\mathrm{S})\tag{5.1.12.2}
\end{equation}
when $X$ is locally Noetherian.

One says that $\sh{F}$ is \emph{equidimensional at $x\in X$} if $\sh{F}_x$ is an equidimensional $\sh{O}_{X,x}$-module $(\mathbf{0}$, 16.1.7), that is if $\Supp(\sh{F}_x)$ is equidimensional as a closed part of $\Spec(\sh{O}_x)$; this amounts to saying that the ring $\sh{O}_{\mathrm{S},x}$ is \emph{equidimensional}.
The \emph{dimension} of $\sh{F}$ is denoted $\dim(\sh{F})$ and means the dimension of the \emph{support} $\Supp(\sh{F})$; it results from \sref{IV.5.1.4} and \sref{IV.5.1.12.1} that
\begin{equation}
\label{IV.5.1.12.3}
  \dim(\sh{F})=\sup_{x\in X}\dim(\sh{F}_x).\tag{5.1.12.3}
\end{equation}

If $X=\Spec(A)$ is an affine scheme and $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module of finite type, one has $\dim(\sh{F})=\dim(M)$ by virtue of $(\mathbf{0}$, 16.1.7) and \sref{IV.5.1.4}.
\end{env}

\begin{proposition}[5.1.13]
\label{IV.5.1.13}
Let $X$ be a prescheme, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent of finite type, $x$ a point of $X$, $x'$ a generisation of $x$ in $X$; then
\begin{equation}
\label{IV.5.1.13.1}
  \dim(\sh{F}_{x'})\leq\dim(\sh{F}_x).\tag{5.1.13.1}
\end{equation}
\end{proposition}

\begin{proof}
This follows immediately from \sref{IV.5.1.12.1} and the definitions.
\end{proof}

\subsection{Dimension of an algebraic prescheme}
\label{subsection:IV.5.2}

\oldpage[IV-2]{90}

\begin{proposition}[5.2.1]
\label{IV.5.2.1}
Let $k$ be a field, $X$ an irreducible prescheme locally of finite type over $k$, $\xi$ its generic point.
Then $X$ is biequidimensional and we have $\dim(X)=\deg.\mathrm{tr}_k\,k(\xi)$.
\end{proposition}

\begin{proof}
Every local ring $\sh{O}_x$ is the local ring of a $k$-algebra of finite type, hence a quotient of a regular local ring $(\mathbf{0}$, 17.3.9), and one knows $(\mathbf{0}$, 16.5.12) that these rings are catenary; since $X$ is irreducible, \sref{IV.5.1.5} shows that $X$ is catenary, and it suffices to show that for every \emph{closed point} $x\in X$, we have
\begin{equation}
\label{IV.5.2.1.1}
  \dim(\sh{O}_x)=\deg.\mathrm{tr}_k\,k(\xi).\tag{5.2.1.1}
\end{equation}

One can obviously suppose $X$ reduced and affine, hence integral, with ring of fractions $A$, $k(\xi)=K$ the field of fractions of $A$.
One knows (Bourbaki, \emph{Alg.~comm.}, chap.~V, \textsection3, no.~1, th.~1) that there exists a sub-$k$-algebra $B=k[t_1,\ldots,t_n]$ of $A$, where the $t_i$ are algebraically independent over $k$, such that $A$ is a finite $B$-algebra.
Let $\mathfrak{m}=\mathfrak{i}_x$, which by hypothesis is a maximal ideal of $A$; $\mathfrak{n}=B\cap\mathfrak{m}$ is therefore also a maximal ideal of $B$ (Bourbaki, \emph{Alg.~comm.}, chap.~V, \textsection2, no.~1, prop.~1), and $A_\mathfrak{m}$ is a local ring of the $B_\mathfrak{n}$-algebra \emph{finite} $\mathrm{S}^{-1}A$, where $\mathrm{S}=B-\mathfrak{n}$; since $B_\mathfrak{n}$ is integrally closed and $\mathrm{S}^{-1}A$ integral, $\dim(A_\mathfrak{m})=\dim(B_\mathfrak{n})$ $(\mathbf{0}$, 16.1.6).
One can therefore restrict to the case where $A=k[t_1,\ldots,t_n]$; one knows then that $k'=k(x)$ is a \emph{finite} extension of $k$ \sref[I]{I.6.4.2}; let $A'=k'[t_1,\ldots,t_n]$; taking into account \sref[I]{I.2.4.6} and \sref{IV.3.3.14}, there is a maximal ideal $\mathfrak{m}'$ of $A'$ above $\mathfrak{m}$ and such that $k'$ is the residue field of $A'_{\mathfrak{m}'}$.
Since $A'$ is integral and is a finite $A$-algebra, the same reasoning as above shows $\dim(A'_{\mathfrak{m}'})=\dim(A_\mathfrak{m})$, so one is reduced to the case where $A/\mathfrak{m}=k$.
Now in this case $\mathfrak{m}$ is generated by the polynomials $t_i-a_i$ (where $a_i\in k$, $1\leq i\leq n$, the $a_i$ being the images of the $t_i$ in $A/\mathfrak{m}$).
Replacing $t_i$ by $t_i-a_i$, one sees that $\mathfrak{m}=\sum_{i=1}^n At_i$.
The completion of the local ring $A_\mathfrak{m}$ is then the ring of formal power series $k[[t_1,\ldots,t_n]]$; one knows that it has the same dimension as $A_\mathfrak{m}$, and on the other hand, the dimension of $k[[t_1,\ldots,t_n]]$ is equal to $n$ $(\mathbf{0}$, 17.1.4); whence the conclusion.
\end{proof}

\begin{corollary}[5.2.2]
\label{IV.5.2.2}
For a prescheme $X$ locally of finite type over a field $k$, $\dim(X)$ coincides with the number defined in \sref{IV.4.1.1}.
\end{corollary}

\begin{proof}
Indeed, if $X_\alpha$ are the reduced subpreschemes having as underlying spaces the irreducible components of $X$, we have $\dim(X)=\sup_\alpha(\dim(X_\alpha))$ $(\mathbf{0}$, 14.1.2.1) and it suffices to apply \sref{IV.5.2.1} to the $X_\alpha$.
\end{proof}

\begin{corollary}[5.2.3]
\label{IV.5.2.3}
Let $X$ be a prescheme locally of finite type over a field $k$, $x$ a point of $X$.
We have
\begin{equation}
\label{IV.5.2.3.1}
  \dim_x(X)=\dim(\sh{O}_x)+\deg.\mathrm{tr}_k\,k(x).\tag{5.2.3.1}
\end{equation}
\end{corollary}

\oldpage[IV-2]{91}

\begin{proof}
One knows that there exists an open neighbourhood $U$ of $x$ in $X$ such that $\dim_x(X)=\dim(U)$ $(\mathbf{0}$, 14.1.4.1), and one can suppose that the irreducible components of $U$ are the $X_i\cap U$, where the $X_i$ are the irreducible components of $X$ containing $x$; since $U\cap X_i$ is dense in $X_i$, it follows from \sref{IV.4.1.1.3} that $\dim(X_i)=\dim(U\cap X_i)$, hence $\dim_x(X)=\sup_i(\dim(X_i))$.
Moreover, the minimal prime ideals of $\sh{O}_x$ correspond to the generic points of the $X_i$, hence $(\mathbf{0}$, 16.1.1.1), $\dim(\sh{O}_x)=\sup_i(\dim(\sh{O}_{X_i,x}))$.
One is therefore reduced to the case where $X$ is irreducible; by \sref{IV.5.2.1}, $X$ is biequidimensional, hence $\dim(X)=\dim(\overline{\{x\}})+\codim(\overline{\{x\}},X)$, and one knows that $\dim(\overline{\{x\}})=\deg.\mathrm{tr}_k\,k(x)$ by \sref{IV.5.1.2} and $\codim(\overline{\{x\}},X)=\dim(\sh{O}_x)$ by \sref{IV.5.1.2}.
\end{proof}

\begin{corollary}[5.2.4]
\label{IV.5.2.4}
Let $X$ be a prescheme locally of finite type over a field $k$, $\sh{L}$ an $\sh{O}_X$-Module invertible, $f$ a section of $\sh{L}$ above $X$, $Y$ the set of $x\in X$ such that $f(x)=0$ and that is rare in $X$.
Then $\dim(Y)\leq\dim(X)-1$; the two members of this inequality are equal if $Y$ meets every irreducible component of maximal dimension of $X$.

If $(X_\alpha)$ is the family of irreducible components of $X$, we have
\[
  \dim(Y)=\sup_\alpha(\dim(Y\cap X_\alpha))
\]
and one is therefore reduced to the case where $X$ is irreducible ($Y\cap X_\alpha$ being rare in $X_\alpha$ since $X_\alpha$ has non-empty interior in $X$).
One can restrict to the case $Y\neq\emp$; then, for every maximal point $x$ of $Y$, we have (since $X$ is biequidimensional) $\dim(\overline{\{x\}})=\dim(X)-\codim(\overline{\{x\}},X)$, and since $Y$ is rare in $X$, we have $\codim(\overline{\{x\}},X)=1$ by \sref{IV.5.1.8}; whence the corollary.
\end{corollary}

\begin{remarks}[5.2.5]
\label{IV.5.2.5}
\begin{enumerate}[\rm(i)]
  \item Contrary to what occurs for algebraic $k$-preschemes, a locally Noetherian prescheme $X$ is not necessarily catenary (cf.\ \sref{IV.5.6.11}); it is however if all its local rings $\sh{O}_x$ are quotients of regular rings $(\mathbf{0}$, 16.5.12) and in particular if $X$ is \emph{regular} \sref[I]{I.4.1.4}.
  But even an integral scheme $X=\Spec(A)$, where $A$ is a regular ring, is not necessarily \emph{biequidimensional}, in other words the codimensions in $X$ of the various closed points of $X$ are not necessarily the same.
  \item When $X$ is a prescheme locally of finite type over a field $k$, one saw in \sref{IV.4.1.1.3} that $\dim(U)=\dim(X)$ for every open everywhere dense subset $U$ of $X$.
  This result does \emph{not extend} to Noetherian preschemes, even if they are biequidimensional.
\end{enumerate}
\end{remarks}

\subsection{Dimension of the support of a Module and Hilbert polynomial}
\label{subsection:IV.5.3}

\oldpage[IV-2]{92}

This section uses results from chapter III; it will not be used later in this chapter.

\begin{proposition}[5.3.1]
\label{IV.5.3.1}
Let $A$ be a local Artinian ring, $X$ a projective scheme over $\Spec(A)$, $\sh{L}$ an $\sh{O}_X$-Module invertible very ample relative to $A$, $\sh{F}$ a coherent $\sh{O}_X$-Module $\neq 0$; if we set $\sh{F}(n)=\sh{F}\otimes_{\sh{O}_X}\sh{L}^{\otimes n}$ for $n\in\bb{Z}$, the degree of the Hilbert polynomial $\mathrm{P}(n)=\chi_A(\sh{F}(n))$ of $\sh{F}$ with respect to $A$ (\textbf{III}, 2.5.3) is equal to the dimension of $\Supp(\sh{F})$.
\end{proposition}

\begin{proof}
We reason by induction on $d=\dim(\Supp(\sh{F}))$.
One knows that there exists a closed subprescheme $Y$ of $X$ whose $\Supp(\sh{F})$ is the underlying space, and an $\sh{O}_Y$-Module coherent $\sh{G}$ such that $\sh{F}=j_*(\sh{G})$, where $j:Y\to X$ is the canonical injection (\textbf{Err}$_\textbf{III}$, 30).
It is immediate that the Hilbert polynomials of $\sh{F}$ and of $\sh{G}$ are the same, so one can restrict to the case where $X=\Supp(\sh{F})$.
Suppose first $d=0$; all the points of $X$ are closed, $X$ is an Artinian scheme \sref[I]{I.6.2.2}, hence $\sh{F}(n)=\sh{F}$ for every integer $n$, and one has by \sref[III]{III.2.5.3}
\[
  \chi_A(\sh{F}(n))=\operatorname{long}_A(\Gamma(X,\sh{F}))
\]
for $n$ large enough, and the degree of the polynomial is therefore 0.
Suppose now $d>0$, and set $Z=\operatorname{Ass}(\sh{F})$, which is a finite set \sref{IV.3.1.6}; there exists an integer $m>0$ and a section $f\in\Gamma(X,\sh{L}^{\otimes m})$ such that $X_f$ is a neighbourhood of $Z$ (\textbf{II}, 4.5.4).
Multiplication by $f$ is a homomorphism
\[
  \mu_f:\sh{F}\to\sh{F}(m)
\]
which is injective \sref{IV.3.1.8}; one therefore has an exact sequence
\begin{equation}
\label{IV.5.3.1.1}
  0\to\sh{F}\xrightarrow{\mu_f}\sh{F}(m)\to\sh{G}\to 0\tag{5.3.1.1}
\end{equation}
where $\sh{G}$ is coherent.
By virtue of the Nakayama lemma, the points $x\in\Supp(\sh{G})$ are exactly those for which $f(x)=0$.
We shall now deduce that
\begin{equation}
\label{IV.5.3.1.2}
  \dim(\Supp(\sh{G}))=d-1.\tag{5.3.1.2}
\end{equation}
This will result from the following lemma:
\end{proof}

\begin{lemma}[5.3.1.3]
\label{IV.5.3.1.3}
Let $A$ be a local Artinian ring, $X$ a projective scheme over $\Spec(A)$, $\sh{L}$ an $\sh{O}_X$-Module invertible ample relative to $A$; then, for every section $g$ of $\sh{L}$ above $X$ such that $g(x)=0$ meets every irreducible component of $X$ of dimension $\neq 0$.
\end{lemma}

\begin{proof}
Indeed (\textbf{II}, 5.5.7), the set $X_g$ is an open affine of $X$, and if it contains an irreducible component $X'$ of $X$, the reduced closed subprescheme of $X$ having $X'$ as underlying space is both projective and affine over $A$ (\textbf{III}, 4.4.2), and is therefore finite over $A$, hence an Artinian scheme, therefore of dimension 0.
\end{proof}

This lemma being established, note that since $Z$ contains the maximal points of $X$, $X_f$ is dense, hence $\Supp(\sh{G})$ is rare in $X$, and the relation \sref{IV.5.3.1.2} results from the lemma and from \sref{IV.5.2.4}.

From the exact sequence \sref{IV.5.3.1.1}, we deduce, for every $n$, the exact sequence
\[
  0\to\sh{F}(n)\to\sh{F}(n+m)\to\sh{G}(n)\to 0
\]
and for $n$ large enough, one therefore has the exact sequence (\textbf{III}, 2.2.3)
\[
  0\to\Gamma(X,\sh{F}(n))\to\Gamma(X,\sh{F}(n+m))\to\Gamma(X,\sh{G}(n))\to 0
\]
whence, taking into account (\textbf{III}, 2.5.3), for $n$ large enough,
\[
  \chi_A(\sh{G}(n))=\chi_A(\sh{F}(n+m))-\chi_A(\sh{F}(n)).
\]
Since by virtue of \sref{IV.5.3.1.2} and the induction hypothesis, the degree of the polynomial $\chi_A(\sh{G}(n))$ is $d-1$, the preceding relation implies that the degree of $\chi_A(\sh{F}(n))$ is $d$, Q.E.D.

\subsection{Dimension of the image of a morphism}
\label{subsection:IV.5.4}

\oldpage[IV-2]{93}

\begin{proposition}[5.4.1]
\label{IV.5.4.1}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism.
\begin{enumerate}[\rm(i)]
  \item If $f$ is quasi-finite, we have $\dim(X)\leq\dim(\overline{f(X)})\leq\dim(Y)$.
  \item If $f$ is surjective and open \emph{(}resp.\ surjective and closed\emph{)}, we have $\dim(X)\geq\dim(Y)$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) One can replace $f$ by $f_\mathrm{red}$ \sref[II]{II.6.2.4}, hence suppose $X$ and $Y$ reduced; if $Z$ is the closed reduced subprescheme of $Y$ having $\overline{f(X)}$ as underlying space, we have $f=j\circ g$, where $j:Z\to Y$ is the canonical injection and $g:X\to Z$ is a quasi-finite morphism \sref[I]{I.5.2.2} and \sref[II]{II.6.2.4}.
One can therefore restrict to the case where $f(X)$ is dense in $Y$.
For every $x\in X$, $\sh{O}_x$ is then an $\sh{O}_{f(x)}$-module quasi-finite \sref[II]{II.6.2.2}, and it follows that $\mathfrak{m}_{f(x)}\sh{O}_x$ is an ideal of definition of $\sh{O}_x$ $(\mathbf{0_I}$, 7.4.4); but one knows $(\mathbf{0}$, 16.3.10) that if $A\to B$ is a local homomorphism of local Noetherian rings, such that, if $\mathfrak{m}$ is the maximal ideal of $A$, $\mathfrak{m}B$ is an ideal of definition of $B$, then $\dim(B)\leq\dim(A)$; this finishes proving (i) by virtue of \sref{IV.5.1.4}.

(ii) The definition of dimension $(\mathbf{0}$, 14.1.2) shows that it suffices to prove that for every suite $(y_i)_{0\leq i\leq n}$ of distinct elements of $Y$ such that $y_i\in\overline{\{y_{i+1}\}}$ for $0\leq i\leq n-1$, there exists a suite $(x_i)_{0\leq i\leq n}$ of points of $X$ such that $x_i\in\overline{\{x_{i+1}\}}$ for $0\leq i\leq n-1$ and $f(x_i)=y_i$ for every $i$.
Suppose first $f$ surjective and \emph{open}, and let us show the existence of $x_i$ by induction on $i$; the existence of $x_0\in X$ such that $f(x_0)=y_0$ results from the fact that $f$ is surjective.
If the $x_i$ have been determined for $i\leq m$ in a way satisfying $f(x_i)=y_i$ for $i\leq m$ and $x_i\in\overline{\{x_{i+1}\}}$ for $i\leq m$, we note that since $f$ is surjective and $y_{m+1}$ a generisation of $y_m$ \sref{IV.1.10.3}, there exists $x_{m+1}\in f^{-1}(y_{m+1})$ which is a generisation of $x_m$ and the induction can continue.

Suppose now $f$ surjective and \emph{closed}, and let us show the existence of $x_i$ by \emph{descending} induction on $i$; the existence of $x_n$ such that $f(x_n)=y_n$ results again from the fact that $f$ is surjective.
Suppose now $f$ surjective and closed and let us show the existence of the $x_i$ by descending induction on $i$: if the $x_i$ have been determined satisfying the conditions sought for $i>m$, we note $f(\overline{\{x_{m+1}\}})=\overline{\{y_{m+1}\}}$ since $f$ is closed (Bourbaki, \emph{Top.~gén.}, chap.~I\textsuperscript{er}, 3\textsuperscript{e} éd., \textsection5, no.~4, prop.~9); there exists therefore $x_m\in\overline{\{x_{m+1}\}}$ such that $f(x_m)=y_m$ and the descending induction can continue.
\end{proof}

\oldpage[IV-2]{94}

\begin{corollary}[5.4.2]
\label{IV.5.4.2}
If $X$, $Y$ are two locally Noetherian preschemes, $f:X\to Y$ a finite morphism \emph{(}hence closed\emph{)}, we have $\dim(X)=\dim(\overline{f(X)})$.
If moreover $f$ is surjective, $\dim(X)=\dim(Y)$.
\end{corollary}

\begin{remarks}[5.4.3]
\label{IV.5.4.3}
\begin{enumerate}[\rm(i)]
  \item One saw in \sref{IV.4.1.2} that if $X$ and $Y$ are two preschemes locally of finite type over a field $k$ and $f$ a $k$-morphism, the inequality $\dim(Y)\leq\dim(X)$ is already verified when $f$ is \emph{quasi-compact and dominant}.
  On the contrary, one can have $\dim(Y)>\dim(X)$ even when $f$ is of \emph{finite type, bijective} and a \emph{local immersion}, if one only supposes $X$ and $Y$ locally Noetherian.
  \item If $A$ and $B$ are two Noetherian rings such that $A\subset B$ and $B$ is a finite $A$-algebra, the corollary \sref{IV.5.4.2} shows again that $\dim(B)=\dim(A)$ $(\mathbf{0}$, 16.1.5).
  Suppose moreover that $A$ is a \emph{local} Noetherian ring; then $B$ is a Noetherian \emph{semi-local} ring (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection2, no.~5, cor.~3 of prop.~9); if $\mathfrak{n}_i$ ($1\leq i\leq r$) are the maximal ideals of $B$, we have therefore
  \begin{equation}
  \label{IV.5.4.3.1}
    \dim(A)=\sup_i\dim(B_{\mathfrak{n}_i}).\tag{5.4.3.1}
  \end{equation}
  One will observe that, under these conditions, one does not necessarily have $\dim(B_{\mathfrak{n}_i})=\dim(A)$ for \emph{every} $i$.
\end{enumerate}
\end{remarks}

\subsection{Dimension formula for a morphism of finite type}
\label{subsection:IV.5.5}

\begin{env}[5.5.1]
\label{IV.5.5.1}
Recall $(\mathbf{0}$, 16.3.9) that if $A$, $B$ are two local Noetherian rings, $k$ the residue field of $A$, $\varphi:A\to B$ a local homomorphism, we have
\begin{equation}
\label{IV.5.5.1.1}
  \dim(B)\leq\dim(A)+\dim(B\otimes_A k).\tag{5.5.1.1}
\end{equation}
\end{env}

We deduce:

\begin{proposition}[5.5.2]
\label{IV.5.5.2}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism, $x$ a point of $X$, $y=f(x)$.
Then
\begin{equation}
\label{IV.5.5.2.1}
  \dim(\sh{O}_x)\leq\dim(\sh{O}_y)+\dim(\sh{O}_x\otimes_{\sh{O}_y}k(y)).\tag{5.5.2.1}
\end{equation}
\end{proposition}

\oldpage[IV-2]{95}

In particular, if $x$ is a maximal point of the fibre $f^{-1}(y)$, we have
\begin{equation}
\label{IV.5.5.2.2}
  \dim(\sh{O}_x)\leq\dim(\sh{O}_y)\tag{5.5.2.2}
\end{equation}
since $\sh{O}_x\otimes_{\sh{O}_y}k(y)$ is the local ring of $x$ in the prescheme $f^{-1}(y)$, and is of dimension 0 by hypothesis.

We shall now obtain a more precise formula than \sref{IV.5.5.2.1} when one supposes that $f$ is a morphism \emph{of finite type}.

\begin{proposition}[5.5.3]
\label{IV.5.5.3}
Let $A$ be a Noetherian ring, $\mathrm{T}$ an indeterminate, $\mathfrak{p}$ a prime ideal of $A$; then $\mathfrak{p}'=\mathfrak{p}A[\mathrm{T}]$ is a prime ideal of $B=A[\mathrm{T}]$ and $\mathfrak{p}'\cap A=\mathfrak{p}$.
There exists an infinity of prime ideals of $B$ distinct from $\mathfrak{p}'$, whose intersection with $A$ is $\mathfrak{p}$; these ideals are pairwise without inclusion relation.
Moreover, if $\mathfrak{q}$ is such an ideal, we have
\begin{equation}
\label{IV.5.5.3.1}
  \dim(B_\mathfrak{q})=\dim(B_{\mathfrak{p}'})+1=\dim(A_\mathfrak{p})+r\tag{5.5.3.1}
\end{equation}
\end{proposition}

\begin{proof}
In the first assertions, we immediately reduce, by replacing $A$ by $A/\mathfrak{p}$ and observing that $A[\mathrm{T}]/\mathfrak{p}A[\mathrm{T}]=(A/\mathfrak{p})[\mathrm{T}]$, to the case $\mathfrak{p}=0$; they then result from the fact that the prime ideals of $B=A[\mathrm{T}]$ whose intersection with $A$ reduces to~$0$ are exactly those which do not meet the multiplicative subset $S=A-\{0\}$ of the integral ring $A$; now there is an increasing bijection from the set of these prime ideals onto the set of prime ideals of $S^{-1}A[\mathrm{T}]=K[\mathrm{T}]$, where $K$ is the field of fractions of $A$ \emph{(Bourbaki, Alg.~comm., chap.~II, \S2, no.~5, prop.~11)}.
Moreover, by \sref{IV.5.5.1.2}, $\dim(B_\mathfrak{q})\leq\dim(A_\mathfrak{p})+\dim(B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q})$, and if $k$ is the field of fractions of $A/\mathfrak{p}$, $B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q}$ identifies canonically with $(k[\mathrm{T}])_{\bar{\mathfrak{q}}}$, hence is a discrete valuation ring, hence of dimension~$1$.
Finally, if $\mathfrak{p}=\mathfrak{p}_0\supset\mathfrak{p}_1\supset\cdots\supset\mathfrak{p}_m$ is a chain of prime ideals of $A$ of maximal length, the ideals $\mathfrak{p}_jB$ $(0\leq j\leq m)$ are prime, pairwise distinct in $B$, and contained in $\mathfrak{q}$; so $\dim(B_\mathfrak{q})\geq m+1=\dim(A_\mathfrak{p})+1$ and hence $\dim(B_\mathfrak{q})=\dim(A_\mathfrak{p})+1$.
This relation also writes $\operatorname{ht}(\mathfrak{q})=\operatorname{ht}(\mathfrak{p})+1$; since $\mathfrak{q}\neq\mathfrak{p}'$, we also have $\operatorname{ht}(\mathfrak{q})\geq\operatorname{ht}(\mathfrak{p}')+1\geq\operatorname{ht}(\mathfrak{p})+1$ by definition of the height of a prime ideal; this finishes proving \sref{IV.5.5.3.1}.
\end{proof}

\begin{corollary}[5.5.4]
\label{IV.5.5.4}
For every Noetherian ring $A$, we have
\begin{equation}
\label{IV.5.5.4.1}
  \dim(A[\mathrm{T}_1,\ldots,\mathrm{T}_r])=\dim(A)+r\tag{5.5.4.1}
\end{equation}
\emph{(}$\mathrm{T}_i$ indeterminates\emph{)}.
\end{corollary}

\begin{corollary}[5.5.5]
\label{IV.5.5.5}
For every locally Noetherian prescheme $X$, the dimension of $X[\mathrm{T}_1,\ldots,\mathrm{T}_r]=X\otimes_\bb{Z}\bb{Z}[\mathrm{T}_1,\ldots,\mathrm{T}_r]$ \emph{(}$\mathrm{T}_i$ indeterminates\emph{)} is $\dim(X)+r$.
\end{corollary}

\begin{proof}
This results from \sref{IV.5.5.4} and $(\mathbf{0}$, 14.1.7).
\end{proof}

\begin{corollary}[5.5.6]
\label{IV.5.5.6}
Under the hypotheses of \sref{IV.5.5.3}, let $\mathfrak{q}$ be a prime ideal of $B=A[\mathrm{T}]$ such that $\mathfrak{q}\cap A=\mathfrak{p}$; if $k$ and $k'$ are the residue fields of $A_\mathfrak{p}$ and $B_\mathfrak{q}$ respectively, we have
\begin{equation}
\label{IV.5.5.6.1}
  \dim(A_\mathfrak{p})+1=\dim(B_\mathfrak{q})+\deg.\mathrm{tr}_k\,k'.\tag{5.5.6.1}
\end{equation}
\end{corollary}

\oldpage[IV-2]{96}

\begin{proof}
If $\mathfrak{q}=\mathfrak{p}B$, by \sref{IV.5.5.3.1}, $\dim(B_\mathfrak{q})=\dim(A_\mathfrak{p})$, and in this case $k'=k(\mathrm{T})$, so formula \sref{IV.5.5.6.1} is indeed verified.
In the contrary case, $\dim(B_\mathfrak{q})=\dim(A_\mathfrak{p})+1$, and since $\mathfrak{q}$ corresponds to a nonzero prime ideal $\mathfrak{q}'\neq 0$ of $k[\mathrm{T}]$, $k'$ is an algebraic extension of $k$, so we again have formula \sref{IV.5.5.6.1}.
\end{proof}

\begin{lemma}[5.5.7]
\label{IV.5.5.7}
Let $A$ be a local integral Noetherian ring, $\mathfrak{m}$ its maximal ideal, $k$ its residue field.
\begin{enumerate}[label=\emph{(\roman*)}]
  \item For every integral ring $B$ containing $A$, such that $B=A[x]$ \emph{(}for some $x\in B$\emph{)} and every prime ideal $\mathfrak{q}$ of $B$ such that $\mathfrak{q}\cap A=\mathfrak{m}$, we have
  \begin{equation}
  \label{IV.5.5.7.1}
    \dim(A)+\deg.\mathrm{tr}_A B\geq\dim(B_\mathfrak{q})+\deg.\mathrm{tr}_k\,k'\tag{5.5.7.1}
  \end{equation}
  where $k'$ denotes the residue field of $B_\mathfrak{q}$ and $\deg.\mathrm{tr}_A B$ the degree of transcendence of the field of fractions of $B$ over that of $A$.
  \item Suppose that for every maximal ideal $\mathfrak{n}$ of $A[\mathrm{T}]$ such that $\mathfrak{n}\cap A=\mathfrak{m}$, the ring $(A[\mathrm{T}])_\mathfrak{n}$ is catenary; then the two members of \sref{IV.5.5.7.1} are equal.
\end{enumerate}
\end{lemma}

\begin{proof}
\medskip\noindent\emph{(i)}
If $x$ is transcendental over the field of fractions of $A$, then $\deg.\mathrm{tr}_A B=1$ and the two members of \sref{IV.5.5.7.1} are equal by virtue of \sref{IV.5.5.6}.
In the contrary case, $B=A[\mathrm{T}]/\mathfrak{p}$, where $\mathfrak{p}$ is a nonzero prime ideal of $A[\mathrm{T}]$ such that $\mathfrak{p}\cap A=0$ since $B$ contains $A$; so $\operatorname{ht}(\mathfrak{p})=1$ by virtue of \sref{IV.5.5.3}.
The ideal $\mathfrak{q}$ of $B$ is of the form $\mathfrak{n}/\mathfrak{p}$, where $\mathfrak{n}\supset\mathfrak{p}$ is a prime ideal of $A[\mathrm{T}]$ such that $\mathfrak{n}\cap A=\mathfrak{m}$, and $B_\mathfrak{q}=(A[\mathrm{T}])_\mathfrak{n}/\mathfrak{p}(A[\mathrm{T}])_\mathfrak{n}$;
formula $(\mathbf{0}$, 16.1.4.1), applied to $X=\operatorname{Spec}((A[\mathrm{T}])_\mathfrak{n})$ and $Y=\operatorname{Spec}(B_\mathfrak{q})$, gives
\begin{equation}
\label{IV.5.5.7.2}
  \dim((A[\mathrm{T}])_\mathfrak{n})\geq\operatorname{ht}(\mathfrak{p}(A[\mathrm{T}])_\mathfrak{n})+\dim(B_\mathfrak{q})\tag{5.5.7.2}
\end{equation}
since $\operatorname{ht}(\mathfrak{p}(A[\mathrm{T}])_\mathfrak{n})=\operatorname{ht}(\mathfrak{p})=1$ by virtue of the bijective correspondence between prime ideals of $A[\mathrm{T}]$ contained in $\mathfrak{n}$ and prime ideals of $(A[\mathrm{T}])_\mathfrak{n}$.
Finally, formula \sref{IV.5.5.6.1} gives
\begin{equation}
\label{IV.5.5.7.3}
  \dim((A[\mathrm{T}])_\mathfrak{n})=\dim(A)+1-\deg.\mathrm{tr}_k\,k'\tag{5.5.7.3}
\end{equation}
since the residue fields of $B_\mathfrak{q}$ and of $(A[\mathrm{T}])_\mathfrak{n}$ are the same; moreover, $\deg.\mathrm{tr}_A B=0$, which finishes proving \sref{IV.5.5.7.1}.

\medskip\noindent\emph{(ii)}
If $(A[\mathrm{T}])_\mathfrak{n}$ is catenary, the two members of \sref{IV.5.5.7.2} are also equal $(\mathbf{0}$, 16.1.4), hence so are the two members of \sref{IV.5.5.7.1}.
\end{proof}

\begin{theorem}[5.5.8]
\label{IV.5.5.8}
\emph{(Dimension formula).}
Let $A$ be a local Noetherian integral ring, $B$ an integral ring containing $A$ and which is an $A$-algebra of finite type, $\mathfrak{q}$ a prime ideal of $B$ such that $\mathfrak{q}\cap A$ is the maximal ideal $\mathfrak{m}$ of $A$, $k$ and $k'$ the residue fields of $A$ and $B_\mathfrak{q}$ respectively.
We then have the inequality
\begin{equation}
\label{IV.5.5.8.1}
  \dim(A)+\deg.\mathrm{tr}_A B\geq\dim(B_\mathfrak{q})+\deg.\mathrm{tr}_k\,k'.\tag{5.5.8.1}
\end{equation}
Moreover, the two members are equal if, for every sub-$A$-algebra $A'$ of finite type of $B[\mathrm{T}]$ and every maximal ideal $\mathfrak{m}'$ of $A'$ such that $\mathfrak{m}'\cap A=\mathfrak{m}$, $A'_{\mathfrak{m}'}$ is catenary.
\end{theorem}

\begin{proof}
Let $B=A[x_1,\ldots,x_n]$ and argue by induction on $n$.
Set $C=A[x_1,\ldots,x_{n-1}]$ and $\mathfrak{r}=\mathfrak{q}\cap C$; $C_\mathfrak{r}$ is a local integral Noetherian ring, and if we set $S=C-\mathfrak{r}$, $B'=S^{-1}B$, $B'=C_\mathfrak{r}[x_n]$, and $\mathfrak{q}'=S^{-1}\mathfrak{q}$, we have $B_\mathfrak{q}=B'_{\mathfrak{q}'}$ and $\mathfrak{q}'\cap C_\mathfrak{r}=\mathfrak{r}C_\mathfrak{r}$; moreover the fields of fractions of $B'$ and $C_\mathfrak{r}$ are those of $B$ and $C$ respectively.
By \sref{IV.5.5.7.1}, we thus have
\begin{equation}
\label{IV.5.5.8.2}
  \dim(C_\mathfrak{r})+\deg.\mathrm{tr}_C B\geq\dim(B_\mathfrak{q})+\deg.\mathrm{tr}_{k_1}k'.\tag{5.5.8.2}
\end{equation}
\oldpage[IV-2]{97}
On the other hand, the induction hypothesis gives
\begin{equation}
\label{IV.5.5.8.3}
  \dim(A)+\deg.\mathrm{tr}_A C\geq\dim(C_\mathfrak{r})+\deg.\mathrm{tr}_k\,k_1\tag{5.5.8.3}
\end{equation}
whence \sref{IV.5.5.8.1} by adding \sref{IV.5.5.8.2} and \sref{IV.5.5.8.3} member by member.
To prove the second assertion, note first that every sub-$A$-algebra of finite type of $C[\mathrm{T}]$ is also a sub-$A$-algebra of finite type of $B[\mathrm{T}]$; so by induction on $n$, we can suppose that the two members of \sref{IV.5.5.8.3} are equal.
On the other hand, to see that the two members of \sref{IV.5.5.8.2} are equal, it suffices, by virtue of \sref{IV.5.5.7}, to verify that if $\mathfrak{n}$ is a maximal ideal of $C_\mathfrak{r}[\mathrm{T}]$ such that $\mathfrak{n}\cap C_\mathfrak{r}=\mathfrak{r}C_\mathfrak{r}$, then $(C_\mathfrak{r}[\mathrm{T}])_\mathfrak{n}$ is catenary;
but $C_\mathfrak{r}[\mathrm{T}]=S'^{-1}C[\mathrm{T}]$ where $S'=C-\mathfrak{r}$; the ideal $\mathfrak{n}$ is thus of the form $S'^{-1}\mathfrak{n}'$, where $\mathfrak{n}'$ is a prime ideal of $C[\mathrm{T}]$ such that $\mathfrak{n}'\cap C=\mathfrak{r}$, whence $(C_\mathfrak{r}[\mathrm{T}])_\mathfrak{n}=(C[\mathrm{T}])_{\mathfrak{n}'}$;
if $\mathfrak{m}'$ is a maximal ideal of $C[\mathrm{T}]$ containing $\mathfrak{n}'$, $(C[\mathrm{T}])_{\mathfrak{m}'}$ is thus a local ring of $(C[\mathrm{T}])_{\mathfrak{n}'}$, and since by hypothesis this last ring is catenary, so is the former by $(\mathbf{0}$, 16.1.4).
\end{proof}

\subsection{Dimension formula and universally catenary rings}
\label{subsection:IV.5.6}

\begin{proposition}[5.6.1]
\label{IV.5.6.1}
Let $A$ be a Noetherian ring. The following properties are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item Every polynomial ring $A[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ is catenary.
  \item Every $A$-algebra of finite type is catenary.
  \item $A$ is catenary, and for every local integral $A$-algebra, essentially of finite type \sref{IV.1.3.8}, $B'$, designating by $\mathfrak{p}$ the inverse image in $A$ of the maximal ideal $\mathfrak{q}$ of $B'$, by $A'$ the image of $A_\mathfrak{p}$ in $B'$, by $k$ and $k'$ the residue fields of $A'$ and $B'$ respectively,
  \begin{equation}
  \label{IV.5.6.1.1}
    \dim(A')+\deg.\mathrm{tr}_{A'}(B')=\dim(B')+\deg.\mathrm{tr}_k\,k'.\tag{5.6.1.1}
  \end{equation}
\end{enumerate}
\end{proposition}

\begin{proof}
The fact that \emph{b)} implies \emph{c)} results from \sref{IV.5.5.8}; indeed, $B'$ is an $A'$-algebra essentially of finite type \sref{IV.1.3.10}, hence of the form $B''_{\mathfrak{q}''}$, where $B''$ is a sub-$A'$-algebra of finite type of $B'$, $\mathfrak{q}''$ a prime ideal of $B''$ above the maximal ideal $\mathfrak{p}'$ of $A'$; moreover the fields of fractions of $B'$ and $B''$ are the same, hence $\deg.\mathrm{tr}_{A'}(B')=\deg.\mathrm{tr}_{A'}(B'')$.
To prove \sref{IV.5.6.1.1}, it suffices to show that (under hypothesis \emph{b)}) every sub-$A'$-algebra $A_1$ of finite type of $B'[\mathrm{T}]$ is catenary; indeed the two members of \sref{IV.5.5.8.1}, where $A$, $B$, and $\mathfrak{q}$ are replaced by $A'$, $B''$, and $\mathfrak{q}''$, will then be equal, whence the equality \sref{IV.5.6.1.1}.
But hypothesis \emph{b)} implies that every $A$-algebra essentially of finite type is catenary $(\mathbf{0}$, 16.1.4), and $A_1$ is such an $A$-algebra \sref{IV.1.3.9}.

It is trivial that \emph{b)} implies \emph{a)}; conversely, \emph{a)} implies \emph{b)}, since every $A$-algebra of finite type is a quotient of a polynomial algebra $(\mathbf{0}$, 16.1.4).

\oldpage[IV-2]{98}
It remains to prove that \emph{c)} implies \emph{b)}.
Since every quotient by a prime ideal of an $A$-algebra of finite type is an integral $A$-algebra of finite type, if $B$ is an integral $A$-algebra of finite type, $\mathfrak{q}$ and $\mathfrak{q}'$ two prime ideals of $B$ such that $\mathfrak{q}'\subset\mathfrak{q}$, by $(\mathbf{0}$, 16.1.4.2)
\begin{equation}
\label{IV.5.6.1.2}
  \dim(B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q})+\dim(B_{\mathfrak{q}'})=\dim(B_\mathfrak{q}).\tag{5.6.1.2}
\end{equation}
Let $\mathfrak{p}$, $\mathfrak{p}'$ be the inverse images of $\mathfrak{q}$, $\mathfrak{q}'$ respectively in $A$, $\mathfrak{n}$ the kernel of the homomorphism $A_\mathfrak{p}\to B_\mathfrak{q}$.
The image $A'$ of $A_\mathfrak{p}$ in $B_\mathfrak{q}$ being isomorphic to $A_\mathfrak{p}/\mathfrak{n}$, formula \sref{IV.5.6.1.1} applied to $A'$ and $B'=B_\mathfrak{q}$ writes
\begin{equation}
\label{IV.5.6.1.3}
  \dim(A_\mathfrak{p}/\mathfrak{n})+\deg.\mathrm{tr}_{A_\mathfrak{p}/\mathfrak{n}}(B_\mathfrak{q})=\dim(B_\mathfrak{q})+\deg.\mathrm{tr}_{k(\mathfrak{p})}\,k(\mathfrak{q}).\tag{5.6.1.3}
\end{equation}
On the other hand, the kernel of $A_{\mathfrak{p}'}\to B_{\mathfrak{q}'}$ is $\mathfrak{n}A_{\mathfrak{p}'}$, so, applying formula \sref{IV.5.6.1.1} where $A'$ is replaced by $A_{\mathfrak{p}'}/\mathfrak{n}A_{\mathfrak{p}'}$ and $B'$ by $B_{\mathfrak{q}'}$, we have
\begin{equation}
\label{IV.5.6.1.4}
  \dim(A_{\mathfrak{p}'}/\mathfrak{n}A_{\mathfrak{p}'})+\deg.\mathrm{tr}_{A_{\mathfrak{p}'}/\mathfrak{n}A_{\mathfrak{p}'}}(B_{\mathfrak{q}'})=\dim(B_{\mathfrak{q}'})+\deg.\mathrm{tr}_{k(\mathfrak{p}')}\,k(\mathfrak{q}').\tag{5.6.1.4}
\end{equation}
Finally, $B/\mathfrak{q}'$ is an integral $A$-algebra of finite type, the inverse image of $\mathfrak{q}/\mathfrak{q}'$ in $A$ is $\mathfrak{p}$, and the kernel of the homomorphism $A_\mathfrak{p}\to B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q}$ is $\mathfrak{p}'A_\mathfrak{p}$, so, applying \sref{IV.5.6.1.1} where $A'$ is replaced by $A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p}$ and $B'$ by $B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q}$, we have
\begin{equation}
\label{IV.5.6.1.5}
  \dim(A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p})+\deg.\mathrm{tr}_{A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p}}(B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q})=\dim(B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q})+\deg.\mathrm{tr}_{k(\mathfrak{p})}\,k(\mathfrak{q}).\tag{5.6.1.5}
\end{equation}
Adding \sref{IV.5.6.1.4} and \sref{IV.5.6.1.5} member by member, and noting that $k(\mathfrak{p}')$ (resp.\ $k(\mathfrak{q}')$) is the field of fractions of $A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p}$ (resp.\ $B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q}$); moreover $A_{\mathfrak{p}'}/\mathfrak{n}A_{\mathfrak{p}'}$ and $A_\mathfrak{p}/\mathfrak{n}$ have the same field of fractions, and $B_{\mathfrak{q}'}$ and $B_\mathfrak{q}$ have the same field of fractions; finally, since $A$ is supposed catenary, by $(\mathbf{0}$, 16.1.4.2)
\[
  \dim(A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p})+\dim(A_{\mathfrak{p}'}/\mathfrak{n}A_{\mathfrak{p}'})=\dim(A_\mathfrak{p}/\mathfrak{n}).
\]
Taking these remarks into account and \sref{IV.5.6.1.3}, we obtain relation \sref{IV.5.6.1.2}.
\end{proof}

\begin{definition}[5.6.2]
\label{IV.5.6.2}
A Noetherian ring $A$ is said to be \emph{universally catenary} if it satisfies the equivalent conditions \emph{a)}, \emph{b)}, \emph{c)} of \sref{IV.5.6.1}.
\end{definition}

\begin{remarks}[5.6.3]
\label{IV.5.6.3}
\medskip\noindent\emph{(i)}
If $A$ is universally catenary, then so is $S^{-1}A$ for every multiplicative subset $S$ of $A$, as follows immediately from (\sref{IV.5.6.1},~\emph{a)}) and the fact that every ring of fractions of a catenary ring is catenary.
Conversely, if for every prime ideal $\mathfrak{p}$ of $A$, the ring $A_\mathfrak{p}$ is universally catenary, then $A$ is universally catenary: this follows from (\sref{IV.5.6.1},~\emph{c)}), noting that by setting $S=A-\mathfrak{p}$, $S^{-1}B$ is an $A_\mathfrak{p}$-algebra of finite type, and $B_\mathfrak{q}=(S^{-1}B)_{S^{-1}\mathfrak{q}}$.

\medskip\noindent\emph{(ii)}
A locally Noetherian prescheme $X$ is said to be \emph{universally catenary} if, for every $x\in X$, the ring $\sh{O}_x$ is universally catenary.
It follows from~(i) that for $A$ to be a universally catenary ring, it is necessary and sufficient that the scheme $\operatorname{Spec}(A)$ be universally catenary.

\medskip\noindent\emph{(iii)}
The criterion (\sref{IV.5.6.1},~\emph{c)}) only involves the images of $A$ in integral $A$-algebras of finite type, hence \emph{integral quotient rings} of $A$.
We conclude that if $\mathfrak{p}_i$ $(1\leq i\leq n)$ are the minimal prime ideals of $A$, it amounts to the same to say that $A$ is universally catenary or that each of the $A/\mathfrak{p}_i$ is.
This also shows that a locally Noetherian prescheme $X$ is universally catenary, or that $X_\mathrm{red}$ is.

\oldpage[IV-2]{99}
\medskip\noindent\emph{(iv)}
The criterion (\sref{IV.5.6.1},~\emph{b)}) and remark~(i) show that, if $A$ is a universally catenary ring, then so is every $A$-algebra \emph{essentially of finite type} \sref{IV.1.3.8}.
\end{remarks}

\begin{proposition}[5.6.4]
\label{IV.5.6.4}
Every quotient of a regular ring is universally catenary.
\end{proposition}

\begin{proof}
It suffices to see that every regular ring $A$ is universally catenary (\sref{IV.5.6.3},~(iv)); or, we know that $A[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ is regular $(\mathbf{0}$, 17.3.7), hence catenary $(\mathbf{0}$, 16.5.12), and we conclude by (\sref{IV.5.6.1},~\emph{a)}).
\end{proof}

In particular, it follows from the Cohen structure theorem $(\mathbf{0}$, 19.8.8) that every \emph{complete local Noetherian ring} is universally catenary.
Similarly, every algebra of finite type over a field, every prescheme being a quotient of a regular prescheme, and every prescheme \emph{locally of finite type over a field} is universally catenary.

We will see further on \sref{IV.6.3.7} that in \sref{IV.5.6.4}, one can replace ``regular ring'' by ``Cohen-Macaulay ring''.

\begin{proposition}[5.6.5]
\label{IV.5.6.5}
Let $Y$ be an irreducible locally Noetherian prescheme, $X$ an irreducible prescheme, $f:X\to Y$ a dominant morphism locally of finite type.
Let $\xi$ \emph{(}resp.\ $\eta$\emph{)} be the generic point of $X$ \emph{(}resp.\ $Y$\emph{)}, and let $e=\dim(f^{-1}(\eta))=\deg.\mathrm{tr}_{k(\eta)}\,k(\xi)$ \emph{(``dimension of the generic fibre'', cf.\ $(\mathbf{0_I}$, 2.1.8) and \sref{IV.4.1.1}).}
For every $x\in X$, setting $y=f(x)$, we have
\begin{equation}
\label{IV.5.6.5.1}
  e+\dim(\sh{O}_y)\geq\deg.\mathrm{tr}_{k(y)}\,k(x)+\dim(\sh{O}_x)\tag{5.6.5.1}
\end{equation}
\begin{equation}
\label{IV.5.6.5.2}
  \dim(\sh{O}_x)\leq\dim(\sh{O}_y)+\dim(\sh{O}_x\otimes_{\sh{O}_y}k(y))-\delta(x)\tag{5.6.5.2}
\end{equation}
where $\delta(x)=\dim_x(f^{-1}(y))-e$.

Moreover, if $Y$ is universally catenary, the two members of \sref{IV.5.6.5.1} \emph{(}resp.\ \sref{IV.5.6.5.2}\emph{)} are equal.
If furthermore $x$ is closed in $f^{-1}(y)$, we have
\begin{equation}
\label{IV.5.6.5.3}
  \dim(\sh{O}_x)=\dim(\sh{O}_y)+e.\tag{5.6.5.3}
\end{equation}
\end{proposition}

\begin{proof}
We can evidently restrict to the case where $X$ and $Y$ are affine (hence $f$ of finite type) and reduced \sref{IV.1.5.4}, hence integral.
Since $f$ is dominant, $\sh{O}_y$ identifies with a subring of $\sh{O}_x$, and the fields of fractions of $\sh{O}_y$ and of $\sh{O}_x$ are $k(\eta)$ and $k(\xi)$ respectively \sref[I]{I.8.2.7};
moreover, $\sh{O}_x$ is a local ring of an $\sh{O}_y$-algebra integral of finite type $B$ at a prime ideal $\mathfrak{q}$ \sref[I]{I.3.6.5}; applying \sref{IV.5.5.8.1} with $A=\sh{O}_y$, $B$, and $\mathfrak{q}$, we obtain \sref{IV.5.6.5.1}.
Moreover, $\sh{O}_x\otimes_{\sh{O}_y}k(y)=\sh{O}_x/\mathfrak{m}_y\sh{O}_x$ is none other than the local ring of the fibre $f^{-1}(y)$ at the point $x$ \sref[I]{I.3.6.1}; since $f^{-1}(y)$ is a prescheme of finite type over $k(y)$, it follows from \sref{IV.5.2.3} that
\[
  \dim(\sh{O}_x\otimes_{\sh{O}_y}k(y))=\dim_x(f^{-1}(y))-\deg.\mathrm{tr}_{k(y)}\,k(x)\,;
\]
so the inequality \sref{IV.5.6.5.2} is just another form of \sref{IV.5.6.5.1}.

\oldpage[IV-2]{100}
The fact that the two members of \sref{IV.5.6.5.1} are equal when $Y$ is universally catenary is none other than the equality \sref{IV.5.6.1.1}, applied to $A=\sh{O}_y$ and $B'=\sh{O}_x$.
To show that moreover \sref{IV.5.6.5.3} holds when $x$ is closed in $f^{-1}(y)$, it suffices to note that, in general, $\deg.\mathrm{tr}_{k(y)}\,k(x)$ is the dimension of $\overline{\{x\}}\cap f^{-1}(y)$, as follows from \sref{IV.5.2.1} applied to the closed reduced sub-prescheme of $f^{-1}(y)$ having this subspace as underlying space.

We will show further on (13.1.1) that, under the conditions of \sref{IV.5.6.5}, we \emph{always} have $\delta(x)\geq 0$, and \sref{IV.5.6.5.2} makes precise in this case \sref{IV.5.2.1}.
\end{proof}

\begin{corollary}[5.6.6]
\label{IV.5.6.6}
Under the general hypotheses of \sref{IV.5.6.5}, we have
\begin{equation}
\label{IV.5.6.6.1}
  \dim(X)\leq\dim(Y)+e.\tag{5.6.6.1}
\end{equation}
If we furthermore suppose $Y$ universally catenary, then, for the two members of \sref{IV.5.6.6.1} to be equal, it is necessary and sufficient that
\begin{equation}
\label{IV.5.6.6.2}
  \dim(Y)=\sup_{y\in f(X)}\dim(\sh{O}_y).\tag{5.6.6.2}
\end{equation}
In particular, the two members of \sref{IV.5.6.6.1} are equal when $Y$ is locally of finite type over a field.
\end{corollary}

\begin{proof}
The inequality \sref{IV.5.6.6.1} results from \sref{IV.5.6.5.1} applied to a closed point $x$ of $X$, taking into account \sref{IV.5.1.4.2} and \sref{IV.5.1.11}.
On the other hand, every nonempty fibre $f^{-1}(y)$ contains a closed point $x$ in this fibre \sref{IV.5.1.11}, and if we suppose $Y$ universally catenary, at this point we have the relation \sref{IV.5.6.5.3}.
Taking the upper bounds of the two members of \sref{IV.5.6.5.3} when $x$ runs through $X$, and noting that every closed point $x$ of $X$ is \emph{a fortiori} closed in $f^{-1}(f(x))$, the second assertion of the corollary follows from \sref{IV.5.1.4.1} and \sref{IV.5.1.4.2}.

We note that condition \sref{IV.5.6.6.2} is trivially satisfied when $f$ is surjective.
\end{proof}

\begin{corollary}[5.6.7]
\label{IV.5.6.7}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $n$ an integer.
If, for every $y\in Y$, $\dim(f^{-1}(y))\leq n$, then
\begin{equation}
\label{IV.5.6.7.1}
  \dim(X)\leq\dim(Y)+n.\tag{5.6.7.1}
\end{equation}
\end{corollary}

\begin{proof}
\oldpage[IV-2]{101}
By virtue of $(\mathbf{0}$, 14.1.7) and of \sref{IV.1.5.4}, we can restrict to the case where $X$ and $Y$ are affine, hence $f$ of finite type, $X$ and $Y$ reduced;
let $X_i$ $(1\leq i\leq m)$ be the sub-preschemes (integral) of $X$ having for underlying spaces the irreducible components of $X$; we have $\dim(X)=\sup(\dim(X_i))$.
If $Z_i$ is the closed reduced sub-prescheme of $Y$ having for underlying space $\overline{f(X_i)}$, $Z_i$ is integral $(\mathbf{0_I}$, 2.1.5); the restriction $X_i\to Y$ of $f$ factors as $X_i\xrightarrow{g_i}Z_i\xrightarrow{j_i}Y$, where $j_i$ is the canonical injection \sref[I]{I.5.2.2}, and $g_i$ is dominant and of finite type \sref{IV.1.5.4}.
We have $\dim(Z_i)\leq\dim(Y)$ for every $i$; on the other hand, if $z_i$ is the generic point of $Z_i$, $\dim(g_i^{-1}(z_i))\leq n$ for every $i$ by hypothesis; so to prove \sref{IV.5.6.7.1}, it suffices to show that $\dim(X_i)\leq\dim(Z_i)+n$ for every $i$, which follows from \sref{IV.5.6.6.1}.
\end{proof}

\begin{corollary}[5.6.8]
\label{IV.5.6.8}
Let $Y$ be a locally Noetherian prescheme, $X$ an irreducible prescheme, $f:X\to Y$ a morphism locally of finite type, $n$ an integer $\geq 0$.
Suppose that $Y$ is universally catenary, and that for every $y\in Y$, $\dim(f^{-1}(y))\geq n$ \emph{(}resp.\ $\dim(f^{-1}(y))=n$\emph{)}.
Then
\begin{equation}
\label{IV.5.6.8.1}
  \dim(X)\geq\dim(Y)+n\tag{5.6.8.1}
\end{equation}
\emph{(}resp.
\begin{equation}
\label{IV.5.6.8.2}
  \dim(X)=\dim(Y)+n\tag{5.6.8.2}
\end{equation}
\emph{)}.
\end{corollary}

\begin{proof}
Since $n\geq 0$, the hypothesis implies that $f$ is \emph{surjective}, hence $Y$ irreducible; moreover, if $\eta$ is the generic point of $Y$, $\dim(f^{-1}(\eta))\geq n$ (resp.\ $\dim(f^{-1}(\eta))=n$).
The conclusion follows from \sref{IV.5.6.6}.
\end{proof}

\begin{remarks}[5.6.9]
\label{IV.5.6.9}
\medskip\noindent\emph{(i)}
Even if $Y$ is regular, $X$ irreducible, $f:X\to Y$ dominant and of finite type, the two members of \sref{IV.5.6.6.1} are not necessarily equal, as shown by the example where $Y=\operatorname{Spec}(A)$, $A$ a discrete valuation ring, $X=\operatorname{Spec}(K)$ where $K$ is the field of fractions of $A$, $f$ being the canonical morphism.

\medskip\noindent\emph{(ii)}
Example (\sref{IV.5.4.3},~(i)) shows that in \sref{IV.5.6.6} and \sref{IV.5.6.8}, one cannot suppress the hypothesis that $X$ is irreducible, the other hypotheses being satisfied.
However, we will see (10.6.1) that with additional hypotheses on $Y$, satisfied for example when $Y$ is a prescheme locally of finite type over a field, or over a Dedekind ring having an infinity of prime ideals, such phenomena cannot occur.
\end{remarks}

\begin{proposition}[5.6.10]
\label{IV.5.6.10}
Let $A$ be a local integral Noetherian universally catenary ring, $B$ an integral ring containing $A$ and which is an $A$-algebra of finite type.
Then, for every maximal ideal $\mathfrak{n}$ of $B$, $\dim(B_\mathfrak{n})=\dim(A)$.
\end{proposition}

\begin{proof}
Indeed, $\deg.\mathrm{tr}_A B=0$ and the residue field $k'$ of $B_\mathfrak{n}$ is an algebraic extension of the field of fractions of $A$.
The conclusion follows from formula \sref{IV.5.6.1.1}, every maximal ideal being above that of $A$.
\end{proof}

\begin{example}[5.6.11]
\label{IV.5.6.11}
Let $A$ be a local Noetherian integral ring and \emph{integrally closed}; then, as we have seen $(\mathbf{0}$, 16.1.6), the conclusion of \sref{IV.5.6.10} is valid for every integral finite $A$-algebra $B$ containing $A$.
On the other hand, we will construct an example of a local Noetherian integral \emph{catenary} ring $A$ for which the conclusion of \sref{IV.5.6.10} will fail.
We use the construction of (\sref{IV.5.2.5},~(i)).
Let $k_0$ be a field, $k$ a pure transcendental extension $k_0(S_i)_{i\in\bb{N}}$ of infinite transcendence degree, $V$ the discrete valuation ring $k[\mathrm{S}]_{(\mathrm{S})}$, local ring of the polynomial ring $k[\mathrm{S}]$ at the prime ideal $(\mathrm{S})$; finally let $E$ be the polynomial ring $V[\mathrm{T}]$; we have seen that in $E$ the maximal ideal $\mathfrak{m}=(\mathrm{S})+(\mathrm{T})$ is of height~$2$ and the maximal ideal $\mathfrak{m}'=(\mathrm{S}\mathrm{T}-1)$ of height~$1$;
the residue fields are $k(\mathfrak{m})=k$ and $k(\mathfrak{m}')=k(\mathrm{S})$, fields of fractions of $V$; by choice of $k$, these fields are isomorphic.
Let $\varepsilon$ and $\varepsilon'$ be the canonical homomorphisms of $E$ onto $E/\mathfrak{m}=k(\mathfrak{m})$ and $E/\mathfrak{m}'=k(\mathfrak{m}')$ respectively; let $\sigma$ be an isomorphism of $k(\mathfrak{m})$ onto $k(\mathfrak{m}')$, and consider the subring $C$ of $E$ formed by $x\in E$ such that $\varepsilon'(x)=\sigma(\varepsilon(x))$ (this construction is a particular case of ``gluing procedures'' which will be systematically studied in chap.~V).
It is immediate that $\mathfrak{n}=\mathfrak{m}\mathfrak{m}'=\mathfrak{m}\cap\mathfrak{m}'$ is a maximal ideal of $C$, $C/\mathfrak{m}\mathfrak{m}'$ identifying with the subfield of $(E/\mathfrak{m})\times(E/\mathfrak{m}')$ formed by the pairs $(z,\sigma(z))$.
We have $E=C+C(\mathrm{S}\mathrm{T}-1)$, hence $E$ is a $C$-algebra of finite type; we will now see that $C$ is Noetherian: this will result from the following lemma:
\end{example}

\oldpage[IV-2]{102}

\begin{lemma}[5.6.11.1]
\label{IV.5.6.11.1}
Let $R$ be a ring, $S$ a subring, $\mathfrak{R}=\operatorname{Ann}_S(R/S)$ the conductor of $S$ in $R$ \emph{(}largest ideal of $S$ which is also an ideal of $R$\emph{)}.
\begin{enumerate}[label=\emph{(\roman*)}]
  \item For every ideal $\mathfrak{J}\subset\mathfrak{R}$ of $R$, there exists a strictly increasing map from the set of ideals $\mathfrak{L}$ of $S$ such that $R\cdot\mathfrak{L}=\mathfrak{J}$ onto the set of sub-$(S/\mathfrak{R})$-modules of $\mathfrak{J}/\mathfrak{R}\mathfrak{J}$.
  \item If $S/\mathfrak{R}$ and $R$ are Noetherian and if $R$ is an $S$-module of finite type, then every increasing sequence of ideals of $S$ contained in $\mathfrak{R}$ is stationary.
\end{enumerate}
\end{lemma}

\begin{proof}
\medskip\noindent\emph{(i)}
This is clear since $\mathfrak{L}$ is an ideal of $S$ and $R\cdot\mathfrak{L}=\mathfrak{J}$ ($\mathfrak{L}$ being an ideal of $S$), whence the conclusion.

\medskip\noindent\emph{(ii)}
Let $(\mathfrak{L}_n)$ be an increasing sequence of ideals of $S$ contained in $\mathfrak{R}$.
The sequence of ideals $R\cdot\mathfrak{L}_n$ of $R$ is stationary since $R$ is Noetherian; we can thus suppose that all the ideals $R\cdot\mathfrak{L}_n$ are equal to the same ideal $\mathfrak{J}$ of $R$.
Since $R$ is Noetherian, $\mathfrak{J}/\mathfrak{R}\mathfrak{J}$ is also a finitely generated $S$-module, hence also an $(S/\mathfrak{R})$-module of finite type.
But since $S/\mathfrak{R}$ is Noetherian, $\mathfrak{J}/\mathfrak{R}\mathfrak{J}$ is an $(S/\mathfrak{R})$-Noetherian module, and the conclusion follows from~(i).
\end{proof}

We apply this lemma with $R=E$, $S=C$; it is clear that $\mathfrak{n}$ is the conductor of $C$ in $E$; moreover, for every ideal $\mathfrak{a}$ of $C$, $\mathfrak{a}/(\mathfrak{n}\cap\mathfrak{a})$ is isomorphic to $(\mathfrak{a}+\mathfrak{n})/\mathfrak{n}$, hence a sub-$C$-module of $C/\mathfrak{n}$, which is a simple $C$-module; it therefore suffices to show that every ideal $\mathfrak{a}\cap\mathfrak{n}$ of $C$ is of finite type, and this follows from \sref{IV.5.6.11.1}.
It follows from $(\mathbf{0}$, 16.1.5) that $\dim(C)=\dim(E)=2$.
Let $A=C_\mathfrak{n}$; if $U=C-\mathfrak{n}$, the ring of fractions $B=U^{-1}E$ is thus the integral closure of $A$, and it is an $A$-module of finite type; moreover, since $\mathfrak{m}$ and $\mathfrak{m}'$ are the only prime ideals of $E$ containing $\mathfrak{n}$, $B$ is a semi-local ring whose local rings at the two maximal ideals are isomorphic to $E_\mathfrak{m}$ and $E_{\mathfrak{m}'}$ respectively, hence are of dimension~$2$ and~$1$ respectively.
This shows that $\dim(B)=2$, hence also $\dim(A)=2$ $(\mathbf{0}$, 16.1.5).
Since $A$ is a local integral ring, it is necessarily catenary (every prime ideal distinct from~$0$ and from the maximal ideal being necessarily of height~$1$); but it does not satisfy the conclusion of \sref{IV.5.6.10}, and \emph{a fortiori} is not universally catenary.

\oldpage[IV-2]{103}
Let us also note that $\mathfrak{n}$ is a prime ideal of height~$2$ in $C$, and that for every prime ideal $\mathfrak{p}$ of height~$1$ in $C$, there exists a \emph{single} prime ideal $\mathfrak{p}'$ of $E$ above $\mathfrak{p}$, necessarily of height~$1$, and such that $C_\mathfrak{p}=E_{\mathfrak{p}'}$.
Indeed, there is at least one prime ideal $\mathfrak{p}'$ of $E$ above $\mathfrak{p}$, and it follows from the Cohen--Seidenberg theorem that such an ideal is necessarily of height~$1$; moreover, since $\mathfrak{p}\not\supset\mathfrak{n}$, $C_\mathfrak{p}$ is integrally closed \emph{(Bourbaki, Alg.~comm., chap.~V, \S1, no.~5, cor.~5 of prop.~16)}, hence $E_\mathfrak{p}=C_\mathfrak{p}$, which proves our assertions.

It would be interesting to know whether every local integral Noetherian ring satisfying the conclusion of \sref{IV.5.6.10} is universally catenary; this is what Nagata~[33] has affirmed, but his proof does not appear to be complete.

\subsection{Depth and property $(\mathrm{S}_k)$}
\label{subsection:IV.5.7}

\begin{definition}[5.7.1]
\label{IV.5.7.1}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ an $\sh{O}_X$-Module coherent.
We call \emph{depth} (resp.\ \emph{codepth}) of $\sh{F}$ at a point $x\in X$ the number $\operatorname{prof}(\sh{F}_x)$ (resp.\ $\operatorname{coprof}(\sh{F}_x)$) $(\mathbf{0}$, 16.4.5 and 16.4.9).
We call \emph{codepth} of $\sh{F}$ the number
\begin{equation}
\label{IV.5.7.1.1}
  \operatorname{coprof}(\sh{F})=\sup_{x\in X}\operatorname{coprof}(\sh{F}_x).\tag{5.7.1.1}
\end{equation}

We say that $\sh{F}$ is an $\sh{O}_X$-\emph{Module de Cohen-Macaulay} at a point $x$ if $\sh{F}_x$ is an $\sh{O}_x$-module de Cohen-Macaulay, that is to say $(\mathbf{0}$, 16.5.1) if $\operatorname{coprof}(\sh{F}_x)=0$.
We say that $\sh{F}$ is an $\sh{O}_X$-Module de Cohen-Macaulay if it is so at every point, that is if $\operatorname{coprof}(\sh{F})=0$.
A point $x\in X$ such that $\sh{O}_x$ is a Cohen-Macaulay ring is also called a \emph{Cohen-Macaulay point} of $X$.

We call \emph{codepth} of $X$ and write $\operatorname{coprof}(X)$ the number $\operatorname{coprof}(\sh{O}_X)$.
We say that $X$ is a \emph{Cohen-Macaulay prescheme} if $\sh{O}_X$ is an $\sh{O}_X$-Module de Cohen-Macaulay, that is if $\operatorname{coprof}(X)=0$.
Every locally Noetherian prescheme of dimension~$0$ is evidently a Cohen-Macaulay prescheme.
To say that $\operatorname{Spec}(A)$ is a Cohen-Macaulay scheme means that $A$ is a Cohen-Macaulay ring $(\mathbf{0}$, 16.5.13).
\end{definition}

\begin{definition}[5.7.2]
\label{IV.5.7.2}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ an $\sh{O}_X$-Module coherent, $k$ a positive or negative integer.
We say that $\sh{F}$ possesses \emph{property $(\mathrm{S}_k)$} if, for every $x\in X$, we have
\begin{equation}
\label{IV.5.7.2.1}
  \operatorname{prof}(\sh{F}_x)\geq\inf(k,\dim(\sh{F}_x)).\tag{5.7.2.1}
\end{equation}
We say that $\sh{F}$ possesses property $(\mathrm{S}_k)$ \emph{at a point} $x\in X$ if, for every generisation $x'$ of $x$ in $X$, we have
\begin{equation}
\label{IV.5.7.2.2}
  \operatorname{prof}(\sh{F}_{x'})\geq\inf(k,\dim(\sh{F}_{x'})).\tag{5.7.2.2}
\end{equation}
We say that $X$ satisfies property $(\mathrm{S}_k)$ at a point $x$ (resp.\ satisfies $(\mathrm{S}_k)$) if $\sh{O}_X$ satisfies $(\mathrm{S}_k)$ at the point $x$ (resp.\ satisfies $(\mathrm{S}_k)$).

For $\sh{F}$ to satisfy $(\mathrm{S}_k)$, it is necessary and sufficient that it satisfy it at every point.
If $U$ is an open subset of $X$ and if $\sh{F}$ satisfies $(\mathrm{S}_k)$, then so does $\sh{F}|U$; reciprocally, if $(U_\alpha)$ is an open covering of $X$ and if $\sh{F}|U_\alpha$ satisfies $(\mathrm{S}_k)$ for every $\alpha$, then $\sh{F}$ satisfies $(\mathrm{S}_k)$.
\end{definition}

\begin{remarks}[5.7.3]
\label{IV.5.7.3}
\oldpage[IV-2]{104}
\medskip\noindent\emph{(i)}
Recall that one always has $\operatorname{prof}(\sh{F}_x)\leq\dim(\sh{F}_x)$ $(\mathbf{0}$, 16.4.5.1) if $\sh{F}_x\neq 0$.
To say that $\sh{F}$ possesses property $(\mathrm{S}_k)$ thus means that one has $\operatorname{prof}(\sh{F}_x)\geq k$ \emph{except} at points $x\in X$ such that $\dim(\sh{F}_x)<k$, and that at these latter points one has $\dim(\sh{F}_x)=\operatorname{prof}(\sh{F}_x)$, that is to say $(\mathbf{0}$, 16.5.1) that $\sh{F}$ is an $\sh{O}_X$-Module de Cohen-Macaulay at these points; we note that only at points where $\dim(\sh{F}_x)=k$ does one have $\operatorname{prof}(\sh{F}_x)=k$, hence $\sh{F}$ is an $\sh{O}_X$-Module de Cohen-Macaulay at these points.
To say that $\sh{F}$ possesses $(\mathrm{S}_k)$ \emph{for all} $k$ signifies that $\sh{F}$ is an $\sh{O}_X$-\emph{Module de Cohen-Macaulay}.
It is clear that for $k'\geq k$, property $(\mathrm{S}_{k'})$ implies $(\mathrm{S}_k)$; for $k\leq 0$, \emph{every} $\sh{O}_X$-Module coherent satisfies $(\mathrm{S}_k)$.

\medskip\noindent\emph{(ii)}
To verify condition \sref{IV.5.7.2.1}, one can restrict to the case $x\in\operatorname{Supp}(\sh{F})$; in the contrary case indeed $\dim(\sh{F}_x)=-\infty$ $(\mathbf{0}$, 14.1.2).

\medskip\noindent\emph{(iii)}
If $X=\operatorname{Spec}(A)$, where $A$ is a Noetherian ring, and $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module of finite type, one says that $M$ possesses property $(\mathrm{S}_k)$ if $\sh{F}$ possesses this property.
For a locally Noetherian prescheme $X$, to say that $\sh{F}$ possesses property $(\mathrm{S}_k)$ \emph{at a point} $x\in X$ signifies that if one sets $Y=\operatorname{Spec}(\sh{O}_x)$, the $\sh{O}_Y$-Module $\widetilde{\sh{F}_x}$ possesses property $(\mathrm{S}_k)$; one notes that condition \sref{IV.5.7.2.1} in general \emph{does not entail} \sref{IV.5.7.2.2} for every generisation $x'$ of $x$, since there is no inequality relation between $\operatorname{prof}(\sh{F}_x)$ and $\operatorname{prof}(\sh{F}_{x'})$ $(\mathbf{0}$, 16.4.6).

\medskip\noindent\emph{(iv)}
It follows immediately from the definition that if $\sh{F}$ satisfies $(\mathrm{S}_k)$ at a point $x$, it also satisfies $(\mathrm{S}_k)$ at every point $x'$ which is a generisation of $x$.

\medskip\noindent\emph{(v)}
Property $(\mathrm{S}_k)$ is especially important for $k=1$ and $k=2$; it was introduced for $k=2$ by Serre, to express his criterion of normality (cf.\ \sref{IV.5.8.5}).

\medskip\noindent\emph{(vi)}
Let $X$ be a locally Noetherian prescheme, $j:Y\to X$ the canonical closed immersion, $\sh{G}$ an $\sh{O}_Y$-Module coherent.
It is clear that for every $x\in Y$, $\dim(\sh{G}_x)=\dim((j_*(\sh{G}))_x)$ and $\operatorname{prof}(\sh{G}_x)=\operatorname{prof}((j_*(\sh{G}))_x)$, whence $\operatorname{coprof}(\sh{G}_x)=\operatorname{coprof}((j_*(\sh{G}))_x)$.
For $\sh{G}$ to satisfy $(\mathrm{S}_k)$, it suffices that $j_*(\sh{G})$ satisfy $(\mathrm{S}_k)$.
\end{remarks}

\begin{proposition}[5.7.4]
\label{IV.5.7.4}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ an $\sh{O}_X$-Module coherent.
Set $S=\operatorname{Supp}(\sh{F})$, and for every $n\geq 0$, denote by $Z_n$ the set of $z\in X$ such that $\operatorname{coprof}(\sh{F}_z)>n$, so that $Z_n\subset S$.
\begin{enumerate}[label=\emph{(\roman*)}]
  \item For $\sh{F}$ to satisfy property $(\mathrm{S}_k)$, it is necessary and sufficient that, for every $n\geq 0$, we have
  \begin{equation}
  \label{IV.5.7.4.1}
    \codim(Z_n,S)>n+k.\tag{5.7.4.1}
  \end{equation}
  \item Suppose furthermore that the $Z_n$ are closed in $X$.
  Then, for $\sh{F}$ to satisfy property $(\mathrm{S}_k)$ at a point $x\in S$, it is necessary and sufficient that, for every $n\geq 0$, we have
  \begin{equation}
  \label{IV.5.7.4.2}
    \codim_x(Z_n,S)>n+k.\tag{5.7.4.2}
  \end{equation}
\end{enumerate}
\end{proposition}

\begin{proof}
\medskip\noindent\emph{(i)}
By definition \sref{IV.5.1.3}, $\codim(Z_n,S)=\inf_{z\in Z_n}(\dim(\sh{O}_{S,z}))$ and the inequality \sref{IV.5.7.4.1} means that, for every $z\in X$, and every $n\geq 0$, the relation
\[
  \dim(\sh{F}_z)-\operatorname{prof}(\sh{F}_z)\geq n+1
\]
implies the relation
\[
  \dim(\sh{F}_z)\geq n+k+1.
\]
But if one sets $a=\dim(\sh{F}_z)$, $b=\operatorname{prof}(\sh{F}_z)$, we have $b\leq a$, and to say that for every $n\geq 0$, $b\leq a-n-1$ implies $k\leq a-n-1$ is equivalent to saying $b\geq\inf(k,a)$, whence the proposition.

\medskip\noindent\emph{(ii)}
The argument is the same as in~(i), except that one restricts to $z\in X$ which are generisations of $x$, and takes account of \sref{IV.5.1.3.2}.
\end{proof}

\begin{proposition}[5.7.5]
\label{IV.5.7.5}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ an $\sh{O}_X$-Module coherent.
For $\sh{F}$ to satisfy $(\mathrm{S}_k)$ $(k\geq 1)$, it is necessary and sufficient that for every integer $r$ such that $0\leq r<k$, every open subset $U$ of $X$, and every $(\sh{F}|U)$-regular suite $(f_i)_{1\leq i\leq r}$ of sections of $\sh{O}_X$ above $U$, $(\sh{F}|U)/(\sum_{i=1}^r f_i(\sh{F}|U))$ be without embedded associated prime cycle.
\end{proposition}

\begin{proof}
Let us first prove the proposition for $k=1$; it then says that \emph{for $\sh{F}$ to satisfy} $(\mathrm{S}_1)$, \emph{it is necessary and sufficient that $\sh{F}$ be without embedded associated prime cycle}.
Indeed, to say that $\sh{F}$ satisfies $(\mathrm{S}_1)$ means that at the points $x\in\operatorname{Supp}(\sh{F})$ such that $\dim(\sh{F}_x)>0$, one has $\operatorname{prof}(\sh{F}_x)\geq 1$ (since at the other points of $\operatorname{Supp}(\sh{F})$ one has $\dim(\sh{F}_x)=0$, hence $\operatorname{prof}(\sh{F}_x)=0$); but to say that $\operatorname{prof}(\sh{F}_x)\geq 1$ means that $\mathfrak{m}_x$ \emph{is not associated} to $\sh{F}$ $(\mathbf{0}$, 16.4.6,~(i)), or equivalently that $x$ is not associated to $\sh{F}$ (3.1.1); on the other hand, if $S$ is a sub-prescheme of $X$ having for underlying space $\operatorname{Supp}(\sh{F})$, by \sref{IV.5.1.12.1} $\dim(\sh{F}_x)=\dim(\sh{O}_{S,x})$; to say that $\dim(\sh{F}_x)>0$ means that $x$ is not a maximal point of $\operatorname{Supp}(\sh{F})$ \sref{IV.5.1.2}, whence the conclusion.

In the second place, let us show that, for $k$ arbitrary $>1$, the condition of the statement is \emph{necessary}.
One can restrict to considering the case $U=X$, and our assertion will be proved (by induction on $k$ and by virtue of the first part of the argument) if we show that when $\sh{F}$ satisfies $(\mathrm{S}_k)$ $(k>1)$ and $f$ is an $\sh{F}$-regular section of $\sh{O}_X$ above $X$, then $\sh{F}/f\sh{F}$ satisfies $(\mathrm{S}_{k-1})$ (i.e.\ satisfies \sref{IV.5.7.2.1} where $k$ is replaced by $k-1$).
Indeed, for every $x\in X$, $f_x$ is an $\sh{F}_x$-regular element; if it is invertible, $\sh{F}_x/f_x\sh{F}_x=0$ and the conclusion is trivial.
If on the contrary $f_x\in\mathfrak{m}_x$, one knows that $\operatorname{prof}(\sh{F}_x/f_x\sh{F}_x)=\operatorname{prof}(\sh{F}_x)-1$ $(\mathbf{0}$, 16.4.6,~(i)), and $\dim(\sh{F}_x/f_x\sh{F}_x)=\dim(\sh{F}_x)-1$ $(\mathbf{0}$, 16.3.4), and our assertion follows.

Let us finally prove that for $k>1$, the condition of the statement is \emph{sufficient}.
We proceed by induction on $k$; let $x$ be a point of $X$, and suppose first that $\dim(\sh{F}_x)\geq k$.
The induction hypothesis implies that $\sh{F}$ satisfies $(\mathrm{S}_{k-1})$, hence $\operatorname{prof}(\sh{F}_x)\geq k-1$; taking account of $(\mathbf{0}$, 15.2.4), there is thus an open neighbourhood $U$ of $x$ and an $(\sh{F}|U)$-regular suite $(f_i)_{1\leq i\leq k-1}$ of sections of $\sh{O}_X$ above $U$, such that $(f_i)_x\in\mathfrak{m}_x$ for $1\leq i\leq k-1$.
The hypothesis implies that $\sh{G}=(\sh{F}|U)/(\sum_{i=1}^{k-1}f_i(\sh{F}|U))$ is without embedded associated prime cycle; but $\dim(\sh{G}_x)=\dim(\sh{F}_x)-(k-1)\geq 1$ $(\mathbf{0}$, 16.3.4), hence $x$ is not associated to $\sh{G}$; so $\operatorname{prof}(\sh{G}_x)\geq 1$ $(\mathbf{0}$, 16.4.6), and since $\operatorname{prof}(\sh{G}_x)=\operatorname{prof}(\sh{F}_x)-(k-1)$ $(\mathbf{0}$, 16.4.6), we have $\operatorname{prof}(\sh{F}_x)\geq k$.
Suppose in the second place that $\dim(\sh{F}_x)=r<k$; since $\sh{F}$ satisfies $(\mathrm{S}_{k-1})$, we have $\operatorname{prof}(\sh{F}_x)\geq\inf(k-1,r)=r$, and this finishes the proof.
\end{proof}

\begin{corollary}[5.7.6]
\label{IV.5.7.6}
Suppose that $\sh{F}$ satisfies $(\mathrm{S}_k)$ $(k\geq 1)$; if $(f_i)$ $(1\leq i\leq r)$ is an $\sh{F}$-regular suite of sections of $\sh{O}_X$ above $X$ and if $r<k$, $\sh{F}/(\sum_{i=1}^r f_i\sh{F})$ satisfies $(\mathrm{S}_{k-r})$.
\end{corollary}

\begin{proof}
This results immediately from \sref{IV.5.7.5}.
\end{proof}

\begin{corollary}[5.7.7]
\label{IV.5.7.7}
For $\sh{F}$ to satisfy $(\mathrm{S}_2)$, it is necessary and sufficient that $\sh{F}$ be without embedded associated prime cycle and that for every open subset $U$ of $X$, every $(\sh{F}|U)$-regular section $f$ of $\sh{O}_X$ above $U$, $(\sh{F}|U)/f(\sh{F}|U)$ be without embedded associated prime cycle.
\end{corollary}

\begin{remarks}[5.7.8]
\label{IV.5.7.8}
Let $X$ be a locally Noetherian prescheme \emph{of dimension}~$1$; it then amounts to the same to say that $X$ is a Cohen-Macaulay prescheme, or that it satisfies $(\mathrm{S}_1)$, or that it satisfies one of the properties $(\mathrm{S}_n)$ for $n\geq 1$, by virtue of definitions \sref{IV.5.7.1} and \sref{IV.5.7.2}.

\begin{proposition}[5.7.9]
\label{IV.5.7.9}
Let $A$, $B$ be two Noetherian rings, $\rho:A\to B$ a homomorphism of rings, $M$ a $B$-module such that $M_{[\mathfrak{p}]}$ is an $A$-module of finite type.
Let $\mathfrak{p}$ be a prime ideal of $A$; the prime ideals of $B$ above $\mathfrak{p}$ and belonging to $\operatorname{Supp}(M)$ are finite in number, and if $(\mathfrak{q}_i)_{1\leq i\leq n}$ is the family of these ideals, we have
\begin{equation}
\label{IV.5.7.9.1}
  \dim_{A_\mathfrak{p}}(M_\mathfrak{p})=\sup_i\dim_{B_{\mathfrak{q}_i}}(M_{\mathfrak{q}_i})\tag{5.7.9.1}
\end{equation}
\begin{equation}
\label{IV.5.7.9.2}
  \operatorname{prof}_{A_\mathfrak{p}}(M_\mathfrak{p})=\inf_i\operatorname{prof}_{B_{\mathfrak{q}_i}}(M_{\mathfrak{q}_i}).\tag{5.7.9.2}
\end{equation}
\end{proposition}

\begin{proof}
If $\mathfrak{b}$ is the annihilator of $M$, $B/\mathfrak{b}$ identifies with a sub-$A$-module of $\operatorname{End}_A(M_{[\mathfrak{p}]})$, hence is of finite type since $A$ is Noetherian; so there are only a finite number of prime ideals of $B$ above $\mathfrak{p}$ and containing $\mathfrak{b}$, and these are precisely those belonging to $\operatorname{Supp}(M)$.
Replacing $B$ by $B/\mathfrak{b}$, which does not change the second members of \sref{IV.5.7.9.1} and \sref{IV.5.7.9.2} (by $(\mathbf{0}$, 16.1.9) and $(\mathbf{0}$, 16.4.8)), one can thus suppose that $B$ is a finite $A$-algebra.
Set $S=A-\mathfrak{p}$ and $B'=S^{-1}B$; $B'$ is a semi-local Noetherian ring whose maximal ideals are $S^{-1}\mathfrak{q}_i$ $(1\leq i\leq n)$ and since $M_\mathfrak{p}$ is an $A_\mathfrak{p}$-module of finite type, $\dim_{A_\mathfrak{p}}(M_\mathfrak{p})=\dim_{B'}(M_\mathfrak{p})$ $(\mathbf{0}$, 16.1.9).
This being so, relation \sref{IV.5.7.9.1} becomes a particular case of $(\mathbf{0}$, 16.1.7.4).
To prove relation \sref{IV.5.7.9.2}, we immediately reduce as in $(\mathbf{0}$, 16.4.8) to the case $\operatorname{prof}_{A_\mathfrak{p}}(M_\mathfrak{p})=0$, and the same argument as in $(\mathbf{0}$, 16.4.8) shows that $M_\mathfrak{p}$ contains a sub-$B'$-module of \emph{finite length}, and hence also a \emph{simple} sub-$B'$-module; but such a sub-module is necessarily isomorphic to the residue field of one of the $B_{\mathfrak{q}_i}$, hence there is at least one index $i$ such that $\operatorname{prof}_{B_{\mathfrak{q}_i}}(M_{\mathfrak{q}_i})=0$ $(\mathbf{0}$, 16.4.6), which finishes the proof.
\end{proof}

\begin{corollary}[5.7.10]
\label{IV.5.7.10}
Suppose verified the hypotheses of \sref{IV.5.7.9}, and suppose in addition that $A$ is a \emph{local} ring; then, for $M$ to be an $A$-module de Cohen-Macaulay, it is necessary and sufficient that, for all prime ideals $\mathfrak{q}_i$ of $B$ above the maximal ideal of $A$, $M_{\mathfrak{q}_i}$ be a $B_{\mathfrak{q}_i}$-module de Cohen-Macaulay, and moreover that all the $\dim_{B_{\mathfrak{q}_i}}(M_{\mathfrak{q}_i})$ be equal.
\end{corollary}

\begin{proof}
It follows from \sref{IV.5.7.9.1} and \sref{IV.5.7.9.2} that these conditions are equivalent to $\dim_A(M)=\operatorname{prof}_A(M)$.
\end{proof}

\begin{corollary}[5.7.11]
\label{IV.5.7.11}
Suppose verified the hypotheses of \sref{IV.5.7.9}.
\begin{enumerate}[label=\emph{(\roman*)}]
  \item If $M_{[\mathfrak{p}]}$ possesses property $(\mathrm{S}_k)$, then so does $M$.
  \item Suppose that for every pair of prime ideals $\mathfrak{q}$, $\mathfrak{q}'$ of $B$ such that $\rho^{-1}(\mathfrak{q})=\rho^{-1}(\mathfrak{q}')$, $\dim_{B_\mathfrak{q}}(M_\mathfrak{q})=\dim_{B_{\mathfrak{q}'}}(M_{\mathfrak{q}'})$.
  Then, if $M$ possesses property $(\mathrm{S}_k)$, so does $M_{[\mathfrak{p}]}$.
\end{enumerate}
\end{corollary}

\begin{proof}
This results immediately from \sref{IV.5.7.9.1} and \sref{IV.5.7.9.2} and the definition of property $(\mathrm{S}_k)$.
\end{proof}

\subsection{Regular preschemes and property $(\mathrm{R}_k)$. Serre's normality criterion}
\label{subsection:IV.5.8}

\begin{env}[5.8.1]
\label{IV.5.8.1}
Recall $(\mathbf{0_I}$, 4.1.4) that a ringed space $(X,\sh{O}_X)$ is called \emph{regular} at a point $x\in X$ if $\sh{O}_x$ is a \emph{regular local ring}.
When dealing with preschemes, we will only use this terminology when $X$ is \emph{locally Noetherian}.
\end{env}

\begin{definition}[5.8.2]
\label{IV.5.8.2}
A locally Noetherian prescheme $X$ is said to be \emph{regular in codimension $\leq k$}, or \emph{possesses property $(\mathrm{R}_k)$} if the set of points where $X$ is \emph{not} regular is of codimension $>k$ \emph{(}that is to say \sref{IV.5.1.3}, for every $x\in X$, the relation $\dim(\sh{O}_x)\leq k$ implies that $\sh{O}_x$ is regular\emph{)}.
\end{definition}

To say that $X$ is regular signifies that $X$ possesses property $(\mathrm{R}_k)$ for every $k$.
If $X=\operatorname{Spec}(A)$, where $A$ is a Noetherian ring, one says that $A$ possesses property $(\mathrm{R}_k)$ if $X$ possesses this property; to say that $X$ is regular signifies that the ring $A$ is \emph{regular} $(\mathbf{0}$, 17.3.6).
For a locally Noetherian prescheme $X$, we say that $X$ possesses property $(\mathrm{R}_k)$ \emph{at a point} $x\in X$ if the local ring $\sh{O}_x$ possesses property $(\mathrm{R}_k)$; this means that for every generisation $x'$ of $x$ in $X$, the relation $\dim(\sh{O}_{x'})\leq k$ implies that $\sh{O}_{x'}$ is regular.
To say that $X$ is regular at a point $x$ is equivalent to saying that $X$ satisfies $(\mathrm{R}_n)$ for \emph{every} $n\geq 0$ at the point $x$, by virtue of $(\mathbf{0}$, 17.3.6).

\begin{proposition}[5.8.3]
\label{IV.5.8.3}
If $k$ is a field, $X$ a prescheme locally of finite type over $k$, then, for every $x\in X$, there exists an open neighbourhood of $x$ in $X$ isomorphic to a sub-scheme of a regular $k$-scheme.
\end{proposition}

\begin{proof}
Indeed, there is an affine open neighbourhood $U$ of $x$ isomorphic to a $k$-scheme of the form $\operatorname{Spec}(A)$, where $A$ is a $k$-algebra of finite type; $A$ is therefore isomorphic to a quotient of a polynomial algebra $k[\mathrm{T}_1,\ldots,\mathrm{T}_n]$, and it is known that this last algebra is a regular ring $(\mathbf{0}$, 17.3.7).
\end{proof}

\begin{env}[5.8.4]
\label{IV.5.8.4}
By virtue of \sref{IV.5.8.2}, to say that $X$ possesses property $(\mathrm{R}_0)$ signifies that for every maximal point $x$ of $X$, $\sh{O}_x$ is a \emph{field} $(\mathbf{0}$, 17.1.4), or equivalently that $X$ is \emph{reduced} at this point.
Since the set $U$ of $x\in X$ where $X$ is reduced is \emph{open} (the nilradical of $X$ being an $\sh{O}_X$-Module coherent \sref[I]{I.6.1.1} and $(\mathbf{0_I}$, 5.2.2)), it amounts to the same to say that $X$ possesses property $(\mathrm{R}_0)$ or that the set $U$ is \emph{everywhere dense}.
Consequently:
\end{env}

\begin{proposition}[5.8.5]
\label{IV.5.8.5}
For a locally Noetherian prescheme $X$ to be reduced, it is necessary and sufficient that it satisfy properties $(\mathrm{S}_1)$ and $(\mathrm{R}_0)$.
\end{proposition}

\begin{proof}
Taking into account \sref{IV.5.7.5}, this results from the preceding remark and from (3.2.1).
\end{proof}

\begin{theorem}[5.8.6]
\label{IV.5.8.6}
\emph{(Serre's criterion).}
Let $X$ be a locally Noetherian prescheme.
For $X$ to be \emph{normal}, it is necessary and sufficient that $X$ satisfy properties $(\mathrm{S}_2)$ and $(\mathrm{R}_1)$, that is to say that, for every $x\in X$, the following properties hold:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item If $\dim(\sh{O}_x)\leq 1$, $\sh{O}_x$ is regular \emph{(}that is, is a field or a discrete valuation ring\emph{)}.
  \item If $\dim(\sh{O}_x)\geq 2$, then $\operatorname{prof}(\sh{O}_x)\geq 2$.
\end{enumerate}
\end{theorem}

\begin{proof}
The conditions are \emph{necessary}.
Indeed, to say that $X$ is normal signifies that for every $x\in X$, $\sh{O}_x$ is a local Noetherian integrally closed ring.
If $\dim(\sh{O}_x)=0$ (resp.\ $\dim(\sh{O}_x)=1$), we conclude that $\sh{O}_x$ is a field since $\sh{O}_x$ is integral (resp.\ that $\sh{O}_x$ is a discrete valuation ring, by virtue of \sref[II]{II.7.1.6}).
On the other hand, for every nonzero element $f_x\neq 0$ of $\sh{O}_x$, one knows (Bourbaki, \emph{Alg.~comm.}, chap.~VII, \S1, no.~4, prop.~8) that the prime ideals associated to $\sh{O}_x/f_x\sh{O}_x$ are non-immersed, hence $\sh{O}_x$ satisfies $(\mathrm{S}_2)$ \sref{IV.5.7.7}.

The conditions are \emph{sufficient}.
The question being local, one can moreover suppose that $X=\operatorname{Spec}(A)$, where $A$ is a local Noetherian ring \emph{reduced} \sref[I]{I.5.1.4}; if $R$ is the total ring of fractions of $A$, $R$ is a direct product of a finite number of fields, and (taking account of \sref[II]{II.6.3.6}), it suffices to prove that $A$ is \emph{integrally closed} in $R$.
So let $h=f/g$ be an element of $R$ integral over $A$, $g$ and $f$ being elements of $A$ such that $g$ is a non-zero-divisor of $0$.
We have a relation of the form
\begin{equation}
\label{IV.5.8.6.1}
  f^n+\sum_{i=1}^n a_if^{n-i}g^i=0\qquad\text{with $a_i\in A$ for $1\leq i\leq n$}.\tag{5.8.6.1}
\end{equation}

Let $\mathfrak{p}$ be a prime ideal of $A$ such that $\dim(A_\mathfrak{p})=1$; if $f_\mathfrak{p}$ and $g_\mathfrak{p}$ are the images of $f$ and $g$ in $A_\mathfrak{p}$, it follows from \sref{IV.5.8.6.1} that $f_\mathfrak{p}/g_\mathfrak{p}$ (which belongs to the total ring of fractions of $A_\mathfrak{p}$, since $g_\mathfrak{p}$ is a non-zero-divisor of~$0$ in $A_\mathfrak{p}$ by flatness $(\mathbf{0_I}$, 5.3.1)) is integral over $A_\mathfrak{p}$; but since $A_\mathfrak{p}$ is regular, hence integrally closed, we have $f_\mathfrak{p}/g_\mathfrak{p}\in A_\mathfrak{p}$.
But hypothesis $(\mathrm{S}_2)$ implies \sref{IV.5.7.7} that $g$ is a non-zero-divisor and that $A/gA$ has no non-immersed associated prime ideals $\mathfrak{p}_i$ $(1\leq i\leq n)$; or, $gA$ is the intersection of the primary ideals $\mathfrak{q}_i$ corresponding to the $\mathfrak{p}_i$, and from what we have just seen, the $\mathfrak{q}_i$ are the inverse images in $A$, by the homomorphisms $A\to A_{\mathfrak{p}_i}$, of the ideals $(gA)_{\mathfrak{p}_i}$ \emph{(Bourbaki, Alg.~comm., chap.~IV, \S2, no.~3, prop.~5)}.
But by the Hauptidealsatz $(\mathbf{0}$, 16.3.2) $\dim(A_{\mathfrak{p}_i})=1$ for $1\leq i\leq n$, hence $(fA)_{\mathfrak{p}_i}\subset(gA)_{\mathfrak{p}_i}$ for every $i$ by what precedes; since $fA$ is contained in the intersection of the inverse images of $(fA)_{\mathfrak{p}_i}$ $(1\leq i\leq n)$, we have $fA\subset gA$, that is to say $f/g\in A$.
\end{proof}

\subsection{$Z$-pure and $Z$-closed Modules}
\label{subsection:IV.5.9}

Some of the notions and results of this section and the following are particular cases of notions and results developed in chapter~III in the theory of local cohomology.
For the convenience of the reader, we give here an independent treatment.

\begin{env}[5.9.1]
\label{IV.5.9.1}
Let $X$ be a locally Noetherian prescheme, $Z$ a subset of $X$ \emph{stable by specialisation}: this means that for every finite subset $M$ of $Z$, the closure of $M$ is contained in $Z$, and hence $Z$ is the union of an increasing filtering family $(Z_\alpha)$ of closed subsets of $X$; conversely, it is clear that such a union is stable by specialisation.

Set $U_\alpha=X-Z_\alpha$, so that $X-Z$ is the intersection of the filtering decreasing family of opens $U_\alpha$; let $i_\alpha:U_\alpha\to X$ be the canonical injection, and for $U_\alpha\supset U_\beta$, let $i_{\alpha\beta}:U_\beta\to U_\alpha$ be the canonical injection, so that $i_\beta=i_\alpha\circ i_{\alpha\beta}$.
Let $\sh{F}$ be an $\sh{O}_X$-Module (not necessarily quasi-coherent); from the canonical homomorphism $(\mathbf{0_I}$, 4.4.3.2)
\[
  \sh{F}|U_\alpha\to(i_{\alpha\beta})_*(\sh{F}|U_\beta)
\]
we deduce, by application of the functor $(i_\alpha)_*$, a homomorphism
\[
  \rho_{\beta\alpha}:(i_\alpha)_*(\sh{F}|U_\alpha)\to(i_\beta)_*(\sh{F}|U_\beta)
\]
and one verifies immediately that $\rho_{\gamma\alpha}=\rho_{\gamma\beta}\circ\rho_{\beta\alpha}$ for $U_\alpha\supset U_\beta\supset U_\gamma$; in other words, the $\sh{O}_X$-Modules $(i_\alpha)_*(\sh{F}|U_\alpha)$ form an \emph{inductive system} for the homomorphisms $\rho_{\beta\alpha}$.
We set
\begin{equation}
\label{IV.5.9.1.1}
  \mathscr{H}^0_{X/Z}(\sh{F})=\varinjlim_\alpha(i_\alpha)_*(\sh{F}|U_\alpha).\tag{5.9.1.1}
\end{equation}

This $\sh{O}_X$-Module does not depend on the increasing family $(Z_\alpha)$ of closed sets whose union is $Z$: indeed, let $V$ be a Noetherian open subset of $X$; one knows (G,~II, 3.10.1) that in the category of $\sh{O}_V$-Modules, the functor $\sh{G}\mapsto\Gamma(V,\sh{G})$ commutes with inductive limits; so by virtue of \sref{IV.5.9.1.1}
\begin{equation}
\label{IV.5.9.1.2}
  \Gamma(V,\mathscr{H}^0_{X/Z}(\sh{F}))=\varinjlim_\alpha\Gamma(V\cap U_\alpha,\sh{F}).\tag{5.9.1.2}
\end{equation}

\oldpage[IV-2]{110}
Let then $(Z'_\lambda)$ be a second increasing filtering family of closed sets of $X$, with union $Z$; $V\cap Z_\alpha$ is therefore the union of the $V\cap Z_\alpha\cap Z'_\lambda$; but $V\cap Z_\alpha$ is locally closed in $X$, hence every closed irreducible subset of $V\cap Z_\alpha$ admits a generic point; since the $V\cap Z_\alpha\cap Z'_\lambda$ are closed in $V\cap Z_\alpha$ and form (for $\alpha$ fixed) an increasing filtering family, there exists an index $\lambda$ such that $V\cap Z_\alpha\cap Z'_\lambda=V\cap Z_\alpha$ $(\mathbf{0_{III}}$, 9.2.4), in other words $V\cap Z_\alpha\subset V\cap Z'_\lambda$.
This proves that the filtering decreasing families $V\cap U_\alpha$, $V\cap U'_\lambda$ (where $U'_\lambda=X-Z'_\lambda$) are cofinal in each other, whence our assertion, by virtue of \sref{IV.5.9.1.2}.

We note that the set $Z$ is not necessarily constructible: one has an example of this by taking $X=\operatorname{Spec}(A)$, where $A$ is an integral Noetherian ring having an infinity of maximal ideals, and $Z$ the complement of the generic point of $X$.

If $Z$ is closed and if $i:X-Z\to X$ is the canonical injection, we have
\begin{equation}
\label{IV.5.9.1.3}
  \mathscr{H}^0_{X/Z}(\sh{F})=i_*(\sh{F}|X-Z)\tag{5.9.1.3}
\end{equation}
and in particular, for $Z=\varnothing$, $\mathscr{H}^0_{X/\varnothing}(\sh{F})=\sh{F}$.
\end{env}

\begin{proposition}[5.9.2]
\label{IV.5.9.2}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item The functor $\sh{F}\mapsto\mathscr{H}^0_{X/Z}(\sh{F})$ is exact on the left.
  \item If $\sh{F}$ is quasi-coherent, then so is $\mathscr{H}^0_{X/Z}(\sh{F})$.
\end{enumerate}
\end{proposition}

\begin{proof}
Assertion~(i) results from the definition \sref{IV.5.9.1.1}, from the fact that $(i_\alpha)_*$ is a left exact functor and from the fact that the inductive limit preserves exactness in the category of $\sh{O}_X$-Modules.
Assertion~(ii) results from \sref[I]{I.9.2.2} and the fact that an inductive limit of quasi-coherent $\sh{O}_X$-Modules is quasi-coherent \sref[I]{I.1.3.9}.
\end{proof}

\begin{remark}[5.9.3]
\label{IV.5.9.3}
If $\sh{F}$ is an $\sh{O}_X$-Algebra, then so is $\mathscr{H}^0_{X/Z}(\sh{F})$ $(\mathbf{0_I}$, 4.2.4); in particular $\mathscr{H}^0_{X/Z}(\sh{O}_X)$ is a quasi-coherent $\sh{O}_X$-Algebra, and for every $\sh{O}_X$-Module $\sh{F}$, $\mathscr{H}^0_{X/Z}(\sh{F})$ is a $\mathscr{H}^0_{X/Z}(\sh{O}_X)$-Module, which is quasi-coherent if $\sh{F}$ is quasi-coherent \sref[I]{I.9.6.1}.
More particularly, suppose that $X=\operatorname{Spec}(A)$, where $A$ is integral and Noetherian; then $\mathscr{H}^0_{X/Z}(\sh{O}_X)$ is the $\sh{O}_X$-Algebra $\widetilde{B}$, where
\begin{equation}
\label{IV.5.9.3.1}
  B=\bigcap_{\mathfrak{p}\in X-Z}A_\mathfrak{p}.\tag{5.9.3.1}
\end{equation}
This results indeed from \sref{IV.5.9.1.2} and \sref[I]{I.8.2.1.1}.
\end{remark}

\begin{proposition}[5.9.4]
\label{IV.5.9.4}
Let $X$ be a locally Noetherian prescheme, $Z$ a subset of $X$ stable by specialisation, $X'$ a locally Noetherian prescheme, $f:X'\to X$ a flat morphism.
Then $Z'=f^{-1}(Z)$ is stable by specialisation and for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, we have a canonical isomorphism
\begin{equation}
\label{IV.5.9.4.1}
  f^*(\mathscr{H}^0_{X/Z}(\sh{F}))\xrightarrow{\sim}\mathscr{H}^0_{X'/Z'}(f^*(\sh{F})).\tag{5.9.4.1}
\end{equation}
\end{proposition}

\begin{proof}
\oldpage[IV-2]{111}
Indeed, with the notations of \sref{IV.5.9.1}, $Z'_\alpha=f^{-1}(Z_\alpha)$ is closed in $X'$ and $Z'$ is the union of the $Z'_\alpha$; moreover, $(i_\alpha)_{(X')}$ is the canonical injection $i'_\alpha:U'_\alpha\to X'$, where $U'_\alpha=X'-Z'_\alpha=f^{-1}(U_\alpha)$.
Since $f$ is flat, one knows (2.3.1) that the canonical homomorphism $f^*((i_\alpha)_*(\sh{F}|U_\alpha))\to(i'_\alpha)_*(f^*(\sh{F})|U'_\alpha)$ is bijective; since, for $\alpha\leq\beta$, the diagram
\[
\xymatrix{
  f^*((i_\alpha)_*(\sh{F}|U_\alpha)) \ar[r]^-{\sim} \ar[d] & (i'_\alpha)_*(f^*(\sh{F})|U'_\alpha) \ar[d] \\
  f^*((i_\beta)_*(\sh{F}|U_\beta)) \ar[r]^-{\sim} & (i'_\beta)_*(f^*(\sh{F})|U'_\beta)
}
\]
is commutative, passing to the limit one obtains a canonical isomorphism $\varinjlim_\alpha(f^*((i_\alpha)_*(\sh{F}|U_\alpha)))\xrightarrow{\sim}\mathscr{H}^0_{X'/Z'}(f^*(\sh{F}))$.
But since the functor $f^*$ commutes with inductive limits $(\mathbf{0_I}$, 4.3.2), this gives by definition the isomorphism \sref{IV.5.9.4.1} sought.
\end{proof}

\begin{corollary}[5.9.5]
\label{IV.5.9.5}
Under the hypotheses of \sref{IV.5.9.4}, if $\mathscr{H}^0_{X/Z}(\sh{F})$ is coherent, then so is $\mathscr{H}^0_{X'/Z'}(f^*(\sh{F}))$.
The converse is true when $f$ is a faithfully flat and quasi-compact morphism.
\end{corollary}

\begin{proof}
The first assertion results from \sref{IV.5.9.4.1} and $(\mathbf{0_I}$, 5.3.11); the second amounts to saying that if $\mathscr{H}^0_{X'/Z'}(f^*(\sh{F}))$ is of finite type, then so is $\mathscr{H}^0_{X/Z}(\sh{F})$; this results from \sref{IV.5.9.4.1} and (2.5.2).
\end{proof}

\begin{corollary}[5.9.6]
\label{IV.5.9.6}
Let $X$ be a locally Noetherian prescheme, $Z$ a subset of $X$ stable by specialisation, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent.
For every $x\in X$, setting $X_x=\operatorname{Spec}(\sh{O}_x)$, $Z_x=Z\cap X_x$, we have a canonical functorial isomorphism
\begin{equation}
\label{IV.5.9.6.1}
  (\mathscr{H}^0_{X/Z}(\sh{F}))_x\xrightarrow{\sim}\mathscr{H}^0_{X_x/Z_x}(\widetilde{\sh{F}_x}).\tag{5.9.6.1}
\end{equation}
\end{corollary}

\begin{proof}
It suffices to apply \sref{IV.5.9.4} to the canonical morphism $X_x\to X$, which is flat, and \sref[I]{I.1.6.5}.
\end{proof}

\begin{env}[5.9.7]
\label{IV.5.9.7}
With the notations of \sref{IV.5.9.1}, for every $\alpha$ one has a canonical functorial homomorphism $\sh{F}\to(i_\alpha)^*(\sh{F}|U_\alpha)$ $(\mathbf{0_I}$, 4.4.3.2), and these homomorphisms form an inductive system; passing to the inductive limit, one deduces a canonical functorial homomorphism
\begin{equation}
\label{IV.5.9.7.1}
  \rho_{X/Z}:\sh{F}\to\mathscr{H}^0_{X/Z}(\sh{F}).\tag{5.9.7.1}
\end{equation}
\end{env}

\begin{proposition}[5.9.8]
\label{IV.5.9.8}
Let $X$ be a locally Noetherian prescheme, $Z$ a subset of $X$ stable by specialisation, $\sh{F}$ an $\sh{O}_X$-Module.
The following properties are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item The homomorphism $\rho_{X/Z}$ \sref{IV.5.9.7.1} is injective \emph{(}resp.\ bijective\emph{)}.
  \item For every Noetherian open subset $V$ of $X$, the homomorphism
  \[
    (\rho_{X/Z})_V:\Gamma(V,\sh{F})\to\varinjlim_\alpha\Gamma(V\cap U_\alpha,\sh{F})
  \]
  is injective \emph{(}resp.\ bijective\emph{)}.
  \item[\emph{a$'$)}] For every closed subset $T\subset Z$ of $X$, the canonical homomorphism $(\mathbf{0_I}$, 4.4.3.2)
  \[
    \sh{F}\to i_*(\sh{F}|X-T)
  \]
  \emph{(}where $i:X-T\to X$ is the canonical injection\emph{)} is injective \emph{(}resp.\ bijective\emph{)}.
  \item[\emph{b$'$)}] For every closed subset $T\subset Z$ of $X$ and every Noetherian open subset $V$ of $X$, the restriction homomorphism
  \[
    \Gamma(V,\sh{F})\to\Gamma(V\cap(X-T),\sh{F})
  \]
  is injective \emph{(}resp.\ bijective\emph{)}.
\end{enumerate}
\end{proposition}

\begin{proof}
\oldpage[IV-2]{112}
Taking account of \sref{IV.5.9.1.2}, the equivalence of \emph{a)} and \emph{b)} (resp.\ \emph{a$'$)} and \emph{b$'$)}) results from the definition of the functor $\Gamma$ and from the fact that it is left exact.
Since the homomorphism $(\rho_{X/Z})_V$ is the composite
\begin{equation}
\label{IV.5.9.8.1}
  \Gamma(V,\sh{F})\to\Gamma(V\cap(X-T),\sh{F})\to\varinjlim_\alpha\Gamma(V\cap U_\alpha,\sh{F})\tag{5.9.8.1}
\end{equation}
for every closed subset $T\subset Z$, if $(\rho_{X/Z})_V$ is injective, then so is $\Gamma(V,\sh{F})\to\Gamma(V\cap(X-T),\sh{F})$; on the other hand, the fact that \emph{b$'$)} implies \emph{b)} results from the definition of an inductive limit.
It remains to show that if $\rho_{X/Z}$ is bijective, then so is $\Gamma(V,\sh{F})\to\Gamma(V\cap(X-T),\sh{F})$, and for this it suffices, by virtue of \sref{IV.5.9.8.1}, to see that if $U'\subset U$ are two open sets contained in $V$ and containing $V\cap Z$, the restriction homomorphism $\Gamma(U,\sh{F})\to\Gamma(U',\sh{F})$ is injective; but this results from the fact that $\rho_{X/Z}$ is injective, by replacing in what precedes $V$ by $U$ and $V\cap(X-T)$ by $U'$.
The equivalence of \emph{a)} and \emph{a$'$)} follows from \sref{IV.5.9.1.3} and the fact that the filtering families $(U_\alpha)$ and $(X-T)_{T\subset Z,\,T\text{ closed}}$ are cofinal.
\end{proof}

\begin{definition}[5.9.9]
\label{IV.5.9.9}
Under the hypotheses of \sref{IV.5.9.8}, we say that $\sh{F}$ is \emph{Z-pure} (resp.\ \emph{Z-closed}) if the homomorphism $\rho_{X/Z}$ is injective (resp.\ bijective).

If $X=\Spec(A)$ is affine, $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module, we say that $M$ is Z-pure (resp.\ Z-closed) when $\sh{F}$ is Z-pure (resp.\ Z-closed).

We say that $\sh{F}$ is \emph{Z-pure} (resp.\ \emph{Z-closed}) \emph{at a point} $x\in X$ if (with the notations of \sref{IV.5.9.6}) $\sh{F}_x$ is $Z_x$-pure (resp.\ $Z_x$-closed); this amounts to the same, by virtue of \sref{IV.5.9.6}, as saying that the canonical homomorphism $\sh{F}_x\to(\mathscr{H}^0_{X/Z}(\sh{F}))_x$ is injective (resp.\ bijective).

We note that for every $x\in X-Z$, $\sh{F}$ is Z-closed at the point $x$, by virtue of \sref{IV.5.9.8}.
\end{definition}

\begin{corollary}[5.9.10]
\label{IV.5.9.10}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item Let $(V_\lambda)$ be an open covering of $X$. For $\sh{F}$ to be Z-pure \emph{(}resp.\ Z-closed\emph{)}, it is necessary and sufficient that for every $\lambda$, $\sh{F}|V_\lambda$ be $(Z\cap V_\lambda)$-pure \emph{(}resp.\ $(Z\cap V_\lambda)$-closed\emph{)}.
  \item Let $Z'$ be a subset of $Z$ stable under specialisation. If $\sh{F}$ is Z-pure \emph{(}resp.\ Z-closed\emph{)}, it is $Z'$-pure \emph{(}resp.\ $Z'$-closed\emph{)}.
\end{enumerate}
\end{corollary}

\begin{proof}
This follows immediately from \sref{IV.5.9.8}, \emph{b$'$)}.
\end{proof}

\begin{proposition}[5.9.11]
\label{IV.5.9.11}
Under the hypotheses of \sref{IV.5.9.8}, the $\sh{O}_X$-Modules $\Ker(\rho_{X/Z})$ and $\Coker(\rho_{X/Z})$ have their support contained in $Z$, and the $\sh{O}_X$-Module $\mathscr{H}^0_{X/Z}(\sh{F})$ is Z-closed.
Furthermore, if $u:\sh{F}\to\sh{F}'$ is a homomorphism of $\sh{O}_X$-Modules such that $\sh{F}'$ is Z-closed, $u$ factors in a unique way as $\sh{F}\xrightarrow{\rho_{X/Z}}\mathscr{H}^0_{X/Z}(\sh{F})\xrightarrow{v}\sh{F}'$.
If in addition the supports of $\Ker(u)$ and of $\Coker(u)$ are contained in $Z$, $v$ is an isomorphism.
\end{proposition}

\begin{proof}
The first assertion means that, for every $x\in X-Z$, we have
\[
  \Ker(\rho_{X/Z})_x=\Coker(\rho_{X/Z})_x=0,
\]
i.e.\ that $(\rho_{X/Z})_x$ is bijective, or in other words that $\sh{F}$ is Z-pure at $x$, as has been noted above \sref{IV.5.9.9}.

To show that $\mathscr{H}^0_{X/Z}(\sh{F})$ is Z-closed, it suffices to see that for every Noetherian open subset $V$ of $X$, $\Gamma(V\cap U_\beta,\mathscr{H}^0_{X/Z}(\sh{F}))$ is equal to $\Gamma(V,\mathscr{H}^0_{X/Z}(\sh{F}))$; but by definition
\[
  \Gamma(V\cap U_\beta,\mathscr{H}^0_{X/Z}(\sh{F}))=\varinjlim_\alpha\Gamma(V\cap U_\alpha\cap U_\beta,\sh{F}).
\]
Now, the double family $(U_\alpha\cap U_\beta)$ is a decreasing filtering family, and our assertion results from \sref{IV.5.9.1} and the theorem of the double inductive limit.

\oldpage[IV-2]{113}
Let us pass to the second part of the proposition.
The existence and uniqueness of $v$ result from the fact that for every $\alpha$, $(i_\alpha)_*(u)$ is the unique homomorphism making the diagram
\[
\xymatrix{
  \sh{F}\ar[r]^u\ar[d] & \sh{F}'\ar[d]\\
  (i_\alpha)_*(\sh{F}|U_\alpha)\ar[r]^-{(i_\alpha)_*(u)} & (i_\alpha)_*(\sh{F}'|U_\alpha)
}
\]
commutative, from which there is a unique homomorphism $w$ making all the diagrams
\[
\xymatrix{
  (i_\alpha)_*(\sh{F}|U_\alpha)\ar[r]^-{(i_\alpha)_*(u)}\ar[d] & (i_\alpha)_*(\sh{F}'|U_\alpha)\ar[d]\\
  \mathscr{H}^0_{X/Z}(\sh{F})\ar[r]^w & \mathscr{H}^0_{X/Z}(\sh{F}')
}
\]
commutative, and finally that $\sh{F}'$ and $\mathscr{H}^0_{X/Z}(\sh{F}')$ identify canonically by hypothesis.

It remains to see that if the supports of $\Ker(u)$ and of $\Coker(u)$ are contained in $Z$, $v$ is an isomorphism.
It suffices to see that for every Noetherian open subset $V$, the corresponding homomorphism $\Gamma(V,\mathscr{H}^0_{X/Z}(\sh{F}))\to\Gamma(V,\sh{F}')$ is then an isomorphism.
Now, if a section $t\in\Gamma(V,\mathscr{H}^0_{X/Z}(\sh{F}))$ has image $0$ in $\Gamma(V,\sh{F}')$, note that for some index $\alpha$, we have $t\in\Gamma(V\cap U_\alpha,\sh{F})$, and by virtue of the hypothesis on $u$, we have $t_y=0$ for every $y\in V\cap(X-Z)$; there is thus an open set containing $V\cap(X-Z)$ such that the restriction of $t$ to this open set is zero, hence by definition $t$ is the element $0$ of $\Gamma(V,\mathscr{H}^0_{X/Z}(\sh{F}))$.
Let us now prove that every section $s'\in\Gamma(V,\sh{F}')$ is the image of a section of $\mathscr{H}^0_{X/Z}(\sh{F})$ above $V$.
By hypothesis, for every $x\in V\cap(X-Z)$ there exists a section $s^{(x)}$ of $\sh{F}$ above a neighbourhood $W^{(x)}$ of $x$ in $X$, whose image by $u$ is $s'|W^{(x)}$; $s'|W^{(x)}$ is thus also the image by $v$ of the section $t^{(x)}$ of $\mathscr{H}^0_{X/Z}(\sh{F})$, the canonical image of $s^{(x)}$.
Furthermore, since as we have seen $v$ is injective, the restrictions of $t^{(x)}$ and $t^{(x')}$ to $W^{(x)}\cap W^{(x')}$ are identical for any two points $x$, $x'$ of $V\cap(X-Z)$; the $t^{(x)}$ are therefore restrictions of the same section $t$ of $\mathscr{H}^0_{X/Z}(\sh{F})$ above an open neighbourhood $U$ of $(X-Z)\cap V$.
But since $\mathscr{H}^0_{X/Z}(\sh{F})$ is Z-closed, $t$ extends in a unique way to a section of $\mathscr{H}^0_{X/Z}(\sh{F})$ above $V$, whose image by $v$ has the same restriction as $s'$ to $U$, and coincides therefore with $s'$ for the same reason.

We say that $\mathscr{H}^0_{X/Z}(\sh{F})$ is the \emph{Z-closure} of $\sh{F}$.
\end{proof}

\begin{remarks}[5.9.12]
\label{IV.5.9.12}
\oldpage[IV-2]{114}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item Let $\mathsf{C}(X)$ be the category of $\sh{O}_X$-Modules, and let $\mathsf{C}_Z(X)$ be the subcategory of $\mathsf{C}(X)$ formed by the $\sh{O}_X$-Modules with support contained in $Z$; this subcategory is localising in the sense of Gabriel, and the functor $\mathscr{H}^0_{X/Z}$ is none other than the localisation functor of Gabriel (cf.~\cite{IV-27}); this would furnish another proof of \sref{IV.5.9.11}).
  When $Z$ is closed, the functor $i^*:\mathsf{C}(X)\to\mathsf{C}(X-Z)$ (where $i:X-Z\to X$ is the canonical injection) defines an \emph{equivalence} of categories $\mathsf{C}(X)/\mathsf{C}_Z(X)\approx\mathsf{C}(X-Z)$.
  \item It results from \sref{IV.5.9.11} that the condition $\mathscr{H}^0_{X/Z}(\sh{F})=0$ is \emph{equivalent} to $\Supp(\sh{F})\subset Z$.
  It entails indeed this last condition since the kernel of $\rho_{X/Z}$ is then equal to $\sh{F}$.
  Conversely, if $\Supp(\sh{F})\subset Z$, it suffices to apply the second part of \sref{IV.5.9.11} to the unique homomorphism $u:\sh{F}\to 0$ to conclude that the corresponding morphism $v:\mathscr{H}^0_{X/Z}(\sh{F})\to 0$ is an isomorphism.
  \item The above developments hold in their entirety for any ringed space which is locally Noetherian, whose every irreducible closed subset admits exactly one generic point.
  In particular they apply, on a locally Noetherian prescheme, to \emph{sheaves of abelian groups} (considered as Modules over the simple sheaf associated to the constant presheaf $\bb{Z}$).
  We have moreover for every $x\in X$ the canonical isomorphism \sref{IV.5.9.6.1}, in which $\widetilde{\sh{F}}_x$ denotes the sheaf induced on the subspace $X_x$ of $X$ by the sheaf $\sh{F}$; the direct proof results immediately from the definition \sref{IV.5.9.1.2} and the theorem of the double inductive limit.
\end{enumerate}
\end{remarks}

\subsection{Property $(\mathrm{S}_2)$ and Z-closure}
\label{subsection:IV.5.10}

\begin{env}[5.10.1]
\label{IV.5.10.1}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module; for every subset $T$ of $X$, we set
\begin{equation}
\label{IV.5.10.1.1}
  \operatorname{prof}_T(\sh{F})=\inf_{x\in T}\operatorname{prof}(\sh{F}_x).\tag{5.10.1.1}
\end{equation}
\end{env}

\begin{proposition}[5.10.2]
\label{IV.5.10.2}
Let $X$ be a locally Noetherian prescheme, $Z$ a subset of $X$ stable under specialisation, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\sh{F}$ is Z-pure.
  \item $\operatorname{Ass}(\sh{F})$ does not meet $Z$.
\end{enumerate}
If furthermore $\sh{F}$ is coherent, these conditions are equivalent to the following:
\begin{enumerate}[resume,label=\emph{\alph*)}]
  \item $\operatorname{prof}_Z(\sh{F})\geq 1$.
\end{enumerate}
\end{proposition}

\begin{proof}
To say that $\sh{F}$ is Z-pure means that for every Noetherian open subset $V$ of $X$, and every open set $U\supset X-Z$, the restriction homomorphism $\Gamma(V,\sh{F})\to\Gamma(V\cap U,\sh{F})$ is injective \sref{IV.5.9.8}; but by \sref{IV.3.1.8} this is equivalent to $V\cap\operatorname{Ass}(\sh{F})\subset U$, whence the equivalence of \emph{a)} and \emph{b)}.
Furthermore, to say that $x\in\operatorname{Ass}(\sh{F})$ means that no element of $\mathfrak{m}_x$ is $\sh{F}_x$-regular \sref{IV.3.1.2}, hence, when $\sh{F}$ is coherent, this can be written $\operatorname{prof}(\sh{F}_x)=0$; from this we immediately deduce in this case the equivalence of \emph{b)} and \emph{c)}.
\end{proof}

\begin{corollary}[5.10.3]
\label{IV.5.10.3}
Let $0\to\sh{F}'\to\sh{F}\to\sh{F}''\to 0$ be an exact sequence of quasi-coherent $\sh{O}_X$-Modules.
If $\sh{F}$ is Z-pure, so is $\sh{F}'$; conversely, if $\sh{F}'$ and $\sh{F}''$ are Z-pure, so is $\sh{F}$.
\end{corollary}

\begin{proof}
This results from the form \sref{IV.5.10.2}, \emph{b)} of the condition for a quasi-coherent $\sh{O}_X$-Module to be Z-pure, and from \sref{IV.3.1.7}.
\end{proof}

\end{corollary}

\begin{proof}
This results immediately from \sref{IV.5.10.2} and \sref{IV.5.9.6}.
\end{proof}

\begin{theorem}[5.10.5]
\label{IV.5.10.5}
Let $X$ be a locally Noetherian prescheme, $Z$ a subset stable under specialisation, $\sh{F}$ a coherent $\sh{O}_X$-Module.
For $\sh{F}$ to be Z-closed, it is necessary and sufficient that $\operatorname{prof}_Z(\sh{F})\geq 2$.
\end{theorem}

\begin{proof}
By virtue of \sref{IV.5.10.2}, we can limit ourselves to the case where $\sh{F}$ is Z-pure and $\operatorname{prof}_Z(\sh{F})\geq 1$.
Furthermore, to say that $\operatorname{prof}_Z(\sh{F})\geq 2$ is equivalent to saying that for every closed subset $Z_\alpha$ of $Z$, $\operatorname{prof}_{Z_\alpha}(\sh{F})\geq 2$; and similarly, it results from \sref{IV.5.9.8} that to say that $\sh{F}$ is Z-closed for every $\alpha$ is equivalent to saying that $\sh{F}$ is Z-closed in the case where $Z$ is \emph{closed}.
The question being local, it suffices, for every $x\in Z$, to prove the theorem for $\sh{F}|U$, where $U$ is an affine neighbourhood of $x$, and one can therefore reduce to the case where $X=U$ is affine.
We know then that $\operatorname{Ass}(\sh{F})$ is finite \sref{IV.3.1.6}, and since $\operatorname{Ass}(\sh{F})\subset X-Z$, there is a section $f$ of $\sh{O}_X$ above $X$ such that $\operatorname{Ass}(\sh{F})\subset X_f\subset X-Z$ (\textbf{II}, 4.5.4); we deduce from this that $f$ is $\sh{F}$-regular \sref{IV.3.1.9} and that for every $y\in Z$, we have $f_y\in\mathfrak{m}_y$, hence $\operatorname{prof}(\sh{F}_y)=1+\operatorname{prof}(\sh{F}_y/f_y\sh{F}_y)$ ($\mathbf{0}$, 16.4.6).
The condition $\operatorname{prof}_Z(\sh{F})\geq 2$ is thus equivalent to $\operatorname{prof}_Z(\sh{F}/f\sh{F})\geq 1$, or equivalently \sref{IV.5.10.2} to the fact that $\sh{F}/f\sh{F}$ is Z-\emph{pure}, and it suffices to see that this last property is equivalent to the fact that $\sh{F}$ is Z-closed.

Consider the exact sequence $0\to\sh{F}\xrightarrow{f}\sh{F}\to\sh{F}/f\sh{F}\to 0$ (the homothety of ratio $f:\sh{F}\xrightarrow{f}\sh{F}$ being injective by hypothesis); setting $W=X-Z$, we have the commutative diagram
\[
\xymatrix{
  0\ar[r] & \Gamma(X,\sh{F})\ar[r]^f\ar[d] & \Gamma(X,\sh{F})\ar[r]\ar[d] & \Gamma(X,\sh{F}/f\sh{F})\ar[r]\ar[d] & 0\\
  0\ar[r] & \Gamma(W,\sh{F})\ar[r]^f & \Gamma(W,\sh{F})\ar[r] & \Gamma(W,\sh{F}/f\sh{F}) &
}
\]
where the two rows are exact ($X$ being affine).
If the restriction homomorphism $\Gamma(X,\sh{F})\to\Gamma(W,\sh{F})$ is bijective, we deduce from this diagram that
\[
  \Gamma(X,\sh{F}/f\sh{F})\to\Gamma(W,\sh{F}/f\sh{F})
\]
is injective, and this shows \sref{IV.5.9.8} that if $\sh{F}$ is Z-closed, $\sh{F}/f\sh{F}$ is Z-pure.
Conversely, suppose that $\sh{F}/f\sh{F}$ is Z-pure, and let $s$ be a section of $\sh{F}$ above $W$; since $X_f\subset W$, as $f^n(s|X_f)$ extends to a section $t$ of $\sh{F}$ above $X$ (\textbf{I}, 1.4.1); moreover, the restrictions of $t$ and of $f^ns$ to $X_f$ being the same, it follows that the restriction of $t$ to $W$ is equal to $f^ns$ by virtue of the relation $\operatorname{Ass}(\sh{F})\subset X_f$ \sref{IV.5.10.2}; since $f$ is $\sh{F}$-regular, it suffices to see that $t$ is of the form $f^nt'$, where $t'\in\Gamma(X,\sh{F})$
\oldpage[IV-2]{116}
to show that the homomorphism $\Gamma(X,\sh{F})\to\Gamma(W,\sh{F})$ is surjective, hence bijective.
Now, to say that $t=f^nt'$ means that the image of $t$ in $\Gamma(X,\sh{F}/f^n\sh{F})$ is zero.
But since $f^k\sh{F}/f^{k+1}\sh{F}$ is isomorphic to $\sh{F}/f\sh{F}$, hence Z-pure by hypothesis, we deduce from \sref{IV.5.10.3} that $\sh{F}/f^n\sh{F}$ is Z-pure.
But by definition the image of $t|W=f^ns$ in $\Gamma(W,\sh{F}/f^n\sh{F})$ is zero, whence the conclusion.
\end{proof}

\begin{corollary}[5.10.6]
\label{IV.5.10.6}
Let $\sh{F}$ be a coherent $\sh{O}_X$-Module. For $\sh{F}$ to be Z-closed at a point $x$, it is necessary and sufficient that $\operatorname{prof}_{Z_x}(\widetilde{\sh{F}}_x)\geq 2$.
\end{corollary}

\begin{proof}
This results from \sref{IV.5.9.6} and \sref{IV.5.10.5}.
\end{proof}

\begin{theorem}[5.10.7]
\label{IV.5.10.7}
\emph{(Hartshorne).}
Let $X$ be a locally Noetherian prescheme, $Y$ a closed subset of $X$.
Suppose that for every $y\in Y$, we have $\operatorname{prof}(\sh{O}_{X,y})\geq 2$; then for every connected component $C$ of $X$, $C-(C\cap Y)$ is connected.
\end{theorem}

\begin{proof}
We can limit ourselves to the case where $X$ is connected; it results then from \sref{IV.5.10.5} that the canonical homomorphism $\sh{O}_X\to i_*(\sh{O}_X|X-Y)$ (where $i:X-Y\to X$ is the canonical injection) is bijective.
It follows that the restriction homomorphism $\Gamma(X,\sh{O}_X)\to\Gamma(X-Y,\sh{O}_X)$ is also bijective.
It now suffices to apply the lemma (\textbf{III}, 7.8.6.1).
\end{proof}

\begin{corollary}[5.10.8]
\label{IV.5.10.8}
Let $X$ be a locally Noetherian prescheme, $d$ an integer such that for every $x\in X$, the relation $\dim(\sh{O}_x)\geq d$ implies $\operatorname{prof}(\sh{O}_x)\geq 2$.
Suppose $X$ connected; then, if $X'$, $X''$ are two distinct irreducible components of $X$, there exists a suite $(X_i)_{0\leq i\leq n}$ of irreducible components of $X$ such that $X_0=X'$, $X_n=X''$, and that, for $1\leq i\leq n$, $\operatorname{codim}(X_{i-1}\cap X_i,X)\leq d-1$ \emph{(}we say then that $X$ is \emph{connected in codimension} $\leq d-1$\emph{)}.

If $Y$ is a closed subset of $X$ such that $\operatorname{codim}(Y,X)\geq d$, we have $\dim(\sh{O}_{X,y})\geq d$ for every $y\in Y$ \sref{IV.5.1.3}, hence $\operatorname{prof}(\sh{O}_{X,y})\geq 2$ for every $y\in Y$, and it results from \sref{IV.5.10.7} that $X-Y$ is connected.
On the other hand, for $\operatorname{codim}(Y,X)\geq d$, it is necessary and sufficient that $\operatorname{codim}(Y\cap V,V)\geq d$ for every open neighbourhood $V$ of $y$ in $X$ ($\mathbf{0}$, 14.2.3).
We note finally that if $\mathfrak{Y}$ denotes the set of closed subsets $Y$ of $X$ of codimension $\geq d$, the union of two sets of $\mathfrak{Y}$ belongs to $\mathfrak{Y}$ ($\mathbf{0}$, 14.2.5), and every closed set contained in a set of $\mathfrak{Y}$ belongs to $\mathfrak{Y}$; this is also expressed by saying that $\mathfrak{Y}$ is an \emph{antifilter} of closed subsets of $X$.
The corollary then results from the following topological lemma:
\end{corollary}

\begin{lemma}[5.10.8.1]
\label{IV.5.10.8.1}
Let $X$ be a connected locally Noetherian topological space, $\mathfrak{Y}$ an antifilter of closed subsets of $X$.
Suppose that if $Y$ is a closed subset of $X$ such that for every $y\in Y$, there is a neighbourhood $V$ of $y$ in $X$ and a $Y_y\in\mathfrak{Y}$ such that $V\cap Y=V\cap Y_y$, then $Y\in\mathfrak{Y}$.
The following conditions are then equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item For every $Y\in\mathfrak{Y}$, $X-Y$ is connected.
  \item If $X'$ and $X''$ are two distinct irreducible components of $X$, there exists a suite $(X_i)_{0\leq i\leq n}$ of irreducible components of $X$ such that $X_0=X'$, $X_n=X''$ and that, for $1\leq i\leq n$, $X_{i-1}\cap X_i\notin\mathfrak{Y}$.
\end{enumerate}
\end{lemma}

\begin{proof}
\oldpage[IV-2]{117}
Suppose \emph{b)} verified and let us prove that $U=X-Y$ is connected for every $Y\in\mathfrak{Y}$.
If $U'$ and $U''$ are two distinct irreducible components of $U$, there exist two irreducible components $X'$, $X''$ of $X$ such that $X'\cap U=U'$, $X''\cap U=U''$ ($\mathbf{0_I}$, 2.1.6); let us form for these two components a suite $(X_i)$ having the property stated in \emph{b)} and set $U_i=X_i\cap U$ for $1\leq i\leq n$; then $U_i$ is an irreducible component of $U$ ($\mathbf{0_I}$, 2.1.6), and moreover $U_i\cap U_{i-1}\neq\emp$ for $1\leq i\leq n$; if we had $X_{i-1}\cap X_i\in\mathfrak{Y}$ for some $i$ with $1\leq i\leq n$, we would deduce $X_{i-1}\cap X_i\subset Y$ by definition of $Y$, whence $U_{i-1}\cap U_i=\emp$, contradicting the hypothesis.
This completes the proof that $U$ is connected.

Let us now show that \emph{a)} implies \emph{b)}.
Denote by $Y$ the union of the family $(X_\alpha\cap X_\beta)$, where $(X_\alpha,X_\beta)$ ranges over the set of couples of \emph{distinct} irreducible components of $X$ such that $X_\alpha\cap X_\beta\in\mathfrak{Y}$.
For every point $y\in Y$ in a neighbourhood $V$ of $y$ in $X$ meeting only a finite number of irreducible components of $X$; it follows on the one hand that $Y$ is closed and on the other that $V\cap Y$ is the intersection of $V$ and of a set of $\mathfrak{Y}$; by virtue of the hypothesis on $\mathfrak{Y}$, we have $Y\in\mathfrak{Y}$, hence $U=X-Y$ is connected; moreover $Y$ is rare in $X$.
Let $X'$, $X''$ be two distinct irreducible components of $X$, $U'$, $U''$ their respective traces on $U$; these are distinct irreducible components of $U$ ($\mathbf{0_I}$, 2.1.6).
Now the union of the irreducible components $W$ of $U$ such that there exist irreducible components $(U_i)_{0\leq i\leq n}$ of $U$ for which $U_0=U'$, $U_{i-1}\neq U_i$ and $U_{i-1}\cap U_i\neq\emp$ for $1\leq i\leq n$ and $U_n=W$ is an \emph{open and closed} set in $U$, since $U$ is locally Noetherian and its irreducible components form a locally finite family of closed sets.
There is thus such a suite $(U_i)$ for which $U_n=U''$; let $X_i$ $(0\leq i\leq n)$ be the irreducible component of $X$ such that $X_i\cap U=U_i$ ($\mathbf{0_I}$, 2.1.6); since $U_{i-1}\neq U_i$, we have $X_{i-1}\neq X_i$ for $1\leq i\leq n$; if we had $X_{i-1}\cap X_i\in\mathfrak{Y}$ for some $i$ with $1\leq i\leq n$, we would deduce $X_{i-1}\cap X_i\subset Y$ by definition of $Y$, whence $U_{i-1}\cap U_i=\emp$, contradicting the hypothesis.
This completes the proof of the lemma.

We note that the hypothesis on $X$ in \sref{IV.5.10.8} is verified when $X$ is a \emph{Cohen-Macaulay prescheme} and $d\geq 2$.
\end{proof}

\begin{corollary}[5.10.9]
\label{IV.5.10.9}
If a Noetherian local ring $A$ satisfies $(\mathrm{S}_2)$ and is catenary, it is equidimensional.
\end{corollary}

\begin{proof}
The hypothesis of \sref{IV.5.10.8} is then verified by $X=\Spec(A)$, with $d=2$.
To show that all the irreducible components of $X$ have the same dimension, it suffices then, by virtue of \sref{IV.5.10.8}, to show that two such components $X'$, $X''$ have the same dimension when one further supposes that $\operatorname{codim}(X'\cap X'',X)=1$.
There is then an irreducible component $Z$ of $X'\cap X''$ such that $\operatorname{codim}(Z,X)=1$, hence $\operatorname{codim}(Z,X')=1$, since $\operatorname{codim}(Z,X')\geq 1$; similarly $\operatorname{codim}(Z,X'')=1$, and as $X$ is catenary, this implies $\dim(X')=\dim(X'')$.
\end{proof}

\begin{proposition}[5.10.10]
\label{IV.5.10.10}
Let $X$ be a locally Noetherian prescheme, $Z$ a subset of $X$ stable under specialisation, $\sh{F}$ a coherent $\sh{O}_X$-Module and suppose that the $\sh{O}_X$-Module $\mathscr{H}^0_{X/Z}(\sh{F})$ is coherent.
Then:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item We have $\operatorname{prof}_Z(\mathscr{H}^0_{X/Z}(\sh{F}))\geq 2$.
  \item For every point $x\in\operatorname{Ass}(\sh{F})\cap(X-Z)$, we have $\operatorname{codim}(Z\cap\overline{\{x\}},\overline{\{x\}})\geq 2$.
  \item The set $U$ of the $x\in X$ such that $\operatorname{prof}_{Z_x}(\widetilde{\sh{F}}_x)\geq 2$ \emph{(}notations of \sref{IV.5.9.6}\emph{)} is open in $X$; we have $X-U\subset Z$, and $U$ is the largest open subset of $X$ such that $\sh{F}|U$ is $(Z\cap U)$-closed.
\end{enumerate}
\end{proposition}

\oldpage[IV-2]{118}
\begin{proof}
Set $\sh{F}'=\mathscr{H}^0_{X/Z}(\sh{F})$ for short.
We know \sref{IV.5.9.11} that $\sh{F}'$ is Z-closed, and assertion (i) thus results from \sref{IV.5.10.5} applied to $\sh{F}'$.
Let $x\in\operatorname{Ass}(\sh{F})\cap(X-Z)$; since the restrictions of $\sh{F}$ and $\sh{F}'$ to $X-Z$ are canonically isomorphic \sref{IV.5.9.11}, we have $x\in\operatorname{Ass}(\sh{F}')$.
Consider a point $y\in Z\cap\overline{\{x\}}$; the prime ideal $\mathfrak{p}$ of $\sh{O}_y$ corresponding to $x$ is associated to the $\sh{O}_y$-module $\sh{F}'_y$, hence we have, by virtue of (i) and ($\mathbf{0}$, 16.4.6.2),
$2\leq\operatorname{prof}(\sh{F}'_y)\leq\dim(\sh{O}_y/\mathfrak{p})=\operatorname{codim}(\{y\},\overline{\{x\}})$,
whence (ii).
To prove (iii), note that $U$ is the set of the $x\in X$ such that $\widetilde{\sh{F}}_x$ is $Z_x$-closed \sref{IV.5.10.5}, or equivalently, by virtue of \sref{IV.5.9.6}, the set of points where the canonical homomorphism $\widetilde{\sh{F}}_x\to\widetilde{\sh{F}}'_x$ is bijective; it is therefore the complement of the union of the supports of $\Ker(\rho_{X/Z})$ and $\Coker(\rho_{X/Z})$, and these last are coherent $\sh{O}_X$-Modules by virtue of the hypothesis ($\mathbf{0_I}$, 5.3.4), hence have closed support ($\mathbf{0_I}$, 5.2.2); this shows that $U$ is open, and $U$ is clearly the largest open set such that $\sh{F}|U$ is $(Z\cap U)$-closed; finally the inclusion $X-U\subset Z$ results from \sref{IV.5.9.11}.

We will see further on \sref{IV.5.11.1} that in the most important cases assertion (ii) conversely implies that $\mathscr{H}^0_{X/Z}(\sh{F})$ is coherent.
\end{proof}

\begin{proposition}[5.10.14]
\label{IV.5.10.14}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module with support equal to $X$.
\begin{enumerate}[label=\emph{(\roman*)}]
  \item For $\sh{F}$ to satisfy $(\mathrm{S}_1)$, it is necessary and sufficient that it be $Z^{(1)}$-pure.
  \item For $\sh{F}$ to satisfy $(\mathrm{S}_2)$, it is necessary and sufficient that it be $Z^{(2)}$-closed and $Z^{(1)}$-pure, or equivalently that it be $Z^{(2)}$-closed and have no associated prime cycle of codimension $1$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item To say that $\sh{F}$ has the property $(\mathrm{S}_1)$ means that $\sh{F}$ has no immersed associated prime cycle \sref{IV.5.7.5}, or equivalently that for every $x\in\operatorname{Ass}(\sh{F})$, we have $\dim(\sh{F}_x)=0$ \sref{IV.3.1.4}, in other words $\dim(\sh{O}_x)=0$ \sref{IV.5.1.12.1}; but this is equivalent to saying that $\operatorname{Ass}(\sh{F})$ does not meet $Z^{(1)}$, and the conclusion results from \sref{IV.5.10.2}.
  \item To say that $\sh{F}$ is $Z^{(2)}$-closed means that $\operatorname{prof}_{Z^{(2)}}(\sh{F})\geq 2$ \sref{IV.5.10.5}, or equivalently that, for every $x\in X$, $\dim(\sh{F}_x)\geq 2$ implies $\operatorname{prof}(\sh{F}_x)\geq 2$; this shows that property $(\mathrm{S}_2)$ implies this condition; it implies in addition that $\sh{F}$ satisfies $(\mathrm{S}_1)$, hence has no immersed associated prime cycle.
  Conversely, suppose that $\sh{F}$ is $Z^{(2)}$-closed and has no associated prime cycle of codimension $1$; to verify $(\mathrm{S}_2)$, it remains to show that if $x\in X$ is such that $\dim(\sh{F}_x)=1$ (or, what amounts to the same, $\dim(\sh{O}_x)=1$), then $\operatorname{prof}(\sh{F}_x)=1$.
  But the hypothesis $\dim(\sh{O}_x)=1$ implies $x\notin\operatorname{Ass}(\sh{F})$, and this last relation is equivalent to $\operatorname{prof}(\sh{F}_x)\neq 0$, i.e.\ $\operatorname{prof}(\sh{F}_x)\geq 1$.
  If $\sh{F}$ is $Z^{(1)}$-pure, it is noted above that it satisfies $(\mathrm{S}_1)$, hence all the associated prime cycles of $\sh{F}$ are of codimension $0$, hence the preceding applies.
\end{enumerate}

We note that it can happen that $\sh{F}$ is $Z^{(2)}$-closed and does not satisfy $(\mathrm{S}_1)$: this is the case for example when $X$ is of dimension $1$ (for then $Z^{(2)}=\emp$, and every $\sh{O}_X$-Module is $Z^{(2)}$-closed) and has immersed associated prime cycles.

Recall that in Chapter~III, in the study of local cohomology, a cohomological characterisation of property $(\mathrm{S}_n)$ for all $n\geq 1$, generalising \sref{IV.5.10.14}, is given.
\end{proof}

\begin{corollary}[5.10.15]
\label{IV.5.10.15}
\oldpage[IV-2]{120}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module with support $X$.
Suppose that $\sh{F}$ has no associated prime cycles of codimension $1$ and that $\sh{F}'=\mathscr{H}^0_{X/Z^{(2)}}(\sh{F})$ is coherent.
Then:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item $\sh{F}'$ satisfies the property $(\mathrm{S}_2)$.
  \item The set $U$ of the $x\in X$ such that $\sh{F}$ satisfies $(\mathrm{S}_2)$ at the point $x$ \sref{IV.5.7.2} is open in $X$ and $\operatorname{codim}(X-U,X)\geq 2$.
\end{enumerate}
\end{corollary}

\begin{proof}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item We know \sref{IV.5.9.11} that $\sh{F}'$ is $Z^{(2)}$-closed, and furthermore $\Supp(\sh{F}')=X$, since the maximal points of $X$ belong to $X-Z^{(2)}$ and at these points $\sh{F}'_x=\sh{F}_x\neq 0$, hence the support of $\sh{F}'$ is dense in $X$, and as $\sh{F}'$ is coherent, $\Supp(\sh{F}')$ is closed, hence $\Supp(\sh{F}')=X$.
  It remains to see that $\sh{F}'$ has no associated prime cycles of codimension $1$.
  But if $x\in\operatorname{Ass}(\sh{F}')$ and $\dim(\sh{F}'_x)=\dim(\sh{O}_x)=1$, we have $x\in X-Z^{(2)}$, hence since $\sh{F}'_x=\sh{F}_x$ we would have $x\in\operatorname{Ass}(\sh{F})$, contrary to the hypothesis, which completes the proof of (i).
  \item We have $Z^{(n)}(X_x)=Z^{(n)}(X)\cap X_x$, with the notations of \sref{IV.5.9.6}, taking into account (\textbf{I}, 2.4.2); on the other hand, the hypothesis that $\sh{F}$ has no associated prime cycles of codimension $1$ implies the same hypothesis for $\widetilde{\sh{F}}_x$; for $\widetilde{\sh{F}}_x$ to satisfy $(\mathrm{S}_2)$, it is therefore necessary and sufficient, by virtue of \sref{IV.5.10.14}, that $\widetilde{\sh{F}}_x$ be $Z^{(2)}(X_x)$-closed; assertion (ii) therefore results from \sref{IV.5.10.6} and \sref{IV.5.10.10}, (iii).
\end{enumerate}
\end{proof}

\begin{proposition}[5.10.16]
\label{IV.5.10.16}
Let $X$ be a locally Noetherian prescheme,
\[
  X'=\Spec(\mathscr{H}^0_{X/Z^{(2)}}(\sh{O}_X))
\]
its $Z^{(2)}$-closure, $g:X'\to X$ the structural morphism.
Suppose that $X$ has no associated prime cycle of codimension $1$.
\begin{enumerate}[label=\emph{(\roman*)}]
  \item For $X$ to satisfy $(\mathrm{S}_2)$ at a point $x\in X$, it is necessary and sufficient that the morphism $X'_x\to X_x$ deduced from $g$ \emph{(}notations of \sref{IV.5.10.12}\emph{)} be an isomorphism.
  This condition is always satisfied if $\operatorname{codim}(\overline{\{x\}},X)\leq 1$.
  \item Suppose furthermore that $g$ be a finite morphism \emph{(}see in \sref{IV.5.11.2} sufficient conditions for this to hold\emph{)}.
  Then the set $U$ of the points where $X$ satisfies $(\mathrm{S}_2)$ is open and $\operatorname{codim}(X-U,X)\geq 2$; furthermore $U$ is the largest open subset of $X$ such that the restriction $g^{-1}(U)\to U$ of $g$ is an isomorphism.
  \item Under the same hypotheses as in \emph{(ii)}, $X'$ satisfies $(\mathrm{S}_2)$ and for every $x'\in X'$ such that $\operatorname{codim}(\overline{\{x'\}},X')\leq 1$, the point $x=g(x')$ is such that $\operatorname{codim}(\overline{\{x\}},X)=\operatorname{codim}(\overline{\{x'\}},X')$.
  \item The hypotheses being those of \emph{(ii)}, let $\sh{F}$ be a coherent $\sh{O}_X$-Module with support $X$, such that $\mathscr{H}^0_{X/Z^{(2)}}(\sh{F})$ is coherent; then the $\sh{O}_{X'}$-Module $\sh{F}'$ such that $g_*(\sh{F}')=\mathscr{H}^0_{X/Z^{(2)}}(\sh{F})$ is coherent and satisfies $(\mathrm{S}_2)$, and its support is a union of irreducible components of $X'$.
\end{enumerate}
\end{proposition}

\begin{proof}
\oldpage[IV-2]{121}
Assertions (i) and (ii) are included for reference, having already been proved in substance: (i) results in fact from \sref{IV.5.10.12}, (i) and \sref{IV.5.10.14}, and (ii) is a particular case of \sref{IV.5.10.15}, (ii).

Let us prove (iii); set $x=g(x')$; since $g$ is finite, it is of the same for the morphism $X'_x\to X_x$, hence $\dim(\sh{O}_{X',x'})\leq\dim(X_x)\leq\dim(\sh{O}_{X,x})$ by virtue of \sref{IV.5.4.1}.
Suppose first that $\dim(\sh{O}_{X',x'})\leq 1$ and let us show that then $\dim(\sh{O}_{X,x})\leq 1$.
Otherwise, we would have $\dim(\sh{O}_{X,x})\geq 2$, i.e.\ $x\in Z^{(2)}$; then we would have $\operatorname{prof}(\sh{O}_{X',x'})\geq 2$ by \sref{IV.5.10.12}, which is absurd.
On the other hand, $x\in X-Z^{(2)}$, and hence $\sh{O}_{X',x'}$ is isomorphic to $\sh{O}_{X,x}$ \sref{IV.5.10.12}, (i), whence $\dim(\sh{O}_{X',x'})=\dim(\sh{O}_{X,x})$.
Furthermore, since $X$ has no associated prime cycles of codimension $1$, the hypothesis $\dim(\sh{O}_{X,x})=1$ implies $x\notin\operatorname{Ass}(\sh{O}_X)$, hence $\operatorname{prof}(\sh{O}_{X,x})\neq 0$, and hence also $\operatorname{prof}(\sh{O}_{X',x'})=1$.
Suppose now $\dim(\sh{O}_{X',x'})\geq 2$, hence $\dim(\sh{O}_{X,x})\geq 2$, i.e.\ $x\in Z^{(2)}$; then we deduce from \sref{IV.5.10.12} that $\operatorname{prof}(\sh{O}_{X',x'})\geq 2$ by \sref{IV.5.10.12}, which establishes the assertions of (iii).

To prove (iv) it suffices to replace $\sh{O}_X$ by $\sh{F}$ in the preceding reasoning, which establishes that $\sh{F}'$ satisfies $(\mathrm{S}_2)$ and that if $\dim(\sh{F}'_{x'})\leq 1$, $\sh{F}_x$ and $\sh{F}'_{x'}$ are isomorphic; in particular if $\dim(\sh{F}'_{x'})=0$, we have $\dim(\sh{O}_{X,x})=0$, hence $\dim(\sh{F}_x)=0$ since $\sh{F}$ has support $X$, and finally $\dim(\sh{O}_{X',x'})=0$; every irreducible component of $\Supp(\sh{F}')$ is therefore an irreducible component of $X'$, since $\sh{F}'$ is coherent, hence $\Supp(\sh{F}')$ is closed, hence $\Supp(\sh{F}')$ is a union of irreducible components of $X'$.
\end{proof}

\begin{proposition}[5.10.17]
\label{IV.5.10.17}
Let $A$ be an integral Noetherian ring, and denote by $A^{(1)}$ the intersection of the local rings $A_\mathfrak{p}$, where $\mathfrak{p}$ ranges over the set of prime ideals of $A$ of height $1$.
Suppose that $A^{(1)}$ is a finite $A$-algebra.
Then:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item The ring $A^{(1)}$ satisfies the condition $(\mathrm{S}_2)$.
  \item The set $U$ of the $\mathfrak{p}\in\Spec(A)$ such that the canonical homomorphism $A_\mathfrak{p}\to(A^{(1)})_\mathfrak{p}$ is bijective is equal to the set of the $\mathfrak{p}$ such that $A_\mathfrak{p}$ satisfies $(\mathrm{S}_2)$; $U$ is open in $X=\Spec(A)$ and $\operatorname{codim}(X-U,X)\geq 2$.
  \item For every multiplicative subset $S$ of $A$, $(S^{-1}A)^{(1)}$ is a $(S^{-1}A)$-algebra that is finite.
  \item Let $B$ be a finite $A$-algebra, integral and containing $A$.
  Then $B^{(1)}$ is a finite $B$-algebra.
  Furthermore, for every prime ideal $\mathfrak{q}$ of $B$ of height $1$, the prime ideal $\mathfrak{q}\cap A$ of $A$ is of height $1$.
\end{enumerate}
\end{proposition}

\begin{proof}
Taking into account the formula \sref{IV.5.9.3.1}, we see that $X'=\Spec(A^{(1)})$ is the $Z^{(2)}$-closure of $X=\Spec(A)$; as $A$ has no immersed prime ideals, properties (i) and (ii) are particular cases of \sref{IV.5.10.16}, (i), (ii) and (iii).
To prove (iii), it suffices to remark that $(S^{-1}A)^{(1)}=S^{-1}A^{(1)}$, which is a particular case of \sref{IV.5.9.4}: indeed $S^{-1}A$ is a flat $A$-module, the prime ideals of $S^{-1}A$ are the $S^{-1}\mathfrak{p}$, where $\mathfrak{p}\in\Spec(A)$ does not meet $S$, and $\operatorname{ht}(S^{-1}\mathfrak{p})=\operatorname{ht}(\mathfrak{p})$, whence (iii).
As $A^{(1)}$ is a finite $A$-algebra, $S^{-1}A^{(1)}$ is a finite $S^{-1}A$-algebra, whence (iii).
To prove (iv), set $Y=\Spec(B)$, and let $f:Y\to X$ be the structural morphism; since it is finite, it results from \sref{IV.5.4.1} that for every $y\in Y$, $\dim(\sh{O}_{Y,y})\leq\dim(\sh{O}_{X,f(y)})$; hence, if $T=f^{-1}(Z^{(2)}(X))$, we have $T\supset Z^{(2)}(Y)$.
Let us show that $\sh{G}=\mathscr{H}^0_{Y/Z^{(2)}(Y)}(f_*(\sh{O}_Y))$ is coherent; indeed $f_*(\sh{O}_Y)=\widetilde{B}$, $B$ being considered as an $A$-module; but since $B$ is an integral finite $A$-algebra, its field of fractions is finite over the field of fractions of $A$, and therefore $B$ is contained in a free $A$-module of finite type, and by \sref{IV.5.9.2}, (i) $\sh{G}$ is a quasi-coherent sub-$\sh{O}_X$-Module of $(\mathscr{H}^0_{X/Z^{(2)}(X)}(\sh{O}_X))^n$ for a suitable $n$; this last being coherent by hypothesis, so is $\sh{G}$.
Now, it results from the definition \sref{IV.5.9.1.2} that $\sh{G}$ is isomorphic to $f_*(\mathscr{H}^0_{Y/T}(\sh{O}_Y))$; this proves \emph{a fortiori} that $\mathscr{H}^0_{Y/T}(\sh{O}_Y)$ is a coherent $\sh{O}_Y$-Module.
It results then from \sref{IV.5.10.10}, (ii), applied to $\sh{O}_Y$ and to codimension $2$, that $Z^{(2)}(Y)\subset T$, and finally $T=Z^{(2)}(Y)$.
This proves the two assertions of (iv).
\end{proof}

\begin{proposition}[5.11.1]
\label{IV.5.11.1}
Let $X$ be a locally Noetherian prescheme, $Z$ a subset of $X$ stable under specialisation, $\sh{F}$ a coherent $\sh{O}_X$-Module.
Denote by $(x_\alpha)$ the family of points of $\operatorname{Ass}(\sh{F})\cap(X-Z)$ and, for every $\alpha$, let $Y_\alpha$ be the reduced closed subprescheme of $X$ with underlying space $\overline{\{x_\alpha\}}$, and $Z_\alpha=Z\cap\overline{\{x_\alpha\}}$.
Then the following two conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\mathscr{H}^0_{X/Z}(\sh{F})$ is a coherent $\sh{O}_X$-Module.
  \item For every $\alpha$, $\mathscr{H}^0_{Y_\alpha/Z_\alpha}(\sh{O}_{Y_\alpha})$ is a coherent $\sh{O}_{Y_\alpha}$-Module.
\end{enumerate}
These two conditions imply the following:
\begin{enumerate}[resume,label=\emph{\alph*)}]
  \item For every $\alpha$, we have $\operatorname{codim}(Z_\alpha,Y_\alpha)\geq 2$.
\end{enumerate}
Moreover, the three conditions \emph{a)}, \emph{b)}, \emph{c)} are equivalent when in addition one of the following properties is satisfied:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item Every point of $X$ admits an open neighbourhood isomorphic to a subscheme of a regular scheme \emph{(}in which case we also say that $X$ is \emph{locally immersible in a regular scheme}\emph{)}.
  \item For every $\alpha$, $Y_\alpha$ is universally catenary \sref{IV.5.6.2} and its normalisation (\textbf{II}, 6.3.8) $Y'_\alpha$ is finite over $Y_\alpha$.
\end{enumerate}
\end{proposition}

\begin{proof}
All the properties under consideration are local on $X$, so we can limit ourselves to the case where $X=\Spec(A)$ is affine, and $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module of finite type.
Then, for every $\alpha$, if $h_\alpha$ is the canonical injection $Y_\alpha\to X$, $(h_\alpha)_*(\sh{O}_{Y_\alpha})=\sh{G}_\alpha$ is the $\sh{O}_X$-Module corresponding to the $A$-module quotient $A/\mathfrak{I}_{x_\alpha}$, and, by definition of $\operatorname{Ass}(\sh{F})$, this $A$-module is isomorphic to a sub-$A$-module of $M$.
Since $\mathscr{H}^0_{X/Z}(\sh{G}_\alpha)$ is a quasi-coherent sub-$\sh{O}_X$-Module of $\mathscr{H}^0_{X/Z}(\sh{F})$ \sref{IV.5.9.2}, the hypothesis that $\mathscr{H}^0_{X/Z}(\sh{F})$ is coherent implies that so is $\mathscr{H}^0_{X/Z}(\sh{G}_\alpha)$.
On the other hand, it results from the definition \sref{IV.5.9.1.2} that $\mathscr{H}^0_{X/Z}(\sh{G}_\alpha)$ is isomorphic to $\mathscr{H}^0_{Y_\alpha/Z_\alpha}(\sh{O}_{Y_\alpha})$; this proves that \emph{a)} implies \emph{b)}.
To see that \emph{b)} implies \emph{a)}, it suffices to show that there is a finite filtration $(\sh{F}_i)_{0\leq i\leq n}$ of $\sh{F}$ formed by coherent $\sh{O}_X$-Modules, with $\sh{F}_0=\sh{F}$, $\sh{F}_n=0$, and such that $\mathscr{H}^0_{X/Z}(\sh{F}_i/\sh{F}_{i+1})$ is coherent for every $i$: this results, by descending induction on $i$, from the exact sequences
\[
  0\to\sh{F}_{i+1}\to\sh{F}_i\to\sh{F}_i/\sh{F}_{i+1}
\]
from the fact that $\mathscr{H}^0_{X/Z}$ is a left exact functor \sref{IV.5.9.2}, and finally from ($\mathbf{0_I}$, 5.3.3) and (\textbf{I}, 6.1.1).
By virtue of \sref{IV.3.2.8}, it therefore suffices to prove that $\mathscr{H}^0_{X/Z}(\sh{F})$ is coherent when $\sh{F}$ is irredundant; in other words, $\operatorname{Ass}(M)=\{\mathfrak{p}\}$ is reduced to a single element.
We now note the

\begin{lemma}[5.11.1.1]
\label{IV.5.11.1.1}
Let $A$ be a Noetherian ring, $M$ an $A$-module of finite type, such that $\operatorname{Ass}(M)=\{\mathfrak{p}\}$.
There exists a finite filtration $(M_h)_{0\leq h\leq m}$ of $M$ such that $M_0=M$, $M_m=0$ and that $M_h/M_{h+1}$ is isomorphic to a submodule of $A/\mathfrak{p}$.
\end{lemma}

\begin{proof}
\oldpage[IV-2]{123}
First note that the canonical homomorphism $M\to M_\mathfrak{p}=N$ is injective (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \S\,1, no~2, prop.~6).
Set $B=A_\mathfrak{p}$, $\mathfrak{m}=\mathfrak{p}A_\mathfrak{p}$, the maximal ideal of $B$; we have $\operatorname{Ass}(N)=\{\mathfrak{m}\}$ (\emph{loc.\ cit.}, prop.~5), and since $N$ is a $B$-module of finite type, there is an integer $r$ such that $\mathfrak{m}^rN=0$; if we set $N'_j=\mathfrak{m}^jN$ for $0\leq j\leq r$, $N'_j/N'_{j+1}$ is an $(A/\mathfrak{p})$-module of finite type, hence a direct sum of a finite number of submodules isomorphic to $B/\mathfrak{m}$; in other words, there is a finite filtration $(N_h)_{0\leq h\leq m}$ of $N$ such that $N_0=N$, $N_m=0$ and that $N_h/N_{h+1}$ is isomorphic to $B/\mathfrak{m}$, i.e.\ to the field of fractions of $A/\mathfrak{p}$; the filtration of the $M_h=M\cap N_h$ answers the question, since $M_h/M_{h+1}$ is isomorphic to a sub-$(A/\mathfrak{p})$-module of finite type of $N_h/N_{h+1}=B/\mathfrak{m}$; but one knows that such a submodule is isomorphic to a submodule of $A/\mathfrak{p}$.
\end{proof}

The existence of the filtration $(M_h)$ then shows, by the same reasoning as above, that to prove (granting \emph{b)}) that $\mathscr{H}^0_{X/Z}(\sh{F})$ is coherent, we can limit ourselves to showing that $\mathscr{H}^0_{X/Z}(\sh{P})$ is coherent, where $\sh{P}=(A/\mathfrak{q})\tilde{\phantom{v}}$, $\mathfrak{q}$ being an ideal associated to $M$.
But if $\mathfrak{q}=\mathfrak{I}_y$ with $y\in Z$, the support of $\sh{P}$ is a closed set contained in $Z$, since $Z$ is stable under specialisation; the definition \sref{IV.5.9.1.2} then shows that $\mathscr{H}^0_{X/Z}(\sh{P})=0$.
If on the contrary $\mathfrak{q}=\mathfrak{I}_{x_\alpha}$ for some $\alpha$, we have $\sh{P}=(h_\alpha)_*(\sh{O}_{Y_\alpha})$ by definition, and we have seen above that $\mathscr{H}^0_{X/Z}(\sh{P})$ is isomorphic to $\mathscr{H}^0_{Y_\alpha/Z_\alpha}(\sh{O}_{Y_\alpha})$, hence is coherent by virtue of \emph{b)}.

We have already seen \sref{IV.5.10.10}, (ii) that \emph{a)} implies \emph{c)}.
It remains to prove that, under one or the other of the hypotheses (i), (ii), \emph{c)} implies \emph{b)}.
Note that if $X$ satisfies (i), so does each $Y_\alpha$.
Similarly \sref{IV.5.11.1}; it therefore suffices to prove \sref{IV.5.9.3.1} the corollary below.
\end{proof}

\begin{corollary}[5.11.2]
\label{IV.5.11.2}
Let $A$ be a Noetherian integral ring, satisfying one of the two following hypotheses:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item $A$ is a quotient of a regular Noetherian ring.
  \item $A$ is universally catenary, and its integral closure $A'$ is a finite $A$-algebra.
\end{enumerate}
Then the ring $A^{(1)}$, intersection of the $A_\mathfrak{p}$ where $\mathfrak{p}$ ranges over the set of prime ideals of $A$ of height $1$, is a finite $A$-algebra.
\end{corollary}

\begin{proof}
(i) Set $X=\Spec(A)$.
The set $U$ of the points $x\in X$ where $X$ satisfies $(\mathrm{S}_2)$ is \emph{open} in $X$ by \sref{IV.6.11.2}\footnote{The reader may verify that \sref{IV.5.11.2} is not used in the proof of \sref{IV.6.11.2}.}.
Furthermore, if one sets $Z=X-U$, we have $\operatorname{codim}(Z,X)\geq 2$: indeed, for every $x\in X$ such that $\dim(A_{x'})\leq 1$, we have $\dim(A_{x'})\leq 1$ for every generalisation $x'$ of $x$, hence $\operatorname{prof}(A_{x'})\geq 1$, hence $X$ satisfies $(\mathrm{S}_2)$ at the point $x$.
We have thus $Z\subset Z^{(2)}$, and $\mathscr{H}^0_{X/Z^{(2)}}(\sh{O}_X)=j_*(\mathscr{H}^0_{U/Z^{(2)}}(\sh{O}_U))$, where $j:U\to X$ is the canonical injection \sref{IV.5.9.1.2}.
But since the prescheme $U$ satisfies $(\mathrm{S}_2)$, it results from \sref{IV.5.10.14} that $\mathscr{H}^0_{U/Z^{(2)}}(\sh{O}_U)$ is isomorphic to $\sh{O}_U$; on the other hand, since $\operatorname{codim}(Z,X)\geq 2$, one knows, from Chapter~III, \S\,9, that $j_*(\sh{O}_U)$ is a coherent $\sh{O}_X$-Module, which proves the proposition in case (i).

\oldpage[IV-2]{124}
(ii) The ring $A'$ is, by virtue of hypothesis (ii), a Noetherian integral ring integrally closed, hence (Bourbaki, \emph{Alg.\ comm.}, chap.~VII, \S\,1, no~6, th.~4) the intersection of its local rings $A'_{\mathfrak{p}'}$, where $\mathfrak{p}'$ ranges over the set of prime ideals of height $1$ of $A'$.
Now, for such a prime ideal $\mathfrak{p}'$, if one sets $\mathfrak{p}=\mathfrak{p}'\cap A$ and $S=A-\mathfrak{p}$, $A'_{\mathfrak{p}'}$ is a local ring at a prime ideal $S^{-1}\mathfrak{p}'$ of $S^{-1}A'$; since $S^{-1}\mathfrak{p}'$ lies above the maximal ideal $\mathfrak{p}A_\mathfrak{p}$ of $A_\mathfrak{p}$, it is a maximal ideal of $S^{-1}A'$; and $S^{-1}A'$ is by hypothesis a finite $A_\mathfrak{p}$-algebra; since $S^{-1}\mathfrak{p}'$ lies above the maximal ideal of $A_\mathfrak{p}$, $A_\mathfrak{p}$ is universally catenary.
We deduce from \sref{IV.5.6.10} that $\dim(A_\mathfrak{p})=\dim(A'_{\mathfrak{p}'})=1$.
It results from this that $A^{(1)}\subset A'$, and as $A'$ is a finite module of $A$ by hypothesis, so is $A^{(1)}$ since $A$ is Noetherian; the proposition is therefore proved in case (ii).
\end{proof}

\begin{remark}[5.11.3]
\label{IV.5.11.3}
The fact that $A^{(1)}$ is a finite $A$-algebra is not strictly necessary in hypothesis (ii) of \sref{IV.5.11.2}: one only supposes that $A$ is \emph{catenary}.
An example is given by the catenary local ring $A$ constructed in \sref{IV.5.6.11}, whose integral closure $A'$ (denoted $B$ in \sref{IV.5.6.11}) is a finite $A$-algebra; if $A^{(1)}$ were a finite $A$-algebra, it would be contained in the field of fractions of $A$, hence contained in $A'$.
But on the other hand, with the notations of \sref{IV.5.6.11}, every prime ideal $\mathfrak{r}$ of height $1$ in $A$ is of the form $\mathfrak{p}A$, where $\mathfrak{p}$ is a prime ideal of height $1$ in $C$, and $A_\mathfrak{pA}=C_\mathfrak{p}$; we know \sref{IV.5.6.11} that $C_\mathfrak{p}=E_{\mathfrak{p}'}$, where $\mathfrak{p}'$ is the unique prime ideal of $E$ above $\mathfrak{p}$, hence $A_\mathfrak{pA}$ is integrally closed and contains $A'$; by definition, $A^{(1)}\subset A'$, and the hypothesis that $A^{(1)}$ is a finite $A$-algebra would imply finally $A^{(1)}=A'$.
But this conclusion is absurd, since one of the two prime ideals of $\mathfrak{n}A$ of $A$ is of height $1$, while $\mathfrak{n}A$ is of height $2$, which would contradict \sref{IV.5.10.17}, (iv).
\end{remark}

\begin{corollary}[5.11.4]
\label{IV.5.11.4}
Let $X$ be a locally Noetherian prescheme, $Z$ a closed subset of $X$, $U=X-Z$, $i:U\to X$ the canonical injection, $\sh{F}$ an $\sh{O}_U$-Module coherent.
For $i_*(\sh{F})$ to be a coherent $\sh{O}_X$-Module, it is necessary that for every $x\in\operatorname{Ass}(\sh{F})$, $\operatorname{codim}(\overline{\{x\}}\cap Z,\overline{\{x\}})\geq 2$.

\begin{enumerate}[label=\emph{(\roman*)}]
  \item The prescheme $X$ is locally immersible in a regular scheme.
  \item The prescheme $X$ is universally catenary and universally Japanese \emph{(}which means, by definition, that every point $x\in X$ admits an affine open neighbourhood whose ring is universally Japanese (\textbf{0}, 23.1.1)\emph{)}.
\end{enumerate}

\begin{proof}
One knows (\textbf{I}, 9.4.7) that there exists a quasi-coherent sub-$\sh{O}_X$-Module $\sh{G}$ of $i_*(\sh{F})$ such that $\sh{G}|U=\sh{F}$.
We have clearly $\operatorname{Ass}(\sh{F})\subset\operatorname{Ass}(\sh{G})\subset\operatorname{Ass}(i_*(\sh{F}))$, and since $\operatorname{Ass}(i_*(\sh{F}))=\operatorname{Ass}(\sh{F})$ \sref{IV.3.1.13}, on has $\operatorname{Ass}(\sh{G})=\operatorname{Ass}(\sh{F})$; it suffices then to apply \sref{IV.5.11.1} to the coherent $\sh{O}_X$-Module $\sh{G}$, noting that $i_*(\sh{F})=\mathscr{H}^0_{X/Z}(\sh{G})$ and that, when $X$ is universally catenary and universally Japanese, hypothesis (ii) of \sref{IV.5.11.1} is satisfied by definition.
\end{proof}

\begin{proof}
Let $B=A/\mathfrak{q}$, where $\mathfrak{q}$ is a prime ideal of $A$, so that $Y=\Spec(B)$ is the closed subset $V(\mathfrak{q})$ of $X$; set $Z=Z^{(2)}(Y)$, which is a subset of $X$ stable under specialisation.
If condition \emph{b)} is satisfied, one can apply it to the $\sh{O}_X$-Module coherent $\sh{F}=(A/\mathfrak{q})\tilde{\phantom{v}}$ and to $Z=V(\mathfrak{J})$; inversely, one proves that \emph{d)} implies \emph{c)} by using the equivalence of \emph{a)} and \emph{b)} in \sref{IV.5.11.1}.
It is clear that \emph{c)} is a particular case of \emph{b)}.
Finally, the fact that \emph{a)} (and hence each of the other conditions) is satisfied when $A$ satisfies one of the hypotheses (i), (ii) results from \sref{IV.5.11.2}, by the definition of universally catenary rings and of universally Japanese rings.
\end{proof}

\begin{remarks}[5.11.7]
\label{IV.5.11.7}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item If, in \sref{IV.5.11.5}, condition \emph{d)} implies \emph{a)} without supplementary hypothesis on $X$; we shall see further on \sref{IV.7.2.4} that this is indeed the case also when $X$ is a \emph{local} scheme.
  Similarly, we shall prove that when $A$ is a local Noetherian ring the four properties \emph{a)}, \emph{b)}, \emph{c)}, \emph{d)} of \sref{IV.5.11.6} are equivalent \sref{IV.7.2.4}.
  If $A$ satisfies property \emph{a)} of \sref{IV.5.11.6}, it is of the same for every \emph{ring of fractions} $S^{-1}A$ \sref{IV.5.10.17}, (iii) and (iv); in particular, it extends to all Noetherian rings.
  \item If $A$ satisfies property \emph{a)} of \sref{IV.5.11.6}, it is of the same for every \emph{integral finite $A$-algebra} $C$.
  Indeed every integral quotient ring of $S^{-1}A$ is of the form $S^{-1}(A/\mathfrak{q})$, where $\mathfrak{q}$ is a prime ideal of $A$ not meeting $S$; and on the other hand, if $\mathfrak{r}$ is a prime ideal of $C$, $\mathfrak{p}$ its inverse image in $A$, $C_\mathfrak{r}$ is an $(A/\mathfrak{p})$-algebra that is finite integral with the field of fractions finite over that of $A/\mathfrak{p}$; our assertions are therefore consequences of \sref{IV.5.10.17}, (iii) and (iv).
\end{enumerate}
\end{remarks}

\subsection{Relations between the properties of a Noetherian local ring $A$ and of a quotient ring $A/tA$}
\label{subsection:IV.5.12}

We have already seen in \sref{IV.3.4} relations between the properties of $A$ and of $A/tA$ concerning the associated prime ideals, as well as the properties of being integral or reduced.
In this section we give other relations between the properties of these rings, related to the notions of dimension and of depth.

\begin{proposition}[5.12.1]
\label{IV.5.12.1}
Let $A$ be a Noetherian local ring, $X=\Spec(A)$, $t$ an element of $A$ belonging to a system of parameters ($\mathbf{0}$, 16.3.6), $X_0$ the closed subspace $V(t)$ of $X$, $X_1$ the complementary open subset $X-X_0$.
Let $Z$ be a subset of $X$ such that every specialisation of a point of $Z$ belongs to $Z$.
Suppose that $X$ is biequidimensional \emph{(}which will be the case if $A$ is equidimensional and a quotient of a regular local ring ($\mathbf{0}$, 16.5.12)\emph{)}.
Then if we set $Z_0=Z\cap X_0$, $Z_1=Z\cap X_1$, we have
\begin{equation}
\label{IV.5.12.1.1}
  \operatorname{codim}(Z_0,X_0)\leq\operatorname{codim}(Z,X)\leq\operatorname{codim}(Z_1,X_1).\tag{5.12.1.1}
\end{equation}
\end{proposition}

\oldpage[IV-2]{127}
\begin{proof}
The second inequality results from the definition \sref{IV.5.1.3}; to prove the first, we can limit ourselves to proving it when $Z$ is \emph{closed}: indeed, by hypothesis $Z$ is a union of closed subsets $Z_\alpha$, and if $\operatorname{codim}(X_0\cap Z_\alpha,X_0)\leq\operatorname{codim}(Z_\alpha,X)$ for every $\alpha$, we will have also $\operatorname{codim}(Z_0,X_0)=\inf_\alpha(\operatorname{codim}(X_0\cap Z_\alpha,X_0))\leq\inf_\alpha(\operatorname{codim}(Z_\alpha,X))=\operatorname{codim}(Z,X)$.
Suppose therefore $Z$ closed, so that
\[
  \operatorname{codim}(Z,X)=\dim(X)-\dim(Z).
\]
On the other hand, $X_0$ is clearly catenary, equidimensional and of dimension $\dim(X)-1$ by virtue of ($\mathbf{0}$, 16.3.4); we thus also have
\[
  \operatorname{codim}(Z_0,X_0)=\dim(X_0)-\dim(Z_0)=\dim(X)-1-\dim(Z_0).
\]
But we have $\dim(Z_0)\geq\dim(Z)-1$ ($\mathbf{0}$, 16.3.4), which completes the proof of the first inequality \sref{IV.5.12.1.1}.
\end{proof}

\begin{proposition}[5.12.2]
\label{IV.5.12.2}
Let $A$ be a Noetherian local ring, $M$ an $A$-module of finite type, $t$ an $M$-regular element belonging to the maximal ideal $\mathfrak{m}$ of $A$, $k$ an integer $\geq 1$.
Suppose that $A$ is a catenary ring.
Then, if $M/tM$ is equidimensional and satisfies $(\mathrm{S}_k)$, the same is true of $M$.
\end{proposition}

\begin{proof}
Taking into account ($\textbf{Err}_{\textbf{III}}$, 30) and \sref{IV.5.7.3}, (vi), we can limit ourselves to the case where $\Supp(M)=\Spec(A)=X$.
Set $X_0=V(t)=\Spec(A/tA)$; since $M/tM$ satisfies $(\mathrm{S}_1)$ by hypothesis, it has no immersed associated prime ideals \sref{IV.5.7.5}.
Applying \sref{IV.3.4.4} to the closed point of $X$ (and of every closed subprescheme of $X$), we see that the irreducible components of $\Supp(M/tM)$ are exactly those of $Y_i\cap X_0$, where $Y_i$ $(1\leq i\leq r)$ are the irreducible components of $X$.
Since $t$ is by hypothesis $M$-regular, $V(t)$ contains no maximal point of $X$ \sref{IV.3.1.8}; each of the irreducible components of $Y_i\cap X_0$ is therefore of codimension $1$ in $Y_i$ \sref{IV.5.1.8}; since $X$ is catenary and since all the irreducible components of $\Supp(M/tM)$ have by hypothesis the same dimension, we conclude that the $Y_i$ all have the same dimension, in other words $X$ is equidimensional, hence \emph{biequidimensional} since $A$ is local and catenary.
To see that $M$ satisfies $(\mathrm{S}_k)$, let us apply the criterion \sref{IV.5.7.4} (with the notations of \sref{IV.5.7.4}): we must verify that, for every integer $n\geq 0$, $\operatorname{codim}(Z_n,X)>n+k$.
But the hypothesis on $M/tM$ and the criterion \sref{IV.5.7.4} show that $\operatorname{codim}(Z_n\cap X_0,X_0)>n+k$ for every integer $n\geq 0$.
But every specialisation of a point of $Z_n$ still belongs to $Z_n$, as we shall see in (6.11.5)\footnote{The reader may verify that \sref{IV.5.12.2} is not used in the proof of \sref{IV.6.11.5}.}; since $X$ is biequidimensional and $t$ belongs to a system of parameters for $M$ ($\mathbf{0}$, 16.4.1), and since $t$ is a non-zero-divisor for $A$ (the annihilator of $M$ being nilpotent by virtue of the hypothesis $\Supp(M)=\Spec(A)$), the conclusion results from \sref{IV.5.12.1}.
\end{proof}

\begin{remark}[5.12.3]
\label{IV.5.12.3}
If, in the statement of \sref{IV.5.12.2}, one only supposes that $M/tM$ is equidimensional, the conclusion does not necessarily follow that $M$ is equidimensional.
\end{remark}

\oldpage[IV-2]{128}
\begin{corollary}[5.12.4]
\label{IV.5.12.4}
Let $A$ be a Noetherian local catenary ring, $t$ a regular element belonging to the maximal ideal of $A$, $k$ an integer $\geq 1$.
If $A/tA$ satisfies $(\mathrm{S}_k)$, the same is true of $A$.
\end{corollary}

\begin{proof}
If $k=1$, the corollary results from \sref{IV.3.4.4}, in view of the interpretation of condition $(\mathrm{S}_1)$ \sref{IV.5.7.5}.
If $k\geq 2$, it results from the criterion of Hartshorne \sref{IV.5.10.9} that $A/tA$ is equidimensional, and one can apply \sref{IV.5.12.2}.
\end{proof}

\begin{proposition}[5.12.5]
\label{IV.5.12.5}
Let $A$ be a Noetherian local catenary ring, $t$ an element of $A$ belonging to the maximal ideal of $A$, $k$ an integer $\geq 0$.
If the ring $A/tA$ is reduced, equidimensional and satisfies the property $(\mathrm{R}_k)$, the ring $A$ also has these three properties.
\end{proposition}

\begin{proof}
It is already known \sref{IV.3.4.6} that $A$ is reduced, and it results from \sref{IV.5.12.2}, applied for $k=1$, that $A$ is equidimensional.
Set $X=\Spec(A)$, $X_0=V(t)=\Spec(A/tA)$, and let $Z$ be the set of points $x\in X$ where $X$ is not regular; by virtue of ($\mathbf{0}$, 17.3.2), every specialisation of a point of $Z$ also belongs to $Z$.
It results on the other hand from ($\mathbf{0}$, 17.1.8) that for every point $x\in X_0$ where $X$ is regular, $X$ is also regular, hence the set $Z'_0$ of the points where $X_0$ is not regular contains $Z\cap X_0$; the hypothesis therefore implies that
\[
  k\leq\operatorname{codim}(Z'_0,X_0)\leq\operatorname{codim}(Z_0,X_0).
\]
But, since $A$ is equidimensional and catenary, $X$ is biequidimensional, hence from \sref{IV.5.12.1}
\[
  \operatorname{codim}(Z_0,X_0)\leq\operatorname{codim}(Z,X)
\]
which proves that $X$ satisfies $(\mathrm{R}_k)$.
\end{proof}

\begin{remark}[5.12.6]
\label{IV.5.12.6}
If, in the statement of \sref{IV.5.12.5}, one only supposes that $A/tA$ is reduced and possesses property $(\mathrm{R}_k)$, it does not necessarily follow that $A$ possesses property $(\mathrm{R}_k)$.
\end{remark}

\begin{corollary}[5.12.7]
\label{IV.5.12.7}
\oldpage[IV-2]{129}
Let $A$ be a Noetherian local catenary ring, $t$ a regular element belonging to the maximal ideal of $A$.
If $A/tA$ is integral and integrally closed, the same is true of $A$.
\end{corollary}

\begin{proof}
By virtue of Serre's criterion \sref{IV.5.8.6} and of the fact that $A$ is local, it suffices to show that $\Spec(A)$ satisfies $(\mathrm{S}_2)$ and $(\mathrm{R}_1)$.
But the hypothesis on $A/tA$ implies that $A$ satisfies $(\mathrm{R}_1)$ by \sref{IV.5.12.5}, and that $A$ satisfies $(\mathrm{S}_2)$ by \sref{IV.5.12.4}.
\end{proof}

\begin{proposition}[5.12.8]
\label{IV.5.12.8}
\emph{(Hironaka's lemma).}
Let $A$ be a Noetherian local ring, reduced, equidimensional and catenary.
Suppose furthermore that for every minimal prime ideal $\mathfrak{q}_i$ of $A$, the ring $B_i=A/\mathfrak{q}_i$ is such that $B_i^{(1)}$ \emph{(}notation of \sref{IV.5.10.17}\emph{)} is a finite $B_i$-algebra \emph{(}which occurs for instance when $A$ possesses one of the properties (i) or (ii) of \sref{IV.5.11.6}\emph{)}.
Let $t$ be an element of $A$, non-zero-divisor in $A$, and such that:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item The $A$-module $A/tA$ has only one non-immersed associated prime ideal $\mathfrak{p}$ \emph{(}necessarily of height $1$ by the Hauptidealsatz ($\mathbf{0}$, 16.3.2)\emph{)}, but one does not suppose that this is the only prime ideal associated to $A/tA$.
  \item The ideal $\mathfrak{p}A_\mathfrak{p}$ of $A_\mathfrak{p}$ is generated by $t/1$.
  \item The ring $A/\mathfrak{p}$ is integrally closed.
\end{enumerate}
Under these conditions, the ring $A$ is integral and integrally closed, and we have $\mathfrak{p}=tA$.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{130}
Since $Z=V(tA)$ is an irreducible closed subset of $X=\Spec(A)$, whose generic point $z$ is such that $\mathfrak{I}_z=\mathfrak{p}$, hypothesis (i) shows that $\mathfrak{p}$ cannot contain more than \emph{one} minimal prime ideal $\mathfrak{q}$ of $A$, in other words $\mathfrak{p}$ belongs to a \emph{single} irreducible component of $X$; moreover $\mathfrak{q}$ is the inverse image of $(0)$ by the canonical homomorphism $A\to A_\mathfrak{p}$, and $A_\mathfrak{p}$ is therefore isomorphic to $(A/\mathfrak{q})_{\mathfrak{p}\mathfrak{q}}$, which proves that the integral quotient $A/\mathfrak{q}$ is \emph{integrally closed} as an intersection of integrally closed rings (\textbf{I}, 8.2.1.1).
If $A$ is \emph{Noetherian and normal}, $A$ is composed of a finite direct product of integral domains that are integrally closed (\textbf{I}, 5.1.4).
Let us now note that since $A^{(1)}$ is a finite $A$-algebra, it is a semi-local Noetherian ring, and every non-immersed prime ideal associated to $A^{(1)}/tA^{(1)}$ is of height $1$ by the Hauptidealsatz ($\mathbf{0}$, 16.3.2); now, for such a prime ideal $\mathfrak{r}$, $\mathfrak{r}\cap A$ is of height $1$ by virtue of \sref{IV.5.10.17}, (iv), and as it contains $tA$, it cannot be $\mathfrak{p}$ by virtue of hypothesis (i).
On the other hand, $A^{(1)}$ is contained in the integral closure $A'$ of $A$; if one sets $S=A-\mathfrak{p}$, the integral closure of $A_\mathfrak{p}$ is $S^{-1}A'$ (Bourbaki, \emph{Alg.\ comm.}, chap.~V, \S\,1, no~5, prop.~17), hence $S^{-1}A'=A_\mathfrak{p}$ since $A_\mathfrak{p}$ is a discrete valuation ring; it follows from this that there exists only \emph{one} prime ideal $\mathfrak{p}'$ of $A'$ above $\mathfrak{p}$, and \emph{a fortiori} one prime ideal $\mathfrak{p}^{(1)}$ of $A^{(1)}$ above $\mathfrak{p}$, and we have $A_\mathfrak{p}=A'_{\mathfrak{p}'}=A^{(1)}_{\mathfrak{p}^{(1)}}$.
Note now that $A^{(1)}$ satisfies $(\mathrm{S}_2)$ \sref{IV.5.10.17}, (i) and since $t$ is a non-zero-divisor in $A^{(1)}$, $A^{(1)}/tA^{(1)}$ has no immersed associated prime ideals
\oldpage[IV-2]{131}
\sref{IV.5.7.7}.
The prime ideal $\mathfrak{p}^{(1)}$ is therefore the unique associated prime ideal of $A^{(1)}/tA^{(1)}$, in other words $tA^{(1)}$ is a $\mathfrak{p}^{(1)}$-primary ideal of $A^{(1)}$; this means also that the inverse image of $tA^{(1)}$ by the canonical projection $A^{(1)}\to A^{(1)}/\mathfrak{p}^{(1)}$ is the maximal ideal of $A_\mathfrak{p}$, from (ii); hence $tA^{(1)}=\mathfrak{p}^{(1)}$ is prime in $A^{(1)}$.
On the other hand, $A^{(1)}/\mathfrak{p}^{(1)}$ is finite over $A/\mathfrak{p}$ and has the same field of fractions (since the residue field of $A^{(1)}_{\mathfrak{p}^{(1)}}=A_\mathfrak{p}$), hence it is identical to $A/\mathfrak{p}$ by virtue of hypothesis (iii).
One can therefore write $A^{(1)}=A+\mathfrak{p}^{(1)}=A+tA^{(1)}$, and since $t$ is contained in the radical of the semi-local Noetherian ring $A^{(1)}$, the lemma of Nakayama implies $A^{(1)}=A$, whence $\mathfrak{p}^{(1)}=\mathfrak{p}=tA$.
But since $A$ is catenary and $A/tA$ is integral and integrally closed by virtue of hypothesis (iii), we conclude from \sref{IV.5.12.7} that $A$ is integrally closed.
\end{proof}

\begin{corollary}[5.12.10]
\label{IV.5.12.10}
Let $A$ be an integral Noetherian local ring, satisfying one of the two following hypotheses:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $A$ is universally catenary and universally Japanese.
  \item $A$ is a quotient of a regular ring.
\end{enumerate}
Let $(x_i)_{1\leq i\leq n}$ be a suite of $n$ elements of $A$; set $\mathfrak{J}=x_1A+\ldots+x_nA$, and suppose the following conditions satisfied:
\begin{enumerate}[label=\emph{(\roman*)}]
  \item There exists only one non-immersed prime ideal $\mathfrak{p}$ associated to $A/\mathfrak{J}$ and $\mathfrak{p}$ is of height $n$.
  \item The maximal ideal $\mathfrak{p}A_\mathfrak{p}$ of $A_\mathfrak{p}$ is generated by the $x_i/1$ \emph{(}hence $A_\mathfrak{p}$ is a regular local ring of dimension $n$\emph{)}.
  \item The ring $A/\mathfrak{p}$ is integrally closed.
\end{enumerate}
Under these conditions, for every integer $i$ such that $0\leq i\leq n$, the quotient ring $A/(x_1A+\ldots+x_iA)$ is integrally closed and of dimension $\dim(A)-i$.
In particular, $\mathfrak{J}$ is equal to $\mathfrak{p}$, $A$ is integrally closed and $(x_i)_{1\leq i\leq n}$ is an $A$-regular sequence.
\end{corollary}

\begin{proof}
We proceed by induction on $n$, the proposition being trivial for $n=0$ and reducing to Hironaka's lemma \sref{IV.5.12.8} for $n=1$.
We may therefore suppose $n\geq 2$.
Set $\mathfrak{J}'=x_1A+\ldots+x_{n-1}A$, and let $\mathfrak{q}$ be a minimal prime ideal among those containing $\mathfrak{J}'$.
If $\mathfrak{q}$ is minimal among the prime ideals of $A$ containing $\mathfrak{q}+x_nA$, $\mathfrak{p}'/\mathfrak{q}$ is of height $1$ by the Hauptidealsatz ($\mathbf{0}$, 16.3.1); let us show that $\mathfrak{q}\subset\mathfrak{p}$.
Indeed, a prime ideal $\mathfrak{n}$ of $B$ minimal among those containing $\mathfrak{q}+x_nA$ is of height $\leq n-1$ by the Hauptidealsatz, and since $\mathfrak{q}$ is minimal, there is at most one such ideal.
Since it contains $\mathfrak{J}$, it is necessarily equal to $\mathfrak{p}$, whence $\mathfrak{q}\subset\mathfrak{p}$.

Consider now the integral ring $B=A/\mathfrak{q}$, and let $t$ be the image of $x_n$ in $B$.
It is immediate (cf.\ \sref{IV.5.6.3}, (iv)) that if $A$ satisfies hypothesis \emph{a)} (resp.\ \emph{b)}), then so does $B$, and $B$ and $t$ satisfy the hypotheses of Hironaka's lemma.
Application of \sref{IV.5.12.8} proves that $B=A/\mathfrak{q}$ is integrally closed.
Let us now use the induction hypothesis, which shows that $A/(x_1A+\ldots+x_iA)$ is integrally closed and of dimension $\dim(A)-i$ for $0\leq i\leq n-1$ and that $\mathfrak{q}=\mathfrak{J}'$, whence $\mathfrak{p}=\mathfrak{J}'+x_nA=\mathfrak{J}$.
We conclude that $A/\mathfrak{J}=A/\mathfrak{p}$ is integrally closed by hypothesis (iii), and that $\dim(A/\mathfrak{p})=n$ and that $\dim(A/\mathfrak{p})=\dim(A)-n$ since $A$ is biequidimensional, which completes the proof.
\end{proof}

\subsection{Permanence properties by passage to the inductive limit}
\label{subsection:IV.5.13}

Throughout this section, $(A_\alpha,\varphi_{\beta\alpha})$ denotes an inductive system of rings whose index set is a pre-ordered filtering set; we set $A=\varinjlim A_\alpha$ and denote by $\varphi_\alpha$ the canonical homomorphism $A_\alpha\to A$.

\begin{proposition}[5.13.1]
\label{IV.5.13.1}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item For every index $\alpha$, let $\mathfrak{a}_\alpha$ be an ideal of $A_\alpha$ such that, for $\alpha\leq\beta$, $\varphi_{\beta\alpha}(\mathfrak{a}_\alpha)\subset\mathfrak{a}_\beta$.
  Then the $\mathfrak{a}_\alpha$ form an inductive system for the restrictions of the $\varphi_{\beta\alpha}$, $\mathfrak{a}=\varinjlim\mathfrak{a}_\alpha$ identifies canonically with an ideal of $A$, $A/\mathfrak{a}$ with $\varinjlim(A_\alpha/\mathfrak{a}_\alpha)$.
  If furthermore $\mathfrak{a}_\alpha=\varphi_{\beta\alpha}^{-1}(\mathfrak{a}_\beta)$ for $\alpha\leq\beta$, then $\mathfrak{a}_\alpha=\varphi_\alpha^{-1}(\mathfrak{a})$ for every $\alpha$.
  \item Conversely, for every ideal $\mathfrak{a}$ of $A$, if we set $\mathfrak{a}_\alpha=\varphi_\alpha^{-1}(\mathfrak{a})$ for every $\alpha$, the $\mathfrak{a}_\alpha$ form an inductive system such that $\mathfrak{a}=\varinjlim\mathfrak{a}_\alpha$.
\end{enumerate}
\end{proposition}

\begin{proof}
It is clear that (ii) is a particular case of (i).
In (i), the fact that $\mathfrak{a}$ identifies canonically with an ideal of $A$ and $A/\mathfrak{a}$ with $\varinjlim(A_\alpha/\mathfrak{a}_\alpha)$ comes from the exactness of the functor $\varinjlim$ in the category of abelian groups.
Furthermore, $\mathfrak{a}$ is the union of the increasing family of the $\varphi_\alpha(\mathfrak{a}_\alpha)$, hence, if $x_\alpha\in A_\alpha$ is such that $\varphi_\alpha(x_\alpha)\in\mathfrak{a}$, there exists $\beta\geq\alpha$ such that $\varphi_\alpha(x_\alpha)=\varphi_\beta(y_\beta)$ for some $y_\beta\in\mathfrak{a}_\beta$; but this implies the existence of a $\gamma\geq\beta$ such that $\varphi_{\gamma\alpha}(x_\alpha)=\varphi_{\gamma\beta}(y_\beta)\in\mathfrak{a}_\gamma$, and hence $x_\alpha\in\mathfrak{a}_\alpha$ if $\mathfrak{a}_\alpha=\varphi_{\gamma\alpha}^{-1}(\mathfrak{a}_\gamma)$.
\end{proof}

\oldpage[IV-2]{132}
\begin{corollary}[5.13.2]
\label{IV.5.13.2}
If $\mathfrak{N}_\alpha$ is the nilradical of $A_\alpha$, the nilradical of $A$ identifies with $\varinjlim\mathfrak{N}_\alpha$, and $A_\mathrm{red}$ with $\varinjlim(A_\alpha)_\mathrm{red}$.
In particular, if the $A_\alpha$ are reduced, so is $A$.
\end{corollary}

\begin{proof}
It is clear that $\varphi_{\beta\alpha}(\mathfrak{N}_\alpha)\subset\mathfrak{N}_\beta$ for $\alpha\leq\beta$, and that $\varinjlim\mathfrak{N}_\alpha$ is a nilpotent ideal of $A$, hence contained in the nilradical $\mathfrak{N}$ of $A$.
Conversely, if $x\in\mathfrak{N}$, we have $x^h=0$ for some integer $h$; one can write $x=\varphi_\alpha(x_\alpha)$ for some index $\alpha$ and some $x_\alpha\in A_\alpha$; the relation $\varphi_\alpha(x_\alpha^h)=0$ implies the existence of a $\beta\geq\alpha$ such that $\varphi_{\beta\alpha}(x_\alpha^h)=0$, or equivalently $x_\beta^h=0$ on setting $x_\beta=\varphi_{\beta\alpha}(x_\alpha)$; hence $x_\beta\in\mathfrak{N}_\beta$ and since $x=\varphi_\beta(x_\beta)$ this completes the proof, taking account of \sref{IV.5.13.1}.
\end{proof}

\begin{proposition}[5.13.3]
\label{IV.5.13.3}
\begin{enumerate}[label=\emph{(\roman*)}]
  \item For every index $\alpha$, let $\mathfrak{p}_\alpha$ be a prime ideal of $A_\alpha$ such that, for $\alpha\leq\beta$, $\varphi_{\beta\alpha}(\mathfrak{p}_\alpha)\subset\mathfrak{p}_\beta$.
  Then $\mathfrak{p}=\varinjlim\mathfrak{p}_\alpha$ is a prime ideal of $A$.
  In particular, if the $A_\alpha$ are integral, so is $A$.
  \item If furthermore $\mathfrak{p}_\alpha=\varphi_{\beta\alpha}^{-1}(\mathfrak{p}_\beta)$ for $\alpha\leq\beta$, then $A_\mathfrak{p}$ identifies canonically with $\varinjlim(A_\alpha)_{\mathfrak{p}_\alpha}$.
  \item If $A$ is a local ring, $\mathfrak{p}$ its maximal ideal, and if the homomorphisms $\varphi_\alpha$ are injective, then, setting $\mathfrak{p}_\alpha=\varphi_\alpha^{-1}(\mathfrak{p})$, the homomorphisms $(A_\alpha)_{\mathfrak{p}_\alpha}\to A$ are injective.
\end{enumerate}
\end{proposition}

\begin{proof}
The two assertions of (i) are in fact equivalent, since $A/\mathfrak{p}=\varinjlim(A_\alpha/\mathfrak{p}_\alpha)$.
Suppose therefore all the $A_\alpha$ integral, and let $x$, $y$ be two elements of $A$ such that $xy=0$.
There exists thus an index $\alpha$ such that $x=\varphi_\alpha(x_\alpha)$, $y=\varphi_\alpha(y_\alpha)$ with $x_\alpha$, $y_\alpha$ in $A_\alpha$, and $\varphi_\alpha(x_\alpha y_\alpha)=0$; but this implies the existence of an index $\beta\geq\alpha$ such that, setting $x_\beta=\varphi_{\beta\alpha}(x_\alpha)$, $y_\beta=\varphi_{\beta\alpha}(y_\alpha)$, $x_\beta y_\beta=0$; since $A_\beta$ is integral, we have $x_\beta=0$ or $y_\beta=0$, which completes the proof that $A$ is integral.

Let us now prove (ii).
The hypothesis implies that the $A_\alpha-\mathfrak{p}_\alpha$ form an inductive system of multiplicative subsets of the $A_\alpha$; hence the local rings $(A_\alpha)_{\mathfrak{p}_\alpha}$ form an inductive system (where the transition homomorphisms are \emph{local}), and the canonical homomorphisms $(A_\alpha)_{\mathfrak{p}_\alpha}\to A_\mathfrak{p}$ an inductive system of homomorphisms, whose inductive limit $\psi:\varinjlim((A_\alpha)_{\mathfrak{p}_\alpha})\to A_\mathfrak{p}$ is a surjective homomorphism.
It remains to see that $\psi$ is injective.
Now, if $x_\alpha\in A_\alpha$ and $s_\alpha\in A_\alpha-\mathfrak{p}_\alpha$ are such that $\varphi_\alpha(x_\alpha)/\varphi_\alpha(s_\alpha)=0$ in $A_\mathfrak{p}$, there exists $s'\in A-\mathfrak{p}$ such that $s'\varphi_\alpha(x_\alpha)=0$, and, replacing $\alpha$ by a larger index, we can suppose $s'=\varphi_\alpha(s'_\alpha)$ with $s'_\alpha\in A_\alpha-\mathfrak{p}_\alpha$, whence $\varphi_\alpha(s'_\alpha x_\alpha)=0$; there is thus a $\beta\geq\alpha$ such that $\varphi_{\beta\alpha}(s'_\alpha x_\alpha)=0$ in $(A_\beta)_{\mathfrak{p}_\beta}$, which proves that the canonical image of $x_\alpha/s_\alpha$ in $\varinjlim((A_\alpha)_{\mathfrak{p}_\alpha})$ is zero.

Finally, the first assertion of (iii) results from (ii) and from the fact that the elements of $A_\alpha-\mathfrak{p}_\alpha$ are invertible in $A$, and that $A_{\mathfrak{p}_\alpha}=A$, and the fact that $(A_\alpha)_{\mathfrak{p}_\alpha}\to A$ is injective follows by flatness from the hypothesis that $\varphi_\alpha$ is injective.
\end{proof}

\begin{corollary}[5.13.4]
\label{IV.5.13.4}
Suppose that the $A_\alpha$ are all integral, and that the $\varphi_{\beta\alpha}$ are injective.
Then, if $A'_\alpha$ is the integral closure of $A_\alpha$, $A'=\varinjlim A'_\alpha$ is the integral closure of $A$.
In particular, if $A_\alpha$ is integrally closed for every $\alpha$, so is $A$.
If the $A_\alpha$ are local rings that are unibranch \emph{(}resp.\ geometrically unibranch\emph{)} ($\mathbf{0}$, 23.2.1) and the $\varphi_{\beta\alpha}$ are injective local homomorphisms, $A$ is unibranch \emph{(}resp.\ geometrically unibranch\emph{)}.
\end{corollary}

\begin{proof}
\oldpage[IV-2]{133}
If $K_\alpha$ is the field of fractions of $A_\alpha$, it results from \sref{IV.5.13.3}, (ii) applied to $\mathfrak{p}_\alpha=0$, that $K=\varinjlim K_\alpha$ is the field of fractions of $A$, and $A'=\varinjlim A'_\alpha$ identifies with a subring of $K$, clearly integral over $A$.
Conversely, let $z\in K$ be an element satisfying an integral dependence equation $z^n+\sum_{i=1}^n c_iz^{n-i}=0$, where the $c_i\in A$.
There exists an index $\alpha$, a $z_\alpha\in K_\alpha$ and $c_{i\alpha}\in A_\alpha$ such that $z$ (resp.\ the $c_i$) are the images of $z_\alpha$ (resp.\ the $c_{i\alpha}$), which proves that $z\in A'$.
It follows immediately that if the $A_\alpha$ are local and the $\varphi_{\beta\alpha}$ injective local homomorphisms, $A'$ is local and the $\varphi_{\beta\alpha}$ local, hence if the $A'_\alpha$ are local, $A'$ is a single extension of $k_\alpha$ for every $\alpha$, $k'=\varinjlim k'_\alpha$, which is the residue field of $A'$, is a purely inseparable (resp.\ radicial) extension of $k$, and $A$ is therefore geometrically unibranch.
\end{proof}

\begin{env}[5.13.5]
\label{IV.5.13.5}
We shall say that a ring $A$ is \emph{normal}, equivalently, if for every prime ideal $\mathfrak{p}$ of $A$, $A_\mathfrak{p}$ is \emph{integral} and \emph{integrally closed}.
We note that this implies that $A$ is \emph{reduced} and that $\mathfrak{p}$ can contain only a \emph{single} minimal prime ideal of $A$, in other words $\mathfrak{p}$ belongs to a single irreducible component of $\Spec(A)$; moreover $\mathfrak{q}$ is the inverse image of $(0)$ by the canonical homomorphism $A\to A_\mathfrak{p}$, and $A_\mathfrak{p}$ is therefore isomorphic to $(A/\mathfrak{q})_{\mathfrak{p}\mathfrak{q}}$, which proves that the integral quotient $A/\mathfrak{q}$ is \emph{integrally closed} as an intersection of integrally closed rings (\textbf{I}, 8.2.1.1).
If $A$ is \emph{Noetherian} and \emph{normal}, $A$ is a composed direct product of a finite number of integral domains that are integrally closed (\textbf{I}, 5.1.4).
\end{env}

\begin{corollary}[5.13.6]
\label{IV.5.13.6}
Suppose that the $A_\beta$ are normal and that for $\alpha\leq\beta$, every irreducible component of $\Spec(A_\beta)$ dominates an irreducible component of $\Spec(A_\alpha)$.
Then $A=\varinjlim A_\alpha$ is normal.
\end{corollary}

\begin{proof}
Indeed, let $\mathfrak{p}$ be a prime ideal of $A$ and for every $\alpha$, set $\mathfrak{p}_\alpha=\varphi_\alpha^{-1}(\mathfrak{p})$.
By hypothesis, for every $\alpha$, $\mathfrak{p}_\alpha$ contains a single minimal ideal $\mathfrak{q}_\alpha$ of $A_\alpha$, and for $\alpha\leq\beta$, $\varphi_{\beta\alpha}^{-1}(\mathfrak{q}_\beta)$ is a minimal ideal of $A_\alpha$ contained in $\mathfrak{p}_\alpha$, hence equal to $\mathfrak{q}_\alpha$.
Setting $B_\alpha=A_\alpha/\mathfrak{q}_\alpha$, the integral rings $B_\alpha$ therefore form an inductive system for the homomorphisms $\varphi'_{\beta\alpha}:B_\alpha\to B_\beta$ deduced from the $\varphi_{\beta\alpha}$ by passage to quotients, which are injective; furthermore the $B_\alpha$ are integrally closed since they are the same by definition; as furthermore $(B_\alpha)_{\mathfrak{p}'_\alpha}=(A_\alpha)_{\mathfrak{p}_\alpha}$; since $\varinjlim(B_\alpha)_{\mathfrak{p}'_\alpha}=\varinjlim(A_\alpha)_{\mathfrak{p}_\alpha}=A_\mathfrak{p}$ is an integral ring integrally closed \sref{IV.5.13.3}, this proves the corollary.
\end{proof}

\begin{proposition}[5.13.7]
\label{IV.5.13.7}
Suppose that the $A_\alpha$ are regular and that, for $\alpha\leq\beta$, $\varphi_{\beta\alpha}$ makes $A_\beta$ a flat $A_\alpha$-module; suppose furthermore that $A=\varinjlim A_\alpha$ is Noetherian.
Then $A$ is regular.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{134}
By definition, it suffices to prove that for every prime ideal $\mathfrak{p}$ of $A$, the local Noetherian ring $A_\mathfrak{p}$ is regular.
Setting $\mathfrak{p}_\alpha=\varphi_\alpha^{-1}(\mathfrak{p})$, we have $A_\mathfrak{p}=\varinjlim((A_\alpha)_{\mathfrak{p}_\alpha})$ by \sref{IV.5.13.3}; since by hypothesis the $(A_\alpha)_{\mathfrak{p}_\alpha}$ are regular and $(A_\beta)_{\mathfrak{p}_\beta}$ is a flat $(A_\alpha)_{\mathfrak{p}_\alpha}$-module for $\alpha\leq\beta$ ($\mathbf{0_I}$, 6.3.2), we see that we are reduced to the case where the $A_\alpha$ and $A$ are \emph{local} rings.
It then suffices to show that, if $M$, $N$ are two $A$-modules of finite type, $\operatorname{Tor}_i^A(M,N)=0$ for $i$ sufficiently large ($\mathbf{0}$, 17.3.1 and 17.2.6).
But we have the following lemma (where the $A_\alpha$ are again arbitrary):

\begin{lemma}[5.13.7.1]
\label{IV.5.13.7.1}
For every $A$-module $M$ of finite presentation, there exists an index $\alpha$ and an $A_\alpha$-module of finite presentation $M_\alpha$ such that $M$ is isomorphic to $M_\alpha\otimes_{A_\alpha}A$.
\end{lemma}

\begin{proof}
Indeed, one can suppose that $M=A^r/P$, where $P$ is a sub-$A$-module of finite type.
Since $A'=\varinjlim(A_\alpha^r)$ and since $P$ is of finite type, there exists an $\alpha$ and a sub-$A_\alpha$-module of finite type $P_\alpha$ of $A_\alpha^r$ such that $P$ is the image canonique of $P_\alpha\otimes_{A_\alpha}A$; the exactness to the right of the tensor product then shows that $M$ is isomorphic to $(A_\alpha^r/P_\alpha)\otimes_{A_\alpha}A$.
\end{proof}

This lemma being established, $M$ and $N$ are of finite presentation, since $A$ is Noetherian; write therefore $M=M_\alpha\otimes_{A_\alpha}A$, $N=N_\alpha\otimes_{A_\alpha}A$; by virtue of the hypothesis, $A$ is a flat $A_\alpha$-module, hence $\operatorname{Tor}_i^A(M,N)=\operatorname{Tor}_i^{A_\alpha}(M_\alpha,N_\alpha)\otimes_{A_\alpha}A$ up to isomorphism (\textbf{III}, 6.3.9); since by hypothesis $A_\alpha$ is regular, there exists $i_0$ such that $\operatorname{Tor}_i^{A_\alpha}(M_\alpha,N_\alpha)=0$ for $i\geq i_0$.
\end{proof}

\begin{corollary}[5.13.8]
\label{IV.5.13.8}
Suppose that the $A_\alpha$ are Noetherian and satisfy property $(\mathrm{R}_k)$ for an integer $k\geq 0$.
Suppose furthermore that for $\alpha\leq\beta$, $\varphi_{\beta\alpha}$ makes $A_\beta$ a flat $A_\alpha$-module, that $A=\varinjlim A_\alpha$ is Noetherian and that, for every $\alpha$, ${}^a\!\varphi_\alpha$ is a homeomorphism of $\Spec(A)$ onto $\Spec(A_\alpha)$.
Then $A$ satisfies $(\mathrm{R}_k)$.
\end{corollary}

\begin{proof}
Indeed let $\mathfrak{p}$ be a prime ideal of $A$ and, for every $\alpha$, set $\mathfrak{p}_\alpha=\varphi_\alpha^{-1}(\mathfrak{p})$, so that $\mathfrak{p}=\varinjlim\mathfrak{p}_\alpha$ and $A_\mathfrak{p}=\varinjlim(A_\alpha)_{\mathfrak{p}_\alpha}$ \sref{IV.5.13.3}.
It results from (\textbf{I}, 2.4.2) and from the hypothesis that the morphism $\Spec(A_\mathfrak{p})\to\Spec((A_\alpha)_{\mathfrak{p}_\alpha})$ is a surjective homeomorphism for every $\alpha$; we thus have $\dim(A_\mathfrak{p})=\dim((A_\alpha)_{\mathfrak{p}_\alpha})$ for every $\alpha$.
Suppose then that $\dim(A_\mathfrak{p})\leq k$; we have thus $\dim((A_\alpha)_{\mathfrak{p}_\alpha})\leq k$, and the hypothesis implies that $(A_\alpha)_{\mathfrak{p}_\alpha}$ is a regular ring; since furthermore, for $\alpha\leq\beta$, $(A_\beta)_{\mathfrak{p}_\beta}$ is a flat $(A_\alpha)_{\mathfrak{p}_\alpha}$-module ($\mathbf{0_I}$, 6.3.2), we conclude from \sref{IV.5.13.7} that $A_\mathfrak{p}$ is regular, which proves the corollary.
\end{proof}
