\section{Dimension and depth of locally Noetherian preschemes}
\label{section:IV.5}

\subsection{Dimension of preschemes}
\label{subsection:IV.5.1}

\oldpage[IV-2]{86}

\begin{env}[5.1.1]
\label{IV.5.1.1}
Let $A$ be a ring, $\mathfrak{J}$ an ideal of $A$.
Recall \sref[0]{0.16.1} that the definitions of the dimension of a ring \sref[0]{0.16.1.1} and those of the theory of the combinatorial dimension of topological spaces \sref[0]{0.14.1.2} give the following relations:
\begin{align}
  \dim(\Spec(A)) &= \dim(A) \tag{5.1.1.1}\label{IV.5.1.1.1}\\
  \dim(\mathrm{V}(\mathfrak{J})) &= \dim(A/\mathfrak{J}) \tag{5.1.1.2}\label{IV.5.1.1.2}\\
  \operatorname{codim}(\mathrm{V}(\mathfrak{J}),\Spec(A)) &= \operatorname{ht}(\mathfrak{J}) \tag{5.1.1.3}\label{IV.5.1.1.3}
\end{align}

\medskip
\begin{enumerate}
  \item[\bf(5.1.1.4)] $\Spec(A)$ is a catenary space $\Leftrightarrow$ $A$ is a catenary ring.
  \item[\bf(5.1.1.5)] $\Spec(A)$ is equidimensional $\Leftrightarrow$ $A$ is equidimensional $\Leftrightarrow$ the minimal prime ideals $\mathfrak{p}_\alpha$ of $A$ are such that the dimensions of the rings $A/\mathfrak{p}_\alpha$ are all equal.
  \item[\bf(5.1.1.6)] $\Spec(A)$ is equicodimensional $\Leftrightarrow$ $A$ is equicodimensional $\Leftrightarrow$ all maximal ideals of $A$ have the same height.
\end{enumerate}

Recall that a Noetherian ring $A$ is said to be \emph{biequidimensional} if $\Spec(A)$ is biequidimensional, that is to say if $\Spec(A)$ is Noetherian and $A$ is equidimensional, equicodimensional, catenary, and of finite dimension.
\end{env}

\begin{proposition}[5.1.2]
\label{IV.5.1.2}
Let $X$ be a prescheme, $Y$ an irreducible closed subset of $X$, $y$ the generic point of $Y$.
Then
\[
\tag{5.1.2.1}\label{IV.5.1.2.1}
  \operatorname{codim}(Y, X) = \dim(\sh{O}_{X,y})
\]
\end{proposition}

\begin{proof}
Indeed \sref[I]{I.2.4.2}, the irreducible closed subsets of $X$ containing $y$ are canonically in bijective correspondence with the irreducible closed subsets of $\Spec(\sh{O}_{X,y})$, and are therefore in bijective correspondence with the prime ideals of $\sh{O}_{X,y}$, and \eqref{IV.5.1.2.1} follows from the definitions.

In particular, we thus recover the fact that the generic points of the irreducible components of $X$ are the \emph{only} points $x\in X$ such that $\dim(\sh{O}_x)=0$ \sref[I]{I.1.1.14}.
\end{proof}

\begin{corollary}[5.1.3]
\label{IV.5.1.3}
For every closed subset $Y$ of a prescheme $X$, we have
\[
\tag{5.1.3.1}\label{IV.5.1.3.1}
  \operatorname{codim}(Y, X) = \inf_{y\in Y} \dim(\sh{O}_{X,y}).
\]
Furthermore, if $X$ is locally Noetherian, then for every $x\in Y$ we have
\[
\tag{5.1.3.2}\label{IV.5.1.3.2}
  \operatorname{codim}_x(Y, X) = \inf_{\substack{y\in Y,\, x\in\overline{\{y\}}}} \dim(\sh{O}_{X,y}).
\]
\end{corollary}

\begin{proof}
The relation \eqref{IV.5.1.3.2} follows from \eqref{IV.5.1.3.1} and \sref[0]{0.14.2.6}.
This corollary allows us to \emph{define}, for any subset $Y$ of a prescheme $X$, the \emph{codimension} $\operatorname{codim}(Y,X)$ of $Y$ in $X$ as equal to the right-hand side of \eqref{IV.5.1.3.1}.
\end{proof}

\begin{proposition}[5.1.4]
\label{IV.5.1.4}
For every prescheme $X$, we have
\[
\tag{5.1.4.1}\label{IV.5.1.4.1}
  \dim(X) = \sup_{x\in X}\big(\dim(\sh{O}_x)\big).
\]
If every irreducible closed subset of $X$ contains a closed point, then we also have
\[
\tag{5.1.4.2}\label{IV.5.1.4.2}
  \dim(X) = \sup_{x\in F}\big(\dim(\sh{O}_x)\big)
\]
where $F$ is the set of closed points of $X$.
\end{proposition}

\begin{proof}
It suffices to note (by virtue of \sref[I]{I.2.4.2}) that the chains of irreducible closed subsets of $X$ correspond bijectively to the chains of irreducible closed subsets of a local scheme of $X$; when every irreducible closed subset of $X$ contains a closed point, we can evidently restrict to the local schemes at the closed points of $X$.

We note that every irreducible closed subset of $X$ contains a closed point when $X$ is \emph{quasi-compact} \sref[0]{0.2.1.3}; we will see a little further on \sref{IV.5.1.11} that the same holds when $X$ is locally Noetherian.
\end{proof}

\begin{corollary}[5.1.5]
\label{IV.5.1.5}
For a prescheme $X$ to be catenary, it is necessary and sufficient that, for every $x\in X$, the local ring $\sh{O}_x$ be catenary.
If in addition every irreducible closed subset of $X$ contains a closed point, it suffices, for $X$ to be catenary, that $\sh{O}_x$ be catenary for every closed point $x$ of $X$.
\end{corollary}

\begin{proof}
The argument is the same as in \sref{IV.5.1.4}, since it amounts to comparing chains of irreducible closed subsets having the same endpoints.
\end{proof}

\begin{env}[5.1.6]
\label{IV.5.1.6}
We will now examine properties that are more special, linked to Noetherian hypotheses.

Recall \sref[0]{0.16.2.3} that a \emph{local} Noetherian ring $A\neq 0$ is of \emph{finite} dimension, equal to the minimum number of generators of an ideal of definition of $A$; for every prime ideal $\mathfrak{p}$ of $A$, the height of $\mathfrak{p}$, equal to $\dim(A_\mathfrak{p})$, is therefore also finite.
These properties, combined with \eqref{IV.5.1.2.1}, \eqref{IV.5.1.3.1}, and \eqref{IV.5.1.4.1}, show that:
\end{env}

\begin{proposition}[5.1.7]
\label{IV.5.1.7}
For every non-empty closed subset $Y$ of a locally Noetherian prescheme $X$, $\operatorname{codim}(Y,X)$ is finite.
If $X$ is Noetherian and affine and $Y$ is an irreducible closed subset of $X$, $\operatorname{codim}(Y,X)$ is equal to the minimum number of sections $s_i$ of $\sh{O}_X$ over $X$ such that $Y$ is an irreducible component of the set of $x\in X$ such that $s_i(x)=0$ for all $i$.
\end{proposition}

\begin{corollary}[5.1.8]
\label{IV.5.1.8}
Let $X$ be a locally Noetherian prescheme, $\sh{L}$ an invertible $\sh{O}_X$-Module, $f$ a section of $\sh{L}$ over $X$.
Then every irreducible component of the set $Z$ of $x\in X$ such that $f(x)=0$ is of codimension $\leq 1$ in $X$; it is of codimension $1$ if $Z$ does not contain any irreducible component of $X$.
\end{corollary}

\begin{proof}
\oldpage[IV-2]{88}
We can restrict to the case where $\sh{L}=\sh{O}_X$.
If $y$ is the generic point of an irreducible component $Y$ of $Z$, the ideal $(f_y)$ of $\sh{O}_{X,y}$ must be such that $\sh{O}_{X,y}/(f_y)$ has no prime ideal other than a single prime ideal, which means that $f_y$ generates an ideal of definition of the local Noetherian ring $\sh{O}_{X,y}$; we therefore have $\operatorname{codim}(Y,X)\leq 1$ \sref{IV.5.1.7}; if $Z$ does not contain any irreducible component of $X$, we cannot have $\operatorname{codim}(Y,X)=0$ by virtue of \sref[0]{0.14.2.1}.
\end{proof}

\begin{proposition}[5.1.9]
\label{IV.5.1.9}
Let $X$ be a locally Noetherian prescheme, $Y$ a closed subset of $X$.
If $x\in Y$ is such that $\sh{O}_{X,x}$ is a catenary ring, then
\[
\tag{5.1.9.1}\label{IV.5.1.9.1}
  \operatorname{codim}_x(Y, X) = \dim(\sh{O}_{X,x}) - \operatorname{codim}(\overline{\{x\}}, Y) = \dim(\sh{O}_{X,x}) - \dim(\sh{O}_{Y,x})
\]
\end{proposition}

\begin{proof}
Let $Y_i$ ($1\leq i\leq n$) be the irreducible components of $Y$ containing $x$ (which are finite in number since $Y$ is locally Noetherian), and let $y_i$ be the generic point of $Y_i$.
If we set $A=\sh{O}_{X,x}$, and if $\mathfrak{p}_i$ is the prime ideal of $A$ corresponding to $Y_i\cap\Spec(A)$, the irreducible closed subsets of $Y$ containing $x$ correspond bijectively to the prime ideals of $A$ containing one of the $\mathfrak{p}_i$, hence $\dim(\sh{O}_{Y,x})=\sup_i\dim(A/\mathfrak{p}_i)$; on the other hand, $\sh{O}_{X,y_i}=A_{\mathfrak{p}_i}$, and the hypothesis on $A$ implies $\dim(A)=\dim(A/\mathfrak{p}_i)+\dim(A_{\mathfrak{p}_i})$ \sref[0]{0.16.1.4}; the conclusion follows from the relation $\operatorname{codim}_x(Y,X)=\inf_i(\dim(\sh{O}_{X,y_i}))$ (\eqref{IV.5.1.2.1} and \sref[0]{0.14.2.6}).
\end{proof}

\begin{proposition}[5.1.10]
\label{IV.5.1.10}
\begin{enumerate}[label=(\roman*)]
  \item In every prescheme $X$, every locally constructible non-empty subset $E$ containing a point $x$ such that $\{x\}$ is a locally closed subset of $X$ (which amounts to the same as saying that $x$ is isolated in $\overline{\{x\}}$).
  \item Let $X$ be a locally Noetherian prescheme, $x$ a point of $X$ such that $\{x\}$ is locally closed in $X$; then $\dim(\{x\})\leq 1$; every point $y\neq x$ of $\{x\}$ is therefore closed in $X$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item This is a particular case of the following lemma:
\end{enumerate}
\end{proof}

\begin{lemma}[5.1.10.1]
\label{IV.5.1.10.1}
Let $X$ be a topological space having the following property: for every locally closed non-empty subset $Z\neq\varnothing$ of $X$, there exists $Z'$, a subset of $Z$, locally closed in $X$ (or, equivalently, in $Z$) and a point $x\in Z'$, \emph{closed} in $Z'$.
Then every locally closed non-empty subset $Z\neq\varnothing$ of $X$ contains a point $x$ isolated in $\overline{\{x\}}$.
\end{lemma}

\begin{proof}
Indeed, let $Z'\subset Z$ be a locally closed subset of $Z$ containing a point $x$ such that $x$ is closed in $Z'$.
There is an open neighbourhood $U$ of $x$ in $X$ such that $Z'\cap U$ is closed in $U$; this means that $x$ is also closed in $U$, hence $U\cap\overline{\{x\}}=\{x\}$, in other words $x$ is isolated in $\overline{\{x\}}$.

The lemma applies to every underlying space of a prescheme $X$, since $Z$ is then also the underlying space of a sub-prescheme \sref[I]{I.5.2.1} and it suffices to take for $Z'$ an affine open subset, which is a Kolmogoroff quasi-compact space, and therefore contains a closed point \sref[0]{0.2.1.3}.
\end{proof}

\begin{proof}[Proof of \emph{(ii)}]
Let $Z$ be the reduced sub-prescheme of $X$ having $\overline{\{x\}}$ as underlying space; the hypothesis implies that $\{x\}$ is open \emph{in} $Z$; hence, for all $z\in Z$, the generic point $x$ of $\Spec(\sh{O}_{Z,z})$ is \emph{isolated} in $\Spec(\sh{O}_{Z,z})$; but the ring $A=\sh{O}_{Z,z}$ is a local Noetherian integral ring, and the hypothesis implies that there exists $f\in A$ such that $A$ is a \emph{field}; we know \sref[0]{0.16.3.3} that this implies $\dim(A)\leq 1$.
Since $\dim(\sh{O}_{Z,z})\leq 1$ for all $z\in Z$, we indeed have $\dim(Z)\leq 1$.
\end{proof}

\begin{corollary}[5.1.11]
\label{IV.5.1.11}
If $X$ is a locally Noetherian prescheme, every non-empty closed subset of $X$ contains a closed point.
\end{corollary}

\begin{proof}
Indeed, every closed subset of $X$ is constructible \sref[0]{0.9.1.1} and \sref{IV.9.1.5}, and it suffices to apply \sref{IV.5.1.10}.
\end{proof}

\begin{env}[5.1.12]
\label{IV.5.1.12}
Let $X$ be a prescheme, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent \emph{of finite type}, $S=\operatorname{Supp}(\sh{F})$ its support, which is \emph{closed} in $X$ \sref[0]{0.5.2.2}.
For every $x\in X$, we consider $\operatorname{Supp}(\sh{F}_x)$ as a closed subset of the local scheme $\Spec(\sh{O}_x)$; by definition \sref[0]{0.16.1.7}, $\dim(\sh{F}_x)=\dim(\operatorname{Supp}(\sh{F}_x))$; but we have
\[
  \operatorname{Supp}(\sh{F}_x) = S\cap\Spec(\sh{O}_{X,x}) = \Spec(\sh{O}_{S,x})
\]
where, in the last term, $S$ denotes any closed sub-prescheme of $X$ having $S$ as underlying space.
If we note that the irreducible components of $\Spec(\sh{O}_{S,x})$ are the intersections of the irreducible components of $\Spec(\sh{O}_{X,x})$ and the irreducible components of $S$ containing $x$, and correspond bijectively to the latter, we see that
\[
\tag{5.1.12.1}\label{IV.5.1.12.1}
  \dim(\sh{F}_x) = \dim(\sh{O}_{S,x})
\]
by virtue of \eqref{IV.5.1.1.1}; it also follows from \eqref{IV.5.1.2.1} that
\[
\tag{5.1.12.2}\label{IV.5.1.12.2}
  \dim(\sh{F}_x) = \operatorname{codim}(\overline{\{x\}}, S)
\]
\oldpage[IV-2]{89}
if $X$ is locally Noetherian.

We say that $\sh{F}$ is \emph{equidimensional at the point} $x\in X$ if $\sh{F}_x$ is an $\sh{O}_{X,x}$-module equidimensional, that is to say \sref[0]{0.16.1.7} if $\operatorname{Supp}(\sh{F}_x)$ is equidimensional as a closed subset of $\Spec(\sh{O}_{X,x})$; this amounts to saying that the ring $\sh{O}_{S,x}$ is equidimensional.

We call the \emph{dimension} of $\sh{F}$ and write $\dim(\sh{F})$ the dimension of the \emph{support} $\operatorname{Supp}(\sh{F})$; it follows from \eqref{IV.5.1.4.1} and \eqref{IV.5.1.12.1} that
\[
\tag{5.1.12.3}\label{IV.5.1.12.3}
  \dim(\sh{F}) = \sup_{x\in X} \dim(\sh{F}_x).
\]

If $X=\Spec(A)$ is an affine scheme and if $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module of finite type, then $\dim(\sh{F})=\dim(M)$ by virtue of \sref[0]{0.16.1.7} and \eqref{IV.5.1.4.1}.
\end{env}

\begin{proposition}[5.1.13]
\label{IV.5.1.13}
Let $X$ be a prescheme, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent of finite type, $x$ a point of $X$, $x'$ a generization of $x$ in $X$; then
\[
\tag{5.1.13.1}\label{IV.5.1.13.1}
  \dim(\sh{F}_{x'}) \leq \dim(\sh{F}_x)
\]
\end{proposition}

\begin{proof}
This follows immediately from \eqref{IV.5.1.12.1} and from the definitions.
\end{proof}

\subsection{Dimension of an algebraic prescheme}
\label{subsection:IV.5.2}

\oldpage[IV-2]{90}

\begin{proposition}[5.2.1]
\label{IV.5.2.1}
Let $k$ be a field, $X$ an irreducible prescheme locally of finite type over $k$, $\xi$ its generic point.
Then $X$ is biequidimensional and $\dim(X)=\deg.\operatorname{tr}_k\kappa(\xi)$.
\end{proposition}

\begin{proof}
Every local ring $\sh{O}_x$ is a local ring of a $k$-algebra of finite type, hence a quotient of a regular local ring \sref[0]{0.17.3.9}, and we know \sref[0]{0.16.5.12} that these rings are catenary; consequently $X$ is a catenary space \sref{IV.5.1.5}, and since $X$ is irreducible, it suffices \eqref{IV.5.1.1.1} to show that for every \emph{closed point} $x\in X$, we have
\[
\tag{5.2.1.1}\label{IV.5.2.1.1}
  \dim(\sh{O}_x) = \deg.\operatorname{tr}_k\kappa(\xi).
\]

We may evidently suppose $X$ is reduced and affine, hence an \emph{integral} ring $A$, an algebra of finite type over $k$.
We know (Bourbaki, \emph{Alg.\ comm.}, chap.~V, \S~3, n\textsuperscript{o}~1, th.~1) that there exists a sub-$k$-algebra $B=k[t_1,\ldots,t_n]$ of $A$, where the $t_i$ are algebraically independent over $k$, such that $A$ is a $B$-algebra \emph{finite}.
Let $\mathfrak{m}=\mathfrak{i}_x$, which is by hypothesis a maximal ideal of $A$; $\mathfrak{n}=B\cap\mathfrak{m}$ is therefore a maximal ideal of $B$ (Bourbaki, \emph{Alg.\ comm.}, chap.~V, \S~2, n\textsuperscript{o}~1, prop.~1), and $A_\mathfrak{m}$ is a local ring of the $B_\mathfrak{n}$-algebra \emph{finite} $S^{-1}A$, where $S=B-\mathfrak{n}$; since $B_\mathfrak{n}$ is integrally closed, $\dim(A_\mathfrak{m})=\dim(B_\mathfrak{n})$ \sref[0]{0.16.1.6}.
We can therefore restrict to the case where $A=k[t_1,\ldots,t_n]$; we then know that $k'=\kappa(x)$ is a \emph{finite} extension of $k$ \sref[I]{I.6.4.2}; let $A'=k'[t_1,\ldots,t_n]$; taking into account \sref[I]{I.2.4.6} and \sref{IV.3.3.14}, there exists a maximal ideal $\mathfrak{m}'$ of $A'$ above $\mathfrak{m}$ and such that $k'$ is the residue field of $A'_{\mathfrak{m}'}$.
Since $A'$ is integral and is a finite $A$-algebra, the same argument as above shows that $\dim(A'_{\mathfrak{m}'})=\dim(A_\mathfrak{m})$, so we are reduced to the case where $A/\mathfrak{m}=k$.
But in this case $\mathfrak{m}$ is generated by the polynomials $t_i-a_i$ (where $a_i\in k$, the $a_i$ being the canonical images of the $t_i$ in $A/\mathfrak{m}$).
Replacing $t_i$ by $t_i-a_i$, we are finally reduced to the case where $\mathfrak{m}=\sum_{i=1}^n At_i$.
The completion of the local ring $A_\mathfrak{m}$ is then the formal power series ring $k[[t_1,\ldots,t_n]]$; we know that it has the same dimension as $A_\mathfrak{m}$ \sref[0]{0.16.2.4}, and on the other hand, the dimension of $k[[t_1,\ldots,t_n]]$ is equal to $n$ \sref[0]{0.17.1.4}; whence the conclusion.
\end{proof}

\begin{corollary}[5.2.2]
\label{IV.5.2.2}
For a prescheme $X$ locally of finite type over a field $k$, $\dim(X)$ coincides with the number defined in \sref{IV.4.1.1}.
\end{corollary}

\begin{proof}
Indeed, if $X_\alpha$ are the reduced sub-preschemes having as underlying spaces the irreducible components of $X$, we have $\dim(X)=\sup_\alpha(\dim(X_\alpha))$ \sref[0]{0.14.2.1} and it suffices to apply \sref{IV.5.2.1} to the $X_\alpha$.
\end{proof}

\begin{corollary}[5.2.3]
\label{IV.5.2.3}
Let $X$ be a prescheme locally of finite type over a field $k$, $x$ a point of $X$.
Then
\[
\tag{5.2.3.1}\label{IV.5.2.3.1}
  \dim_x(X) = \dim(\sh{O}_x) + \deg.\operatorname{tr}_k\kappa(x).
\]
\end{corollary}

\begin{proof}
\oldpage[IV-2]{91}
We know that there exists an open neighbourhood $U$ of $x$ in $X$ such that $\dim_x(X)=\dim(U)$ \sref[0]{0.14.1.4.1}, and we can suppose that the irreducible components of $U$ are the $X_i\cap U$, where the $X_i$ are the irreducible components of $X$ containing $x$; since $U\cap X_i$ is dense in $X_i$, it follows from \sref{IV.4.1.1.3} that $\dim(X_i)=\dim(U\cap X_i)$, hence $\dim_x(X)=\sup_i(\dim(X_i))$.
Furthermore, the minimal prime ideals of $\sh{O}_x$ correspond to the generic points of the $X_i$, hence \sref[0]{0.16.1.1.1}, $\dim(\sh{O}_{X,x})=\sup_i(\dim(\sh{O}_{X_i,x}))$.
We are therefore reduced to the case where $X$ is \emph{irreducible}; as $X$ is biequidimensional by \sref{IV.5.2.1}, $\dim(X)=\dim(\overline{\{x\}})+\operatorname{codim}(\overline{\{x\}},X)$ \sref[0]{0.14.3.5.1}, and we know that $\dim(\overline{\{x\}})=\deg.\operatorname{tr}_k\kappa(x)$ by \sref{IV.5.2.1} and $\operatorname{codim}(\overline{\{x\}},X)=\dim(\sh{O}_x)$ by \sref{IV.5.1.2}.
\end{proof}

\begin{corollary}[5.2.4]
\label{IV.5.2.4}
Let $X$ be a prescheme locally of finite type over a field $k$, $\sh{L}$ an invertible $\sh{O}_X$-Module, $f$ a section of $\sh{L}$ over $X$, such that the set $Y$ of $x\in X$ such that $f(x)=0$ is rare in $X$.
Then $\dim(Y)\leq\dim(X)-1$; the two members of this inequality are equal if $Y$ meets every irreducible component of maximum dimension of $X$.
\end{corollary}

\begin{proof}
If $(X_\alpha)$ is the family of irreducible components of $X$, we have
\[
  \dim(Y) = \sup_\alpha\big(\dim(Y\cap X_\alpha)\big)
\]
and we are thus reduced to the case where $X$ is irreducible (since $Y\cap X_\alpha$ is rare in $X_\alpha$, each $X_\alpha$ having a non-empty interior in $X$).
We can then restrict to the case where $Y\neq\varnothing$; then, for every maximal point $x$ of $Y$, we have (since $X$ is biequidimensional) $\dim(\overline{\{x\}})=\dim(X)-\operatorname{codim}(\overline{\{x\}},X)$, and since $Y$ is rare in $X$, we have
\[
  \operatorname{codim}(\overline{\{x\}},X) = 1
\]
by \sref{IV.5.1.8}; whence the corollary.
\end{proof}

\begin{remarks}[5.2.5]
\label{IV.5.2.5}
\begin{enumerate}[label=(\roman*)]
  \item Contrary to what holds for algebraic $k$-preschemes, a locally Noetherian prescheme $X$ is not necessarily catenary (cf.\ \sref{IV.5.6.11}); however, it is so if all its local rings $\sh{O}_x$ are quotients of regular local rings \sref[0]{0.16.5.12} and in particular if $X$ is \emph{regular} \sref[I]{I.4.1.4}.
However, even a scheme (integral) $X=\Spec(A)$, where $A$ is a \emph{regular} ring, is not necessarily \emph{biequidimensional}, in other words \sref[0]{0.14.3.3} the codimensions in $X$ of the various closed points of $X$ are not necessarily the same.
For example, let $B$ be a discrete valuation ring, $\mathfrak{m}=B\pi$ its maximal ideal, $k=B/\mathfrak{m}$ its residue field, $K$ the field of fractions of $B$; let $A$ be the polynomial ring $B[T]$.
There are in $A$ maximal ideals of \emph{height} $2$, for example $(\pi)+(T)$; but there are also maximal ideals of \emph{height} $1$, for example the principal ideal $(\pi T-1)$: indeed, a prime ideal, not the zero ideal, is of height $1$ \sref{IV.5.1.8}; on the other hand $A/(\pi T-1)$ is isomorphic to the ring of fractions $B_\pi$ \sref[0]{0.1.2.3}, which is nothing other than $K$, which proves that the ideal $(\pi T-1)$ is maximal and of height $1$.
  \item When $X$ is a prescheme locally of finite type over a field, we have seen \sref{IV.4.1.3} that $\dim(U)=\dim(X)$ for every open subset $U$ everywhere dense in $X$.
This result \emph{does not extend} to Noetherian regular preschemes, even if they are biequidimensional.
For example, let $B$ be a discrete valuation ring, $\mathfrak{m}$ its maximal ideal; $X=\Spec(A)$ has two points, $(0)$ and $\mathfrak{m}$, the latter being the only closed point; we have $\dim(X)=1$, but the open subset $U=\{(0)\}$ is of dimension $0$ (cf.\ \S~10).
\end{enumerate}
\end{remarks}

\subsection{Dimension of the support of a Module and Hilbert polynomial}
\label{subsection:IV.5.3}

\oldpage[IV-2]{92}

This section uses the results of chapter III; it will not be used in the rest of this chapter.

\begin{proposition}[5.3.1]
\label{IV.5.3.1}
Let $A$ be a local Artinian ring, $X$ a projective scheme over $\Spec(A)$, $\sh{L}$ an $\sh{O}_X$-Module invertible very ample relative to $A$, $\sh{F}$ an $\sh{O}_X$-Module coherent $\neq 0$; we set $\sh{F}(n)=\sh{F}\otimes_{\sh{O}_X}\sh{L}^{\otimes n}$ for $n\in\bb{Z}$.
Then the degree of the Hilbert polynomial $P(n)=\chi_A(\sh{F}(n))$ of $\sh{F}$ with respect to $A$ \sref[III]{III.2.5.3} is equal to the dimension of $\operatorname{Supp}(\sh{F})$.
\end{proposition}

\begin{proof}
Let us argue by induction on $d=\dim(\operatorname{Supp}(\sh{F}))$.
We know that there is a closed sub-prescheme $Y$ of $X$ whose $\operatorname{Supp}(\sh{F})$ is the underlying space, and an $\sh{O}_Y$-Module coherent $\sh{G}$ such that $\sh{F}=j_*(\sh{G})$, where $j:Y\to X$ is the canonical injection (\textbf{Err}\textsubscript{\textbf{III}},~30).
It is immediate that the Hilbert polynomials of $\sh{F}$ and $\sh{G}$ are the same, hence we can restrict to the case where $X=\operatorname{Supp}(\sh{F})$.
Suppose first that $d=0$; all the points of $X$ are closed, $X$ is an Artinian scheme \sref[I]{I.6.2.2}, hence $\sh{F}(n)=\sh{F}$ for all integers $n$, and we have by \sref[III]{III.2.5.3}
\[
  \chi_A(\sh{F}(n)) = \operatorname{length}_A(\Gamma(X,\sh{F}))
\]
for $n$ sufficiently large, and the degree of the Hilbert polynomial is therefore $0$.
Suppose now that $d>0$, and set $Z=\operatorname{Ass}(\sh{F})$, which is a finite set \sref{IV.3.1.6}; there exist an integer $m>0$ and a section $f\in\Gamma(X,\sh{L}^{\otimes m})$ such that $X_f$ is a neighbourhood of $Z$ \sref[II]{II.4.5.4}.
Multiplication by $f$ is a homomorphism
\[
  \mu_f : \sh{F} \to \sh{F}(m)
\]
which is injective \sref{IV.3.1.8}; we therefore have an exact sequence
\[
\tag{5.3.1.1}\label{IV.5.3.1.1}
  0\to\sh{F}\xrightarrow{\mu_f}\sh{F}(m)\to\sh{G}\to 0
\]
where $\sh{G}$ is coherent.
By Nakayama's lemma, the points $x\in\operatorname{Supp}(\sh{G})$ are exactly those for which $f(x)=0$.
We will now deduce that
\[
\tag{5.3.1.2}\label{IV.5.3.1.2}
  \dim(\operatorname{Supp}(\sh{G})) = d-1.
\]

This will follow from the following lemma:
\end{proof}

\begin{lemma}[5.3.1.3]
\label{IV.5.3.1.3}
Let $A$ be a local Artinian ring, $X$ a projective scheme over $\Spec(A)$, $\sh{L}$ an $\sh{O}_X$-Module invertible ample relative to $A$; then, for every section $g$ of $\sh{L}$ over $X$, the set $X_g$ of $x\in X$ such that $g(x)=0$ meets every irreducible component of $X$ of dimension $\neq 0$.
\end{lemma}

\begin{proof}
Indeed \sref[II]{II.5.5.7}, the set $X_g$ is an affine open subset of $X$, and if it contains an irreducible component $X'$ of $X$, the closed reduced sub-prescheme of $X$ having $X'$ as underlying space is both projective and affine, hence finite over $A$ \sref[III]{III.4.4.2}, and is therefore an Artinian scheme, hence of dimension $0$.
\end{proof}

\begin{proof}[Proof of \emph{(5.3.1)}, continued]
This lemma being established, note that since $Z$ contains the maximal points of $X$, $X_f$ is dense, hence $\operatorname{Supp}(\sh{G})$ is rare in $X$, and the relation \eqref{IV.5.3.1.2} follows from the lemma and from \sref{IV.5.2.4}.

\oldpage[IV-2]{93}
Since from the exact sequence \eqref{IV.5.3.1.1}, we deduce, for all $n$, the exact sequence
\[
  0\to\sh{F}(n)\to\sh{F}(n+m)\to\sh{G}(n)\to 0
\]
and for $n$ sufficiently large, we also have the exact sequence \sref[III]{III.2.2.3}
\[
  0\to\Gamma(X,\sh{F}(n))\to\Gamma(X,\sh{F}(n+m))\to\Gamma(X,\sh{G}(n))\to 0
\]
whence, taking into account \sref[III]{III.2.5.3}, for $n$ sufficiently large,
\[
  \chi_A(\sh{G}(n)) = \chi_A(\sh{F}(n+m)) - \chi_A(\sh{F}(n)).
\]
By virtue of \eqref{IV.5.3.1.2} and the induction hypothesis, the degree of the polynomial $\chi_A(\sh{G}(n))$ is $d-1$, the preceding relation implies that the degree of $\chi_A(\sh{F}(n))$ is $d$.
\end{proof}

\subsection{Dimension of the image of a morphism}
\label{subsection:IV.5.4}

\begin{proposition}[5.4.1]
\label{IV.5.4.1}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism.
\begin{enumerate}[label=(\roman*)]
  \item If $f$ is quasi-finite, $\dim(X)\leq\dim(\overline{f(X)})\leq\dim(Y)$.
  \item If $f$ is surjective and \emph{open} (resp.\ surjective and \emph{closed}), $\dim(X)\geq\dim(Y)$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item We can replace $f$ by $f_{\mathrm{red}}$ \sref[II]{II.6.2.4}, hence suppose $X$ and $Y$ reduced; if $Z$ is the closed reduced sub-prescheme of $Y$ having $\overline{f(X)}$ as underlying space, $f=j\circ g$, where $j:Z\to Y$ is the canonical injection and $g:X\to Z$ is a quasi-finite morphism \sref[I]{I.5.2.2} and \sref[II]{II.6.2.4}.
We can therefore restrict to the case where $f(X)$ is dense in $Y$.
For every $x\in X$, $\sh{O}_x$ is then an $\sh{O}_{f(x)}$-module quasi-finite \sref[II]{II.6.2.2}, and consequently $\mathfrak{m}_{f(x)}\sh{O}_x$ is an ideal of definition of $\sh{O}_x$ \sref[0]{0.7.4.4}; but we know \sref[0]{0.16.3.10} that if $A\to B$ is a local homomorphism of local Noetherian rings, such that if $\mathfrak{m}$ is the maximal ideal of $A$, $\mathfrak{m}B$ is an ideal of definition of $B$, then $\dim(B)\leq\dim(A)$; this proves (i) by virtue of \sref{IV.5.1.4}.
  \item The definition of the dimension \sref[0]{0.14.1.2} shows that it suffices to prove that for every sequence $(y_i)_{0\leq i\leq n}$ of distinct elements of $Y$ such that $y_i\in\overline{\{y_{i+1}\}}$ for $0\leq i\leq n-1$, there exists a sequence $(x_i)_{0\leq i\leq n}$ of points of $X$ such that $x_i\in\overline{\{x_{i+1}\}}$ for $0\leq i\leq n-1$ and $f(x_i)=y_i$ for all $i$.
Suppose first $f$ is surjective and \emph{open}, and show the existence of $x_i$ by induction on $i$; the existence of $x_0\in X$ such that $f(x_0)=y_0$ follows from the fact that $f$ is surjective.
If the $x_i$ have been determined for $i\leq m$ so as to satisfy $f(x_i)=y_i$ for $i\leq m$ and $x_i\in\overline{\{x_{i+1}\}}$ for $i<m$, we note that since $f$ is open and $y_{m+1}$ is a generization of $y_m$, there exists $x_{m+1}\in f^{-1}(y_{m+1})$ which is a generization of $x_m$ \sref{IV.1.10.3} and the induction can continue.

\oldpage[IV-2]{94}
Suppose now $f$ is surjective and \emph{closed}, and show the existence of $x_i$ by \emph{descending} induction on $i$, the existence of $x_n$ such that $f(x_n)=y_n$ resulting again from the fact that $f$ is surjective.
If the $x_i$ have been determined so as to satisfy the required conditions for $i>m$, we note that $f(\overline{\{x_{m+1}\}})$ is the closure of $\{f(x_{m+1})\}=\{y_{m+1}\}$ since $f$ is closed (Bourbaki, \emph{Top.\ g\'{e}n.}, chap.~I\textsuperscript{er}, 3\textsuperscript{e}~\'{e}d., \S~5, n\textsuperscript{o}~4, prop.~9); there exists therefore $x_m\in\overline{\{x_{m+1}\}}$ such that $f(x_m)=y_m$ and the descending induction can continue.
\end{enumerate}
\end{proof}

\begin{corollary}[5.4.2]
\label{IV.5.4.2}
If $X$, $Y$ are two locally Noetherian preschemes, $f:X\to Y$ a finite morphism (hence closed), then $\dim(X)=\dim(\overline{f(X)})$.
If furthermore $f$ is surjective, then $\dim(X)=\dim(Y)$.
\end{corollary}

\begin{remarks}[5.4.3]
\label{IV.5.4.3}
\begin{enumerate}[label=(\roman*)]
  \item We have seen in \sref{IV.4.1.2} that if $X$ and $Y$ are preschemes locally of finite type over a field $k$ and $f$ is a $k$-morphism, the inequality $\dim(Y)\leq\dim(X)$ is already satisfied when $f$ is \emph{quasi-compact} and \emph{dominant}.
On the other hand, one can have $\dim(Y)>\dim(X)$ even when $f$ is \emph{of finite type}, \emph{bijective}, and a \emph{local immersion}, if we only suppose $X$ and $Y$ locally Noetherian.
For example, let $A$ be a discrete valuation ring, $K$ its field of fractions, $k$ its residue field; if $Y=\Spec(A)$, and if $a$ is the generic point of $Y$ and $b$ its closed point, then $\kappa(a)=K$, $\kappa(b)=k$.
Let $X$ be the prescheme \emph{sum} of $\Spec(K)$ and $\Spec(k)$, $f:X\to Y$ the morphism which coincides with the canonical respective morphisms \sref[I]{I.2.4.5}; it is clear that $f$ is a local bijective immersion, $\{a\}$ being open in $Y$; on the other hand, $f$ is of finite type, for $K=A[\pi^{-1}]$, where $\pi$ is a uniformisant of $A$, hence $K$ is a finite $A$-algebra.
However $\dim(X)=0$ and $\dim(Y)=1$.
  \item If $A$ and $B$ are two Noetherian rings such that $A\subset B$ and $B$ is a finite $A$-algebra, the corollary \sref{IV.5.4.2} shows again that $\dim(B)=\dim(A)$ \sref[0]{0.16.1.5}.
Suppose furthermore that $A$ is a local Noetherian ring; then $B$ is a Noetherian semi-local ring (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \S~2, n\textsuperscript{o}~5, cor.~3 of prop.~9); if $\mathfrak{n}_i$ ($1\leq i\leq r$) are the maximal ideals of $B$, we have therefore
\[
\tag{5.4.3.1}\label{IV.5.4.3.1}
  \dim(A) = \sup_i \dim(B_{\mathfrak{n}_i}).
\]

We observe that, under these conditions, one does not necessarily have $\dim(B_{\mathfrak{n}_i})=\dim(A)$ for \emph{all} $i$: it suffices to see this by replacing $B$ by the direct composition of $B$ and of the residue field $k$ of $A$.
But we have at the same time examples where $A$ and $B$ are furthermore \emph{integral} rings of \emph{dimension} $2$ and where certain $B_{\mathfrak{n}_i}$ have dimension $<\dim(A)$ \sref{IV.5.6.11}.
We will see further on (\sref{IV.5.6.4} and \sref{IV.5.6.10}) that this last phenomenon cannot arise when $A$ is a quotient of a regular local ring.
\end{enumerate}
\end{remarks}

\subsection{Dimension formula for a morphism of finite type}
\label{subsection:IV.5.5}

\begin{env}[5.5.1]
\label{IV.5.5.1}
\oldpage[IV-2]{94}
Recall \sref[0]{0.16.3.9} that if $A$, $B$ are two local Noetherian rings, $k$ the residue field of $A$, $\varphi:A\to B$ a local homomorphism, then
\[
\tag{5.5.1.1}\label{IV.5.5.1.1}
  \dim(B) \leq \dim(A) + \dim(B\otimes_A k).
\]
\end{env}

We deduce:

\begin{proposition}[5.5.2]
\label{IV.5.5.2}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism, $x$ a point of $X$, $y=f(x)$.
Then
\[
\tag{5.5.2.1}\label{IV.5.5.2.1}
  \dim(\sh{O}_x) \leq \dim(\sh{O}_y) + \dim(\sh{O}_x\otimes_{\sh{O}_y}\kappa(y)).
\]
In particular, if $x$ is a maximal point of the fibre $f^{-1}(y)$, then
\[
\tag{5.5.2.2}\label{IV.5.5.2.2}
  \dim(\sh{O}_x) \leq \dim(\sh{O}_y)
\]
since $\sh{O}_x\otimes_{\sh{O}_y}\kappa(y)=\sh{O}_x/\mathfrak{m}_y\sh{O}_x$ is the local ring of $x$ in the prescheme $f^{-1}(y)$, and is of dimension $0$ by hypothesis.
\end{proposition}

\oldpage[IV-2]{95}

We will now obtain a more precise formula when we suppose that $f$ is a morphism \emph{of finite type}.

\begin{proposition}[5.5.3]
\label{IV.5.5.3}
Let $A$ be a Noetherian ring, $T$ an indeterminate, $\mathfrak{p}$ a prime ideal of $A$; then $\mathfrak{p}'=\mathfrak{p}A[T]$ is prime in $B=A[T]$ and $\mathfrak{p}'\cap A=\mathfrak{p}$.
There exists an infinity of prime ideals of $B$ distinct from $\mathfrak{p}'$, whose intersection with $A$ is $\mathfrak{p}$; these ideals are pairwise without inclusion relation.
Furthermore, if $\mathfrak{q}$ is such an ideal, then
\[
\tag{5.5.3.1}\label{IV.5.5.3.1}
  \dim(B_\mathfrak{q}) = \dim(B_{\mathfrak{p}'}) + 1 = \dim(A_\mathfrak{p}) + 1.
\]
\end{proposition}

\begin{proof}
The first assertions reduce immediately, by replacing $A$ by $A/\mathfrak{p}$ and observing that $A[T]/\mathfrak{p}A[T]=(A/\mathfrak{p})[T]$, in the case $\mathfrak{p}=0$; they follow then from the fact that the prime ideals of $B=A[T]$ whose intersection with $A$ reduces to $0$ are exactly those which do not meet the multiplicative set $S=A-\{0\}$ of the \emph{integral} ring $A$; but we know that there is a bijection on the set of prime ideals of $S^{-1}A[T]=K[T]$, where $K$ is the field of fractions of $A$ (Bourbaki, \emph{Alg.\ comm.}, chap.~II, \S~2, n\textsuperscript{o}~5, prop.~11).
Furthermore, we have, from \eqref{IV.5.5.1.1} and \eqref{IV.5.1.2.1}, $\dim(B_\mathfrak{q})\leq\dim(A_\mathfrak{p})+\dim(B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q})$, and if $k$ is the field of fractions of $A/\mathfrak{p}$, $B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q}$ is canonically identified with $(k[T])_\mathfrak{q}$, and is therefore a discrete valuation ring, hence of dimension $1$.
Finally, if $\mathfrak{p}=\mathfrak{p}_0\supset\mathfrak{p}_1\supset\cdots\supset\mathfrak{p}_m$ is a chain of prime ideals of $A$, the ideals $\mathfrak{p}_jB$ ($0\leq j\leq m$) are prime in $B$, pairwise distinct and contained in $\mathfrak{q}$; hence $\dim(B_\mathfrak{q})\geq m+1=\dim(A_\mathfrak{p})+1$.
This relation can also be written $\operatorname{ht}(\mathfrak{q})=\operatorname{ht}(\mathfrak{p})+1$; since $\mathfrak{q}\neq\mathfrak{p}'$, $\operatorname{ht}(\mathfrak{q})\geq\operatorname{ht}(\mathfrak{p}')+1\geq\operatorname{ht}(\mathfrak{p})+1$ by definition of the height of a prime ideal; this completes the proof of \eqref{IV.5.5.3.1}.
\end{proof}

\begin{corollary}[5.5.4]
\label{IV.5.5.4}
For every Noetherian ring $A$, we have
\[
\tag{5.5.4.1}\label{IV.5.5.4.1}
  \dim(A[T_1,\ldots,T_r]) = \dim(A) + r
\]
($T_i$ indeterminates).
\end{corollary}

\begin{proof}
We note by contrast that there are examples of local rings (not Noetherian) $A$ such that $\dim(A)=1$ and $\dim(A[T])=3$ [30].
\end{proof}

\begin{corollary}[5.5.5]
\label{IV.5.5.5}
For every locally Noetherian prescheme $X$, the dimension of $X[T_1,\ldots,T_r]=X\otimes_\bb{Z}\bb{Z}[T_1,\ldots,T_r]$ ($T_i$ indeterminates) is $\dim(X)+r$.
\end{corollary}

\begin{proof}
This follows from \eqref{IV.5.5.4.1} and from \sref[0]{0.14.1.7}.
\end{proof}

\begin{corollary}[5.5.6]
\label{IV.5.5.6}
Under the hypotheses of \sref{IV.5.5.3}, let $\mathfrak{q}$ be a prime ideal of $B=A[T]$ such that $\mathfrak{q}\cap A=\mathfrak{p}$; if $k$ and $k'$ are the residue fields of $A_\mathfrak{p}$ and $B_\mathfrak{q}$ respectively, then
\[
\tag{5.5.6.1}\label{IV.5.5.6.1}
  \dim(A_\mathfrak{p}) + 1 = \dim(B_\mathfrak{q}) + \deg.\operatorname{tr}_k k'
\]
\end{corollary}

\begin{proof}
\oldpage[IV-2]{96}
If $\mathfrak{q}=\mathfrak{p}B$, we have, from \eqref{IV.5.5.3.1}, $\dim(B_\mathfrak{q})=\dim(A_\mathfrak{p})$, and in this case $k'=k(T)$, hence the formula \eqref{IV.5.5.6.1} is indeed satisfied.
In the opposite case, $\dim(B_\mathfrak{q})=\dim(A_\mathfrak{p})+1$, and since $\mathfrak{q}$ corresponds to a prime ideal $\neq 0$ of $k[T]$, $k'$ is an algebraic extension of $k$, hence the formula \eqref{IV.5.5.6.1} is again the formula \eqref{IV.5.5.6.1}.
\end{proof}

\begin{lemma}[5.5.7]
\label{IV.5.5.7}
Let $A$ be a local Noetherian integral ring, $\mathfrak{m}$ its maximal ideal, $k$ its residue field.
\begin{enumerate}[label=(\roman*)]
  \item For every integral ring $B$ containing $A$, such that $B=A[x]$ (for some $x\in B$) and every prime ideal $\mathfrak{q}$ of $B$ such that $\mathfrak{q}\cap A=\mathfrak{m}$, then
\[
\tag{5.5.7.1}\label{IV.5.5.7.1}
  \dim(A) + \deg.\operatorname{tr}_A B \geq \dim(B_\mathfrak{q}) + \deg.\operatorname{tr}_k k'
\]
where $k'$ denotes the residue field of $B_\mathfrak{q}$ and $\deg.\operatorname{tr}_A B$ the transcendence degree of the field of fractions of $B$ over that of $A$.
  \item Suppose that for every maximal ideal $\mathfrak{n}$ of $A[T]$ such that $\mathfrak{n}\cap A=\mathfrak{m}$, the ring $(A[T])_\mathfrak{n}$ is catenary; then the two members of \eqref{IV.5.5.7.1} are equal.
\end{enumerate}
\end{lemma}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item If $x$ is transcendental over the field of fractions of $A$, $\deg.\operatorname{tr}_A B=1$ and the two members of \eqref{IV.5.5.7.1} are equal by virtue of \eqref{IV.5.5.6.1}.
In the opposite case, $B=A[T]/\mathfrak{p}$, where $\mathfrak{p}$ is a prime ideal $\neq 0$ of $A[T]$, such that $\mathfrak{p}\cap A=0$ since $B$ contains $A$; we have therefore $\operatorname{ht}(\mathfrak{p})=1$ by virtue of \eqref{IV.5.5.3.1}.
The ideal $\mathfrak{q}$ of $B$ is of the form $\mathfrak{n}/\mathfrak{p}$, where $\mathfrak{n}\supset\mathfrak{p}$ is a prime ideal of $A[T]$ such that $\mathfrak{n}\cap A=\mathfrak{m}$, and $B_\mathfrak{q}=(A[T])_\mathfrak{n}/\mathfrak{p}(A[T])_\mathfrak{n}$; the formula \eqref{IV.5.5.6.1}, applied to $X=\Spec((A[T])_\mathfrak{n})$ and $Y=\Spec(B_\mathfrak{q})$, gives
\[
\tag{5.5.7.2}\label{IV.5.5.7.2}
  \dim((A[T])_\mathfrak{n}) \geq \operatorname{ht}(\mathfrak{p}(A[T])_\mathfrak{n}) + \dim(B_\mathfrak{q}) = 1 + \dim(B_\mathfrak{q})
\]
since $\operatorname{ht}(\mathfrak{p}(A[T])_\mathfrak{n})=\operatorname{ht}(\mathfrak{p})=1$ by the bijective correspondence between prime ideals contained in $\mathfrak{n}$ and prime ideals of $(A[T])_\mathfrak{n}$.
Finally, the formula \eqref{IV.5.5.6.1} gives
\[
\tag{5.5.7.3}\label{IV.5.5.7.3}
  \dim((A[T])_\mathfrak{n}) = \dim(A) + 1 - \deg.\operatorname{tr}_k k'
\]
which completes the proof of \eqref{IV.5.5.7.1}.
  \item If $(A[T])_\mathfrak{n}$ is catenary, the two members of \eqref{IV.5.5.7.2} are equal \sref[0]{0.16.1.4}, hence also the two members of \eqref{IV.5.5.7.1}.
\end{enumerate}
\end{proof}

\begin{theorem}[5.5.8]
\label{IV.5.5.8}
\textbf{(dimension formula).}
Let $A$ be a local Noetherian integral ring, $B$ an integral ring containing $A$ and which is an $A$-algebra of finite type, $\mathfrak{q}$ an ideal of $B$ such that $\mathfrak{q}\cap A$ is the maximal ideal $\mathfrak{m}$ of $A$, $k$ and $k'$ the residue fields of $A$ and $B_\mathfrak{q}$ respectively.
Then the inequality
\[
\tag{5.5.8.1}\label{IV.5.5.8.1}
  \dim(A) + \deg.\operatorname{tr}_A B \geq \dim(B_\mathfrak{q}) + \deg.\operatorname{tr}_k k'.
\]
Furthermore the two members are equal if, for every sub-$A$-algebra $A'$ of finite type of $B[T]$ and every maximal ideal $\mathfrak{m}'$ of $A'$ such that $\mathfrak{m}'\cap A=\mathfrak{m}$, $A'_{\mathfrak{m}'}$ is catenary.
\end{theorem}

\begin{proof}
\oldpage[IV-2]{97}
Let $B=A[x_1,\ldots,x_n]$ and let us argue by induction on $n$.
Set $C=A[x_1,\ldots,x_{n-1}]$ and $\mathfrak{r}=\mathfrak{q}\cap C$; $C_\mathfrak{r}$ is an integral local Noetherian ring, and if we set $S=C-\mathfrak{r}$, $\mathfrak{q}'=S^{-1}\mathfrak{q}$, $B'=S^{-1}B$, $B'=C_\mathfrak{r}[x_n]$ and $\mathfrak{q}'\cap C_\mathfrak{r}=\mathfrak{r}C_\mathfrak{r}$; furthermore the fields of fractions of $B'$ and $C_\mathfrak{r}$ are those of $B$ and $C$ respectively.
If $k_1$ is the residue field of $C_\mathfrak{r}$, we have therefore, from \eqref{IV.5.5.7.1}
\[
\tag{5.5.8.2}\label{IV.5.5.8.2}
  \dim(C_\mathfrak{r}) + \deg.\operatorname{tr}_\mathfrak{c} B \geq \dim(B_\mathfrak{q}) + \deg.\operatorname{tr}_{k_1} k'.
\]
On the other hand, the induction hypothesis gives
\[
\tag{5.5.8.3}\label{IV.5.5.8.3}
  \dim(A) + \deg.\operatorname{tr}_A C \geq \dim(C_\mathfrak{r}) + \deg.\operatorname{tr}_k k_1
\]
whence \eqref{IV.5.5.8.1} by adding \eqref{IV.5.5.8.2} and \eqref{IV.5.5.8.3} member by member.
To show the second assertion, note first that any sub-$A$-algebra of finite type of $C[T]$ is also a sub-$A$-algebra of finite type of $B[T]$; by induction on $n$, we can therefore suppose that the two members of \eqref{IV.5.5.8.3} are equal.
On the other hand, to see that the two members of \eqref{IV.5.5.8.2} are equal, it suffices, by virtue of \sref{IV.5.5.7}, to verify that if $\mathfrak{n}$ is a maximal ideal of $C_\mathfrak{r}[T]$ such that $\mathfrak{n}\cap C_\mathfrak{r}=\mathfrak{r}C_\mathfrak{r}$, then $(C_\mathfrak{r}[T])_\mathfrak{n}$ is catenary; but $C_\mathfrak{r}[T]=S'^{-1}C[T]$ where $S'=C-\mathfrak{r}$, the ideal $\mathfrak{n}$ is therefore of the form $S'^{-1}\mathfrak{n}'$, where $\mathfrak{n}'$ is a prime ideal of $C[T]$ such that $\mathfrak{n}'\cap C=\mathfrak{r}$, whence $(C_\mathfrak{r}[T])_\mathfrak{n}=(C[T])_{\mathfrak{n}'}$; if $\mathfrak{m}'$ is a maximal ideal of $C[T]$ containing $\mathfrak{n}'$, $(C[T])_{\mathfrak{m}'}$ is a local ring and by hypothesis this last is catenary, hence so is $(C[T])_{\mathfrak{n}'}$ \sref[0]{0.16.1.4}.
\end{proof}

\subsection{Dimension formula and universally catenary rings}
\label{subsection:IV.5.6}

\oldpage[IV-2]{97}

\begin{proposition}[5.6.1]
\label{IV.5.6.1}
Let $A$ be a Noetherian ring.
The following properties are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item Every polynomial ring $A[T_1,\ldots,T_n]$ is catenary.
  \item Every $A$-algebra of finite type is catenary.
  \item $A$ is catenary, and for every integral $A$-algebra $B'$, local, essentially of finite type \sref{IV.1.3.8}, if we denote by $\mathfrak{p}$ the inverse image of the maximal ideal $\mathfrak{q}$ of $B'$ in $A$, by $A'$ the image of $A_\mathfrak{p}$ in $B'$, by $k$ and $k'$ the residue fields of $A'$ and $B'$ respectively,
\[
\tag{5.6.1.1}\label{IV.5.6.1.1}
  \dim(A') + \deg.\operatorname{tr}_{A'}(B') = \dim(B') + \deg.\operatorname{tr}_k k'.
\]
\end{enumerate}
\end{proposition}

\begin{proof}
The fact that $b)$ implies $c)$ follows from \eqref{IV.5.5.8.1}; indeed, $B'$ is an $A'$-algebra local, essentially of finite type, hence of the form $B''_\mathfrak{q}$, where $B''$ is a sub-$A'$-algebra of type fini of $B'$, $\mathfrak{q}''$ a prime ideal of $B''$ above the maximal ideal $\mathfrak{p}'$ of $A'$; in addition the fields of fractions of $B'$ and $B''$ are the same, hence $\deg.\operatorname{tr}_{A'}(B')=\deg.\operatorname{tr}_{A'}(B'')$.
To show \eqref{IV.5.6.1.1}, it suffices to show that (under hypothesis $b)$) every sub-$A'$-algebra $A_1$ of finite type of $B'[T]$ is catenary; indeed the two members of \eqref{IV.5.5.8.1}, where we replace $A$, $B$, and $\mathfrak{q}$ by $A'$, $B''$, and $\mathfrak{q}''$, will then be equal, whence the equality \eqref{IV.5.6.1.1}.
But hypothesis $b)$ implies that every $A$-algebra essentially of finite type is catenary \sref[0]{0.16.1.4}, and $A_1$ is such an $A$-algebra \sref{IV.1.3.9}.

It is trivial that $b)$ implies $a)$; conversely, $a)$ implies $b)$, every $A$-algebra of finite type being a quotient of a polynomial algebra \sref[0]{0.16.1.4}.

It remains to prove that $c)$ implies $b)$.
Since every quotient by a prime ideal of a finite type $A$-algebra is a finite type integral $A$-algebra, if $B$ is a finite type integral $A$-algebra, $\mathfrak{q}$ and $\mathfrak{q}'$ two prime ideals of $B$ such that $\mathfrak{q}'\subset\mathfrak{q}$, by \sref[0]{0.16.1.4.2}
\[
\tag{5.6.1.2}\label{IV.5.6.1.2}
  \dim(B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q}) + \dim(B_{\mathfrak{q}'}) = \dim(B_\mathfrak{q}).
\]

\oldpage[IV-2]{98}

Let $\mathfrak{p}$, $\mathfrak{p}'$ be the inverse images of $\mathfrak{q}$, $\mathfrak{q}'$ respectively in $A$, $\mathfrak{n}$ the kernel of the homomorphism $A_\mathfrak{p}\to B_\mathfrak{q}$.
The image $A'$ of $A_\mathfrak{p}$ in $B_\mathfrak{q}$ being isomorphic to $A_\mathfrak{p}/\mathfrak{n}$, the formula \eqref{IV.5.6.1.1} applied to $A'$ and $B'=B_\mathfrak{q}$ gives
\[
\tag{5.6.1.3}\label{IV.5.6.1.3}
  \dim(A_\mathfrak{p}/\mathfrak{n}) + \deg.\operatorname{tr}_{A_\mathfrak{p}/\mathfrak{n}}(B_\mathfrak{q}) = \dim(B_\mathfrak{q}) + \deg.\operatorname{tr}_{\kappa(\mathfrak{p})}\kappa(\mathfrak{q}).
\]

On the other hand, the kernel of $A_{\mathfrak{p}'}\to B_{\mathfrak{q}'}$ is $\mathfrak{n}A_{\mathfrak{p}'}$, hence, applying the formula \eqref{IV.5.6.1.1} where we replace $A'$ by $A_{\mathfrak{p}'}/\mathfrak{n}A_{\mathfrak{p}'}$ and $B'$ by $B_{\mathfrak{q}'}$, we have
\[
\tag{5.6.1.4}\label{IV.5.6.1.4}
  \dim(A_{\mathfrak{p}'}/\mathfrak{n}A_{\mathfrak{p}'}) + \deg.\operatorname{tr}_{A_{\mathfrak{p}'}/\mathfrak{n}A_{\mathfrak{p}'}}(B_{\mathfrak{q}'}) = \dim(B_{\mathfrak{q}'}) + \deg.\operatorname{tr}_{\kappa(\mathfrak{p}')}\kappa(\mathfrak{q}').
\]

Finally, $B/\mathfrak{q}'$ is an integral $A$-algebra of finite type, the inverse image of $\mathfrak{q}/\mathfrak{q}'$ being $\mathfrak{p}'A_\mathfrak{p}$, hence, applying the formula \eqref{IV.5.6.1.1} where we replace $A'$ by $A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p}$ and $B'$ by $B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q}$, we have
\[
\tag{5.6.1.5}\label{IV.5.6.1.5}
  \dim(A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p}) + \deg.\operatorname{tr}_{A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p}}(B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q}) = \dim(B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q}) + \deg.\operatorname{tr}_{\kappa(\mathfrak{p}')}\kappa(\mathfrak{q}).
\]

Adding now \eqref{IV.5.6.1.4} and \eqref{IV.5.6.1.5} member by member, and noting that $\kappa(\mathfrak{p}')$ (resp.\ $\kappa(\mathfrak{q}')$) is the field of fractions of $A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p}$ (resp.\ $B_\mathfrak{q}/\mathfrak{q}'B_\mathfrak{q}$); by the same token $A_\mathfrak{p}/\mathfrak{n}$ have the same field of fractions, and $B_\mathfrak{q}$ and $B_\mathfrak{q}$ have the same field of fractions; finally, since $A$ is supposed catenary, by \sref[0]{0.16.1.4.2}
\[
  \dim(A_\mathfrak{p}/\mathfrak{p}'A_\mathfrak{p}) + \dim(A_{\mathfrak{p}'}/\mathfrak{n}A_{\mathfrak{p}'}) = \dim(A_\mathfrak{p}/\mathfrak{n})
\]

Taking into account these remarks and \eqref{IV.5.6.1.3}, we obtain the relation \eqref{IV.5.6.1.2}.
\end{proof}

\begin{definition}[5.6.2]
\label{IV.5.6.2}
A Noetherian ring $A$ is said to be \emph{universally catenary} if it satisfies the equivalent conditions $a)$, $b)$, $c)$ of \sref{IV.5.6.1}.
\end{definition}

\begin{remarks}[5.6.3]
\label{IV.5.6.3}
\begin{enumerate}[label=(\roman*)]
  \item If $A$ is universally catenary, so is $S^{-1}A$ for every multiplicative subset $S$ of $A$, as it follows immediately from \sref{IV.5.6.1},~$a)$) and from the fact that every ring of fractions of a catenary ring is catenary.
Conversely, if for every prime ideal $\mathfrak{p}$ of $A$, the ring $A_\mathfrak{p}$ is universally catenary, then $A$ is universally catenary: this follows from \sref{IV.5.6.1},~$c)$), noting that if $S=A-\mathfrak{p}$, $S^{-1}B$ is an $A_\mathfrak{p}$-algebra of finite type, and $B_\mathfrak{q}=(S^{-1}B)_{S^{-1}\mathfrak{q}}$.
  \item We say that a locally Noetherian prescheme $X$ is \emph{universally catenary} if, for every $x\in X$, the ring $\sh{O}_x$ is universally catenary.
It follows from (i) that for $A$ to be a universally catenary ring, it is necessary and sufficient that the scheme $\Spec(A)$ be universally catenary.
  \item The criterion \sref{IV.5.6.1},~$c)$) only involves the images of $A$ in integral $A$-algebras of finite type, hence the \emph{integral quotient rings} of $A$.
We conclude that if $A$ is a universally catenary ring, or that $X_{\mathrm{red}}$ is universally catenary, it is necessary and sufficient that the scheme $\Spec(A)$ be so.
  \item The criterion \sref{IV.5.6.1},~$b)$) and the remark (i) show that, if $A$ is a universally catenary ring, the same is true of every $A$-algebra \emph{essentially of finite type} \sref{IV.1.3.8}.
\end{enumerate}
\end{remarks}

\begin{proposition}[5.6.4]
\label{IV.5.6.4}
Every quotient of a regular ring is universally catenary.
\end{proposition}

\begin{proof}
It amounts to seeing that a ring $A$ is universally catenary \sref{IV.5.6.3},~(iv)); or, we know that $A[T_1,\ldots,T_n]$ is regular \sref[0]{0.17.3.7}, hence catenary \sref[0]{0.16.5.12}, and we conclude by \sref{IV.5.6.1},~$a)$).

In particular, it follows from the theorem of Cohen \sref[0]{0.19.8.8} that every \emph{complete local Noetherian ring} is universally catenary.
Similarly, every algebra of finite type over a field is universally catenary.
Every prescheme \emph{locally of finite type over a field} is universally catenary.

We will see further on \sref{IV.6.3.7} that in \sref{IV.5.6.4}, one can replace ``regular'' by ``Cohen-Macaulay''.
\end{proof}

\begin{proposition}[5.6.5]
\label{IV.5.6.5}
Let $Y$ be an irreducible locally Noetherian prescheme, $X$ a prescheme irreducible, $f:X\to Y$ a dominant morphism locally of finite type.
Let $\xi$ (resp.\ $\eta$) be the generic point of $X$ (resp.\ $Y$), and let $e=\dim(f^{-1}(\eta))=\deg.\operatorname{tr}_{\kappa(\eta)}\kappa(\xi)$ (``dimension of the generic fibre'', cf.\ \sref[0]{0.2.1.8} and \sref{IV.4.1.1}).
For every $x\in X$, setting $y=f(x)$, we have
\[
\tag{5.6.5.1}\label{IV.5.6.5.1}
  e + \dim(\sh{O}_y) \geq \deg.\operatorname{tr}_{\kappa(y)}\kappa(x) + \dim(\sh{O}_x)
\]
\[
\tag{5.6.5.2}\label{IV.5.6.5.2}
  \dim(\sh{O}_x) \leq \dim(\sh{O}_y) + \dim(\sh{O}_x\otimes_{\sh{O}_y}\kappa(y)) - \delta(x)
\]
\oldpage[IV-2]{99}
where $\delta(x)=\dim_x(f^{-1}(y))-e$.

Furthermore, if $Y$ is universally catenary, the two members of \eqref{IV.5.6.5.1} (resp.\ \eqref{IV.5.6.5.2}) are equal.
If furthermore $x$ is closed in $f^{-1}(y)$, then
\[
\tag{5.6.5.3}\label{IV.5.6.5.3}
  \dim(\sh{O}_x) = \dim(\sh{O}_y) + e.
\]
\end{proposition}

\begin{proof}
We can restrict to the case where $X$ and $Y$ are affine (hence $f$ of finite type) and integral.
Since $f$ is dominant, $\sh{O}_y$ identifies with a sub-ring of $\sh{O}_x$, and the fields of fractions of $\sh{O}_x$ and $\sh{O}_y$ are respectively $\kappa(\xi)$ and $\kappa(\eta)$ \sref[I]{I.8.2.7}; furthermore, $\sh{O}_x$ is a local ring of an $\sh{O}_y$-algebra integral of finite type $B$ and an ideal prime $\mathfrak{q}$ \sref[I]{I.3.6.5}; setting $A=\sh{O}_y$, $\mathfrak{q}$ and $\mathfrak{q}$ \sref{IV.5.5.8.1} to $A=\sh{O}_y$, we obtain \eqref{IV.5.6.5.1}.
Furthermore, $\sh{O}_x\otimes_{\sh{O}_y}\kappa(y)=\sh{O}_x/\mathfrak{m}_y\sh{O}_x$ is nothing other than the local ring of the fibre $f^{-1}(y)$ at the point $x$ \sref[I]{I.3.6.1}; since $f^{-1}(y)$ is a prescheme of finite type over $\kappa(y)$, it follows from \eqref{IV.5.2.3.1} that
\[
  \dim(\sh{O}_x\otimes_{\sh{O}_y}\kappa(y)) = \dim_x(f^{-1}(y)) - \deg.\operatorname{tr}_{\kappa(y)}\kappa(x)\,;
\]
the inequality \eqref{IV.5.6.5.2} is therefore nothing other than another form of \eqref{IV.5.6.5.1}.

\oldpage[IV-2]{100}
The fact that the two members of \eqref{IV.5.6.5.1} are equal when $Y$ is universally catenary is nothing other than the equality \eqref{IV.5.6.1.1}, applied to $A=\sh{O}_y$ and $B'=\sh{O}_x$.
To show that we also have \eqref{IV.5.6.5.3} when $x$ is closed in $f^{-1}(y)$, it suffices to remark that, in a general way, $\deg.\operatorname{tr}_{\kappa(y)}\kappa(x)$ is the dimension of $\overline{\{x\}}\cap f^{-1}(y)$, as it follows from \sref{IV.5.2.1} applied to the closed reduced sub-prescheme of $f^{-1}(y)$ having this sub-space as underlying space.
\end{proof}

\begin{corollary}[5.6.6]
\label{IV.5.6.6}
Under the general hypotheses of \sref{IV.5.6.5}, we have
\[
\tag{5.6.6.1}\label{IV.5.6.6.1}
  \dim(X) \leq \dim(Y) + e.
\]
If we suppose furthermore $Y$ universally catenary, then, for the two members of \eqref{IV.5.6.6.1} to be equal, it is necessary and sufficient that
\[
\tag{5.6.6.2}\label{IV.5.6.6.2}
  \dim(Y) = \sup_{y\in f(X)} \dim(\sh{O}_y).
\]
In particular, the two members of \eqref{IV.5.6.6.1} are equal when $Y$ is locally of finite type over a field.
\end{corollary}

\begin{proof}
The inequality \eqref{IV.5.6.6.1} follows from \eqref{IV.5.6.5.1} applied at a closed point $x$ of $X$, taking into account \sref{IV.5.1.4.2} and \sref{IV.5.1.4.2}.
On the other hand, every non-empty fibre $f^{-1}(y)$ contains a closed point $x$ \sref{IV.5.1.11}.
If we suppose $Y$ universally catenary, we therefore have at this point the relation \eqref{IV.5.6.5.3}.
Taking the upper bounds of \eqref{IV.5.6.5.3} as $x$ runs over $X$, and noting that every closed point $x$ is \emph{a fortiori} closed in $f^{-1}(f(x))$, the second assertion of the corollary follows from \eqref{IV.5.1.4.1} and \eqref{IV.5.1.4.2}.

To prove the last assertion, note that there is an open affine neighbourhood $U$ of the generic point of $Y$ such that $f|U$ is of finite type, hence $f(U)$ is a \emph{constructible} subset of $Y$ \sref{IV.1.8.5}, and therefore contains a non-empty open subset (hence everywhere dense) of $Y$ \sref[0]{0.9.2.2}.
The hypothesis that $Y$ is locally of finite type over a field assures, on the one hand, that $Y$ is universally catenary \sref{IV.5.6.4}, and on the other hand that the two members of \eqref{IV.5.6.6.2} are equal \sref{IV.5.2.2} and \sref{IV.4.1.1.3}.

We note that the condition \eqref{IV.5.6.6.2} is trivially satisfied when $f$ is \emph{surjective}.
\end{proof}

\begin{corollary}[5.6.7]
\label{IV.5.6.7}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $n$ an integer.
If, for every $y\in Y$, $\dim(f^{-1}(y))\leq n$, then
\[
\tag{5.6.7.1}\label{IV.5.6.7.1}
  \dim(X) \leq \dim(Y) + n.
\]
\end{corollary}

\begin{proof}
\oldpage[IV-2]{101}
By virtue of \sref[0]{0.14.1.7} and \sref{IV.1.5.4}, we can restrict to the case where $X$ and $Y$ are affine, hence $f$ of finite type, $X$ and $Y$ reduced; let $X_i$ ($1\leq i\leq m$) be the sub-preschemes (integral) of $X$ having as underlying spaces the irreducible components of $X$; $\dim(X)=\sup_i(\dim(X_i))$.
If $Z_i$ is the closed reduced sub-prescheme of $Y$ having $\overline{f(X_i)}$ as underlying space, $Z_i$ is integral \sref[0]{0.2.1.5}; the restriction $X_i\to Y$ of $f$ factorises as $X_i\xrightarrow{g_i}Z_i\xrightarrow{j_i}Y$, where $j_i$ is the canonical injection \sref[I]{I.5.2.2}, and $g_i$ is dominant and of finite type \sref{IV.1.5.4}; if $z_i$ is the generic point of $Z_i$, $\dim(g_i^{-1}(z_i))\leq n$ for all $i$ by hypothesis; we see thus that to prove \sref{IV.5.6.7.1} it suffices to show that $\dim(X_i)\leq\dim(Z_i)+n$ for all $i$, which follows from \eqref{IV.5.6.6.1}.
\end{proof}

\begin{corollary}[5.6.8]
\label{IV.5.6.8}
Let $Y$ be a locally Noetherian prescheme, $X$ a prescheme irreducible, $f:X\to Y$ a morphism locally of finite type, $n$ an integer $\geq 0$.
Suppose that $Y$ is universally catenary, and that for every $y\in Y$, $\dim(f^{-1}(y))\geq n$ (resp.\ $\dim(f^{-1}(y))=n$).
Then
\[
\tag{5.6.8.1}\label{IV.5.6.8.1}
  \dim(X) \geq \dim(Y) + n
\]
(resp.
\[
\tag{5.6.8.2}\label{IV.5.6.8.2}
  \dim(X) = \dim(Y) + n).
\]
\end{corollary}

\begin{proof}
Since $n\geq 0$, the hypothesis implies that $f$ is \emph{surjective}, hence $Y$ is irreducible; furthermore, $\dim(f^{-1}(\eta))\geq n$ (resp.\ $\dim(f^{-1}(\eta))=n$).
The conclusion follows from \sref{IV.5.6.6}.
\end{proof}

\begin{remarks}[5.6.9]
\label{IV.5.6.9}
\begin{enumerate}[label=(\roman*)]
  \item Even if $Y$ is regular, $X$ irreducible, $f:X\to Y$ dominant and of finite type, the two members of \eqref{IV.5.6.6.1} are not necessarily equal, as is shown by the example where $Y=\Spec(A)$ where $A$ is a discrete valuation ring, $K$ the field of fractions of $A$, $f$ being the canonical morphism, $X=\Spec(K)$ where $K$ is the field of fractions of $A$.
  \item The example \sref{IV.5.4.3},~(i)) shows that in \sref{IV.5.6.6} and \sref{IV.5.6.8}, one cannot suppress the hypothesis that $X$ is irreducible, the other hypotheses being satisfied.
We will see however \sref{IV.10.6.1} that with supplementary hypotheses on $Y$, which are satisfied for example when $Y$ is a prescheme locally of finite type over a field, or over a Dedekind ring having an infinity of prime ideals, such phenomena cannot arise.
\end{enumerate}
\end{remarks}

\begin{proposition}[5.6.10]
\label{IV.5.6.10}
Let $A$ be a local Noetherian integral ring universally catenary, $B$ an integral ring containing $A$ and which is a finite $A$-algebra.
Then, for every maximal ideal $\mathfrak{n}$ of $B$, $\dim(B_\mathfrak{n})=\dim(A)$.
\end{proposition}

\begin{proof}
Indeed, $\deg.\operatorname{tr}_A B=0$ and the residue field $k'$ of $B_\mathfrak{n}$ is an algebraic extension of the field of fractions of $A$.
The conclusion follows from the formula \eqref{IV.5.6.1.1}, every maximal ideal of $B$ being above that of $A$.
\end{proof}

\begin{example}[5.6.11]
\label{IV.5.6.11}
Let $A$ be a local Noetherian integral ring and \emph{integrally closed}; then, since \sref[0]{0.16.1.6}, the conclusion of \sref{IV.5.6.10} is valid for every integral finite $A$-algebra $B$ containing $A$.
On the contrary, we will now construct an example of a local Noetherian integral \emph{catenary} ring $A$ for which the conclusion of \sref{IV.5.6.10} will be at fault.
We will use the construction of \sref{IV.5.2.5},~(i)).
Let $k_0$ be a field, $k$ a pure transcendental extension $k_0(S_i)_{i\in\bb{N}}$ of infinite transcendence degree, $V$ the discrete valuation ring, a local ring of the polynomial ring $k[S]$ at the prime ideal $(S)$; finally let $E$ be the polynomial ring $V[T]$; we have seen that in $E$ the maximal ideal $\mathfrak{m}=(S)+(T)$ is of height $2$ and the maximal ideal $\mathfrak{m}'=(ST-1)$ is of height $1$;
\oldpage[IV-2]{102}
the corresponding residue fields are $k(\mathfrak{m})=k$ and $k(\mathfrak{m}')=k(S)$, field of fractions of $V$; by virtue of the choice of $k$, these fields are \emph{isomorphic}.
Let us denote by $\varepsilon$ and $\varepsilon'$ the canonical homomorphisms of $E/\mathfrak{m}=k(\mathfrak{m})$ and $E/\mathfrak{m}'=k(\mathfrak{m}')$ respectively; let $\sigma$ be an isomorphism of $k(\mathfrak{m})$ onto $k(\mathfrak{m}')$, and consider the sub-ring $C$ of $E$ formed of $x\in E$ such that $\varepsilon'(x)=\sigma(\varepsilon(x))$ (this construction is a particular case of the ``glueing procedures'' which will be systematically studied in chap.~V).
It is immediate that $\mathfrak{n}=\mathfrak{m}\mathfrak{m}'=\mathfrak{m}\cap\mathfrak{m}'$ is a maximal ideal of $C$, $C/\mathfrak{m}\mathfrak{m}'$ identifying with the sub-field of $(E/\mathfrak{m})\times(E/\mathfrak{m}')$ formed of the pairs $(z,\sigma(z))$.
We have $E=C+C(ST-1)$, and it is evidently the integral closure of $C$; we will moreover see that $C$ is \emph{Noetherian}: this will follow from the following lemma:

\begin{lemma}[5.6.11.1]
\label{IV.5.6.11.1}
Let $R$ be a ring, $S$ a sub-ring, $\mathfrak{R}=\operatorname{Ann}_S(R/S)$ the conductor of $S$ in $R$ (which is also an ideal of $R$).
\begin{enumerate}[label=(\roman*)]
  \item For every ideal $\mathfrak{J}\subset\mathfrak{R}$ of $R$, there exists a strictly increasing map from the set of ideals $\mathfrak{L}$ of $S$ such that $R.\mathfrak{L}=\mathfrak{J}$ to the set of sub-$(S/\mathfrak{R})$-modules of $\mathfrak{J}/\mathfrak{R}\mathfrak{J}$.
  \item If $S/\mathfrak{R}$ and $R$ are Noetherian and if $R$ is a finite type $S$-module, then every increasing sequence of ideals of $S$ contained in $K$ is stationary.
\end{enumerate}
\end{lemma}

We will apply this lemma by taking $R=E$, $S=C$; it is clear that $\mathfrak{n}$ is the conductor of $C$ in $E$; furthermore, for every ideal $\mathfrak{a}$ of $C$, $\mathfrak{a}/(\mathfrak{n}\cap\mathfrak{a})$ is isomorphic to $(\mathfrak{a}+\mathfrak{n})/\mathfrak{n}$, which is a sub-$C$-module of $C/\mathfrak{n}$, hence a simple $C$-module; it suffices therefore to show that every ideal $\mathfrak{a}\cap\mathfrak{n}$ of $C$ is of finite type, and this follows from \sref{IV.5.6.11.1}.

It follows from \sref[0]{0.16.1.5} that $\dim(C)=\dim(E)=2$.
Taking $A=C_\mathfrak{n}$; if $U=C-\mathfrak{n}$, the ring of fractions $B=U^{-1}E$ is therefore the integral closure of $A$, and it is an $A$-module of finite type; moreover, since $\mathfrak{m}$ and $\mathfrak{m}'$ are the only prime ideals of $E$ containing $\mathfrak{n}$, $B$ is a semi-local ring whose local rings at the two maximal ideals are isomorphic to $E_\mathfrak{m}$ and $E_{\mathfrak{m}'}$, of dimension respectively $2$ and $1$.
This shows that $\dim(B)=2$, hence $\dim(A)=2$ \sref[0]{0.16.1.5}.
Since $A$ is a local integral ring, it is necessarily catenary (every prime ideal distinct from $0$ and from the maximal ideal being necessarily of height $1$); but it does not satisfy the conclusion of \sref{IV.5.6.10}, and \emph{a fortiori} is not universally catenary.

Note also that $\mathfrak{n}$ is an ideal of height $2$ in $C$, and that for every prime ideal $\mathfrak{p}$ of height $1$ in $C$, there is a \emph{single} prime ideal $\mathfrak{p}'$ of $E$ above $\mathfrak{p}$, necessarily of height $1$, and such that $C_\mathfrak{p}=E_{\mathfrak{p}'}$.
Indeed, there is at least one prime ideal $\mathfrak{p}'$ of $E$ above $\mathfrak{p}$, and it follows from the Cohen-Seidenberg theorem that
\oldpage[IV-2]{103}
such an ideal is necessarily of height $1$; furthermore, since $E$ is a finite $C$-algebra and $C_\mathfrak{p}$ is integrally closed (Bourbaki, \emph{Alg.\ comm.}, chap.~V, \S~1, n\textsuperscript{o}~5, cor.~5 of prop.~16), this proves our assertions.

It would be interesting to know whether every local Noetherian integral ring satisfying the conclusion of \sref{IV.5.6.10} is universally catenary; this is what Nagata has claimed [33], but his proof does not seem complete.
\end{example}

\subsection{Depth and property \texorpdfstring{$(S_k)$}{(Sk)}}
\label{subsection:IV.5.7}

\oldpage[IV-2]{103}

\begin{definition}[5.7.1]
\label{IV.5.7.1}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ an $\sh{O}_X$-Module coherent.
We call \emph{depth} (resp.\ \emph{codepth}) of $\sh{F}$ at a point $x\in X$ the number $\operatorname{prof}(\sh{F}_x)$ (resp.\ $\operatorname{coprof}(\sh{F}_x)$) \sref[0]{0.16.4.5} and \sref[0]{0.16.4.9}.
We call the \emph{codepth} of $\sh{F}$ the number
\[
\tag{7.5.1.1}\label{IV.7.5.1.1}
  \operatorname{coprof}(\sh{F}) = \sup_{x\in X}\operatorname{coprof}(\sh{F}_x).
\]

We say that $\sh{F}$ is an $\sh{O}_X$-Module de \emph{Cohen-Macaulay} at a point $x$ if $\sh{F}_x$ is an $\sh{O}_x$-module de Cohen-Macaulay, that is to say \sref[0]{0.16.5.1} if $\operatorname{coprof}(\sh{F}_x)=0$.
We say that $\sh{F}$ is an $\sh{O}_X$-Module de \emph{Cohen-Macaulay} if it is so at every point, that is to say if $\operatorname{coprof}(\sh{F})=0$.
A point $x\in X$ such that $\sh{O}_x$ is a Cohen-Macaulay ring is also called a \emph{Cohen-Macaulay point} of $X$.

We call \emph{codepth} of $X$ and write $\operatorname{coprof}(X)$ the number $\operatorname{coprof}(\sh{O}_X)$.
We say that $X$ is a \emph{prescheme de Cohen-Macaulay} if $\sh{O}_X$ is an $\sh{O}_X$-Module de Cohen-Macaulay, that is to say if $\operatorname{coprof}(X)=0$.
Every locally Noetherian prescheme of \emph{dimension} $0$ is evidently a prescheme de Cohen-Macaulay.
Saying that $\Spec(A)$ is a Cohen-Macaulay scheme means that $A$ is a Cohen-Macaulay ring \sref[0]{0.16.5.13}.
\end{definition}

\begin{definition}[5.7.2]
\label{IV.5.7.2}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ an $\sh{O}_X$-Module coherent, $k$ a positive or negative integer.
We say that $\sh{F}$ satisfies property $(S_k)$ if, for every $x\in X$,
\[
\tag{5.7.2.1}\label{IV.5.7.2.1}
  \operatorname{prof}(\sh{F}_x) \geq \inf(k, \dim(\sh{F}_x))
\]

We say that $\sh{F}$ satisfies property $(S_k)$ at a point $x\in X$ if, for every generization $x'$ of $x$ in $X$,
\[
\tag{5.7.2.2}\label{IV.5.7.2.2}
  \operatorname{prof}(\sh{F}_{x'}) \geq \inf(k, \dim(\sh{F}_{x'})).
\]

We say that $X$ satisfies property $(S_k)$ (resp.\ $(S_k)$ at a point $x$) if $\sh{O}_X$ satisfies property $(S_k)$ (resp.\ $(S_k)$ at the point $x$).

For $\sh{F}$ to satisfy property $(S_k)$, it is necessary and sufficient that it satisfy this property at every point of $X$.
If $U$ is an open subset of $X$ and $\sh{F}$ satisfies $(S_k)$, the same holds for $\sh{F}|U$; conversely, if $(U_\alpha)$ is an open covering of $X$ and $\sh{F}|U_\alpha$ satisfies $(S_k)$ for all $\alpha$, $\sh{F}$ satisfies $(S_k)$.
\end{definition}

\begin{remarks}[5.7.3]
\label{IV.5.7.3}
\begin{enumerate}[label=(\roman*)]
  \item Recall that we always have $\operatorname{prof}(\sh{F}_x)\leq\dim(\sh{F}_x)$ \sref[0]{0.16.4.5.1} if $\sh{F}_x\neq 0$.
To say that $\sh{F}$ satisfies property $(S_k)$ therefore means that $\operatorname{prof}(\sh{F}_x)\geq k$ \emph{except} at points $x\in X$ such that $\dim(\sh{F}_x)<k$, and that at these last points we have $\dim(\sh{F}_x)=\operatorname{prof}(\sh{F}_x)$, that is to say \sref[0]{0.16.5.1} that $\sh{F}$ is an $\sh{O}_X$-Module de Cohen-Macaulay at these points; we note that the points where $\dim(\sh{F}_x)=k$ are also points where $\operatorname{prof}(\sh{F}_x)=k$,
\oldpage[IV-2]{104}
hence $\sh{F}$ is again an $\sh{O}_X$-Module de Cohen-Macaulay at these points.
To say that $\sh{F}$ satisfies property $(S_k)$ \emph{for all} $k$ therefore means that $\sh{F}$ is an $\sh{O}_X$-Module \emph{de Cohen-Macaulay}.
It is clear that for $k'\geq k$, the property $(S_{k'})$ implies $(S_k)$; for $k\leq 0$, \emph{every} $\sh{O}_X$-Module coherent satisfies property $(S_k)$.
  \item To verify the condition \eqref{IV.5.7.2.1}, we can restrict to the case where $x\in\operatorname{Supp}(\sh{F})$; in the opposite case indeed $\dim(\sh{F}_x)=-\infty$ \sref[0]{0.14.1.2}.
  \item If $X=\Spec(A)$, where $A$ is a Noetherian ring, and $\sh{F}=\widetilde{M}$, where $M$ is a finite type $A$-module, we say that $M$ satisfies property $(S_k)$ if $\sh{F}$ satisfies this property.
For a locally Noetherian prescheme $X$ and any point $x\in X$, to say $\sh{F}$ satisfies property $(S_k)$ \emph{at a point} $x\in X$ means that if we set $Y=\Spec(\sh{O}_x)$, the $\sh{O}_Y$-Module $\widetilde{\sh{F}_x}$ satisfies $(S_k)$; we note that the condition \eqref{IV.5.7.2.1} in general \emph{does not} imply \eqref{IV.5.7.2.2} for every generization $x'$ of $x$, seeing as in general there is no relation between $\operatorname{prof}(\sh{F}_{x'})$ and $\operatorname{prof}(\sh{F}_{x})$ \sref[0]{0.16.4.6}.
  \item It follows immediately from the definition that if $\sh{F}$ satisfies $(S_k)$ at a point $x$, it satisfies also $(S_k)$ at every generization $x'$ of $x$.
  \item Property $(S_k)$ is above all important for $k=1$ and $k=2$; it was introduced for $k=2$ by Serre to express his criterion of normality (cf.\ \sref{IV.5.8.5}).
  \item Let $X$ be a locally Noetherian prescheme, $Y$ a closed sub-prescheme of $X$, $j:Y\to X$ the canonical injection, $\sh{G}$ a coherent $\sh{O}_Y$-Module.
It is clear that for every $x\in Y$, $\dim(\sh{G}_x)=\dim((j_*(\sh{G}))_x)$ and $\operatorname{prof}(\sh{G}_x)=\operatorname{prof}((j_*(\sh{G}))_x)$, whence $\operatorname{coprof}(\sh{G}_x)=\operatorname{coprof}((j_*(\sh{G}))_x)$.
For $\sh{G}$ to satisfy $(S_k)$, it is necessary and sufficient that $j_*(\sh{G})$ satisfy $(S_k)$.
\end{enumerate}
\end{remarks}

\begin{proposition}[5.7.4]
\label{IV.5.7.4}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ an $\sh{O}_X$-Module coherent.
Setting $S=\operatorname{Supp}(\sh{F})$, and for every $n\geq 0$, denote by $Z_n$ the set of $z\in X$ such that $\operatorname{coprof}(\sh{F}_z)>n$, so that $Z_n\subset S$.
\begin{enumerate}[label=(\roman*)]
  \item For $\sh{F}$ to satisfy property $(S_k)$, it is necessary and sufficient that, for every $n\geq 0$, we have
\[
\tag{5.7.4.1}\label{IV.5.7.4.1}
  \operatorname{codim}(Z_n, S) > n+k.
\]
  \item Suppose furthermore that the $Z_n$ are closed in $X$.
Then, for $\sh{F}$ to satisfy property $(S_k)$ at a point $x\in S$, it is necessary and sufficient that, for every $n\geq 0$, we have
\[
\tag{5.7.4.2}\label{IV.5.7.4.2}
  \operatorname{codim}_x(Z_n, S) > n+k.
\]
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item By definition \sref{IV.5.1.3}, $\operatorname{codim}(Z_n,S)=\inf_{z\in Z_n}(\dim(\sh{O}_{S,z}))$ and the inequality \eqref{IV.5.7.4.1} signifies therefore \sref{IV.5.1.12.2} that, for every $z\in X$, and every $n\geq 0$, the relation
\[
  \dim(\sh{F}_z) - \operatorname{prof}(\sh{F}_z) \geq n+1
\]
implies the relation
\[
  \dim(\sh{F}_z) \geq n+k+1
\]

But if we set $a=\dim(\sh{F}_z)$, $b=\operatorname{prof}(\sh{F}_z)$, we have $b\leq a$, and to say that for every $n\geq 0$, the relation $b\leq a-n-1$ implies $k\leq a-n-1$ is equivalent to saying that $b\geq\inf(k,a)$, whence the proposition.
  \item The argument is the same as in (i), up to restricting to $z\in X$ that are \emph{generizations} of $x$, and taking into account \sref{IV.5.1.3.2}.
\end{enumerate}
\end{proof}

\begin{proposition}[5.7.5]
\label{IV.5.7.5}
\oldpage[IV-2]{105}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ an $\sh{O}_X$-Module coherent.
For $\sh{F}$ to satisfy $(S_k)$ ($k\geq 1$), it is necessary and sufficient that for every integer $r$ such that $0\leq r<k$, every $(\sh{F}|U)$-regular suite $(f_i)_{1\leq i\leq r}$ of sections of $\sh{O}_X$ over an open subset $U$ of $X$,
\[
  (\sh{F}|U)/\Big(\sum_{i=1}^r f_i(\sh{F}|U)\Big)
\]
is without associated immersed prime cycle.
\end{proposition}

\begin{proof}
We first show the proposition for $k=1$; it is then stated as follows: \emph{for $\sh{F}$ to satisfy $(S_1)$, it is necessary and sufficient that $\sh{F}$ be without associated immersed prime cycle}.
Indeed, to say that $\sh{F}$ satisfies $(S_1)$ means that at the points $x\in\operatorname{Supp}(\sh{F})$ such that $\dim(\sh{F}_x)>0$, $\operatorname{prof}(\sh{F}_x)\geq 1$ (since at all other points of $\operatorname{Supp}(\sh{F})$, $\operatorname{prof}(\sh{F}_x)=0$, hence $\operatorname{prof}(\sh{F}_x)\geq 1$ signifies that $\mathfrak{m}_x$ is not associated to $\sh{F}_x$ \sref[0]{0.16.4.6},~(i)), or again that $x$ is not a maximal point of a sub-prescheme of $X$ having $\operatorname{Supp}(\sh{F})$ as underlying space.
On the other hand, if $\dim(\sh{F}_x)>0$, then $\dim(\sh{O}_{S,x})>0$ signifies that $x$ is not a maximal point of $\operatorname{Supp}(\sh{F})$ \sref{IV.5.1.12.1}, $\dim(\sh{F}_x)=\dim(\sh{O}_{S,x})=\operatorname{codim}(\overline{\{x\}},S)$ \sref{IV.5.1.2}, whence the conclusion.

In the second place, we show that, for $k$ arbitrary $>1$, the condition of the statement is \emph{necessary}.
We can restrict to considering the case $U=X$, and let us suppose first that $\dim(\sh{F}_x)\geq k$; then $\operatorname{prof}(\sh{F}_x)\geq k-1$ \sref[0]{0.15.2.4}; there is then an open neighbourhood $U$ of $x$ and a sequence $(f_i)_{1\leq i\leq k-1}$ of sections of $\sh{O}_X$ over $U$ such that $(f_i)_x\in\mathfrak{m}_x$ for $1\leq i\leq k-1$.
The hypothesis implies that $\sh{G}=(\sh{F}|U)/(\sum_{i=1}^{k-1}f_i(\sh{F}|U))$ is without associated prime cycle immersed, since $\dim(\sh{G}_x)=\dim(\sh{F}_x)-(k-1)\geq 1$ \sref[0]{0.16.3.4}, hence $x$ is not associated to $\sh{G}$ \sref[0]{0.16.4.6},~(i)), and our assertion follows.
Suppose in the second place that $\dim(\sh{F}_x)=r<k$; the condition of the statement requires first of all that $\dim(\sh{F}_x)\geq k$, hence $\operatorname{prof}(\sh{F}_x)\geq k-1$; as $\sh{F}$ satisfies $(S_{k-1})$, we have $\operatorname{prof}(\sh{F}_x)\geq\inf(k-1,r)=r$, hence $\operatorname{prof}(\sh{F}_x)=\operatorname{prof}(\sh{F}_x)-(k-1)$ \sref[0]{0.16.3.4}, and our assertion follows.

Let us prove finally that for $k>1$, the condition in the statement is \emph{sufficient}.
We proceed by induction on $k$; let $x$ be a point of $X$, and let us suppose first $\dim(\sh{F}_x)\geq k$.
The induction hypothesis implies $\operatorname{prof}(\sh{F}_x)\geq k-1$; taking into account \sref[0]{0.15.2.4}, there is then an open neighbourhood $U$ of $x$ and a sequence $(f_i)_{1\leq i\leq k-1}$ of sections of $\sh{O}_X$ over $U$, such that $(f_i)_x\in\mathfrak{m}_x$ for $1\leq i\leq k-1$.
The hypothesis implies that $\sh{G}=(\sh{F}|U)/(\sum_{i=1}^{k-1}f_i(\sh{F}|U))$ is without associated prime cycle immersed; but $\dim(\sh{G}_x)=\dim(\sh{F}_x)-(k-1)\geq 1$ \sref[0]{0.16.3.4}, hence $x$ is not associated to $\sh{G}$; we therefore have $\operatorname{prof}(\sh{G}_x)\geq 1$ \sref[0]{0.16.4.6}, and since $\operatorname{prof}(\sh{G}_x)=\operatorname{prof}(\sh{F}_x)-(k-1)$ \sref[0]{0.16.4.6}, $\operatorname{prof}(\sh{F}_x)\geq k$.
Suppose in the second place that $\dim(\sh{F}_x)=r<k$; as $\sh{F}$ satisfies $(S_{k-1})$, $\operatorname{prof}(\sh{F}_x)\geq\inf(k-1,r)=r$, and this completes the proof.
\end{proof}

\begin{corollary}[5.7.6]
\label{IV.5.7.6}
Suppose $\sh{F}$ satisfies $(S_k)$ ($k\geq 1$); if $(f_i)$ ($1\leq i\leq r$) is an $\sh{F}$-regular sequence of sections of $\sh{O}_X$ over $X$ and if $r<k$, $\sh{F}/(\sum_{i=1}^r f_i\sh{F})$ satisfies $(S_{k-r})$.
\end{corollary}

\begin{proof}
This follows immediately from \sref{IV.5.7.5}.
\end{proof}

\begin{corollary}[5.7.7]
\label{IV.5.7.7}
For $\sh{F}$ to satisfy $(S_2)$, it is necessary and sufficient that $\sh{F}$ be without associated immersed prime cycle and that for every open subset $U$ of $X$ and every $(\sh{F}|U)$-regular section $f$ of $\sh{O}_X$ over $U$, $(\sh{F}|U)/f(\sh{F}|U)$ be without associated immersed prime cycle.
\end{corollary}

\begin{proof}
This follows immediately from \sref{IV.5.7.5}.
\end{proof}
\begin{remarks}[5.7.8]
\label{IV.5.7.8}
Let $X$ be a locally Noetherian prescheme \emph{of dimension} $1$;
it then amounts to the same to say that $X$ is a Cohen-Macaulay prescheme, or that it satisfies $(\mathrm{S}_1)$, or that it satisfies one of the properties $(\mathrm{S}_n)$ for $n\geqslant 1$, by virtue of the definitions \sref{IV.5.7.1} and \sref{IV.5.7.2}.
By virtue of \sref{IV.5.7.5}, it therefore amounts to the same, for preschemes of dimension $1$, to say that $X$ is a Cohen-Macaulay prescheme or that it has no associated immersed prime cycle.
For example, a locally Noetherian \emph{reduced} prescheme of dimension $1$ is a Cohen-Macaulay prescheme.
\end{remarks}

\begin{proposition}[5.7.9]
\label{IV.5.7.9}
Let $A$, $B$ be two Noetherian rings, $\rho:A\to B$ a homomorphism of rings, $M$ a $B$-module such that $M_{[\mathfrak{p}]}$ is an $A$-module of finite type.
Let $\mathfrak{p}$ be a prime ideal of $A$; let the prime ideals of $B$ above $\mathfrak{p}$ and belonging to $\operatorname{Supp}(M)$ be finite in number, and let $(\mathfrak{q}_i)_{1\leqslant i\leqslant n}$ be the family of these ideals.
Then we have
\end{proposition}

\begin{align}
\label{IV.5.7.9.1}
\dim_{A_\mathfrak{p}}(M_\mathfrak{p}) &= \sup_i \dim_{B_{\mathfrak{q}_i}}(M_{\mathfrak{q}_i}) \tag{5.7.9.1}\\
\label{IV.5.7.9.2}
\operatorname{prof}_{A_\mathfrak{p}}(M_\mathfrak{p}) &= \inf_i \operatorname{prof}_{B_{\mathfrak{q}_i}}(M_{\mathfrak{q}_i}). \tag{5.7.9.2}
\end{align}

\begin{proof}
If $\mathfrak{b}$ is the annihilator of $M$, $B/\mathfrak{b}$ is identified with a sub-$A$-module of $\operatorname{End}_A(M_{[\mathfrak{p}]})$, hence is of finite type since $A$ is Noetherian; there are thus only a finite number of prime ideals of $B$ above $\mathfrak{p}$ and containing $\mathfrak{b}$, and these are precisely those which belong to $\operatorname{Supp}(M)$.
Replacing $B$ by $B/\mathfrak{b}$, which does not change the second members of \sref{IV.5.7.9.1} and \sref{IV.5.7.9.2} (by \sref[0]{0.16.1.9} and \sref[0]{0.16.4.8}), we may thus suppose that $B$ is a finite $A$-algebra.
Set $S=A-\mathfrak{p}$, and $B'=S^{-1}B$; $B'$ is a semi-local Noetherian ring whose maximal ideals are $S^{-1}\mathfrak{q}_i$ $(1\leqslant i\leqslant n)$, and as $M_\mathfrak{p}$ is an $A_\mathfrak{p}$-module of finite type, we have $\dim_{A_\mathfrak{p}}(M_\mathfrak{p})=\dim_{B'}(M_\mathfrak{p})$ \sref[0]{0.16.1.9}.
This being the case, the relation \sref{IV.5.7.9.1} becomes a particular case of \sref[0]{0.16.1.7.4}.
To prove the relation \sref{IV.5.7.9.2}, we reduce immediately, as in \sref[0]{0.16.4.8}, to the case where $\operatorname{prof}_{A_\mathfrak{p}}(M_\mathfrak{p})=0$, and the same reasoning as in \sref[0]{0.16.4.8} shows that $M_\mathfrak{p}$ contains a sub-$B'$-module of \emph{finite length}, and hence also a simple sub-$B'$-module, which is necessarily isomorphic to the residue field of one of the $B_{\mathfrak{q}_i}$; thus there is at least one index $i$ such that $\operatorname{prof}_{B_{\mathfrak{q}_i}}(M_{\mathfrak{q}_i})=0$ \sref[0]{0.16.4.6}, which finishes the proof.
\end{proof}

\oldpage[IV-2]{107}

\begin{corollary}[5.7.10]
\label{IV.5.7.10}
Suppose the hypotheses of \sref{IV.5.7.9} satisfied, and suppose in addition that $A$ is a local ring; then, for $M$ to be an $A$-module of Cohen-Macaulay, it is necessary and sufficient that, for all the maximal ideals $\mathfrak{q}_i$ of $B$ above the maximal ideal of $A$, $M_{\mathfrak{q}_i}$ be a $B_{\mathfrak{q}_i}$-module of Cohen-Macaulay, and that in addition all the numbers $\dim_{B_{\mathfrak{q}_i}}(M_{\mathfrak{q}_i})$ be equal.
\end{corollary}

It follows immediately from \sref{IV.5.7.9.1} and \sref{IV.5.7.9.2} that these conditions are equivalent to the relation $\dim_A(M)=\operatorname{prof}_A(M)$.

\begin{corollary}[5.7.11]
\label{IV.5.7.11}
Suppose the hypotheses of \sref{IV.5.7.9} satisfied.
\begin{enumerate}[label=\emph{\roman*)}]
  \item If $M_{[\mathfrak{p}]}$ satisfies the property $(\mathrm{S}_k)$, so does $M$.
  \item Suppose that for every pair of prime ideals $\mathfrak{q}$, $\mathfrak{q}'$ of $B$ such that $\rho^{-1}(\mathfrak{q})=\rho^{-1}(\mathfrak{q}')$, we have $\dim_{B_\mathfrak{q}}(M_\mathfrak{q})=\dim_{B_{\mathfrak{q}'}}(M_{\mathfrak{q}'})$.
Then, if $M$ satisfies the property $(\mathrm{S}_k)$, so does $M_{[\mathfrak{p}]}$.
\end{enumerate}
\end{corollary}

This follows immediately from the relations \sref{IV.5.7.9.1} and \sref{IV.5.7.9.2} and from the definition of the property $(\mathrm{S}_k)$.

\begin{env}[5.7.12]
\label{IV.5.7.12}
In accordance with the definitions of \sref{IV.5.7.1}, given a Noetherian ring $A$ and a finite type $A$-module $M$, one defines $\operatorname{coprof}_A(M)$ as equal to $\operatorname{coprof}(M)=\sup_{x\in X}(\operatorname{coprof}_{A_x}(M_x))$, where $X=\Spec(A)$; we shall see later on \sref{IV.6.11.5} that this definition \emph{coincides} with that of \sref[0]{0.16.4.9} when $A$ is a Noetherian \emph{local} ring.
\end{env}

\begin{corollary}[5.7.13]
\label{IV.5.7.13}
Let $A$, $B$ be two Noetherian rings, $\rho:A\to B$ a homomorphism of rings, $M$ a $B$-module such that $M_{[\mathfrak{p}]}$ is an $A$-module of finite type.
Then we have
\end{corollary}

\begin{equation}
\label{IV.5.7.13.1}
\operatorname{coprof}_A(M_{[\mathfrak{p}]})\geqslant\operatorname{coprof}_B(M). \tag{5.7.13.1}
\end{equation}

This follows from the preceding definition and from the relations \sref{IV.5.7.9.1} and \sref{IV.5.7.9.2}.

\subsection{Regular preschemes and property \texorpdfstring{$(\mathrm{R}_k)$}{(Rk)}. Serre's normality criterion}
\label{subsection:IV.5.8}

\begin{env}[5.8.1]
\label{IV.5.8.1}
Recall \sref[0]{0.4.1.4} that a ringed space $(X,\sh{O}_X)$ is said to be \emph{regular} at a point $x\in X$, if $\sh{O}_x$ is a regular local ring.
When dealing with preschemes, we shall only use this terminology in this chapter when $X$ is \emph{locally Noetherian}.
\end{env}

\begin{definition}[5.8.2]
\label{IV.5.8.2}
We say that a locally Noetherian prescheme $X$ is \emph{regular in codimension $\leqslant k$}, or possesses the property $(\mathrm{R}_k)$, if the set of points where $X$ is \emph{not regular} is of codimension $>k$ (in other words \sref{IV.5.1.3}), if, for every $x\in X$, the relation $\dim(\sh{O}_x)\leqslant k$ implies that $\sh{O}_x$ is regular).
\end{definition}

To say that $X$ is regular means that $X$ possesses the property $(\mathrm{R}_k)$ for every $k$.

If $X=\Spec(A)$, where $A$ is a Noetherian ring, we say that $A$ possesses the property $(\mathrm{R}_k)$ if $X$ possesses this property; to say that $X$ is regular means that the ring $A$ is \emph{regular} \sref[0]{0.17.3.6}.
For a locally Noetherian prescheme $X$, we say that $X$ \emph{possesses the property} $(\mathrm{R}_k)$ \emph{at a point} $x\in X$ if the local ring $\sh{O}_x$ possesses the property $(\mathrm{R}_k)$; this means that for every \emph{generization} $x'$ of $x$ in $X$, the relation $\dim(\sh{O}_{x'})\leqslant k$ implies that $\sh{O}_{x'}$ is a regular local ring.
To say that $X$ is regular at a point $x$ amounts to saying that $X$ satisfies $(\mathrm{R}_n)$ for \emph{all} $n\geqslant 0$ at the point $x$, by virtue of \sref[0]{0.17.3.6}.

\begin{proposition}[5.8.3]
\label{IV.5.8.3}
If $k$ is a field, $X$ a prescheme locally of finite type over $k$, then, for every $x\in X$, there exists an open neighbourhood of $x$ in $X$ isomorphic to a sub-scheme of a regular $k$-scheme.
\end{proposition}

\begin{proof}
Indeed, there is an open affine neighbourhood $U$ of $x$ isomorphic to a $k$-scheme of the form $\Spec(A)$, where $A$ is a $k$-algebra of finite type; $A$ is thus isomorphic to a quotient of a polynomial algebra $k[T_1,\ldots,T_n]$, and one knows that this latter is a regular ring \sref[0]{0.17.3.7}.
\end{proof}

\begin{env}[5.8.4]
\label{IV.5.8.4}
By virtue of \sref{IV.5.8.2}, to say that $X$ possesses the property $(\mathrm{R}_0)$ means that for every maximal point $x$ of $X$, the ring $\sh{O}_x$ is a \emph{field} \sref[0]{0.17.1.4}, in other words that $X$ is \emph{reduced} at this point.
As the set $U$ of $x\in X$ where $X$ is reduced is \emph{open} (the

\oldpage[IV-2]{108}

nilradical of $X$ being a coherent $\sh{O}_X$-Module \sref[I]{I.6.1.1} and \sref[0]{0.5.2.2}), this amounts to saying that $X$ possesses the property $(\mathrm{R}_0)$ or that the set $U$ is \emph{everywhere dense}.
Consequently:
\end{env}

\begin{proposition}[5.8.5]
\label{IV.5.8.5}
For a locally Noetherian prescheme $X$ to be reduced, it is necessary and sufficient that it satisfies the properties $(\mathrm{S}_1)$ and $(\mathrm{R}_0)$.
\end{proposition}

\begin{proof}
Taking into account \sref{IV.5.7.5}, this follows from the preceding remark and from \sref{IV.3.2.1}.
\end{proof}

\begin{theorem}[5.8.6]
\label{IV.5.8.6}
\emph{(Serre's criterion).}
Let $X$ be a locally Noetherian prescheme.
For $X$ to be normal, it is necessary and sufficient that $X$ satisfies the properties $(\mathrm{S}_2)$ and $(\mathrm{R}_1)$; in other words, it is necessary and sufficient that, for every $x\in X$, the following properties are satisfied:
\begin{enumerate}[label=\emph{\roman*)}]
  \item If $\dim(\sh{O}_x)\leqslant 1$, $\sh{O}_x$ is regular (that is to say, a field or a discrete valuation ring \sref[0]{0.17.1.4}).
  \item If $\dim(\sh{O}_x)\geqslant 2$, then $\operatorname{prof}(\sh{O}_x)\geqslant 2$.
\end{enumerate}
\end{theorem}

\begin{proof}
The conditions are \emph{necessary}.
Indeed, to say that $X$ is normal means that, for every $x\in X$, $\sh{O}_x$ is an integral local Noetherian ring.
If $\dim(\sh{O}_x)=0$ (resp.\ $\dim(\sh{O}_x)=1$), we conclude that $\sh{O}_x$ is a field since $\sh{O}_x$ is integral (resp.\ that $\sh{O}_x$ is a discrete valuation ring, by virtue of \sref[II]{II.7.1.6}).
On the other hand, for every element $f_x\neq 0$ of $\sh{O}_x$, one knows (Bourbaki, \emph{Alg.\ comm.}, chap.~VII, \S~1, no~4, prop.~8) that the prime ideals associated to $\sh{O}_x/f_x\sh{O}_x$ are not immersed, hence $\sh{O}_x$ satisfies $(\mathrm{S}_2)$ \sref{IV.5.7.7}.

The conditions are \emph{sufficient}.
Indeed, it follows first from \sref{IV.5.8.5} that $X$ is reduced.
The question being local, one can further suppose that $X=\Spec(A)$, where $A$ is a Noetherian integral \emph{reduced} ring \sref[I]{I.5.1.4}; if $R$ is the total ring of fractions of $A$, $R$ is composed of a direct sum of a finite number of fields, and (taking into account \sref[II]{II.6.3.6}), it suffices to prove that $A$ is \emph{integrally closed} in $R$.
Let thus $h=f/g$ be an element of $R$ integral over $A$, $g$ and $f$ being elements of $A$ such that $g$ is not a divisor of $0$.
We have a relation of the form
\begin{equation}
\label{IV.5.8.6.1}
f^n+\sum_{i=1}^{n} a_i f^{n-i} g^i = 0 \quad\text{with}\quad a_i\in A \quad\text{for}\quad 1\leqslant i\leqslant n. \tag{5.8.6.1}
\end{equation}

Let $\mathfrak{p}$ be a prime ideal of $A$ such that $\dim(A_\mathfrak{p})=1$; if $f_\mathfrak{p}$ and $g_\mathfrak{p}$ are the images of $f$ and $g$ in $A_\mathfrak{p}$, it follows from \sref{IV.5.8.6.1} that $f_\mathfrak{p}/g_\mathfrak{p}$ (which belongs to the total ring of fractions of $A_\mathfrak{p}$, since $g_\mathfrak{p}$ is not a divisor of $0$ in $A_\mathfrak{p}$ by flatness \sref[0]{0.5.3.1}) is integral over $A_\mathfrak{p}$; but as $A_\mathfrak{p}$ is regular, hence integrally closed, we have $f_\mathfrak{p}/g_\mathfrak{p}\in A_\mathfrak{p}$.
In other words, we have $(fA)_\mathfrak{p}\subset(gA)_\mathfrak{p}$.
But the hypothesis $(\mathrm{S}_2)$ implies \sref{IV.5.7.7}, since $g$ is not a divisor of $0$ in $A$, that $A/gA$ has no associated prime ideal of codimension $1$; or, $gA$ is the intersection of primary ideals $\mathfrak{q}_i$ corresponding to the $\mathfrak{p}_i$, and from what we have just seen, the $\mathfrak{q}_i$ are, by the homomorphisms $A\to A_{\mathfrak{p}_i}$, the ideals $(gA)_{\mathfrak{p}_i}$ (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \S~2, no~3, prop.~5).
But by virtue of the Hauptidealsatz \sref[0]{0.16.3.2}, we have $\dim(A_{\mathfrak{p}_i})=1$ for $1\leqslant i\leqslant n$, hence $(fA)_{\mathfrak{p}_i}\subset(gA)_{\mathfrak{p}_i}$ for every $i$ by what precedes; as $fA$ is contained in the intersection of the $(fA)_{\mathfrak{p}_i}$ $(1\leqslant i\leqslant n)$, we have $fA\subset gA$, that is to say $f/g\in A$.
\end{proof}

\subsection{\texorpdfstring{$Z$}{Z}-pure and \texorpdfstring{$Z$}{Z}-closed modules}
\label{subsection:IV.5.9}

Some of the notions and results of this section and of the next are special cases of notions and results developed in chapter~III in the theory of local cohomology.
For the convenience of the reader, we give here an independent exposition.

\begin{env}[5.9.1]
\label{IV.5.9.1}
Let $X$ be a locally Noetherian prescheme, $Z$ a part of $X$ \emph{stable by specialisation}: this means that for every finite part $M$ of $Z$, the closure of $M$ is contained in $Z$, and hence $Z$ is the union of a filtered increasing family $(Z_\alpha)$ of closed parts of $X$; conversely, it is clear that such a union is stable by specialisation.

Set $U_\alpha=X-Z_\alpha$, so that $X-Z$ is the intersection of the filtered decreasing family of open sets $U_\alpha$; let $i_\alpha:U_\alpha\to X$ be the canonical injection, and for $U_\alpha\supset U_\beta$, let $i_{\alpha\beta}:U_\beta\to U_\alpha$ be the canonical injection, so that we have $i_\beta=i_\alpha\circ i_{\alpha\beta}$.
Let $\sh{F}$ be an $\sh{O}_X$-Module (not necessarily quasi-coherent); we have therefore $(i_\beta)_*(\sh{F}|U_\beta)=(i_\alpha)_*((i_{\alpha\beta})_*(\sh{F}|U_\beta))$; by the canonical homomorphism \sref[0]{0.4.4.3.2}
\[
  \sh{F}|U_\alpha \to (i_{\alpha\beta})_*(\sh{F}|U_\beta)
\]
we thus deduce, by application of the functor $(i_\alpha)_*$, a homomorphism
\[
  \rho_{\beta\alpha}:(i_\alpha)_*(\sh{F}|U_\alpha)\to(i_\beta)_*(\sh{F}|U_\beta)
\]
and one verifies immediately that $\rho_{\gamma\alpha}=\rho_{\gamma\beta}\circ\rho_{\beta\alpha}$ for $U_\alpha\supset U_\beta\supset U_\gamma$; in other words, the $\sh{O}_X$-Modules $(i_\alpha)_*(\sh{F}|U_\alpha)$ form an \emph{inductive system} for the homomorphisms $\rho_{\beta\alpha}$.
We set
\end{env}

\begin{equation}
\label{IV.5.9.1.1}
\sh{H}^0_{X/Z}(\sh{F})=\varinjlim_\alpha (i_\alpha)_*(\sh{F}|U_\alpha). \tag{5.9.1.1}
\end{equation}

This $\sh{O}_X$-Module does not depend on the increasing family $(Z_\alpha)$ of closed sets whose union is $Z$: indeed, let $V$ be a Noetherian open of $X$; one knows \sref[G]{G.II.3.10.1} that in the category of $\sh{O}_V$-Modules, the functor $\sh{G}\mapsto\Gamma(V,\sh{G})$ commutes with inductive limits; we thus have, by virtue of \sref{IV.5.9.1.1}

\begin{equation}
\label{IV.5.9.1.2}
\Gamma(V,\sh{H}^0_{X/Z}(\sh{F}))=\varinjlim_\alpha \Gamma(V\cap U_\alpha,\sh{F}). \tag{5.9.1.2}
\end{equation}

Let then $(Z'_\lambda)$ be a second filtered increasing family of closed sets of $X$, of union $Z$; $V\cap Z_\alpha$ is thus the union of the $V\cap Z_\alpha\cap Z'_\lambda$; but $V\cap Z_\alpha$ is locally closed in $X$, hence every closed irreducible part of $V\cap Z_\alpha$ admits a generic point; as the $V\cap Z_\alpha\cap Z'_\lambda$ are closed in $V\cap Z_\alpha$ and form (for $\alpha$ fixed) a filtered increasing family, there exists an index $\lambda$ such that $V\cap Z_\alpha\cap Z'_\lambda=V\cap Z_\alpha$ \sref[0]{0.9.2.4}, in other words $V\cap Z_\alpha\subset V\cap Z'_\lambda$.
This proves that the filtered decreasing families $V\cap U_\alpha$, $V\cap U'_\lambda$ (where $U'_\lambda=X-Z'_\lambda$) are cofinal with each other, whence our assertion, by virtue of \sref{IV.5.9.1.2}.

\oldpage[IV-2]{110}

One notes that the set $Z$ is not necessarily constructible: one has an example of this by taking $X=\Spec(A)$, where $A$ is a Noetherian integral ring having an infinity of maximal ideals, and $Z$ the complement of the generic point of $X$.

If $Z$ is closed and if $i:X-Z\to X$ is the canonical injection, we have

\begin{equation}
\label{IV.5.9.1.3}
\sh{H}^0_{X/Z}(\sh{F})=i_*(\sh{F}|X-Z) \tag{5.9.1.3}
\end{equation}

and in particular, for $Z=\varnothing$, $\sh{H}^0_{X/Z}(\sh{F})=\sh{F}$.

\begin{proposition}[5.9.2]
\label{IV.5.9.2}
\begin{enumerate}[label=\emph{\roman*)}]
  \item The functor $\sh{F}\mapsto\sh{H}^0_{X/Z}(\sh{F})$ is exact on the left.
  \item If $\sh{F}$ is quasi-coherent, so is $\sh{H}^0_{X/Z}(\sh{F})$.
\end{enumerate}
\end{proposition}

\begin{proof}
Assertion~(i) follows from the definition \sref{IV.5.9.1.1}, from the fact that $(i_\alpha)_*$ is a functor exact on the left, and from the fact that the inductive limit preserves exactness in the category of $\sh{O}_X$-Modules.
Assertion~(ii) follows from \sref[I]{I.9.2.2} and from the fact that an inductive limit of quasi-coherent $\sh{O}_X$-Modules is quasi-coherent \sref[I]{I.1.3.9}.
\end{proof}

\begin{remark}[5.9.3]
\label{IV.5.9.3}
If $\sh{F}$ is an $\sh{O}_X$-\emph{Algebra}, so is $\sh{H}^0_{X/Z}(\sh{F})$ \sref[0]{0.4.2.4}; in particular $\sh{H}^0_{X/Z}(\sh{O}_X)$ is a quasi-coherent $\sh{O}_X$-Algebra, and for every $\sh{O}_X$-Module $\sh{F}$, $\sh{H}^0_{X/Z}(\sh{F})$ is an $\sh{H}^0_{X/Z}(\sh{O}_X)$-Module, which is quasi-coherent if $\sh{F}$ is quasi-coherent \sref[I]{I.9.6.1}.
More particularly, suppose that $X=\Spec(A)$, where $A$ is integral and Noetherian; then $\sh{H}^0_{X/Z}(\sh{O}_X)$ is the $\sh{O}_X$-Algebra $\widetilde{B}$, where
\end{remark}

\begin{equation}
\label{IV.5.9.3.1}
B=\bigcap_{\mathfrak{p}\in X-Z} A_\mathfrak{p}. \tag{5.9.3.1}
\end{equation}

This follows immediately from \sref{IV.5.9.1.2} and from \sref[I]{I.8.2.1.1}.

\begin{proposition}[5.9.4]
\label{IV.5.9.4}
Let $X$ be a locally Noetherian prescheme, $Z$ a part of $X$ stable by specialisation, $X'$ a locally Noetherian prescheme, $f:X'\to X$ a flat morphism.
Then $Z'=f^{-1}(Z)$ is stable by specialisation, and for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, there is a canonical isomorphism
\end{proposition}

\begin{equation}
\label{IV.5.9.4.1}
f^*(\sh{H}^0_{X/Z}(\sh{F}))\xrightarrow{\sim}\sh{H}^0_{X'/Z'}(f^*(\sh{F})) \tag{5.9.4.1}
\end{equation}

\begin{proof}
Indeed, with the notations of \sref{IV.5.9.1}, $Z'_\alpha=f^{-1}(Z_\alpha)$ is closed in $X'$ and $Z'$ is the union of the $Z'_\alpha$; furthermore, $(i_\alpha)_{(X')}$ is the canonical injection $i'_\alpha:U'_\alpha\to X'$, and if $U'_\alpha=X'-Z'_\alpha=f^{-1}(U_\alpha)$.
As $f$ is flat, one knows \sref{IV.2.3.1} that the canonical homomorphism $f^*((i_\alpha)_*(\sh{F}|U_\alpha))\to(i'_\alpha)_*(f^*(\sh{F})|U'_\alpha)$ is bijective; as, for $\alpha\leqslant\beta$, the diagram
\[
\xymatrix{
  f^*((i_\alpha)_*(\sh{F}|U_\alpha)) \ar[r]^-{\sim} \ar[d] & (i'_\alpha)_*(f^*(\sh{F})|U'_\alpha) \ar[d] \\
  f^*((i_\beta)_*(\sh{F}|U_\beta)) \ar[r]^-{\sim} & (i'_\beta)_*(f^*(\sh{F})|U'_\beta)
}
\]

\oldpage[IV-2]{111}

is commutative, one has therefore, passing to the limit, a canonical isomorphism $\varinjlim_\alpha(f^*((i_\alpha)_*(\sh{F}|U_\alpha)))\xrightarrow{\sim}\sh{H}^0_{X'/Z'}(f^*(\sh{F}))$.
But as the functor $f^*$ commutes with inductive limits \sref[0]{0.4.3.2}, this gives by definition the sought isomorphism \sref{IV.5.9.4.1}.
\end{proof}

\begin{corollary}[5.9.5]
\label{IV.5.9.5}
Under the hypotheses of \sref{IV.5.9.4}, if $\sh{H}^0_{X/Z}(\sh{F})$ is coherent, so is $\sh{H}^0_{X'/Z'}(f^*(\sh{F}))$.
The converse is true when $f$ is a faithfully flat and quasi-compact morphism.
\end{corollary}

\begin{proof}
The first assertion follows from \sref{IV.5.9.4.1} and from \sref[0]{0.5.3.11}; the second is equivalent to saying that if $\sh{H}^0_{X'/Z'}(f^*(\sh{F}))$ is of finite type, so is $\sh{H}^0_{X/Z}(\sh{F})$; this follows from \sref{IV.5.9.4.1} and from \sref{IV.2.5.2}.
\end{proof}

\begin{corollary}[5.9.6]
\label{IV.5.9.6}
Let $X$ be a locally Noetherian prescheme, $Z$ a part of $X$ stable by specialisation, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
For every $x\in X$, setting $X_x=\Spec(\sh{O}_x)$, $Z_x=Z\cap X_x$, we have a functorial isomorphism
\end{corollary}

\begin{equation}
\label{IV.5.9.6.1}
(\sh{H}^0_{X/Z}(\sh{F}))_x\xrightarrow{\sim}\sh{H}^0_{X_x/Z_x}(\widetilde{\sh{F}}_x) \tag{5.9.6.1}
\end{equation}

\begin{proof}
It suffices to apply \sref{IV.5.9.4} to the canonical morphism $X_x\to X$, which is flat, and to take into account \sref[I]{I.1.6.5}.
\end{proof}

\begin{env}[5.9.7]
\label{IV.5.9.7}
With the notations of \sref{IV.5.9.1}, for every $\alpha$ we have a canonical functorial homomorphism $\sh{F}\to(i_\alpha)^*(\sh{F}|U_\alpha)$ \sref[0]{0.4.4.3.2}, and these homomorphisms form an inductive system; by passage to the inductive limit, we thus deduce a canonical functorial homomorphism
\end{env}

\begin{equation}
\label{IV.5.9.7.1}
\rho_{X/Z}:\sh{F}\to\sh{H}^0_{X/Z}(\sh{F}) \tag{5.9.7.1}
\end{equation}

\begin{proposition}[5.9.8]
\label{IV.5.9.8}
Let $X$ be a locally Noetherian prescheme, $Z$ a part of $X$ stable by specialisation, $\sh{F}$ an $\sh{O}_X$-Module.
The following properties are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item The homomorphism $\rho_{X/Z}$ \sref{IV.5.9.7.1} is injective (resp.\ bijective).
  \item For every Noetherian open $V$ of $X$, the homomorphism
\[
  (\rho_{X/Z})_V:\Gamma(V,\sh{F})\to\varinjlim_\alpha\Gamma(V\cap U_\alpha,\sh{F})
\]
is injective (resp.\ bijective).
\end{enumerate}
\begin{enumerate}
  \item[\rm a${}'$)] For every closed part $T\subset Z$ of $X$, the canonical homomorphism \sref[0]{0.4.4.3.2}
\[
  \sh{F}\to i_*(\sh{F}|X-T)
\]
(where $i:X-T\to X$ is the canonical injection) is injective (resp.\ bijective).
  \item[\rm b${}'$)] For every closed part $T\subset Z$ of $X$ and every Noetherian open $V$ of $X$, the homomorphism of restriction
\[
  \Gamma(V,\sh{F})\to\Gamma(V\cap(X-T),\sh{F})
\]
is injective (resp.\ bijective).
\end{enumerate}
\end{proposition}

\begin{proof}
Taking into account \sref{IV.5.9.1.2}, the equivalence of \emph{a)} and \emph{b)} (resp.\ \emph{a${}'$)} and \emph{b${}'$)}) follows from

\oldpage[IV-2]{112}

the definition of the functor $\Gamma$ and from the fact that it is exact on the left.
As the homomorphism $(\rho_{X/Z})_V$ is the composite
\[
  \Gamma(V,\sh{F})\to\Gamma(V\cap(X-T),\sh{F})\to\varinjlim_\alpha\Gamma(V\cap U_\alpha,\sh{F})
\]
for every closed part $T\subset Z$, if $(\rho_{X/Z})_V$ is injective, then so is $\Gamma(V,\sh{F})\to\Gamma(V\cap(X-T),\sh{F})$; on the other hand, the fact that \emph{b${}'$)} implies \emph{b)} follows from the definition of an inductive limit.
It remains to show that if $\rho_{X/Z}$ is bijective, then so is $\Gamma(V,\sh{F})\to\Gamma(V\cap(X-T),\sh{F})$, and for this it suffices, by virtue of \sref{IV.5.9.8.1}, to see that if $U'\subset U$ are two opens contained in $V$ and containing $V\cap Z$, the homomorphism of restriction $\Gamma(U,\sh{F})\to\Gamma(U',\sh{F})$ is injective; but this follows from the fact that $\rho_{X/Z}$ is injective, by replacing $V$ by $U$ and $V\cap(X-T)$ by $U'$.
\end{proof}

\begin{definition}[5.9.9]
\label{IV.5.9.9}
Under the hypotheses of \sref{IV.5.9.8}, we say that $\sh{F}$ is \emph{$Z$-pure} (resp.\ \emph{$Z$-closed}) if $\rho_{X/Z}$ is injective (resp.\ bijective).
\end{definition}

If $X=\Spec(A)$ is affine, $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module, we say that $M$ is $Z$-pure (resp.\ $Z$-closed) when $\sh{F}$ is $Z$-pure (resp.\ $Z$-closed).

We say that $\sh{F}$ is \emph{$Z$-pure} (resp.\ \emph{$Z$-closed}) \emph{at a point} $x\in X$ if (with the notations of \sref{IV.5.9.6}) $\widetilde{\sh{F}}_x$ is $Z_x$-pure (resp.\ $Z_x$-closed); this amounts to the same, by virtue of \sref{IV.5.9.6}, to saying that the canonical homomorphism $\sh{F}_x\to(\sh{H}^0_{X/Z}(\sh{F}))_x$ is injective (resp.\ bijective).

One notes that for every $x\in X-Z$, $\sh{F}$ is $Z$-closed at the point $x$, by virtue of \sref{IV.5.9.8}.

\begin{corollary}[5.9.10]
\label{IV.5.9.10}
\begin{enumerate}[label=\emph{\roman*)}]
  \item Let $(V_\lambda)$ be an open covering of $X$.
For $\sh{F}$ to be $Z$-pure (resp.\ $Z$-closed), it is necessary and sufficient that, for every $\lambda$, $\sh{F}|V_\lambda$ be $(Z\cap V_\lambda)$-pure (resp.\ $(Z\cap V_\lambda)$-closed).
  \item Let $Z'$ be a part of $Z$ stable by specialisation.
If $\sh{F}$ is $Z$-pure (resp.\ $Z$-closed), it is $Z'$-pure (resp.\ $Z'$-closed).
\end{enumerate}
\end{corollary}

\begin{proof}
This follows immediately from \sref{IV.5.9.8}, \emph{b${}'$)}).
\end{proof}

\begin{proposition}[5.9.11]
\label{IV.5.9.11}
Under the hypotheses of \sref{IV.5.9.8}, the $\sh{O}_X$-Modules $\operatorname{Ker}(\rho_{X/Z})$ and $\operatorname{Coker}(\rho_{X/Z})$ have their support contained in $Z$, and the $\sh{O}_X$-Module $\sh{H}^0_{X/Z}(\sh{F})$ is $Z$-closed.
Furthermore, if $u:\sh{F}\to\sh{F}'$ is a homomorphism of $\sh{O}_X$-Modules such that $\sh{F}'$ is $Z$-closed, $u$ factorises in a unique way as $\sh{F}\xrightarrow{\rho_{X/Z}}\sh{H}^0_{X/Z}(\sh{F})\xrightarrow{v}\sh{F}'$.
If in addition the supports of $\operatorname{Ker}(u)$ and of $\operatorname{Coker}(u)$ are contained in $Z$, $v$ is an isomorphism.
\end{proposition}

\begin{proof}
The first assertion means that, for every $x\in X-Z$, we have
\[
  \operatorname{Ker}(\rho_{X/Z})_x=\operatorname{Coker}(\rho_{X/Z})_x=0,
\]
i.e.\ that $(\rho_{X/Z})_x$ is bijective, or again that $\sh{F}$ is $Z$-pure at $x$, as has been indicated above \sref{IV.5.9.9}.

To show that $\sh{H}^0_{X/Z}(\sh{F})$ is $Z$-closed, it suffices to see that for every Noetherian open $V$ of $X$, $\varinjlim_\beta\Gamma(V\cap U_\beta,\sh{H}^0_{X/Z}(\sh{F}))$ is equal to $\Gamma(V,\sh{H}^0_{X/Z}(\sh{F}))$; but by definition $\Gamma(V\cap U_\beta,\sh{H}^0_{X/Z}(\sh{F}))=\varinjlim_\alpha\Gamma(V\cap U_\alpha\cap U_\beta,\sh{F})$.
Now, the double family $(U_\alpha\cap U_\beta)$ is a filtered decreasing family, and our assertion follows from \sref{IV.5.9.1} and from the theorem on double inductive limits.

\oldpage[IV-2]{113}

Let us pass to the second part of the proposition.
The existence and the uniqueness of $v$ follow from the fact that for every $\alpha$, $(i_\alpha)_*(u)$ is the unique homomorphism making the diagram
\[
\xymatrix{
  \sh{F} \ar[r]^{u} & \sh{F}' \\
  (i_\alpha)_*(\sh{F}|U_\alpha) \ar[r]^{(i_\alpha)_*(u)} \ar[u] & (i_\alpha)_*(\sh{F}'|U_\alpha) \ar[u]
}
\]
commutative; from this there is a unique homomorphism $w$ making all the diagrams
\[
\xymatrix{
  (i_\alpha)_*(\sh{F}|U_\alpha) \ar[r]^{(i_\alpha)_*(u)} & (i_\alpha)_*(\sh{F}'|U_\alpha) \\
  \sh{H}^0_{X/Z}(\sh{F}) \ar[r]^{w} \ar[u] & \sh{H}^0_{X/Z}(\sh{F}') \ar[u]
}
\]
commutative, and finally that $\sh{F}'$ and $\sh{H}^0_{X/Z}(\sh{F}')$ identify canonically by hypothesis.

It remains to see that if the supports of $\operatorname{Ker}(u)$ and of $\operatorname{Coker}(u)$ are contained in $Z$, $v$ is an isomorphism.
It suffices to see that for every Noetherian open $V$ of $X$, the corresponding homomorphism $\Gamma(V,\sh{H}^0_{X/Z}(\sh{F}))\to\Gamma(V,\sh{F}')$ is an isomorphism.
Now, if $t\in\Gamma(V,\sh{H}^0_{X/Z}(\sh{F}))$ has image $0$ in $\Gamma(V,\sh{F}')$, let us note that for a given index $\alpha$, we have $t\in\Gamma(V\cap U_\alpha,\sh{F})$, and by the hypothesis on $u$, $t_y=0$ for every $y\in V\cap(X-Z)$; it follows that there is an open set containing $V\cap(X-Z)$ on which the restriction of $t$ is zero, hence by definition $t$ is the element $0$ of $\Gamma(V,\sh{H}^0_{X/Z}(\sh{F}))$.
Let us now prove that every section $s'\in\Gamma(V,\sh{F}')$ is the image of a section of $\sh{H}^0_{X/Z}(\sh{F})$ above $V$.
By hypothesis, for every $x\in V\cap(X-Z)$ there exists an open neighbourhood $W^{(x)}$ of $x$ in $X$, and a section $s^{(x)}$ of $\sh{F}$ above $W^{(x)}$, whose image by $u$ is $s'|W^{(x)}$; $s'|W^{(x)}$ is thus also the image by $v$ of the section $t^{(x)}$ of $\sh{H}^0_{X/Z}(\sh{F})$.
Furthermore, as one has seen that $v$ is injective, the restrictions of $t^{(x)}$ and $t^{(x')}$ to $W^{(x)}\cap W^{(x')}$ are identical for any two points $x$, $x'$ of $V\cap(X-Z)$; the $t^{(x)}$ are therefore restrictions of a single section $t$ of $\sh{H}^0_{X/Z}(\sh{F})$ above an open set $U$ of $(X-Z)\cap V$.
But as $\sh{H}^0_{X/Z}(\sh{F})$ is $Z$-closed, $t$ extends in a unique way to a section of $\sh{H}^0_{X/Z}(\sh{F})$ above $V$, whose image by $v$ has the same restriction as $s'$ to $U$, and hence coincides with $s'$ for the same reason.

We say that $\sh{H}^0_{X/Z}(\sh{F})$ is the \emph{$Z$-closure} of $\sh{F}$.
\end{proof}

\oldpage[IV-2]{114}

\begin{remarks}[5.9.12]
\label{IV.5.9.12}
\begin{enumerate}[label=\emph{\roman*)}]
  \item Let $\mathcal{C}(X)$ be the category of $\sh{O}_X$-Modules, and let $\mathcal{C}_Z(X)$ be the sub-category of $\mathcal{C}(X)$ formed of $\sh{O}_X$-Modules of support contained in $Z$; this sub-category is localisante in the sense of Gabriel, and the functor $\sh{H}^0_{X/Z}$ is none other than the localisation functor of Gabriel (cf.~\cite[27]); this would furnish another proof of \sref{IV.5.9.11}).
When $Z$ is closed, the functor $i^*:\mathcal{C}(X)\to\mathcal{C}(X-Z)$ (where $i:X-Z\to X$ is the canonical injection) defines an \emph{equivalence} of categories $\mathcal{C}(X)/\mathcal{C}_Z(X)\approx\mathcal{C}(X-Z)$.
  \item It follows from \sref{IV.5.9.11} that the condition $\sh{H}^0_{X/Z}(\sh{F})=0$ is \emph{equivalent} to $\operatorname{Supp}(\sh{F})\subset Z$.
Indeed, it implies that the kernel of $\rho_{X/Z}$ is equal to $\sh{F}$.
Conversely, if $\operatorname{Supp}(\sh{F})\subset Z$, it suffices to apply the second part of \sref{IV.5.9.11} to the unique homomorphism $u:\sh{F}\to 0$ to conclude that the corresponding homomorphism $v:\sh{H}^0_{X/Z}(\sh{F})\to 0$ is an isomorphism.
  \item The preceding developments hold for every annulated space locally Noetherian where every closed irreducible part admits exactly one generic point.
In particular they apply, on a locally Noetherian prescheme, to any \emph{sheaves of abelian groups} (considered as Modules over the simple sheaf associated to the constant presheaf $\mathbf{Z}$).
One has moreover for every $x\in X$ the canonical isomorphism \sref{IV.5.9.6.1}, in which $\widetilde{\sh{F}}_x$ denotes the sheaf \emph{induced} on the subspace $X_x$ of $X$ by the sheaf $\sh{F}$; the direct proof follows immediately from the definition \sref{IV.5.9.1.2} and from the theorem on double inductive limits.
\end{enumerate}
\end{remarks}

\subsection{Property \texorpdfstring{$(\mathrm{S}_2)$}{(S2)} and \texorpdfstring{$Z$}{Z}-closure}
\label{subsection:IV.5.10}

\begin{env}[5.10.1]
\label{IV.5.10.1}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module; for every part $T$ of $X$, we shall set
\end{env}

\begin{equation}
\label{IV.5.10.1.1}
\operatorname{prof}_T(\sh{F})=\inf_{z\in T}\operatorname{prof}(\sh{F}_z) \tag{5.10.1.1}
\end{equation}

\begin{proposition}[5.10.2]
\label{IV.5.10.2}
Let $X$ be a locally Noetherian prescheme, $Z$ a part of $X$ stable by specialisation, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\sh{F}$ is $Z$-pure.
  \item $\operatorname{Ass}(\sh{F})$ does not meet $Z$.
\end{enumerate}
If in addition $\sh{F}$ is coherent, these conditions are also equivalent to the following:
\begin{enumerate}
  \item[\rm c)] $\operatorname{prof}_Z(\sh{F})\geqslant 1$.
\end{enumerate}
\end{proposition}

\begin{proof}
To say that $\sh{F}$ is $Z$-pure means that for every Noetherian open $V$ of $X$, and every open $U\supset X-Z$, the homomorphism of restriction $\Gamma(V,\sh{F})\to\Gamma(V\cap U,\sh{F})$ is injective \sref{IV.5.9.8}; but by \sref{IV.3.1.8}, this is equivalent to $V\cap\operatorname{Ass}(\sh{F})\subset U$, whence the equivalence of \emph{a)} and \emph{b)}.
Furthermore, to say that $x\in\operatorname{Ass}(\sh{F})$ means that no element of $\mathfrak{m}_x$ is $\sh{F}_x$-regular \sref{IV.3.1.2}, hence, when $\sh{F}$ is coherent, this is written $\operatorname{prof}(\sh{F}_x)=0$; one deduces immediately from this the equivalence of \emph{b)} and \emph{c)}.
\end{proof}

If $\sh{F}$ is $Z$-pure, so is $\sh{F}'$; conversely, if $\sh{F}'$ and $\sh{F}''$ are $Z$-pure, so is $\sh{F}$.
\end{corollary}

\begin{proof}
This follows from the form \sref{IV.5.10.2}, \emph{b)}) of the condition for a quasi-coherent $\sh{O}_X$-Module to be $Z$-pure, and from \sref{IV.3.1.7}.
\end{proof}

\begin{theorem}[5.10.5]
\label{IV.5.10.5}
Let $X$ be a locally Noetherian prescheme, $Z$ a part of $X$ stable by specialisation, $\sh{F}$ a coherent $\sh{O}_X$-Module.
For $\sh{F}$ to be $Z$-closed, it is necessary and sufficient that $\operatorname{prof}_Z(\sh{F})\geqslant 2$.
\end{theorem}

\begin{proof}
By virtue of \sref{IV.5.10.2}, we may restrict ourselves to the case where $\sh{F}$ is $Z$-pure and $\operatorname{prof}_Z(\sh{F})\geqslant 1$; $\operatorname{prof}_{Z_\alpha}(\sh{F})\geqslant 2$; and similarly, it follows from \sref{IV.5.9.8} that to say that $\sh{F}$ is $Z$-closed is equivalent to saying that $\sh{F}$ is $Z_\alpha$-closed for every $\alpha$.
We may thus already restrict ourselves to the case where $Z$ is \emph{closed}.
The question being local, it suffices, for every $x\in Z$, to prove the theorem for $\sh{F}|U$, where $U$ is an affine open neighbourhood of $x$, and we can therefore restrict ourselves to the case where $X=U$ is affine.
We know then that $\operatorname{Ass}(\sh{F})$ is finite \sref{IV.3.1.6}, and as $\operatorname{Ass}(\sh{F})\subset X-Z$, there is a section $f$ of $\sh{O}_X$ above $X$ such that $\operatorname{Ass}(\sh{F})\subset X_f\subset X-Z$ \sref[II]{II.4.5.4}; we deduce from this that $f$ is $\sh{F}$-regular \sref{IV.3.1.9} and that for every $y\in Z$, we have $f_y\in\mathfrak{m}_y$, hence $\operatorname{prof}(\sh{F}_y)=1+\operatorname{prof}(\sh{F}_y/f_y\sh{F}_y)$ \sref[0]{0.16.4.6}.
The condition $\operatorname{prof}_Z(\sh{F})\geqslant 2$ is thus equivalent to $\operatorname{prof}_Z(\sh{F}/f\sh{F})\geqslant 1$, or again \sref{IV.5.10.2} to the fact that $\sh{F}/f\sh{F}$ is $Z$-pure, and it suffices to see that this last property is equivalent to the fact that $\sh{F}$ is $Z$-closed.

Consider the exact sequence $0\to\sh{F}\xrightarrow{f}\sh{F}\to\sh{F}/f\sh{F}\to 0$ (the homothety of ratio $f:\sh{F}\xrightarrow{f}\sh{F}$ being injective by hypothesis); if we set $W=X-Z$, we have the commutative diagram
\[
\xymatrix{
  0 \ar[r] & \Gamma(X,\sh{F}) \ar[r]^{f} \ar[d] & \Gamma(X,\sh{F}) \ar[r] \ar[d] & \Gamma(X,\sh{F}/f\sh{F}) \ar[r] \ar[d] & 0 \\
  0 \ar[r] & \Gamma(W,\sh{F}) \ar[r]^{f} & \Gamma(W,\sh{F}) \ar[r] & \Gamma(W,\sh{F}/f\sh{F}) &
}
\]
where the two rows are exact ($X$ being affine).
If the homomorphism of restriction $\Gamma(X,\sh{F})\to\Gamma(W,\sh{F})$ is bijective, we deduce from this diagram that
\[
  \Gamma(X,\sh{F}/f\sh{F})\to\Gamma(W,\sh{F}/f\sh{F})
\]
is injective, and this shows \sref{IV.5.9.8} that $\sh{F}/f\sh{F}$ is $Z$-pure.
Conversely, suppose that $\sh{F}/f\sh{F}$ is $Z$-pure, and let $s$ be a section of $\sh{F}$ above $W$; as $X_f\subset W$, there is an integer $n>0$ such that $f^n(s|X_f)$ extends to a section $t$ of $\sh{F}$ above $X$ \sref[I]{I.1.4.1}; besides, the restrictions of $t$ and of $f^ns$ to $X_f$ being the same, it follows that the restriction of $t$ to $W$ is equal to $f^ns$ by virtue of the relation $\operatorname{Ass}(\sh{F})\subset X_f$ \sref{IV.5.10.2}; as $f$ is $\sh{F}$-regular, it suffices to see that $t$ is of the form $f^nt'$, where $t'\in\Gamma(X,\sh{F})$

\oldpage[IV-2]{116}

to show that the homomorphism $\Gamma(X,\sh{F})\to\Gamma(W,\sh{F})$ is surjective, hence bijective.
Now, to say that $t=f^nt'$ means that the image of $t$ in $\Gamma(X,\sh{F}/f^n\sh{F})$ is zero.
But as $f^k\sh{F}/f^{k+1}\sh{F}$ is isomorphic to $\sh{F}/f\sh{F}$, hence $Z$-pure by hypothesis, we deduce from \sref{IV.5.10.3} by induction on $n$ that $\sh{F}/f^n\sh{F}$ is $Z$-pure.
But by definition the image of $t|W=f^ns$ in $\Gamma(W,\sh{F}/f^n\sh{F})$ is zero, whence the conclusion.
\end{proof}

For $\sh{F}$ to be $Z$-closed at a point $x$, it is necessary and sufficient that $\operatorname{prof}_{Z_x}(\widetilde{\sh{F}}_x)\geqslant 2$.
\end{corollary}

\begin{proof}
This follows from \sref{IV.5.9.6} and \sref{IV.5.10.5}.

\begin{theorem}[5.10.7]
\label{IV.5.10.7}
\emph{(Hartshorne).}
Let $X$ be a locally Noetherian prescheme, $Y$ a closed part of $X$.
Suppose that for every $y\in Y$, we have $\operatorname{prof}(\sh{O}_{X,y})\geqslant 2$; then for every connected component $C$ of $X$, $C-(C\cap Y)$ is connected.
\end{theorem}

\begin{proof}
We may restrict ourselves to the case where $X$ is connected; it follows then from \sref{IV.5.10.5} that the canonical homomorphism $\sh{O}_X\to i_*(\sh{O}_X|X-Y)$ (where $i:X-Y\to X$ is the canonical injection) is bijective.
Consequently, the homomorphism of restriction $\Gamma(X,\sh{O}_X)\to\Gamma(X-Y,\sh{O}_X)$ is also bijective.
It suffices now to apply the lemma \sref[III]{III.7.8.6.1}.
\end{proof}

\begin{corollary}[5.10.8]
\label{IV.5.10.8}
Let $X$ be a locally Noetherian prescheme, $d$ an integer such that for every $x\in X$, the relation $\dim(\sh{O}_x)\geqslant d$ implies $\operatorname{prof}(\sh{O}_x)\geqslant 2$.
Suppose $X$ connected; then, if $X'$, $X''$ are two distinct irreducible components of $X$, there exists a suite $(X_i)_{0\leqslant i\leqslant n}$ of irreducible components of $X$ such that $X_0=X'$, $X_n=X''$, and that, for $1\leqslant i\leqslant n$, we have $\codim(X_{i-1}\cap X_i,X)\leqslant d-1$ (we then say that $X$ is \emph{connected in codimension $\leqslant d-1$}).

If $Y$ is a closed part of $X$ such that $\codim(Y,X)\geqslant d$, we have $\dim(\sh{O}_{X,y})\geqslant d$ for every $y\in Y$ \sref{IV.5.1.3}, hence $\operatorname{prof}(\sh{O}_{X,y})\geqslant 2$ for every $y\in Y$, and it follows from \sref{IV.5.10.7} that $X-Y$ is connected.
On the other hand, for $\codim(Y,X)\geqslant d$, it is necessary and sufficient that for every $y\in Y$, there exists an open neighbourhood $V$ of $y$ in $X$ such that $\codim(Y\cap V,V)\geqslant d$ \sref[0]{0.14.2.3}.
Let us finally note that if $\mathfrak{Y}$ denotes the set of closed parts $Y$ of $X$ of codimension $\geqslant d$, the union of two sets of $\mathfrak{Y}$ belongs to $\mathfrak{Y}$ \sref[0]{0.14.2.5}, and every closed set contained in a set of $\mathfrak{Y}$ also belongs to $\mathfrak{Y}$; properties which we may also express by saying that $\mathfrak{Y}$ is an \emph{antifilter} of closed parts of $X$.
The corollary is then the following topological lemma:
\end{corollary}

\begin{lemma}[5.10.8.1]
\label{IV.5.10.8.1}
Let $X$ be a connected locally Noetherian topological space, $\mathfrak{Y}$ an antifilter of closed parts of $X$.
Suppose that if $Y$ is a closed part of $X$ such that for every $y\in Y$, there is a neighbourhood open $V$ of $y$ in $X$ and a $Y_y\in\mathfrak{Y}$ such that $V\cap Y=V\cap Y_y$, then $Y\in\mathfrak{Y}$.
The following conditions are then equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item For every $Y\in\mathfrak{Y}$, $X-Y$ is connected.
  \item If $X'$ and $X''$ are two distinct irreducible components of $X$, there exists a suite $(X_i)_{0\leqslant i\leqslant n}$ of irreducible components of $X$ such that $X_0=X'$, $X_n=X''$ and that, for $1\leqslant i\leqslant n$, we have $X_{i-1}\cap X_i\notin\mathfrak{Y}$.
\end{enumerate}
\end{lemma}

\begin{proof}
Suppose \emph{b)} satisfied and let us prove that $U=X-Y$ is connected for every $Y\in\mathfrak{Y}$.
If $U'$ and $U''$ are two distinct irreducible components of $U$, there exist two irreducible components $X'$, $X''$ of $X$ such that $X'\cap U=U'$, $X''\cap U=U''$ \sref[0]{0.2.1.6};

\oldpage[IV-2]{117}

let us form for these two components a suite $(X_i)$ having the property stated in \emph{b)} and set $U_i=X_i\cap U$ for $1\leqslant i\leqslant n$; then $U_i$ is an irreducible component of $U$ \sref[0]{0.2.1.6} and furthermore $U_i\cap U_{i-1}\neq\varnothing$ for $1\leqslant i\leqslant n$, otherwise one would have $X_i\cap X_{i-1}\subset Y$, hence $X_i\cap X_{i-1}\in\mathfrak{Y}$, contrary to the hypothesis.
This shows that $U$ is connected.

Let us now show that \emph{a)} implies \emph{b)}.
Denote by $Y$ the union of the family $(X_\alpha\cap X_\beta)$, where $(X_\alpha,X_\beta)$ runs through the set of pairs of \emph{distinct} irreducible components of $X$ such that $X_\alpha\cap X_\beta\in\mathfrak{Y}$.
For every point $y\in Y$, in every open neighbourhood $V$ of $y$ in $X$ there is only a finite number of irreducible components; this shows on the one hand that $Y$ is closed, and on the other part that $V\cap Y$ is an intersection of $V$ and of a set of $\mathfrak{Y}$; by the hypothesis made on $\mathfrak{Y}$, we have $Y\in\mathfrak{Y}$, hence $U=X-Y$ is connected.
Let now $X'$, $X''$ be two distinct irreducible components of $X$, $U'$, $U''$ their respective traces on $U$; these are irreducible components of $U$ \sref[0]{0.2.1.6}.
Take the union $W$ of the irreducible components of $U$ such that there exists a suite $(U_i)_{0\leqslant i\leqslant n}$ of irreducible components of $U$ for which $U_0=U'$, $U_{i-1}\cap U_i\neq\varnothing$ for $1\leqslant i\leqslant n$ and $U_n=W$; this is an \emph{open and closed} set in $U$, since $U$ is locally Noetherian and its irreducible components form a locally finite family of closed sets.
So there is such a suite $(U_i)$ for which $U_n=U''$; let $X_i$ $(0\leqslant i\leqslant n)$ be the irreducible component of $X$ such that $X_i\cap U=U_i$ \sref[0]{0.2.1.6}; as $U_{i-1}\neq U_i$, on has $X_{i-1}\neq X_i$ for $1\leqslant i\leqslant n$; if one had $X_{i-1}\cap X_i\in\mathfrak{Y}$ for some $i$ such that $1\leqslant i\leqslant n$, one would deduce $X_{i-1}\cap X_i\subset Y$, contrary to the definition of the $X_i$.
This achieves the proof of the lemma.
\end{proof}

One notes that the hypothesis made on $X$ in \sref{IV.5.10.8} is satisfied when $X$ is a Cohen-Macaulay \emph{prescheme} and $d\geqslant 2$.

\begin{corollary}[5.10.9]
\label{IV.5.10.9}
If a Noetherian local ring $A$ satisfies $(\mathrm{S}_2)$ and is catenary, it is equidimensional.
\end{corollary}

\begin{proof}
The hypothesis of \sref{IV.5.10.8} is then satisfied by $X=\Spec(A)$, with $d=2$.
To show that all the irreducible components of $X$ have the same dimension, it suffices then, by virtue of \sref{IV.5.10.8}, to show that any two such components $X'$, $X''$ have the same dimension, when in addition one supposes that $\codim(X'\cap X'',X)=1$.
There is then an irreducible component $Z$ of $X'\cap X''$ such that $\codim(Z,X)=1$, hence $\codim(Z,X')=1$, hence $\codim(Z,X')\geqslant 1$; of the same, $\codim(Z,X'')\geqslant 1$; as $X$ is catenary, this implies $\dim(X')=\dim(X'')$.
\end{proof}

\begin{proposition}[5.10.10]
\label{IV.5.10.10}
Let $X$ be a locally Noetherian prescheme, $Z$ a part of $X$ stable by specialisation, $\sh{F}$ a coherent $\sh{O}_X$-Module, and suppose that the $\sh{O}_X$-Module $\sh{H}^0_{X/Z}(\sh{F})$ is coherent.
Then:
\begin{enumerate}[label=\emph{\roman*)}]
  \item We have $\operatorname{prof}_Z(\sh{H}^0_{X/Z}(\sh{F}))\geqslant 2$.
  \item For every point $x\in\operatorname{Ass}(\sh{F})\cap(X-Z)$, we have $\codim(Z\cap\overline{\{x\}},\overline{\{x\}})\geqslant 2$.
  \item The set $U$ of $x\in X$ such that $\operatorname{prof}_{Z_x}(\widetilde{\sh{O}}_x)\geqslant 2$ and $\operatorname{prof}_{Z_x}(\widetilde{\sh{F}}_x)\geqslant 2$ (notations of \sref{IV.5.9.6}) is open in $X$; we have $X-U\subset Z$, and $U$ is the largest open of $X$ such that $\sh{F}|U$ is $(Z\cap U)$-closed.

\oldpage[IV-2]{118}

\end{enumerate}
\end{proposition}

\begin{proof}
Let us set, for brevity, $\sh{F}'=\sh{H}^0_{X/Z}(\sh{F})$.
We know \sref{IV.5.9.11} that $\sh{F}'$ is $Z$-closed, and the restrictions of $\sh{F}$ and $\sh{F}'$ to $X-Z$ are canonically isomorphic \sref{IV.5.9.11}, hence $x\in\operatorname{Ass}(\sh{F}')$.
Consider a point $y\in Z\cap\overline{\{x\}}$; the prime ideal $\mathfrak{p}$ of $\sh{O}_y$ corresponding to $x$ is associated to the $\sh{O}_y$-module $\sh{F}'_y$, hence on the one hand, by virtue of~(i) and of \sref[0]{0.16.4.6.2}, $2\leqslant\operatorname{prof}(\sh{F}'_y)\leqslant\dim(\sh{O}_y/\mathfrak{p})=\codim(\overline{\{y\}},\overline{\{x\}})$, whence~(ii).
Finally, to prove~(iii), let us note that $U$ is the set of $x\in X$ such that $\widetilde{\sh{F}}_x$ is $Z_x$-closed \sref{IV.5.10.5}, or again, by virtue of \sref{IV.5.9.6}, the set of points where the canonical homomorphism $\sh{F}_x\to\sh{F}'_x$ is bijective; it is therefore the complement of the union of the supports of $\operatorname{Ker}(\rho_{X/Z})$ and $\operatorname{Coker}(\rho_{X/Z})$, and these last are coherent $\sh{O}_X$-Modules by virtue of the hypothesis \sref[0]{0.5.3.4}, hence have a closed support \sref[0]{0.5.2.2}; this shows that $U$ is open, and $U$ is evidently the largest open such that $\sh{F}|U$ is $(Z\cap U)$-closed; finally the inclusion $X-U\subset Z$ follows from \sref{IV.5.9.11}.
\end{proof}

We shall see further on \sref{IV.5.11.1} that in the most important cases assertion~(ii) implies conversely that $\sh{H}^0_{X/Z}(\sh{F})$ is coherent.

  X'=\Spec(\sh{H}^0_{X/Z^{(2)}}(\sh{O}_X))
\]
its $Z^{(2)}$-closure, $g:X'\to X$ the structural morphism.
Suppose that $X$ has no associated prime cycle of codimension~$1$.
\begin{enumerate}[label=\emph{\roman*)}]
  \item For a point $x\in X$ to satisfy $(\mathrm{S}_2)$, it is necessary and sufficient that the morphism $X'_x\to X_x$ deduced from $g$ (notations of \sref{IV.5.10.12}) be an isomorphism.
This condition is always satisfied if $\codim(\overline{\{x\}},X)\leqslant 1$.
  \item Suppose in addition that $g$ is a finite morphism (see in \sref{IV.5.11.2} for sufficient conditions for this to be so).
Then the set $U$ of points where $X$ satisfies $(\mathrm{S}_2)$ is open and $\codim(X-U,X)\geqslant 2$; furthermore $U$ is the largest open of $X$ such that the restriction $g^{-1}(U)\to U$ of $g$ is an isomorphism.
  \item Under the same hypotheses as in~(ii), $X'$ satisfies $(\mathrm{S}_2)$, and for every $x'\in X'$ such that $\codim(\overline{\{x'\}},X')\leqslant 1$, the point $x=g(x')$ is such that $\codim(\overline{\{x\}},X)=\codim(\overline{\{x'\}},X')$.
  \item Under the hypotheses of~(ii), let $\sh{F}$ be an $\sh{O}_X$-Module coherent of support $X$, such that $\sh{H}^0_{X/Z^{(2)}}(\sh{F})$ is coherent; then the $\sh{O}_{X'}$-Module $\sh{F}'$ such that $g_*(\sh{F}')=\sh{H}^0_{X/Z^{(2)}}(\sh{F})$ is coherent and satisfies $(\mathrm{S}_2)$, and its support is a union of irreducible components of $X'$.
\end{enumerate}
\end{proposition}

\begin{proof}
Assertions~(i) and~(ii) are in substance already proved, having been already demonstrated as a particular case of \sref{IV.5.10.12}, (i) and \sref{IV.5.10.14}, and~(ii) is a particular case of \sref{IV.5.10.15}, (ii).

Let us prove~(iii); set $x=g(x')$; as $g$ is finite, it is the same for the morphism $X'_x\to X_x$, hence $\dim(\sh{O}_{X',x'})\leqslant\dim(X'_x)\leqslant\dim(X_x)=\dim(\sh{O}_{X,x})$ by virtue of \sref{IV.5.4.1}.
Suppose first that $\dim(\sh{O}_{X,x})\leqslant 1$ and let us show that $\dim(\sh{O}_{X',x'})\leqslant 1$; then by \sref{IV.5.10.12}, (ii), applied to $\sh{O}_X$, we would have $\operatorname{prof}(\sh{O}_{X',x'})\geqslant 2$, which is absurd, i.e.\ $x\in Z^{(2)}$; we deduce that $\operatorname{prof}(\sh{O}_{X',x'})\geqslant 2$ by \sref{IV.5.10.12}, (ii).
Suppose now that $\dim(\sh{O}_{X,x})\geqslant 2$, and let us show similarly that $\sh{O}_{X',x'}$ is isomorphic to $\sh{O}_{X,x}$ \sref{IV.5.10.12}, (i), whence $\dim(\sh{O}_{X',x'})=\dim(\sh{O}_{X,x})$.
Furthermore, as $X$ has no associated prime cycles of codimension $1$, the hypothesis $\dim(\sh{O}_x)=1$ implies $x\notin\operatorname{Ass}(\sh{O}_X)$, hence $\operatorname{prof}(\sh{O}_{X,x})=1$; but by hypothesis the relation $\dim(\sh{O}_x)\leqslant 1$ implies that $\sh{F}_x=\sh{F}'_x$; hence every irreducible component of $\operatorname{Supp}(\sh{F}')$ is a component of $X'$, since $\sh{F}'$ is coherent, hence $\operatorname{Supp}(\sh{F}')$ is closed.
This establishes the assertions of~(iii).

\oldpage[IV-2]{121}

To prove~(iv), it suffices to replace $\sh{O}_X$ by $\sh{F}$ in the preceding reasoning, which establishes that $\sh{F}'$ satisfies $(\mathrm{S}_2)$ and that if $\dim(\sh{F}'_{x'})=1$, $\sh{F}_x$ and $\sh{F}'_{x'}$ are di-isomorphic; in particular if $\dim(\sh{F}'_{x'})=0$, hence $\dim(\sh{F}_x)=0$, hence $\dim(\sh{O}_{X,x})=0$ and $\dim(\sh{O}_{X',x'})=0$ since $\sh{F}$ has support $X$; every irreducible component of $\operatorname{Supp}(\sh{F}')$ is therefore a component of $X'$, since $\sh{F}'$ is coherent, hence $\operatorname{Supp}(\sh{F}')$ is closed.
\end{proof}

\begin{proposition}[5.10.17]
\label{IV.5.10.17}
Let $A$ be a Noetherian integral ring, and denote by $A^{(1)}$ the intersection of the local rings $A_\mathfrak{p}$, where $\mathfrak{p}$ runs through the set of prime ideals of $A$ of height~$1$.
Suppose that $A^{(1)}$ is a finite $A$-algebra.
Then:
\begin{enumerate}[label=\emph{\roman*)}]
  \item The ring $A^{(1)}$ satisfies the condition $(\mathrm{S}_2)$.
  \item The set $U$ of $\mathfrak{p}\in\Spec(A)$ such that the canonical homomorphism $A_\mathfrak{p}\to(A^{(1)})_\mathfrak{p}$ is bijective is equal to the set of $\mathfrak{p}$ such that $A_\mathfrak{p}$ satisfies $(\mathrm{S}_2)$; $U$ is open in $X=\Spec(A)$ and we have $\codim(X-U,X)\geqslant 2$.
  \item For every multiplicative subset $S$ of $A$, $(S^{-1}A)^{(1)}$ is a finite $(S^{-1}A)$-algebra.
  \item Let $B$ be a finite integral $A$-algebra containing $A$.
Then $B^{(1)}$ is a finite $B$-algebra.
Furthermore, for every prime ideal $\mathfrak{q}$ of $B$ of height~$1$, the prime ideal $\mathfrak{q}\cap A$ is of height~$1$ in $A$.
\end{enumerate}
\end{proposition}

\begin{proof}
If one takes into account formula \sref{IV.5.9.3.1}, one sees that $X'=\Spec(A^{(1)})$ is the $Z^{(2)}$-closure of $X=\Spec(A)$; as $A$ has no associated immersed prime ideals, properties~(i) and~(ii) are particular cases of \sref{IV.5.10.16}, (i), (ii) and~(iii).
To prove~(iii), it suffices to remark that $(S^{-1}A)^{(1)}=S^{-1}A^{(1)}$, which is a particular case of \sref{IV.5.9.4}; as this is a flat $A$-module, the prime ideals associated to $S^{-1}A$ do not meet $S$, and we have $\operatorname{ht}(S^{-1}\mathfrak{p})=\operatorname{ht}(\mathfrak{p})$ for $\mathfrak{p}\in\Spec(A)$ such that $S^{-1}\mathfrak{p}$ is defined, whence~(iii).

To prove~(iv), set $Y=\Spec(B)$, and let $f:Y\to X$ be the structural morphism; as $f$ is finite, it follows from \sref{IV.5.4.1} that for every $y\in Y$, $\dim(\sh{O}_{X,f(y)})\leqslant\dim(\sh{O}_{X,f(y)})$; hence, if $T=f^{-1}(Z^{(2)}(X))$, we have $T\supset Z^{(2)}(Y)$.
Let us show that $\sh{G}=\sh{H}^0_{X/Z^{(2)}(X)}(f_*(\sh{O}_Y))$ is coherent; indeed, $B$ being considered as an $A$-module; but $B$ is a finite integral $A$-algebra, its field of fractions is finite over the field of fractions of $A$, hence $B$ is contained in a free $A$-module of finite type, and hence by \sref{IV.5.9.2}, (i) $((\sh{H}^0_{X/Z^{(2)}(X)}(\sh{O}_X))^n$ for a suitable $n$; this last being by hypothesis quasi-coherent of $(\sh{H}^0_{X/Z^{(2)}(X)}(\sh{O}_X))^n$, the result follows.
Now, it follows from the definition \sref{IV.5.9.1.2} that $\sh{G}$ is isomorphic to $f_*(\sh{H}^0_{Y/T}(\sh{O}_Y))$; this proves \emph{a fortiori} that $\sh{H}^0_{Y/T}(\sh{O}_Y)$ is a coherent $\sh{O}_Y$-Module.
It follows then from \sref{IV.5.10.10}, (ii), applied to $\sh{O}_Y$ and at the generic point of $Y$, that $\codim(T,Y)\geqslant 2$, that is to say $T\subset Z^{(2)}(Y)$, and finally $T=Z^{(2)}(Y)$.
This proves the two assertions of~(iv).
\end{proof}

\subsection{Coherence criteria for the Modules \texorpdfstring{$\sh{H}^0_{X/Z}(\sh{F})$}{H\^{}0\_\{X/Z\}(F)}}
\label{subsection:IV.5.11}

\begin{proposition}[5.11.1]
\label{IV.5.11.1}
Let $X$ be a locally Noetherian prescheme, $Z$ a part of $X$ stable by specialisation, $\sh{F}$ a coherent $\sh{O}_X$-Module.
Denote by $(x_\alpha)$ the family of points of $\operatorname{Ass}(\sh{F})\cap(X-Z)$ and, for every $\alpha$, let $Y_\alpha$ be the reduced closed sub-prescheme of $X$ having $\overline{\{x_\alpha\}}$ for underlying space, $Z_\alpha=Z\cap\overline{\{x_\alpha\}}$.
Then the following two conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\sh{H}^0_{X/Z}(\sh{F})$ is a coherent $\sh{O}_X$-Module.
  \item For every $\alpha$, $\sh{H}^0_{Y_\alpha/Z_\alpha}(\sh{O}_{Y_\alpha})$ is a coherent $\sh{O}_{Y_\alpha}$-Module.
\end{enumerate}
These two conditions imply the following:
\begin{enumerate}
  \item[\rm c)] For every $\alpha$, we have $\codim(Z_\alpha,Y_\alpha)\geqslant 2$.
\end{enumerate}
Moreover, the three conditions \emph{a)}, \emph{b)}, \emph{c)} are equivalent when in addition one of the following properties is satisfied:
\begin{enumerate}
  \item[\rm(i)] Every point of $X$ admits a neighbourhood isomorphic to a sub-scheme of a regular scheme (in which case one also says that $X$ is \emph{locally immersible in a regular scheme}).
  \item[\rm(ii)] For every $\alpha$, $Y_\alpha$ is universally catenary \sref[0]{0.5.6.2} and its normalised $(\sref[II]{II.6.3.8})$ $Y_\alpha'$ is finite over $Y_\alpha$.
\end{enumerate}
All the properties under consideration are local on $X$, so one can reduce to the case where $X=\Spec(A)$ is affine, $A$ is a Noetherian ring, and $\sh{F}=\widetilde{M}$, where $M$ is a finite type $A$-module.
Then, for every $\alpha$, if $h_\alpha$ is the canonical injection $Y_\alpha\to X$, $(h_\alpha)_*(\sh{O}_{Y_\alpha})=\sh{G}_\alpha$ is the $\sh{O}_X$-Module corresponding to the quotient $A$-module $A/\mathfrak{i}_{x_\alpha}$, and, by definition of $\operatorname{Ass}(\sh{F})$, this $A$-module is isomorphic to a sub-$A$-module of $M$.
As $\sh{H}^0_{X/Z}(\sh{G}_\alpha)$ is a sub-$\sh{O}_X$-Module quasi-coherent of $\sh{H}^0_{X/Z}(\sh{F})$ \sref{IV.5.9.2}, the hypothesis that $\sh{H}^0_{X/Z}(\sh{F})$ is coherent implies that it is also coherent.
On the other hand, it follows from the definition \sref{IV.5.9.1.2} that $\sh{H}^0_{X/Z}(\sh{G}_\alpha)$ is isomorphic to $\sh{H}^0_{Y_\alpha/Z_\alpha}(\sh{O}_{Y_\alpha})$; this proves that \emph{a)} implies \emph{b)}.
To see that \emph{b)} implies \emph{a)}, it suffices to show that there is a finite filtration $(\sh{F}_i)_{0\leqslant i\leqslant n}$ of $\sh{F}$ formed of coherent $\sh{O}_X$-Modules, with $\sh{F}_0=\sh{F}$, $\sh{F}_n=0$, and such that $\sh{H}^0_{X/Z}(\sh{F}_i/\sh{F}_{i+1})$ is coherent for every $i$: this follows, by induction on $i$, from the exact sequences
\[
  0\to\sh{F}_{i+1}\to\sh{F}_i\to\sh{F}_i/\sh{F}_{i+1},
\]
from the fact that $\sh{H}^0_{X/Z}$ is a left exact functor \sref{IV.5.9.2}, and finally from \sref[0]{0.5.3.3} and $(\sref[I]{I.6.1.1})$.
By virtue of \sref[I]{I.3.2.8}, it thus suffices to prove that $\sh{H}^0_{X/Z}(\sh{F})$ is coherent when $\sh{F}$ is \emph{irredundant}; in other words, when $\operatorname{Ass}(M)=\{\mathfrak{p}\}$ is reduced to a single element.
Let us now state the
\begin{lemma}[5.11.1.1]
\label{IV.5.11.1.1}
Let $A$ be a Noetherian ring, $M$ a finite type $A$-module, such that $\operatorname{Ass}(M)=\{\mathfrak{p}\}$.
There exists a finite filtration $(M_h)_{0\leqslant h\leqslant m}$ of $M$ such that $M_0=M$, $M_m=0$ and that $M_h/M_{h+1}$ is isomorphic to a sub-module of $A/\mathfrak{p}$.
\end{lemma}

\oldpage[IV-2]{122}

Let us first note that the canonical homomorphism $M\to M_\mathfrak{p}=N$ is injective (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \S~1, no~2, prop.~6).
Set $B=A_\mathfrak{p}$, $\mathfrak{m}=\mathfrak{p}A_\mathfrak{p}$, maximal ideal of $B$; on has $\operatorname{Ass}(N)=\{\mathfrak{m}\}$ (\emph{loc.\ cit.}, prop.~5), and as $N$ is a finite type $B$-module,
\oldpage[IV-2]{123}
there exists an integer $r$ such that $\mathfrak{m}^rN=0$; if one sets $N_j'=\mathfrak{m}^jN$ for $0\leqslant j\leqslant r$, $N_j'/N_{j+1}'$ is a $(B/\mathfrak{m})$-module of finite type, hence a direct sum of a finite number of sub-modules isomorphic to $B/\mathfrak{m}$, since $B/\mathfrak{m}$ is a field; in other words, there is a finite filtration $(N_h)_{0\leqslant h\leqslant m}$ of $N$ such that $N_m=0$ and that $N_h/N_{h+1}$ is isomorphic to $B/\mathfrak{m}$, i.e.\ to the field of fractions of $A/\mathfrak{p}$; the filtration of the $M_h=M\cap N_h$ satisfies the requirements, for $M_h/M_{h+1}$ is isomorphic to a sub-$(A/\mathfrak{p})$-module of finite type of $N_h/N_{h+1}=B/\mathfrak{m}$; but it is known that such a sub-module is isomorphic to a sub-module of $A/\mathfrak{p}$.

The existence of the filtration $(M_h)$ shows, by the same reasoning as above, that to prove (averaging \emph{b)}) that $\sh{H}^0_{X/Z}(\sh{F})$ is coherent, one can reduce to showing that $\sh{H}^0_{X/Z}(\sh{P})$ is coherent, where $\sh{P}=(A/\mathfrak{p})^\sim$, $\mathfrak{p}$ being an ideal associated to $M$.
But if $\mathfrak{p}=\mathfrak{i}_y$ with $y\in Z$, the support of $\sh{P}$ is a closed set contained in $Z$, since $Z$ is stable by specialisation; the definition \sref{IV.5.9.1.2} shows then that $\sh{H}^0_{X/Z}(\sh{P})=0$.
If on the contrary $\mathfrak{p}=\mathfrak{i}_{x_\alpha}$ for some $\alpha$, on has $\sh{P}=(h_\alpha)_*(\sh{O}_{Y_\alpha})$ by definition, and one has seen above that $\sh{H}^0_{X/Z}(\sh{P})$ is isomorphic to $\sh{H}^0_{Y_\alpha/Z_\alpha}(\sh{O}_{Y_\alpha})$, hence is coherent by virtue of the hypothesis \emph{b)}.

We have already seen \sref{IV.5.10.10}[, (ii)] that \emph{a)} implies \emph{c)}.
It remains to prove that, under one or the other of hypotheses (i), (ii), \emph{c)} implies \emph{b)}.
Let us note that if $X$ satisfies (i), it is the same for each $Y_\alpha$.
It therefore suffices \sref{IV.5.9.3.1} to prove the following corollary.
\end{proposition}

\begin{remark}[5.11.3]
\label{IV.5.11.3}
The fact that $A^{(1)}$ is a finite $A$-algebra is no longer necessarily true if, in hypothesis~(ii) of \sref{IV.5.11.2}, one assumes only that $A$ is \emph{catenary}.
An example is given by the catenary local ring $A$ constructed in \sref{IV.5.6.11}, whose integral closure $A'$ (denoted $B$ in \sref{IV.5.6.11}) is a finite $A$-algebra; if $A^{(1)}$ were also a finite $A$-algebra, it would be contained in the field of fractions of $A$, and hence also contained in $A'$.
But on the other hand, with the notations of \sref{IV.5.6.11}, every prime ideal of height $1$ in $A$ is of the form $\mathfrak{p}A$, where $\mathfrak{p}$ is a prime ideal of height $1$ in $C$, and one has $A_{\mathfrak{p}A}=C_\mathfrak{p}$; on says \sref{IV.5.6.11} that $C_\mathfrak{p}=E_{\mathfrak{p}'}$, where $\mathfrak{p}'$ is the unique prime ideal of $E$ above $\mathfrak{p}$; by definition, $A_{\mathfrak{p}A}$ is integrally closed and contains $A'$; one would therefore have $A'\subset A^{(1)}$, and the hypothesis $A^{(1)}$ is a finite $A$-algebra would finally imply $A^{(1)}=A'$.
But this conclusion is absurd, for one of the two prime ideals of $A'$ above the maximal ideal $\mathfrak{n}A$ of $A$ is of height $1$, whereas $\mathfrak{n}A$ is of height $2$, which would contradict \sref{IV.5.10.17}[, (iv)].
\end{remark}

Conversely, suppose \emph{a)}; then, if $\sh{F}$ is a coherent $\sh{O}_X$-Module such that for every $x\in\operatorname{Ass}(\sh{F})\cap(X-Z)$ we have $\codim(\overline{\{x\}}\cap Z,\overline{\{x\}})\geqslant 2$, one can apply (with the notations of \sref{IV.5.11.1}) to each of the affine schemes $Y_\alpha=\Spec(B_\alpha)$, where $B_\alpha$ is an integral quotient ring of $A$, the result of condition \emph{a)}; since by hypothesis $Z_\alpha$ is contained in $Z^{(2)}(Y_\alpha)$, condition \emph{a)} (taking into account \sref{IV.5.10.2})

\oldpage[IV-2]{126}

implies that $\sh{H}^0_{Y_\alpha/Z_\alpha}(\sh{O}_{Y_\alpha})$ is a coherent $\sh{O}_{Y_\alpha}$-Module, and \emph{b)} follows then from \sref{IV.5.11.1}.

To see that \emph{c)} implies \emph{d)}, one reasons as above by applying \emph{c)} to the case where $\sh{F}=(A/\mathfrak{q})^\sim$ and $Z=V(\mathfrak{I})$; conversely, one proves that \emph{d)} implies \emph{c)} by using the equivalence of \emph{a)} and \emph{b)} in \sref{IV.5.11.1}.
It is evident that \emph{c)} is a particular case of \emph{b)}.
Finally, the fact that \emph{a)} (and hence each of the other conditions) is satisfied when $A$ satisfies one of the hypotheses (i), (ii) follows from \sref{IV.5.11.2}, in view of the definition of universally catenary rings and of universally Japanese rings.
\end{corollary}

\begin{remarks}[5.11.7]
\label{IV.5.11.7}
\begin{enumerate}
  \item[\rm(i)] If one ignores, in \sref{IV.5.11.5}, condition \emph{d)}, then \emph{a)} without any supplementary hypothesis on $X$; we will see further on \sref{IV.7.2.4} that it is indeed so when $X$ is a \emph{local} scheme.

  \item[\rm(ii)] If $A$ satisfies property \emph{a)} of \sref{IV.5.11.6}, the same holds for every \emph{ring of fractions} $S^{-1}A$ and for every finite $A$-algebra $C$.

\subsection{Relations between the properties of a Noetherian local ring \texorpdfstring{$A$}{A} and of a quotient ring \texorpdfstring{$A/tA$}{A/tA}}
\label{subsection:IV.5.12}

\oldpage[IV-2]{126}

We have already seen in \sref{IV.3.4} relations between the properties of $A$ and of $A/tA$ concerning the associated prime ideals, as well as the properties of being integral or reduced.
One gives in this section other relations between the properties of these rings, linked to the notions of dimension and of depth.

\begin{proposition}[5.12.1]
\label{IV.5.12.1}
Let $A$ be a Noetherian local ring, $X=\Spec(A)$, $t$ an element of $A$ forming part of a system of parameters $(\sref[0]{0.16.3.6})$, $X_0$ the closed sub-space $V(t)$ of $X$, $X_1$ the open complement $X-X_0$.
Let $Z$ be a part of $X$ such that every specialisation of a point of $Z$ belongs to $Z$.
Suppose that $X$ is \emph{biequidimensional} (which will be the case if $A$ is equidimensional and a quotient of a regular local ring).
Then, if one sets $Z_0=Z\cap X_0$, $Z_1=Z\cap X_1$, we have
\begin{equation}
\tag{5.12.1.1}
\label{IV.5.12.1.1}
  \codim(Z_0,X_0)\leqslant\codim(Z,X)\leqslant\codim(Z_1,X_1).
\end{equation}
\end{proposition}

The second inequality follows from the definition \sref{IV.5.1.3}; to prove the first, one can reduce to proving it when $Z$ is \emph{closed}: indeed, by hypothesis $Z$ is a union of closed parts $Z_\alpha$, and if one has $\codim(X_0\cap Z_\alpha,X_0)\leqslant\codim(Z_\alpha,X)$ for all $\alpha$, one also has $\codim(Z_0,X_0)=\inf_\alpha(\codim(X_0\cap Z_\alpha,X_0))\leqslant\inf_\alpha(\codim(Z_\alpha,X))=\codim(Z,X)$.
Suppose therefore $Z$ closed, so that one has
\[
  \codim(Z,X)=\dim(X)-\dim(Z)
\]

\oldpage[IV-2]{127}

by virtue of $(\sref[0]{0.14.3.5.1})$.
On the other hand, $X_0$ is evidently catenary, equidimensional and of dimension $\dim(X)-1$ by virtue of $(\sref[0]{0.16.3.4})$; one therefore also has
\[
  \codim(Z_0,X_0)=\dim(X_0)-\dim(Z_0)=\dim(X)-1-\dim(Z_0)
\]

But one has $\dim(Z_0)\geqslant\dim(Z)-1$ $(\sref[0]{0.16.3.4})$, which completes the proof of the first inequality \sref{IV.5.12.1.1}.

\begin{proposition}[5.12.2]
\label{IV.5.12.2}
Let $A$ be a Noetherian local ring, $M$ a finite type $A$-module, $t$ an $M$-regular element belonging to the maximal ideal $\mathfrak{m}$ of $A$, $k$ an integer $\geqslant 1$.
Suppose that $M/tM$ is equidimensional and satisfies $(\mathrm{S}_k)$, it is of the same for $M$.
\end{proposition}

Taking into account (\textbf{Err}\textsubscript{\textbf{III}},~30) and \sref{IV.5.7.3}[, (vi)], one can reduce to the case where $\operatorname{Supp}(M)=\Spec(A)=X$.
Set $X_0=V(t)=\Spec(A/tA)$; as $M/tM$ satisfies $(\mathrm{S}_1)$ by hypothesis, it has no associated immersed prime ideals \sref{IV.3.4.4}.
Applying \sref{IV.3.4.4} to a closed point of $X$ (and to every closed sub-prescheme of $X$), one sees that the irreducible components of $\operatorname{Supp}(M/tM)$ are exactly those of $Y_i\cap X_0$, where $Y_i$ ($1\leqslant i\leqslant r$) are the irreducible components of $X$.
As $t$ is by hypothesis $M$-regular, $V(t)$ does not contain any maximal point of $X$ \sref{IV.3.1.8}; each of the irreducible components of $Y_i\cap X_0$ is therefore of codimension $1$ in $Y_i$ \sref{IV.5.1.8}; as $X$ is catenary and that all the irreducible components of $\operatorname{Supp}(M/tM)$ have by hypothesis the same dimension, one concludes that the $Y_i$ all have the same dimension, in other words $X$ is equidimensional, hence \emph{biequidimensional} since $A$ is local and catenary.
To see that $M$ satisfies $(\mathrm{S}_k)$, let us apply the criterion \sref{IV.5.7.4}: one has to verify (with the notations of \sref{IV.5.7.4}) that, for every integer $n\geqslant 0$, we have $\codim(Z_n,X)>n+k$.
Now, the hypothesis on $M/tM$ and the criterion \sref{IV.5.7.4} show that $\codim(Z_n\cap X_0,X_0)>n+k$ for every integer $n\geqslant 0$.
But every specialisation of a point of $Z_n$ still belongs to $Z_n$, as we shall see in \sref{IV.6.11.5}; as $X$ is biequidimensional and that $t$ belongs to a system of parameters, hence also for $A$ (the annihilator of $M$ being nilpotent by virtue of the hypothesis $\operatorname{Supp}(M)=\Spec(A)$), the conclusion follows from \sref{IV.5.12.1}.

% Footnote in original: The reader will verify that \sref{IV.5.12.2} is not used in the proof of \sref{IV.6.11.5}.

\begin{remark}[5.12.3]
\label{IV.5.12.3}
If, in the statement of \sref{IV.5.12.2}, one assumes only that $M/tM$ is equidimensional, it does not follow that $M$ is equidimensional.
For example, let $k$ be a field, $B$ the ring of polynomials $k[T,U,V]$ with $3$ indeterminates, $C$ the local ring of $B$ at the maximal ideal $BT+BU+BV$ of $B$, and let $\mathfrak{p}=CU$, $\mathfrak{q}=CV+C(T+U)$; consider then the local ring $A=C/\mathfrak{p}\mathfrak{q}$ (geometrically, if $X$ is the closed sub-prescheme of affine space with $3$ dimensions over $k$, formed of a plane and of a line meeting the plane at a point $x$, $A$ is the local ring of $X$ at the point $x$).
Let $t$, $u$, $v$ be the canonical images of $T$, $U$, $V$ in $A$; it is immediate that $t$ is not a zero divisor in $A$, and $A/tA$ is isomorphic to $C_0/\mathfrak{p}_0\mathfrak{q}_0$, where $C_0$ is

\oldpage[IV-2]{128}

the local ring of $B_0=k[U,V]$ at the maximal ideal $B_0U+B_0V$, and $\mathfrak{p}_0=C_0U$, $\mathfrak{q}_0=C_0V$ ($\mathfrak{q}_0$ being the ideal of $C_0$).
It is immediate that $\Spec(A/tA)$ is irreducible and of dimension $1$, $A/tA$ does not satisfy $(\mathrm{S}_1)$ and $\Spec(A)$ is not equidimensional.
\end{remark}

\begin{corollary}[5.12.4]
\label{IV.5.12.4}
Let $A$ be a catenary Noetherian local ring, $t$ a regular element belonging to the maximal ideal of $A$, $k$ an integer $\geqslant 1$.
If $A/tA$ satisfies $(\mathrm{S}_k)$, it is of the same for $A$.
\end{corollary}

If $k=1$, the corollary follows from \sref{IV.3.4.4}, in view of the interpretation of condition $(\mathrm{S}_1)$ \sref{IV.5.7.5}.
If $k\geqslant 2$, it follows from the Hartshorne criterion \sref{IV.5.10.9} that $A/tA$ is equidimensional, and one can apply \sref{IV.5.12.2}.

\begin{proposition}[5.12.5]
\label{IV.5.12.5}
Let $A$ be a catenary Noetherian local ring, $t$ a regular element of $A$ belonging to the maximal ideal of $A$, $k$ an integer $\geqslant 0$.
If the ring $A/tA$ is reduced, equidimensional and satisfies $(\mathrm{R}_k)$, the ring $A$ also possesses these three properties.
\end{proposition}

One already knows \sref{IV.3.4.6} that $A$ is reduced, and it follows from \sref{IV.5.12.2}, applied for $k=1$, that $A$ is equidimensional.
Set $X=\Spec(A)$, $X_0=V(t)=\Spec(A/tA)$, and let $Z$ be the set of points $x\in X$ where $X$ is not regular; by virtue of $(\sref[0]{0.17.3.2})$ every specialisation of a point of $Z$ belongs to $Z$.
It follows on the other hand from $(\sref[0]{0.17.1.8})$ that for every point $x\in X_0$ where $X$ is regular, $X$ is also regular, hence the set $Z_0'$ of the points where $X_0$ is not regular contains $Z\cap X_0$; the hypothesis implies therefore that
\[
  k\leqslant\codim(Z_0',X_0)\leqslant\codim(Z_0,X_0).
\]

But, since $A$ is equidimensional and catenary, one has, by \sref{IV.5.12.1}
\[
  \codim(Z_0,X_0)\leqslant\codim(Z,X)
\]
which proves that $X$ satisfies $(\mathrm{R}_k)$.

\begin{remark}[5.12.6]
\label{IV.5.12.6}
If, in the statement of \sref{IV.5.12.5}, one assumes only that $A/tA$ is reduced and possesses property $(\mathrm{R}_k)$, it does not follow necessarily that $A$ possesses property $(\mathrm{R}_k)$.
For example, let $k$ be a field, $P(U,V)$ a polynomial of $k[U,V]$ irreducible such that the curve $\Gamma$ defined by the principal ideal $(P)$ in the affine plane $\Spec(k[U,V])$ has a singular point at the maximal ideal $(U)+(V)$ (for example $P(U,V)=U(U^2+V^2)+(U^2-V^2)$, giving a cubic with a double point).
Let then $B$ be the ring of polynomials with $4$ indeterminates $k[T,U,V,W]$, $C$ the local ring of $B$ at the maximal ideal $BT+BU+BV+BW$, and let
\[
  \mathfrak{p}=CW+CP(U-T,V),\quad\mathfrak{q}=CU.
\]

Consider then the local ring $A=C/\mathfrak{p}\mathfrak{q}$ (geometrically, if $X$ is the closed sub-prescheme of affine space with $4$ dimensions, formed of a hyperplane $H$ and of a ``cylinder'' $L$ with $2$ dimensions having for ``base'' the curve $\Gamma$ and whose ``singular line'' $Y$ is not contained in the hyperplane $H$, then $A$ is the local ring of $X$ at the point $x$ where $Y$ meets $H$).
One then sees immediately that $\Spec(A/tA)$ (where $t$ is the image of $T$ in $A$) has for sole singular point $x$, which is reduced and of dimension $2$; on the contrary, the generic point $y$ of the ``singular line'' $Y$ (defined by the image in $A$ of the ideal $C(U-T)+CV+CW$) is a singular point of $X$, and $\sh{O}_{X,y}$ is of dimension $1$; in other words, $A/tA$ is reduced and satisfies $(\mathrm{R}_1)$, but $A$ does not satisfy $(\mathrm{R}_1)$.
\end{remark}

\begin{corollary}[5.12.7]
\label{IV.5.12.7}
Let $A$ be a catenary Noetherian local ring, $t$ a regular element of $A$ belonging to the maximal ideal of $A$.
If $A/tA$ is integral and integrally closed, it is of the same for $A$.
\end{corollary}

By virtue of the Serre criterion \sref{IV.5.8.6} and the fact that the ring $A$ is local, it suffices to show that $\Spec(A)$ satisfies $(\mathrm{S}_2)$ and $(\mathrm{R}_1)$.
But the hypothesis on $A/tA$ implies that $A$ satisfies $(\mathrm{R}_1)$ by \sref{IV.5.12.5}, and that $A$ satisfies $(\mathrm{S}_2)$ by \sref{IV.5.12.4}.

\begin{proposition}[5.12.8]
\label{IV.5.12.8}
{\normalfont(Hironaka's lemma).}
Let $A$ be a Noetherian local ring, reduced, equidimensional and catenary.
Suppose moreover that for every minimal prime ideal $\mathfrak{q}_i$ of $A$, the ring $B_i=A/\mathfrak{q}_i$ is such that $B_i^{(1)}$ $($notation of \sref{IV.5.10.17}$)$ is a finite $B_i$-algebra $($which will be the case for example when $A$ possesses one of properties {\rm(i), (ii)} of \sref{IV.5.11.6}$)$.
Let $t$ be an element of $A$, not a zero divisor in $A$, and such that:
\begin{enumerate}
  \item[\rm(i)] The $A$-module $A/tA$ has only one associated non-immersed prime ideal $\mathfrak{p}$ $($necessarily of height $1$ by virtue of the Hauptidealsatz $(\sref[0]{0.16.3.2})$, but one does not assume that this is the sole prime ideal associated to $A/tA$$)$.
  \item[\rm(ii)] The ideal $\mathfrak{p}A_\mathfrak{p}$ of $A_\mathfrak{p}$ is generated by $t/1$.
  \item[\rm(iii)] The ring $A/\mathfrak{p}$ is integrally closed.
\end{enumerate}
Under these conditions, the ring $A$ is integral and integrally closed, and one has $\mathfrak{p}=tA$.
\end{proposition}

\oldpage[IV-2]{129}

If $Z=V(tA)$, hypothesis (i) implies that $Z$ is an irreducible closed part of $X=\Spec(A)$, whose generic point $z$ is such that $\mathfrak{i}_z=\mathfrak{p}$.
Hypothesis (ii) shows that the maximal ideal $\mathfrak{p}A_\mathfrak{p}$ of the Noetherian local ring $A_\mathfrak{p}$ is generated by a single element, hence $A_\mathfrak{p}$ is a \emph{discrete valuation ring} whose $t/1$ is a uniformiser (Bourbaki, \emph{Alg.\ comm.}, chap.~VI, \S~3, no~6, prop.~9); $A_\mathfrak{p}/tA_\mathfrak{p}$ is the residue field of $A_\mathfrak{p}$, hence an $A_\mathfrak{p}$-module of length $1$.
By virtue of \sref{IV.3.4.2}, this implies that $Z$ is contained in only \emph{one} irreducible component $X_j$ of $X$, for every other irreducible component $X_j$ of $X$, one has therefore $\dim(X_j\cap Z)<\dim(Z)$.
On the other hand, as $t$ is not a zero divisor in $A$, it is not contained in any of the $\mathfrak{q}_i$, and one has therefore \sref{IV.5.1.8} $\codim(X_i\cap Z,X_i)=\codim(X_j\cap Z,X_j)=1$, for all $j\neq i$.
As $A$ is supposed biequidimensional, the relation $\dim(X_i)\neq\dim(X_i\cap Z)+\dim(X_j\cap Z)$ would imply $\dim(X_i)\neq\dim(X_j)$ $(\sref[0]{0.14.3.3.1})$, which is absurd.
One sees therefore that there is only \emph{one minimal prime ideal} in $A$; as $A$ is by hypothesis reduced, this implies that it is \emph{integral}.
Let us note now that since $A^{(1)}$ is a finite $A$-algebra, it is a semi-local Noetherian ring, and every non-immersed prime ideal associated to $A^{(1)}/tA^{(1)}$ is therefore of height $1$ by virtue of the Hauptidealsatz $(\sref[0]{0.16.3.2})$; or, for such an ideal $\mathfrak{r}$, $\mathfrak{r}\cap A$ is of height $1$ by virtue of \sref{IV.5.10.17}[, (iv)] and as it contains $tA$, this can only be $\mathfrak{p}$ by virtue of hypothesis~(i).
On the other hand, $A^{(1)}$ is contained in the integral closure $A'$ of $A_\mathfrak{p}$; if one sets $S=A-\mathfrak{p}$, the integral closure of $A_\mathfrak{p}$ is $S^{-1}A'$ (Bourbaki, \emph{Alg.\ comm.}, chap.~V, \S~1, no~5, prop.~17), hence $S^{-1}A'=A_\mathfrak{p}$ since $A_\mathfrak{p}$ is a discrete valuation ring; it follows (\emph{loc.\ cit.}, \S~2, no~1, lemma~1) that there exists only \emph{one} prime ideal $\mathfrak{p}'$ of $A'$ above $\mathfrak{p}$, and \emph{a fortiori} only one prime $\mathfrak{p}^{(1)}$ of $A^{(1)}$ above $\mathfrak{p}$, and one has $A_\mathfrak{p}=A'_{\mathfrak{p}'}=A^{(1)}_{\mathfrak{p}^{(1)}}$.
Let us note now that $A^{(1)}$ satisfies $(\mathrm{S}_2)$ \sref{IV.5.10.17}[, (i)] and as $t$ is not a zero divisor in $A^{(1)}$, $A^{(1)}/tA^{(1)}$ has no associated immersed prime ideals

\oldpage[IV-2]{130}

\sref{IV.5.7.5}.
The ideal $\mathfrak{p}^{(1)}$ is therefore the sole prime ideal associated to $A^{(1)}/tA^{(1)}$, in other words, $tA^{(1)}$ is a $\mathfrak{p}^{(1)}$-primary ideal of $A^{(1)}$; this means also that the inverse image of $A^{(1)}$ of $tA^{(1)}_{\mathfrak{p}^{(1)}}=tA_\mathfrak{p}$, and this last is the maximal ideal of $A_\mathfrak{p}$ by (ii); hence $tA^{(1)}=\mathfrak{p}^{(1)}$ is prime in $A^{(1)}$.
On the other hand, $A^{(1)}/\mathfrak{p}^{(1)}$ is finite over $A/\mathfrak{p}$ and has the same field of fractions (the residue field of $A^{(1)}_{\mathfrak{p}^{(1)}}=A_\mathfrak{p}$), hence it is identical to $A/\mathfrak{p}$ by virtue of hypothesis~(iii).
One can therefore write $A^{(1)}=A+\mathfrak{p}^{(1)}=A+tA^{(1)}$, and as $t$ is contained in the radical of the semi-local Noetherian ring $A^{(1)}$, the Nakayama lemma implies $A^{(1)}=A$, whence $\mathfrak{p}^{(1)}=\mathfrak{p}=tA$.
But $A$ is catenary and $A/tA$ is integral and integrally closed by virtue of hypothesis~(iii), one concludes by \sref{IV.5.12.7} that $A$ is integrally closed.

\begin{remark}[5.12.9]
\label{IV.5.12.9}
The conclusion of \sref{IV.5.12.8} fails if one does not further assume $A$ equidimensional.
For example, let $k$ be a field, $B$ the ring of polynomials $k[X,Y,Z]$ with $3$ indeterminates, $C$ the ring $B/\mathfrak{r}_1\mathfrak{r}_2$, where $\mathfrak{r}_1=BZ$ and $\mathfrak{r}_2=BX+BY$; let $\mathfrak{m}$ be the maximal ideal $BX+BY+BZ$, $\mathfrak{n}=\mathfrak{m}/\mathfrak{r}_1\mathfrak{r}_2$ its image in $C$, $A$ the local ring $C_\mathfrak{n}$.
Finally, let $t_0$ be the image in $C$ of the element $Y+Z$ of $B$, $t$ its image in $A$ ($\Spec(C)$ is therefore formed of a plane $P$ and of a line $D$ not contained in this plane, $\Spec(C/t_0C)$ is formed of a line $D'$ contained in $P$ and passing through the point $D\cap P$).
The ring $A$ is reduced and catenary (being a quotient of a regular ring) and its minimal prime ideals $\mathfrak{q}_1$, $\mathfrak{q}_2$ are the images of $\mathfrak{r}_1$, $\mathfrak{r}_2$.
The $A$-module $A/tA$ has only one associated non-immersed prime ideal $\mathfrak{p}$, image of $BY+BZ$, $\mathfrak{p}A_\mathfrak{p}$ is generated by $t/1$ and $A/\mathfrak{p}$ is isomorphic to $k[X]$; but $A$ is not integral.
\end{remark}

\begin{corollary}[5.12.10]
\label{IV.5.12.10}
Let $A$ be an integral Noetherian local ring, satisfying one of the following hypotheses:
\begin{enumerate}
  \item[\rm a)] $A$ is a quotient of a regular ring.
  \item[\rm b)] $A$ is universally catenary and universally Japanese.
\end{enumerate}
Let $(x_i)_{1\leqslant i\leqslant n}$ be a sequence of $n$ elements of $A$; set $\mathfrak{I}=x_1A+\ldots+x_nA$, and suppose the following conditions are satisfied:
\begin{enumerate}
  \item[\rm(i)] There exists only one non-immersed prime ideal $\mathfrak{p}$ associated to $A/\mathfrak{I}$ and $\mathfrak{p}$ is of height $n$.
  \item[\rm(ii)] The maximal ideal $\mathfrak{p}A_\mathfrak{p}$ of $A_\mathfrak{p}$ is generated by the $x_i/1$ (hence $A_\mathfrak{p}$ is a regular local ring of dimension $n$).
  \item[\rm(iii)] The ring $A/\mathfrak{p}$ is integrally closed.
\end{enumerate}
Under these conditions, for every integer $i$ such that $0\leqslant i\leqslant n$, the quotient ring $A/(x_1A+\ldots+x_iA)$ is integrally closed and of dimension $\dim(A)-i$.
In particular, $\mathfrak{I}$ is prime, equal to $\mathfrak{p}$, $A$ is integrally closed and $(x_i)_{1\leqslant i\leqslant n}$ is a regular $A$-sequence.
\end{corollary}

We proceed by induction on $n$, the proposition being trivial for $n=0$ and we can therefore suppose $n\geqslant 2$.
Let $\mathfrak{I}'=x_1A+\ldots+x_{n-1}A$, and let $\mathfrak{q}$ be a prime ideal minimal among those that contain $\mathfrak{I}'$; on has $\operatorname{ht}(\mathfrak{q})\leqslant n-1$ $(\sref[0]{0.16.3.1})$; let us show that one has $\mathfrak{q}\subset\mathfrak{p}$.
Indeed, if $\mathfrak{p}'$ is minimal among the prime ideals of $A$ containing $\mathfrak{q}+x_nA$, $\mathfrak{p}'/\mathfrak{q}$ is of height $1$ in $A/\mathfrak{q}$; therefore $\mathfrak{p}'$ is of height $\leqslant n$ since $A$ is catenary $(\sref[0]{0.16.3.2})$; but as it contains $\mathfrak{I}$, whence $\mathfrak{p}'=\mathfrak{I}'+x_nA$, and hence $\mathfrak{q}\subset\mathfrak{p}$.
But by virtue of hypothesis~(ii) and of $(\sref[0]{0.17.1.7})$, $\mathfrak{I}'A_\mathfrak{p}$ is prime in $A_\mathfrak{p}$ and it is of height $n-1$ and $\mathfrak{q}$ of height $\leqslant n-1$; let us show that $\mathfrak{q}\cap\mathfrak{I}'A_\mathfrak{p}$ is the \emph{unique} non-immersed prime ideal associated to $A/\mathfrak{I}'$ and it is of height $n-1$, since $\mathfrak{I}'A_\mathfrak{p}$ is of height $n-1$ and $\mathfrak{q}A_\mathfrak{p}=\mathfrak{I}'A_\mathfrak{p}$ and as $A_\mathfrak{q}=(A_\mathfrak{p})_{\mathfrak{q}A_\mathfrak{p}}$, the maximal ideal $\mathfrak{q}A_\mathfrak{q}$ is generated by $\mathfrak{I}'$.
One sees therefore that the sequence $(x_i)_{1\leqslant i\leqslant n-1}$ already satisfies conditions (i) and (ii) of the statement.
Let us show also that it satisfies~(iii).
Consider for this the integral ring $B=A/\mathfrak{q}$, and let $t$ be the image of $x_n$ in $B$.
It is immediate (cf. \sref{IV.5.6.3}[, (iv)]) that if $A$ satisfies hypothesis \emph{a)} (resp.\ \emph{b)}) of \sref{IV.5.12.10}, let us show that $B$ and $t$ satisfy the hypotheses of Hironaka's lemma.
Indeed, an ideal $\mathfrak{n}$ of $B$ minimal among those containing $\mathfrak{q}+x_nA$, and one has seen that this unique ideal; since $B_\mathfrak{n}=A_\mathfrak{p}/\mathfrak{q}A_\mathfrak{p}$ and $\mathfrak{n}B_\mathfrak{n}=\mathfrak{p}A_\mathfrak{p}/\mathfrak{q}A_\mathfrak{p}$, the fact that $\mathfrak{q}A_\mathfrak{p}$ is generated by $\mathfrak{I}'$ implies that $\mathfrak{n}B_\mathfrak{n}$ is generated by $t$.
Finally, $B/\mathfrak{n}=A/\mathfrak{p}$ is integrally closed.
The application of \sref{IV.5.12.8} proves therefore that $B=A/\mathfrak{q}$ is integrally closed, and that $\mathfrak{p}=\mathfrak{q}+x_nA$.
Let us now use the induction hypothesis.
Considering for this the integral ring $B=A/\mathfrak{q}$, and let $t$ be the canonical image of $x_n$ in $B$.
It is immediate (cf. \sref{IV.5.6.3}[, (iv)]) that if $A$ satisfies hypothesis \emph{a)}, then $B$ and $t$ satisfy the hypotheses of the lemma of Hironaka.
In effect, a prime ideal $\mathfrak{n}$ of $B$ minimal among the prime ideals of $A$ containing $\mathfrak{q}+x_nA$, is induced on $A$ by a prime ideal minimal among those containing $\mathfrak{q}+x_nA$; and one has shown that the only such ideal is $\mathfrak{p}$; since $B_\mathfrak{n}=A_\mathfrak{p}/\mathfrak{q}A_\mathfrak{p}$ and $\mathfrak{n}B_\mathfrak{n}=\mathfrak{p}A_\mathfrak{p}/\mathfrak{q}A_\mathfrak{p}$, the fact that $\mathfrak{q}A_\mathfrak{p}$ is generated by $\mathfrak{I}'$ implies that $\mathfrak{n}B_\mathfrak{n}$ is generated by $t$.
Finally, $B/\mathfrak{n}=A/\mathfrak{p}$ is integrally closed.
The application of \sref{IV.5.12.8} proves therefore that $B=A/\mathfrak{q}$ is integrally closed, and that $\mathfrak{p}=\mathfrak{q}+x_nA$.
Let us use now the induction hypothesis, which shows that $A/(x_1A+\ldots+x_iA)$ is integrally closed and of dimension $\dim(A)-i$ for $0\leqslant i\leqslant n-1$ and that $\mathfrak{q}=\mathfrak{I}'$, whence $\mathfrak{p}=\mathfrak{I}'+x_nA=\mathfrak{I}$.
One concludes that $A/\mathfrak{I}=A/\mathfrak{p}$ is integrally closed, and that $\dim(A/\mathfrak{p})=\dim(A)-n$ and that $\dim(A_\mathfrak{p})=n=\dim(A)-\dim(A_\mathfrak{p})$ since $A$ is biequidimensional, which completes the proof.


\subsection{Properties of permanence by passage to the inductive limit}
\label{subsection:IV.5.13}

\oldpage[IV-2]{131}

Throughout this paragraph, $(A_\alpha,\varphi_{\beta\alpha})$ denotes an \emph{inductive system of rings} whose indexing set is pre-ordered and \emph{filtered}; one sets $A=\varinjlim A_\alpha$ and denotes by $\varphi_\alpha$ the canonical homomorphism $A_\alpha\to A$.

\begin{proposition}[5.13.1]
\label{IV.5.13.1}
\begin{enumerate}
  \item[\rm(i)] For every index $\alpha$, let $\mathfrak{a}_\alpha$ be an ideal of $A_\alpha$ such that, for $\alpha\leqslant\beta$, one has $\varphi_{\beta\alpha}(\mathfrak{a}_\alpha)\subset\mathfrak{a}_\beta$.
Then the $\mathfrak{a}_\alpha$ form an inductive system for the restrictions of the $\varphi_{\beta\alpha}$, $\mathfrak{a}=\varinjlim\mathfrak{a}_\alpha$ is identified canonically to an ideal of $A$, $A/\mathfrak{a}$ to $\varinjlim(A_\alpha/\mathfrak{a}_\alpha)$.
If moreover one has $\mathfrak{a}_\alpha=\varphi_{\beta\alpha}^{-1}(\mathfrak{a}_\beta)$ for $\alpha\leqslant\beta$, then $\mathfrak{a}_\alpha=\varphi_\alpha^{-1}(\mathfrak{a})$ for every $\alpha$.
  \item[\rm(ii)] Conversely, for every ideal $\mathfrak{a}$ of $A$, if one sets $\mathfrak{a}_\alpha=\varphi_\alpha^{-1}(\mathfrak{a})$ for every $\alpha$, the $\mathfrak{a}_\alpha$ form an inductive system such that $\mathfrak{a}=\varinjlim\mathfrak{a}_\alpha$.
\end{enumerate}
\end{proposition}

It is clear that (ii) is a particular case of (i).
In (i), the fact that $\mathfrak{a}$ is identified canonically to an ideal of $A$, and $A/\mathfrak{a}$ to $\varinjlim(A_\alpha/\mathfrak{a}_\alpha)$ follows from the exactness of the inductive limit functor in the category of abelian groups.
Moreover, $\mathfrak{a}$ is the union of the increasing family of the $\varphi_\alpha(\mathfrak{a}_\alpha)$, so that if $x_\alpha\in E_\alpha$ is such that $\varphi_\alpha(x_\alpha)\in\mathfrak{a}$, there exists $\beta\geqslant\alpha$ such that $\varphi_\alpha(x_\alpha)\in\mathfrak{a}$; but this implies the existence of an index $\beta\geqslant\alpha$ such that, if one sets $x_\beta=\varphi_{\beta\alpha}(x_\alpha)$, $y_\beta=\varphi_{\beta\alpha}(y_\alpha)$, one has $x_\beta y_\beta=0$; since $A_\beta$ is integral, one has $x_\beta=0$ or $y_\beta=0$ and as $x=\varphi_\beta(x_\beta)$ and $y=\varphi_\beta(y_\beta)$, this completes the proof that $A$ is integral.

\begin{corollary}[5.13.2]
\label{IV.5.13.2}
If $\mathfrak{N}_\alpha$ is the nilradical of $A_\alpha$, the nilradical of $A$ is identified to $\varinjlim\mathfrak{N}_\alpha$, and $A_{\mathrm{red}}$ to $\varinjlim(A_\alpha)_{\mathrm{red}}$.
In particular, if the $A_\alpha$ are reduced, the same holds for $A$.
\end{corollary}

\oldpage[IV-2]{132}

It is clear that $\varphi_{\beta\alpha}(\mathfrak{N}_\alpha)\subset\mathfrak{N}_\beta$ for $\alpha\leqslant\beta$, and that $\varinjlim\mathfrak{N}_\alpha$ is a nilpotent ideal contained in the nilradical $\mathfrak{N}$ of $A$.
Conversely, if $x\in\mathfrak{N}$, one has $x^h=0$ for an integer $h$; one can write $x=\varphi_\alpha(x_\alpha)$ for an index $\alpha$ and a $x_\alpha\in A_\alpha$; the relation $\varphi_\alpha(x_\alpha^h)=0$ implies the existence of a $\beta\geqslant\alpha$ such that $\varphi_{\beta\alpha}(x_\alpha^h)=0$, or else $x_\beta^h=0$ where $x_\beta=\varphi_{\beta\alpha}(x_\alpha)$; hence $x_\beta\in\mathfrak{N}_\beta$ and as $x=\varphi_\beta(x_\beta)$, this completes the proof of the corollary, taking into account \sref{IV.5.13.1}.

\begin{proposition}[5.13.3]
\label{IV.5.13.3}
\begin{enumerate}
  \item[\rm(i)] For every index $\alpha$, let $\mathfrak{p}_\alpha$ be a prime ideal of $A_\alpha$ such that, for $\alpha\leqslant\beta$, one has $\varphi_{\beta\alpha}(\mathfrak{p}_\alpha)\subset\mathfrak{p}_\beta$.
Then $\mathfrak{p}=\varinjlim\mathfrak{p}_\alpha$ is a prime ideal of $A$.
In particular, if the $A_\alpha$ are integral, the same holds for $A$.
  \item[\rm(ii)] If moreover one has $\mathfrak{p}_\alpha=\varphi_{\beta\alpha}^{-1}(\mathfrak{p}_\beta)$ for $\alpha\leqslant\beta$, then $A_\mathfrak{p}$ is identified canonically to $\varinjlim(A_\alpha)_{\mathfrak{p}_\alpha}$.
  \item[\rm(iii)] If $A$ is a local ring, $\mathfrak{p}$ a maximal ideal, and if the local homomorphisms $\varphi_\alpha$ are injective, then, setting $\mathfrak{p}_\alpha=\varphi_\alpha^{-1}(\mathfrak{p})$, one has $A=\varinjlim(A_\alpha)_{\mathfrak{p}_\alpha}$ and the homomorphisms $(A_\alpha)_{\mathfrak{p}_\alpha}\to A$ are injective.
\end{enumerate}
\end{proposition}

The first two assertions of (i) are in fact equivalent, since $A/\mathfrak{p}=\varinjlim(A_\alpha/\mathfrak{p}_\alpha)$.
Suppose therefore all the $A_\alpha$ integral, and let $x$, $y$ be two elements of $A$ such that $xy=0$.
There exists an index $\alpha$ such that $x=\varphi_\alpha(x_\alpha)$, $y=\varphi_\alpha(y_\alpha)$ with $x_\alpha$, $y_\alpha$ in $A_\alpha$, and $\varphi_\alpha(x_\alpha y_\alpha)=0$; but this implies the existence of an index $\beta\geqslant\alpha$ such that, if one sets $x_\beta=\varphi_{\beta\alpha}(x_\alpha)$, $y_\beta=\varphi_{\beta\alpha}(y_\alpha)$, one has $x_\beta y_\beta=0$; as $A_\beta$ is integral, one has $x_\beta=0$ or $y_\beta=0$ and as $x=\varphi_\beta(x_\beta)$ and $y=\varphi_\beta(y_\beta)$, this completes the proof that $A$ is integral.

Let us now prove (ii).
The hypothesis implies that the $A_\alpha-\mathfrak{p}_\alpha$ form an inductive system of multiplicative parts of the $A_\alpha$, whence $A-\mathfrak{p}$ is the inductive limit; hence the local rings $(A_\alpha)_{\mathfrak{p}_\alpha}$ form an inductive system (where the transition homomorphisms are local), and the canonical homomorphisms $(A_\alpha)_{\mathfrak{p}_\alpha}\to A_\mathfrak{p}$ form an inductive system of homomorphisms, whose inductive limit $\psi:\varinjlim((A_\alpha)_{\mathfrak{p}_\alpha})\to A_\mathfrak{p}$ is a surjective homomorphism.
It remains to see that $\psi$ is injective.
Now, if $x_\alpha\in A_\alpha$ and $s_\alpha\in A_\alpha-\mathfrak{p}_\alpha$ are such that $\varphi_\alpha(x_\alpha)/\varphi_\alpha(s_\alpha)=0$ in $A_\mathfrak{p}$, there exists $s'\in A-\mathfrak{p}$ such that $s'\varphi_\alpha(x_\alpha)=0$, and, replacing $\alpha$ by a larger index if needed, one can suppose that $s'=\varphi_\alpha(s_\alpha')$ with $s_\alpha'\in A_\alpha-\mathfrak{p}_\alpha$, whence $\varphi_\alpha(s_\alpha'x_\alpha)=0$; there is therefore a $\beta\geqslant\alpha$ such that $\varphi_{\beta\alpha}(s_\alpha'x_\alpha)=0$ in $A_\beta$, which proves therefore that the image of $x_\alpha/s_\alpha$ in $\varinjlim((A_\alpha)_{\mathfrak{p}_\alpha})$ is zero.

Finally, the first assertion of (iii) follows from (ii) and from the fact that $A_\mathfrak{p}=A$.
Furthermore, the elements of $A_\alpha-\mathfrak{p}_\alpha$ are invertible in $A$, hence also $A_{\mathfrak{p}_\alpha}=A$, and the fact that $(A_\alpha)_{\mathfrak{p}_\alpha}\to A$ is injective follows by flatness from the hypothesis that $\varphi_\alpha$ is injective.

\begin{corollary}[5.13.4]
\label{IV.5.13.4}
Suppose that the $A_\alpha$ are all integral, and that the $\varphi_{\beta\alpha}$ are injective.
Then, if $A_\alpha'$ is the integral closure of $A_\alpha$, $A'=\varinjlim A_\alpha'$ is the integral closure of $A$.
In particular, if $A_\alpha$ is integrally closed for every $\alpha$, the same holds for $A$.
If the $A_\alpha$ are local rings with local homomorphisms $($resp.\ geometrically unibranch$)$ $(\sref[0]{0.23.2.1})$ and the $\varphi_{\beta\alpha}$ local injective homomorphisms $($resp.\ geometrically unibranch$)$, $A$ is unibranch $($resp.\ geometrically unibranch$)$.
\end{corollary}

If $K_\alpha$ is the field of fractions of $A_\alpha$, it follows from \sref{IV.5.13.3}[, (ii)] applied to $\mathfrak{p}_\alpha=0$, that $K=\varinjlim K_\alpha$ is the field of fractions of $A$, and $A'=\varinjlim A_\alpha'$ is identified to

\oldpage[IV-2]{133}

a subring of $K$, evidently integral over $A$.
Conversely, let $z\in K$ be an element satisfying an integral dependence equation $z^n+\sum_{i=1}^n c_i z^{n-i}=0$, where the $c_i\in A$.
There exists an index $\alpha$, a $z_\alpha\in K_\alpha$ and $c_{i\alpha}\in A_\alpha$ such that $z$ (resp.\ the $c_i$) are the images of $z_\alpha$ (resp.\ of the $c_{i\alpha}$), hence one has $z_\alpha\in A_\alpha'$, which proves that $z\in A'$.
It follows immediately that if the $A_\alpha$ are integrally closed, the same holds for $A$.
If the $A_\alpha$ are local and the $\varphi_{\beta\alpha}$ local homomorphisms, one knows (\sref[0]{0.10.3.1.3}) that $A$ is a local ring and if $k_\alpha$ is the residue field of $A_\alpha$, $k=\varinjlim k_\alpha$ is that of $A$; if the $A_\alpha'$ are local, $A'$ is therefore local and $A$ is unibranch; if moreover the residue field $k_\alpha'$ of $A_\alpha'$ is a radicial extension of $k_\alpha$ for every $\alpha$, $k'=\varinjlim k_\alpha'$, which is the residue field of $A'$, is a radicial extension of $k$, and $A$ is therefore geometrically unibranch.

\begin{env}[5.13.5]
\label{IV.5.13.5}
We say that a ring $A$ is \emph{normal} if $\Spec(A)$ is normal, in other words if, for every prime ideal $\mathfrak{p}$ of $A$, $A_\mathfrak{p}$ is \emph{integral and integrally closed}.
One notes that this implies that $A$ is \emph{reduced} and that $\mathfrak{p}$ cannot contain more than \emph{one} minimal prime ideal of $A$, in other words that $\mathfrak{p}$ can belong to only one \emph{single} irreducible component of $\Spec(A)$; hence $\mathfrak{q}$ is the inverse image of (o) by the canonical homomorphism $A\to A_\mathfrak{p}$, and $A_\mathfrak{p}$ is therefore isomorphic to $(A/\mathfrak{q})_{\mathfrak{p}/\mathfrak{q}}$, which proves that the integral ring $A/\mathfrak{q}$ is \emph{integrally closed} as intersection of integrally closed rings $(\sref[I]{I.8.2.1.1})$.
If $A$ is \emph{Noetherian} and \emph{normal}, $A$ is the composed direct product of a finite number of integral and integrally closed rings $(\sref[I]{I.5.1.4})$.
\end{env}

\begin{corollary}[5.13.6]
\label{IV.5.13.6}
Suppose that the $A_\beta$ are normal and that for $\alpha\leqslant\beta$, every irreducible component of $\Spec(A_\beta)$ dominates an irreducible component of $\Spec(A_\alpha)$.
Then $A=\varinjlim A_\alpha$ is normal.
\end{corollary}

Indeed, let $\mathfrak{p}$ be a prime ideal of $A$ and set, for every $\alpha$, $\mathfrak{p}_\alpha=\varphi_\alpha^{-1}(\mathfrak{p})$.
By hypothesis, for every $\alpha$, $\mathfrak{p}_\alpha$ contains a unique minimal ideal $\mathfrak{q}_\alpha$ of $A_\alpha$, and for $\alpha\leqslant\beta$, $\varphi_{\beta\alpha}^{-1}(\mathfrak{q}_\beta)$ is a minimal ideal of $A_\alpha$ contained in $\mathfrak{p}_\alpha$, hence equal to $\mathfrak{q}_\alpha$.
If one sets $B_\alpha=A_\alpha/\mathfrak{q}_\alpha$, the integral rings $B_\alpha$ form an inductive system for the homomorphisms $\varphi'_{\beta\alpha}:\mathrm{B}_\alpha\to\mathrm{B}_\beta$ induced by the $\varphi_{\beta\alpha}$ by passage to quotients, and which are injective by definition; moreover the rings $B_\alpha$ are integrally closed, hence the same holds for $B=\varinjlim B_\alpha$ \sref{IV.5.13.4}.
If one sets $\mathfrak{p}_\alpha'=\mathfrak{p}_\alpha/\mathfrak{q}_\alpha$, one has moreover $(\mathrm{B}_\alpha)_{\mathfrak{p}_\alpha'}=(A_\alpha)_{\mathfrak{p}_\alpha}$; as $\varinjlim(B_\alpha)_{\mathfrak{p}_\alpha'}=\varinjlim(A_\alpha)_{\mathfrak{p}_\alpha}=A_\mathfrak{p}$ is a local ring in an ideal of a prime of $B$ \sref{IV.5.13.3}, it is an integral ring and integrally closed, which proves the corollary.

\begin{proposition}[5.13.7]
\label{IV.5.13.7}
Suppose that the $A_\alpha$ are regular and that, for $\alpha\leqslant\beta$, $\varphi_{\beta\alpha}$ makes $A_\beta$ a flat $A_\alpha$-module; suppose in addition that $A=\varinjlim A_\alpha$ is Noetherian.
Then $A$ is regular.
\end{proposition}

By definition, it suffices to prove that for every prime ideal $\mathfrak{p}$ of $A$, the local Noetherian ring $A_\mathfrak{p}$ is regular; or, if $\mathfrak{p}_\alpha=\varphi_\alpha^{-1}(\mathfrak{p})$, on has $\mathfrak{p}=\varinjlim\mathfrak{p}_\alpha$ and $A_\mathfrak{p}=\varinjlim(A_\alpha)_{\mathfrak{p}_\alpha}$ \sref{IV.5.13.3}; as by hypothesis the $(A_\alpha)_{\mathfrak{p}_\alpha}$ are regular and that $(A_\beta)_{\mathfrak{p}_\beta}$ is an $(A_\alpha)_{\mathfrak{p}_\alpha}$-module flat for $\alpha\leqslant\beta$ $(\sref[0]{0.6.3.2})$, one sees that one is reduced to the case where the $A_\alpha$ and $A$ are also \emph{local} rings.
To prove that $A$ is regular, it suffices then to show that if $M$, $N$ are two finite type $A$-modules, there exists $i_0$ such that, for $i\geqslant i_0$,

\oldpage[IV-2]{134}

on has $\operatorname{Tor}_i^A(M,N)=0$ $(\sref[0]{0.17.3.1}$ and $17.2.6)$.
But one has the following lemma (where the $A_\alpha$ are again arbitrary):

\begin{lemma}[5.13.7.1]
\label{IV.5.13.7.1}
For every finite type $A$-module $M$ of finite presentation, there exists an index $\alpha$ and a finite type $A_\alpha$-module $M_\alpha$ of finite presentation such that $M$ is isomorphic to $M_\alpha\otimes_{A_\alpha}A$.
\end{lemma}

Indeed, one can suppose that $M=A^r/P$, where $P$ is a sub-$A$-module of finite type of $A^r$.
As $A^r=\varinjlim(A_\alpha^r)$ and that $P$ is of finite type, there exists $\alpha$ and a sub-$A_\alpha$-module of finite type $P_\alpha$ of $A_\alpha^r$ such that $P$ is the canonical image of $P_\alpha\otimes_{A_\alpha}A$; the exactness of the tensor product shows therefore that $M$ is isomorphic to $(A_\alpha^r/P_\alpha)\otimes_{A_\alpha}A$.

Having established this lemma, since $A$ is Noetherian, write therefore $M=M_\alpha\otimes_{A_\alpha}A$, $N=N_\alpha\otimes_{A_\alpha}A$; by virtue of the hypothesis, $A$ is an $A_\alpha$-module flat $(\sref[0]{0.6.3.2})$, hence on has $\operatorname{Tor}_i^A(M,N)=\operatorname{Tor}_i^{A_\alpha}(M_\alpha,N_\alpha)\otimes_{A_\alpha}A$ an isomorphism approximately $(\sref[III]{III.6.3.9})$; as by hypothesis $A_\alpha$ is regular, there exists $i_0$ such that $\operatorname{Tor}_i^{A_\alpha}(M_\alpha,N_\alpha)=0$ for $i\geqslant i_0$.

\begin{corollary}[5.13.8]
\label{IV.5.13.8}
Suppose that the $A_\alpha$ are Noetherian and satisfy property $(\mathrm{R}_k)$ for an integer $k\geqslant 0$.
Suppose furthermore that for $\alpha\leqslant\beta$, $\varphi_{\beta\alpha}$ makes $A_\beta$ a flat $A_\alpha$-module, that $A=\varinjlim A_\alpha$ is Noetherian and that, for every $\alpha$, ${}^a\!\varphi_\alpha$ is a homeomorphism of $\Spec(A)$ over $\Spec(A_\alpha)$.
Then $A$ satisfies $(\mathrm{R}_k)$.
\end{corollary}

Let in fact $\mathfrak{p}$ be a prime ideal of $A$ and, for every $\alpha$, set $\mathfrak{p}_\alpha=\varphi_\alpha^{-1}(\mathfrak{p})$, so that $\mathfrak{p}=\varinjlim\mathfrak{p}_\alpha$ and $A_\mathfrak{p}=\varinjlim(A_\alpha)_{\mathfrak{p}_\alpha}$ \sref{IV.5.13.3}.
It follows from $(\sref[I]{I.2.4.2})$ and from the hypothesis that the morphism $\Spec(A_\mathfrak{p})\to\Spec((A_\alpha)_{\mathfrak{p}_\alpha})$ is a surjective homeomorphism for every $\alpha$; one therefore has $\dim(A_\mathfrak{p})=\dim((A_\alpha)_{\mathfrak{p}_\alpha})$ for every $\alpha$.
Suppose then that $\dim(A_\mathfrak{p})\leqslant k$, and the hypothesis implies that $(A_\alpha)_{\mathfrak{p}_\alpha}$ is a regular ring; since furthermore, for $\alpha\leqslant\beta$, $(A_\beta)_{\mathfrak{p}_\beta}$ is an $(A_\alpha)_{\mathfrak{p}_\alpha}$-module flat $(\sref[0]{0.6.3.2})$, one concludes by \sref{IV.5.13.7} that $A_\mathfrak{p}$ is regular, which proves the corollary.
