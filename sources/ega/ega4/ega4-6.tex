\section{Flat morphisms of locally Noetherian preschemes}
\label{section:IV.6}

\oldpage[IV-2]{134}

Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism.
For every $y\in Y$, the fibre $f^{-1}(y)=X\times_Y\Spec(k(y))$ is also a locally Noetherian prescheme, since it suffices to see this when $Y=\Spec(A)$ and $X=\Spec(B)$ are spectra of Noetherian rings, and then $B\otimes_A k(y)$ is a ring of fractions of the quotient ring $B/\mathfrak{i}_yB$, and hence is Noetherian.
We first propose in this section to relate the properties of $X$, of $Y$, and of the fibres $f^{-1}(y)$ (where $y$ ranges over $Y$), under the hypothesis that the morphism $f$ is \emph{flat}.
The questions treated will reduce to the study of the relations between the local Noetherian rings $\sh{O}_y$, $\sh{O}_x$, and $\sh{O}_x\otimes_{\sh{O}_y}k(y)$, where the homomorphism $\sh{O}_y\to\sh{O}_x$ is local and makes $\sh{O}_x$ a flat $\sh{O}_y$-module.
In (6.11) to (6.13) (which are fairly separate from the rest of the section, by virtue of their ``absolute'' rather than ``relative'' nature), we apply some of the preceding results to find criteria allowing us to assert that the singular locus (or certain analogous sets) of certain preschemes are closed sets; these criteria will play an important role in \textsection\ref{section:IV.7}.

\subsection{Flatness and dimension}
\label{subsection:IV.6.1}

\oldpage[IV-2]{135}

\begin{proposition}[6.1.1]
\label{IV.6.1.1}
Let $A$, $B$ be two local Noetherian rings, $\mathfrak{m}$ the maximal ideal of $A$, $k=A/\mathfrak{m}$ its residue field, $\varphi:A\to B$ a local homomorphism.
Suppose that for every prime ideal $\mathfrak{p}$ of $A$, distinct from $\mathfrak{m}$, and for every minimal element $\mathfrak{q}$ of the set of prime ideals of $B$ containing $\mathfrak{p}B$, $\varphi^{-1}(\mathfrak{q})$ is distinct from $\mathfrak{m}$ (in other words, that no irreducible component of $\Spec(B/\mathfrak{p}B)$ is contained in the inverse image of the closed point $\mathfrak{m}$ of $\Spec(A)$ in $\Spec(B)$).
Then we have
\begin{equation}
\label{IV.6.1.1.1}
  \dim(B)=\dim(A)+\dim(B\otimes_A k).\tag{6.1.1.1}
\end{equation}
\end{proposition}

\begin{proof}
We argue by induction on $n=\dim(A)$; the assertion is trivial for $n=0$, since $\mathfrak{m}B$ is then contained in the nilradical of $B$ because $\mathfrak{m}$ is the nilradical of $A$.
We can thus suppose $n>0$.
Let $\mathfrak{q}_i$ $(1\leq i\leq m)$ be the minimal prime ideals of $B$, and set $\mathfrak{p}_i=\varphi^{-1}(\mathfrak{q}_i)$; for every $i$, we have $\mathfrak{p}_i\neq\mathfrak{m}$; otherwise there would exist (since $n>0$) a prime ideal $\mathfrak{p}\neq\mathfrak{m}$ contained in $\mathfrak{p}_i$ and since $\mathfrak{q}_i$ is minimal among the prime ideals of $B$ containing $\mathfrak{p}B$, one would arrive at a contradiction with the hypothesis.
It follows that $\mathfrak{m}$ is distinct from the union of the $\mathfrak{p}_i$ and the minimal prime ideals $\mathfrak{p}'_j$ $(1\leq j\leq r)$ of $A$ (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection1, n\textsuperscript{o}~1, prop.~2) and there exists an $a\in\mathfrak{m}$ belonging to none of the $\mathfrak{p}_i$ nor the $\mathfrak{p}'_j$.
Set $A'=A/aA$, $B'=B/aB$; we have $(\mathbf{0}$, 16.3.4)
\[
  \dim(A')=\dim(A)-1,\quad \dim(B')=\dim(B)-1
\]
by construction of $a$, and on the other hand $B'\otimes_{A'}k=B\otimes_A k=B/\mathfrak{m}B$, hence
\[
  \dim(B'\otimes_{A'}k)=\dim(B\otimes_A k)
\]
and it therefore suffices to prove \sref{IV.6.1.1.1} for $A'$ and $B'$; but, by virtue of the bijective correspondence between ideals of $A$ (resp.\ $B$) containing $a$ and ideals of $A'$ (resp.\ $B'$), the hypotheses of the statement are also valid for $A'$ and $B'$; one can thus apply the induction hypothesis, which completes the proof.
\end{proof}

\begin{corollary}[6.1.2]
\label{IV.6.1.2}
Let $A$, $B$ be two local Noetherian rings, $k$ the residue field of $A$, $\varphi:A\to B$ a local homomorphism, $M\neq 0$ a finite type $A$-module, $N\neq 0$ a finite type $B$-module.
If $N$ is a flat $A$-module (resp.\ if $B$ is a flat $A$-module), then we have
\begin{equation}
\label{IV.6.1.2.1}
  \dim_B(M\otimes_A N)=\dim_A(M)+\dim_{B\otimes_A k}(N\otimes_A k)\tag{6.1.2.1}
\end{equation}
(resp.\ \sref{IV.6.1.1.1}).
\end{corollary}

\begin{proof}
It suffices to prove the assertion concerning $N$; on the other hand, if $\mathfrak{b}$ is the annihilator of $N$, one can replace $B$ by $B/\mathfrak{b}$, and hence suppose that $\Supp(N)=\Spec(B)$; the hypothesis then means that the morphism $f:\Spec(B)\to\Spec(A)$ corresponding to $\varphi$ is quasi-flat \sref{IV.2.3.3}; we conclude \sref{IV.2.3.4} that for every prime ideal $\mathfrak{p}\neq\mathfrak{m}$ of $A$,
\oldpage[IV-2]{136}
every irreducible component of $f^{-1}(\mathrm{V}(\mathfrak{p}))$ dominates $\mathrm{V}(\mathfrak{p})$, and hence the condition of \sref{IV.6.1.1} is satisfied, whence the conclusion.
\end{proof}

\begin{corollary}[6.1.3]
\label{IV.6.1.3}
Let $A$, $B$ be two local Noetherian rings, $\mathfrak{m}$ the maximal ideal of $A$, $k$ its residue field, $\varphi:A\to B$ a local homomorphism.
Suppose that $\dim(B\otimes_A k)=0$ (that is to say $(\mathbf{0}$, 16.2.1) that $\mathfrak{m}B$ is an ideal of definition of $B$).
Then $\dim(B)\leq\dim(A)$.
If in addition there exists a finite type $B$-module $N$ that is a flat $A$-module and whose support equals $\Spec(B)$ (which is the case in particular if $B$ is a flat $A$-module), then $\dim(B)=\dim(A)$.
\end{corollary}

\begin{proof}
The first assertion follows from \sref{IV.5.5.1} and the second from \sref{IV.6.1.2}.
\end{proof}

\begin{corollary}[6.1.4]
\label{IV.6.1.4}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a surjective morphism.
For every closed subset $Z$ of $Y$, we then have
\begin{equation}
\label{IV.6.1.4.1}
  \codim(f^{-1}(Z),X)\leq\codim(Z,Y).\tag{6.1.4.1}
\end{equation}
Moreover, if $f$ is quasi-flat \sref{IV.2.3.3}, the two sides of \sref{IV.6.1.4.1} are equal.
\end{corollary}

\begin{proof}
Indeed, if $y$ is a maximal point of $Z$ and $x$ a maximal point of the fibre $f^{-1}(y)$, we have $\dim(\sh{O}_{X,x})\leq\dim(\sh{O}_{Y,y})$ by \sref{IV.5.5.2}; the inequality \sref{IV.6.1.4.1} then follows from \sref{IV.5.1.2.1} and \sref{IV.5.1.3.1} and the fact that $f$ is surjective.
If in addition $f$ is quasi-flat, we know \sref{IV.2.3.4} that every irreducible component of $f^{-1}(Z)$ dominates an irreducible component of $Z$; the maximal points of $f^{-1}(Z)$ are thus the maximal points of the fibres $f^{-1}(y)$, where $y$ ranges over the set of maximal points of $Z$ $(\mathbf{0_I}$, 2.1.8), and at each of these maximal points $x$ we have $\dim(\sh{O}_x\otimes_{\sh{O}_y}k(y))=0$; since in addition $\sh{O}_x$ is a flat $\sh{O}_y$-module, we have $\dim(\sh{O}_x)=\dim(\sh{O}_y)$ by \sref{IV.6.1.1}, whence the equality in \sref{IV.6.1.4.1}, taking into account that $f$ is surjective.
\end{proof}

Proposition \sref{IV.6.1.1} admits the following partial converse:

\begin{proposition}[6.1.5]
\label{IV.6.1.5}
Let $A$, $B$ be two local Noetherian rings, $k$ the residue field of $A$, $\varphi:A\to B$ a local homomorphism, $M$ a finite type $B$-module.
Suppose that:
\begin{enumerate}
  \item[\rm 1\textsuperscript{o}] $A$ is a regular ring.
  \item[\rm 2\textsuperscript{o}] $M$ is a Cohen-Macaulay $B$-module.
  \item[\rm 3\textsuperscript{o}] We have $\dim_B(M)=\dim(A)+\dim_{B\otimes_A k}(M\otimes_A k)$.
\end{enumerate}
Then $M$ is a flat $A$-module.
\end{proposition}

\begin{proof}
We proceed by induction on $n=\dim(A)$.
If $n=0$, $A$ is a field since it is regular and the assertion is trivial.
Suppose $n>0$; let $\mathfrak{m}$ be the maximal ideal of $A$, and let $x\in\mathfrak{m}$ be an element not belonging to $\mathfrak{m}^2$; we then know $(\mathbf{0}$, 17.1.8) that $A'=A/xA$ is regular and $\dim(A')=\dim(A)-1$.
Set
\[
  B'=B/xB,\quad M'=M/xM=M\otimes_A A';
\]
we thus have
\[
  B'\otimes_{A'}k=B\otimes_A k,\quad M'\otimes_{A'}k=M\otimes_A k
\]
and by \sref{IV.5.5.1.2} we have
\[
  \dim_{B'}(M')\leq\dim(A')+\dim_{B'\otimes_{A'}k}(M'\otimes_{A'}k);
\]
\oldpage[IV-2]{137}
we thus conclude from $(\mathbf{0}$, 16.3.4) that
\[
  \dim_B(M)\leq\dim_{B'}(M')+1\leq\big(\dim(A')+\dim_{B\otimes_A k}(M\otimes_A k)\big)+1=\dim(A)+\dim_{B\otimes_A k}(M\otimes_A k).
\]
Since by hypothesis the extreme members of these inequalities are equal, we necessarily have:
(i) $\dim_{B'}(M')=\dim_B(M)-1$, and since $M$ is by hypothesis a Cohen-Macaulay $B$-module, this means that $x$ is $M$-regular $(\mathbf{0}$, 16.1.9 and 16.5.5);
(ii) $\dim_{B'}(M')=\dim(A')+\dim_{B'\otimes_{A'}k}(M'\otimes_{A'}k)$;
since $M'$ is a Cohen-Macaulay $B'$-module by (i) and $(\mathbf{0}$, 16.1.9 and 16.5.5) and $A'$ is regular, the induction hypothesis shows that $M'$ is a flat $A'$-module; we then deduce from $(\mathbf{0_{III}}$, 10.2.7) that $M$ is a flat $A$-module, since $x$ is $M$-regular by (i).
\end{proof}

\subsection{Flatness and projective dimension}
\label{subsection:IV.6.2}

\begin{proposition}[6.2.1]
\label{IV.6.2.1}
\begin{enumerate}
  \item[\rm(i)] Let $A$, $B$ be two rings, $\varphi:A\to B$ a homomorphism such that $B$ is a flat $A$-module.
  Then, for every $A$-module $M$, we have
  \begin{equation}
  \label{IV.6.2.1.1}
    \operatorname{dim.proj}_B(M\otimes_A B)\leq\operatorname{dim.proj}_A(M).\tag{6.2.1.1}
  \end{equation}
  \item[\rm(ii)] Suppose in addition that $A$ is a Noetherian ring, $B$ a faithfully flat $A$-module, and $M$ a finite type $A$-module; then
  \begin{equation}
  \label{IV.6.2.1.2}
    \operatorname{dim.proj}_B(M\otimes_A B)=\operatorname{dim.proj}_A(M).\tag{6.2.1.2}
  \end{equation}
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}
  \item[\rm(i)] We can restrict to the case where $n=\operatorname{dim.proj}_A(M)$ is finite; there thus exists a left resolution
  \[
    0\to P_n\to P_{n-1}\to\ldots\to P_0\to M\to 0
  \]
  of $M$ by projective $A$-modules; since $B$ is a flat $A$-module, the sequence
  \[
    0\to P_n\otimes_A B\to P_{n-1}\otimes_A B\to\ldots\to P_0\otimes_A B\to M\otimes_A B\to 0
  \]
  is exact; on the other hand, the $P_i\otimes_A B$ are projective $B$-modules; whence the conclusion.

  \item[\rm(ii)] Suppose that $\operatorname{dim.proj}_B(M\otimes_A B)=m$, and consider an exact sequence
  \[
    0\to R\to P_{m-1}\to P_{m-2}\to\ldots\to P_0\to M\to 0
  \]
  where the $P_i$ $(0\leq i\leq m-1)$ are projective $A$-modules of finite type; since $A$ is Noetherian, $R$ is also a finite type $A$-module.
  Since $B$ is a flat $A$-module, we also have an exact sequence
  \[
    0\to R\otimes_A B\to P_{m-1}\otimes_A B\to\ldots\to P_0\otimes_A B\to M\otimes_A B\to 0
  \]
  and the hypothesis on $M\otimes_A B$ implies that $R\otimes_A B$ is a projective $B$-module $(\mathbf{0}$, 17.2.1).
  Since $B$ is a faithfully flat $A$-module, we conclude that $R$ is a projective $A$-module (Bourbaki, \emph{Alg.~comm.}, chap.~I\textsuperscript{er}, \textsection3, n\textsuperscript{o}~6, prop.~12); hence $\operatorname{dim.proj}_A(M)\leq m$, which completes the proof.
\end{enumerate}
\end{proof}

\begin{corollary}[6.2.2]
\label{IV.6.2.2}
Let $f:X\to Y$ be a flat morphism of preschemes, $\sh{E}$ an $\sh{O}_Y$-Module quasi-coherent.
If $\operatorname{dim.proj}(\sh{E})\leq n$ $(\mathbf{0}$, 17.2.14), then $\operatorname{dim.proj}(f^*(\sh{E}))\leq n$.
\end{corollary}

\oldpage[IV-2]{138}

\begin{proposition}[6.2.3]
\label{IV.6.2.3}
Let $A$, $B$ be two local Noetherian rings, $k$ the residue field of $A$, $\varphi:A\to B$ a local homomorphism, $N$ a finite type $B$-module.
Suppose that $B$ and $N$ are flat $A$-modules.
Then we have
\begin{equation}
\label{IV.6.2.3.1}
  \operatorname{dim.proj}_B(N)=\operatorname{dim.proj}_{B\otimes_A k}(N\otimes_A k).\tag{6.2.3.1}
\end{equation}
\end{proposition}

\begin{proof}
Consider a left resolution of $N$ by free $B$-modules of finite type
\[
  \ldots\to L_i\to L_{i-1}\to\ldots\to L_0\to N\to 0.
\]
Since the $L_i$ and $N$ are flat $A$-modules ($B$ being a flat $A$-module) it follows from \sref{IV.2.1.10} that the $\mathrm{Z}_i(L_.)$ are flat $A$-modules, that $L_.\otimes_A k$ is a left resolution of the $(B\otimes_A k)$-module $N\otimes_A k$, and that $\mathrm{Z}_i(L_.\otimes_A k)=\mathrm{Z}_i(L_.)\otimes_A k$ for every $i$.
Note that since $B$ is Noetherian, the $\mathrm{Z}_i(L_.)$ are $B$-modules of finite type, hence the $\mathrm{Z}_i(L_.\otimes_A k)$ are $(B\otimes_A k)$-modules of finite type.
Now, to say that a $B$-module (resp.\ a $(B\otimes_A k)$-module) of finite type is flat is equivalent to saying that it is \emph{free} or equivalently \emph{projective} $(\mathbf{0_{III}}$, 10.1.3).
Since the $\mathrm{Z}_i(L_.)$ are flat $A$-modules, it follows moreover from $(\mathbf{0_{III}}$, 10.2.5) (taking $C=B$) that, for $\mathrm{Z}_i(L_.)$ to be a flat $B$-module, it is necessary and sufficient that $\mathrm{Z}_i(L_.\otimes_A k)$ be a flat $(B\otimes_A k)$-module.
The smallest integer $n$ such that $\mathrm{Z}_{n-1}(L_.)$ is a free $B$-module is thus also the smallest integer such that $\mathrm{Z}_{n-1}(L_.\otimes_A k)$ is a free $(B\otimes_A k)$-module, which proves the proposition $(\mathbf{0}$, 17.2.1).
\end{proof}

\subsection{Flatness and depth}
\label{subsection:IV.6.3}

\begin{proposition}[6.3.1]
\label{IV.6.3.1}
Let $A$, $B$ be two local Noetherian rings, $k$ the residue field of $A$, $\varphi:A\to B$ a local homomorphism, $M$ a finite type $A$-module, $N$ a finite type $B$-module.
If $N$ is a flat $A$-module $\neq 0$, then we have
\begin{equation}
\label{IV.6.3.1.1}
  \operatorname{prof}_B(M\otimes_A N)=\operatorname{prof}_A(M)+\operatorname{prof}_{B\otimes_A k}(N\otimes_A k).\tag{6.3.1.1}
\end{equation}
\end{proposition}

\begin{proof}
We can restrict to the case where $M\neq 0$, since otherwise the two sides of \sref{IV.6.3.1.1} are both equal to $+\infty$.
We proceed by induction on the integer $n$ equal to the second member of \sref{IV.6.3.1.1} (which is finite by the hypotheses, $(\mathbf{0}$, 16.4.6.2) and the Nakayama lemma applied to $N$).
Suppose first that $n=0$; then, if $\mathfrak{m}$ and $\mathfrak{n}$ denote the maximal ideals of $A$ and $B$ respectively, $\mathfrak{m}$ is an ideal premier associated to $M$ and $\mathfrak{n}/\mathfrak{m}B$ an ideal premier associated to $N\otimes_A k$ $(\mathbf{0}$, 16.4.6, (i)).
We then conclude from \sref{IV.3.3.1} that $\mathfrak{n}$ is a prime ideal of $B$ associated to $M\otimes_A N$, hence $(\mathbf{0}$, 16.4.6, (i)) $\operatorname{prof}_B(M\otimes_A N)=0$.
Suppose now $n>0$, and distinguish two cases:

\emph{a)} Suppose $\operatorname{prof}_A(M)>0$.
Let $x\in\mathfrak{m}$ be an $M$-regular element, and set
\[
  A'=A/xA,\quad B'=B/xB,\quad M'=M/xM,\quad N'=N/xN;
\]
\oldpage[IV-2]{139}
we have
\[
  B'\otimes_{A'}k=B\otimes_A k,\quad N'\otimes_{A'}k=N\otimes_A k
\]
and, since $N$ is a flat $A$-module, the hypothesis that $x$ is $M$-regular implies that $x$ is also $(M\otimes_A N)$-regular $(\mathbf{0_I}$, 6.1.1).
We have by $(\mathbf{0}$, 16.4.6, (ii) and 16.4.8)
\[
  \operatorname{prof}_{A'}(M')=\operatorname{prof}_A(M)-1,\quad \operatorname{prof}_{B'}(M'\otimes_{A'}N')=\operatorname{prof}_B(M\otimes_A N)-1
\]
and
\[
  \operatorname{prof}_{B'\otimes_{A'}k}(N'\otimes_{A'}k)=\operatorname{prof}_{B\otimes_A k}(N\otimes_A k).
\]
The equality \sref{IV.6.3.1.1} is thus a consequence of the same relation for $A'$, $B'$, $M'$, and $N'$; but since $N$ is a flat $A$-module, $N'=N\otimes_A A'$ is a flat $A'$-module $(\mathbf{0_I}$, 6.2.1); one can therefore apply the induction, which proves \sref{IV.6.3.1.1} in this case.

\emph{b)} Suppose $\operatorname{prof}_{B\otimes_A k}(N\otimes_A k)>0$.
Consider a $y\in\mathfrak{n}$ that is $(N\otimes_A k)$-regular; it follows from $(\mathbf{0_{III}}$, 10.2.4) that $y$ is $N$-regular and that
\[
  N'=N/yN
\]
is a flat $A$-module, since $N$ is assumed to be a flat $A$-module.
Applying $(\mathbf{0_I}$, 6.1.2) to the exact sequence of $A$-modules
\[
  0\to N\xrightarrow{y} N\to N'\to 0
\]
we obtain isomorphisms
\[
  (M\otimes_A N)/y(M\otimes_A N)\xrightarrow{\sim} M\otimes_A N'\quad\text{and}\quad N'\otimes_A k\xrightarrow{\sim}(N\otimes_A k)/y(N\otimes_A k)
\]
and moreover $y$ is $(M\otimes_A N)$-regular.
We thus have
\[
  \operatorname{prof}_{B\otimes_A k}(N'\otimes_A k)=\operatorname{prof}_{B\otimes_A k}(N\otimes_A k)-1
\]
and
\[
  \operatorname{prof}_B(M\otimes_A N')=\operatorname{prof}_B(M\otimes_A N)-1;
\]
the induction hypothesis shows that the relation \sref{IV.6.3.1.1} holds for $A$, $B$, $M$, and $N'$, and from what precedes, we deduce \sref{IV.6.3.1.1} for $A$, $B$, $M$, and $N$.
\end{proof}

\begin{corollary}[6.3.2]
\label{IV.6.3.2}
Under the hypotheses of \sref{IV.6.3.1}, suppose in addition that $M\neq 0$ and $N\neq 0$; then we have
\begin{equation}
\label{IV.6.3.2.1}
  \operatorname{coprof}_B(M\otimes_A N)=\operatorname{coprof}_A(M)+\operatorname{coprof}_{B\otimes_A k}(N\otimes_A k).\tag{6.3.2.1}
\end{equation}
\end{corollary}

\begin{proof}
This follows immediately from \sref{IV.6.3.1.1} and \sref{IV.6.1.2} and from the definition of coprofondeur $(\mathbf{0}$, 16.4.9).
\end{proof}

\begin{corollary}[6.3.3]
\label{IV.6.3.3}
Under the hypotheses of \sref{IV.6.3.1}, suppose in addition that $M\neq 0$ and $N\neq 0$; for $M\otimes_A N$ to be a Cohen-Macaulay $B$-module, it is necessary and sufficient that $M$ be a Cohen-Macaulay $A$-module and that $N\otimes_A k$ be a $(B\otimes_A k)$-module de Cohen-Macaulay.
\end{corollary}

\begin{proof}
This follows from the corollary \sref{IV.6.3.2} and the definition of Cohen-Macaulay modules, by the fact that their coprofondeur is zero $(\mathbf{0}$, 16.5.1), taking into account that the condition $N\neq 0$ is equivalent to $N\otimes_A k\neq 0$ by the Nakayama lemma.
\end{proof}

\begin{corollary}[6.3.4]
\label{IV.6.3.4}
Under the hypotheses of \sref{IV.6.3.1}, suppose in addition that $N\otimes_A k$ is a $B$-module of finite length; then we have
\begin{equation}
\label{IV.6.3.4.1}
  \operatorname{prof}_B(M\otimes_A N)=\operatorname{prof}_A(M).\tag{6.3.4.1}
\end{equation}
\oldpage[IV-2]{140}
If in addition $M\neq 0$ and $N\neq 0$, then we have
\begin{equation}
\label{IV.6.3.4.2}
  \operatorname{coprof}_B(M\otimes_A N)=\operatorname{coprof}_A(M).\tag{6.3.4.2}
\end{equation}
\end{corollary}

\begin{proof}
It amounts to saying that $N\otimes_A k$ is a $B$-module of finite length, or a $(B\otimes_A k)$-module of finite length, and we know $(\mathbf{0}$, 16.2.3) that modules of finite length $\neq 0$ are of dimension $0$ and depth $0$.
The corollary \sref{IV.6.3.4} is applicable in particular when $B\otimes_A k$ is a ring of finite length, that is to say when $\mathfrak{m}B$ is an \emph{ideal of definition} of the ring $B$.
\end{proof}

\begin{corollary}[6.3.5]
\label{IV.6.3.5}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a flat morphism.
\begin{enumerate}
  \item[\rm(i)] If, at a point $x\in X$, $\operatorname{coprof}(\sh{O}_x)\leq n$, then, setting $y=f(x)$, $\operatorname{coprof}(\sh{O}_y)\leq n$ and $\operatorname{coprof}(\sh{O}_x\otimes_{\sh{O}_y}k(y))\leq n$; in particular, if $\sh{O}_x$ is a Cohen-Macaulay ring, the same is true of $\sh{O}_y$ and of $\sh{O}_x\otimes_{\sh{O}_y}k(y)$.
  \item[\rm(ii)] Suppose conversely that $\sh{O}_x\otimes_{\sh{O}_y}k(y)$ is a Cohen-Macaulay ring.
  Then, if $\sh{O}_y$ is a Cohen-Macaulay ring, $\operatorname{coprof}(\sh{O}_x)\leq n$ and $\operatorname{coprof}(\sh{O}_y)\leq n$ (resp.\ $\sh{O}_x$ is a Cohen-Macaulay ring).
\end{enumerate}
\end{corollary}

\begin{proof}
It suffices to apply \sref{IV.6.3.2} for $M=A=\sh{O}_y$ and $N=B=\sh{O}_x$.
\end{proof}

\begin{corollary}[6.3.6]
\label{IV.6.3.6}
Let $A$ be a Cohen-Macaulay ring.
Then $A'=A[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ ($\mathrm{T}_i$ indeterminates) is a Cohen-Macaulay ring.
\end{corollary}

\begin{proof}
Indeed, if we set $Y=\Spec(A)$, $X=\Spec(A')$ and $f:X\to Y$ is the canonical morphism, $f$ is \emph{flat} (since $A'$ is a free $A$-module \sref{IV.2.1.2}) and for every $y\in Y$, $k(y)[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ is a regular ring $(\mathbf{0}$, 17.3.7), hence \emph{a fortiori} a Cohen-Macaulay ring; it therefore suffices to apply \sref{IV.6.3.5}.
\end{proof}

\begin{corollary}[6.3.7]
\label{IV.6.3.7}
Every quotient of a Cohen-Macaulay ring is universally catenary.
\end{corollary}

\begin{proof}
This follows from \sref{IV.6.3.6}, \sref{IV.5.6.1}, and $(\mathbf{0}$, 16.5.12).
\end{proof}

\begin{proposition}[6.3.8]
\label{IV.6.3.8}
Let $A$ be a local Noetherian ring; suppose that there exists a Cohen-Macaulay finite type $A$-module $M$ whose support is equal to $\Spec(A)$.
Let $B$ be a quotient ring of $A$, and let $f:\Spec(\widehat{B})\to\Spec(B)$ be the canonical morphism; then, for every $x\in\Spec(B)$, $f^{-1}(x)$ is a Cohen-Macaulay prescheme.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{141}
Since $B$ is a finite type $A$-module, on a $\widehat{B}=B\otimes_A\widehat{A}$ $(\mathbf{0_I}$, 7.3.3); hence $f$ is flat and finite, and the fibres of the two morphisms $\Spec(\widehat{B})\to\Spec(B)$ and $\Spec(\widehat{A})\to\Spec(A)$ at the same point are the same.
One is therefore reduced to proving the proposition when $B=A$.
Let $\mathfrak{p}=\mathfrak{i}_x$ be a prime ideal; since $\mathfrak{p}\in\Supp(M)$, we know $(\mathbf{0}$, 16.5.6 and 16.5.10) that $M_\mathfrak{p}$ is a Cohen-Macaulay $A_\mathfrak{p}$-module.
Taking into account \sref[I]{I.3.6.5}, we see that if we set $A''=A'\otimes_{A_\mathfrak{p}}A_\mathfrak{p}$ and $M''=M'\otimes_{A_\mathfrak{p}}A_\mathfrak{p}$, $M''$ is an $A''$-module de Cohen-Macaulay; on the other hand, $M_\mathfrak{p}$ is by hypothesis a Cohen-Macaulay $A_\mathfrak{p}$-module and $A''_{\mathfrak{p}''}$ is a flat $A_\mathfrak{p}$-module $(\mathbf{0_I}$, 7.3.3 and 6.3.2).
We conclude from \sref{IV.6.3.3} that, if $k$ is the residue field of $A_\mathfrak{p}$, $k\otimes_{A_\mathfrak{p}}A''_{\mathfrak{p}''}$ is a Cohen-Macaulay ring.
But since $A_\mathfrak{p}$ is of dimension $0$, the prime ideals of $A''$ correspond bijectively to those of $k\otimes_{A_\mathfrak{p}}A''$ \sref[I]{I.3.5.7}, and if $\mathfrak{q}''$ is the prime ideal of $k\otimes_{A_\mathfrak{p}}A''$ corresponding to $\mathfrak{p}''$, the local rings $(k\otimes_{A_\mathfrak{p}}A'')_{\mathfrak{q}''}$ and $k\otimes_{A_\mathfrak{p}}A''_{\mathfrak{p}''}$ are isomorphic.
It follows that $k\otimes_{A_\mathfrak{p}}A''_{\mathfrak{p}''}$ is a Cohen-Macaulay ring. Q.E.D.
\end{proof}

\subsection{Flatness and property $(\mathrm{S}_k)$}
\label{subsection:IV.6.4}

\begin{proposition}[6.4.1]
\label{IV.6.4.1}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism, $\sh{E}$ a coherent $\sh{O}_Y$-Module, $\sh{F}$ a coherent $\sh{O}_X$-Module that is $f$-flat.
\begin{enumerate}
  \item[\rm(i)] Let $x\in\Supp(\sh{F})$ be such that $\sh{E}\otimes_Y\sh{F}$ has property $(\mathrm{S}_k)$ at the point $x$; then $\sh{E}$ has property $(\mathrm{S}_k)$ at the point $y=f(x)$.
  \item[\rm(ii)] Suppose that for every $y\in f(\Supp(\sh{F}))$, $\sh{F}_y=\sh{F}\otimes_{\sh{O}_y}k(y)$ has property $(\mathrm{S}_k)$; if for a point $y\in f(\Supp(\sh{F}))$, $\sh{E}$ has property $(\mathrm{S}_k)$ at the point $y$, then $\sh{E}\otimes_Y\sh{F}$ has property $(\mathrm{S}_k)$ at every point of $f^{-1}(y)$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}
  \item[\rm(i)] Setting $y=f(x)$ and let $y'$ be a generisation of $y$; as in the proof of \sref{IV.6.4.1} there exists a generic point $x'$ of an irreducible component of $f^{-1}(y')$ that is a generisation of $x$; hence on a $\dim(\sh{O}_{x'}\otimes_{\sh{O}_{y'}}k(y'))=0$ and by the hypothesis and \sref{IV.6.1.2}, $\dim(\sh{O}_{x'})=\dim(\sh{O}_{y'})$.
  The hypothesis implies, either that $\dim(\sh{O}_{x'})>k$, in which case $\dim(\sh{O}_{y'})>k$, or that $\sh{O}_{x'}$ is a regular ring, and then $\sh{O}_{y'}$ is a regular ring by \sref{IV.6.5.1}.

  \item[\rm(ii)] For every generisation $x'$ of $x\in f^{-1}(y)$, $y'=f(x')$ is a generisation of $y$, we can restrict to verifying that if $x\in\Supp(\sh{F})$ and $f(x)=y$, we have $\operatorname{prof}(\sh{G}_x)\geq\inf(k,\dim(\sh{G}_x))$; it follows from \sref{IV.6.1.2} and \sref{IV.6.3.1} that
  \begin{align*}
    \dim(\sh{G}_x)&=\dim(\sh{E}_y)+\dim(\sh{F}_x\otimes_{\sh{O}_y}k(y))\\
    \operatorname{prof}(\sh{G}_x)&=\operatorname{prof}(\sh{E}_y)+\operatorname{prof}(\sh{F}_x\otimes_{\sh{O}_y}k(y)).
  \end{align*}
  By hypothesis, we have
  \[
    \operatorname{prof}(\sh{E}_y)\geq\inf(k,\dim(\sh{E}_y))
  \]
  \[
    \operatorname{prof}(\sh{F}_x\otimes_{\sh{O}_y}k(y))\geq\inf(k,\dim(\sh{F}_x\otimes_{\sh{O}_y}k(y)))
  \]
  whence, by adding member by member,
  \begin{multline*}
    \operatorname{prof}(\sh{G}_x)\geq\inf(k,\dim(\sh{E}_y))+\inf(k,\dim(\sh{F}_x\otimes_{\sh{O}_y}k(y)))\geq{}\\
    {}\geq\inf(k,\dim(\sh{E}_y)+\dim(\sh{F}_x\otimes_{\sh{O}_y}k(y)))=\inf(k,\dim(\sh{G}_x))
  \end{multline*}
  which proves (ii).
\end{enumerate}
\end{proof}

\oldpage[IV-2]{142}

\begin{corollary}[6.4.2]
\label{IV.6.4.2}
Let $X$, $Y$ be two locally Noetherian preschemes, $\sh{E}$ a coherent $\sh{O}_Y$-Module, $f:X\to Y$ a flat morphism.
Suppose that for every $y\in Y$, the prescheme $f^{-1}(y)$ verifies $(\mathrm{S}_k)$ at every point $x$; then, for $f^*(\sh{E})$ to verify $(\mathrm{S}_k)$ at a point $x$, it is necessary and sufficient that $\sh{E}$ verify $(\mathrm{S}_k)$ at the point $f(x)$.
\end{corollary}

\begin{remarks}[6.4.3]
\label{IV.6.4.3}
Let $A$, $B$ be two local Noetherian rings, $k$ the residue field of $A$, $\varphi:A\to B$ a local homomorphism, $M$ a finite type $B$-module that is a flat $A$-module, and suppose in addition that the $A$-module $A$ and the $(B\otimes_A k)$-module $M\otimes_A k$ both have property $(\mathrm{S}_n)$; can one then conclude that the $B$-module $M$ has property $(\mathrm{S}_n)$?
We do not know the answer even when $n=1$, $M=B$ and $B=\widehat{A}$, in other words, we do not know whether in general the \emph{completion} of a local Noetherian ring satisfying $(\mathrm{S}_n)$ also satisfies $(\mathrm{S}_n)$, even for $n=1$.
Setting $Y=\Spec(A)$, $X=\Spec(B)$, and $\sh{F}=\widetilde{M}$, it would suffice, by \sref{IV.6.4.1}, (ii), to show that for \emph{every} $y\in Y$, $\sh{F}_y$ has property $(\mathrm{S}_k)$ (and not only when $y$ is the closed point of $Y$); it would also suffice to show that the set $U$ of $x\in X$ such that $\sh{F}_x\otimes_{\sh{O}_{f(x)}}k(f(x))$ satisfies $(\mathrm{S}_k)$ is \emph{open} in $X$ (since it contains the closed point of $X$ by hypothesis).
We shall show later \sref{IV.12.1.4} that $U$ is open when $B$ is a local ring of a finite type $A$-algebra, and that all the preschemes $f^{-1}(y)$ are Cohen-Macaulay preschemes (otherwise said satisfy $(\mathrm{S}_n)$ for \emph{all} $n$) when the hypotheses of \sref{IV.6.3.8} are satisfied.
On the other hand, for $B=\widehat{A}$ and $M=B$, we have seen in \sref{IV.6.3.8} that the preschemes $f^{-1}(y)$ are Cohen-Macaulay preschemes for all $n$, for which it suffices that $A$ satisfies $(\mathrm{S}_n)$, \emph{it is necessary and sufficient that its completion $\widehat{A}$ satisfies} $(\mathrm{S}_n)$.
It remains to see whether this property holds without restriction on $A$.
\end{remarks}

\subsection{Flatness and property $(\mathrm{R}_k)$}
\label{subsection:IV.6.5}

\oldpage[IV-2]{143}

\begin{proposition}[6.5.1]
\label{IV.6.5.1}
Let $A$, $B$ be two local Noetherian rings, $k$ the residue field of $A$, $\varphi:A\to B$ a local homomorphism, for which $B$ is a flat $A$-module.
Then:
\begin{enumerate}
  \item[\rm(i)] If $B$ is regular, $A$ is regular.
  \item[\rm(ii)] If $A$ and $B\otimes_A k$ are regular, $B$ is regular.
\end{enumerate}
This proposition is given for the record, having already been proved in $(\mathbf{0}$, 17.3.3).
\end{proposition}

\begin{corollary}[6.5.2]
\label{IV.6.5.2}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a flat morphism.
\begin{enumerate}
  \item[\rm(i)] If $X$ is regular at a point $x$, $Y$ is regular at the point $f(x)$.
  \item[\rm(ii)] If for some $y\in f(X)$, $Y$ is regular at the point $y$ and if $f^{-1}(y)$ is a regular prescheme at a point $x$, then $X$ is regular at the point $x$.
\end{enumerate}
\end{corollary}

\begin{proposition}[6.5.3]
\label{IV.6.5.3}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a flat morphism.
\begin{enumerate}
  \item[\rm(i)] If $X$ has property $(\mathrm{R}_k)$ at a point $x$ (resp.\ if $X$ has property $(\mathrm{R}_k)$), $Y$ has property $(\mathrm{R}_k)$ at the point $f(x)$ (resp.\ $Y$ has property $(\mathrm{R}_k)$ at every point of $f(X)$).
  \item[\rm(ii)] Suppose that for every $y\in f(X)$, the prescheme $f^{-1}(y)$ has property $(\mathrm{R}_k)$; then, if for a point $y\in f(X)$, $Y$ has property $(\mathrm{R}_k)$ at the point $y$, $X$ has property $(\mathrm{R}_k)$ at every point of $f^{-1}(y)$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}
  \item[\rm(i)] Set $y=f(x)$ and let $y'$ be a generisation of $y$; as in the proof of \sref{IV.6.4.1}, there exists a generic point $x'$ of an irreducible component of $f^{-1}(y')$ that is a generisation of $x$; hence $\dim(\sh{O}_{x'}\otimes_{\sh{O}_{y'}}k(y'))=0$ and by the hypothesis and \sref{IV.6.1.2}, $\dim(\sh{O}_{x'})=\dim(\sh{O}_{y'})$.
  The hypothesis implies, either that $\dim(\sh{O}_{x'})>k$, in which case $\dim(\sh{O}_{y'})>k$, or that $\sh{O}_{x'}$ is a regular ring, and then $\sh{O}_{y'}$ is a regular ring by \sref{IV.6.5.1}.

  \item[\rm(ii)] For every generisation $x'$ of $x\in f^{-1}(y)$, $y'=f(x')$ is a generisation of $y$; we can restrict to verifying that if $\dim(\sh{O}_x)\leq k$, then $\sh{O}_x$ is a regular ring.
  Now, by \sref{IV.6.1.2}
  \[
    \dim(\sh{O}_x)=\dim(\sh{O}_y)+\dim(\sh{O}_x\otimes_{\sh{O}_y}k(y))
  \]
  hence if $\dim(\sh{O}_x)\leq k$, we have \emph{a fortiori} $\dim(\sh{O}_y)\leq k$ and $\dim(\sh{O}_x\otimes_{\sh{O}_y}k(y))\leq k$ and the hypothesis implies that $\sh{O}_y$ and $\sh{O}_x\otimes_{\sh{O}_y}k(y)$ are regular rings.
  We then conclude from \sref{IV.6.5.1} that $\sh{O}_x$ is a regular ring.
\end{enumerate}
\end{proof}

\begin{corollary}[6.5.4]
\label{IV.6.5.4}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a flat morphism.
\begin{enumerate}
  \item[\rm(i)] If $X$ is normal at a point $x$, $Y$ is normal at the point $f(x)$.
  \item[\rm(ii)] If, for every $y\in f(X)$, $f^{-1}(y)$ is a normal prescheme and if $Y$ is normal at a point $y\in f(X)$, then $X$ is normal at every point of $f^{-1}(y)$.
\end{enumerate}
\end{corollary}

\begin{proof}
This follows immediately from the Serre normality criterion \sref{IV.5.8.6} and from \sref{IV.6.4.1} applied for $k=2$ and \sref{IV.6.5.3} applied for $k=1$.
\end{proof}

\oldpage[IV-2]{144}

\begin{remarks}[6.5.5]
\label{IV.6.5.5}
\begin{enumerate}
  \item[\rm(i)] Let $A$, $B$ be two local Noetherian rings, $\varphi:A\to B$ a local homomorphism such that $B$ is a flat $A$-module.
  Let $k$ be the residue field of $A$, and suppose that the two rings $A$ and $B\otimes_A k$ satisfy property $(\mathrm{R}_i)$ \sref{IV.5.8.2}; it does \emph{not} then necessarily follow that $B$ satisfies $(\mathrm{R}_i)$, even in the particular case where $i=0$ or $i=1$ and where $B$ is the completion $\widehat{A}$ of $A$.
  Nagata has indeed given an example where $A$ is normal (hence satisfies $(\mathrm{R}_1)$) but where $\widehat{A}$ is not even reduced (hence does not satisfy $(\mathrm{R}_0)$) \emph{[30]}.
  One cannot apply the proposition \sref{IV.6.5.3} in this case because the fibres $f^{-1}(y)$ do not necessarily satisfy property $(\mathrm{R}_i)$ at the points distinct from the closed point of $Y=\Spec(A)$.
  We shall show however later \sref{IV.7.8.3}, (v) that such phenomena do not occur for the local Noetherian rings most commonly encountered in applications.

  \item[\rm(ii)] The property for a prescheme of being \emph{integral} does not at all behave like the properties we have just examined in this number and the preceding ones: it can happen that $f:X\to Y$ is a flat morphism of finite type, that all the fibres $f^{-1}(y)$ are regular (and even geometrically regular \sref{IV.6.7.6}) and that $Y$ is integral, without $X$ even being locally integral.
  For example, let $k$ be an algebraically closed field of characteristic $0$, and let $A$ be the integral ring $k[\mathrm{U},\mathrm{V}]/(\mathrm{UV}-(\mathrm{U}+\mathrm{V})^3)$ (whose spectrum is therefore a ``cubic with a double point''); $A$ is not integrally closed, and if $u$, $v$ are the canonical images of $\mathrm{U}$, $\mathrm{V}$ in $A$, the integral closure of $A$ is $C=A[t]$, where $t=(u-v)/(u+v)$, which satisfies the equation $t^2=1+u+v$, whence $u=\frac{1}{2}(t^3+t^2-t)$, $v=\frac{1}{2}(-t^3+t^2+t)$ and hence $C=k[t]$, isomorphic to the polynomial ring in one indeterminate over $k$.
  If $\mathfrak{m}=Au+Av$, the maximal ideal of $A$ (corresponding to the ``double point'' of the cubic), there are two maximal ideals $\mathfrak{n}_1$, $\mathfrak{n}_2$ of $C$, generated by $t-1$ and $t+1$ respectively.
  Then $B$ is the subring of the product $C\times C$ consisting of the pairs of polynomials $(f,g)$ such that $f(1)=g(-1)$ and $f(-1)=g(1)$ ($\Spec(B)$ is the scheme obtained by ``gluing'' two copies of $\Spec(C)$, the point $\mathfrak{n}_1$ (resp.\ $\mathfrak{n}_2$) of one being ``glued'' to the point $\mathfrak{n}_2$ (resp.\ $\mathfrak{n}_1$) of the other; cf.\ chap.~V, where this operation will be discussed in a general fashion).
  There are thus two maximal ideals $\mathfrak{r}_1$, $\mathfrak{r}_2$ of $B$ above the maximal ideal $\mathfrak{m}$ of $A$.
  Moreover, since the ``gluing'' process commutes with localisation and completion, one easily verifies that the canonical homomorphisms $\widehat{A}_\mathfrak{m}\to\widehat{B}_{\mathfrak{r}_1}$ and $\widehat{A}_\mathfrak{m}\to\widehat{B}_{\mathfrak{r}_2}$ are bijective, and hence $B_{\mathfrak{r}_1}$ and $B_{\mathfrak{r}_2}$ are flat $A_\mathfrak{m}$-modules (Bourbaki, \emph{Alg.~comm.}, chap.~III, \textsection3, n\textsuperscript{o}~5, prop.~10) having moreover the same residue field as $A_\mathfrak{m}$.
  For every other maximal ideal $\mathfrak{p}$ of $A$, it is immediate that there are two maximal ideals $\mathfrak{q}_1$, $\mathfrak{q}_2$ of $B$ above $\mathfrak{p}$, and that the homomorphisms $A_\mathfrak{p}\to B_{\mathfrak{q}_1}$ and $A_\mathfrak{p}\to B_{\mathfrak{q}_2}$ are bijective.
  One thus sees that the morphism $\Spec(B)\to\Spec(A)$ is flat and finite, and that all its fibres are geometrically regular (it is even \emph{\'etale}, as we shall see later \sref{IV.17.6.3}); however it is immediate that $B$ is not integral.
\end{enumerate}
\end{remarks}

\subsection{Transitivity properties}
\label{subsection:IV.6.6}

\oldpage[IV-2]{145}

\begin{proposition}[6.6.1]
\label{IV.6.6.1}
For a locally Noetherian prescheme $Z$, let us denote by $\mathbf{P}(Z)$ any one of the following properties:
\begin{enumerate}
  \item[\rm a)] $Z$ is a Cohen-Macaulay prescheme.
  \item[\rm b)] $Z$ satisfies $(\mathrm{S}_n)$.
  \item[\rm c)] $Z$ is regular.
  \item[\rm d)] $Z$ satisfies $(\mathrm{R}_n)$.
  \item[\rm e)] $Z$ is normal.
  \item[\rm f)] $Z$ is reduced.
\end{enumerate}
Let $X$, $Y$, $Z$ be three locally Noetherian preschemes, $f:X\to Y$, $g:Y\to Z$ two morphisms.
\begin{enumerate}
  \item[\rm(i)] Suppose that $f$ and $g$ are flat and that for every $y\in f(X)$ (resp.\ every $z\in g(Y)$), the prescheme $f^{-1}(y)$ (resp.\ $g^{-1}(z)$) satisfies $\mathbf{P}$.
  Then $h=g\circ f$ is flat and for every $z\in h(X)$, $h^{-1}(z)$ satisfies $\mathbf{P}$.
  \item[\rm(ii)] Suppose the following conditions are satisfied: $f$ is faithfully flat, $h=g\circ f$ is flat, for every $z\in h(X)$, the prescheme $f^{-1}(z)$ satisfies $\mathbf{P}$, and for every $z\in g(Y)$, the prescheme $h^{-1}(z)$ satisfies $\mathbf{P}$.
  Then $g$ is flat and for every $z\in g(Y)$, $g^{-1}(z)$ satisfies $\mathbf{P}$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}
  \item[\rm(i)] We already know that if $f$ and $g$ are flat, so is $u=g\circ f$, and that if $f$ is flat (resp.\ faithfully flat, $g$ is flat \sref{IV.2.1.6} and \sref{IV.2.2.13}).
  Furthermore, for every $z\in Z$, $f_z=f\otimes 1_{k(z)}:h^{-1}(z)\to g^{-1}(z)$ is flat \sref{IV.2.1.4}, and for every $y\in g^{-1}(z)$, $f_z^{-1}(y)$ is isomorphic to $f^{-1}(y)$ \sref[I]{I.3.6.4}.
  The conclusion follows respectively from \sref{IV.6.3.5}, (ii), \sref{IV.6.4.2}, \sref{IV.6.5.2}, (ii), \sref{IV.6.5.3}, (ii), \sref{IV.6.5.4}, (ii), and \sref{IV.3.3.5}, (ii).

  \item[\rm(ii)] We already know that $g$ is flat \sref{IV.2.2.13}; moreover, for every $z\in Z$, $f_z$ is faithfully flat \sref{IV.2.2.13}.
  The conclusion follows respectively from \sref{IV.6.3.5}, (i), \sref{IV.6.4.2}, (i), \sref{IV.6.5.2}, (i), \sref{IV.6.5.3}, (i), \sref{IV.6.5.4}, (i), and \sref{IV.2.1.13}.
\end{enumerate}
\end{proof}

\begin{remark}[6.6.2]
\label{IV.6.6.2}
Suppose $h$ flat, $f$ faithfully flat, and suppose that for every $z\in Z$, $h^{-1}(z)$ is of \emph{coprofondeur} $\leq n$ \sref{IV.5.7.1}; then it follows from the reasoning of \sref{IV.6.6.1}, (ii), and from \sref{IV.6.3.2} that $g^{-1}(z)$ is of coprofondeur $\leq n$ for every $z\in g(Y)$ and that for every $y\in Y$, $f^{-1}(y)$ is of coprofondeur $\leq n$.
\end{remark}

\subsection{Application to base changes in algebraic preschemes}
\label{subsection:IV.6.7}

\begin{proposition}[6.7.1]
\label{IV.6.7.1}
Let $k$ be a field, $X$ a locally Noetherian $k$-prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module.
Let $k'$ be an extension of $k$; set $X'=X\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$ and let $p:X'\to X$ be the canonical projection.
Suppose, either that $X$ is locally of finite type over $k$, or that $k'$ is a finite extension of $k$, so that in both cases $X'$ is locally Noetherian.
Let $x'$ be a point of $X'$, $x=p(x')$.
Then:
\begin{enumerate}
  \item[\rm(i)] We have $\operatorname{coprof}(\sh{F}_x)=\operatorname{coprof}(\sh{F}'_{x'})$; in particular, for $\sh{F}_x$ to be a Cohen-Macaulay $\sh{O}_x$-module, it is necessary and sufficient that $\sh{F}'_{x'}$ be a Cohen-Macaulay $\sh{O}_{x'}$-module.
  \item[\rm(ii)] For $\sh{F}$ to satisfy property $(\mathrm{S}_n)$ at the point $x$, it is necessary and sufficient that $\sh{F}'$ satisfy $(\mathrm{S}_n)$ at the point $x'$.
\end{enumerate}
\end{proposition}

\begin{proof}
We know that $p$ is faithfully flat \sref{IV.2.2.13}; the two assertions are therefore consequences respectively of \sref{IV.6.3.2} and \sref{IV.6.4.2}, provided that we prove that $p^{-1}(x)=\Spec(k(x)\otimes_k k')$ is a Cohen-Macaulay prescheme; since one of the two fields $k(x)$, $k'$ is a finite extension of $k$ by hypothesis, we are reduced to proving the following:
\end{proof}

\oldpage[IV-2]{146}

\begin{lemma}[6.7.1.1]
\label{IV.6.7.1.1}
Let $K$, $L$ be two extensions of a field $k$, one of which is of finite type (so that $K\otimes_k L$ is a Noetherian ring).
Then $K\otimes_k L$ is a Cohen-Macaulay ring.
\end{lemma}

\begin{proof}
Suppose for example that $L$ is a finite type extension of $k$, so that $L$ is a finite extension of a pure transcendental extension $k(\mathbf{t})$ of $k$ ($\mathbf{t}$ being a finite system $(t_i)_{1\leq i\leq m}$ of indeterminates).
Then $A=K\otimes_k k(\mathbf{t})$, $B=K\otimes_k L$; $B$ is a finite type $A$-module, and $B$ is an $A$-module flat $(\mathbf{0_I}$, 6.2.1) and of finite type; the morphism $h:\Spec(B)\to\Spec(A)$ is therefore finite, and since every Artinian prescheme is Cohen-Macaulay \sref{IV.6.3.5}, or if $S$ is the set of elements $\neq 0$ of $k[\mathbf{t}]$, we have $A'=S^{-1}A'$, where $A'=K\otimes_k k[\mathbf{t}]=K[\mathbf{t}]$; by $(\mathbf{0}$, 16.5.13), it suffices to prove that $A'$ is a Cohen-Macaulay ring, and this follows from the fact that $A'$ is regular $(\mathbf{0}$, 17.3.7).
\end{proof}

\begin{corollary}[6.7.2]
\label{IV.6.7.2}
For $\sh{O}_x$ to be a Cohen-Macaulay ring (resp.\ for $X$ to satisfy $(\mathrm{S}_n)$ at the point $x$), it is necessary and sufficient that $\sh{O}_{x'}$ be a Cohen-Macaulay ring (resp.\ that $X'$ satisfy $(\mathrm{S}_n)$ at the point $x'$).
\end{corollary}

\begin{corollary}[6.7.3]
\label{IV.6.7.3}
Let $X$, $Y$ be two $k$-preschemes locally Noetherian, at least one of which is locally of finite type over $k$.
Let $\sh{F}$ (resp.\ $\sh{G}$) be a coherent $\sh{O}_Y$-Module (resp.\ a coherent $\sh{O}_X$-Module).
If $\sh{F}$ and $\sh{G}$ satisfy property $(\mathrm{S}_n)$, then so does $\sh{F}\otimes_k\sh{G}$.
\end{corollary}

\begin{proof}
The hypothesis implies that $X\times_k Y$ is locally Noetherian; let $p:X\times_k Y\to X$ and $q:X\times_k Y\to Y$ be the canonical projections, which are flat morphisms.
Suppose for example that $X$ is locally of finite type over $k$; on can write $\sh{F}\otimes_k\sh{G}=p^*(\sh{F})\otimes_k\sh{G}$, and since $\sh{F}$ is flat relative to the structural morphism $X\to\Spec(k)$, $p^*(\sh{F})$ is $q$-flat \sref{IV.2.1.4}; it suffices to see that for every $y\in Y$, $p^*(\sh{F})\otimes_{\sh{O}_Y}k(y)$ has property $(\mathrm{S}_n)$; but $p^*(\sh{F})\otimes_{\sh{O}_Y}k(y)=\sh{F}\otimes_k k(y)$, and since $X$ is locally of finite type over $k$, the conclusion follows from \sref{IV.6.7.1}, (ii).
\end{proof}

\begin{proposition}[6.7.4]
\label{IV.6.7.4}
For a locally Noetherian prescheme $Z$, and a point $z\in Z$, let $\mathbf{P}(Z,z)$ be one of the following properties:
\begin{enumerate}
  \item[\rm a)] $Z$ is regular at the point $z$.
  \item[\rm b)] $Z$ satisfies $(\mathrm{R}_n)$ at the point $z$.
  \item[\rm c)] $Z$ is normal at the point $z$.
  \item[\rm d)] $Z$ is reduced at the point $z$.
\end{enumerate}
This being so, under the hypotheses and with the notations of \sref{IV.6.7.1}, if $\mathbf{P}(X',x')$ is true, then so is $\mathbf{P}(X,x)$; the converse is true if $k'$ is a separable extension of $k$.
\end{proposition}

\begin{proof}
The case where $\mathbf{P}$ is property $d)$ has already been treated for the record, having already been treated \sref{IV.2.1.13} and \sref{IV.4.6.1}.
The same reasoning as at the beginning of the proof of \sref{IV.6.7.1} shows that the first assertion follows respectively from \sref{IV.6.5.2}, (i), \sref{IV.6.5.3}, (i), \sref{IV.6.5.4}, (i); of the same kind, the second assertion follows from \sref{IV.6.5.2}, (ii), \sref{IV.6.5.3}, (ii), \sref{IV.6.5.4}, (ii).
\end{proof}

\begin{lemma}[6.7.4.1]
\label{IV.6.7.4.1}
Let $K$, $L$ be two extensions of a field $k$.
If $L$ is a separable extension of $k$, then $K\otimes_k L$ is a regular ring.
\end{lemma}

\begin{proof}
We distinguish two cases:

A) $L$ is a \emph{finite type} extension of $k$.
Then, with the notations of \sref{IV.6.7.1.1}, we can suppose that $L$ is a \emph{finite separable} extension of $k(\mathbf{t})$.
For every $s\in\Spec(A)$, $k(s)\otimes_{k(t)} L$ is composed of a direct sum of a finite number of fields, finite separable extensions of $k(s)$, hence a regular ring; it follows from \sref{IV.6.5.2}, (ii) that it suffices to prove that $A$ is a regular ring; as $A=S^{-1}A'$, it suffices to prove that $A'$ is regular $(\mathbf{0}$, 17.3.6), but one has seen that this was indeed the case in the proof of \sref{IV.6.7.1.1}.

B) Let $(L_\lambda)$ be the filtered family of sub-extensions of $L$ that are of finite type over $k$; by A), each of the rings $C_\lambda=K\otimes_k L_\lambda$ is regular; on the other hand, for $L_\lambda\subset L_\mu$, $L_\mu$ is an $L_\lambda$-module plat, hence $C_\mu$ is a $C_\lambda$-module plat $(\mathbf{0_I}$, 6.2.1).
Since $C=K\otimes_k L=\varinjlim(K\otimes_k L_\lambda)$ is Noetherian, we can apply \sref{IV.5.13.7}, and $C$ is therefore regular.
\end{proof}

\begin{remarks}[6.7.5]
\label{IV.6.7.5}
\begin{enumerate}
  \item[\rm(i)] In the proof, the fact that $\mathbf{P}(X,x)$ holds implies $\mathbf{P}(X',x')$, without the hypothesis that $k'$ is a \emph{separable} extension of $k$ \sref{IV.4.3.1}; but even when $X$ is regular it can happen that $X'$ is not regular without $k'$ separable \sref{IV.6.7.6}.
  We have already seen \sref{IV.4.6.1} that when $\mathbf{P}$ is property $d)$, it can happen that $X$ is regular without $X'$ being regular; it is interesting to give an example of this in which $k$ is \emph{algebraically closed} in the field of rational functions $K$ of $X$.
  \item[\rm(ii)] As recalled in (i), it follows from \sref{IV.4.6.1} that when $X$ is a prescheme that is algebraic over $k$ which is not geometrically reduced, there are finite radiciel extensions $k'$ of $k$ such that $X'=X\otimes_k k'$ is not reduced.
\end{enumerate}
\end{remarks}

\begin{definition}[6.7.6]
\label{IV.6.7.6}
Let $k$ be a field, $X$ a $k$-prescheme locally Noetherian.
We say that $X$ is \emph{geometrically regular} at a point $x$ (resp.\ has property $(\mathrm{R}_n)$ geometric, resp.\ is \emph{geometrically normal} at a point $x$) if, for every finite extension $k'$ of $k$, $X'=X_{(k')}=X\otimes_k k'$ is regular (resp.\ has property $(\mathrm{R}_n)$, resp.\ is normal) at every point $x'$ whose projection in $X$ is $x$.
We say that $X$ is \emph{geometrically regular} (or equivalently \emph{geometrically regular in codimension $\leq n$}), resp.\ \emph{geometrically normal}) if $X$ is geometrically regular (resp.\ has the geometric property $(\mathrm{R}_n)$, resp.\ is geometrically normal) at every point.

We say that an algebra $A$ over $k$ is a \emph{geometrically regular} (resp.\ \emph{geometrically normal}, resp.\ \emph{geometrically reduced}, resp.\ \emph{having geometric property} $(\mathrm{R}_n)$) ring if $\Spec(A)$ has the same property.

If $X=\Spec(K)$, where $K$ is an extension of $k$, it amounts to the same thing to say that $X$ is geometrically regular, or geometrically normal, or geometrically reduced, or that $K$ is a \emph{separable} extension of $k$: this follows from \sref{IV.4.6.1} and from the fact that if $K$ is separable from $k$ and $k'$ a finite extension, $K\otimes_k k'$ is composed of a direct sum of a finite number of fields (Bourbaki, \emph{Alg.}, chap.~VIII, \textsection7, n\textsuperscript{o}~3, cor.~1 of th.~1).
\end{definition}

\begin{proposition}[6.7.7]
\label{IV.6.7.7}
Let $k$ be a field, $X$ a locally Noetherian $k$-prescheme, $x$ a point of $X$; let us denote by $\mathbf{Q}(k')$ the relation ``$k'$ is an extension of $k$, and $\mathbf{P}(X_{(k')},x')$ is true for every point $x'$ of $X_{(k')}$ whose projection is $x$'', where $\mathbf{P}(Z,z)$ is one of the properties a), b), c) of \sref{IV.6.7.4}.
Then the following properties are equivalent:
\begin{enumerate}
  \item[\rm a)] $\mathbf{Q}(k')$ is true for every finite extension $k'$ of $k$.
  \item[\rm b)] $\mathbf{Q}(k')$ is true for every finite radiciel extension $k'$ of $k$.
  \item[\rm c)] $\mathbf{Q}(k')$ is true for every finite type extension $k'$ of $k$.
\end{enumerate}
Suppose in addition that $X$ is locally of finite type over $k$; then the three preceding properties are also equivalent to the following:
\begin{enumerate}
  \item[\rm d)] $\mathbf{Q}(k')$ is true for every extension $k'$ of $k$.
  \item[\rm e)] $\mathbf{Q}(k')$ is true for every perfect extension $k'$ of $k$.
  \item[\rm f)] $\mathbf{Q}(k')$ is true for every extension $k'=k^{p^{-s}}$, where $p$ is the characteristic exponent of $k$, and $s>0$.
\end{enumerate}
\end{proposition}

\begin{proof}
To prove the equivalence of a), b), and c), it clearly suffices to establish that b) implies c).
Let $K$ be a finite type extension of $k$, which is therefore the field of fractions of a $k$-algebra of finite type $A$; set $Y=\Spec(A)$.
We know \sref{IV.4.6.6} that there exists a finite radiciel extension $k'$ of $k$ such that $Y'=(Y\otimes_k k')_\mathrm{red}$ is separable over $k'$; if $\eta'$ is the generic point of $Y'$, $K'=k(\eta')$ is a separable extension of $k'$ by \sref{IV.4.6.1}.
Since, by hypothesis, $\mathbf{P}(X_{(k')},x')$ is true for every point $x'$ of $X_{(k')}$ above $x$, it follows from \sref{IV.6.7.4} that $\mathbf{P}(X_{(K')},x'')$ is true for every point $x''$ of $X_{(K')}$ above $x$, since $X_{(K')}=(X_{(k')})_{(K')}$, and $x''$ is above a point $x'$ of $X_{(k')}$ above $x$, and $K'$ is separable over $k'$.
But we also have $X_{(K')}=(X_{(K)})_{(K')}$ and for every $x_0\in X_{(K)}$ above $x$, there exists $x''\in X_{(K')}$ above $x_0$; it follows from \sref{IV.6.7.4} that $\mathbf{P}(X_{(K)},x_0)$ is true.

Suppose now $X$ locally of finite type over $k$, so that for \emph{every} extension $K$ of $k$, $X_{(K)}$ is locally Noetherian.
Since a radiciel extension of $k$ is isomorphic to a sub-extension of a perfect extension of $k$, it follows immediately from \sref{IV.6.7.4} that $e)$ implies $f)$; similarly, $f)$ implies $b)$; $d)$ implies trivially $e)$, and finally $e)$ implies $d)$.
We therefore need to prove that $b)$ implies $e)$.
We take $k'=k^{p^{-\infty}}$ in $e)$.
Since the question is local on $X$, one can moreover restrict to the case where $X$ is affine and of finite type over $k$, by replacing $X$ by a neighbourhood of $x$; let thus $X=\Spec(B)$, where $B$ is a $k$-algebra of finite type; moreover, by definition of property $\mathbf{P}$ in all cases considered, one can also replace $X$ by the local scheme of $X$ at the point $x$, hence suppose that $B$ is a \emph{local} Noetherian ring.
Set $B'=B\otimes_k k'$; $k'$ is the inductive limit of its \emph{finite} sub-extensions $k'_\lambda$ and if we set $B'_\lambda=B\otimes_k k'_\lambda$, the morphism $f_\lambda:\Spec(B')\to\Spec(B'_\lambda)$ is a \emph{homeomorphism} of $\Spec(B')$ onto $\Spec(B'_\lambda)$ for every $\lambda$;
\oldpage[IV-2]{150}
indeed $f_\lambda$ is bijective by \sref[I]{I.3.5.2}, \sref[I]{I.3.5.7}, and \sref[I]{I.3.5.8}; on the other hand, it is closed by \sref[II]{II.6.1.10}.
One can now apply \sref{IV.5.13.6} which proves (by hypothesis $b)$) that $b)$ implies $e)$ when $\mathbf{P}(Z,z)$ is property $(\mathrm{R}_n)$ at the point $z$.
The case where $\mathbf{P}(Z,z)$ is the property of being regular (i.e.\ of having property $(\mathrm{R}_n)$ for all $n$) follows trivially.
Finally, taking into account the Serre normality criterion \sref{IV.5.8.6}, $b)$ implies $e)$ when $\mathbf{P}(Z,z)$ is the property of being normal at the point $z$, since being normal at $z$ is equivalent to having both properties $(\mathrm{S}_2)$ and $(\mathrm{R}_1)$: one then applies what precedes for $n=1$, and \sref{IV.6.7.2}, (ii) for $n=2$.
\end{proof}

\begin{corollary}[6.7.8]
\label{IV.6.7.8}
Let $k$ be a field; for a $k$-prescheme locally Noetherian $X$ and a point $x\in X$, let us denote by $\mathbf{P}(X,x)$ one of the following properties:
\begin{enumerate}
  \item[\rm(i)] $\operatorname{coprof}(\sh{O}_x)\leq n$.
  \item[\rm(ii)] $\sh{O}_x$ is a Cohen-Macaulay ring.
  \item[\rm(iii)] $X$ satisfies property $(\mathrm{S}_n)$ at the point $x$.
  \item[\rm(iv)] $X$ is geometrically regular at the point $x$.
  \item[\rm(v)] $X$ has property $(\mathrm{R}_n)$ geometric at the point $x$.
  \item[\rm(vi)] $X$ is geometrically normal at the point $x$.
  \item[\rm(vii)] $X$ is geometrically reduced (i.e.\ \emph{separable}) at the point $x$.
\end{enumerate}
Let $k'$ be an extension of $k$; suppose, either that $k'$ is of finite type, or that $X$ is locally of finite type over $k$, so that $X'=X\otimes_k k'$ is locally Noetherian.
Let $x'$ be a point of $X'$ whose projection in $X$ is $x$.
Then, for $\mathbf{P}(X,x)$ to hold, it is necessary and sufficient that $\mathbf{P}(X',x')$ hold.
\end{corollary}

\begin{proof}
This has already been seen for property (vii) \sref{IV.4.6.11}, and for properties (i), (ii), and (iii) \sref{IV.6.7.1}.
For (iv), (v), and (vi), this follows from \sref{IV.6.7.7}: indeed, the fact that the condition is necessary follows from the equivalence of the criteria $a)$ and $c)$, and from the equivalence of $c)$ and $d)$ when $X$ is locally of finite type over $k$.
To see that the condition is sufficient, either $k''$ is a finite radiciel extension of $k$; one can always consider $k'$ and $k''$ as sub-extensions of an extension $K$ of $k$; set $X''=X\otimes_k k''$, $X_0=X\otimes_k K$, and note that since $k''$ is a radiciel extension of $k$, there is only \emph{one} point $x''$ of $X''$ above $x$ \sref[I]{I.3.5.7} and \sref[I]{I.3.5.8}.
Let $x_0$ be any point of $X_0$ above $x$; if $\mathbf{P}(X',x')$ is true, then so is $\mathbf{P}(X_0,x_0)$ by \sref{IV.6.7.7}, $c)$ and $d)$) (since $k'$ is a finite type extension of $k$ and $K$ is an extension of $k'$, one can suppose that $k'$ is of finite type, and if $X$ is locally of finite type over $k$, $X'$ is locally of finite type over $k'$).
We deduce from \sref{IV.6.7.4} that the property $\mathbf{Q}(k'')$ is true (with the notations of \sref{IV.6.7.7}), hence $\mathbf{P}(X,x)$ is true by \sref{IV.6.7.7}, $b)$).
\end{proof}

\subsection{Regular, normal, reduced, smooth morphisms}
\label{subsection:IV.6.8}

\begin{definition}[6.8.1]
\label{IV.6.8.1}
Let $X$, $Y$ be two preschemes, $f:X\to Y$ a morphism such that the fibre $f^{-1}(y)$ is a locally Noetherian prescheme for every $y\in Y$, $x$ a point of $X$.
We say respectively that $f$ is a morphism:
\begin{enumerate}
  \item[\rm(i)] of coprofondeur $\leq n$ at the point $x$;
  \item[\rm(ii)] of Cohen-Macaulay at the point $x$;
  \item[\rm(iii)] $(\mathrm{S}_n)$ at the point $x$;
  \item[\rm(iv)] regular at the point $x$;
  \item[\rm(v)] $(\mathrm{R}_n)$ at the point $x$;
  \item[\rm(vi)] normal at the point $x$;
  \item[\rm(vii)] reduced at the point $x$;
\end{enumerate}
if $f$ is flat at the point $x$, and if in addition the corresponding property $\mathbf{P}(f^{-1}(f(x)),x)$ (notation of \sref{IV.6.7.8}) is true.

We say that $f$ is \emph{smooth} at the point $x$ if it is locally of finite presentation in a neighbourhood of $x$ in $X$, and if it is regular at the point $x$.
We say that $f$ is a morphism of coprofondeur $\leq n$, Cohen-Macaulay, $(\mathrm{S}_n)$, regular, $(\mathrm{R}_n)$, normal, reduced, smooth, if it has the corresponding property at every point of $X$.
\end{definition}

\oldpage[IV-2]{151}

\begin{proposition}[6.8.2]
\label{IV.6.8.2}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism, $x$ a point of $X$.
Let us denote by $\mathbf{M}(f,x)$ one of the properties (i) to (vii) of the definition \sref{IV.6.8.1}, or the property for $f$ to be smooth at the point $x$.
Let $Y'$ be a locally Noetherian prescheme, $g:Y'\to Y$ a morphism, $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$.
Suppose that $f$ or $g$ is locally of finite type.
Then, for every $x'\in X'$ above $x$, the property $\mathbf{M}(f,x)$ implies $\mathbf{M}(f',x')$.
\end{proposition}

\begin{proof}
Set $y=f(x)$, $y'=f'(x')$; by transitivity of fibres \sref[I]{I.3.6.4}, on a $f'^{-1}(y')=f^{-1}(y)\otimes_{k(y)}k(y')$; since $f^{-1}(y)$ is locally of finite type over $k(y)$ or $k(y')$ is of finite type over $k(y)$ \sref[I]{I.6.4.11}, it follows from \sref{IV.6.7.8} that the properties $\mathbf{P}(f^{-1}(y),x)$ and $\mathbf{P}(f'^{-1}(y'),x')$ are equivalent; moreover, if $f$ is flat at the point $x$, $f'$ is flat at the point $x'$ \sref{IV.2.1.4}, which proves the proposition, taking into account \sref{IV.1.4.3}, (iii) and \sref{IV.2.7.1}, (iv).
\end{proof}

\begin{proposition}[6.8.3]
\label{IV.6.8.3}
For a morphism $f$ of locally Noetherian preschemes, let $\mathbf{M}(f)$ be one of the following properties: to be Cohen-Macaulay, $(\mathrm{S}_n)$, regular, $(\mathrm{R}_n)$, normal, reduced.
\begin{enumerate}
  \item[\rm(i)] Let $X$, $Y$, $Z$ be three locally Noetherian preschemes, $f:X\to Y$, $g:Y\to Z$ two morphisms.
  If $\mathbf{M}(f)$ and $\mathbf{M}(g)$ are true, $\mathbf{M}(g\circ f)$ is true.
  \item[\rm(ii)] Conversely, if $f$ is surjective and if $\mathbf{M}(f)$ and $\mathbf{M}(g\circ f)$ are true, then $\mathbf{M}(g)$ is true.
  \item[\rm(iii)] Let $X$, $Y$, $Y'$ be three locally Noetherian preschemes, $f:X\to Y$, $h:Y'\to Y$ two morphisms; set $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$.
  Suppose that $f$ or $h$ is locally of finite type.
  Then, if $\mathbf{M}(f)$ is true, so is $\mathbf{M}(f')$; the converse is true when $h$ is faithfully flat.
\end{enumerate}
The conclusions of (i) and (iii) are still true when $\mathbf{M}$ is the property of being smooth and (in (iii)) $h$ is quasi-compact.
\end{proposition}

\begin{proof}
\begin{enumerate}
  \item[\rm(i)] We already know that if $f$ and $g$ are flat, so is $u=g\circ f$, and that if $f$ is flat and $f$ surjective, $g$ is flat \sref{IV.2.1.6} and \sref{IV.2.2.13}.
  Moreover, for every $z\in Z$, $f_z=f\otimes 1_{k(z)}:u^{-1}(z)\to g^{-1}(z)$ is flat (resp.\ faithfully flat if $f$ is) and the morphism $f_z^{-1}(y)\otimes_{k(y)}k(y')$ is isomorphic to $f^{-1}(y)$ dans $g^{-1}(z)$ \sref[I]{I.3.6.4}.
  When $\mathbf{M}$ is the property of being regular, $(\mathrm{R}_n)$, normal or reduced, the hypothesis implies that $\mathbf{M}(f)$ and $\mathbf{M}(g)$ hold, and each of the fibres $f_z^{-1}(y')$ has for $y'\in Y'_z$ the corresponding property $c)$, $d)$, $e)$, $f)$ of \sref{IV.6.6.1}; we deduce from \sref{IV.6.6.1}, (i) that $X'_z$ has the same property, hence $\mathbf{M}(g\circ f)$ is true.

  \item[\rm(ii)] Conversely, the hypothesis that $\mathbf{M}(g\circ f)$ and $\mathbf{M}(f)$ are true and that $f$ is surjective implies that $Y'_z$ has the corresponding property for every $z$, by \sref{IV.6.6.1}, (ii), $f_z$ being surjective for every $z$.

  \item[\rm(iii)] The first assertion follows from \sref{IV.6.8.2}.
  On the other hand, if $h$ is faithfully flat, $f$ is flat \sref{IV.2.4.1}; moreover, with the notations of \sref{IV.6.8.2}, the properties $\mathbf{P}(f^{-1}(y),x)$ and $\mathbf{P}(f'^{-1}(y'),x')$ are equivalent \sref{IV.6.7.8}, hence $\mathbf{M}(f)$ and $\mathbf{M}(f')$ are equivalent.
  Finally, the last assertion of the proposition follows from \sref{IV.1.4.3}, (iii) and \sref{IV.2.7.1}, (iv).
\end{enumerate}
\end{proof}

\begin{remarks}[6.8.4]
\label{IV.6.8.4}
\begin{enumerate}
  \item[\rm(i)] If $f$ is faithfully flat, $g$ flat, and $g\circ f$ is of coprofondeur $\leq n$, then $g$ is of coprofondeur $\leq n$, as follows from \sref{IV.6.6.2}.
  \item[\rm(ii)] When, in \sref{IV.6.8.1}, one takes $Y$ to be the spectrum of a field $k$, the notions (iv), (v), and (vi) reduce to those defined in \sref{IV.6.7.5}.
  It is clear that these latter are the ones relative to the base field $k$.
  The definition \sref{IV.6.8.1} leads then to saying that a prescheme $X$ is ``regular (resp.\ $(\mathrm{R}_n)$, resp.\ normal) \emph{over $k$}'' instead of saying that it is ``geometrically regular (resp.\ $(\mathrm{R}_n)$, resp.\ normal)''; one will avoid confusing this notion with the property of being regular (resp.\ $(\mathrm{R}_n)$, resp.\ normal) which is independent of $k$.

\begin{proposition}[6.8.5]
\label{IV.6.8.5}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes locally Noetherian, at least one of which is locally of finite type over $k$.
For a $k$-prescheme $Z$, and a point $z\in Z$, let us denote by $\mathbf{P}(Z)$ one of the properties c), d), e) (resp.\ f)) of \sref{IV.6.6.1}; the property ``$\mathbf{P}(Z)$ geometric'' is then defined in \sref{IV.6.7.6} (resp.\ \sref{IV.4.6.1}).
Then:
\begin{enumerate}
  \item[\rm(i)] If $X$ has property $\mathbf{P}$ and $Y$ has property $\mathbf{P}$ geometric, $X\times_k Y$ has the geometric property $\mathbf{P}$.
  \item[\rm(ii)] If $X$ and $Y$ have property $\mathbf{P}$ geometric, so does $X\times_k Y$.
\end{enumerate}
\end{proposition}

\begin{proof}
Let $f:X\times_k Y\to X$, $g:X\to\Spec(k)$ be the structural morphisms, which are faithfully flat \sref{IV.2.2.13}.
The hypothesis that $Y$ has the geometric property $\mathbf{P}$ implies by \sref{IV.6.8.2} that $\mathbf{M}(f)$ is true, $\mathbf{M}$ being the property corresponding to $\mathbf{P}$; under hypothesis (ii), $\mathbf{M}(g)$ is also true, hence assertion (ii) follows from \sref{IV.6.8.3}, (i).
As for assertion (i), it follows directly from \sref{IV.6.6.1}, (i).
\end{proof}

\begin{theorem}[6.8.6]
\label{IV.6.8.6}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, $y=f(x)$.
The following properties are equivalent:
\begin{enumerate}
  \item[\rm a)] $f$ is smooth at the point $x$.
  \item[\rm b)] $f$ is regular at the point $x$.
  \item[\rm c)] $\sh{O}_x$ is a formally smooth $\sh{O}_y$-algebra $(\mathbf{0}$, 19.3.1) for the preadic topologies on $\sh{O}_x$ and $\sh{O}_y$.
  \item[\rm c')] $\sh{O}_x$ is a formally smooth $\sh{O}_y$-algebra $(\mathbf{0}$, 19.3.1) for the discrete topologies on $\sh{O}_x$ and $\sh{O}_y$.
\end{enumerate}
\oldpage[IV-2]{153}
The equivalence of a) and b) follows from the definitions \sref{IV.6.8.1} and from the fact that for locally Noetherian preschemes, morphisms locally of finite type are locally of finite presentation \sref{IV.1.4.2}.
Secondly, for $\sh{O}_x$ to be an $\sh{O}_y$-algebra smooth for the preadic topologies, it is necessary and sufficient that $\sh{O}_x$ be a flat $\sh{O}_y$-module and that $\sh{O}_x\otimes_{\sh{O}_y}k(y)$ be a $k(y)$-algebra formally smooth for its preadic topology $(\mathbf{0}$, 19.7.1); but for $\sh{O}_x\otimes_{\sh{O}_y}k(y)$ to be a $k(y)$-algebra formally smooth for its preadic topology, it is necessary and sufficient that it be a $k(y)$-algebra geometrically regular $(\mathbf{0}$, 19.6.6); this shows the equivalence of b) and c).
Finally, to prove the equivalence of $c)$ and $c')$, we can restrict to the case where $Y=\Spec(A)$ and $X=\Spec(C)$ are affine, $A$ being Noetherian and $C$ a finite type $A$-algebra, and one can therefore write $C$ in the form $A[\mathrm{T}_1,\ldots,\mathrm{T}_n]/\mathfrak{J}$.
Since here $\mathfrak{J}/\mathfrak{J}^2$ is a $C$-module of finite presentation, the equivalence of $c)$ and $c')$ follows from $(\mathbf{0}$, 22.6.4) applied to $A$, $B=A[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ and $C=B/\mathfrak{J}$.
\end{theorem}

\begin{corollary}[6.8.7]
\label{IV.6.8.7}
Let $X$, $Y$ be two locally Noetherian preschemes, $f:X\to Y$ a morphism locally of finite type.
Then the set of points $x\in X$ where $f$ is smooth (or regular) is open in $X$.
\end{corollary}

\begin{proof}
It follows from $(\mathbf{0}$, 22.6.5) that the set of $x\in X$ satisfying condition $c')$ of \sref{IV.6.8.6} is open in $X$, and the conclusion follows from \sref{IV.6.8.6}.
\end{proof}

\begin{remark}[6.8.8]
\label{IV.6.8.8}
In \sref{IV.17.5.1}, we shall show the equivalence of $b)$ and $c')$ in \sref{IV.6.8.6}, as well as the corollary \sref{IV.6.8.7}, remain valid without the Noetherian hypothesis on $X$ and $Y$, provided one restricts to morphisms \emph{locally of finite presentation}.
\end{remark}

\subsection{The theorem of generic flatness}
\label{subsection:IV.6.9}

\begin{theorem}[6.9.1]
\label{IV.6.9.1}
Let $Y$ be a locally Noetherian prescheme, integral, $u:X\to Y$ a morphism of finite type, $\sh{F}$ a coherent $\sh{O}_X$-Module.
There then exists a non-empty open subset $U$ of $Y$ such that $\sh{F}|u^{-1}(U)$ is flat over $U$.
\end{theorem}

\begin{proof}
One can clearly restrict to the case where $Y=\Spec(A)$ is affine; then $X$ is a finite union of open affines $X_i$ of finite type over $Y$; if, for each $i$, there is an open non-empty $U_i$ in $Y$ such that $\sh{F}|(X_i\cap u^{-1}(U_i))$ is flat over $U_i$, it is clear that taking $U$ as the intersection of the $U_i$ answers the question; one can therefore suppose that $X=\Spec(B)$, where $B$ is a finite type $A$-algebra.
We then have $\sh{F}=\widetilde{M}$, where $M$ is a finite type $B$-module; the theorem will therefore follow from the following:
\end{proof}

\begin{lemma}[6.9.2]
\label{IV.6.9.2}
Let $A$ be an integral Noetherian ring, $B$ a finite type $A$-algebra, $M$ a finite type $B$-module.
Then there exists $f\neq 0$ in $A$ such that $M_f$ is a free $A_f$-module.
\end{lemma}

\begin{proof}
\oldpage[IV-2]{154}
Let $K$ be the field of fractions of $A$; then $B\otimes_A K$ is a finite type algebra over $K$ and $M\otimes_A K$ a $(B\otimes_A K)$-module of finite type.
We argue by induction on the \emph{dimension} $n$ of the support of $M\otimes_A K$, which is $-\infty$ or finite and $\geq 0$.
Suppose first $n=-\infty$, that is to say $M\otimes_A K=0$; if $(m_i)_{1\leq i\leq r}$ is a generating system of the $B$-module $M$, there exists $f\neq 0$ in $A$ such that $fm_i=0$ for $1\leq i\leq r$; hence $M_f=0$ and the lemma is true in this case.
Suppose now $n\geq 0$.
We know that there exists a composition series $M=M_1\supset M_2\supset\ldots\supset M_q=0$ of the $B$-module $M$ such that each of the quotients $N_i=M_i/M_{i+1}$ is isomorphic to a $B$-module of the form $B/\mathfrak{p}_i$, where $\mathfrak{p}_i$ is a prime ideal of $B$ (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, n\textsuperscript{o}~4, th.~1).
If the theorem is true for each of the $N_i$, there is for each $i$ an $f_i\neq 0$ in $A$ such that $(N_i)_{f_i}$ is a free $A_{f_i}$-module; setting $f=f_1 f_2\cdots f_{q-1}$, it follows that $(N_i)_f$ is a free $A_f$-module for $1\leq i\leq q-1$.
But $(N_i)_f=(M_i)_f/(M_{i+1})_f$ $(\mathbf{0_I}$, 1.3.2) and since an extension of free modules is free, we deduce that $M_f$ is a free $A_f$-module.
Replacing $B$ by $B/\mathfrak{p}$ ($\mathfrak{p}$ a prime ideal of $B$), which is still of finite type over $A$, one can thus restrict to the case where $M=B$ and $B$ is \emph{integral}.
We then know (Bourbaki, \emph{Alg.~comm.}, chap.~V, \textsection3, n\textsuperscript{o}~1, cor.~1 of th.~1) that there exists an element $g\neq 0$ in $A$ and elements $t_i$ $(1\leq i\leq m)$ of $B$, algebraically independent over $A$ and such that $B_g$ is entire over $A_g[t_1,\ldots,t_m]$.
One can replace $A$ by $A_g$, $B$ by $B_g$ and hence assume that $B$ is entire over $C=A[t_1,\ldots,t_m]$, hence a $C$-module of finite type and \emph{without torsion}.
We know moreover \sref{IV.4.1.2} that the dimension of $\Spec(B\otimes_A K)$ is equal to $m$, hence on a $m=n$.

This being so, if $h$ is the rank of the torsion-free $C$-module $B$, there exists an exact sequence of $C$-modules
\[
  0\to C^h\to B\to M'\to 0
\]
where $M'$ is a $C$-module of \emph{torsion} of finite type; the support of $M'$ therefore does not contain the generic point of $\Spec(C)$ \sref[I]{I.7.4.6} and hence the support of $M'\otimes_A K$ does not contain the generic point of $\Spec(C\otimes_A K)$ \sref[I]{I.9.1.13.1}; we thus conclude \sref{IV.4.1.2.1} that its dimension is $<n$.
By the induction hypothesis, there exists an $f\neq 0$ in $A$ such that $M'_f$ is a free $A_f$-module; on the other hand $C_f^h$ is an $A_f$-module free; hence it follows that $M_f$ is a free $A_f$-module.
Q.E.D.
\end{proof}

\begin{corollary}[6.9.3]
\label{IV.6.9.3}
Let $S$ be a Noetherian prescheme, $u:X\to S$ a morphism of finite type, $\sh{F}$ a coherent $\sh{O}_X$-Module.
Then there is a partition $(\mathrm{S}_i)_{1\leq i\leq n}$ of $S$ in a finite number of locally closed subsets of $S$ such that, if one denotes by $S_i$ the reduced sub-prescheme having space $\mathrm{S}_i$ as underlying space, and if one sets $X_i=X\times_S S_i$, $\sh{F}_i=\sh{F}\otimes_{\sh{O}_S}\sh{O}_{S_i}$, the $\sh{O}_{X_i}$-Module $\sh{F}_i$ is flat over $S_i$.
\end{corollary}

\begin{proof}
We proceed by Noetherian induction $(\mathbf{0_I}$, 2.2.2) on the set of closed subsets $T$ of $S$ such that the reduced sub-prescheme having $T$ for underlying space satisfies the conclusion of \sref{IV.6.9.3}.
One can thus restrict to proving the corollary for $S$ having for underlying space a closed subset $\neq S$.
Since the morphism $S_\mathrm{red}\to S$ is of finite type and surjective, one can replace $S$ by $S_\mathrm{red}$, hence suppose $S$ reduced and non-empty.
Since $S$ is Noetherian, the interior $T$ of an irreducible component of $S$ is non-empty, and the prescheme induced on $T$ is integral; there is therefore by \sref{IV.6.9.1} a non-empty open subset $U\subset T$
\oldpage[IV-2]{155}
such that $\sh{F}|u^{-1}(U)$ is flat over $U$.
If $Y$ is the sub-prescheme reduced by $S$ having for underlying space $S-U$, it is clear by hypothesis that there is a partition $(Y_i)$ of $Y$ into locally closed sets in $S$, such that the $Y_i$ and $U$ form a partition answering the question.
\end{proof}

\subsection{Dimension and depth of a Module normally flat along a closed sub-prescheme}
\label{subsection:IV.6.10}

\begin{env}[6.10.1]
\label{IV.6.10.1}
Let $X$ be a locally Noetherian prescheme, $\sh{J}$ a quasi-coherent Ideal of $\sh{O}_X$, $Y$ the closed sub-prescheme of $X$ defined by $\sh{J}$, $j:Y\to X$ the canonical injection.
For every integer $k\geq 0$, the $\sh{O}_X$-Module $\sh{J}^k\sh{F}/\sh{J}^{k+1}\sh{F}$ is annihilated by $\sh{J}$, hence is of the form $j_*(\sh{G}_k)$, where $\sh{G}_k=j^*(\sh{J}^k\sh{F}/\sh{J}^{k+1}\sh{F})=j^*(\sh{J}^k\sh{F})$ is a coherent $\sh{O}_Y$-Module.
By abuse of language, we denote $\operatorname{gr}^*_\sh{J}(\sh{F})$ the graded $\sh{O}_Y$-Module equal to the direct sum
\[
  \bigoplus_{k=0}^\infty\sh{G}_k=j^*\bigg(\bigoplus_{k=0}^\infty\sh{J}^k\sh{F}/\sh{J}^{k+1}\sh{F}\bigg);
\]
in particular, we have $\operatorname{gr}^0_\sh{J}(\sh{F})=\sh{G}_0=\sh{F}\otimes_{\sh{O}_X}\sh{O}_Y=j^*(\sh{F})$.
We shall say (following Hironaka) that $\sh{F}$ is \emph{normally flat along} $Y$ if $\operatorname{gr}^*_\sh{J}(\sh{F})$ is a flat $\sh{O}_Y$-Module.
This is the same as saying ((2.1.12) and $(\mathbf{0_I}$, 6.1.2)) that each of the $\sh{O}_Y$-Modules $\sh{G}_k$ is \emph{locally free}.
\end{env}

\begin{proposition}[6.10.2]
\label{IV.6.10.2}
Let $X$ be a locally Noetherian prescheme, $Y$ a closed integral sub-prescheme of $X$.
For every coherent $\sh{O}_X$-Module $\sh{F}$, there exists an open subset $U$ of $X$ such that $Y\cap U\neq\varnothing$ and that $\sh{F}|U$ is normally flat along $Y\cap U$.
\end{proposition}

\begin{proof}
Let $\sh{J}$ be the coherent Ideal of $\sh{O}_X$ defining $Y$; the $\sh{O}_Y$-Algebra $\sh{B}=\operatorname{gr}^*_\sh{J}(\sh{O}_X)$, an image of $\sh{J}/\sh{J}^2$, is quasi-coherent and \emph{of finite type} since it is generated by $\operatorname{gr}^1_\sh{J}(\sh{O}_X)$, image of $\sh{J}/\sh{J}^2$; as $\operatorname{gr}^*_\sh{J}(\sh{F})$ is a graded $\sh{B}$-module quasi-coherent generated by $\operatorname{gr}^1_\sh{J}(\sh{F})$, it is a finite type $\sh{B}$-Module.
If we set $Z=\Spec(\sh{B})$, the structural morphism $u:Z\to Y$ is then of finite type, and if $\sh{H}$ is the coherent $\sh{O}_Z$-Module such that $u_*(\sh{H})=\operatorname{gr}^*_\sh{J}(\sh{F})$, it suffices to apply $u$ and to $\sh{H}$ the theorem of generic flatness \sref{IV.6.9.1} to prove the proposition.
\end{proof}

\begin{proposition}[6.10.3]
\label{IV.6.10.3}
With the notations of \sref{IV.6.10.1}, suppose that $\sh{F}$ is normally flat along $Y$.
Then:
\begin{enumerate}
  \item[\rm(i)] $Y\cap\Supp(\sh{F})$ is a part that is both open and closed in $Y$ (in other words, a union of connected components of $Y$).
  \item[\rm(ii)] If $\sh{J}$ is locally nilpotent, $\Supp(\sh{F})$ is an open and closed part of $X$, and for every $x\in\Supp(\sh{F})$, we have
  \begin{align}
    \dim(\sh{F}_x)&=\dim(\sh{O}_{Y,x})\tag{6.10.3.1}\label{IV.6.10.3.1}\\
    \operatorname{prof}(\sh{F}_x)&=\operatorname{prof}(\sh{O}_{Y,x})\tag{6.10.3.2}\label{IV.6.10.3.2}\\
    \operatorname{coprof}(\sh{F}_x)&=\operatorname{coprof}(\sh{O}_{Y,x}).\tag{6.10.3.3}\label{IV.6.10.3.3}
  \end{align}
\end{enumerate}
\end{proposition}

\begin{corollary}[6.10.4]
\label{IV.6.10.4}
Let $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ (resp.\ $\mathrm{U}_{\mathrm{C}_n}(\sh{F})$) be the set of $x\in X$ such that $\sh{F}$ satisfies $(\mathrm{S}_n)$ at the point $x$ (resp.\ such that $\operatorname{coprof}(\sh{F}_x)\leq n$).
If $\sh{F}$ is normally flat along $Y$, then $\mathrm{U}_{\mathrm{S}_n}(\sh{F})=\mathrm{U}_{\mathrm{S}_n}(\sh{O}_Y)$ and $\mathrm{U}_{\mathrm{C}_n}(\sh{F})=\mathrm{U}_{\mathrm{C}_n}(\sh{O}_Y)$.
\end{corollary}

\begin{proposition}[6.10.5]
\label{IV.6.10.5}
With the notations of \sref{IV.6.10.1}, suppose that $Y$ is connected and non-empty, and that $\sh{F}$ is normally flat along $Y$.
For every integer $n\geq 0$, we set
\begin{equation}
\label{IV.6.10.5.1}
  r(n)=\operatorname{rg}_{\sh{O}_Y}(\operatorname{gr}^n_\sh{J}(\sh{F}))\tag{6.10.5.1}
\end{equation}
(the $\sh{O}_Y$-Module locally free $\operatorname{gr}^n_\sh{J}(\sh{F})$ being necessarily of constant rank).
Then:
\begin{enumerate}
  \item[\rm(i)] There exists a polynomial $P\in\mathbf{Q}[\mathrm{T}]$ such that $P(n)=r(n)$ for $n$ sufficiently large.
  \item[\rm(ii)] We have $Y\cap\Supp(\sh{F})=\varnothing$, or $Y\cap\Supp(\sh{F})=Y$ (in other words $Y\subset\Supp(\sh{F})$).
\end{enumerate}
In the second case, let $d-1$ denote the degree of $P$; for every maximal point $y$ of $Y$, we have
\begin{equation}
\label{IV.6.10.5.2}
  \dim(\sh{F}_y)=d\tag{6.10.5.2}
\end{equation}
and in particular
\begin{equation}
\label{IV.6.10.5.3}
  \codim(Y,\Supp(\sh{F}))=d.\tag{6.10.5.3}
\end{equation}
\begin{enumerate}
  \item[\rm(iii)] Suppose $Y\subset\Supp(\sh{F})$.
  For every $x\in Y$, we have
  \begin{equation}
  \label{IV.6.10.5.4}
    \dim(\sh{F}_x)=\dim(\sh{O}_{Y,x})+d.\tag{6.10.5.4}
  \end{equation}
\end{enumerate}
\end{proposition}

\begin{proposition}[6.10.6]
\label{IV.6.10.6}
\oldpage[IV-2]{157}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module, $Y$ a closed irreducible sub-prescheme of $X$, of generic point $y\in\Supp(\sh{F})$.
There then exists a neighbourhood $U$ of $y$ in $X$ such that, for every $x\in U\cap Y$, we have
\begin{align}
  \dim(\sh{F}_x)&=\dim(\sh{F}_y)+\dim(\sh{O}_{Y,x})\tag{6.10.6.1}\label{IV.6.10.6.1}\\
  \operatorname{prof}(\sh{F}_x)&=\operatorname{prof}(\sh{F}_y)+\operatorname{prof}(\sh{O}_{Y,x})\tag{6.10.6.2}\label{IV.6.10.6.2}\\
  \operatorname{coprof}(\sh{F}_x)&=\operatorname{coprof}(\sh{F}_y)+\operatorname{coprof}(\sh{O}_{Y,x}).\tag{6.10.6.3}\label{IV.6.10.6.3}
\end{align}
\end{proposition}

\subsection{Criteria for the sets $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ or $\mathrm{U}_{\mathrm{C}_n}(\sh{F})$ to be open}
\label{subsection:IV.6.11}

\oldpage[IV-2]{158}

\begin{lemma}[6.11.1]
\label{IV.6.11.1}
Let $X$ be a locally Noetherian prescheme, locally immersible in a regular scheme \sref{IV.5.11.1}; let $\sh{F}$ be a coherent $\sh{O}_X$-Module.
Then:
\begin{enumerate}
  \item[\rm(i)] (M.~Auslander) The function $x\mapsto\operatorname{coprof}(\sh{F}_x)$ is semi-continuous superiorly in $X$ (in other words, for every integer $n$, the set $\mathrm{U}_{\mathrm{C}_n}(\sh{F})$ of $x\in X$ such that $\operatorname{coprof}(\sh{F}_x)\leq n$ is open).
  \item[\rm(ii)] For every integer $n$, the set $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ of $x\in X$ such that $\sh{F}$ has property $(\mathrm{S}_n)$ at the point $x$ is open.
\end{enumerate}
\end{lemma}

\begin{proposition}[6.11.2]
\label{IV.6.11.2}
Let $X$ be a locally Noetherian prescheme, regular and locally immersible in a regular scheme \sref{IV.5.11.1}; let $\sh{F}$ be a coherent $\sh{O}_X$-Module.
Then:
\begin{enumerate}
  \item[\rm(i)] (M.~Auslander) The function $x\mapsto\operatorname{coprof}(\sh{F}_x)$ is semi-continuous superiorly in $X$ (in other words, for every integer $n$, the set $\mathrm{U}_{\mathrm{C}_n}(\sh{F})$ of $x\in X$ such that $\operatorname{coprof}(\sh{F}_x)\leq n$ is open).
  \item[\rm(ii)] For every integer $n$, the set $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ of $x\in X$ such that $\sh{F}$ has property $(\mathrm{S}_n)$ at the point $x$ is open.
\end{enumerate}
\end{proposition}

\begin{corollary}[6.11.3]
\label{IV.6.11.3}
Under the hypotheses of \sref{IV.6.11.2}, the set $\mathrm{CM}(\sh{F})$ of points $x\in X$ such that $\sh{F}_x$ is a Cohen-Macaulay module is open in $X$.
\end{corollary}

\begin{proof}
Indeed, it is the set $\mathrm{U}_{\mathrm{C}_0}(\sh{F})$.
\end{proof}

\begin{remarks}[6.11.4]
\label{IV.6.11.4}
\begin{enumerate}
  \item[\rm(i)] The reasoning of \sref{IV.6.11.2}, (ii) shows that (without hypothesis on $X$) when $\mathrm{U}_{\mathrm{C}_n}(\sh{F})$ is open for every integer $n$, then $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ is also open.
  \item[\rm(ii)] We have $\mathrm{CM}(\sh{F})=\bigcap_n\mathrm{U}_{\mathrm{S}_n}(\sh{F})$.
  If every point $x\in X$ admits a neighbourhood $V$ that is open such that $V$ is of \emph{finite} dimension, the decreasing sequence of intersections $V\cap\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ is \emph{stationary} since $\dim(\sh{F}_x)$ is finite for every $x$; hence, if the $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ are open, $\mathrm{CM}(\sh{F})$ is then also open in $X$.
  \item[\rm(iii)] We write $\mathrm{U}_{\mathrm{S}_n}(X)$, $\mathrm{U}_{\mathrm{C}_n}(X)$, $\mathrm{CM}(X)$ instead of $\mathrm{U}_{\mathrm{S}_n}(\sh{O}_X)$, $\mathrm{U}_{\mathrm{C}_n}(\sh{O}_X)$, $\mathrm{CM}(\sh{O}_X)$.
\end{enumerate}
\end{remarks}

\begin{proposition}[6.11.5]
\label{IV.6.11.5}
Let $A$ be a local Noetherian ring, $M$ a finite type $A$-module.
For every prime ideal $\mathfrak{p}$ of $A$, we have
\begin{equation}
\label{IV.6.11.5.1}
  \operatorname{coprof}_{A_\mathfrak{p}}(M_\mathfrak{p})\leq\operatorname{coprof}_A(M).\tag{6.11.5.1}
\end{equation}
\end{proposition}

\begin{proof}
We can restrict to the case where $M_\mathfrak{p}\neq 0$.
Since $\widehat{A}$ is a faithfully flat $A$-module $(\mathbf{0_I}$, 6.5.1), there exists a prime ideal $\mathfrak{q}$ of $\widehat{A}$ above $\mathfrak{p}$ $(\mathbf{0_I}$, 6.5.1); since $\widehat{M}_\mathfrak{q}=(M\otimes_A\widehat{A})_\mathfrak{q}=M_\mathfrak{p}\otimes_{A_\mathfrak{p}}\widehat{A}_\mathfrak{q}$ and $\widehat{A}_\mathfrak{q}$ is a flat $A_\mathfrak{p}$-module, hence a flat $A$-module flat $(\mathbf{0_I}$, 6.2.1), it follows from \sref{IV.6.3.2} that $\operatorname{coprof}_{A_\mathfrak{p}}(M_\mathfrak{p})\leq\operatorname{coprof}_{\widehat{A}_\mathfrak{q}}(\widehat{M}_\mathfrak{q})$.
On the other hand, $X=\Spec(\widehat{A})$ is a sub-scheme of a regular scheme by the Cohen structure theorem $(\mathbf{0}$, 19.8.8, (i)); on deduces from \sref{IV.6.11.2} that $\operatorname{coprof}_{\widehat{A}_\mathfrak{q}}(\widehat{M}_\mathfrak{q})\leq\operatorname{coprof}_{\widehat{A}}(\widehat{M})$; finally, we know that $\operatorname{coprof}_{\widehat{A}}(\widehat{M})=\operatorname{coprof}_A(M)$ $(\mathbf{0}$, 16.4.10), which completes the proof of \sref{IV.6.11.5.1}.
\end{proof}

\begin{proposition}[6.11.6]
\label{IV.6.11.6}
\oldpage[IV-2]{160}
Let $X$ be a locally Noetherian prescheme, $\sh{F}$ a coherent $\sh{O}_X$-Module, $n$ an integer $>0$.
Suppose that for every closed integral sub-prescheme $Y$ of $X$, there exists a non-empty open subset $W$ of $Y$ such that the sub-prescheme of $Y$ induced on the open $W$ satisfies $(\mathrm{S}_n)$.
Then the set $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ is open in $X$.
\end{proposition}

\begin{corollary}[6.11.7]
\label{IV.6.11.7}
With the notations of \sref{IV.6.11.6}:
\begin{enumerate}
  \item[\rm(i)] The set $\mathrm{U}_{\mathrm{S}_1}(\sh{F})$ is open in $X$.
  \item[\rm(ii)] For $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ to be open, it suffices that every maximal point $x$ of $\Supp(\sh{F})$, belonging to $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$, be interior to $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$.
\end{enumerate}
\end{corollary}

\begin{proposition}[6.11.8]
\label{IV.6.11.8}
Let $X$ be a locally Noetherian prescheme satisfying the following property:
\begin{itemize}
  \item[\rm(CMU)] Every closed integral sub-prescheme $Y$ of $X$ contains a non-empty open $W$ such that the prescheme induced by $X$ on $W$ is a Cohen-Macaulay prescheme.
\end{itemize}
Then, for every coherent $\sh{O}_X$-Module $\sh{F}$, the function $x\mapsto\operatorname{coprof}(\sh{F}_x)$ is locally constructible and semi-continuous superiorly; the sets $\mathrm{U}_{\mathrm{C}_n}(\sh{F})$ and $\mathrm{U}_{\mathrm{S}_n}(\sh{F})$ are open in $X$.
\end{proposition}

\begin{proof}
Indeed, let $Y$ be a closed integral sub-prescheme of $X$, of generic point $y$; by \sref{IV.6.10.6}, there is a neighbourhood $V$ of $y$ in $X$ such that, for every $x\in V\cap Y$, we have
\begin{equation}
\label{IV.6.11.8.1}
  \operatorname{coprof}(\sh{F}_x)=\operatorname{coprof}(\sh{F}_y)+\operatorname{coprof}(\sh{O}_{Y,x}).\tag{6.11.8.1}
\end{equation}
But by hypothesis there exists a non-empty open $W$ of $Y$ such that, for $x\in V\cap W$ we have $\operatorname{coprof}(\sh{O}_{Y,x})=0$, hence $\operatorname{coprof}(\sh{F}_x)$ is constant in a neighbourhood of $y$ in $Y$, which proves that the function $x\mapsto\operatorname{coprof}(\sh{F}_x)$ is locally constructible $(\mathbf{0_{III}}$, 9.3.2); moreover it follows from \sref{IV.6.11.5} and from $(\mathbf{0_{III}}$, 9.3.4) that this function is semi-continuous superiorly.
The last assertion follows from \sref{IV.6.11.4}, (i), or equivalently from \sref{IV.6.11.6}.
\end{proof}

\label{IV.6.11.9}
\begin{enumerate}
  \item[\rm(i)] If $X$ satisfies the hypothesis (CMU) of \sref{IV.6.11.8}, then, for every morphism locally of finite type $u:X'\to X$, $X'$ also satisfies (CMU).
  \item[\rm(ii)] Suppose that there exists a coherent $\sh{O}_X$-Module $\sh{F}$ such that $\Supp(\sh{F})=X$ and that $\sh{F}$ is a Cohen-Macaulay $\sh{O}_X$-Module.
  Then $X$ satisfies condition (CMU).
\end{enumerate}
\end{remarks}

\subsection{Nagata's criteria for $\mathrm{Reg}(X)$ to be open}
\label{subsection:IV.6.12}

\begin{env}[6.12.1]
\label{IV.6.12.1}
Given a locally Noetherian prescheme $X$, the \emph{singular locus} of $X$ is defined to be the set of points $x\in X$ such that $X$ is not regular at the point $x$ (in other words, such that the local ring $\sh{O}_x$ is not regular); it is denoted $\mathrm{Sing}(X)$; the complementary set $X-\mathrm{Sing}(X)$, consisting of the $x\in X$ where $X$ is regular, is denoted $\mathrm{Reg}(X)$.
We propose in this number to give conditions under which $\mathrm{Sing}(X)$ is \emph{closed} (i.e.\ $\mathrm{Reg}(X)$ is \emph{open}).
\end{env}

\oldpage[IV-2]{163}

\begin{proposition}[6.12.2]
\label{IV.6.12.2}
Let $X$ be a locally Noetherian prescheme.
The following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] $\mathrm{Reg}(X)$ is open in $X$.
  \item[\rm b)] For every $x\in\mathrm{Reg}(X)$, there exists a non-empty open subset of $\overline{\{x\}}$ contained in $\mathrm{Reg}(X)$.
\end{enumerate}
Moreover, these conditions are implied by the following:
\begin{enumerate}
  \item[\rm c)] For every $x\in\mathrm{Reg}(X)$, denoting by $Y$ the closed reduced sub-prescheme of $X$ having $\overline{\{x\}}$ as underlying space, $\mathrm{Reg}(Y)$ is a neighbourhood of $x$ in $Y$.
\end{enumerate}
\end{proposition}

\begin{proof}
The equivalence of a) and b) follows from the fact that $\mathrm{Reg}(X)$ is stable by generisation $(\mathbf{0}$, 17.3.2) and from $(\mathbf{0_{III}}$, 9.2.6).
To prove that c) implies b), consider a closed integral sub-prescheme $Z$ of $Y$ of generic point $z$; if $Y'$ is the sub-prescheme integral of $X$ having as underlying space $Y'=\overline{\{z\}}$ of $z$ in $X$, $Z$ is open in $Y'$ and the sub-prescheme $Z$ of $Y'$, being reduced, is induced by $Y'$ on the open $Z$ of the underlying space of $Y'$.
The hypothesis implies that $\mathrm{Reg}(Y')$ is a neighbourhood of $z$ in $Y'$; it therefore suffices to apply \sref{IV.6.12.2} by replacing $X$ by $Y$ and $Y$ by $Z$.
\end{proof}

\begin{corollary}[6.12.3]
\label{IV.6.12.3}
Let $X$ be a locally Noetherian prescheme.
The following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] For every sub-prescheme $Y$ of $X$, $\mathrm{Reg}(Y)$ is open in $Y$.
  \item[\rm b)] For every closed integral sub-prescheme $Y$ of $X$, $\mathrm{Reg}(Y)$ contains a non-empty open part of $Y$.
\end{enumerate}
\end{corollary}

\begin{proof}
It is clear that a) implies b), since if $Y$ is integral and $y$ is its generic point, $\sh{O}_{Y,y}$ is a field, hence $y\in\mathrm{Reg}(Y)$.
Conversely, to see that b) implies a), consider a closed integral sub-prescheme $Z$ of $Y$ of generic point $z$; if $Y'$ is the sub-prescheme integral of $X$ having $\overline{\{z\}}$ as underlying space, $Y'=\overline{\{z\}}$ in $X$, and the sub-prescheme $Z$ of $Y'$, being reduced, is induced by $Y'$ on the open part of its underlying space.
The hypothesis implies that $\mathrm{Reg}(Y')$ is a neighbourhood of $z$ in $Y'$; it suffices then to apply \sref{IV.6.12.2} by replacing $X$ and $Y$ by $Z$.
\end{proof}

\begin{theorem}[6.12.4]
\label{IV.6.12.4}
\emph{(Nagata).} ---
Let $A$ be a Noetherian ring, $X=\Spec(A)$.
The following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] For every prescheme $X'$ locally of finite type over $X$, $\mathrm{Reg}(X')$ is open in $X'$.
  \item[\rm b)] For every finite integral $A$-algebra $A'$, there exists a non-empty open of $X'=\Spec(A')$ contained in $\mathrm{Reg}(X')$.
  \item[\rm c)] For every prime ideal $\mathfrak{p}$ of $A$ and every finite radiciel extension $K'$ of the field of fractions $K$ of $A/\mathfrak{p}$, finite on $A$, having $K'$ for field of fractions, and such that in $X'=\Spec(A')$, a non-empty open subset is contained in $\mathrm{Reg}(X')$.
\end{enumerate}
\end{theorem}

\begin{corollary}[6.12.5]
\label{IV.6.12.5}
\emph{(Zariski).} ---
Let $k$ be a field; for every $k$-prescheme $X$ locally of finite type over $k$, the set of $x\in X$ where $X$ is regular (resp.\ \emph{geometrically regular} over $k$) is open in $X$.
\end{corollary}

\begin{corollary}[6.12.6]
\label{IV.6.12.6}
Let $A$ be a ring having one of the following properties:
\begin{enumerate}
  \item[\rm(i)] $A$ is a Dedekind ring and its field of fractions $K$ is of characteristic $0$.
  \item[\rm(ii)] $A$ is a semi-local Noetherian ring of dimension $\leq 1$.
\end{enumerate}
Then, for every prescheme $X$ locally of finite type over $A$, $\mathrm{Reg}(X)$ is open in $X$.
\end{corollary}

\oldpage[IV-2]{166}

\begin{theorem}[6.12.7]
\label{IV.6.12.7}
\emph{(Nagata).} ---
Let $A$ be a local Noetherian \emph{complete} ring, $X=\Spec(A)$.
Then $\mathrm{Reg}(X)$ is open in $X$.
\end{theorem}

\begin{proof}
By \sref{IV.6.12.2}, we are reduced to the case where $A$ is moreover \emph{integral}, and we must show that $\mathrm{Reg}(X)$ contains a non-empty open subset of $X$.
We distinguish two cases:

\medskip\noindent
I) \emph{The field of fractions $K$ of $A$ is of characteristic $0$.} ---
We know then $(\mathbf{0}$, 19.8.8) that there exists a complete discrete valuation ring $C$ and a sub-ring $B$ of $A$ such that $A$ is a \emph{finite} $B$-algebra and such that $B$ is isomorphic to a formal power series ring $C[[T_1,\ldots,T_r]]$.
As $C$ is regular $(\textbf{II}$, 7.1.6), so is $B$ $(\mathbf{0}$, 17.3.8);
moreover, the field of fractions $L$ of $B$ being of characteristic $0$, $K$ is a \emph{separable} finite extension of $L$, so we can apply \sref{IV.6.12.4}, I) to $B$, which proves the proposition in this case.

\medskip\noindent
II) \emph{The field of fractions $K$ of $A$ is of characteristic $p>0$.} ---
Then $A$ contains the prime field $\bb{F}_p$, and the theorem has already been proved in this case $(\mathbf{0}$, 22.7.6).
\end{proof}

\oldpage[IV-2]{167}

\begin{corollary}[6.12.8]
\label{IV.6.12.8}
Let $A$ be a local Noetherian complete ring.
Then, for every prescheme $X$ locally of finite type over $A$, $\mathrm{Reg}(X)$ is open in $X$.
\end{corollary}

\begin{proof}
We verify condition $c)$ of \sref{IV.6.12.4}.
If $\mathfrak{p}$ is prime in $A$, $A/\mathfrak{p}$ is again a local Noetherian complete ring; if $K'$ is a finite extension of the field of fractions $K$ of $A/\mathfrak{p}$, $K'$ is the field of fractions of a sub-$A$-algebra $A'$ of $K'$, generated by a finite system of generators of $K'$ over $K$, integral over $A$; $A'$ is a semi-local integral ring of dimension $\leq 1$ $(\mathbf{0}$, 16.1.5), and therefore, in $X'=\Spec(A')$, the set reduced to the generic point is open and evidently contained in $\mathrm{Reg}(X')$, which proves condition $c)$ of \sref{IV.6.12.4} in this case.
This corollary applies in particular when $A=\bb{Z}$.
\end{proof}

\begin{proposition}[6.12.9]
\label{IV.6.12.9}
Let $X$ be a locally Noetherian prescheme such that $\mathrm{Reg}(X)$ is open; then, for every $n\geq 0$, the set $\mathrm{U}_{\mathrm{R}_n}(X)$ of the points $x\in X$ where $X$ satisfies $(\mathrm{R}_n)$ is open.
\end{proposition}

\begin{proof}
Indeed \sref[0]{5.8.2}, $\mathrm{U}_{\mathrm{R}_n}(X)$ is the union of $\mathrm{Reg}(X)$ and the set of $x\in\mathrm{Sing}(X)$ such that the generic points $z_i$ of the irreducible components of $\mathrm{Sing}(X)$ containing $x$ satisfy the relation $\dim(\sh{O}_{X,z_i})\geq n+1$ (since we have $z_i\notin\mathrm{Reg}(X)$); in other words, these points $x\in\mathrm{Sing}(X)$ are those for which $\codim_x(\mathrm{Sing}(X),X)\geq n+1$ \sref[0]{5.1.2}; as the function $x\mapsto\codim_x(\mathrm{Sing}(X),X)$ is semi-continuous inferiorly $(\mathbf{0}$, 14.2.6), the set of these points is open in $\mathrm{Sing}(X)$, which completes the proof.
\end{proof}

We also note the following elementary result:

\begin{proposition}[6.12.10]
\label{IV.6.12.10}
Let $X$ be a locally Noetherian prescheme.
The set $\mathrm{U}_{\mathrm{R}_1}(X)$ is open in $X$.
For $\mathrm{U}_{\mathrm{R}_0}(X)$ to be open in $X$, it is necessary and sufficient that every maximal point $x\in X$ such that $x\in\mathrm{U}_{\mathrm{R}_1}(X)$ (which means that $X$ is reduced at $x$) be interior to $\mathrm{U}_{\mathrm{R}_1}(X)$.
\end{proposition}

\begin{proof}
By definition \sref[0]{5.8.2}, $\mathrm{U}_{\mathrm{R}_1}(X)$ is the set $X-\bigcup X'_\alpha$, where $X'_\alpha$ ranges over the set of irreducible components of $X$ such that $X$ is not reduced at their generic point; as the set of irreducible components of $X$ is locally finite, $\mathrm{U}_{\mathrm{R}_1}(X)$ is open.
Regarding $\mathrm{U}_{\mathrm{R}_0}(X)$, the condition of the statement being trivially necessary, let us prove that it is sufficient.
Let $X''_\beta$ be the irreducible components of $X$ such that at their generic point $x''_\beta$ the prescheme $X$ is reduced; by hypothesis there exists an open $U\subset\mathrm{U}_{\mathrm{R}_1}(X)$ containing all the $x''_\beta$.
Denote then by $Z_\lambda$ the irreducible components of the closed set $Z=X-U$ whose generic point $z_\lambda$ is such that $\dim(\sh{O}_{X,z_\lambda})\leq 1$ and that $\sh{O}_{X,z_\lambda}$ is non-regular.
No point belonging to one of the $Z_\lambda$ can belong to $\mathrm{U}_{\mathrm{R}_1}(X)$;
\oldpage[IV-2]{168}
but conversely, if $x\in Z$ does not belong to any of the $Z_\lambda$, then, for every generisation $x'$ of $x$, either $x'$ belongs to $U$, or $\dim(\sh{O}_{X,x'})\geq 2$, or $\dim(\sh{O}_{X,x'})=1$ and, as $x'$ does not belong to any of the $Z_\lambda$, $\overline{\{x'\}}$ is necessarily an irreducible component of $Z$, of codimension $1$ in $X$, distinct from the $Z_\lambda$, hence $\sh{O}_{X,x'}$ is regular by definition.
We conclude that $\mathrm{U}_{\mathrm{R}_1}(X)=X-\bigcup Z_\lambda$; as the set of the $Z_\lambda$ is locally finite in $Z$, $\bigcup Z_\lambda$ is closed, which finishes proving that $\mathrm{U}_{\mathrm{R}_1}(X)$ is open in $X$.
\end{proof}

\subsection{Criteria for $\mathrm{Nor}(X)$ to be open}
\label{subsection:IV.6.13}

\begin{env}[6.13.1]
\label{IV.6.13.1}
Given a locally Noetherian prescheme $X$, we denote by $\mathrm{Nor}(X)$ the set of $x\in X$ where $X$ is \emph{normal}; this set contains $\mathrm{Reg}(X)$ and is contained in the (open) set of points $x$ such that $\sh{O}_{X,x}$ is integral (i.e.\ the set of points $x$ where $X$ is reduced, and which belong to only a single irreducible component of $X$).
\end{env}

\begin{proposition}[6.13.2]
\label{IV.6.13.2}
Let $X$ be a locally Noetherian prescheme, normalis\'{e} $(\textbf{II}$, 6.3.8), $X'$ its normalisation.
If the canonical morphism $f:X'\to X$ is finite, $\mathrm{Nor}(X)$ is open in $X$.
\end{proposition}

\begin{proof}
Indeed, on a $X'=\Spec(f_*(\sh{O}_{X'}))$ and $f_*(\sh{O}_{X'})$ is a coherent $\sh{O}_X$-Algebra by hypothesis $(\textbf{II}$, 6.1.3).
To say that $X$ is normal at a point $x$ means that the canonical homomorphism $\sh{O}_x\to (f_*(\sh{O}_{X'}))_x$ is bijective $(\textbf{II}$, 6.3.4); but the set of such points is open since $\sh{O}_X$ and $f_*(\sh{O}_{X'})$ are coherent $(\mathbf{0_I}$, 5.2.7).
\end{proof}

\begin{corollary}[6.13.3]
\label{IV.6.13.3}
If $A$ is an integral Noetherian \emph{Japanese} ring, $\mathrm{Nor}(\Spec(A))$ is open in $\Spec(A)$.
\end{corollary}

\begin{proposition}[6.13.4]
\label{IV.6.13.4}
Let $X$ be a locally Noetherian prescheme.
For $\mathrm{Nor}(X)$ to be open, it is necessary and sufficient that every maximal point $x$ of $X$ such that $X$ is reduced at the point $x$ (which means that $x\in\mathrm{Nor}(X)$) be interior to $\mathrm{Nor}(X)$.
\end{proposition}

\begin{proof}
It suffices to prove the sufficiency of the condition.
By virtue of the remark made in \sref{IV.6.13.1}, we can restrict (by replacing $X$ by an open subset of $X$ in which $X$ is reduced) to the case where $X$ is \emph{reduced}, i.e.\ suppose the maximal points of $X$ belong to $\mathrm{U}_{\mathrm{S}_1}(X)$ and to $\mathrm{U}_{\mathrm{R}_1}(X)$;
as $\mathrm{Nor}(X)=\mathrm{U}_{\mathrm{S}_1}(X)\cap\mathrm{U}_{\mathrm{R}_1}(X)$ by Serre's criterion \sref[0]{5.8.6}, we deduce from \sref{IV.6.11.7} and \sref{IV.6.12.10} that these two sets are open; hence $\mathrm{Nor}(X)$ is open in $X$.
\end{proof}

\begin{corollary}[6.13.5]
\label{IV.6.13.5}
If $X$ is such that $\mathrm{Reg}(X)$ is open in $X$, then $\mathrm{Nor}(X)$ is open in $X$.
\end{corollary}

\begin{proof}
Indeed, every maximal point $x$ is reduced (hence $\sh{O}_{X,x}$ is a field), hence belongs to $\mathrm{Reg}(X)$, hence is interior to $\mathrm{Reg}(X)$ by hypothesis, and \emph{a fortiori} is interior to $\mathrm{Nor}(X)$.
\end{proof}

\begin{proposition}[6.13.6]
\label{IV.6.13.6}
\oldpage[IV-2]{169}
Let $A$ be an integral Noetherian ring, $K$ its field of fractions, $K'$ a finite extension of $K$, $A'$ the integral closure of $A$ in $K'$.
For $A'$ to be a finite $A$-algebra, it is necessary and sufficient that the two following conditions be satisfied:
\begin{enumerate}
  \item[\rm(i)] There exists $f\neq 0$ in $A$ such that the integral closure $A'_f$ of the ring of fractions $A_f$ is a finite $A_f$-algebra.
  \item[\rm(ii)] For every prime ideal $\mathfrak{p}\in\Spec(A)$, the integral closure $A'_\mathfrak{p}$ of the local ring $A_\mathfrak{p}$ is a finite $A_\mathfrak{p}$-algebra.
\end{enumerate}
\end{proposition}

\begin{proof}
The conditions are necessary, for if $S$ is any multiplicative subset of $A$, $S^{-1}A'$ is the integral closure of $S^{-1}A$ in $K'$ and is by hypothesis a $S^{-1}A$-module of finite type.
To see that the conditions are sufficient, consider the increasing filtrant family $(B_\lambda)$ of sub-$A$-algebras of $K'$ which are finite $A$-algebras and have $K'$ as field of fractions.
Set $Y=\Spec(A)$, $X_\lambda=\Spec(B_\lambda)$, and denote by $u_\lambda$ the morphism $X_\lambda\to Y$, and let $S_\lambda=X_\lambda-\mathrm{Nor}(X_\lambda)$, $T_\lambda=u_\lambda(S_\lambda)$.
We can restrict to those $B_\lambda$ which contain a finite system of generators of the $A$-module $A'_f$; this ensures that $(B_\lambda)_f=A'_f$ for all $\lambda$, or equivalently that $u_\lambda^{-1}(\mathrm{D}(f))$ is contained in $\mathrm{Nor}(X_\lambda)$.
By \sref{IV.6.13.2}, $S_\lambda$ is therefore \emph{closed} in $X_\lambda$, and as $u_\lambda$ is a finite morphism, $T_\lambda$ is \emph{closed} in $Y$.
But for every $\mathfrak{p}\in\Spec(A)$, there exists by hypothesis (ii) a $\lambda$ such that $(B_\lambda)_\mathfrak{p}=A'_\mathfrak{p}$, and hence all the points of $X_\lambda$ above $\mathfrak{p}$ belong to $\mathrm{Nor}(X_\lambda)$; in other words on a $\bigcap_\lambda T_\lambda=\varnothing$;
as $Y$ is Noetherian and the $T_\lambda$ are closed, there exists $\lambda_0$ such that $T_{\lambda_0}=\varnothing$, hence $B_{\lambda_0}=A'$. \textsc{q.e.d.}
\end{proof}

\begin{proposition}[6.13.7]
\label{IV.6.13.7}
Let $A$ be a Noetherian ring, $X=\Spec(A)$.
The following conditions are equivalent:
\begin{enumerate}
  \item[\rm a)] For every prescheme $X'$ locally of finite type over $X$, $\mathrm{Nor}(X')$ is open in $X'$.
  \item[\rm b)] For every finite integral $A$-algebra $A'$, $\mathrm{Nor}(\Spec(A'))$ is open in $\Spec(A')$.
  \item[\rm c)] For every prime ideal $\mathfrak{p}$ of $A$ and every finite radiciel extension $K'$ of the field of fractions $K$ of $A/\mathfrak{p}$, there exists a sub-$A$-algebra $A'$ of $K'$, finite over $A$, having $K'$ as field of fractions and such that $\mathrm{Nor}(\Spec(A'))$ is open in $\Spec(A')$.
\end{enumerate}
\end{proposition}

\begin{proof}
The fact that a) implies b) and that b) implies c) is proved as in \sref{IV.6.12.4}.
To show that c) implies a), by \sref{IV.6.13.2}, we are reduced as in \sref{IV.6.12.4} to proving that if $A'$ is an integral $A$-algebra of finite type, the generic point of $\Spec(A')$ is interior to $\mathrm{Nor}(\Spec(A'))$.
We proceed as in \sref{IV.6.12.4} by first proving the following:
\end{proof}

\begin{lemma}[6.13.7.1]
\label{IV.6.13.7.1}
Let $A$ be an integral Noetherian ring, $A'$ an integral $A$-algebra of finite type, containing $A$ and such that the field of fractions $K'$ of $A'$ is a separable extension of the field of fractions $K$ of $A$.
If $\mathrm{Nor}(\Spec(A))$ is open in $\Spec(A)$, then $\mathrm{Nor}(\Spec(A'))$ is open in $\Spec(A')$.
\end{lemma}

\begin{proof}
We must prove (by \sref{IV.6.13.2}) that the generic point of $\Spec(A')$ is interior to $\mathrm{Nor}(\Spec(A'))$.
The proof follows the same approach as that of \sref{IV.6.12.4}, I), and we preserve the notations.
The proof of \sref{IV.6.12.4}, I) shows (by replacing $A$ by an $A_f$ with $f\neq 0$) that the fibres $g^{-1}(z)$ of the morphism $g:X'\to X$ are regular and \emph{a fortiori} normal.
Moreover $g$ is flat and $X$ is normal, hence \sref[0]{5.4.2}, (ii)) $X'$ is normal.
This lemma being established, we pass to the general case as in \sref{IV.6.12.4}, II), whose notations we still preserve; applying hypothesis $c)$, we see each time that $\mathrm{Nor}(X'_1)$ is open and we are reduced as before to the case where $X'$ is normal and $g:X'\to X'$ flat and surjective; we conclude each time that $X'$ is normal with the aid of \sref[0]{5.4.2}, (i)).
\end{proof}

\subsection{Base change and integral closure}
\label{subsection:IV.6.14}

\begin{proposition}[6.14.1]
\label{IV.6.14.1}
Let $Y$, $Y'$ be two locally Noetherian preschemes, $g:Y'\to Y$ a \emph{normal} morphism \sref{IV.6.8.1}.
Then, for every normal $Y$-prescheme $X$, the prescheme $X'=X\times_Y Y'$ is normal.
\end{proposition}

\begin{proof}[Proof sketch]
We note that in this statement we do not suppose $X$ locally Noetherian.
The proof proceeds in several stages.
\end{proof}

\begin{lemma}[6.14.1.1]
\label{IV.6.14.1.1}
Let $R$ be a ring that is a finite direct product of a finite number of fields.
\begin{enumerate}
  \item[\rm(i)] For a sub-ring $A$ of $R$ having $R$ as its total ring of fractions to be normal, it is necessary and sufficient that it be integrally closed in $R$.
  \item[\rm(ii)] Let $(A_\lambda)$ be a family of normal sub-rings of $R$; if $A=\bigcap_\lambda A_\lambda$ admits $R$ as total ring of fractions, $A$ is normal.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) As $A$ is a sub-ring of $R$, $A_\mathfrak{p}$ is a sub-ring of $R_\mathfrak{p}$ for every prime ideal $\mathfrak{p}$ of $A$; moreover $R_\mathfrak{p}$ is a ring of fractions of $A_\mathfrak{p}$, and is necessarily a finite direct product of fields, hence every element which is not a zero divisor of $0$ in $R_\mathfrak{p}$ is invertible; this proves that $R_\mathfrak{p}$ is the total ring of fractions of $A_\mathfrak{p}$.
If $A$ is integrally closed in $R$, $A_\mathfrak{p}$ is hence integrally closed in $R_\mathfrak{p}$; but if $R_\mathfrak{p}$ is a finite direct product of $2$ fields at least, the integral closure of $A_\mathfrak{p}$ in $R_\mathfrak{p}$ is a direct product of $2$ rings at least which are not reduced to $0$, which is absurd since $A_\mathfrak{p}$ is a local ring; hence $R_\mathfrak{p}$ is necessarily a field and $A_\mathfrak{p}$ is integral and integrally closed, which by definition means that $A$ is normal.
Conversely, if $A$ is normal, $R_\mathfrak{p}$, total ring of fractions of an integral ring $A_\mathfrak{p}$, is a field and $A_\mathfrak{p}$ is integrally closed in $R_\mathfrak{p}$; if $x\in R$ is an integer over $A$, its image in each $R_\mathfrak{p}$ belongs to $A_\mathfrak{p}$, hence $x\in A$ (Bourbaki, \emph{Alg.\ comm.}, chap.~II, \textsection~3, n\textsuperscript{o}~1, cor.~1 du th.~1), and hence $A$ is integrally closed in $R$.

(ii) As $A\subset A_\lambda\subset R$ for each $\lambda$, $R$ is also the total ring of fractions of each $A_\lambda$.
In view of the characterisation of normal rings having $R$ as total ring of fractions given in (i), assertion (ii) follows from Bourbaki, \emph{Alg.\ comm.}, chap.~V, \textsection~1, n\textsuperscript{o}~3, prop.~12.
\end{proof}

The proof of \sref{IV.6.14.1} proceeds in several stages:

\medskip\noindent
I) \emph{Reduction to the case where} $Y=\Spec(A)$, $Y'=\Spec(A')$, $X=\Spec(B)$, $A$, $A'$ \emph{are local Noetherian rings, $A$ integral, $B$ the integral closure of $A$.} ---
We must prove that for every $x'\in X'$, the local ring $\sh{O}_{X',x'}$ is integral and integrally closed; let $x$, $y$, $y'$ be the images of $x'$ in $X$, $Y$, $Y'$ respectively; if we set $A=\sh{O}_{Y,y}$, $A'=\sh{O}_{Y',y'}$, $B=\sh{O}_{X,x}$, the rings $\sh{O}_{X',x'}$, for $x$, $y$, $y'$ fixed, are the localisations of $B'=B\otimes_A A'$ at the prime ideals of $B'$ $(\textbf{I}$, 3.6.5), and it suffices to prove that $B'$ is a \emph{normal} ring.
Note moreover that $\Spec(A')\to\Spec(A)$ is a normal morphism by definition \sref{IV.6.8.1} and $\textbf{I}$, 3.6.5); one is hence reduced to the case of local schemes.
Denote by $(B_\alpha)$ the family of \emph{integral closures} of the sub-$A$-algebras \emph{of finite type} of $B$;
it is clear that $B$ is the union of the increasing filtrant family of the $B_\alpha$.
As $\varinjlim$ commutes with the tensor product, $B'$ is hence isomorphic to $\varinjlim B'_\alpha$, where we have set $B'_\alpha=B_\alpha\otimes_A A'$.
If we prove that $C'_\alpha$ is the integral closure of $B'_\alpha$ in $C'$, then it will follow that $C'$ is the integral closure of $A'$ in $B'$.
We can hence restrict to the case where $B$ is a $A$-algebra of finite type; as $B$ is supposed \emph{reduced}, it is sufficient to prove the claim when $B$ is \emph{integral} and is the \emph{integral closure} of $A$.

\medskip\noindent
II)-V) The remaining reductions and the completion of the proof are technically involved.
In II), one reduces to the case where $A$ is a local integral ring of dimension $1$, $B$ a discrete valuation ring, and the morphism $\Spec(B)\to\Spec(A)$ \emph{radiciel}.
In III), one establishes a key lemma:

\begin{lemma}[6.14.1.2]
\label{IV.6.14.1.2}
Let $A$ be a local Noetherian integral ring of dimension $1$, $A'$ a local Noetherian ring, $A\to A'$ a local homomorphism such that the corresponding morphism $\Spec(A')\to\Spec(A)$ is \emph{normal}.
Let $K$ be the field of fractions of $A$, $\mathfrak{m}$ the maximal ideal of $A$, $k=A/\mathfrak{m}$ its residue field.
\begin{enumerate}
  \item[\rm(i)] If $\mathfrak{q}'_j$ $(1\leq j\leq r)$ are the minimal ideals of $A'$, $K'=K\otimes_A A'$ is a finite direct product of integral and integrally closed rings $K'_j\supset A'/\mathfrak{q}'_j$ $(1\leq j\leq r)$, $K_j$ having for field of fractions the field of fractions $L'_j$ of $A'/\mathfrak{q}'_j$, so that the total ring of fractions $L'$ of $A'$ identifies to the direct product of the $L'_j$ and $A'$ to a sub-ring of $K'$.
  \item[\rm(ii)] The ideal $\mathfrak{p}'=\mathfrak{m} A'$ of $A'$ is prime; the ring $A'_{\mathfrak{p}'}$ is of dimension $1$ and identifies to a sub-product of the fields $L'_j$ for the indices $j$ such that $\mathfrak{q}'_j\subset\mathfrak{p}'$.
  \item[\rm(iii)] If $\overline{A}'_{\mathfrak{p}'}$ denotes the sub-ring of $L'$, product of $A'_{\mathfrak{p}'}$ and of the $L'_j$ such that $\mathfrak{q}'_j\not\subset\mathfrak{p}'$, one has
\[
  A'=K'\cap\overline{A}'_{\mathfrak{p}'}.
\]
\end{enumerate}
\end{lemma}

\begin{lemma}[6.14.1.4]
\label{IV.6.14.1.4}
Let $R$ be a ring, $\mathfrak{a}$ and $\mathfrak{b}$ two ideals of $R$; if $R/\mathfrak{a}$ and $R/\mathfrak{b}$ are Noetherian, so is $R/(\mathfrak{a}\cap\mathfrak{b})$.
\end{lemma}

\begin{proof}
Indeed, let $\mathfrak{c}$ be an ideal of $R$ such that $\mathfrak{a}\cap\mathfrak{b}\subset\mathfrak{c}$; we have therefore $\mathfrak{a}\cap\mathfrak{b}\subset\mathfrak{a}\cap\mathfrak{c}\subset\mathfrak{c}$;
but $\mathfrak{c}/(\mathfrak{a}\cap\mathfrak{c})$ is an $R$-module isomorphic to $(\mathfrak{a}+\mathfrak{c})/\mathfrak{a}$, hence an ideal of $R/\mathfrak{a}$, and is therefore of finite type; on the other hand $(\mathfrak{a}\cap\mathfrak{c})/(\mathfrak{a}\cap\mathfrak{b})$ is a sub-$R$-module of $\mathfrak{a}/(\mathfrak{a}\cap\mathfrak{b})$, and this last is isomorphic to $(\mathfrak{a}+\mathfrak{b})/\mathfrak{b}$, hence is of finite type as ideal of $R/\mathfrak{b}$;
hence $(\mathfrak{a}\cap\mathfrak{c})/(\mathfrak{a}\cap\mathfrak{b})$ is of finite type, and it is so finally for $\mathfrak{c}/(\mathfrak{a}\cap\mathfrak{b})$.
\end{proof}

\oldpage[IV-2]{173}

The proof of \sref{IV.6.14.1} is completed using these lemmas along with the normality criteria of \sref{IV.6.8.2} and the results of \sref{IV.5.13.6}.

\begin{corollary}[6.14.2]
\label{IV.6.14.2}
Let $k$ be a field, $X$ a normal $k$-prescheme (not necessarily locally Noetherian).
Then, for every separable extension $k'$ of $k$, $X_{(k')}=X\otimes_k k'$ is normal.
\end{corollary}

\begin{proof}
We know indeed by \sref{IV.6.7.6} that the morphism $\Spec(k')\to\Spec(k)$ is then normal.
\end{proof}

\begin{corollary}[6.14.3]
\label{IV.6.14.3}
Let $k$ be a field, $X$, $Y$ two integral $k$-preschemes, normal, whose fields of rational functions $K=R(X)$, $L=R(Y)$ are separable extensions of $k$.
Then the prescheme $X\times_k Y$ is normal.
\end{corollary}

\begin{proof}
The question being local on $X$ and $Y$, we can suppose that $X=\Spec(A)$, $Y=\Spec(B)$, where $A$ and $B$ are two integral and integrally closed rings, of fields of fractions $K$ and $L$ respectively;
suppose first that $A$ is a $k$-algebra of finite type, hence $K$ a finite type extension of $k$.
By flatness, $A\otimes_k B$ is a sub-$k$-algebra of $K\otimes_k L$;
but $K\otimes_k L$ is a Noetherian normal ring (by \sref{IV.6.7.4}, I) or \sref{IV.6.14.2}),
hence a finite direct product of integral rings $C_i$, so that if $E_i$ is the field of fractions of $C_i$, the direct product $E$ of the $E_i$ is the total ring of fractions of $K\otimes_k L$ and also of $A\otimes_k B$.
This shows that $A\otimes_k L$ and $K\otimes_k B$ are normal (as they are intersections of the $C_i$ with appropriate direct factors),
and it follows from \sref{IV.6.14.2} that $A\otimes_k B=(A\otimes_k L)\cap(K\otimes_k B)$, hence $A\otimes_k B$ is normal by \sref{IV.6.14.1.1}.

In the general case, $A$ is the union of the increasing filtrant family of its sub-$k$-algebras of finite type $A_\alpha$, and $A\otimes_k B$ is the limit of the $A_\alpha\otimes_k B$.
It follows from \sref[0]{5.13.6} that to prove that every irreducible component of $\Spec(A_\beta\otimes_k B)$ dominates an irreducible component of $\Spec(A'_\alpha)$ for $\alpha\leq\beta$, it suffices to use \sref[I]{2.3.7}, (ii) and the fact that $A_\alpha$ and $A_\beta$ are integral rings.
\end{proof}

\oldpage[IV-2]{174}

\begin{proposition}[6.14.4]
\label{IV.6.14.4}
Let $A$ be a Noetherian ring, $A'$ a Noetherian $A$-algebra such that the morphism $\Spec(A')\to\Spec(A)$ is \emph{normal}.
Let $B$ be an $A$-algebra, $C$ the integral closure of $A$ in $B$; set $B'=B\otimes_A A'$, $C'=C\otimes_A A'$, $C'$ identifying to a sub-ring of $B'$; then $C'$ is the integral closure of $A'$ in $B'$.
\end{proposition}

\begin{proof}
The proof proceeds in several steps:

\medskip\noindent
I) \emph{Reduction to the case where $B$ is reduced.} ---
Set $B_0=B_{\mathrm{red}}=B/\mathfrak{N}$, where $\mathfrak{N}$ is the nilradical of $B$, and let $C_0$ be the integral closure of $A$ in $B_0$.
By \sref{IV.6.14.4.1} below, the inverse image $C$ of $C_0$ by the canonical homomorphism $\varphi:B\to B_0$ is the integral closure of $A$ in $B$.

\medskip\noindent
II) \emph{Reduction to the case where $B$ is an integral $A$-algebra of finite type containing $A$.} ---
Let $(B_\alpha)$ be the increasing filtrant family of sub-$A$-algebras of finite type of $B$, and let $C_\alpha$ be the integral closure of $A$ in $B_\alpha$.
It follows from the definition of integral closure that $C$ is the union of the $C_\alpha$; if we set $B'_\alpha=B_\alpha\otimes_A A'$, $C'_\alpha=C_\alpha\otimes_A A'$, $C'_\alpha$ is contained in $B'_\alpha$ and it suffices to show that $C'_\alpha$ is the integral closure of $A'$ in $B'_\alpha$.

\medskip\noindent
III-V) The remaining cases reduce eventually to the case where $A$ is a field and $B$ is algebraically closed over $A$, where the conclusion follows from \sref{IV.6.14.1}.
\end{proof}

\begin{lemma}[6.14.4.1]
\label{IV.6.14.4.1}
Let $A$ be a ring, $B$ an $A$-algebra, $B_0=B/\mathfrak{n}$ the quotient of $B$ by a nilideal.
If $C_0$ is the integral closure of $A$ in $B_0$, the inverse image $C$ of $C_0$ by the canonical homomorphism $\varphi:B\to B_0$ is the integral closure of $A$ in $B$.
\end{lemma}

\begin{proof}
Indeed, if $x\in B$ is such that $\varphi(x)$ satisfies a monic integral dependence equation with coefficients in $A$, we deduce that $x$ satisfies a relation of the form $x^n+a_1 x^{n-1}+\ldots+a_n\in\mathfrak{n}$ with $a_i\in A$, whence, raising to a sufficiently large power, we obtain a monic integral dependence equation for $x$ with coefficients in $A$.
We have therefore an exact sequence of $A$-modules
$0\to C\to B\to B_0/C_0$,
hence, by flatness, the exact sequence
$0\to C'\to B'\to B'_0/C'_0$,
where we have set $B'_0=B_0\otimes_A A'$, $C'_0=C_0\otimes_A A'$.
If we prove that $C'_0$ is the integral closure of $A'$ in $B'_0$, the lemma \sref{IV.6.14.4.1} will show that $C'$ is the integral closure of $A'$ in $B'$.
\end{proof}

\begin{corollary}[6.14.5]
\label{IV.6.14.5}
Let $A$ be a Noetherian ring, $A'$ a Noetherian $A$-algebra such that the morphism $\Spec(A')\to\Spec(A)$ is \emph{normal}.
Let $B$, $C$ be two $A$-algebras, $\varphi:B\to C$ an $A$-homomorphism making $C$ a $B$-algebra, $D$ the integral closure of $B$ in $C$; if we set $B'=B\otimes_A A'$, $C'=C\otimes_A A'$, $D'=D\otimes_A A'$, $D'$ is the integral closure of $B'$ in $C'$.
\end{corollary}

\begin{proof}
Let $(B_\lambda)$ be the increasing filtrant family of sub-$A$-algebras of finite type of $B$, and set $B'_\lambda=B'_\lambda\otimes_A A'$, so that $B'$ is the union of the filtrant increasing family of the sub-$A'$-algebras of finite type $B'_\lambda$.
Let $D_\lambda$ be the integral closure of $B_\lambda$ in $C$, and set $D'_\lambda=D_\lambda\otimes_A A'$; if we prove that $D'_\lambda$ is the integral closure of $B'_\lambda$ in $C'$, it will follow that $D'$ is the integral closure of $B'$ in $C'$.
But $B_\lambda$ and $B'_\lambda$ are Noetherian and the morphism $\Spec(B'_\lambda)\to\Spec(B_\lambda)$ is normal \sref{IV.6.8.2}; we can hence apply \sref{IV.6.14.4} by replacing $A$, $A'$ and $B$ by $B_\lambda$, $B'_\lambda$ and $C$ respectively, hence the corollary.
\end{proof}

One can for example apply \sref{IV.6.14.5} when $A$ is a local ring and $A'$ its completion $\widehat{A}$, a \emph{local excellent ring}, since in this case $\Spec(A')\to\Spec(A)$ is a regular morphism (7.8.2).

\subsection{Geometrically unibranch preschemes}
\label{subsection:IV.6.15}

\begin{env}[6.15.1]
\label{IV.6.15.1}
We say that a prescheme $X$ is \emph{unibranch} (resp.\ \emph{geometrically unibranch}) at a point $x$, or that the point $x$ is \emph{unibranch} (resp.\ \emph{geometrically unibranch}) in $X$ if the local ring $\sh{O}_{X,x}$ is unibranch (resp.\ geometrically unibranch) $(\mathbf{0}$, 23.2.1);
we say that $X$ is \emph{unibranch} (resp.\ \emph{geometrically unibranch}) if it is so at every point.
It follows from this definition that, for $X$ to be unibranch (resp.\ geometrically unibranch) at a point, it is necessary and sufficient that $X_{\mathrm{red}}$ be so at this point.
\end{env}

\begin{env}[6.15.2]
\label{IV.6.15.2}
It should be noted that with the definitions of \sref{IV.6.15.1}, a \emph{local} ring $A$ can be unibranch (resp.\ geometrically unibranch) \emph{without} $\Spec(A)$ being so; in other words, it can happen that there exist prime ideals $\mathfrak{p}$ of $A$ such that $A_\mathfrak{p}$ is not unibranch; this also amounts to saying that the notion of a geometrically unibranch point on a prescheme \emph{is not stable by generisation}.
We shall see this in the following example.
\end{env}

\begin{env}[6.15.3]
\label{IV.6.15.3}
We say, to abbreviate, that a morphism $f:X\to Y$ is \emph{radiciel at a point} $y\in Y$, if $f^{-1}(y)$ is empty or reduced to a single point $x$ and if $k(x)$ is a radiciel extension of $k(y)$.
It amounts to the same to say that $f$ is radiciel at all the points of $Y$, or that $f$ is radiciel $(\textbf{I}$, 3.5.8).
If $g:f^{-1}(y)\to\Spec(k(y))$ is the morphism obtained from $f$ by the base change $\Spec(k(y))\to Y$, it amounts to the same to say that $f$ is radiciel at the point $y\in Y$, or that $g$ is radiciel.
\end{env}

\begin{lemma}[6.15.3.1]
\label{IV.6.15.3.1}
\begin{enumerate}
  \item[\rm(i)] Let $f:X\to Y$, $g:Y\to Z$ be two morphisms.
  If $f$ is radiciel at a point $y$ and $g$ radiciel at the point $z=g(y)$, then $g\circ f$ is radiciel at the point $g(y)$.
  The converse is true if $f$ is surjective.
  \item[\rm(ii)] Let $f:X\to Y$, $h:Y'\to Y$ be two morphisms, and let $f'=f_{(Y')}:X\times_Y Y'\to Y'$.
  Let $y'\in Y'$ and $y=h(y')$; then for $f$ to be radiciel at the point $y$, it is necessary and sufficient that $f'$ be radiciel at the point $y'$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) If $f$ is radiciel at $y$ and $g$ radiciel at $z=g(y)$, $g^{-1}(z)$ is reduced to $y$ and $f^{-1}(g^{-1}(z))=f^{-1}(y)$ is empty or reduced to a single point $x$; moreover $k(x)$ is radiciel extension of $k(z)$, hence $k(x)$ is radiciel extension of $k(z)$.
Conversely, if $f$ is surjective and $g\circ f$ is radiciel at $z=g(y)$, $g^{-1}(z)$ is reduced to at most two points; but $f^{-1}(g^{-1}(z))$ would have at least two distinct points, a contradiction; furthermore, $f^{-1}(y)=f^{-1}(g^{-1}(z))$ is reduced to a single point $x$, and by hypothesis $k(z)\subset k(y)\subset k(x)$ with $k(x)$ radiciel over $k(z)$, hence $k(y)$ radiciel over $k(z)$ and $k(x)$ radiciel over $k(y)$.

(ii) Let $g:f^{-1}(y)\to\Spec(k(y))$, $g':f'^{-1}(y')\to\Spec(k(y'))$ be the fibre morphisms of $f$ and $f'$ respectively; as $f'^{-1}(y')=f^{-1}(y)\otimes_{k(y)} k(y')$ $(\textbf{I}$, 3.6.4), $g'$ is the base change of $g$, and the assertion follows from \sref[I]{2.6.1}, (v)).
\end{proof}

\begin{env}[6.15.4]
\label{IV.6.15.4}
Generalising the definition given in $(\textbf{I}$, 2.2.9), for two reduced preschemes $X$, $Y$, we say that a morphism $f:X\to Y$ is \emph{birational} if the restriction of $f$ to the set of maximal points of $X$ is a bijection onto the set of maximal points of $Y$, and if, for every maximal point $y$ of $Y$, the morphism $X\times_Y\Spec(\sh{O}_y)\to\Spec(\sh{O}_y)$ induced by $f$ is an isomorphism (in other words, if the fibre $f^{-1}(y)$ is reduced to a single point $x$ (maximal in $X$) and if the homomorphism $k(y)\to k(x)$ induced by $f$ is bijective); this notion reduces to that of $(\textbf{I}$, 2.2.9) when $X$ and $Y$ have only a \emph{finite} number of irreducible components.
\end{env}

\begin{lemma}[6.15.4.1]
\label{IV.6.15.4.1}
Let $f:X\to Y$ be a morphism, $g:Y'\to Y$ a \emph{flat} morphism.
If $f_{\mathrm{red}}$ is birational, so is $f'_{\mathrm{red}}$.
\end{lemma}

\begin{proof}
Indeed, if $y'$ is a maximal point of $Y'$, we know \sref[I]{2.3.4}, (ii)) that $y=g(y')$ is a maximal point of $Y$; there exists a unique point $x$ of $X$ in $f^{-1}(y)$ by hypothesis; since moreover $x$ is maximal in $X$ and $k(x)=k(y)$.
It follows therefore by $(\textbf{I}$, 3.4.9) that $f'^{-1}(y')$ is reduced to a single point $x'$ and that $k(x')=k(y')$.
We conclude by the remark that, by \sref[I]{2.3.7}, (ii)), every irreducible component of $X'$ dominates one irreducible component of $Y'$.
\end{proof}

\begin{proposition}[6.15.5]
\label{IV.6.15.5}
Let $f:X\to Y$ be a morphism such that $f_{\mathrm{red}}$ is entire and birational \sref{IV.6.15.4}, hence surjective.
\begin{enumerate}
  \item[\rm(i)] For $Y$ to be geometrically unibranch at a point $y$, it is necessary and sufficient that $f$ be radiciel at the point $y$ and that $X$ be geometrically unibranch at the unique point $x$ of $f^{-1}(y)$.
  \item[\rm(ii)] For $Y$ to be geometrically unibranch, it is necessary and sufficient that $X$ be geometrically unibranch and that $f$ be radiciel.
\end{enumerate}
\end{proposition}

\begin{proof}
We can evidently restrict to the case where $X$ and $Y$ are \emph{reduced}; the fact that $f$ is surjective results from $f$ being a closed morphism $(\textbf{II}$, 6.1.10) and from $f(X)$ containing all the maximal points of $Y$.
This immediately shows that (i) implies (ii).
Using $(\textbf{I}$, 3.6.5) and $(\textbf{II}$, 6.1.5), we can suppose, to prove (i), that $Y=\Spec(\sh{O}_{X,y})$, and similarly $X=\Spec(B)$.
If $A$ is geometrically unibranch, it is \emph{integral}, and hence $B$ is also integral (since $f$ is birational).
Conversely, if $f$ is radiciel at $y$ and $X$ is geometrically unibranch at $x$, $B$ has a single maximal ideal (since every ideal of $A$ lies above the maximal of $A$ and $A$ is local), hence $B$ is a local ring, \emph{integral}.
As $f$ is dominant by hypothesis and $A$ is reduced $(\textbf{I}$, 1.2.7), we conclude that $A$ is also \emph{integral}.
Thus, to prove (i), it amounts to showing that $A$ is integral and geometrically unibranch au point $x$ means $B$ is integral and $B$ is the integral closure, i.e.\ the morphism $g:Y\to X$ is radiciel at the point $x$ \sref{IV.6.15.3}.
\end{proof}

\begin{proposition}[6.15.6]
\label{IV.6.15.6}
Let $k$ be a field, $X$ a $k$-prescheme.
If $X$ is normal, then, for every extension $k'$ of $k$, $X'=X\otimes_k k'$ is geometrically unibranch.
\end{proposition}

\begin{proof}
We know that $k'$ is the algebraic extension of a pure extension $k_0$ of $k$, and if $k''$ is the largest separable extension of $k_0$ contained in $k'$, $k'$ is a radiciel extension of $k''$.
We know \sref{IV.6.14.2} that $X\otimes_k k''$ is normal; as $X''\otimes_{k''} k'=X''\otimes_{k''} k'$, we see that we can restrict to the case where the extension $k'$ of $k$ is \emph{radiciel}.
Moreover $(\textbf{I}$, 3.6.5), we can suppose that $X=\Spec(A)$, where $A$ is a local integral and integrally closed ring (since $X$ is normal).
The projection morphism $f:X'\to X$ is a universal homeomorphism, since $\Spec(k')\to\Spec(k)$ is a universal homeomorphism \sref[I]{2.4.5}; as $X'=\Spec(A')$ where $A'=A\otimes_k k'$, we see that $A'$ is a local ring whose nilradical $\mathfrak{N}'$ is the unique minimal prime ideal, hence $X'_{\mathrm{red}}=\Spec(A'_0)$, where $A'_0=A'/\mathfrak{N}'$ is a local integral ring; moreover, if $K$ is the field of fractions of $A$, the field of fractions $K'_0$ of $A'_0$ is radiciel over $K$, since the morphism $f$ is radiciel.
As $A'_0$ is entire over $A$, its integral closure $B$ is also the integral closure of $A$ in $K'_0$; moreover, every element of $K'_0$ whose $p^m$-th power (for some $m$) belongs to $A$ belongs to $K$ (for $p$ the characteristic exponent of $k$); in particular $B$ is an integral closure of $A$ in $K'_0$.
But as $A$ is integrally closed and $K'_0$ is radiciel over $K$, every element of $K'_0$ integral over $A$ belongs to $A$ (Bourbaki, \emph{Alg.\ comm.}, chap.~V, \textsection~2, n\textsuperscript{o}~3, lemme~3), and \emph{a fortiori} $A'_0$ is geometrically unibranch, and it is so also of $X'$.
\end{proof}

\begin{proposition}[6.15.7]
\label{IV.6.15.7}
Let $k$ be a field, $X$ a $k$-prescheme, $k'$ an extension of $k$, $X'=X\otimes_k k'$.
Let $x'$ be a point of $X'$, $x$ its projection in $X$.
For $X$ to be geometrically unibranch at the point $x$, it is necessary and sufficient that $X'$ be geometrically unibranch at the point $x'$.
For $X$ to be geometrically unibranch, it is necessary and sufficient that $X'$ be so.
\end{proposition}

\begin{proof}
The second assertion follows from the first and from the fact that the projection $f:X'\to X$ is surjective.
To demonstrate the first assertion, we can, by $(\textbf{I}$, 5.1.8), restrict to the case where $X$ is reduced, and by $(\textbf{I}$, 3.6.5), to the case where $X=\Spec(A)$ (with $A=\sh{O}_{X,x}$) is a local scheme.
Note that $A'=\sh{O}_{X',x'}$ is by hypothesis an $A$-module faithfully flat, hence containing $A$, and hence the hypothesis that $X$ is geometrically unibranch at $x$, or the hypothesis that $X'$ is geometrically unibranch at $x'$, both imply that $A$ is a local integral ring (since $A$, being reduced, is also isomorphic to a sub-ring of $A'_{\mathrm{red}}$).
Let $K$ be the field of fractions of $A$, $B$ the integral closure of $A$, and set $Y=\Spec(B)$;
it amounts to saying that $X$ is geometrically unibranch at the point $x$, or that the morphism $g:Y\to X$ is radiciel at the point $x$ \sref{IV.6.15.3}.
Let $Y'=Y\otimes_k k'=Y\times_X X'$, so that we have the commutative diagram:
\[
\begin{tikzcd}
  Y \arrow[leftarrow]{r}{h} & Y' \\
  X \arrow[swap]{u}{g} \arrow[leftarrow]{r}{f} & X' \arrow[swap]{u}{g'}
\end{tikzcd}
\]
Note that $f$ is a flat morphism; hence by \sref{IV.6.15.4.1} the entire morphism $g'_{\mathrm{red}}$ is birational.
Moreover, as $Y$ is normal, $Y'$ is geometrically unibranch by \sref{IV.6.15.6}.
As $f$ is entire and birational, $f$ is radiciel at this point \sref{IV.6.15.5}; on the other hand, $x$ is the projection of $x'$ in $X$ and it is equivalent to say that $g$ is radiciel at the point $x$ or that $g'$ is radiciel at the point $x'$ \sref{IV.6.15.3.1};
finally, for $X$ to be geometrically unibranch at $x$, it suffices that $g$ be radiciel at this point, hence the proposition.
\end{proof}

\begin{lemma}[6.15.8]
\label{IV.6.15.8}
Let $k$ be a \emph{separably closed} field (in other words, such that the algebraic closure of $k$ is \emph{radiciel} over $k$), $X$ a $k$-prescheme locally of finite type over $k$, $x$ a closed point of $X$.
If $X$ is unibranch at the point $x$, it is geometrically unibranch at this point.
\end{lemma}

\begin{proof}
Indeed, we know $(\textbf{I}$, 6.4.2) that $k(x)$ is an algebraic extension of $k$; as the integral closure of $(\sh{O}_x)_{\mathrm{red}}$ (integral closure which is by hypothesis a local ring) is an extension \emph{algebraic} of $k(x)$, it is therefore an extension \emph{radiciel} of $k(x)$, hence also of $k(x)$ by hypothesis, hence also of $k(x)$.
\end{proof}

\begin{corollary}[6.15.9]
\label{IV.6.15.9}
Let $k$ be a field, $X$ a $k$-prescheme locally of finite type over $k$, $k'$ a separably closed extension of $k$ (in other words, such that the algebraic closure of $k'$ is radiciel over $k'$); then, for $X$ to be geometrically unibranch, it is necessary and sufficient that $X'=X\otimes_k k'$ be unibranch.
\end{corollary}

\begin{proof}
By \sref{IV.6.15.7}, we are reduced to proving that if $X$ is unibranch and $k$ separably closed, then $X$ is geometrically unibranch.
But $X$ is geometrically unibranch at its closed points \sref{IV.6.15.8};
we conclude from (10.4.9) that $X$ is geometrically unibranch at all its points, relying on the fact (noted at the end of \sref{IV.6.15.2}) that the set of points where $X$ is geometrically unibranch is constructible (the corollary \sref{IV.6.15.9} will of course not be used up to this point).
\end{proof}

\begin{proposition}[6.15.10]
\label{IV.6.15.10}
Let $Y$, $Y'$ be two locally Noetherian preschemes, $g:Y'\to Y$ a normal morphism \sref{IV.6.8.1}, $f:X\to Y$ a morphism; set $X'=X\times_Y Y'$ and let $p:X'\to X$ be the canonical projection, $x'$ a point of $X'$.
Then, if $X$ is reduced (resp.\ geometrically unibranch) at the point $x=p(x')$, $X'$ is reduced (resp.\ geometrically unibranch, resp.\ integral and geometrically unibranch) at the point $x'$.
\end{proposition}

\begin{proof}
By $(\textbf{I}$, 3.6.5), we can restrict to the case where $Y=\Spec(A)$, $Y'=\Spec(A')$, $X=\Spec(B)$, where $A$, $A'$ and $B$ are local Noetherian rings, the homomorphisms $A\to A'$ and $A\to B$ being local, and $x$ being the closed point of $X$;
it suffices to prove that $B'=B\otimes_A A'$ is reduced (resp.\ geometrically unibranch, resp.\ integral and geometrically unibranch at its points).
Suppose first $B$ reduced only; $B$ is the union of the increasing filtrant family of its sub-$A$-algebras of finite type $B_\lambda$, and by flatness, $B'$ is the union of the filtrant increasing family of the sub-$A'$-algebras of finite type $B'_\lambda=B_\lambda\otimes_A A'$.
But, the morphism $\Spec(B'_\lambda)\to\Spec(B_\lambda)$ is normal \sref{IV.6.8.2}, and \emph{a fortiori} reduced; the fact that $B'_\lambda$ is reduced follows then; one concludes that $B'$ is itself reduced by \sref[0]{5.13.2}.

Suppose now $B$ geometrically unibranch; by $(\textbf{I}$, 5.1.8), we can moreover suppose $B$ reduced, hence integral.
Let $C$ be its integral closure, which is by hypothesis a local ring; set $Z=\Spec(C)$, $C'=C\otimes_A A'$, $Z'=\Spec(C')=Z\times_X X'$, so that we have the commutative diagram:
\[
\begin{tikzcd}
  Z \arrow[leftarrow]{r}{h} & Z' \\
  X \arrow[swap]{u}{l} \arrow[leftarrow]{r}{p} & X' \arrow[swap]{u}{l'}
\end{tikzcd}
\]
By the first part of the argument, $X'$ is \emph{reduced}; moreover, as $Z'=Z\times_Y Y'$, it follows from \sref{IV.6.14.1} that $Z'$ is \emph{normal} (and \emph{a fortiori} geometrically unibranch).
As $f$ is entire and birational, so is $f'$ by \sref{IV.6.15.4.1}, since $p$ is flat.
Finally, the hypothesis that $X$ is geometrically unibranch at $x$ implies that $f$ is radiciel at this point \sref{IV.6.15.5};
we conclude by \sref{IV.6.15.3.1} that $f'$ is radiciel at every point $x'\in p^{-1}(x)$, and it follows from \sref{IV.6.15.5} that $X'$ is geometrically unibranch (hence integral since it is reduced) at each of these points.
\end{proof}

\begin{remarks}[6.15.11]
\label{IV.6.15.11}
\begin{enumerate}
  \item[\rm(i)] In the proof of \sref{IV.6.15.10}, one cannot avoid appealing to \sref{IV.6.14.1}, even when $X=Y$, as one makes use of the integral closure of the ring $B$, which is not necessarily a Noetherian ring even if $B$ is.
  \item[\rm(ii)] The example \sref[0]{5.5.2}, (ii)) shows that in \sref{IV.6.15.10}, one cannot replace ``geometrically unibranch'' by ``integral'' in the statement, even if the residue field $k(x)$ is algebraically closed and the morphism $g$ is \'{e}tale.
\end{enumerate}
\end{remarks}
