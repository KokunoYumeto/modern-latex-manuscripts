\section{Change of base field for algebraic preschemes}
\label{section:IV.4}

\oldpage[IV-2]{52}

We will develop in \textsection\hyperref[section:IV.5]{5} the general theory of the dimension of preschemes; but the theory of the dimension of \emph{algebraic} preschemes can be developed in a more elementary way, and since it moreover presents numerous special features, we will give here a rapid exposition, independent of the general theory.

If $K$ is a field and $L$ an extension of $K$, we will denote by $\deg.\mathrm{tr}_K L$ the \emph{transcendence degree} of $L$ over $K$.

\subsection{Dimension of algebraic preschemes}
\label{subsection:IV.4.1}

\begin{definition}[4.1.1]
\label{IV.4.1.1}
Let $k$ be a field, $X$ a prescheme locally of finite type over $k$.
The \emph{dimension} of $X$ is defined to be the number
\begin{equation}
\label{IV.4.1.1.1}
  \dim X=\sup_x\deg.\mathrm{tr}_k\,k(x)\tag{4.1.1.1}
\end{equation}
where $x$ ranges over the set of maximal points of $X$.
\end{definition}

We will see in \textsection5 \sref{IV.5.2.2} that the definition \sref{IV.4.1.1} depends only in appearance on the base field $k$ over which $X$ is locally of finite type, and that the number $\dim(X)$ thus defined coincides with the dimension of the \emph{underlying space} $X$ (defined in $(\mathbf{0_{IV}}$, 14.1.2)).
We obviously have $\dim X=\dim X_\mathrm{red}$.

We note that each $k(x)$ is a finitely generated extension of $k$ \sref[I]{I.6.3.3} and therefore has a finite transcendence degree over $k$.
If $X$ is of finite type over $k$ and non-empty, it is Noetherian \sref[I]{I.6.3.7}, hence has only a finite number of irreducible components, and therefore $\dim(X)$ is finite and $\geq 0$; the definition \sref{IV.4.1.1} gives
\[
  \dim(\emp)=-\infty.
\]

It is clear that, if we denote by $(X_\alpha)$ the family of reduced closed subpreschemes of $X$ having as underlying spaces the irreducible components of $X$ \sref[I]{I.5.2.1}, then
\begin{equation}
\label{IV.4.1.1.2}
  \dim(X)=\sup_\alpha\dim(X_\alpha).\tag{4.1.1.2}
\end{equation}
This reduces the calculation of the dimension to the case of \emph{integral} preschemes (locally of finite type over $k$).
Finally, we obviously have
\begin{equation}
\label{IV.4.1.1.3}
  \dim(X)=\dim(U)\tag{4.1.1.3}
\end{equation}
for every sub-prescheme induced on an everywhere dense open subset $U$ of $X$; this finally reduces the notion of dimension to the case of an \emph{affine scheme of finite type} over $k$.

\begin{theorem}[4.1.2]
\label{IV.4.1.2}
Let $X$, $Y$ be two preschemes locally of finite type over a field $k$, $f:X\to Y$ a $k$-morphism.
\begin{enumerate}[\rm(i)]
  \item If $f$ is quasi-compact and dominant, then $\dim(Y)\leq\dim(X)$.
  \item If $f$ is quasi-finite, then $\dim(X)\leq\dim(Y)$.
  \item Suppose that $X$ is of finite type over $k$.
  For $\dim(X)\geq n$ \emph{(}resp.\ $\dim(X)\leq n$, $\dim(X)=n$\emph{)}, it is necessary and sufficient that there exist an open dense subset $U$ of $X$, and a $k$-morphism $g:U\to\bb{V}(k^n)$ \emph{(}$=\Spec(k[\mathrm{T}_1,\ldots,\mathrm{T}_n])$, which we will also denote $\bb{V}_k^n$\emph{)} which is surjective \emph{(}resp.\ finite, resp.\ finite and surjective\emph{)}.
  If $X$ is an affine scheme, one can take $U=X$.
\end{enumerate}
\end{theorem}

\oldpage[IV-2]{53}

\begin{proof}
(i) Let $y$ be a maximal point of $Y$; we know that $f^{-1}(y)$ contains a maximal point $x$ of $X$ \sref{IV.1.1.5}, hence $k(x)$ is an extension of $k(y)$, whence the inequality $\deg.\mathrm{tr}_k\,k(y)\leq\deg.\mathrm{tr}_k\,k(x)\leq\dim(X)$, which proves (i).

(ii) If $x$ is a maximal point of $X$, $k(x)$ is a finite extension of $k(f(x))$ \sref[II]{II.6.2.2}, hence of the same transcendence degree over $k$.
If one considers the reduced closed subprescheme of $Y$ having $\overline{\{f(x)\}}$ as underlying space, one sees that one is reduced to proving the

\begin{corollary}[4.1.2.1]
\label{IV.4.1.2.1}
Let $Y$ be a $k$-prescheme locally of finite type; for every subprescheme $Z$ of $Y$, $\dim(Z)\leq\dim(Y)$.
Suppose moreover that the irreducible components of $Y$ all have the same dimension; then, for $Z$ to be \emph{rare} in $Y$, it is necessary and sufficient that $\dim(Z)<\dim(Y)$.
\end{corollary}

Indeed, let $z$ be a maximal point of $Z$; by considering one of the reduced closed subpreschemes of $Y$ having as underlying space one of the irreducible components of $Y$ containing $z$, one reduces to the case where $Y$ is integral with generic point $y$, then, in considering within $Y$ an affine open containing $z$, to the case where $Y$ is affine and of finite type over $k$, say $Y=\Spec(A)$, and $Z$ closed in $Y$, say $Z=\Spec(A/\mathfrak{a})$ where $\mathfrak{a}$ is an integral ideal of $A$.
By virtue of the Noether normalisation lemma (Bourbaki, \emph{Alg.~comm.}, chap.~V, \textsection3, no.~1, th.~1) there exists a finite sequence $(x_i)_{1\leq i\leq n}$ of elements of $A$, algebraically independent over $k$, such that $A$ is integral over the ring $B=k[x_1,\ldots,x_n]$ and that $\mathfrak{a}\cap B$ is generated by a subfamily $(x_i)_{1\leq i\leq p}$ (possibly empty) of $A$.
Let $g:Y\to\Spec(B)=\bb{V}_k^n$ be the finite dominant morphism corresponding to the injection canonical $B\to A$; since $C=B/(\mathfrak{a}\cap B)$ is isomorphic to $k[x_{p+1},\ldots,x_n]$, $g$ induces on $Z$ a dominant morphism $Z\to\bb{V}_k^{n-p}$; we have therefore $\deg.\mathrm{tr}_k\,k(y)=n$ and $\deg.\mathrm{tr}_k\,k(z)=n-p$, hence $\deg.\mathrm{tr}_k\,k(z)\leq\deg.\mathrm{tr}_k\,k(y)=\dim(Y)$, which demonstrates (i).
Moreover, if $p=0$, $Z$ contains an open non-empty subset of $Y$, hence $Z$ is not rare; if on the contrary $p>0$, we have $\overline{\{z\}}$ is rare in $\overline{\{y\}}$, which finishes proving \sref{IV.4.1.2.1}.

(iii) The proof that the conditions stated are sufficient follows from (i) and (ii).
To prove that they are necessary, one can consider an open dense subset $U$ of $X$, union of affine opens $U_i$ pairwise disjoint and irreducible ($1\leq i\leq m$), each containing one of the maximal points of $X$ $(\mathbf{0_I}$, 2.1.6); one can moreover suppose $X$ reduced and if $X$ is affine, one can of course take $U=X$.
The same reasoning shows that if $\dim(U_i)=n_i$, there exists a $k$-morphism $g_i:U_i\to\bb{V}_k^{n_i}$ which is finite and dominant, hence surjective \sref[II]{II.6.1.10}; for each $n_i$ there is then a morphism $h_i:\bb{V}_k^{n_i}\to\bb{V}_k^n$ which is a closed immersion corresponding to the canonical homomorphism $k[\mathrm{T}_1,\ldots,\mathrm{T}_n]\to k[\mathrm{T}_1,\ldots,\mathrm{T}_{n_i}]$, and which is the identity for $n_i=n$.
\oldpage[IV-2]{54}
Since $U$ is the sum of the $U_i$, one takes for $g$ the morphism which on each $U_i$ coincides with $h_i\circ g_i$; it is obviously finite and surjective.
When $n'\geq n$, one composes with the canonical closed immersion $\bb{V}_k^n\to\bb{V}_k^{n'}$; when $n'\leq n$, one composes with $g$ the canonical morphism $p:\bb{V}_k^n\to\bb{V}_k^{n'}$ corresponding to the canonical injection $k[\mathrm{T}_1,\ldots,\mathrm{T}_{n'}]\to k[\mathrm{T}_1,\ldots,\mathrm{T}_n]$, noting that $p$ is faithfully flat, hence surjective \sref[I]{I.3.5.2} and \sref[II]{II.6.1.5}, whence the corollary.
\end{proof}

\begin{remark}[4.1.3]
\label{IV.4.1.3}
The corollary \sref{IV.4.1.2.1} shows that in the formula \sref{IV.4.1.1.1}, one can suppose that $x$ ranges over the set of \emph{all} points of $X$.
\end{remark}

\begin{corollary}[4.1.4]
\label{IV.4.1.4}
Let $X$ be a prescheme locally of finite type over a field $k$, $K$ an extension of $k$; then $\dim(X\otimes_k K)=\dim(X)$.
\end{corollary}

\begin{proof}
One can restrict to the case where $X$ is of finite type over $k$; then the morphism $u:X\otimes_k K\to X$ is faithfully flat \sref{IV.2.2.13}, (i), hence if $U$ is an open dense subset in $X$, $u^{-1}(U)=U\otimes_k K$ is dense in $X\otimes_k K$ \sref{IV.2.3.10}; if $g:U\to\bb{V}_k^n$ is finite and surjective, it is the same for $g_{(K)}:U\otimes_k K\to\bb{V}_k^n$ \sref[I]{I.3.5.2} and \sref[II]{II.6.1.5}, whence the corollary.
\end{proof}

\begin{corollary}[4.1.5]
\label{IV.4.1.5}
Let $X$ and $Y$ be two preschemes locally of finite type over a field $k$; then $\dim(X\times_k Y)=\dim(X)+\dim(Y)$.
\end{corollary}

\begin{proof}
It suffices to prove that if $U$ (resp.\ $V$) is an affine neighbourhood of a point $x$ (resp.\ $y$) of $X$ (resp.\ $Y$) such that $\deg.\mathrm{tr}_k\,k(x)=m=\dim U$ and $\deg.\mathrm{tr}_k\,k(y)=n=\dim V$, then $\dim(U\times_k V)=m+n$, in other words, by virtue of \sref{IV.4.1.2} that there exist finite surjective $k$-morphisms $f:X\to\bb{V}_k^m$, $g:Y\to\bb{V}_k^n$; then $f\times g:X\times_k Y\to\bb{V}_k^{m+n}$ is finite and surjective \sref[I]{I.3.5.2} and \sref[II]{II.6.1.5}, whence the corollary.
\end{proof}

\subsection{Associated prime cycles on algebraic preschemes}
\label{subsection:IV.4.2}

\begin{proposition}[4.2.1]
\label{IV.4.2.1}
Let $K$ and $L$ be two extensions of a field $k$, such that $K\otimes_k L$ is Noetherian.
Then the prime ideals associated to $K\otimes_k L$ are minimal, and if $E$ is the residue field of the local ring of such an ideal, we have
\begin{equation}
\label{IV.4.2.1.1}
  \deg.\mathrm{tr}_K E=\deg.\mathrm{tr}_k L,\qquad\deg.\mathrm{tr}_L E=\deg.\mathrm{tr}_k K\tag{4.2.1.1}
\end{equation}
whence
\begin{equation}
\label{IV.4.2.1.2}
  \deg.\mathrm{tr}_k E=\deg.\mathrm{tr}_k K+\deg.\mathrm{tr}_k L.\tag{4.2.1.2}
\end{equation}
\end{proposition}

\oldpage[IV-2]{55}

\begin{proof}
We know that $K$ is an algebraic extension of a purely transcendental extension $K'=k(\mathbf{t})$, where $\mathbf{t}=(t_\alpha)_{\alpha\in I}$ is a family of indeterminates; the ring $k[\mathbf{t}]\otimes_k L=L[\mathbf{t}]$ is integral, hence the same is true of $K'\otimes_k L$, which is a ring of fractions, and the field of fractions of $K'\otimes_k L$ is $L(\mathbf{t})$; we have the commutative diagram of canonical homomorphisms
\[
  \xymatrix{
    K\ar[r] & K\otimes_k L\ar[r] & K\otimes_{K'} L(\mathbf{t}) \\
    K'\ar[r]\ar[u] & K'\otimes_k L\ar[r]\ar[u] & L(\mathbf{t})\ar[u] \\
    & k\ar[u]\ar[ul] & L\ar[u]\ar[ul]
  }
\]
Since $K$ is faithfully flat over $K'$, $K\otimes_k L=K\otimes_{K'}(K'\otimes_k L)$ is faithfully flat over $K'\otimes_k L$ $(\mathbf{0_I}$, 6.5.2); moreover $K'\otimes_k L$ identifies with a subring of $K\otimes_k L$; the trace on $K'\otimes_k L$ of a prime ideal $\mathfrak{p}$ associated to $K\otimes_k L$ is the ideal $0$, an element $\neq 0$ of $K'\otimes_k L$ not being a zero divisor in $K\otimes_k L$ $(\mathbf{0_I}$, 6.3.4 and Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, no.~1, cor.~3 of prop.~2).
Since moreover $K$ is algebraic over $K'$, $K\otimes_k L$ is an integral algebra over $K'\otimes_k L$, and the prime ideals of $K\otimes_k L$ inducing $0$ on $K'\otimes_k L$ are necessarily without mutual inclusion relation (Bourbaki, \emph{Alg.~comm.}, chap.~V, \textsection2, no.~1, cor.~1 of prop.~1); this proves the first assertion (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection1, no.~3, cor.~1 of prop.~7).
Furthermore, the residue field $E$ of $\mathfrak{p}$ is algebraic over the residue field of the ideal $(0)$ of $K'\otimes_k L$, that is to say $L(\mathbf{t})$; hence $\deg.\mathrm{tr}_k E=\mathrm{Card}(I)=\deg.\mathrm{tr}_k K$, in other words we have the first relation \sref{IV.4.2.1.1}; exchanging the roles of $K$ and $L$, we have the second relation \sref{IV.4.2.1.1}, whence \sref{IV.4.2.1.2}.
\end{proof}

\begin{corollary}[4.2.2]
\label{IV.4.2.2}
Under the hypotheses of \sref{IV.3.3.6}, if the preschemes $\mathrm{T}_{x,y}$ are locally Noetherian, they have no immersed associated prime cycle.
\end{corollary}

\begin{corollary}[4.2.3]
\label{IV.4.2.3}
Under the hypotheses of \sref{IV.3.3.6} \emph{(}resp.\ \sref{IV.3.3.7}\emph{)}, if the $\mathrm{T}_{x,y}$ are locally Noetherian and if $\sh{F}$ and the $\sh{G}_s$ \emph{(}for $s\in\mathrm{S}$\emph{)} \emph{(}resp.\ $\sh{F}$ and $\sh{G}$\emph{)} are without immersed associated prime cycle, the same is true of $\sh{F}\otimes_\mathrm{S}\sh{G}$.
\end{corollary}

\begin{proof}
This follows from \sref{IV.4.2.2}, \sref{IV.3.3.2} and the proof of \sref{IV.3.3.6}.
In particular, since every prescheme over a field $k$ is \emph{flat} over $k$, we have thus proved assertion (i) of the following:
\end{proof}

\begin{proposition}[4.2.4]
\label{IV.4.2.4}
Let $k$ be a field, $X$ and $Y$ two $k$-preschemes locally Noetherian such that $X\times_k Y$ is locally Noetherian.
Suppose moreover $X$ and $Y$ integral.
Then:
\begin{enumerate}[\rm(i)]
  \item $X\times_k Y$ is without immersed associated prime cycle; each of the irreducible components of $X\times_k Y$ dominates $X$ and $Y$, and the set of these components is in bijective correspondence with the set of the irreducible components of $\Spec(\mathrm{R}(X)\otimes_k\mathrm{R}(Y))$ \emph{(}in other words the set of the minimal prime ideals of $\mathrm{R}(X)\otimes_k\mathrm{R}(Y)$\emph{)}, where $\mathrm{R}(X)$ and $\mathrm{R}(Y)$ are the fields of rational functions of $X$ and $Y$ respectively.
  \item If a maximal point $z$ of $X\times_k Y$ is identified with a minimal prime ideal $\mathfrak{p}$ of $\mathrm{R}(X)\otimes_k\mathrm{R}(Y)$, the local ring $\sh{O}_{X\times_k Y,z}$ is isomorphic to the ring of fractions $(\mathrm{R}(X)\otimes_k\mathrm{R}(Y))_\mathfrak{p}$.
  In particular, if $\mathrm{R}(X)$ or $\mathrm{R}(Y)$ is separable over $k$, $X\times_k Y$ is reduced.
  \item If moreover $X$ and $Y$ are locally of finite type over $k$, every irreducible component of $X\times_k Y$ is of dimension $\dim(X)+\dim(Y)$.
\end{enumerate}
\end{proposition}

\oldpage[IV-2]{56}

\begin{proof}
Assertion (iii) follows from \sref{IV.4.1.2}, taking into account (i) and (ii).
To prove (ii), one can obviously restrict to the case where $X=\Spec(A)$, $Y=\Spec(B)$ are affine, $A$ and $B$ being integral rings with fields of fractions $K=\mathrm{R}(X)$ and $L=\mathrm{R}(Y)$; assertion (i) shows that every minimal prime ideal $\mathfrak{q}$ of $A\otimes_k B$ is the trace on $A\otimes_k B$ of a minimal prime ideal $\mathfrak{p}$ of $K\otimes_k L$; since $K\otimes_k L$ is a ring of fractions of $A\otimes_k B$, the isomorphism of the rings of fractions $(A\otimes_k B)_\mathfrak{q}$ and $(K\otimes_k L)_\mathfrak{p}$ follows from $(\mathbf{0_I}$, 1.2.5).
Finally, if for example $\mathrm{R}(Y)$ is separable over $k$, we know that the ring $\mathrm{R}(X)\otimes_k\mathrm{R}(Y)$ is reduced (Bourbaki, \emph{Alg.}, chap.~VIII, \textsection7, no.~3, th.~1); so the same is true of the local rings at the maximal points of $X\times_k Y$.
One deduces that $X\times_k Y$ is reduced \sref{IV.3.2.1}.
\end{proof}

\begin{proposition}[4.2.5]
\label{IV.4.2.5}
Let $k$ be a field, $X$ and $Y$ two $k$-preschemes locally Noetherian, $\sh{F}$ \emph{(}resp.\ $\sh{G}$\emph{)} a quasi-coherent $\sh{O}_X$-Module \emph{(}resp.\ $\sh{O}_Y$-Module\emph{)}.
Let $(Z'_\lambda)$ \emph{(}resp.\ $(Z''_\mu)$\emph{)} be the family of the associated prime cycles of $\sh{F}$ \emph{(}resp.\ $\sh{G}$\emph{)}, and denote also by $Z'_\lambda$ \emph{(}resp.\ $Z''_\mu$\emph{)} the reduced subprescheme of $X$ \emph{(}resp.\ $Y$\emph{)} having $Z'_\lambda$ \emph{(}resp.\ $Z''_\mu$\emph{)} as underlying space.
Then, if $Z'_\lambda\times_k Z''_\mu$ is locally Noetherian, the irreducible components $Z_{\lambda\mu\nu}$ of $Z'_\lambda\times_k Z''_\mu$ dominate $Z'_\lambda$ and $Z''_\mu$, and $(Z_{\lambda\mu\nu})$ is the family of the distinct associated prime cycles of $\sh{F}\otimes_k\sh{G}$.
\end{proposition}

\begin{proof}
It suffices to apply \sref{IV.4.2.4} to the product $\Spec(k(x))\times_k\Spec(k(y))$.

In particular:
\end{proof}

\begin{corollary}[4.2.6]
\label{IV.4.2.6}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes locally Noetherian such that $X\times_k Y$ is locally Noetherian.
Let $(Z'_\lambda)$ \emph{(}resp.\ $(Z''_\mu)$\emph{)} be the family of reduced subpreschemes having as underlying spaces the irreducible components of $X$ \emph{(}resp.\ $Y$\emph{)}.
Then the irreducible components $Z_{\lambda\mu\nu}$ of $Z'_\lambda\times_k Z''_\mu$ dominate $Z'_\lambda$ and $Z''_\mu$, and $(Z_{\lambda\mu\nu})$ is the family of the irreducible components of $X\times_k Y$.
\end{corollary}

\begin{proof}
Indeed, one can restrict to the case where $X$ and $Y$ are reduced \sref[I]{I.5.1.8}; the irreducible components of $X$ (resp.\ $Y$) are then the associated prime cycles of $\sh{O}_X$ (resp.\ $\sh{O}_Y$) \sref{IV.3.2.1}.
Applying \sref{IV.4.2.5} to $\sh{F}=\sh{O}_X$ and $\sh{G}=\sh{O}_Y$, and noting that by definition $\sh{O}_X\otimes_k\sh{O}_Y=\sh{O}_{X\times_k Y}$, the corollary follows from the fact that $\sh{O}_X$ and $\sh{O}_Y$ have no immersed associated prime cycles by hypothesis \sref{IV.4.2.3}.
\end{proof}

Let us apply the preceding results to the case where $Y$ is the spectrum of an extension $K$ of $k$:

\begin{proposition}[4.2.7]
\label{IV.4.2.7}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$ such that $X\otimes_k K$ is locally Noetherian, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module, $x'$ a point of $X\otimes_k K$, $x$ its image in $X$.
\begin{enumerate}[\rm(i)]
  \item Let $(Z_\lambda)$ be the family of reduced subpreschemes of $X$ having as underlying spaces the associated prime cycles of $\sh{F}$; then the irreducible components $Z_{\lambda\mu}$ of the $Z_\lambda\otimes_k K$ are the cycles premiers associated to $\sh{F}\otimes_k K$, and $Z_{\lambda\mu}$ dominates $Z_\lambda$; moreover, for $Z_{\lambda\mu}$ to be immersed, it is necessary and sufficient that $Z_\lambda$ be so.
  \item For $x$ to belong to an immersed associated prime cycle of $\sh{F}$, it is necessary and sufficient that $x'$ belong to an immersed associated prime cycle of $\sh{F}\otimes_k K$; for $\sh{F}$ to be without immersed associated prime cycle, it is necessary and sufficient that the same hold for $\sh{F}\otimes_k K$.
  \item The indices $\lambda$ such that $x\in Z_\lambda$ are the same as those for which there exists an index $\mu$ such that $x'\in Z_{\lambda\mu}$.
  In particular, if $x'$ belongs to only one associated prime cycle of $\sh{F}\otimes_k K$, $x$ belongs to only one associated prime cycle of $\sh{F}$.
  \item If $X$ is locally of finite type over $k$, we have $\dim(Z_{\lambda\mu})=\dim(Z_\lambda)$.
\end{enumerate}
\end{proposition}

\begin{proof}
One will note that the hypothesis implies that $X$ itself is locally Noetherian \sref{IV.2.2.13} and \sref{IV.2.2.14}; assertion (i) results from \sref{IV.4.2.5} and the proof of \sref{IV.4.2.3}, for $\sh{G}=\sh{O}_Y$, with $Y=\Spec(K)$; (ii) and (iii) result from (i) and \sref{IV.2.3.5}; finally, (iv) is a particular case of \sref{IV.4.2.4}, (iii).
\end{proof}

\begin{corollary}[4.2.8]
\label{IV.4.2.8}
If $X$ is locally of finite type over $k$, the set of dimensions of associated prime cycles is the same for $\sh{F}$ and $\sh{F}\otimes_k K$; the set of dimensions of the irreducible components is the same for $X$ and $X\otimes_k K$.
\end{corollary}

\begin{proposition}[4.2.9]
\label{IV.4.2.9}
Suppose the hypotheses of \sref{IV.4.2.5} are satisfied, and suppose moreover that $\sh{F}$ and $\sh{G}$ are coherent.
Let $(\sh{F}_\lambda)_{\lambda\in\mathrm{L}}$ and $(\sh{G}_\mu)_{\mu\in\mathrm{M}}$ be irredundant decompositions of $\sh{F}$ and $\sh{G}$ respectively; for every couple $(\lambda,\mu)\in\mathrm{L}\times\mathrm{M}$, let $(\sh{H}_{\lambda\mu\nu})_{\nu\in\mathrm{S}(\lambda,\mu)}=\operatorname{Ass}(\sh{F}_\lambda\otimes_k\sh{G}_\mu)$ \sref{IV.3.2.5}.
Then $(\sh{H}_{\lambda\mu\nu})$ \emph{(}for all triples $(\lambda,\mu,\nu)$\emph{)} is an irredundant decomposition of $\sh{F}\otimes_k\sh{G}$; it is reduced if $(\sh{F}_\lambda)$ and $(\sh{G}_\mu)$ are.
\end{proposition}

\begin{proof}
Note that $\sh{F}\otimes_k\sh{G}$ and each of the $\sh{F}_\lambda\otimes_k\sh{G}_\mu$ are coherent, and each of the $\sh{F}_\lambda\otimes_k\sh{G}_\mu$ identifies with a quotient of $\sh{F}\otimes_k\sh{G}$; each of the $\sh{H}_{\lambda\mu\nu}$ identifies by definition with a coherent quotient of $\sh{F}_\lambda\otimes_k\sh{G}_\mu$, hence also with a coherent quotient of $\sh{F}\otimes_k\sh{G}$.
The family of supports of the $\sh{H}_{\lambda\mu\nu}$ is locally finite, for $\Supp(\sh{F}_\lambda\otimes_k\sh{G}_\mu)$ is contained in the underlying space of $\Supp(\sh{F}_\lambda)\times_k\Supp(\sh{G}_\mu)$ (where $\Supp(\sh{F}_\lambda)$ and $\Supp(\sh{G}_\mu)$ denote the corresponding reduced closed subpreschemes), and for $\lambda$ and $\mu$ given, the family of supports of the $\sh{H}_{\lambda\mu\nu}$ is locally finite by hypothesis; our assertion therefore results from the fact that the families $(\Supp(\sh{F}_\lambda))$ and $(\Supp(\sh{G}_\mu))$ are locally finite.
To show that $(\sh{H}_{\lambda\mu\nu})$ is an irredundant decomposition of $\sh{F}\otimes_k\sh{G}$, it suffices therefore to prove that the canonical homomorphism $\sh{F}\otimes_k\sh{G}\to\bigoplus_{\lambda,\mu,\nu}\sh{H}_{\lambda\mu\nu}$ is injective; but it is the composite of the homomorphisms $\sh{F}\otimes_k\sh{G}\to\bigoplus_{\lambda,\mu}(\sh{F}_\lambda\otimes_k\sh{G}_\mu)\to\bigoplus_{\lambda,\mu,\nu}\sh{H}_{\lambda\mu\nu}$; the second of these homomorphisms is injective by definition, and the same holds for the first, which is none other than the tensor product of the canonical injective homomorphisms $\sh{F}\to\bigoplus_\lambda\sh{F}_\lambda$, $\sh{G}\to\bigoplus_\mu\sh{G}_\mu$ (recall that $\sh{F}$ and $\sh{G}$ are flat over $k$).
Finally, if $(\sh{F}_\lambda)$ and $(\sh{G}_\mu)$ are reduced, one can suppose that $\mathrm{M}=\operatorname{Ass}(\sh{F})$ and $\mathrm{M}=\operatorname{Ass}(\sh{G})$; the fact that the $\sh{H}_{\lambda\mu\nu}$ then form a partition of $\operatorname{Ass}(\sh{F}\otimes\sh{G})$ follows from \sref{IV.4.2.2} (cf.\ \sref{IV.3.2.5}).
\end{proof}

\begin{corollary}[4.2.10]
\label{IV.4.2.10}
Under the hypotheses of \sref{IV.4.2.7}, suppose moreover $\sh{F}$ coherent, and let $(\sh{F}_\lambda)_{\lambda\in\mathrm{L}}$ be an irredundant decomposition of $\sh{F}$; for every $\lambda\in\mathrm{L}$, let $(\sh{F}_{\lambda\mu})_{\mu\in\operatorname{Ass}(\sh{F}_\lambda\otimes_k K)}$ be a reduced irredundant decomposition of $\sh{F}_\lambda\otimes_k K$.
Then $(\sh{F}_{\lambda\mu})$ is an irredundant decomposition of $\sh{F}\otimes_k K$; it is reduced if $(\sh{F}_\lambda)$ is.
\end{corollary}

\begin{proof}
One applies \sref{IV.4.2.9} with $Y=\Spec(K)$, $\sh{G}=\sh{O}_Y=K$, $\mathrm{M}$ reduced to a single element.
\end{proof}

\subsection{Reminders on tensor products of fields}
\label{subsection:IV.4.3}

\oldpage[IV-2]{58}

For the convenience of the reader, we recall here certain properties of tensor products of fields which we will use in the following sections; for proofs, we refer to \cite{I}.

\begin{env}[4.3.1]
\label{IV.4.3.1}
Recall that an extension $L$ of a field $k$ is called \emph{primary} if the largest algebraic separable extension of $k$ contained in $L$ is $k$ itself.
\end{env}

\begin{proposition}[4.3.2]
\label{IV.4.3.2}
Let $K$, $L$ be two extensions of a field $k$.
If $L$ is a primary extension of $k$, then $\Spec(L\otimes_k K)$ is irreducible, and if $\xi$ is its generic point, $k(\xi)$ is a primary extension of $K$.
Conversely, if for every finite separable extension $K$ of $k$, $\Spec(L\otimes_k K)$ is irreducible, $L$ is a primary extension of $k$.
\end{proposition}

\begin{proof}
See the proof in \cite{I}, p.~14-03 to 14-06.
\end{proof}

\begin{corollary}[4.3.3]
\label{IV.4.3.3}
If $k$ is separably closed (i.e.\ if its algebraic closure is purely inseparable over $k$), then, for any two extensions $K$, $L$ of $k$, $\Spec(L\otimes_k K)$ is irreducible, and conversely.
\end{corollary}

\begin{proof}
This follows immediately from \sref{IV.4.3.2}.
\end{proof}

\begin{corollary}[4.3.4]
\label{IV.4.3.4}
Let $L$ be an extension of a field $k$, $L_s$ the largest algebraic separable extension of $k$ contained in $L$ (sometimes called the \emph{separable algebraic closure} of $k$ in $L$); suppose $L_s$ of finite degree over $k$, and let $K$ be a Galois extension of $k$ containing $L_s$.
Then $\Spec(L\otimes_k K)$ has $[L_s:k]$ irreducible components, the sum of whose residue fields at the generic points are the primary extensions of $K$.
\end{corollary}

\begin{proof}
Indeed, we have $L\otimes_k K=L\otimes_{L_s}(L_s\otimes_k K)$ and one knows that $L_s\otimes_k K$ is isomorphic to a product of $[L_s:k]$ fields isomorphic to $K$ (Bourbaki, \emph{Alg.}, chap.~VIII, \textsection7, no.~3, cor.~2 of th.~1); hence $L\otimes_k K$ is isomorphic to a product of $[L_s:k]$ rings isomorphic to $L\otimes_{L_s} K$; since $L$ is a primary extension of $L_s$, the conclusion follows from \sref{IV.4.3.2}.
\end{proof}

\begin{proposition}[4.3.5]
\label{IV.4.3.5}
Let $K$, $L$ be two extensions of a field $k$.
If $L$ is a separable extension of $k$, the ring $L\otimes_k K$ is reduced.
Conversely, if for every finite purely inseparable extension $K$ of $k$, the ring $L\otimes_k K$ is reduced, $L$ is a separable extension of $k$.
\end{proposition}

\begin{proof}
For the proof, see Bourbaki, \emph{Alg.}, chap.~VIII, \textsection7, no.~3, th.~1.
\end{proof}

\begin{corollary}[4.3.6]
\label{IV.4.3.6}
If $k$ is perfect, then for any two extensions $K$, $L$ of $k$, $K\otimes_k L$ is reduced, and conversely.
\end{corollary}

\begin{proof}
This follows immediately from \sref{IV.4.3.5}.
\end{proof}

\begin{corollary}[4.3.7]
\label{IV.4.3.7}
Let $L$ be a separable extension of $k$ and let $K$ be an extension of $k$ such that $K$ or $L$ is finite over $k$; then the residue fields of the semi-local ring $L\otimes_k K$ are separable extensions of $K$.
\end{corollary}

\begin{proof}
One knows indeed that $L\otimes_k K$ is a direct product of these residue fields $E_i$; for every extension $K'$ of $K$, $L\otimes_k K'$ is therefore also isomorphic to the direct product of the rings $E_i\otimes_K K'$; since $L\otimes_k K'$ is reduced, the $E_i$ are separable over $K$ by virtue of \sref{IV.4.3.5}.
\end{proof}

\begin{corollary}[4.3.8]
\label{IV.4.3.8}
If $K$ and $L$ are two finite separable extensions of $k$, $K\otimes_k L$ is a product of a finite number of separable extensions of $k$.
\end{corollary}

\begin{proposition}[4.3.9]
\label{IV.4.3.9}
If $k$ is algebraically closed, then, for any two extensions $K$, $L$ of $k$, $K\otimes_k L$ is integral, and conversely.
\end{proposition}

\begin{proof}
This is a consequence of \sref{IV.4.3.3} and \sref{IV.4.3.6}, a perfect and separably closed field being algebraically closed.
\end{proof}

\subsection{Irreducible and connected preschemes over an algebraically closed field}
\label{subsection:IV.4.4}

\oldpage[IV-2]{59}

\begin{env}[4.4.1]
\label{IV.4.4.1}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$.
Since the morphism $\Spec(K)\to\Spec(k)$ is faithfully flat, quasi-compact, and universally open \sref{IV.2.4.9}, the same holds for the projection morphism $p:X\otimes_k K\to X$ \sref{IV.2.2.13}.
It therefore follows from \sref{IV.2.3.5} that every irreducible component of $X\otimes_k K$ \emph{dominates} an irreducible component of $X$, and (since $p$ is surjective) every irreducible component of $X$ is dominated by an irreducible component of $X\otimes_k K$; one thus deduces from $p$ a \emph{surjective} application from the set of irreducible components of $X\otimes_k K$ to the set of irreducible components of $X$.
In the same way, it follows from the fact that $p$ is continuous and surjective that every connected component of $X$ is the image by $p$ of a union of connected components of $X\otimes_k K$, whence a surjective application from the set of connected components of $X\otimes_k K$ to the set of connected components of $X$.
One concludes that every irreducible (resp.\ connected) component $Z'$ of $X\otimes_k K$ is contained in a \emph{unique} set of the form $p^{-1}(Z)$, where $Z$ is an irreducible (resp.\ connected) component of $X$; for every subprescheme $Z$ of $X$ having $Z$ as underlying space (and still denoted $Z$), $Z'$ is an irreducible (resp.\ connected) component of $Z\otimes_k K$.
In particular, if $X\otimes_k K$ has a finite number $n$ (resp.\ $n'$) of irreducible (resp.\ connected) components (which is the case when $X\otimes_k K$ is of \emph{finite type} over $K$, for then $X\otimes_k K$ is of finite type over $K$, hence Noetherian), the number of irreducible (resp.\ connected) components of $X$ is $\leq n$ (resp.\ $\leq n'$); to say that it is equal to $n$ (resp.\ $n'$) means that for every reduced subprescheme $Z$ of $X$ having as underlying space an irreducible (resp.\ connected) component of $X$, $Z\otimes_k K$ is irreducible (resp.\ connected) \sref[I]{I.5.1.8}; the number of irreducible (resp.\ connected) components of $X\otimes_k K$ is therefore independent of $K$ if and only if for every irreducible (resp.\ connected) component $Z$ of $X$, $Z\otimes_k K$ is irreducible (resp.\ connected) for \emph{every} extension $K$ of $k$.

In particular, if $X\otimes_k K$ is irreducible (resp.\ connected), the same holds for $X$ (which already follows from the fact that $p$ is surjective).
In the same way, if $X\otimes_k K$ is \emph{reduced} (resp.\ \emph{integral}), the same holds for $X$ since $p$ is faithfully flat \sref{IV.2.1.13}.
\end{env}

In this section and the following, we shall examine more closely what one can say about the irreducible (resp.\ connected) components of $X\otimes_k K$ when $K$ varies; what precedes already shows that the number of these components is a \emph{increasing} function of $K$.

\begin{lemma}[4.4.2]
\label{IV.4.4.2}
Let $X'$, $X$ be two topological spaces, $f:X'\to X$ a continuous map.
Suppose the following conditions are satisfied:
\begin{enumerate}[\rm(i)]
  \item $f$ is surjective and open \emph{(}resp.\ such that the topological space $X$ identifies canonically with the quotient of $X'$ by the equivalence relation defined by $f$\emph{)}.
  \item For every $x\in X$, $f^{-1}(x)$ is irreducible \emph{(}resp.\ connected\emph{)}.
\end{enumerate}
Then for $X'$ to be irreducible \emph{(}resp.\ connected\emph{)}, it is necessary and sufficient that $X$ be so.
\end{lemma}

\oldpage[IV-2]{60}

\begin{proof}
The assertion relative to connectedness is none other than Bourbaki, \emph{Top.~gén.}, chap.~I\textsuperscript{er}, 3\textsuperscript{e} éd., \textsection11, no.~3, prop.~7.
Let us prove the assertion relative to irreducibility; the condition being trivially necessary since $f$ is surjective, let us show it is sufficient.
Let $X'_1$, $X'_2$ be two closed parts of $X'$ such that $X'=X'_1\cup X'_2$, and let $X_i$ ($i=1,2$) denote the set of $x\in X$ such that $f^{-1}(x)\subset X'_i$; since $X_i=X-f(X'-X'_i)$, and since $f$ is open, we see that $X_i$ is a \emph{closed} part of $X$ ($i=1,2$).
On the other hand, for every $x\in X$, $f^{-1}(x)$ is irreducible by hypothesis, and is the union of the two closed parts $X'_1\cap f^{-1}(x)$ and $X'_2\cap f^{-1}(x)$; one of these two parts must therefore be equal to $f^{-1}(x)$, which means that $x\in X_1$ or $x\in X_2$; hence $X=X_1\cup X_2$, and since $X$ is irreducible, we deduce $X=X_1$ or $X=X_2$, and therefore $X'=X'_1$ or $X'=X'_2$.
\end{proof}

\begin{remark}[4.4.3]
\label{IV.4.4.3}
If the topology of $X$ is not the quotient of that of $X'$ by the equivalence relation defined by $f$, it can happen that $X$ and all the fibres $f^{-1}(x)$ are irreducible, without $X'$ being connected: one obtains an example by taking for $X$ an irreducible scheme (for example the affine line $\Spec(k[\mathrm{T}])$) over a field $k$ algebraically closed, considering a closed point $x_0$ of $X$ and taking for $X'$ the space that is the \emph{sum} of $\{x_0\}$ and of the open subspace $X-\{x_0\}$ of $X$.
The same if one replaces the hypothesis ``$f$ open'' by ``$f$ closed'' (and even ``$f$ proper'') (which still implies that the topology of $X$ is the quotient of that of $X'$ by the relation defined by $f$), it can happen that $X$ and all the fibres $f^{-1}(x)$ are irreducible without $X'$ being so.
Let us take for $X$ the affine line over $k$, denote by $\mathrm{S}$ the product $X\times_k\bb{P}$, where $\bb{P}$ is the projective line over $k$; if $x_0$ is a closed point of $X$, $t_0$ a closed point of $\bb{P}$, $p:\mathrm{S}\to X$, $q:\mathrm{S}\to\bb{P}$ the projections, we denote by $X'$ the reduced subprescheme having as underlying space the closed set $p^{-1}(x_0)\cup q^{-1}(t_0)$, and we take for $f$ the restriction of $p$ to $X'$; $f$ is proper but $X'$ is not irreducible.
\end{remark}

\begin{theorem}[4.4.4]
\label{IV.4.4.4}
Let $k$ be an algebraically closed field, $X$ a $k$-prescheme.
If $X$ is irreducible \emph{(}resp.\ connected\emph{)}, then $X\otimes_k K$ is also irreducible \emph{(}resp.\ connected\emph{)} for every extension $K$ of $k$.
\end{theorem}

\begin{proof}
One saw in \sref{IV.4.4.1} that the morphism $p:X\otimes_k K\to X$ is faithfully flat and open; to be able to apply the lemma \sref{IV.4.4.2}, it suffices therefore to verify that the fibres $p^{-1}(x)$ are irreducible for every $x\in X$.
But the $k(x)$-prescheme $p^{-1}(x)$ is isomorphic to $\Spec(k(x)\otimes_k K)$ \sref[I]{I.3.6.2}, hence is integral by virtue of the hypothesis on $k$ and of \sref{IV.4.3.9}.
\end{proof}

\begin{corollary}[4.4.5]
\label{IV.4.4.5}
Let $k$ be an algebraically closed field, $X$ a $k$-prescheme, $K$ an extension of $k$, $p:X\otimes_k K\to X$ the canonical projection.
If $Z$ is an irreducible \emph{(}resp.\ connected\emph{)} part of $X$, $p^{-1}(Z)$ is irreducible \emph{(}resp.\ connected\emph{)}; in particular, if $X_0$ is an irreducible \emph{(}resp.\ connected\emph{)} component of $X$ containing $Z$, $p^{-1}(X_0)$ is an irreducible \emph{(}resp.\ connected\emph{)} component of $X\otimes_k K$ containing $p^{-1}(Z)$.
\end{corollary}

\begin{proof}
The second assertion results from \sref{IV.4.4.1} and the first applied by replacing $Z$ by $X_0$; to prove the first, let $X'$ denote the sub-prescheme of $X\otimes_k K$ having $X_0$ as underlying space, $Z'=p^{-1}(Z)$; the equivalence relation $\mathrm{R}$ defined by $p$ on $X'$ is open, hence the same holds for the one it induces on the saturated part $Z'$, and $Z$ identifies with the quotient space $Z'/\mathrm{R}_{Z'}$ (Bourbaki, \emph{Top.~gén.}, chap.~I\textsuperscript{er}, 3\textsuperscript{e} éd., \textsection5, no.~2, prop.~4); one can therefore apply the lemma \sref{IV.4.4.2} to $Z$ and $Z'$, whence the first assertion from \sref{IV.4.4.4}.
\end{proof}

\oldpage[IV-2]{61}

\begin{corollary}[4.4.6]
\label{IV.4.4.6}
Under the hypotheses and with the notation of \sref{IV.4.4.5}, the map $Z\mapsto p^{-1}(Z)$ is a bijection from the set of irreducible \emph{(}resp.\ connected\emph{)} components of $X$ to the set of irreducible \emph{(}resp.\ connected\emph{)} components of $X\otimes_k K$, whose inverse bijection is $Z'\mapsto p(Z')$.
\end{corollary}

\subsection{Geometrically irreducible and geometrically connected preschemes}
\label{subsection:IV.4.5}

\begin{proposition}[4.5.1]
\label{IV.4.5.1}
Let $k$ be a field, $X$ a $k$-prescheme, $\Omega$ an algebraically closed extension of $k$.
Let $n$ \emph{(}resp.\ $n'$\emph{)} be the number of irreducible \emph{(}resp.\ connected\emph{)} components of $X\otimes_k\Omega$; this number is independent of the algebraically closed extension $\Omega$ chosen.
Moreover, for every extension $K$ of $k$, the number of irreducible \emph{(}resp.\ connected\emph{)} components of $X\otimes_k K$ is $\leq n$ \emph{(}resp.\ $\leq n'$\emph{)}.
\end{proposition}

\begin{proof}
Indeed, two algebraically closed extensions $\Omega$, $\Omega'$ of $k$ can always be considered as sub-extensions of a same extension $\mathrm{E}$ of $k$; it suffices then to apply \sref{IV.4.4.6} to $X\otimes_k\Omega$ and $X\otimes_k\Omega'$ and to their common extension $\mathrm{E}$ of $\Omega$ and $\Omega'$ to show the first assertion.
The second is obtained by taking for $\Omega$ an algebraically closed extension of $K$ and using \sref{IV.4.4.1}.
\end{proof}

\begin{definition}[4.5.2]
\label{IV.4.5.2}
The number $n$ \emph{(}resp.\ $n'$\emph{)} of \sref{IV.4.5.1} is called the \emph{geometric number of irreducible} \emph{(}resp.\ \emph{connected}\emph{)} \emph{components} of $X$ (relative to $k$).
If $n=1$ (resp.\ $n'\leq 1$), one says that $X$ is a $k$-prescheme \emph{geometrically irreducible} \emph{(}resp.\ \emph{geometrically connected}\emph{)}.
\end{definition}

One thus recovers the definition given for a particular case in (\textbf{III}, 4.3.4).
By virtue of \sref{IV.4.5.1}, the following properties are equivalent:
\begin{enumerate}[\rm a)]
  \item $X$ is geometrically irreducible (resp.\ geometrically connected).
  \item For an algebraically closed extension $\Omega$ of $k$, $X\otimes_k\Omega$ is irreducible (resp.\ connected).
  \item For every extension $K$ of $k$, $X\otimes_k K$ is irreducible (resp.\ connected).
\end{enumerate}

\begin{env}[4.5.3]
\label{IV.4.5.3}
Let $X$ be a $k$-prescheme, $Z$ a locally closed part of $X$.
If we still denote by $Z$ any of the subpreschemes of $X$ having $Z$ as underlying space, it results from \sref[I]{I.5.1.8} that for an extension $K$ of $k$, the number of irreducible (resp.\ connected) components of $Z\otimes_k K$ is independent of the subprescheme with underlying space $Z$ that one has chosen; one thus also defines the geometric number of irreducible (resp.\ connected) components of $Z$ as that of any subprescheme of $X$ having $Z$ as underlying space.
\end{env}

\begin{proposition}[4.5.4]
\label{IV.4.5.4}
Let $X$, $Y$ be two $k$-preschemes, $f:X\to Y$ a $k$-morphism surjective irreducible \emph{(}resp.\ geometrically connected\emph{)}; it is the same for $Y$.
\end{proposition}

\begin{proof}
Indeed, for every extension $K$ of $k$, $f_{(K)}:X_{(K)}\to Y_{(K)}$ is surjective \sref[I]{I.3.5.2}, and $X_{(K)}$ being irreducible (resp.\ connected) by hypothesis, the same holds for $Y_{(K)}$.
\end{proof}

\oldpage[IV-2]{62}

\begin{definition}[4.5.5]
\label{IV.4.5.5}
One says that a morphism $f:X\to Y$ of preschemes is \emph{irreducible} \emph{(}resp.\ \emph{connected}\emph{)}, if, for every $y\in Y$, the $k(y)$-prescheme $f^{-1}(y)$ is geometrically irreducible \emph{(}resp.\ geometrically connected\emph{)}.
\end{definition}

\begin{proposition}[4.5.6]
\label{IV.4.5.6}
\begin{enumerate}[\rm(i)]
  \item Let $X$ be a $k$-prescheme, $K$ an extension of $k$.
  Then the geometric number of irreducible \emph{(}resp.\ connected\emph{)} components of $X_{(K)}$ relative to $K$ is equal to the geometric number of irreducible \emph{(}resp.\ connected\emph{)} components of $X$ relative to $k$.
  In particular, for the $K$-prescheme $X_{(K)}$ to be geometrically irreducible \emph{(}resp.\ geometrically connected\emph{)}, it is necessary and sufficient that the $k$-prescheme $X$ be so.
  \item Let $f:X\to Y$, $g:Y'\to Y$ be two morphisms, and set $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$.
  If $f$ is irreducible \emph{(}resp.\ connected\emph{)}, the same holds for $f'$.
  The converse is true if $g$ is surjective.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) If $\Omega$ is an algebraically closed extension of $K$, we have $X\otimes_k\Omega=X_{(K)}\otimes_K\Omega$, whence the assertion.

(ii) For every $y'\in Y'$, if $y=g(y')$, we have $f'^{-1}(y')=f^{-1}(y)\otimes_{k(y)}k(y')$ \sref[I]{I.3.6.4}, hence the two assertions result from (i).
\end{proof}

\begin{proposition}[4.5.7]
\label{IV.4.5.7}
Let $f:X\to Y$ be a surjective irreducible \emph{(}resp.\ connected\emph{)} morphism, $g:Y'\to Y$ a morphism, and set $X'=X\times_Y Y'$.
If moreover $Y'$ is irreducible \emph{(}resp.\ connected\emph{)}, and $f$ universally open \emph{(}resp.\ flat and quasi-compact, or universally open, or universally closed\emph{)}, then $X'$ is irreducible \emph{(}resp.\ connected\emph{)}.
\end{proposition}

\begin{proof}
The properties of $f$ that one considers being all stable by base change, one can reduce to the case where $Y'=Y$; it then suffices to apply \sref{IV.4.4.2} (taking into account \sref{IV.2.3.12}).
\end{proof}

\begin{corollary}[4.5.8]
\label{IV.4.5.8}
Let $X$, $Y$ be two $k$-preschemes.
\begin{enumerate}[\rm(i)]
  \item If $X$ is geometrically irreducible \emph{(}resp.\ geometrically connected\emph{)} and $Y$ is irreducible \emph{(}resp.\ connected\emph{)}, then $X\times_k Y$ is irreducible \emph{(}resp.\ connected\emph{)}.
  \item If $X$ and $Y$ are geometrically irreducible \emph{(}resp.\ geometrically connected\emph{)}, the same holds for $X\times_k Y$.
\end{enumerate}
\end{corollary}

\begin{proof}
(i) The structural morphism $X\to\Spec(k)$ is surjective and universally open \sref{IV.2.4.9} and in addition irreducible (resp.\ connected); it suffices therefore to apply \sref{IV.4.5.7}.

(ii) If $\Omega$ is an algebraically closed extension of $k$, we have $(X\times_k Y)_{(\Omega)}=X_{(\Omega)}\times_\Omega Y_{(\Omega)}$ \sref[III]{III.3.3.10}; by hypothesis $X_{(\Omega)}$ and $Y_{(\Omega)}$ are geometrically irreducible (resp.\ geometrically connected) $\Omega$-preschemes \sref{IV.4.5.6}, and it suffices therefore to apply (i).
\end{proof}

\begin{proposition}[4.5.9]
\label{IV.4.5.9}
Let $X$ be a $k$-prescheme.
The following conditions are equivalent:
\begin{enumerate}[\rm a)]
  \item $X$ is geometrically irreducible \emph{(}that is to say, for every extension $K$ of $k$, $X\otimes_k K$ is irreducible\emph{)}.
  \item For every finite separable extension $K$ of $k$, $X\otimes_k K$ is irreducible.
  \item $X$ is irreducible, and if $x$ is its generic point, $k(x)$ is a primary extension of $k$.
\end{enumerate}
\end{proposition}

\oldpage[IV-2]{63}

\begin{proof}
It is clear that $a)$ implies $b)$.
To see that $b)$ implies $c)$, consider a finite separable extension $K$ of $k$; if $p:X\otimes_k K\to X$ is the canonical projection, the maximal points of $X\otimes_k K$ are those of the fibre $p^{-1}(x)=\Spec(k(x)\otimes_k K)$ \sref{IV.2.3.4} and $(\mathbf{0_I}$, 2.1.8); to say that this fibre is irreducible for every finite separable extension $K$ of $k$ amounts to saying that $k(y)$ is a primary extension of $k$, by virtue of \sref{IV.4.3.2}.
Conversely, if $c)$ is satisfied, the same reasoning (taking into account \sref{IV.4.3.2}) shows that for every \emph{extension} $K$ of $k$, $X\otimes_k K$ is irreducible, hence $c)$ implies $a)$.
\end{proof}

\begin{corollary}[4.5.10]
\label{IV.4.5.10}
Let $X$ be a $k$-prescheme irreducible, $x$ its generic point, $k'$ the algebraic separable closure of $k$ in $k(x)$, $k''$ a Galois extension of $k$ \emph{(}of finite degree or not\emph{)} containing $k'$.
Suppose $k'$ of finite degree over $k$; then the irreducible components of $X\otimes_k k''$ are geometrically irreducible; their number, equal to $[k':k]$ and is also the geometric number of irreducible components of $X$.
\end{corollary}

\begin{proof}
As in \sref{IV.4.5.9}, one is reduced to considering the maximal points of the fibre $\Spec(k(x)\otimes_k k'')$; by virtue of \sref{IV.4.3.4}, they are equal in number to $[k':k]$ and the residue fields at these points are primary extensions of $k''$, which, taking into account \sref{IV.4.5.9}, proves the corollary.
\end{proof}

\begin{corollary}[4.5.11]
\label{IV.4.5.11}
Let $X$ be a $k$-prescheme.
Suppose that $X$ has only a finite number of maximal points $x_i$ ($1\leq i\leq r$) and that for each $i$, the algebraic separable closure $k'_i$ of $k$ in $k(x_i)$ is a finite extension of $k$.
Then there exists a finite separable extension $L$ of $k$ such that the irreducible components of $X\otimes_k L$ are geometrically irreducible; their number, equal to $\sum_i[k'_i:k]$, is the geometric number of irreducible components of $X$.
\end{corollary}

\begin{proof}
The maximal points of $X\otimes_k L$ are those of the various fibres $\Spec(k(x_i)\otimes_k L)$ \sref{IV.2.3.4} and $(\mathbf{0_I}$, 2.1.8), applied to the irreducible components of $X$ and to their inverse images in $X\otimes_k L$.
It suffices therefore to take for $L$ a Galois extension finite of $k$ containing all the $k'_i$ and to apply \sref{IV.4.5.10}.
\end{proof}

One will note that \sref{IV.4.5.11} applies in particular to every $k$-prescheme $X$ of \emph{finite type} over $k$, for $X$ is then Noetherian, hence has only a finite number of irreducible components, and each of the $k(x_i)$ is a finite type extension of $k$, hence the same holds for the algebraic separable closure of $k$ in $k(x_i)$ (\cite{I}, p.~6-06, lemma 5).

\begin{remarks}[4.5.12]
\label{IV.4.5.12}
\begin{enumerate}[\rm(i)]
  \item The notions defined in \sref{IV.4.5.2} depend on the base field $k$.
  For short, and whenever it is necessary to mention the base field, one says ``$k$-irreducible'' (resp.\ ``$k$-connected'') instead of ``geometrically irreducible (resp.\ connected) relative to $k$''.
  One will note that this terminology, quite natural in the context of schemes, is exactly the \emph{opposite} of that of Weil: for this author, ``$k$-irreducible'' (resp.\ ``$k$-connected'', ``$k$-normal'', etc.) denotes an \emph{intrinsic} property of a prescheme $X$, independent of the choice of the field $k$ taken as ``base field'' (in other words, of the morphism $X\to\Spec(k)$ chosen); we express these properties by saying simply that $X$ is irreducible (resp.\ connected, normal, etc.).
  Weil says on the other hand ``absolutely irreducible'', ``absolutely connected'', ``absolutely normal'', etc., for the corresponding notions relative to the prescheme $X\otimes_k\Omega$, where $\Omega$ is an algebraically closed suitable extension of $k$.
  We will avoid in what follows using the qualifier ``absolute'', which may lend to confusion, and whenever we use the abbreviated terminology introduced above (in conflict with the accepted terminology, see above), we will refer to \sref{IV.4.5.12} to avoid all ambiguity.
  \item The proposition \sref{IV.4.5.9} gives a ``birational'' criterion (that is, depending only on the residue field at the generic point) for a $k$-prescheme $X$ to be geometrically irreducible.
  There is no analogous criterion for $X$ to be geometrically connected.
\end{enumerate}
\end{remarks}

\begin{proposition}[4.5.13]
\label{IV.4.5.13}
Let $X$, $Y$ be two $k$-preschemes, $f:Y\to X$ a $k$-morphism.
Suppose $Y$ geometrically connected and non-empty, and $X$ connected.
Then $X$ is geometrically connected.
\end{proposition}

\begin{proof}
Let $\Omega$ be an algebraic closure of $k$, $X'=X_{(\Omega)}$, $Y'=Y_{(\Omega)}$, $f'=f_{(\Omega)}:Y'\to X'$; note that the morphisms $p:X'\to X$ and $q:Y'\to Y$ are \emph{open} \sref{IV.2.4.10} and \emph{closed} since $\Omega$ is an algebraic extension of $k$ \sref[II]{II.6.1.10}.
Hence $U=p(\mathrm{U'})$ is open and closed, non-empty in $X$, and hence is equal to $X$.
Since $Y$ is non-empty, one concludes \sref[I]{I.3.4.7} that $\mathrm{V'}=f'^{-1}(\mathrm{U'})$ is non-empty; furthermore, it is an open and closed part of $Y'$, and since this last space is connected by hypothesis, $\mathrm{V'}=Y'$.
We deduce that $\mathrm{U'}=X'$, without which the open and closed set $X'-\mathrm{U'}$ would show, applying the same reasoning to $X'-\mathrm{U'}$, that $f'^{-1}(X'-\mathrm{U'})=Y'$, which is absurd since $Y'$ is non-empty.
\end{proof}

\begin{corollary}[4.5.13.1]
\label{IV.4.5.13.1}
\begin{enumerate}[\rm(i)]
  \item Let $X$, $Y$ be two $k$-preschemes, $f:Y\to X$ a $k$-morphism.
  If $Y$ is geometrically connected and non-empty, the connected component $X_0$ of $X$ containing $f(Y)$ is geometrically connected.
  \item Let $X$ be a $k$-prescheme.
  If $Y$ is an irreducible component of $X$ which is geometrically irreducible, the connected component $X_0$ of $X$ containing $Y$ is geometrically connected.
\end{enumerate}
\end{corollary}

\begin{proof}
(i) One can reduce to the case where $Y$ is reduced, so that $f$ factorizes as $f:Y\xrightarrow{g}X_0\xrightarrow{j}X$, denoting $X_0$ as the reduced subprescheme having $X_0$ as underlying space \sref[I]{I.5.2.2}; it suffices to apply \sref{IV.4.5.13} to $g$.

(ii) Considering a prescheme having $Y$ as underlying space, and remarking that $Y$ is \emph{a fortiori} geometrically connected, it suffices to apply (i) to the canonical injection $Y\to X$.
\end{proof}

\begin{corollary}[4.5.14]
\label{IV.4.5.14}
Let $X$ be a $k$-prescheme.
If $x$ is a point of $X$ such that $k(x)$ is a primary extension of $k$ \emph{(}in particular if $x$ is a rational point of $X$\emph{)}, the connected component of $x$ in $X$ is geometrically connected.
\end{corollary}

\begin{proof}
It suffices to apply \sref{IV.4.5.13.1} to $Y=\Spec(k(x))$ and to the canonical morphism $Y\to X$, taking into account that $Y$ is geometrically irreducible by \sref{IV.4.5.9}.
\end{proof}

\begin{proposition}[4.5.15]
\label{IV.4.5.15}
Let $X$ be a $k$-prescheme, $x$ a point of $X$, $k'$ the algebraic separable closure of $k$ in $k(x)$.
Suppose:
\begin{enumerate}[\rm(i)]
  \item $X$ is connected.
  \item $k'$ is a finite extension of $k$.
\end{enumerate}
Then the geometric number of connected components of $X$ is $\leq[k':k]$, and if $k''$ is a Galois extension finite of $k$ containing $k'$, the connected components of $X\otimes_k k''$ are geometrically connected.
\end{proposition}

\begin{proof}
There exist finite Galois extensions $k''$ of $k$ containing $k'$ by virtue of condition (ii).
Set $X''=X\otimes_k k''$; the projection morphism $p:X''\to X$ is faithfully flat and finite, hence closed \sref[II]{II.6.1.10} and it follows \sref{IV.2.3.6}, (ii)) the image by $p$ of every connected component $X''_\alpha$ of $X''$ is equal to $X$; $X''_\alpha$ therefore contains a point $x''_\alpha\in p^{-1}(x)$.
But $p^{-1}(x)$ has a number of points equal to the number of irreducible components of $\Spec(k(x)\otimes_k k'')$ \sref[I]{I.3.4.9}, that is $[k':k]$ \sref{IV.4.3.4}, and \emph{a fortiori} the number of connected components of $X''$ is $\leq[k':k]$.
For every $x''_\alpha\in p^{-1}(x)$, $k(x''_\alpha)$ is a primary extension of $k''$ by \sref{IV.4.3.5} and \sref[I]{I.3.4.9}.
One can therefore apply the corollary \sref{IV.4.5.14} to $x''_\alpha$ and $X''$, which proves that all the connected components of $X''$ are geometrically connected.
\end{proof}

\begin{corollary}[4.5.16]
\label{IV.4.5.16}
Let $X$ be a $k$-prescheme, $(X_\alpha)$ the family of its connected components, and for every $\alpha$, let $x_\alpha\in X_\alpha$.
Suppose:
\begin{enumerate}[\rm(i)]
  \item The family $(X_\alpha)$ is finite.
  \item For every $\alpha$, the algebraic separable closure $k'_\alpha$ of $k$ in $k(x_\alpha)$ is a finite extension of $k$.
\end{enumerate}
Then the geometric number of connected components of $X$ is at most $\sum_\alpha[k'_\alpha:k]$, and there exists a finite separable extension $k''$ of $k$ such that all the connected components of $X\otimes_k k''$ are geometrically connected.
\end{corollary}

\begin{proof}
Noting that the connected components of $X$ are open by virtue of condition (i), it suffices to apply to each of the subpreschemes of $X$ induced on the open sets $X_\alpha$ the result of \sref{IV.4.5.15}; for an extension $k''$ answering the question, it suffices to take a Galois extension finite of $k$ containing all the $k'_\alpha$.
\end{proof}

\begin{corollary}[4.5.17]
\label{IV.4.5.17}
Suppose that the $k$-prescheme $X$ contains a point $x$ such that the algebraic separable closure of $k$ in $k(x)$ is finite over $k$.
For $X$ to be geometrically connected, it is necessary and sufficient that for every finite separable extension $K$ of $k$, $X\otimes_k K$ be connected.
\end{corollary}

\label{IV.4.5.18}
We shall see in \textsection8 \sref{IV.4.8.5} that the conclusion of \sref{IV.4.5.17} is still valid if, instead of supposing verified the condition of the statement, one supposes that $X$ is \emph{quasi-compact}.
\end{remark}

\begin{proposition}[4.5.19]
\label{IV.4.5.19}
Let $X$ be a $k$-prescheme, $k'$ an algebraically closed extension of $k$, $X'=X\otimes_k k'$, $p:X'\to X$ the canonical projection.
Suppose $p^{-1}(Z)$ is contained in a single irreducible component $X'_0$ of $X'$.
Then $a=X_0=p(X'_0)$ is an irreducible component of $X$ containing $Z$, and moreover $X_0$ is geometrically irreducible.
If moreover $X'_0$ is the only irreducible component of $X'$ that meets $Z'$, then $X_0$ is the only irreducible component of $X$ that meets $Z$.
\end{proposition}

\begin{proof}
To show that $X'_0=p^{-1}(X_0)$, we will apply the following:
\end{proof}

\begin{lemma}[4.5.19.1]
\label{IV.4.5.19.1}
Let $\mathrm{T}$, $\mathrm{T'}$ be two preschemes, $p:\mathrm{T'}\to\mathrm{T}$ a morphism; consider the prescheme $\mathrm{T''}=\mathrm{T'}\times_{\mathrm{T}}\mathrm{T'}$, and let $p_1:\mathrm{T''}\to\mathrm{T'}$, $p_2:\mathrm{T''}\to\mathrm{T'}$ be the canonical projections.
For an open subset $\mathrm{U'}$ of $\mathrm{T'}$ to be of the form $p^{-1}(\mathrm{U})$, where $\mathrm{U}\subset\mathrm{T}$, it is necessary and sufficient that $p_1^{-1}(\mathrm{U'})=p_2^{-1}(\mathrm{U'})$.
\end{lemma}

\begin{proof}
The structural morphism $q:\mathrm{T''}\to\mathrm{T}$ is equal to $p\circ p_1$ and to $p\circ p_2$ by definition, hence the relation is necessary.
Conversely, suppose it is verified; if $u'$ is a point of $\mathrm{U'}$, $t$ a point of $p^{-1}(p(u'))$; since $p(t')=p(u')$, there exists a point $t''\in\mathrm{T''}$ such that $p_1(t'')=u'$, $p_2(t'')=t'$ \sref[I]{I.3.4.7}; hence $t''\in p_1^{-1}(\mathrm{U'})=p_2^{-1}(\mathrm{U'})$ by hypothesis, hence $t'=p_2(t'')\in\mathrm{U'}$, which proves the lemma.
\end{proof}

This lemma being established, to show that $X'_0=p^{-1}(X_0)$, let us form the product $X''=X'\times_X X'$ (relative to the morphism $p:X'\to X$); if $p_1$ and $p_2$ are the canonical projections, it suffices to show that $p_1^{-1}(X'_0)=p_2^{-1}(X'_0)$.
Now, if we set $\mathrm{S}=\Spec(k)$, $\mathrm{S'}=\Spec(k')$, observe that if $\mathrm{S''}=\mathrm{S'}\times_{\mathrm{S}}\mathrm{S'}=\Spec(k'\otimes_k k')$, we can also write $X''=X'\times_{\mathrm{S'}}\mathrm{S''}$.
If $\varphi'':\mathrm{X''}\to\mathrm{S''}$ is the structural morphism, it suffices to prove that for every $s''\in\mathrm{S''}$, $p_1^{-1}(X'_0)\cap{\varphi''}^{-1}(s'')=p_2^{-1}(X'_0)\cap{\varphi''}^{-1}(s'')$.
If one sets $\mathrm{T}=\Spec(k(s''))$, $\mathrm{U}=X''\times_{\mathrm{S''}}\mathrm{T}$, and if $v:U\to X''$ is the canonical projection, and since we have the diagram

\[
\xymatrix{
  X & X' \ar@<0.5ex>[l]^{p_1} & X'' \ar@<0.5ex>[l]^{v} & \mathrm{U} \ar[l]\\
  \mathrm{S} \ar[u] & \mathrm{S'} \ar@<0.5ex>[l] \ar[u] & \mathrm{S''} \ar@<0.5ex>[l] \ar[u] & \mathrm{T} \ar[l] \ar[u]
}
\]

By construction, $\mathrm{U}_1$ and $\mathrm{U}_2$ contain both the set $w^{-1}(Z)$, where $w=p\circ p_1\circ v$, and the two irreducible components of $X'$ containing $Z'$; they are therefore all equal to $X'_0$ by \sref{IV.4.4.5}, and finally $\mathrm{U}_1=\mathrm{U}_2$.

Since $p$ is surjective and $X'_0$ dominates an irreducible component of $X$, $X_0=p(X'_0)$ is an irreducible component of $X$ containing $Z$, and since $p^{-1}(X_0)$ is an irreducible component of $X\otimes_k K$, $X_0$ is geometrically irreducible.

To prove the last assertion, let $X_1$ be a second irreducible component of $X$ meeting $Z$.
Taking $z\in Z\cap X_1$, one sees that there exists an irreducible component of $X'$ meeting $Z'$ and dominating $X_1$ \sref{IV.2.3.5}; but since $X'_0$ is by hypothesis the only irreducible component of $X'$ meeting $Z'$, we have necessarily $X_1=X_0$.

\begin{remark}[4.5.20]
\label{IV.4.5.20}
The fact that $Z'$ is contained in a single irreducible component $X'_0$ of $X'$ does \emph{not imply} that $X_0=p(X'_0)$ is the only irreducible component of $X$ containing $Z$.
\end{remark}

\begin{proposition}[4.5.21]
\label{IV.4.5.21}
Let $k$ be a separably closed field, and let $k'$ be an algebraic closure of $k$.
For every $k$-prescheme $X$, the canonical projection $X\otimes_k k'\to X$ is a universal homeomorphism.
In particular the irreducible \emph{(}resp.\ connected\emph{)} components of $X$ are geometrically irreducible \emph{(}resp.\ geometrically connected\emph{)}.
\end{proposition}

\begin{proof}
By definition, $k'$ is a purely inseparable extension of $k$; the morphism $\Spec(k')\to\Spec(k)$ is integral, surjective, and radical, hence the first assertion results from \sref{IV.2.4.5}, (i)); the second follows from \sref{IV.4.5.1}.
\end{proof}

\label{subsection:IV.4.6}

\begin{proposition}[4.6.1]
\label{IV.4.6.1}
Let $k$ be a field, $X$ a $k$-prescheme, $\Omega$ a perfect extension of $k$.
The following conditions are equivalent:
\begin{enumerate}[\rm a)]
  \item For every reduced $k$-prescheme $\mathrm{S}$, $X\times_k\mathrm{S}$ is reduced.
  \item For every extension $K$ of $k$, $X\otimes_k K$ is reduced.
  \item $X\otimes_k\Omega$ is reduced.
  \item For every finite purely inseparable extension $k'$ of $k$, $X\otimes_k k'$ is reduced.
  \item $X$ is reduced, and for every irreducible component $X_\alpha$ of $X$, with generic point $x_\alpha$, $k(x_\alpha)$ is a separable extension of $k$.
\end{enumerate}
\end{proposition}

\begin{proof}
It is trivial that $a)\Rightarrow b)\Rightarrow c)$; since $k'$ can be considered as a sub-extension of $\Omega$, we saw \sref{IV.4.4.1} that $c)$ implies $d)$.
Let us show that $d)$ implies $e)$.
Taking $k'=k$, one sees first that $X$ is reduced; to prove that $k(x_\alpha)$ is separable over $k$, it suffices to show that $k(x_\alpha)\otimes_k k'$ is a reduced ring for every finite purely inseparable extension $k'$ of $k$ \sref{IV.4.3.5}.
One can restrict to the case where $X=\Spec(A)$ is affine, $A$ being a reduced $k$-algebra; then by hypothesis the ring $A\otimes_k k'$ is reduced \sref[I]{I.5.1.4} and $k(x_\alpha)\otimes_k k'$ is a ring of fractions of this ring, hence is reduced $(\mathbf{0_I}$, 1.2.8).
Finally, to prove that $e)$ implies $a)$, one can restrict to the case where $X=\Spec(A)$, $\mathrm{S}=\Spec(B)$ are affine, $A$ and $B$ being $k$-algebras.
The $k(x_\alpha)$ are the fields of fractions of the quotient rings $A/\mathfrak{p}_\alpha$, where $\mathfrak{p}_\alpha=\mathfrak{i}_{x_\alpha}$ are the minimal prime ideals of $A$; since by hypothesis $\bigcap_\alpha\mathfrak{p}_\alpha=0$, $A$ is contained in the product $\prod A/\mathfrak{p}_\alpha$, hence in $\prod k(x_\alpha)$; one can similarly consider $B$ as a subalgebra of a product $\prod L_\beta$ of extensions of $k$.
It follows that $A\otimes_k B$ identifies with a subalgebra of $(\prod_\alpha k(x_\alpha))\otimes_k(\prod_\beta L_\beta)$, and this tensor product itself identifies with a subalgebra of $\prod_{\alpha,\beta}(k(x_\alpha)\otimes_k L_\beta)$ (Bourbaki, \emph{Alg.}, chap.~II, 3\textsuperscript{e} éd., \textsection7, no.~7, prop.~15).
Now by hypothesis the $k(x_\alpha)\otimes_k L_\beta$ are reduced \sref{IV.4.3.5}, hence the same holds for $A\otimes_k B$.
\end{proof}

\begin{definition}[4.6.2]
\label{IV.4.6.2}
When the equivalent conditions a) to e) of the proposition \sref{IV.4.6.1} are satisfied, one says that $X$ is \emph{geometrically reduced} (or \emph{separable}, or \emph{universally reduced}) over $k$.
One says that a $k$-prescheme $X$ is \emph{geometrically integral} over $k$ if, for every extension $K$ of $k$, $X\otimes_k K$ is integral; this amounts (by virtue of \sref{IV.4.6.1}) to saying that $X$ is separable and geometrically irreducible over $k$.
\end{definition}

One says that a $k$-algebra $A$ (commutative) is \emph{separable} if $\Spec(A)$ is separable over $k$; this means therefore that for every extension $K$ of $k$, the ring $A_{(K)}=A\otimes_k K$ is reduced.
One will note that this definition coincides with that of Bourbaki, \emph{Alg.}, chap.~VIII, \textsection7, no.~5, def.~1, when $A$ is a ring of finite rank over $k$ (loc.~cit., cor.\ of prop.~7), but not in general, a $k$-algebra being able to have zero radical $\neq 0$ even if it is integral.

\begin{corollary}[4.6.3]
\label{IV.4.6.3}
Let $X$ be a $k$-prescheme integral; for $X$ to be geometrically reduced \emph{(}resp.\ geometrically integral\emph{)} over $k$, it is necessary and sufficient that its field of rational functions $\mathrm{R}(X)$ be a separable \emph{(}resp.\ separable and primary\emph{)} extension of $k$.
\end{corollary}

\begin{proof}
This follows immediately from \sref{IV.4.5.9} and \sref{IV.4.6.1}.
\end{proof}

\begin{corollary}[4.6.4]
\label{IV.4.6.4}
Let $X$ be a $k$-prescheme reduced.
Then, for every separable extension $k'$ of $k$, $X'=X\otimes_k k'$ is reduced.
\end{corollary}

\begin{proof}
It suffices to apply the equivalence of $a)$ and $e)$ in \sref{IV.4.6.1}, replacing $X$ by $\Spec(k')$ and $\mathrm{S}$ by $X$.
\end{proof}

\begin{proposition}[4.6.5]
\label{IV.4.6.5}
\begin{enumerate}[\rm(i)]
  \item Let $X$ be a $k$-prescheme, $K$ an extension of $k$.
  For $X$ to be geometrically reduced \emph{(}resp.\ geometrically integral\emph{)} over $k$, it is necessary and sufficient that $X\otimes_k K$ be geometrically reduced \emph{(}resp.\ geometrically integral\emph{)} over $K$.
  \item Let $X$, $Y$ be two $k$-preschemes.
  If $X$ and $Y$ are geometrically reduced \emph{(}resp.\ geometrically integral\emph{)} over $k$, the same holds for $X\times_k Y$.
\end{enumerate}
\end{proposition}

\begin{proof}
Assertion (i) is a trivial consequence of the definitions and of \sref{IV.4.4.1}.
For the part of assertion (ii) concerning separability, observe that if $\Omega$ is an algebraically closed extension of $k$, $(X\times_k Y)\otimes_k\Omega=X\times_k(Y\otimes_k\Omega)$; since $Y\otimes_k\Omega$ is reduced, the same holds for $X\times_k(Y\otimes_k\Omega)$ by virtue of \sref{IV.4.6.1}, $a)$).
The rest of assertion (ii) results from what precedes and from \sref{IV.4.5.8}, (ii).
\end{proof}

\begin{proposition}[4.6.6]
\label{IV.4.6.6}
Let $X$ be a $k$-prescheme of finite type.
There exists a finite purely inseparable extension $k'$ of $k$ such that $(X_{(k')})_\mathrm{red}$ is geometrically reduced over $k'$.
\end{proposition}

\begin{proof}
Since $X$ can be covered by a finite number of open affines $U_i$, one can restrict to showing that for each $i$, there is a finite purely inseparable extension $k_i$ of $k$ such that $(U_i\otimes_k k_i)_\mathrm{red}$ is geometrically reduced over $k_i$; one can then suppose that the $k_i$ are contained in a same finite purely inseparable extension $k'$ of $k$; $(U_i\otimes_k k')_\mathrm{red}$ identifies with $(U_i\otimes_k k_i)_\mathrm{red}\otimes_{k_i} k'$ by virtue of \sref[I]{I.5.1.8}, hence is geometrically reduced over $k'$ by hypothesis, and the same holds for $(X\otimes_k k')_\mathrm{red}$.
Suppose therefore $X=\Spec(A)$, where $A$ is a $k$-algebra of finite type, and let $\Omega$ be the algebraic closure of $k$.
Denote by $\mathfrak{N}'$ the nilradical of $A\otimes_k\Omega$; since $A\otimes_k\Omega$ is a Noetherian ring, $\mathfrak{N}'$ is an ideal generated by a finite number of elements of the form $y_i=\sum_j x_{ij}\otimes\xi_{ij}$ where $x_{ij}\in A$, $\xi_{ij}\in\Omega$.
Let $K$ be a finite sub-extension of $\Omega$ containing the $\xi_{ij}$, and $\mathfrak{N}$ the ideal of $A\otimes_k K$ generated by the $y_i$ ($A\otimes_k K$ being identified with a subring of $A\otimes_k\Omega$); it is clear that the $y_i$ are nilpotent in $A\otimes_k K$; on the other hand, since $\mathfrak{N}\otimes_K\Omega=\mathfrak{N}'$ (and hence $\mathfrak{N}\cap(A\otimes_k K)=\mathfrak{N}$), $\mathfrak{N}$ contains the nilradical of $A\otimes_k K$, hence it is equal to this nilradical.
Since $(A\otimes_k\Omega)/\mathfrak{N}'=((A\otimes_k K)/\mathfrak{N})\otimes_K\Omega$, one sees that $(X\otimes_k K)_\mathrm{red}\otimes_K\Omega$ is reduced, and by \sref{IV.4.6.1}, $(X_{(K)})_\mathrm{red}$ is geometrically reduced over $K$.
Replacing $K$ by the quasi-Galois extension over $k$ that it generates in $\Omega$, one sees by \sref[I]{I.5.1.8} that one can suppose moreover $K$ quasi-Galois over $k$, hence a separable extension of a finite purely inseparable extension $k'$ of $k$ (Bourbaki, \emph{Alg.}, chap.~V, \textsection10, no.~9, prop.~14).
By virtue of \sref[I]{I.5.1.8}, $(X_{(k')})_\mathrm{red}\otimes_{k'}\overline{k}$ is isomorphic to $(X_{(K)})_\mathrm{red}$, hence is geometrically reduced; the same holds for $(X_{(k')})_\mathrm{red}\otimes_{k'}\overline{k}\cdot K$ by \sref{IV.4.6.5}, (i); replacing $K$ again we are done.
\end{proof}

If $K$ is an extension of finite type of $k$, there exists a finite purely inseparable extension $k'$ of $k$ such that the residue fields of the semi-local ring $K\otimes_k k'$ are separable over $k'$.
\end{corollary}

\begin{proof}

There exists a finite extension $k'$ of $k$ such that $(X_{(k')})_\mathrm{red}$ is geometrically reduced over $k'$ and the irreducible components of $X_{(k')}$ are geometrically irreducible and the connected components of $X_{(k')}$ are geometrically connected.

\begin{definition}[4.6.9]
\label{IV.4.6.9}
Let $k$ be a field, $X$ a $k$-prescheme, $x$ a point of $X$.
One says that $X$ is \emph{geometrically reduced} (or \emph{separable}) (resp.\ \emph{geometrically punctually integral}) at the point $x$ over $k$, if, for every extension $k'$ of $k$ and every point $x'$ of $X'=X\otimes_k k'$ above $x$, $X'$ is reduced (resp.\ integral) at $x'$ (i.e., $\sh{O}_{X',x'}$ is reduced (resp.\ integral)).
One says that $X$ is \emph{geometrically punctually integral} if $X$ is geometrically punctually integral at all its points, that is, if for every extension $k'$ of $k$, all the local rings of $X\otimes_k k'$ are integral (in which case one says also that $X\otimes_k k'$ is \emph{punctually integral}).
\end{definition}

Note that, for $X$ to be geometrically reduced over $k$ \sref{IV.4.6.2}, it is necessary and sufficient that it be geometrically reduced over $k$ at every point $x\in X$.

\begin{proposition}[4.6.10]
\label{IV.4.6.10}
Let $k$ be a field, $X$ a $k$-prescheme, $k'$ an extension of $k$, $x'$ a point of $X'=X\otimes_k k'$, $x$ its image in $X$.
For $X$ to be geometrically punctually reduced \emph{(}resp.\ geometrically punctually integral\emph{)} at $x$ over $k$, it is necessary and sufficient that $X'$ be so at $x'$ over $k'$.
In particular, for $X$ to be geometrically punctually reduced \emph{(}resp.\ geometrically punctually integral\emph{)} over $k$, it is necessary and sufficient that $X'$ be so over $k'$.
\end{proposition}

\begin{proof}
It suffices to prove the first assertion.
The condition being obviously necessary, let us prove it is sufficient.
Suppose therefore $X'$ geometrically punctually reduced (resp.\ geometrically punctually integral) at $x'$ over $k'$, and let us prove that $X$ is so at $x$ over $k$, that is, for every extension $k''$ of $k$ and every point $x''$ of $X''=X\otimes_k k''$ above $x$, $\sh{O}_{X'',x''}$ is reduced (resp.\ integral).
For this, note that there exists a point $z$ of $X'\times_X X''$ projecting to $x'$ and $x''$ \sref[I]{I.3.4.7}.
Let $s$ be the point of $\Spec(k'\otimes_k k'')$ image of $z$ \sref[I]{I.3.4.9}; set $K=k(s)$, $Z=X\otimes_k K$, so that $K$ can be considered as a composite extension of $k'$ and $k''$, and $z$ as a point of $Z$ whose image in $X'$ (resp.\ $X''$) is $x'$ (resp.\ $x''$).
Since $X'$ is geometrically punctually reduced (resp.\ geometrically punctually integral) at $x'$ over $k'$, the ring $\sh{O}_{Z,z}$ is reduced (resp.\ integral), and since $\sh{O}_{Z,z}$ is an $\sh{O}_{X'',x''}$-module faithfully flat, it follows that $\sh{O}_{X'',x''}$ is reduced (resp.\ integral) $(\mathbf{0_I}$, 6.5.1).
\end{proof}

\begin{corollary}[4.6.11]
\label{IV.4.6.11}
Suppose $k$ perfect \emph{(}resp.\ algebraically closed\emph{)}.
For $X$ to be geometrically punctually reduced \emph{(}resp.\ geometrically punctually integral\emph{)} at the point $x$ over $k$, it is necessary and sufficient that $\sh{O}_{X,x}$ be reduced \emph{(}resp.\ integral\emph{)}.
\end{corollary}

\begin{proof}
This is obviously necessary.
Conversely, if this condition is satisfied, then, for every extension $k'$ of $k$, $\Spec(\sh{O}_{X,x})\otimes_k k'$ is reduced \sref{IV.4.6.4} (resp.\ integral \sref{IV.4.4.4}); hence, for every point $x'$ of $X'=X\otimes_k k'$ above $x$, $\sh{O}_{X',x'}$, which is isomorphic to a local ring of $\Spec(\sh{O}_{X,x})\otimes_k k'$, is reduced (resp.\ integral).
\end{proof}

\begin{proposition}[4.6.12]
\label{IV.4.6.12}
Let $k$ be a field, $X$ a $k$-prescheme, $x$ a point of $X$, $\Omega$ a perfect extension of $k$.
The following conditions are equivalent:
\begin{enumerate}[\rm a)]
  \item $X$ is geometrically reduced at $x$ over $k$, that is, for every extension $k'$ of $k$ and every point $x'$ of $X'=X\otimes_k k'$ above $x$, $\sh{O}_{X',x'}$ is reduced.
  \item The prescheme $X\otimes_k\Omega$ is reduced at every point above $x$.
  \item For every finite purely inseparable extension $k'$ of $k$, $X'=X\otimes_k k'$ is reduced at the unique point above $x$.
  \item $\Spec(\sh{O}_{X,x})$ is geometrically reduced over $k$.
  \item $\sh{O}_{X,x}$ is reduced, and for every irreducible component $Z$ of $X$ containing $x$, with generic point $z$, $k(z)$ is a separable extension of $k$.
\end{enumerate}
\end{proposition}

\begin{proof}
The implications $d)\Rightarrow a)\Rightarrow b)\Rightarrow c)$ are immediate, taking into account for the last that every purely inseparable extension of $k$ is isomorphic to a sub-extension of $\Omega$.
Let us prove that $c)$ implies $d)$.
It suffices to show that for every finite purely inseparable extension $k'$ of $k$, $\sh{O}_{X,x}\otimes_k k'$ is reduced \sref{IV.4.6.1}; now this ring is the local ring of $X'=X\otimes_k k'$ at the unique point $x'$ of $X'$ above $x$ \sref[I]{I.3.5.8} and \sref{IV.3.5.7}; it is therefore reduced by hypothesis $c)$.
Finally, the equivalence of $d)$ and $e)$ results from the equivalence of $b)$ and $e)$ in \sref{IV.4.6.1}, since the irreducible components of $X$ containing $x$ correspond bijectively to the irreducible components of $\Spec(\sh{O}_{X,x})$ \sref[I]{I.2.4.2}.
\end{proof}

\begin{corollary}[4.6.13]
\label{IV.4.6.13}
Under the hypotheses of \sref{IV.4.6.12}, suppose moreover that $X$ is locally Noetherian.
Then conditions a) to e) of \sref{IV.4.6.12} are also equivalent to the following:
\begin{enumerate}
  \item[\rm f)] There exists a neighbourhood $U$ of $x$ that is geometrically reduced over $k$.
\end{enumerate}
\end{corollary}

\begin{proof}
Indeed, this condition trivially implies that $X$ is geometrically reduced at $x$ over $k$.
Conversely, if this is so, since $\sh{O}_{X,x}$ is reduced, there exists a reduced open neighbourhood $U$ of $x$ in $X$ \sref[I]{I.6.1.13}; taking $U$ small enough, we can moreover suppose that $U$ meets no irreducible components of $X$ not containing $x$.
Then the criterion \sref{IV.4.6.12}, $e)$) proves that $U$ is geometrically reduced over $k$, taking into account the criterion \sref{IV.4.6.1}, $e)$).
\end{proof}

It follows from \sref{IV.4.6.13} that when $X$ is locally Noetherian, the set of $x\in X$ at which $X$ is geometrically reduced over $k$ is \emph{open}, and it is the largest open subprescheme of $X$ that is geometrically reduced over $k$.

\begin{env}[4.6.14]
\label{IV.4.6.14}
It results from \sref{IV.4.6.10} and \sref{IV.4.6.11} that if $\Omega$ is an algebraically closed extension of $k$, it amounts to the same to say that $X$ is geometrically punctually integral at $x$, or that $X\otimes_k\Omega$ is integral at every point above $x$.

For $X$ to be geometrically punctually integral at $x$, it is necessary that $X$ be integral at $x$, and that if $z$ is the generic point of the unique irreducible component of $X$ containing $x$, $k(z)$ be a separable extension of $k$; this follows from \sref{IV.4.6.10}.
But these conditions are \emph{not sufficient}, as shown by the example in \sref{IV.4.5.12}, (ii)).
A sufficient but non-necessary condition is that $X$ be geometrically reduced at $x$ and that $x$ belong to only a single irreducible component of $X$, and that this component be moreover geometrically irreducible; this follows from \sref{IV.4.6.1} and \sref{IV.4.5.9}.
\end{env}

\oldpage[IV-2]{72}

Recall that if a prescheme \emph{locally Noetherian} is punctually integral, it is \emph{locally integral} \sref[I]{I.6.1.13}.
If a $k$-prescheme $X$, locally of finite type over $k$, is geometrically punctually integral over $k$, it follows from this remark that, for every extension $k'$ of $k$, $X\otimes_k k'$ is \emph{locally integral} (one says in this case that $X$ is \emph{geometrically locally integral}).

\begin{proposition}[4.6.15]
\label{IV.4.6.15}
\begin{enumerate}[\rm(i)]
  \item Let $k$ be a field, $X$ a $k$-prescheme of finite type.
  For $X$ to be geometrically punctually integral, it is necessary and sufficient that it be geometrically reduced and that the geometric number of its irreducible components be equal to the geometric number of its connected components.
  \item Let $X$ be a prescheme locally of finite type over $k$.
  If $X$ is geometrically punctually integral at a point $x$, there exists an open neighbourhood $U$ of $x$ that is geometrically punctually integral.
  In other words, the set of $x\in X$ at which $X$ is geometrically punctually integral is open.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Given an algebraically closed extension $\Omega$ of $k$, $X_{(\Omega)}$ is punctually integral (or locally integral, which amounts to the same since $X_{(\Omega)}$ is Noetherian) if and only if it is reduced and the number of its connected components is equal to the number of its irreducible components \sref[I]{I.6.1.10}.
It then suffices to apply \sref{IV.4.5.1}, \sref{IV.4.6.1} and the first remark of \sref{IV.4.6.14}.

(ii) The question being local on $X$, one can restrict to the case where $X$ is a prescheme of finite type over $k$.
Moreover, one knows that there is an open neighbourhood of $x$ in $X$ which is geometrically reduced \sref{IV.4.6.13}, so one can suppose $X$ geometrically reduced.
There then exists a finite extension $k'$ of $k$ such that $X'=X\otimes_k k'$ is reduced and its irreducible components are geometrically irreducible \sref{IV.4.6.8}.
Let $p:X'\to X$ be the canonical projection, which is a finite surjective morphism, and let $x'_j$ ($1\leq j\leq r$) be the points of $p^{-1}(x)$; by hypothesis, $X'$ is integral at each $x'_j$, hence each $x'_j$ has an open integral neighbourhood $V'_j$ in $X'$.
Since $p$ is closed \sref[II]{II.6.1.10}, each of the $p(V'_j)$ is a neighbourhood of $x$, hence there exists an open neighbourhood $U$ of $x$ such that $p^{-1}(U)$ is contained in the union of the $V'_j$, and is therefore integral at each of its points, hence locally integral.
The irreducible components of $p^{-1}(U)$ are therefore open and pairwise disjoint, and since they are geometrically irreducible \sref{IV.4.5.9}, $p^{-1}(U)=U\otimes_k k'$ is geometrically punctually integral by (i), and the same holds for $U$ by definition.
\end{proof}

\begin{proposition}[4.6.16]
\label{IV.4.6.16}
Let $k$ be a field, $X$ a $k$-prescheme locally Noetherian, $\sh{F}$ a coherent $\sh{O}_X$-Module.
The following conditions are equivalent:
\begin{enumerate}[\rm a)]
  \item For every extension of finite type $k'$ of $k$, setting $X'=X\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$, $\sh{F}'$ is a reduced $\sh{O}_{X'}$-Module \sref{IV.3.2.2}.
  \item For every finite purely inseparable extension $k'$ of $k$, $\sh{F}'$ is a reduced $\sh{O}_{X'}$-Module.
  \item $\sh{F}$ is reduced, and if $\sh{J}$ is the Ideal of $\sh{O}_X$ annihilating $\sh{F}$, the closed subprescheme defined by $\sh{J}$ is geometrically reduced over $k$.
\end{enumerate}
Moreover, if $X$ is locally of finite type over $k$, these conditions are also equivalent to:
\begin{enumerate}
  \item[\rm d)] For every extension $k'$ of $k$ \emph{(}or for an algebraically closed extension $k'$ of $k$\emph{)}, $\sh{F}'$ is a reduced $\sh{O}_{X'}$-Module.
\end{enumerate}
\end{proposition}

\oldpage[IV-2]{73}

\begin{proof}
It is clear that $a)$ implies $b)$.
Condition $b)$ implies first that $\sh{F}$ is reduced: this means \sref{IV.3.2.3} that if $Y$ is the closed subprescheme of $X$ defined by the coherent Ideal $\sh{J}$ of $\sh{O}_X$, annihilator of $\sh{F}$, $Y$ is reduced, and there is an $\sh{O}_Y$-Module coherent without torsion $\sh{G}$ of rank 1 on every irreducible component of $Y$, such that $j_*(\sh{G})=\sh{F}$, $j:Y\to X$ being the canonical injection.
Now, for \emph{every} extension $k'$ of $k$, the annihilator of $\sh{F}'$ is $\sh{J}'=\sh{J}\otimes_k k'$, the morphism projection $X'\to X$ being flat \sref{IV.2.1.11}; the closed subprescheme of $X'$ defined by $\sh{J}'$ is $Y'=Y\otimes_k k'$, and if $j':Y'\to X'$ is the canonical injection, and $\sh{G}'=\sh{G}\otimes_k k'$, we have $\sh{F}'=j'_*(\sh{G}')$.
Note also that, for every extension $k'$ of $k$ for which $X'$ is locally Noetherian, $\sh{F}'$ is without immersed associated prime cycle \sref{IV.4.2.7}.
Moreover, every maximal point $y'$ of $Y'$ is above a maximal point $y$ of $Y$; since $\sh{G}_{y'}$ is isomorphic to $\sh{O}_{Y,y}$, $\sh{G}'_{y'}$ is isomorphic to $\sh{O}_{Y',y'}$.
One sees therefore (by \sref{IV.3.2.3}) that, whether $\sh{F}'$ is reduced or the prescheme $Y'$ is reduced, it amounts to the same thing.
The fact that $b)$ implies $c)$ and that $c)$ implies $a)$ are therefore consequences of \sref{IV.4.6.1}, $d)$ and $b)$); in $a)$, the hypothesis that $k'$ is an extension of finite type of $k$ is only used to ensure that $X'$ is locally Noetherian.
When $X$ is locally of finite type over $k$, $X'$ is locally Noetherian for every extension $k'$ of $k$, which proves in this case the equivalence of $d)$ and the other conditions.
\end{proof}

\begin{definition}[4.6.17]
\label{IV.4.6.17}
When $\sh{F}$ satisfies the equivalent conditions of \sref{IV.4.6.16}, one says that $\sh{F}$ is \emph{geometrically reduced} (or \emph{separable}) over $k$.
\end{definition}

\begin{proposition}[4.6.18]
\label{IV.4.6.18}
Let $k$ be a field, $X$ a $k$-prescheme locally Noetherian, $\sh{F}$ a coherent $\sh{O}_X$-Module.
The following conditions are equivalent:
\begin{enumerate}[\rm a)]
  \item For every extension of finite type $k'$ of $k$, setting $X'=X\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$, $\sh{F}'$ is an integral $\sh{O}_{X'}$-Module \sref{IV.3.2.4}.
  \item For every finite extension $k'$ of $k$, $\sh{F}'$ is an integral $\sh{O}_{X'}$-Module.
  \item $\sh{F}$ is reduced (or integral), and if $\sh{J}$ is the Ideal of $\sh{O}_X$ annihilating $\sh{F}$, the closed subprescheme defined by $\sh{J}$ is geometrically integral over $k$.
\end{enumerate}
Moreover, if $X$ is locally of finite type over $k$, these conditions are also equivalent to:
\begin{enumerate}
  \item[\rm d)] For every extension $k'$ of $k$ \emph{(}or for an algebraically closed extension $k'$ of $k$\emph{)}, $\sh{F}'$ is an integral $\sh{O}_{X'}$-Module.
\end{enumerate}
\end{proposition}

\begin{proof}
Conditions $a)$, $b)$ and $c)$ imply that $\sh{F}$ is integral, hence $\operatorname{Ass}(\sh{F})$ reduces to a single point $x$, which is the generic point of $\Supp(\sh{F})$.
For \emph{every} extension $k'$ of $k$ for which $X'$ is locally Noetherian, the associated prime cycles of $\sh{F}'$ are therefore non-immersed, and are the maximal points of $\Supp(\sh{F}')$, which are all above $x$.
On the other hand, by virtue of \sref{IV.4.6.16}, each of the conditions $a)$, $b)$, $c)$ implies that $\sh{F}$ is geometrically reduced; it is trivial that $a)$ implies $b)$, and $b)$ implies that $\Supp(\sh{F})$ is geometrically irreducible \sref{IV.4.5.9}, hence $c)$, since the subprescheme $Y$ defined by the annihilator $\sh{J}$ of $\sh{F}$ is then geometrically reduced and geometrically irreducible, hence geometrically integral; this shows that $c)$ implies $a)$, and $c)$ implies $d)$ when $X$ is locally of finite type over $k$.
\end{proof}

\begin{definition}[4.6.19]
\label{IV.4.6.19}
When $\sh{F}$ satisfies the equivalent conditions of \sref{IV.4.6.18}, one says that $\sh{F}$ is \emph{geometrically integral} over $k$.
\end{definition}

\begin{proposition}[4.6.20]
\label{IV.4.6.20}
Let $k$ be a field, $X$ a $k$-prescheme locally Noetherian, $\sh{F}$ a coherent $\sh{O}_X$-Module.
Let $K$ be an extension of $k$ such that $X\otimes_k K$ is locally Noetherian.
Then, if $\sh{F}$ is geometrically reduced \emph{(}resp.\ geometrically integral\emph{)} over $k$, $\sh{F}\otimes_k K$ is geometrically reduced \emph{(}resp.\ geometrically integral\emph{)} over $K$.
\end{proposition}

\begin{proof}
Indeed, for every finite extension $K'$ of $K$, $(X\otimes_k K)\otimes_K K'=X\otimes_k K'$ is locally Noetherian, and one saw in the proofs of \sref{IV.4.6.16} and \sref{IV.4.6.18} that the hypothesis on $\sh{F}$ implies that $\sh{F}\otimes_k K'$ is reduced (resp.\ integral) over $K$, whence the conclusion.
\end{proof}

\begin{proposition}[4.6.21]
\label{IV.4.6.21}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes locally Noetherian such that $X\times_k Y$ is locally Noetherian.
Let $\sh{F}$ be a coherent $\sh{O}_X$-Module, $\sh{G}$ a coherent $\sh{O}_Y$-Module.
\begin{enumerate}[\rm(i)]
  \item If $\sh{F}$ is geometrically reduced \emph{(}resp.\ geometrically integral\emph{)} and $\sh{G}$ reduced \emph{(}resp.\ integral\emph{)}, then $\sh{F}\otimes_k\sh{G}$ is reduced \emph{(}resp.\ integral\emph{)}.
  \item If $\sh{F}$ and $\sh{G}$ are geometrically reduced \emph{(}resp.\ geometrically integral\emph{)}, the same holds for $\sh{F}\otimes_k\sh{G}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Using \sref{IV.3.2.3}, \sref{IV.4.6.16} and \sref{IV.4.6.18}, one can restrict to the case where $\Supp(\sh{F})=X$, $\Supp(\sh{G})=Y$, $X$ being geometrically reduced (resp.\ geometrically integral), $Y$ reduced (resp.\ integral), $\sh{F}$ and $\sh{G}$ without immersed associated prime cycles, and $\sh{F}_x$ isomorphic to $\sh{O}_x$ at every maximal point $x$ of $X$, $\sh{G}_y$ isomorphic to $\sh{O}_y$ at every maximal point $y$ of $Y$.
One knows then \sref{IV.4.2.3} that $\sh{F}\otimes_k\sh{G}$ is without immersed associated prime cycle, and that the associated prime cycles of $\sh{F}\otimes_k\sh{G}$ are exactly the irreducible components of $X\times_k Y$ \sref{IV.4.2.5}.
One can therefore restrict to the case where $X$ and $Y$ are irreducible; then, if $X$ is geometrically reduced and $Y$ reduced, one knows \sref{IV.4.2.4}, (ii)) that $X\times_k Y$ is reduced; if $X$ is geometrically integral and $Y$ integral, one knows that $X\times_k Y$ is also irreducible \sref{IV.4.5.8}, (i)).
Finally, every maximal point $z$ of $X\times_k Y$ being above $x$ and $y$, the hypotheses on $\sh{F}$ and $\sh{G}$ imply, by virtue of \sref[I]{I.9.1.12} that $(\sh{F}\otimes_k\sh{G})_z$ is isomorphic to $\sh{O}_z$, which finishes the proof in this case.

(ii) The reasoning is analogous, but here (after reduction to the case where $X=\Supp(\sh{F})$, $Y=\Supp(\sh{G})$), $X\times_k Y$ is geometrically reduced (resp.\ geometrically integral) by virtue of \sref{IV.4.6.5}, (ii).
\end{proof}

The definitions \sref{IV.4.6.18} and \sref{IV.4.6.19} localize:

\begin{definition}[4.6.22]
\label{IV.4.6.22}
Let $k$ be a field, $X$ a $k$-prescheme locally Noetherian, $\sh{F}$ a coherent $\sh{O}_X$-Module.
One says that $\sh{F}$ is \emph{geometrically reduced} (or \emph{separable}) (resp.\ \emph{geometrically punctually integral}) at a point $x\in X$ if, for every finite purely inseparable \emph{(}resp.\ finite\emph{)} extension $k'$ of $k$, $\sh{F}\otimes_k k'$ is reduced (resp.\ integral) at each of the points $x'$ of $X\otimes_k k'$ above $x$ (cf.\ \sref{IV.3.2.2} and \sref{IV.3.2.4}).
\end{definition}

\oldpage[IV-2]{75}

If $\sh{F}$ is reduced at $x$, to say that $\sh{F}$ is geometrically reduced at $x$ is equivalent to saying that $\sh{F}$ is reduced at $x$ and that the closed subprescheme $Y$ defined by the annihilator $\sh{J}$ of $\sh{F}$ is geometrically reduced at $x$; it follows that if $X$ is locally of finite type over $k$, saying that $\sh{F}$ is geometrically punctually integral at $x$ is equivalent to saying that $\sh{F}$ is reduced at $x$ and that the subprescheme $Y$ is \emph{geometrically punctually integral at $x$}.

\subsection{Multiplicities in the primary decomposition on an algebraic prescheme}
\label{subsection:IV.4.7}

\begin{lemma}[4.7.1]
\label{IV.4.7.1}
Let $A$, $B$ be two local rings, $\mathfrak{m}$ the maximal ideal of $A$, $\rho:A\to B$ a local homomorphism.
Suppose that $B$ is a flat $A$-module, and $B/\mathfrak{m}B$ a $B$-module of finite length.
Then, for an $A$-module $M$ of finite length, it is necessary and sufficient that $M_{(B)}$ be a $B$-module of finite length, and we have
\begin{equation}
\label{IV.4.7.1.1}
  \operatorname{long}_B(M_{(B)})=\operatorname{long}_A(M)\cdot\operatorname{long}_B(B/\mathfrak{m}B).\tag{4.7.1.1}
\end{equation}
\end{lemma}

\begin{proof}
This is a particular case of \sref{IV.2.5.5.2}.
\end{proof}

\begin{corollary}[4.7.2]
\label{IV.4.7.2}
Let $A$, $B$ be two Noetherian rings, $\rho:A\to B$ a ring homomorphism.
Let $\mathfrak{q}$ be a minimal prime ideal of $B$, $\mathfrak{p}=\rho^{-1}(\mathfrak{q})$.
If $B$ is a flat $A$-module, then $\mathfrak{p}$ is a minimal prime ideal of $A$, $M_\mathfrak{p}$ is an $A_\mathfrak{p}$-module of finite length, and we have
\begin{equation}
\label{IV.4.7.2.1}
  \operatorname{long}_{B_\mathfrak{q}}(M_{(B)})_\mathfrak{q}=(\operatorname{long}_{A_\mathfrak{p}}(M_\mathfrak{p}))(\operatorname{long}_{B_\mathfrak{q}}(B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q})).\tag{4.7.2.1}
\end{equation}
\end{corollary}

\begin{proof}
Indeed, $B_\mathfrak{q}$ is a flat $A_\mathfrak{p}$-module $(\mathbf{0_I}$, 6.3.2) and the homomorphism $A_\mathfrak{p}\to B_\mathfrak{q}$ is $A_\mathfrak{p}$-flat, hence faithfully flat $(\mathbf{0_I}$, 6.6.2); the fact that $\mathfrak{p}$ is minimal follows from \sref{IV.2.3.5}.
Moreover, $A_\mathfrak{p}$ is then an Artinian ring (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection2, no.~5, prop.~9) and $M_\mathfrak{p}$ is therefore an $A_\mathfrak{p}$-module of finite length; the formula \sref{IV.4.7.2.1} is therefore a particular case of \sref{IV.4.7.1.1}.
\end{proof}

\begin{proposition}[4.7.3]
\label{IV.4.7.3}
Let $k$ be a field of characteristic exponent $p$, $X$ an integral prescheme locally Noetherian, $K=\mathrm{R}(X)$ the field of rational functions of $X$.
Let $k'$ be an extension of $k$, $(X'_\alpha)$ the family of irreducible components of $X'=X_{(k')}$, $x'_\alpha$ the generic point of $X'_\alpha$.
\begin{enumerate}[\rm(i)]
  \item Suppose that $X'$ is locally Noetherian \emph{(}which occurs if $X$ is of finite type over $k$, or if $k'$ is an extension of finite type of $k$\emph{)}.
  Then for $k''$ an extension of $k'$, $(X''_\beta)$ the family of irreducible components of $X''$, $x''_\beta$ the generic point of $X''_\beta$.
  If $x''_\beta$ is above $x'_\alpha$, we have
  \begin{equation}
  \label{IV.4.7.3.1}
    \operatorname{long}(\sh{O}_{x''_\beta})=\operatorname{long}(\sh{O}_{x'_\alpha})\cdot\operatorname{long}(\sh{O}_{Z'',x''_\beta})\tag{4.7.3.1}
  \end{equation}
  where we have set $Z''=(X'_\mathrm{red})\otimes_k k''$.
  In particular, if $k''$ is a separable extension of $k'$ such that $Z''$ is locally Noetherian, we have $\operatorname{long}(\sh{O}_{x''_\beta})=\operatorname{long}(\sh{O}_{x'_\alpha})$.
  \item If $X$ is locally of finite type over $k$, or if $k'$ is an extension of finite type of $k$, the numbers $\operatorname{long}(\sh{O}_{x'_\alpha})$ are powers of $p$.
  \item Suppose $X$ is locally of finite type over $k$.
  Then, if $k'$ is perfect, all the numbers $\operatorname{long}(\sh{O}_{x'_\alpha})$ are equal and do not depend on the particular perfect extension $K$ of $k$ considered.
  Moreover, there exists a finite purely inseparable extension $k_1$ of $k$ such that, if $x_1$ is the generic point of $X_1=X\otimes_k k_1$ \emph{(}which is irreducible\emph{)}, we have $\operatorname{long}(\sh{O}_{x'_\alpha})=\operatorname{long}(\sh{O}_{x_1})$ for every $\alpha$.
\end{enumerate}
\end{proposition}

\oldpage[IV-2]{76}

\begin{proof}
(i) Recall \sref{IV.4.2.4} that the irreducible components $X'_\alpha$, which are finite in number, correspond bijectively to the minimal prime ideals $\mathfrak{p}_\alpha$ of $K\otimes_k k'$, and that $\sh{O}_{x'_\alpha}$ is an Artinian ring, isomorphic to $(K\otimes_k k')_{\mathfrak{p}_\alpha}$; these rings therefore do not depend on the extensions $K$ and $k'$ of $k$ (and are therefore the same, $k'$ being fixed, for all $k$-preschemes integral having the same fields of rational functions).
The formula \sref{IV.4.7.3.1} results from \sref{IV.4.7.2.1} applied to the Noetherian ring $A$ of an affine neighbourhood of $x'_\alpha$ in $X'$, and to the ring $B$ of an affine neighbourhood small enough of $x''_\beta$ in $X''$, $\mathfrak{p}$ and $\mathfrak{q}$ being the minimal prime ideals corresponding to $x'_\alpha$ and $x''_\beta$ respectively, and taking $M=A$: one notes that $A/\mathfrak{p}$ is the ring of an affine neighbourhood of $x'_\alpha$ in $X'_\mathrm{red}$, and that $B=A\otimes_{k'}k''$, and that $\sh{O}_{Z'',x''_\beta}=(B/\mathfrak{p}B)_\mathfrak{q}=B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q}$, whence \sref{IV.4.7.3.1}.
The last assertion of (i) results from the fact that in this case $Z''$ is reduced \sref{IV.4.2.4}, hence $\mathfrak{p}B_\mathfrak{q}=\mathfrak{q}B_\mathfrak{q}$.

(ii) If $k'$ is an extension of finite type of $k$, there is a purely transcendental extension $k'_0$ of $k$ such that $k'$ is a finite extension of $k'_0$; since $k'_0$ is separable over $k$, it results from (i) applied with $k'$ replaced by $k$ and $k''$ by $k'_0$ respectively, that we can replace $X$ by $X\otimes_k k'_0$, in other words suppose $k'$ is an extension \emph{finite} of $k$.
Then there is a finite quasi-Galois extension $k''$ of $k$ containing $k'$, and the formula \sref{IV.4.7.3.1} shows that it suffices to prove the assertion for $k''$.
But $k''$ is a separable extension of a finite purely inseparable extension $k_1$ of $k$, hence, by virtue of (i), one is reduced to proving the assertion when $k'=k_1$ is \emph{purely inseparable}.
One knows that $K\otimes_k k_1$ is an Artinian ring having only a single minimal prime ideal \sref{IV.4.3.2}; if $\overline{K}$ is the algebraic closure of $K$, the length of $K\otimes_k k_1$ divides that of the Artinian ring $\overline{K}\otimes_k k_1$ by virtue of \sref{IV.4.7.1.1}.
But $\overline{K}\otimes_k k_1$ is a $\overline{K}$-algebra having only a single prime ideal (necessarily maximal), and its quotient by this ideal is necessarily a finite extension of $\overline{K}$, hence necessarily equal to $\overline{K}$; the length of $\overline{K}\otimes_k k_1$ is therefore equal to its rank over $\overline{K}$, that is, $[k_1:k]$, which is a power of $p$.

Suppose next that $X$ is locally of finite type over $k$; then, $X''$ is locally Noetherian for every extension $k''$ of $k'$.
Taking for $k''$ an algebraically closed extension, one is reduced, by virtue of \sref{IV.4.7.3.1}, to proving the assertion when $k'$ is the \emph{algebraic closure} $\overline{k}$.
Now, one knows (replacing $X$ by an open affine of finite type over $k$) that there is a finite purely inseparable extension $k_1$ of $k$ such that $(X\otimes_k k_1)_\mathrm{red}$ is separable over $k_1$ \sref{IV.4.6.6}; hence $(X\otimes_k k_1)_\mathrm{red}\otimes_{k_1}\overline{k}$ is reduced \sref{IV.4.6.4}, and the same holds for one and we conclude as above.

(iii) Note that two extensions $k'$, $k''$ of $k$ can always be contained in a same extension $E$ of $k$; it suffices then to compare the numbers for $k'$ and for an algebraically closed extension $E$ common to $\Omega$ and $\Omega'$ to show the first assertion.
The second is obtained by taking for $\Omega$ an algebraically closed extension $\overline{k}$ and using the fact that the extension $k_1$ provided by (ii) determines as unique generic point of $X_1=X\otimes_k k_1$, and the relation $\operatorname{long}(\sh{O}_{x'_\alpha})=\operatorname{long}(\sh{O}_{x_1})$ for every $\alpha$, follows from \sref{IV.4.7.3.1}.
\end{proof}

\begin{definition}[4.7.4]
\label{IV.4.7.4}
Let $k$ be a field, $X$ a $k$-prescheme integral, $K$ the field of rational functions of $X$.
The \emph{purely inseparable multiplicity} of $K$ (or of $X$) over $k$ is defined to be the upper bound of the lengths of the Artinian rings $K\otimes_k k_1$, when $k_1$ ranges over the set of finite purely inseparable extensions of $k$.
The \emph{separable multiplicity} of $K$ (or of $X$) over $k$ is defined to be the geometric number of irreducible components of $X$ if this number is finite, and $+\infty$ otherwise.
The \emph{total multiplicity} of $K$ (or of $X$) over $k$ is the product of the purely inseparable multiplicity and the separable multiplicity.
\end{definition}

One will note that if $p$ is the characteristic exponent of $k$ and if the purely inseparable multiplicity of $K$ over $k$ is finite, it is a power $q=p^f$ of $p$, as follows from \sref{IV.4.7.3}, (ii)); when $p\neq 1$, $f$ is well-determined and is called the \emph{exponent of inseparability} of $K$ over $k$; when $p=1$, the purely inseparable multiplicity is always equal to 1.

To say that the purely inseparable multiplicity (resp.\ separable, resp.\ total) of $X$ over $k$ is 1 means that $X$ is geometrically reduced over $k$ (resp.\ geometrically irreducible, resp.\ geometrically integral) over $k$.

When $X$ is locally of finite type over $k$, the purely inseparable multiplicity of $X$ over $k$ is also the common length of the local rings $A_i$ at the minimal prime ideals of $A=K\otimes_k k'$, for every perfect extension $k'$ of $k$; the separable multiplicity of $X$ over $k$ is the number of minimal prime ideals of $A=K\otimes_k\Omega$ for every algebraically closed extension $\Omega$ of $k$, and finally the total multiplicity is the sum of the lengths of the local rings $A_i$ of $A=K\otimes_k\Omega$ at its minimal prime ideals; these numbers are all three \emph{finite} in this case.

\begin{definition}[4.7.5]
\label{IV.4.7.5}
Let $k$ be a field, $X$ a $k$-prescheme locally of finite type over $k$, $\sh{F}$ a coherent $\sh{O}_X$-Module, $x$ a point of $X$ such that $\sh{F}_x$ is an $\sh{O}_x$-module of finite length.
The \emph{geometric length} of $\sh{F}$ at $x$ \emph{(}relative to $k$\emph{)}, or the \emph{purely inseparable multiplicity of $x$ for $\sh{F}$}, is the product $\lambda_x(\sh{F})=(\operatorname{long}_{\sh{O}_x}(\sh{F}_x))\mu_i(k(x)\,|\,k)$, where $\mu_i(k(x)\,|\,k)$ denotes the purely inseparable multiplicity of $k(x)$ over $k$.
\end{definition}

When $X$ is \emph{integral}, and $x$ the generic point of $X$, $\operatorname{long}(\sh{O}_x)=1$, hence the purely inseparable multiplicity of $x$ for $\sh{O}_X$ is none other than the purely inseparable multiplicity of $X$ over $k$, defined in \sref{IV.4.7.4}.

\begin{proposition}[4.7.6]
\label{IV.4.7.6}
Let $0\to\sh{F}'\to\sh{F}\to\sh{F}''\to 0$ be an exact sequence of coherent $\sh{O}_X$-Modules.
If at a point $x\in X$ the geometric length $\lambda_x(\sh{F})$ of $\sh{F}$ at this point is defined, the same holds for $\lambda_x(\sh{F}')$ and $\lambda_x(\sh{F}'')$, of $\sh{F}'$ and $\sh{F}''$ at the point $x$, and conversely; moreover, we have
\begin{equation}
\label{IV.4.7.6.1}
  \lambda_x(\sh{F})=\lambda_x(\sh{F}')+\lambda_x(\sh{F}'').\tag{4.7.6.1}
\end{equation}
\end{proposition}

\begin{proof}
The proposition results immediately from the definition, for $\operatorname{long}(\sh{F}_x)$ is finite if and only if $\operatorname{long}(\sh{F}'_x)$ and $\operatorname{long}(\sh{F}''_x)$ are finite, and we have $\operatorname{long}(\sh{F}_x)=\operatorname{long}(\sh{F}'_x)+\operatorname{long}(\sh{F}''_x)$.
\end{proof}

\begin{env}[4.7.7]
\label{IV.4.7.7}
Under the hypotheses of \sref{IV.4.7.5}, the only points $x\in X$ such that $\sh{F}_x$ is a non-zero $\sh{O}_x$-module of finite length are (by virtue of \sref{IV.3.1.2}) the \emph{maximal points} of $\Supp(\sh{F})$, as follows from Bourbaki, \emph{Alg.~comm.}, chap.~IV, \textsection2, no.~5, cor.~2 of prop.~7, the question being local on $X$.
One knows that there exists a closed subprescheme $Y$ having $\Supp(\sh{F})$ as underlying space, and a coherent $\sh{O}_Y$-Module $\sh{G}$ such that $\sh{F}=j_*(\sh{G})$, $j:Y\to X$ being the canonical injection.
If $x$ is a maximal point of $Y$, we see that the geometric lengths of $\sh{F}$ and of $\sh{G}$ at $x$ are equal, $\sh{O}_{X,x}$ and $\sh{O}_{Y,x}$ having the same residue fields.
\end{env}

\oldpage[IV-2]{78}

\begin{proposition}[4.7.8]
\label{IV.4.7.8}
Under the hypotheses of \sref{IV.4.7.5}, let $x$ be a maximal point of $\Supp(\sh{F})$.
Let $k'$ be a perfect extension of $k$; set $X'=X\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$; then the geometric length $\lambda_x(\sh{F})$ of $\sh{F}$ at $x$ is equal to the length of the $\sh{O}_{x'}$-module $\sh{F}'_{x'}$ at every maximal point $x'$ of $\Supp(\sh{F}')$ above $x$.
Moreover, there exists a finite purely inseparable extension $k_1$ of $k$ such that, setting $X_1=X\otimes_k k_1$, $\sh{F}_1=\sh{F}\otimes_k k_1$, $\lambda_x(\sh{F})$ is also equal to the length of the $\sh{O}_{x_1}$-module $(\sh{F}_1)_{x_1}$ at every maximal point $x_1$ of $\Supp(\sh{F}_1)$ above $x$.
\end{proposition}

\begin{proof}
The remark of \sref{IV.4.7.7} shows that one can restrict to the case where $x$ is a maximal point of $X$ and one can therefore suppose $X=\Spec(A)$ affine Noetherian, $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module of finite type; let $\mathfrak{p}$ (resp.\ $\mathfrak{q}$) be the minimal prime ideal of $A$ (resp.\ $B$) corresponding to $x$ (resp.\ $x'$).
We have then, applying the formula \sref{IV.4.7.2.1}:
\[
  \operatorname{long}_{\sh{O}_{x'}}(\sh{F}'_{x'})=\operatorname{long}_{\sh{O}_x}(\sh{F}_x)\cdot\operatorname{long}_{B_\mathfrak{q}}(B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q}).
\]
But since $k(x)=A_\mathfrak{p}/\mathfrak{p}A_\mathfrak{p}$, the same reasoning as in \sref{IV.4.7.3} shows that $\operatorname{long}_{B_\mathfrak{q}}(B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q})$ is equal to the length of the local ring $(B/\mathfrak{p}B)_{\mathfrak{q}'}$, where $\mathfrak{q}'=\mathfrak{q}/\mathfrak{p}B$, and this length is by definition $\mu_i(k(x)\,|\,k)$, whence the first assertion.
The second is proved similarly, replacing $k'$ by $k_1$ and using \sref{IV.4.7.3}, (iii).
\end{proof}

One says, in the case where $x$ is a maximal point of $\Supp(\sh{F})$, that the geometric length of $\sh{F}$ at $x$ is the \emph{purely inseparable multiplicity of the maximal prime cycle $\{x\}$ associated to $\sh{F}$}.

\begin{corollary}[4.7.9]
\label{IV.4.7.9}
Under the hypotheses of \sref{IV.4.7.5}, let $k'$ be an extension of $k$, $X'=X\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$; let $Z$ be an irreducible component of $\Supp(\sh{F})$, and $Z'$ an irreducible component of $\Supp(\sh{F}')$ that dominates $Z$ \sref{IV.4.2.7}.
Then the purely inseparable multiplicities of $Z$ and of $Z'$ with respect to $\sh{F}$ and $\sh{F}'$ respectively are the same.
\end{corollary}

\begin{proof}
It suffices to consider a perfect extension $k''$ of $k'$ and, setting $X''=X\otimes_k k''$, $\sh{F}''=\sh{F}\otimes_k k''$, a maximal point $x''$ of $\Supp(\sh{F}'')$ above the generic point of $Z'$; then by \sref{IV.4.7.8} $\operatorname{long}(\sh{F}''_{x''})$ is equal to the purely inseparable multiplicities of $Z$ and of $Z'$.
\end{proof}

\begin{proposition}[4.7.10]
\label{IV.4.7.10}
Let $k$ be a field, $X$ a $k$-prescheme locally of finite type over $k$, $\sh{F}$ a coherent $\sh{O}_X$-Module, $x$ a point of $X$, $z_i$ the generic points of the irreducible components of $\Supp(\sh{F})$ containing $x$ \sref{IV.4.6.22}.
For $\sh{F}$ to be geometrically reduced at $x$ over $k$, it is necessary and sufficient that $x$ not belong to any immersed associated prime cycle of $\sh{F}$ and that, for every $i$, the geometric length $\lambda_{z_i}(\sh{F})$ of $\sh{F}$ at the point $z_i$ be equal to $1$.
\end{proposition}

\begin{proof}
Let us first show that the conditions of the statement are sufficient.
Indeed, if $k'$ is a finite extension of $k$, $x'$ a point of $X'=X\otimes_k k'$ above $x$, and $\sh{F}'=\sh{F}\otimes_k k'$, $x'$ does not belong to any immersed associated prime cycle of $\sh{F}'$ by \sref{IV.4.2.7}; moreover, by \sref{IV.4.7.8}, the geometric length of $\sh{F}'$ is equal to 1 at every generic point $z'_j$ of an irreducible component of $\Supp(\sh{F}')$ containing $x'$, such a point being above one of the $z_i$ by \sref{IV.2.3.4}; \emph{a fortiori} $\operatorname{long}(\sh{F}'_{z'_j})=1$, which proves that $\sh{F}'$ is reduced at $x'$.

Conversely, it results first from \sref{IV.4.2.7} that if, for \emph{one} finite extension $k'$ of $k$, $\sh{F}'$ is reduced at all points $x'$ of $X'$ above $x$, then $x$ does not belong to any immersed associated prime cycle of $\sh{F}$.
Moreover, taking for $k'$ the field $k_1$ from the statement of \sref{IV.4.7.8} (where $x$ is replaced by one of the $z_i$), the hypothesis that $\sh{F}$ is geometrically reduced at $x$ implies $\lambda_{z_i}(\sh{F})=1$.
\end{proof}

\begin{proposition}[4.7.11]
\label{IV.4.7.11}
Let $k$ be a field, $X$ a prescheme locally of finite type over $k$, $\sh{F}$ a coherent $\sh{O}_X$-Module, $k'$ an extension of $k$, $X'=X\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$, $x$ a point of $X$, $x'$ a point of $X'$ above $x$.
For $\sh{F}$ to be geometrically punctually reduced \emph{(}resp.\ geometrically punctually integral\emph{)} at $x$, it is necessary and sufficient that $\sh{F}'$ be geometrically punctually reduced \emph{(}resp.\ geometrically punctually integral\emph{)} at $x'$.
\end{proposition}

\begin{proof}
The irreducible components of $\Supp(\sh{F}')$ containing $x'$ each dominate an irreducible component of $\Supp(\sh{F})$ containing $x$, and conversely each of these latter is dominated by an irreducible component of $\Supp(\sh{F}')$ containing $x'$ (\sref{IV.4.2.7}, (i)) and \sref{IV.2.3.4}).
The assertion concerning separability results therefore from \sref{IV.4.2.7}, (ii), \sref{IV.4.7.9} and \sref{IV.4.7.10}.
Let $Y$ be the closed subprescheme of $X$ defined by the Ideal $\sh{J}$ annihilator of $\sh{F}$, so that $Y'=Y\otimes_k k'$ is the closed subprescheme of $X'$ defined by the Ideal $\sh{J}'=\sh{J}\otimes_k k'$ annihilator of $\sh{F}'$; to say that $\sh{F}$ (resp.\ $\sh{F}'$) is geometrically punctually reduced (resp.\ integral) at $x$ (resp.\ $x'$) is the same as saying that $\sh{F}$ (resp.\ $\sh{F}'$) is geometrically punctually reduced at $x$ (resp.\ $x'$) and that $Y$ (resp.\ $Y'$) is geometrically punctually integral at $x$ (resp.\ $x'$).
By virtue of the assertion of the statement concerning separability, the hypothesis that $\sh{F}$ is geometrically punctually integral at $x$, or the hypothesis that $\sh{F}'$ is geometrically punctually integral at $x'$, imply both that there exists in $Y$ a neighbourhood of $x$ that is separable, and the conclusion results from \sref{IV.4.6.10}.
\end{proof}

\begin{definition}[4.7.12]
\label{IV.4.7.12}
Let $k$ be a field, $X$ a $k$-prescheme locally of finite type over $k$, $\sh{F}$ a coherent $\sh{O}_X$-Module, $x$ a maximal point of the support of $\sh{F}$.
The \emph{total multiplicity} of $x$ \emph{(}or of the prime cycle $\{x\}$\emph{)} for $\sh{F}$ \emph{(}relative to $k$\emph{)} is the product of the purely inseparable multiplicity of $x$ for $\sh{F}$ by the separable multiplicity of $k(x)$ over $k$.
\end{definition}

\oldpage[IV-2]{80}

It amounts to the same to say that the total multiplicity of $x$ for $\sh{F}$ is the product of the \emph{length} $\operatorname{long}_{\sh{O}_x}(\sh{F}_x)$ by the total multiplicity of $k(x)$ over $k$ \sref{IV.4.7.4}.
With the notation of \sref{IV.4.7.11}, the total multiplicity of $x$ for $\sh{F}$ is equal to the \emph{sum} of the total multiplicities for $\sh{F}'$ of the maximal points $x'_i$ of $\Supp(\sh{F}')$ that are above $x$; moreover, there exists a finite extension $k'$ of $k$ such that the total multiplicity of $x$ for $\sh{F}$ be \emph{equal to} $\sum_i\operatorname{long}(\sh{F}'_{x'_i})$.

\begin{proposition}[4.7.13]
\label{IV.4.7.13}
Let the hypotheses and notation be those of \sref{IV.4.7.10}.
For $\sh{F}$ to be geometrically punctually integral at the point $x$, it suffices that $x$ belong to only one associated prime cycle \emph{(}necessarily non-immersed\emph{)} of $\sh{F}$ and that the total multiplicity for $\sh{F}$ of the generic point $z$ of this cycle be equal to $1$.
\end{proposition}

\begin{proof}
Indeed, if $k'$ is a finite extension of $k$, $x'$ a point of $X'=X\otimes_k k'$ above $x$, and $\sh{F}'=\sh{F}\otimes_k k'$, $x'$ can belong to only one associated prime cycle of $\sh{F}'$, since there is only one maximal point of $\Supp(\sh{F}')$ above $z$ by hypothesis; moreover, $\sh{F}$ is geometrically reduced by \sref{IV.4.7.10}, whence the conclusion.
\end{proof}

\label{subsection:IV.4.8}

\begin{env}[4.8.1]
\label{IV.4.8.1}
Given a prescheme $X$, we will denote, in this section, by $\mathfrak{S}(X)$ the set of subpreschemes of $X$, and as usual by $\mathfrak{P}(X)$ the set of subsets of the underlying space of $X$; for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, we denote by $\Phi(\sh{F})$ the set of quasi-coherent sub-$\sh{O}_X$-Modules of $\sh{F}$.
\end{env}

\begin{env}[4.8.2]
\label{IV.4.8.2}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes, $\sh{F}$, $\sh{G}$ two quasi-coherent $\sh{O}_X$-Modules, $K$, $K'$ two extensions of $k$ such that $K'\subset K$.
We then have, corresponding to the morphism $\Spec(K)\to\Spec(K')$, canonical maps
\begin{align}
  \Phi(\sh{F}\otimes_k K') &\to \Phi(\sh{F}\otimes_k K) \label{IV.4.8.2.1}\tag{4.8.2.1}\\
  \Hom(\sh{F}\otimes_k K',\sh{G}\otimes_k K') &\to \Hom(\sh{F}\otimes_k K,\sh{G}\otimes_k K) \label{IV.4.8.2.2}\tag{4.8.2.2}\\
  \mathfrak{S}(X_{(K')}) &\to \mathfrak{S}(X_{(K)}) \label{IV.4.8.2.3}\tag{4.8.2.3}\\
  \Hom_{K'}(X_{(K')},Y_{(K')}) &\to \Hom_K(X_{(K)},Y_{(K)}) \label{IV.4.8.2.4}\tag{4.8.2.4}\\
  \mathfrak{P}(X_{(K')}) &\to \mathfrak{P}(X_{(K)}). \label{IV.4.8.2.5}\tag{4.8.2.5}
\end{align}

If $p_X:X_{(K)}\to X_{(K')}$ is the canonical projection, the map \hyperref[IV.4.8.2.5]{(4.8.2.5)} is $M\mapsto p_X^{-1}(M)$, and likewise \hyperref[IV.4.8.2.3]{(4.8.2.3)} is $Z\mapsto p_X^{-1}(Z)=Z\otimes_{K'}K$ \sref[I]{I.4.4.1}.
The map \hyperref[IV.4.8.2.1]{(4.8.2.1)} is $\sh{H}\mapsto p_X^*(\sh{H})=\sh{H}\otimes_{\sh{O}_{X_{(K')}}}\sh{O}_{X_{(K)}}$; \hyperref[IV.4.8.2.2]{(4.8.2.2)} is $u\mapsto p_X^*(u)$ and finally \hyperref[IV.4.8.2.4]{(4.8.2.4)} is the map $f\mapsto f_{(K)}$.

If we denote by $\varepsilon_{K,K'}$ any one of these canonical maps, it is clear that if $K''$, $K'$, $K$ are three extensions of $k$ such that $K''\subset K'\subset K$, then we have
\[
  \varepsilon_{K,K''} = \varepsilon_{K,K'}\circ\varepsilon_{K',K''}.
  \label{IV.4.8.2.6}\tag{4.8.2.6}
\]
\end{env}

\begin{proposition}[4.8.3]
\label{IV.4.8.3}
The canonical maps \hyperref[IV.4.8.2.1]{(4.8.2.1)} to \hyperref[IV.4.8.2.5]{(4.8.2.5)} are injective.
\end{proposition}

\begin{proof}
Indeed, we know that $p_X$ is faithfully flat and quasi-compact; the fact that \hyperref[IV.4.8.2.5]{(4.8.2.5)} is injective thus follows from the surjectivity of $p_X$.
The injectivity of \hyperref[IV.4.8.2.1]{(4.8.2.1)} follows from \sref{IV.2.2.2}, that of \hyperref[IV.4.8.2.2]{(4.8.2.2)} from \sref{IV.2.2.7}, that of \hyperref[IV.4.8.2.3]{(4.8.2.3)} from \sref{IV.2.2.15}, and finally that of \hyperref[IV.4.8.2.4]{(4.8.2.4)} from \sref{IV.2.2.16}.
\end{proof}

\begin{definition}[4.8.4]
\label{IV.4.8.4}
The notations being those of \sref{IV.4.8.2}, we say that a quasi-coherent sub-$\sh{O}_{X_{(K)}}$-Module of $\sh{F}\otimes_k K$ (resp.~a homomorphism $\sh{F}\otimes_k K\to\sh{G}\otimes_k K$, resp.~a subprescheme of $X_{(K)}$, resp.~a $K$-morphism $X_{(K)}\to Y_{(K)}$, resp.~a subset of $X_{(K)}$) is \emph{defined over $K'$} if it belongs to the image of the map \hyperref[IV.4.8.2.1]{(4.8.2.1)} (resp.~\hyperref[IV.4.8.2.2]{(4.8.2.2)}, resp.~\hyperref[IV.4.8.2.3]{(4.8.2.3)}, resp.~\hyperref[IV.4.8.2.4]{(4.8.2.4)}, resp.~\hyperref[IV.4.8.2.5]{(4.8.2.5)}).
We then say that $K'$ is a \emph{field of definition} of the object considered.
\end{definition}

It is clear that $K$ itself is a field of definition of an element of any one of the five target sets of the maps \hyperref[IV.4.8.2.1]{(4.8.2.1)} to \hyperref[IV.4.8.2.5]{(4.8.2.5)}.
We note however that for a $K$-prescheme $Z$, to say for example that a quasi-coherent $\sh{O}_Z$-Module $\sh{H}$ is ``defined over a sub-corps $K'$ of $K$'' has no meaning unless $Z$ has been \emph{given} in the form $X_{(K)}$ for a prescheme $X$ over a sub-corps $k$ of $K$, and if $\sh{H}$ is a sub-$\sh{O}_Z$-Module which has been \emph{given} in the form $\sh{F}\otimes_k K$, where $\sh{F}$ is a quasi-coherent $\sh{O}_X$-Module; analogous remarks apply for the other notions.
It follows from \hyperref[IV.4.8.2.6]{(4.8.2.6)} that if an element of one of the target sets of the maps of \sref{IV.4.8.2} is defined on $K'$, it is also defined on every field $K''$ such that $K'\subset K''\subset K$.
Finally, with the notations of \hyperref[IV.4.8.2.6]{(4.8.2.6)}, if $K_1$ is an extension of $K$, for an element of the target set of $\varepsilon_{K,K'}$ to be defined over $K'$, it suffices that its image by $\varepsilon_{K_1,K}$ be defined over $K'$, by virtue of the relation $\varepsilon_{K_1,K'}=\varepsilon_{K_1,K}\circ\varepsilon_{K,K'}$.

\begin{proposition}[4.8.5]
\label{IV.4.8.5}
With the notations of \sref{IV.4.8.2}, let $(X_\alpha)$ be an open covering of $X$.
For an element $\sh{H}\in\Phi(\sh{F}\otimes_k K)$ \emph{(}resp.~$u\in\Hom(\sh{F}\otimes_k K,\sh{G}\otimes_k K)$, resp.~$Z\in\mathfrak{S}(X_{(K)})$, resp.~$M\in\mathfrak{P}(X_{(K)})$\emph{)} to be defined over $K'$, it is necessary and sufficient that for all $\alpha$, $\sh{H}|(X_\alpha)_{(K)}$ \emph{(}resp.~$u|(X_\alpha)_{(K)}$, $Z\cap(X_\alpha)_{(K)}$, $M\cap(X_\alpha)_{(K)}$\emph{)} be defined over $K'$.
\end{proposition}

\begin{proof}
This follows immediately from the injectivity of the maps \hyperref[IV.4.8.2.1]{(4.8.2.1)}, \hyperref[IV.4.8.2.2]{(4.8.2.2)}, \hyperref[IV.4.8.2.3]{(4.8.2.3)}, and \hyperref[IV.4.8.2.5]{(4.8.2.5)}: for example, if for all $\alpha$, there is a quasi-coherent sub-$\sh{O}_{(X_\alpha)_{(K')}}$-Module $\sh{H}'_\alpha$ of $(\sh{F}|X_\alpha)\otimes_k K'$ such that $\sh{H}|(X_\alpha)_{(K)}=\sh{H}'_\alpha\otimes_{K'}K$, for any two indices $\alpha$, $\beta$, we will have
$\sh{H}|(X_\alpha\cap X_\beta)_{(K)}=(\sh{H}'_\alpha|(X_\alpha\cap X_\beta)_{(K')})\otimes_{K'}K=(\sh{H}'_\beta|(X_\alpha\cap X_\beta)_{(K')})\otimes_{K'}K$,
hence necessarily $\sh{H}'_\alpha|(X_\alpha\cap X_\beta)_{(K')}=\sh{H}'_\beta|(X_\alpha\cap X_\beta)_{(K')}$ whatever $\alpha$ and $\beta$, and therefore there is a quasi-coherent sub-$\sh{O}_{X_{(K')}}$-Module $\sh{H}'$ of $\sh{F}\otimes_k K'$ such that $\sh{H}'|(X_\alpha)_{(K')}=\sh{H}'_\alpha$ for all $\alpha$, hence $\sh{H}=\sh{H}'\otimes_{K'}K$.
The reasoning is the same in the other cases.
\end{proof}

\begin{lemma}[4.8.6]
\label{IV.4.8.6}
Let $k$ be a field, $K$ an extension of $k$, $A$ a $k$-algebra, $M$ an $A$-module, $\overline{N}$ a sub-$A_{(K)}$-module of $M_{(K)}$, $K'$ a sub-extension of $K$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] $\overline{N}$ is of the form $N'\otimes_{K'}K$, where $N'$ is a sub-$A_{(K')}$-module of $M_{(K')}$.
  \item[b)] If $X=\Spec(A)$, the quasi-coherent $\sh{O}_{X_{(K)}}$-Module $\widetilde{\overline{N}}$ on $X_{(K)}=\Spec(A_{(K)})$ is defined over $K'$.
  \item[c)] $\overline{N}$ is of the form $N'\otimes_{K'}K$, where $N'$ is a sub-$K'$-vector space of $M_{(K')}$.
\end{enumerate}
\end{lemma}

\begin{proof}
The equivalence of \emph{a)} and \emph{b)} follows trivially from \sref[I]{I.1.6.5}; it is clear that \emph{a)} implies \emph{c)}.
Inversely, if \emph{c)} is satisfied, we know that we can write (up to canonical identification) $N'=\overline{N}\cap M_{(K')}$.
Since $\overline{N}$ and $M_{(K')}$ are by hypothesis $A_{(K')}$-modules, the same is true of $N'$.
\end{proof}

\begin{corollary}[4.8.7]
\label{IV.4.8.7}
Under the hypotheses of \sref{IV.4.8.6}, there exists a smallest sub-corps $K'$ of $K$ containing $k$ such that the equivalent conditions of \sref{IV.4.8.6} are satisfied.
\end{corollary}

\begin{proof}
It suffices to see this for condition \emph{c)}, which follows from Bourbaki, \emph{Alg.}, chap.~II, 3\textsuperscript{rd}~ed., \textsection8, n\textsuperscript{o}~6, prop.~6.
\end{proof}

\begin{lemma}[4.8.8]
\label{IV.4.8.8}
The notations being those of \sref{IV.4.8.2}:
\begin{enumerate}
  \item[\rm(i)] Let $\overline{u}:\sh{F}\otimes_k K\to\sh{G}\otimes_k K$ be an $\sh{O}_{X_{(K)}}$-homomorphism, and let $\sh{H}\subset(\sh{F}\otimes_k K)\oplus(\sh{G}\otimes_k K)$ be its graph.
    For $\overline{u}$ to be defined over $K'$, it is necessary and sufficient that $\sh{H}$ be so.
  \item[\rm(ii)] Let $\overline{Z}$ be a closed subprescheme of $X_{(K)}$, and let $\overline{\sh{J}}$ be the quasi-coherent Ideal of $\sh{O}_{X_{(K)}}$ defining $\overline{Z}$.
    For $\overline{Z}$ to be defined over $K'$, it is necessary and sufficient that $\overline{\sh{J}}$ be so.
  \item[\rm(iii)] Let $\overline{f}$ be a $K$-morphism $X_{(K)}\to Y_{(K)}$ and let $\overline{Z}$ be the subprescheme of $X_{(K)}\times_K Y_{(K)}$, graph of $\overline{f}$ \sref[I]{I.5.3.11}; for $\overline{f}$ to be defined over $K'$, it is necessary and sufficient that $\overline{Z}$ be so.
\end{enumerate}
\end{lemma}

\begin{proof}
Assertion (ii) follows immediately from \sref[I]{I.4.4.5}.
The necessity in assertion (iii) is evident; suppose conversely that $\overline{Z}=Z'_{(K')}$, where $Z'$ is a subprescheme of $(X\times_k Y)_{(K')}$; if $p':Z'\to X_{(K')}$ is the restriction to $Z'$ of the first projection, we know that $p'_{(K)}$ is an isomorphism of preschemes, hence the same is true of $p'$ \sref{IV.2.7.1}[, (viii)]; but this means \sref[I]{I.5.3.11} that $Z'$ is the graph of a $K'$-morphism $f':X_{(K')}\to Y_{(K')}$, and we then have $\overline{f}=f'_{(K)}$ \sref[I]{I.5.3.12}.

Finally, to prove (i), we can, by virtue of \sref{IV.4.8.5}, suppose that $X=\Spec(A)$ is affine, $\sh{F}=\widetilde{M}$, $\sh{G}=\widetilde{N}$, where $M$ and $N$ are $A$-modules, and $\overline{u}=(\overline{v})^\sim$, where $\overline{v}:M_{(K)}\to N_{(K)}$ is an $A_{(K)}$-homomorphism.
Suppose that the graph of $\overline{v}$ is of the form $P'_{(K')}$, where $P'$ is a sub-$A_{(K')}$-module of $M_{(K')}\oplus N_{(K')}$; if $p':P'\to M_{(K')}$ is the restriction to $P'$ of the first projection, we know that $p'\otimes 1_K$ is an isomorphism of $A_{(K')}$-modules, hence, by faithful flatness ($\mathbf{0_I}$, 6.4.1), $p'$ is an isomorphism of $A_{(K')}$-modules, which proves that $P'$ is the graph of an $A_{(K')}$-homomorphism $v':M_{(K')}\to N_{(K')}$ and we obviously have $\overline{v}=v'\otimes 1_K$.
\end{proof}

\begin{proposition}[4.8.9]
\label{IV.4.8.9}
Let $k$ be a field, $K$ an extension of $k$, $X$ a $k$-prescheme, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
Let $\overline{\sh{H}}$ be a quasi-coherent sub-$\sh{O}_{X_{(K)}}$-Module of $\sh{F}\otimes_k K$.
Then there exists a smallest field of definition of $\overline{\sh{H}}$.
\end{proposition}

\begin{proof}
When $X$ is affine, the assertion is none other than \sref{IV.4.8.7}.
In the general case, consider an open covering $(X_\alpha)$ of $X$ by affine opens; by virtue of what precedes, there exists for all $\alpha$ a smallest sub-corps $K'_\alpha$ of $K$ containing $k$ and such that $\overline{\sh{H}}|(X_\alpha)_{(K)}$ is defined over $K'_\alpha$.
By virtue of \sref{IV.4.8.5}, the sub-corps of $K$ generated by the $K'_\alpha$ is the smallest field of definition of $\overline{\sh{H}}$.
\end{proof}

\begin{corollary}[4.8.10]
\label{IV.4.8.10}
The notations being those of \sref{IV.4.8.9}, let $\sh{G}$ be a second quasi-coherent $\sh{O}_X$-Module, and let $\overline{u}:\sh{F}\otimes_k K\to\sh{G}\otimes_k K$ be an $\sh{O}_{X_{(K)}}$-homomorphism; then there exists a smallest field of definition of $\overline{u}$.
\end{corollary}

\begin{proof}
Indeed, it follows from \sref{IV.4.8.8} that such a field is also a smallest field of definition of the graph of $\overline{u}$, and the existence of the latter follows from \sref{IV.4.8.9}.
\end{proof}

Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$, $\overline{Z}$ a closed subprescheme of $X_{(K)}$; then there exists a smallest field of definition of $\overline{Z}$.

\begin{corollary}[4.8.12]
\label{IV.4.8.12}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes, $Y$ being supposed separated, $K$ an extension of $k$, $\overline{f}:X_{(K)}\to Y_{(K)}$ a $K$-morphism.
There then exists a smallest field of definition of $\overline{f}$.
\end{corollary}

\begin{proof}
Indeed, the existence of such a field, by virtue of \sref{IV.4.8.8}, amounts to that of a smallest field of definition of the graph $\overline{Z}$ of $\overline{f}$; but since $\overline{Z}$ is a closed subprescheme of $(X\times_k Y)_{(K)}$ \sref[I]{I.5.4.3}, the existence of a smallest field of definition of $\overline{Z}$ follows from \sref{IV.4.8.11}.
\end{proof}

\begin{proposition}[4.8.13]
\label{IV.4.8.13}
Under the hypotheses of \sref{IV.4.8.11} \emph{(}resp.~\sref{IV.4.8.9}, \sref{IV.4.8.10}, \sref{IV.4.8.12}\emph{)} suppose further that $X$ is of finite type over $k$ \emph{(}resp.~that $X$ is of finite type over $k$ and $\sh{F}$ coherent, resp.~that $X$ is of finite type over $k$ and $\sh{F}$ and $\sh{G}$ coherent, resp.~that $X$ and $Y$ are of finite type over $k$\emph{)}.
Then the smallest field of definition of $\overline{Z}$ \emph{(}resp.~of $\overline{\sh{H}}$, of $\overline{u}$, of $\overline{f}$\emph{)} is a finite type extension of $k$.
\end{proposition}

\begin{proof}
We can restrict ourselves to treating the case of \sref{IV.4.8.9}, and suppose $X$ affine since $X$ is a finite union of affine opens of finite type over $k$.
With the notations of \sref{IV.4.8.6}, it suffices to show that if $A$ is a $k$-algebra of finite type and $M$ an $A$-module of finite type, there exists a sub-extension $K'$ of $K$ which is of finite type over $k$ (every sub-extension of a finite type extension being of finite type) and which satisfies conditions \emph{a)}, \emph{b)}, or \emph{c)} of \sref{IV.4.8.6}.
Now, let $(m_i)_{1\leq i\leq n}$ be a system of generators of the $A$-module $M$; since $A$ is Noetherian, the same is true of $A_{(K)}$, hence $\overline{N}$ is an $A_{(K)}$-module of finite type, admitting a finite system of generators $n_j$ ($1\leq j\leq r$) such that $n_j=\sum_i(m_i\otimes 1)b_{ij}$, where $b_{ij}\in A_{(K)}$; each $b_{ij}$ can be written $\sum_h a_{ijh}\otimes c_{ijh}$, where $a_{ijh}\in A$ and $c_{ijh}\in K$; it is clear that the extension $K'$ of $k$ generated by the $c_{ijh}$ answers the question.
\end{proof}

\begin{proposition}[4.8.14]
\label{IV.4.8.14}
Let $k$ be a field, $X$ a $k$-prescheme of finite type, $K$ an extension of $k$ containing a perfect extension of $k$.
Then the smallest field of definition of the closed subprescheme $(X_{(K)})_{\mathrm{red}}$ of $X_{(K)}$ is a finite radicial extension of $k$.
\end{proposition}

\begin{proof}
If $p$ is the characteristic exponent of $k$, the hypothesis implies that $k'=k^{p^{-\infty}}$ is contained in $K$; we know that $(X_{(k')})_{\mathrm{red}}$ is geometrically reduced over $k'$ \sref{IV.4.6.1}, hence $(X_{(k')})_{\mathrm{red}}\otimes_{k'}K$ is reduced, and is therefore equal to $(X_{(K)})_{\mathrm{red}}$ \sref[I]{I.5.1.8}; in other words, $k'$ is a field of definition of $(X_{(K)})_{\mathrm{red}}$.
Since $k'$ is an algebraic extension of $k$, the conclusion follows from \sref{IV.4.8.13}.
\end{proof}

\textbf{Remarks.}

\subsection{Fields of definition of a part of a prescheme}
\label{subsection:IV.4.9}

\begin{proposition}[4.9.1]
\label{IV.4.9.1}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$, $\overline{T}$ a closed part of $X_{(K)}$, $K'$ a sub-extension of $K$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] $\overline{T}$ is defined over $K'$.
  \item[b)] The open part $\overline{U}=X_{(K)}-\overline{T}$ of $X_{(K)}$ is defined over $K'$.
  \item[b')] The subprescheme of $X_{(K)}$ induced on the open $\overline{U}$ is defined over $K'$.
  \item[c)] There exists a closed subprescheme $\overline{Z}$ of $X_{(K)}$ having $\overline{T}$ as underlying space, and defined over $K'$.
\end{enumerate}
\end{proposition}

\begin{proof}
If $g:X_{(K)}\to X_{(K')}$ is the canonical projection, to say that $\overline{T}$ is defined over $K'$ means that there exists a closed part $T'$ of $X_{(K')}$ such that $\overline{T}=g^{-1}(T')$; since $g^{-1}(X_{(K')}-T')=X_{(K)}-g^{-1}(T')$ for all closed parts $T'$ of $X_{(K')}$, \emph{a)}, \emph{b)}, and \emph{b')} are equivalent.
If $\overline{Z}$, having $\overline{T}$ as underlying space, is defined over $K'$, we have $\overline{Z}=Z'\otimes_{K'}K$, where $Z'$ is a closed subprescheme of $X_{(K')}$, and if $T'$ is the underlying space of $Z'$, we have $\overline{T}=g^{-1}(T')$ \sref[I]{I.4.4.1}, hence $\overline{T}$ is defined over $K'$.
Inversely, to see that \emph{a)} implies \emph{c)}, it suffices to consider a closed subprescheme $Z'$ of $X_{(K')}$ having $T'$ as underlying space \sref[I]{I.5.2.1} and to take $\overline{Z}=Z'\otimes_{K'}K$.
\end{proof}

\begin{corollary}[4.9.2]
\label{IV.4.9.2}
With the notations of \sref{IV.4.9.1}, suppose that $K'$ is a field of definition of $\overline{T}$; every extension $K''\subset K'$ of $k$, such that $K'$ is a radicial extension of $K''$, is also a field of definition of $\overline{T}$.
\end{corollary}

\begin{proof}
It suffices to observe that the canonical projection $g:X_{(K')}\to X_{(K'')}$ is a morphism that is integer, surjective, and radicial, hence a homeomorphism \sref{IV.2.4.5}.
\end{proof}

\begin{env}[4.9.3]
\label{IV.4.9.3}
\textbf{Remark.}
It follows from \sref{IV.4.9.2} that a closed part $\overline{T}$ of $X_{(K)}$ does not necessarily admit a \emph{smallest} field of definition.

However:

\begin{proposition}[4.9.4]
\label{IV.4.9.4}
With the notations of \sref{IV.4.9.1}, suppose that $K$ is a separable extension of $K'$; for $K'$ to be a field of definition of $\overline{T}$, it is necessary and sufficient that $K'$ be a field of definition of the reduced subprescheme of $X_{(K)}$ having $\overline{T}$ as underlying space.
\end{proposition}

\begin{proof}
The condition is evidently sufficient by virtue of \sref{IV.4.9.1}; to see that it is necessary, note that if $\overline{T}=g^{-1}(T')$, where $T'$ is a closed part of $X_{(K')}$, and if $Z'$ is the reduced subprescheme of $X_{(K')}$ having $T'$ as underlying space, then $\overline{Z}=Z'\otimes_{K'}K=\Spec(K)\times_{K'}Z'$ is reduced by virtue of the criterion \sref{IV.4.6.1}[, \emph{e)}] and the hypothesis on $K$, and has $\overline{T}$ as underlying space.
\end{proof}

\begin{corollary}[4.9.5]
\label{IV.4.9.5}
Suppose one of the following hypotheses satisfied:
\begin{enumerate}
  \item[a)] $k$ is of characteristic $0$.
  \item[b)] $K$ is an algebraic extension of $k$ and $X$ is of finite type over $k$.
\end{enumerate}
Then every closed part $\overline{T}$ of $X_{(K)}$ has a smallest field of definition, which is a finite separable extension of $k$ \emph{(}and finite over $k$ in hypothesis \emph{b)}\emph{)}.
\end{corollary}

\begin{proof}
In hypothesis \emph{a)}, $K$ is separable on each of its sub-corps, and it follows from \sref{IV.4.9.4} that the fields of definition of $\overline{T}$ are the same as those of the reduced subprescheme of $X_{(K)}$ having $\overline{T}$ as underlying space.
The corollary then follows from \sref{IV.4.8.11}.
In hypothesis \emph{b)}, we know that $K$ is a radicial extension of a separable extension $K_1$ of $k$.
Then, for every sub-extension $K'$ of $K$, $K'$ is radicial on $K'\cap K_1$, hence, by virtue of \sref{IV.4.9.2}, $K'$ is a field of definition of $\overline{T}$ if and only if $K'\cap K_1$ is.
This reduces us to the case where $K'=K_1$, hence the case where $K'$ is a separable algebraic extension of $k$, and one finishes the reasoning as in case \emph{a)}.
\end{proof}

\begin{proposition}[4.9.6]
\label{IV.4.9.6}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$, $\overline{U}$ a part that is both open and closed of $X_{(K)}$, $\overline{V}$ its complement.
Let $K'$ be a sub-extension of $K$; then the following conditions are equivalent:
\begin{enumerate}
  \item[a)] $K'$ is a field of definition of the part $\overline{U}$ of $X_{(K)}$.
  \item[a')] $K'$ is a field of definition of the closed subprescheme of $X_{(K)}$ induced on the open $\overline{U}$.
  \item[b)] $K'$ is a field of definition of the part $\overline{V}$ of $X_{(K)}$.
  \item[b')] $K'$ is a field of definition of the closed subprescheme of $X_{(K)}$ induced on the open $\overline{V}$.
\end{enumerate}
\end{proposition}

\begin{proof}
It is clear that \emph{a')} implies \emph{a)}, and \emph{a)} implies \emph{b')} by virtue of \sref{IV.4.9.1}.
Since $\overline{U}$ and $\overline{V}$ play symmetric roles, \emph{b')} implies \emph{b)} and \emph{b)} implies \emph{a')}, which completes the proof.
\end{proof}

\begin{corollary}[4.9.7]
\label{IV.4.9.7}
Under the hypotheses of \sref{IV.4.9.6}, there exists a smallest field of definition $K'$ for the part $\overline{U}$ of $X_{(K)}$, and $K'$ is also the smallest field of definition of the part $\overline{V}$ of $X_{(K)}$ and of the closed subpreschemes of $X_{(K)}$ induced on the opens $\overline{U}$ and $\overline{V}$.
If in addition $X$ is of finite type over $k$, $K'$ is a finite separable extension of $k$.
\end{corollary}

\begin{proof}
Taking into account \sref{IV.4.9.6} and \sref{IV.4.8.11}, we only have to demonstrate the last assertion.
By virtue of \sref{IV.4.8.15}[, (ii)], one can restrict oneself to the case where $K$ is algebraically closed.
Let $\overline{k}$ be the algebraic closure of $k$ contained in $K$, and denote by $W_i$ the connected components of $X_{(\overline{k})}$ ($1\leq i\leq m$); if $p:X_{(K)}\to X_{(\overline{k})}$ is the canonical projection, we know that the $p^{-1}(W_i)$ are the connected components of $X_{(K)}$ \sref{IV.4.4.6}, hence $\overline{U}$ is a union of a certain number of these components, and is therefore defined over $\overline{k}$; the conclusion follows from \sref{IV.4.9.5}[, \emph{b)}].
\end{proof}
