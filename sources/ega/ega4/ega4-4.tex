\section{Change of base field for algebraic preschemes}
\label{section:IV.4}

\subsection{If \texorpdfstring{$X$}{X} is an algebraic prescheme over a field \texorpdfstring{$k$}{k}, and \texorpdfstring{$K$}{K} is an extension of \texorpdfstring{$k$}{k}, the prescheme \texorpdfstring{$X\otimes_k K$}{X x\_k K} and its underlying topological space}
\label{subsection:IV.4.1}

\oldpage[IV-2]{52}

\begin{env}[4.1.1]
\label{IV.4.1.1}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$.
We write $X_{(\mathrm{K})}=X\otimes_k K$ to denote the product $X\times_k\Spec(K)$; recall that $\Spec(K)\to\Spec(k)$ is faithfully flat, so $X_{(\mathrm{K})}$ is faithfully flat over $X$ \sref[I]{I.3.5.2}.
If $\sh{F}$ is an $\sh{O}_X$-Module, we write $\sh{F}_{(\mathrm{K})}=\sh{F}\otimes_k K$ for the $\sh{O}_{X_{(\mathrm{K})}}$-Module inverse image of $\sh{F}$ under the canonical projection $X_{(\mathrm{K})}\to X$.
Let $x$ be a point of $X$ and $x'$ a point of $X_{(\mathrm{K})}$ above $x$; we have $\sh{O}_{X_{(\mathrm{K})},x'}=\sh{O}_{X,x}\otimes_k K'$, where $K'$ is the local ring of $\Spec(\kappa(x)\otimes_k K)$ at the point that is the image of $x'$ under the canonical isomorphism $p^{-1}(x)\xrightarrow{\sim}\Spec(\kappa(x)\otimes_k K)$ (where $p:X_{(\mathrm{K})}\to X$ is the projection) \sref[I]{I.3.4.9}.
\end{env}

\begin{env}[4.1.2]
\label{IV.4.1.2}
Let $k$ be a field, $K$ and $L$ two extensions of $k$, $K'$ a subextension of $K$.
We write $\mathfrak{t}=\operatorname{tr.deg}_k$.
Suppose that there exists a $k$-monomorphism $u:K'\to L$; we consider $L$ as a $K'$-algebra via $u$, and $K$ as a $K'$-algebra via the inclusion map.
We have the commutative diagram
\[
\tag{4.1.2.1}
k\to K'\to K, \quad k\to K'\xrightarrow{u} L
\]
which gives us a commutative diagram
\[
\tag{4.1.2.2}
K\otimes_k L \xleftarrow{\sim} K\otimes_{K'}L \xleftarrow{} L.
\]

Recall the following relation (cf.\ Bourbaki, \emph{Alg.}, chap.~V, \S~6, n\textsuperscript{o}~6, th.~3 and Bourbaki, \emph{Alg.~comm.}, chap.~VIII, \S~1, n\textsuperscript{o}~1, cor.~2 and~3 de la prop.~1):
\[
\tag{4.1.2.3}
\operatorname{deg.tr}_k(K\otimes_k L) = \mathfrak{t}\,K + \mathfrak{t}\,L - \mathfrak{t}\,K'
\]
(the degree of transcendence of a ring without zero divisors over a field $k$ being defined in Bourbaki, \emph{Alg.~comm.}, chap.~VIII, \S~1, n\textsuperscript{o}~1).
When $K$ and $L$ are extensions of $k$ that are subextensions of the same extension $\Omega$, we take $K'=K\cap L$, $u$ the inclusion map, and ($4.1.2.3$) becomes
\[
\tag{4.1.2.4}
\operatorname{deg.tr}_k(K\otimes_k L) = \mathfrak{t}\,K + \mathfrak{t}\,L - \mathfrak{t}(K\cap L).
\]

Recall also the following result: under the hypotheses of the first relation ($4.2.1.1$), let us denote further by $E$ the field of fractions of $K\otimes_{K'} L = K\otimes_k L$, and by $I$ the set of minimal prime ideals of $K\otimes_k L$; then $\operatorname{Card}(I)\leq[\mathrm{K}':\mathrm{k}]_s$ (the degree of the largest separable algebraic extension of $k$ contained in $K'$) (Bourbaki, \emph{Alg.~comm.}, chap.~V, \S~2, n\textsuperscript{o}~1, cor.~1 de la prop.~1).
We also have the first relation
\[
\tag{4.2.1.1}
\operatorname{deg.tr}_k E = \mathfrak{t}\,K + \mathfrak{t}\,L - \mathfrak{t}\,K';
\]
(Bourbaki, \emph{Alg.}, chap.~IV, \S~1, n\textsuperscript{o}~1, cor.~3 de la prop.~1).
Moreover, if in addition $K'$ and $L$ are algebraic extensions of $k$, we have
\[
\operatorname{Card}(I) = [\mathrm{K}' : \mathrm{k}]_s.
\]

\oldpage[IV-2]{53}
Suppose $K$ and $L$ are extensions of $k$ such that $K\cap L=k$ (inside some common overfield), $K'=k$; let $E$ be the field of fractions of $K\otimes_k L$, $I$ the set of minimal prime ideals, so $\operatorname{Card}(I)=1$ and $\operatorname{deg.tr}_k E = \mathfrak{t}\,K + \mathfrak{t}\,L$.
In general, if we suppose only that $K'$ is an algebraic extension of $k$, we have $\operatorname{Card}(I)\leq [K':k]_s$, and equality holds when $K'$ is a finite extension of $k$.

Now suppose further that $K$ and $L$ are \emph{subextensions of the same extension} $\Omega$ of $k$, and let $K'=K\cap L$; then $K'$ is a subextension of $K$ and of $L$, and there is a canonical $k$-monomorphism $u:K'\to L$ (the inclusion map).
The field of fractions of $K\otimes_k L$ is then equal to the composite field $KL$ inside $\Omega$.
\end{env}

\begin{env}[4.1.3]
\label{IV.4.1.3}
Keep the notation of \sref{IV.4.1.2}.
We have the canonical $k$-homomorphisms
\[
k\to K'\to K, \quad k\to L
\]
and, setting $K'\otimes_k L=K'\otimes_k L$, the field of fractions of $K'\otimes_k L$ is $L(\mathfrak{t})$; we have the commutative diagram of canonical homomorphisms
\[
\xymatrix{
K \ar[r] & K\otimes_k L \ar[r] & K\otimes_{K'} L(\mathfrak{t})\\
K' \ar[u] \ar[r] & K'\otimes_k L \ar[u] \ar[r] & L(\mathfrak{t}) \ar[u]\\
& k \ar[ul] \ar[r] & L \ar[u]
}
\tag{4.2.1.3}
\]

Since $K$ is faithfully flat over $K'$, $K\otimes_k L = K\otimes_{K'}(K'\otimes_k L)$ is faithfully flat over $K'\otimes_k L$, so $K'\otimes_k L$ is Noetherian $(\mathbf{0}_I,\,6.5.2)$; furthermore $K'\otimes_k L$ identifies with a subring of $K\otimes_k L$; the trace on $K'\otimes_k L$ of a prime ideal $\mathfrak{p}$ associated to $K\otimes_k L$ is the ideal $\mathfrak{o}$ of $K'\otimes_k L$ not being a zero divisor in $K\otimes_k L$ $(\mathbf{0}_I,\,6.3.4$ and Bourbaki, \emph{Alg.~comm.}, chap.~IV, \S~1, n\textsuperscript{o}~1, cor.~3 de la prop.~2).
Moreover, $K$ being algebraic over $K'$, $K\otimes_k L$ is an integral algebra over $K'\otimes_k L$, and the prime ideals of $K\otimes_k L$ inducing $\mathfrak{o}$ on $K'\otimes_k L$ are necessarily without mutual inclusion relation (Bourbaki, \emph{Alg.~comm.}, chap.~V, \S~2, n\textsuperscript{o}~1, cor.~1 de la prop.~1); this proves the first assertion (Bourbaki, \emph{Alg.~comm.}, chap.~IV, \S~1, n\textsuperscript{o}~3, cor.~1 de la prop.~7).
Furthermore, the residue field $E$ of $\mathfrak{p}$ is algebraic over the residue field of the ideal $(\mathfrak{o})$ of $K'\otimes_k L$, that is to say $L(\mathfrak{t})$; hence $\operatorname{deg.tr}_k E = \operatorname{deg.tr}_k L(\mathfrak{t}) = \operatorname{Card}(I) = \operatorname{deg.tr}_k K$, which by switching the roles of $K$ and $L$ gives the second relation $(4.2.1.1)$, whence $(4.2.1.2)$.
\end{env}

\begin{corollary}[4.2.2]
\label{IV.4.2.2}
Under the hypotheses of \sref{IV.3.3.6}, if the preschemes $\mathrm{T}_{x,y}$ are locally Noetherian, they have no associated immersed prime cycle.
\end{corollary}

\begin{corollary}[4.2.3]
\label{IV.4.2.3}
Under the hypotheses of \sref{IV.3.3.6} (resp.\ \sref{IV.3.3.7}), if the $\mathrm{T}_{x,y}$ are locally Noetherian and if $\sh{F}$ and the $\sh{G}_s$ (for $s\in S$) (resp.\ $\sh{F}$ and $\sh{G}$) are without associated immersed prime cycle, then so is $\sh{F}\otimes_\sh{O}\sh{G}$.
\end{corollary}

This follows from $(4.2.2)$, $(3.3.2)$ and the proof of \sref{IV.3.3.6}.
In particular, since every prescheme over a field $k$ is \emph{flat} over $k$, we have thus proved assertion~(i) of the following:

\begin{proposition}[4.2.4]
\label{IV.4.2.4}
Let $k$ be a field, $X$ and $Y$ two locally Noetherian $k$-preschemes such that $X\times_k Y$ is locally Noetherian.
Suppose further that $X$ and $Y$ are integral.
Then:
\begin{enumerate}[label=(\roman*)]
  \item $X\times_k Y$ has no associated immersed prime cycle; each of the irreducible components of $X\times_k Y$ dominates $X$ and $Y$, and the set of these components is in bijective correspondence with the set of irreducible components of $\Spec(\mathrm{R}(X)\otimes_k\mathrm{R}(Y))$ (in other words, the set of minimal prime ideals of $\mathrm{R}(X)\otimes_k\mathrm{R}(Y))$, where $\mathrm{R}(X)$ and $\mathrm{R}(Y)$ are the fields of rational functions of $X$ and $Y$ respectively.
  \item If a maximal point $z$ of $X\times_k Y$ is identified with a minimal prime ideal $\mathfrak{p}$ of $\mathrm{R}(X)\otimes_k\mathrm{R}(Y)$, the local ring $\sh{O}_{X\times_k Y,z}$ is isomorphic to the ring of fractions $(\mathrm{R}(X)\otimes_k\mathrm{R}(Y))_\mathfrak{p}$.
In particular, if $\mathrm{R}(X)$ or $\mathrm{R}(Y)$ is separable over $k$, $X\times_k Y$ is reduced.
\end{enumerate}
\end{proposition}

\oldpage[IV-2]{56}
\begin{enumerate}
  \item[\rm(iii)] If in addition $X$ and $Y$ are locally of finite type over $k$, every irreducible component of $X\times_k Y$ is of dimension $\dim(X)+\dim(Y)$.
\end{enumerate}

Assertion~(iii) follows from $(4.2.1.2)$, taking into account~(i) and~(ii).
To prove~(ii), we can evidently restrict to the case where $X=\Spec(A)$, $Y=\Spec(B)$ are affine, $A$ and $B$ being integral rings with fields of fractions $K=\mathrm{R}(X)$ and $L=\mathrm{R}(Y)$ respectively; assertion~(i) shows that every minimal prime ideal $\mathfrak{q}$ of $A\otimes_k B$ is the trace on $A\otimes_k B$ of a minimal prime ideal $\mathfrak{p}$ of $K\otimes_k L$; since $K\otimes_k L$ is a ring of fractions of $A\otimes_k B$, the isomorphism of the rings $(A\otimes_k B)_\mathfrak{q}$ and $(K\otimes_k L)_\mathfrak{p}$ follows from $(\mathbf{0}_I,\,1.2.5)$.

Finally, if for example $\mathrm{R}(Y)$ is separable over $k$, we know that the ring $\mathrm{R}(X)\otimes_k\mathrm{R}(Y)$ is reduced (Bourbaki, \emph{Alg.}, chap.~VIII, \S~7, n\textsuperscript{o}~3, th.~1); it is therefore also reduced for the local rings of the maximal points of $X\times_k Y$.
We conclude that $X\times_k Y$ is reduced \sref{IV.3.2.1}.

\begin{proposition}[4.2.5]
\label{IV.4.2.5}
Let $k$ be a field, $X$ and $Y$ two $k$-preschemes, locally Noetherian, $\sh{F}$ (resp.\ $\sh{G}$) an $\sh{O}_X$-Module (resp.\ an $\sh{O}_Y$-Module) quasi-coherent.
Let $(Z'_\lambda)$ (resp.\ $(Z''_\mu)$) be the family of prime cycles associated to $\sh{F}$ (resp.\ $\sh{G}$), and denote further by $Z'_\lambda$ (resp.\ $Z''_\mu$) the reduced sub-prescheme of $X$ (resp.\ $Y$) having $Z'_\lambda$ (resp.\ $Z''_\mu$) as underlying space.
Then, if $Z'_\lambda\times_k Z''_\mu$ is locally Noetherian, the irreducible components $Z_{\lambda\mu\nu}$ of $Z'_\lambda\times_k Z''_\mu$ dominate $Z'_\lambda$ and $Z''_\mu$, and $(Z_{\lambda\mu\nu})$ is the family of distinct prime cycles associated to $\sh{F}\otimes_k\sh{G}$.
\end{proposition}

\begin{proof}
It suffices to apply $(4.2.4)$ to the product $\Spec(\kappa(x))\times_k\Spec(\kappa(y))$.
In particular:
\end{proof}

\begin{corollary}[4.2.6]
\label{IV.4.2.6}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes, locally Noetherian, such that $X\times_k Y$ is locally Noetherian.
Let $(Z'_\lambda)$ (resp.\ $(Z''_\mu)$) be the family of the reduced sub-preschemes of $X$ (resp.\ $Y$) having for underlying spaces the irreducible components of $X$ (resp.\ $Y$).
Then the irreducible components $Z_{\lambda\mu\nu}$ of $Z'_\lambda\times_k Z''_\mu$ dominate $Z'_\lambda$ and $Z''_\mu$, and $(Z_{\lambda\mu\nu})$ is the family of irreducible components of $X\times_k Y$.
\end{corollary}

\begin{proof}
Indeed, we can restrict to the case where $X$ and $Y$ are reduced \sref[I]{I.5.1.8}; the irreducible components of $X$ (resp.\ $Y$) are then the prime cycles associated to $\sh{O}_X$ (resp.\ $\sh{O}_Y$) \sref{IV.3.2.1}.
We apply $(4.2.5)$ to $\sh{F}=\sh{O}_X$ and $\sh{G}=\sh{O}_Y$, noting that by definition $\sh{O}_X\otimes_k\sh{O}_Y=\sh{O}_{X\times_k Y}$; the corollary follows from the fact that $\sh{O}_X\otimes_k\sh{O}_Y$ has no associated immersed prime cycles, since this is so for $\sh{O}_X$ and $\sh{O}_Y$ by hypothesis \sref{IV.4.2.3}.

Applying the preceding results to the case where $Y$ is the spectrum of an extension $K$ of $k$:
\end{proof}

\begin{proposition}[4.2.7]
\label{IV.4.2.7}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$ such that $X\otimes_k K$ is locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent, $x'$ a point of $X\otimes_k K$, $x$ its image in $X$.
\begin{enumerate}[label=(\roman*)]
  \item Let $(Z_\lambda)$ be the family of reduced sub-preschemes of $X$ having for underlying spaces the prime cycles associated to $\sh{F}$; then the irreducible components $Z_{\lambda\mu}$ of $Z_\lambda\otimes_k K$ are the prime cycles associated to $\sh{F}\otimes_k K$, and $Z_{\lambda\mu}$ dominates $Z_\lambda$; furthermore, for $Z_{\lambda\mu}$ to be immersed, it is necessary and sufficient that $Z_\lambda$ be so.
  \item For $x$ to belong to a prime cycle associated to an immersed $\sh{F}$, it is necessary and sufficient that $x'$ belong to a prime cycle associated to an immersed $\sh{F}\otimes_k K$; for $\sh{F}$ to be without associated immersed prime cycle, it is necessary and sufficient that $\sh{F}\otimes_k K$ be so.
\end{enumerate}
\end{proposition}

\oldpage[IV-2]{57}
\begin{enumerate}
  \item[\rm(iii)] The indices $\lambda$ such that $x\in Z_\lambda$ are the same as those for which there exists an index $\mu$ such that $x'\in Z_{\lambda\mu}$.
In particular, if $x'$ belongs to a single prime cycle associated to $\sh{F}\otimes_k K$, $x$ belongs to a single prime cycle associated to $\sh{F}$.
  \item[\rm(iv)] If $X$ is locally of finite type over $k$, we have $\dim(Z_{\lambda\mu})=\dim(Z_\lambda)$.
\end{enumerate}

We note that the hypothesis implies that $X$ itself is locally Noetherian \sref{IV.2.2.13} and \sref{IV.2.2.14}; assertion~(i) follows from $(4.2.5)$ and the proof of $(4.2.3)$, for $\sh{G}=\sh{O}_Y$, with $Y=\Spec(K)$; (ii) and~(iii) follow from~(i) and \sref{IV.2.3.5}; finally, (iv) is a particular case of $(4.2.4)$,~(iii)).

\begin{corollary}[4.2.8]
\label{IV.4.2.8}
If $X$ is locally of finite type over $k$, the set of dimensions of the prime cycles associated to $\sh{F}$ and $\sh{F}\otimes_k K$; the set of dimensions of the irreducible components is the same for $X$ and $X\otimes_k K$.
\end{corollary}

\begin{proposition}[4.2.9]
\label{IV.4.2.9}
Suppose the hypotheses of \sref{IV.4.2.5} are satisfied, and suppose in addition that $\sh{F}$ and $\sh{G}$ are coherent.
Let $(\sh{F}_\lambda)_{\lambda\in\mathrm{L}}$ and $(\sh{G}_\mu)_{\mu\in\mathrm{M}}$ be irredundant decompositions of $\sh{F}$ and $\sh{G}$ respectively; for every pair $(\lambda,\mu)\in\mathrm{L}\times\mathrm{M}$, let $(\sh{H}_{\lambda\mu\nu})_{\nu\in\mathrm{S}(\lambda,\mu)}$ be an irredundant decomposition of $\sh{F}_\lambda\otimes_k\sh{G}_\mu$, where $\mathrm{S}(\lambda,\mu)=\operatorname{Ass}(\sh{F}_\lambda\otimes_k\sh{G}_\mu)$ \sref{IV.3.2.5}.
Then $(\sh{H}_{\lambda\mu\nu})$ (for all triples $(\lambda,\mu,\nu)$) is an irredundant decomposition of $\sh{F}\otimes_k\sh{G}$; it is reduced if $(\sh{F}_\lambda)$ and $(\sh{G}_\mu)$ are.
\end{proposition}

\begin{proof}
Note that $\sh{F}\otimes_k\sh{G}$ and each of the $\sh{F}_\lambda\otimes_k\sh{G}_\mu$ are coherent, and each of the $\sh{F}_\lambda\otimes_k\sh{G}_\mu$ identifies with a quotient of $\sh{F}\otimes_k\sh{G}$; each $\sh{H}_{\lambda\mu\nu}$ identifies by definition with a coherent quotient of $\sh{F}_\lambda\otimes_k\sh{G}_\mu$, hence also with a coherent quotient of $\sh{F}\otimes_k\sh{G}$.
The family of supports of the $\sh{H}_{\lambda\mu\nu}$ is locally finite, since $\operatorname{Supp}(\sh{F}_\lambda\otimes_k\sh{G}_\mu)$ is contained in $\operatorname{Supp}(\sh{F}_\lambda)\times_k\operatorname{Supp}(\sh{G}_\mu)$ (where $\operatorname{Supp}(\sh{F}_\lambda)$ and $\operatorname{Supp}(\sh{G}_\mu)$ denote the corresponding reduced closed sub-preschemes), and for given $\lambda$ and $\mu$, the family of supports of the $\sh{H}_{\lambda\mu\nu}$ is locally finite by hypothesis; our assertion then follows from the fact that the families $(\operatorname{Supp}(\sh{F}_\lambda))$ and $(\operatorname{Supp}(\sh{G}_\mu))$ are locally finite.
To show that $(\sh{H}_{\lambda\mu\nu})$ is an irredundant decomposition of $\sh{F}\otimes_k\sh{G}$, it suffices therefore to prove that the canonical homomorphism $\sh{F}\otimes_k\sh{G}\to\bigoplus_{\lambda,\mu,\nu}\sh{H}_{\lambda\mu\nu}$ is injective; or, it is the composite of the homomorphisms $\sh{F}\otimes_k\sh{G}\to\bigoplus_{\lambda,\mu}(\sh{F}_\lambda\otimes_k\sh{G}_\mu)\to\bigoplus_{\lambda,\mu,\nu}\sh{H}_{\lambda\mu\nu}$; the second of these homomorphisms is injective by definition, and it is the same for the first, which is nothing other than the tensor product of the canonical injective homomorphisms $\sh{F}\to\bigoplus_\lambda\sh{F}_\lambda$, $\sh{G}\to\bigoplus_\mu\sh{G}_\mu$ (remembering that $\sh{F}$ and $\sh{G}$ are flat over $k$).
Finally, if $(\sh{F}_\lambda)$ and $(\sh{G}_\mu)$ are reduced, we can suppose that $\mathrm{L}=\operatorname{Ass}(\sh{F})$ and $\mathrm{M}=\operatorname{Ass}(\sh{G})$; the fact that the $\sh{H}_{\lambda\mu\nu}$ then form a reduced decomposition follows from the fact that the $\operatorname{Ass}(\sh{F}_\lambda\otimes_k\sh{G}_\mu)$ form a partition of $\operatorname{Ass}(\sh{F}\otimes\sh{G})$ by virtue of $(4.2.2)$ (cf.\ $(3.2.5)$).
\end{proof}

\begin{corollary}[4.2.10]
\label{IV.4.2.10}
Under the hypotheses of \sref{IV.4.2.7}, suppose in addition that $\sh{F}$ is coherent, and let $(\sh{F}_\lambda)_{\lambda\in\mathrm{L}}$ be an irredundant decomposition of $\sh{F}$; for every $\lambda\in\mathrm{L}$, let $(\sh{F}_{\lambda\mu})_{\mu\in\operatorname{Ass}(\sh{F}_\lambda\otimes_k K)}$ be an irredundant decomposition of $\sh{F}_\lambda\otimes_k K$.
Then $(\sh{F}_{\lambda\mu})$ is an irredundant decomposition of $\sh{F}\otimes_k K$; it is reduced if $(\sh{F}_\lambda)$ is.
\end{corollary}

\begin{proof}
We apply $(4.2.9)$ with $Y=\Spec(K)$, $\sh{G}=\sh{O}_Y=K$, $M$ reduced to a single element.
\end{proof}

\subsection{Reminders on tensor products of fields}
\label{subsection:IV.4.3}

\oldpage[IV-2]{58}

For the convenience of the reader, we recall here certain properties of tensor products of fields that we will use in the following sections; for the proofs, we refer to \cite{I}.

\begin{env}[4.3.1]
\label{IV.4.3.1}
Recall that an extension $L$ of a field $k$ is said to be \emph{primary} if the largest separable algebraic extension of $k$ contained in $L$ is $k$ itself.
\end{env}

\begin{proposition}[4.3.2]
\label{IV.4.3.2}
Let $K$, $L$ be two extensions of a field $k$.
If $L$ is a primary extension of $k$, then $\Spec(L\otimes_k K)$ is irreducible, and if $\xi$ is its generic point, $\kappa(\xi)$ is a primary extension of $K$.
Conversely, if for every finite separable extension $K$ of $k$, $\Spec(L\otimes_k K)$ is irreducible, $L$ is a primary extension of $k$.
\end{proposition}

\begin{proof}
See the proof in \cite{I}, p.~14-03 to 14-06.
\end{proof}

\begin{corollary}[4.3.3]
\label{IV.4.3.3}
If $k$ is separably closed (i.e.\ if its separable algebraic closure is radicial over $k$), then, for any two extensions $K$, $L$ of $k$, $\Spec(L\otimes_k K)$ is irreducible, and conversely.
\end{corollary}

\begin{proof}
This follows immediately from $(4.3.2)$.
\end{proof}

\begin{corollary}[4.3.4]
\label{IV.4.3.4}
Let $L$ be an extension of a field $k$, $\mathrm{L}_s$ the largest separable algebraic extension of $k$ contained in $L$ (sometimes called the \emph{separable closure} of $k$ in $L$); suppose $\mathrm{L}_s$ is of finite degree over $k$, and let $K$ be a Galois extension of $k$ containing $\mathrm{L}_s$.
Then $\Spec(L\otimes_k K)$ has $[\mathrm{L}_s:k]$ irreducible components, and the residue fields at the generic points of these components are primary extensions of $K$.
\end{corollary}

\begin{proof}
Indeed, we have $L\otimes_k K = L\otimes_{\mathrm{L}_s}(\mathrm{L}_s\otimes_k K)$ and we know that $\mathrm{L}_s\otimes_k K$ is isomorphic to a product of $[\mathrm{L}_s:k]$ fields isomorphic to $K$ (Bourbaki, \emph{Alg.}, chap.~VIII, \S~7, n\textsuperscript{o}~3, cor.~2 du th.~1); hence $L\otimes_k K$ is isomorphic to a product of $[\mathrm{L}_s:k]$ rings isomorphic to $L\otimes_{\mathrm{L}_s} K$; since $L$ is a primary extension of $\mathrm{L}_s$, the conclusion follows from $(4.3.2)$.
\end{proof}

\begin{proposition}[4.3.5]
\label{IV.4.3.5}
Let $K$, $L$ be two extensions of a field $k$.
If $L$ is a separable extension of $k$, the ring $L\otimes_k K$ is reduced.
Conversely, if for every finite radicial extension $K$ of $k$, the ring $L\otimes_k K$ is reduced, $L$ is a separable extension of $k$.
\end{proposition}

\begin{proof}
For the proof, see Bourbaki, \emph{Alg.}, chap.~VIII, \S~7, n\textsuperscript{o}~3, th.~1.
\end{proof}

\begin{corollary}[4.3.6]
\label{IV.4.3.6}
If $k$ is perfect, then, for any two extensions $K$, $L$ of $k$, $K\otimes_k L$ is reduced, and conversely.
\end{corollary}

\begin{proof}
This follows immediately from $(4.3.5)$.
\end{proof}

\begin{corollary}[4.3.7]
\label{IV.4.3.7}
Let $L$ be a separable extension of $k$ and let $K$ be an extension of $k$ such that $K$ or $L$ is finite over $k$; then the residue fields of the semi-local ring $L\otimes_k K$ are separable extensions of $K$.
\end{corollary}

\begin{proof}
Indeed, we know that $L\otimes_k K$ is the direct sum of these residue fields $\mathrm{E}_i$; for every extension $K'$ of $K$, $L\otimes_k K'$ is therefore isomorphic to the direct sum of the rings $\mathrm{E}_i\otimes_K K'$; since $L\otimes_k K'$ is reduced, the $\mathrm{E}_i$ are separable over $K$ by virtue of $(4.3.5)$.
\end{proof}

\begin{corollary}[4.3.8]
\label{IV.4.3.8}
If $K$ and $L$ are two finite separable extensions of $k$, $K\otimes_k L$ is a product of a finite number of separable extensions of $k$.
\end{corollary}

\begin{proposition}[4.3.9]
\label{IV.4.3.9}
If $k$ is algebraically closed, then, for any two extensions $K$, $L$ of $k$, $L\otimes_k K$ is integral, and conversely.
\end{proposition}

\begin{proof}
This is a consequence of $(4.3.3)$ and $(4.3.6)$, a perfect field being separably closed if and only if it is algebraically closed.
\end{proof}

\subsection{Irreducible preschemes and connected preschemes over an algebraically closed field}
\label{subsection:IV.4.4}

\oldpage[IV-2]{59}

\begin{env}[4.4.1]
\label{IV.4.4.1}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$.
Since the morphism $\Spec(K)\to\Spec(k)$ is faithfully flat, quasi-compact, and universally open \sref{IV.2.4.9}, the same is true of the projection morphism $p:X\otimes_k K\to X$ \sref{IV.2.2.13}.
It follows therefore from \sref{IV.2.3.5} that every irreducible component of $X\otimes_k K$ \emph{dominates} an irreducible component of $X$, and (since $p$ is surjective) every irreducible component of $X$ is dominated by an irreducible component of $X\otimes_k K$; we thus deduce that $p$ is a \emph{surjective} map from the set of irreducible components of $X\otimes_k K$ to the set of irreducible components of $X$.
Similarly, it follows from the fact that $p$ is continuous and surjective that every connected component of $X$ is the image by $p$ of a union of connected components of $X\otimes_k K$, whence a \emph{surjective} map from the set of connected components of $X\otimes_k K$ to the set of connected components of $X$.
We conclude that every irreducible (resp.\ connected) component $Z'$ of $X\otimes_k K$ is contained in a \emph{unique} set of the form $p^{-1}(Z)$, where $Z$ is an irreducible (resp.\ connected) component of $X$; for every sub-prescheme $Z$ of $X$ having $Z$ as underlying space (and still denoted $Z$), $Z'$ is an irreducible (resp.\ connected) component of $Z\otimes_k K$.
In particular, if $X\otimes_k K$ has a finite number $n$ (resp.\ $n'$) of irreducible (resp.\ connected) components (which will be the case when $X$ is of \emph{finite type} over $k$, since then $X\otimes_k K$ is of finite type over $K$, hence Noetherian), the number of irreducible (resp.\ connected) components of $X$ is $\leq n$ (resp.\ $\leq n'$); to say that this number is equal to $n$ (resp.\ $n'$) means that for every reduced sub-prescheme $Z$ of $X$ having an irreducible (resp.\ connected) underlying space, $Z\otimes_k K$ is irreducible (resp.\ connected) \sref[I]{I.5.1.8}; the number of irreducible (resp.\ connected) components of $X\otimes_k K$ is therefore independent of $K$ if and only if for every irreducible (resp.\ connected) component $Z$ of $X$, $Z\otimes_k K$ is irreducible (resp.\ connected) for \emph{every} extension $K$ of $k$.

In particular, if $X\otimes_k K$ is irreducible (resp.\ connected), so is $X$ (which already follows from the fact that $p$ is surjective).
Similarly, if $X\otimes_k K$ is \emph{reduced} (resp.\ \emph{integral}), so is $X$ since $p$ is faithfully flat $(2.1.13)$.

In this section and the next, we will examine more closely what can be said about the irreducible (resp.\ connected) components of $X\otimes_k K$ as $K$ varies; the above already shows that the number of these components is a \emph{non-decreasing} function of $K$.
\end{env}

\begin{lemma}[4.4.2]
\label{IV.4.4.2}
Let $X'$, $X$ be two topological spaces, $f:X'\to X$ a continuous map.
Suppose that the following conditions are satisfied:
\begin{enumerate}[label=(\roman*)]
  \item $f$ is a surjective open map (resp.\ such that the topological space $X$ is canonically identified with the quotient of $X'$ by the equivalence relation defined by $f$).
\end{enumerate}
\end{lemma}

\oldpage[IV-2]{60}
\begin{enumerate}
  \item[\rm(ii)] For every $x\in X$, $f^{-1}(x)$ is irreducible (resp.\ connected).
\end{enumerate}

Then for $X'$ to be irreducible (resp.\ connected), it is necessary and sufficient that $X$ be so.

\begin{proof}
The assertion concerning connectedness is nothing other than Bourbaki, \emph{Top.\ g\'en.}, chap.~I\textsuperscript{er}, 3\textsuperscript{e} \'ed., \S~11, n\textsuperscript{o}~3, prop.~7.
Let us prove the assertion concerning irreducibility; the condition being trivially necessary since $f$ is surjective, let us prove that it is sufficient.
Let $X'_1$, $X'_2$ be two closed subsets of $X'$ such that $X'=X'_1\cup X'_2$, and denote by $X_i$ ($i=1,2$) the set of $x\in X$ such that $f^{-1}(x)\subset X'_i$; we have $X_i=X-f(X'-X'_i)$, and since $f$ is open, we conclude that $X_i$ is a closed part of $X$ ($i=1,2$).
On the other hand, for every $x\in X$, $f^{-1}(x)$ is irreducible by hypothesis, and is the union of the two closed subsets $X'_1\cap f^{-1}(x)$ and $X'_2\cap f^{-1}(x)$; one of these two subsets must therefore be equal to $f^{-1}(x)$, which means that $x\in X_1$ or $x\in X_2$; we therefore have $X=X_1\cup X_2$, and since $X$ is irreducible by hypothesis, we deduce $X=X_1$ or $X=X_2$, hence $X'=X'_1$ or $X'=X'_2$.
\end{proof}

\begin{remark}[4.4.3]
\label{IV.4.4.3}
If the topology of $X$ is not the quotient of that of $X'$ by the equivalence relation defined by $f$, it can happen that $X$ and all the fibres $f^{-1}(x)$ are irreducible, without $X'$ being so: one can construct an example by taking for $X$ an irreducible scheme (for example the affine line $\Spec(k[\mathrm{T}])$) over an algebraically closed field $k$ and, considering a closed point $x_0$ of $X$ and taking for $X'$ the direct \emph{sum} of $\{x_0\}$ and the open subset $X-\{x_0\}$ of $X$.
Similarly, if one replaces the hypothesis ``$f$ open'' by ``$f$ closed'' (or even ``$f$ proper'') (which implies, however, that the topology of $X$ is then the quotient of that of $X'$ by the relation defined by $f$), it can happen that $X$ and all the fibres $f^{-1}(x)$ are irreducible without $X'$ being so.
Let us take for $X$ the affine line over $k$; if $x_0$ is a closed point of $X$, $t_0$ a closed point of $\mathrm{P}=X\times_k\mathrm{P}$, where $\mathrm{P}$ is the projective line over $k$, $p:S\to X$, $q:S\to\mathrm{P}$ the projections, denote by $X'$ the reduced sub-prescheme having for underlying space $p^{-1}(x_0)\cup q^{-1}(t_0)$, and let us take $f$ to be the restriction of $p$ to $X'$ of $p$; $f$ is proper but $X'$ is not irreducible.
\end{remark}

\begin{theorem}[4.4.4]
\label{IV.4.4.4}
Let $k$ be an algebraically closed field, $X$ a $k$-prescheme.
If $X$ is irreducible (resp.\ connected), so is $X\otimes_k K$ for every extension $K$ of $k$.
\end{theorem}

\begin{proof}
We saw indeed in $(4.4.1)$ that the morphism $p:X\otimes_k K\to X$ is faithfully flat and open; to be able to apply the lemma $(4.4.2)$, it suffices therefore to verify that the fibres $p^{-1}(x)$ are irreducible for every $x\in X$.
But the $\kappa(x)$-prescheme $p^{-1}(x)$ is isomorphic to $\Spec(\kappa(x)\otimes_k K)$ \sref[I]{I.3.6.2}, hence integral by virtue of the hypothesis on $k$ and of $(4.3.9)$.
\end{proof}

\begin{corollary}[4.4.5]
\label{IV.4.4.5}
Let $k$ be an algebraically closed field, $X$ a $k$-prescheme, $K$ an extension of $k$, $p:X\otimes_k K\to X$ the canonical projection.
If $Z$ is an irreducible (resp.\ connected) part of $X$, $p^{-1}(Z)$ is an irreducible (resp.\ connected) part; in particular, if $X_0$ is an irreducible (resp.\ connected) component of $X$ containing $Z$, $p^{-1}(X_0)$ is an irreducible (resp.\ connected) component of $X\otimes_k K$ containing $p^{-1}(Z)$.
\end{corollary}

\begin{proof}
The second assertion follows from the first applied to $(4.4.1)$ replacing $Z$ by $X_0$; to prove the first, let $X'$ be the underlying space of $X\otimes_k K$, $Z'=p^{-1}(Z)$; the equivalence relation $R$ defined by $p$ on $X'$ is open, hence so is the relation $R_{Z'}$ it induces on the saturated part $Z'$, and $Z$ identifies with the quotient space $Z'/R_{Z'}$ (Bourbaki, \emph{Top.\ g\'en.}, chap.~I\textsuperscript{er}, 3\textsuperscript{e} \'ed., \S~5, n\textsuperscript{o}~2, prop.~4); we can therefore apply the lemma $(4.4.2)$ to $Z$ and $Z'$, whence the first assertion of $(4.4.5)$.
\end{proof}

\begin{corollary}[4.4.6]
\label{IV.4.4.6}
The hypotheses and notation being those of \sref{IV.4.4.5}, the map $Z\mapsto p^{-1}(Z)$ is a bijection from the set of irreducible (resp.\ connected) components of $X$ to the set of irreducible (resp.\ connected) components of $X\otimes_k K$, whose inverse bijection is $Z'\mapsto p(Z')$.
\end{corollary}

\subsection{Geometrically irreducible and geometrically connected preschemes}
\label{subsection:IV.4.5}

\begin{proposition}[4.5.1]
\label{IV.4.5.1}
Let $k$ be a field, $X$ a $k$-prescheme, $\Omega$ an algebraically closed extension of $k$.
Let $n$ (resp.\ $n'$) be the cardinal of the set of irreducible (resp.\ connected) components of $X\otimes_k\Omega$; this number is independent of the algebraically closed extension $\Omega$ chosen.
Furthermore, for every extension $K$ of $k$, the cardinal of the set of irreducible (resp.\ connected) components of $X\otimes_k K$ is $\leq n$ (resp.\ $\leq n'$).
\end{proposition}

\begin{proof}
Indeed, two algebraically closed extensions $\Omega$, $\Omega'$ of $k$ can always be considered as subextensions of the same extension $E$ of $k$; it suffices then to apply $(4.4.6)$ to $X\otimes_k\Omega$ and $X\otimes_k\Omega'$ and to the common extension $E$ of $\Omega$ and $\Omega'$ to prove the first assertion.
The second is obtained by taking for $\Omega$ an algebraically closed extension of $k$ and using $(4.4.1)$.
\end{proof}

\begin{definition}[4.5.2]
\label{IV.4.5.2}
The cardinal $n$ (resp.\ $n'$) of \sref{IV.4.5.1} is called the \emph{geometric number of irreducible components} (resp.\ \emph{of connected components}) of $X$ (relative to $k$).
If $n=1$ (resp.\ $n'\leq 1$) we say that $X$ is \emph{geometrically irreducible} (resp.\ \emph{geometrically connected}).
\end{definition}

We thus recover a definition given for a particular case in \sref[III]{III.4.3.4}.
By virtue of $(4.5.1)$, the following properties are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $X$ is geometrically irreducible (resp.\ geometrically connected).
  \item For an algebraically closed extension $\Omega$ of $k$, $X\otimes_k\Omega$ is irreducible (resp.\ connected).
  \item For every extension $K$ of $k$, $X\otimes_k K$ is irreducible (resp.\ connected).
\end{enumerate}

\begin{env}[4.5.3]
\label{IV.4.5.3}
Let $X$ be a $k$-prescheme, $Z$ a locally closed subset of $X$.
If we again denote by $Z$ any of the sub-preschemes of $X$ having $Z$ as underlying space, it follows from \sref[I]{I.5.2.1} that for an extension $K$ of $k$, the number of irreducible (resp.\ connected) components of $Z\otimes_k K$ is independent of the sub-prescheme with underlying space $Z$ that one has chosen; we thus define the geometric number of irreducible (resp.\ connected) components of $Z$ as that of any sub-prescheme of $X$ having $Z$ as underlying space.
\end{env}

\begin{proposition}[4.5.4]
\label{IV.4.5.4}
Let $X$, $Y$ be two $k$-preschemes, $f:X\to Y$ a surjective $k$-morphism.
If $X$ is geometrically irreducible (resp.\ geometrically connected), so is $Y$.
\end{proposition}

\begin{proof}
Indeed, for every extension $K$ of $k$, $f_{(\mathrm{K})}:X_{(\mathrm{K})}\to Y_{(\mathrm{K})}$ is surjective \sref[I]{I.3.5.2}, and $X_{(\mathrm{K})}$ being irreducible (resp.\ connected) by hypothesis, the same holds for $Y_{(\mathrm{K})}$.
\end{proof}

\oldpage[IV-2]{62}

\begin{definition}[4.5.5]
\label{IV.4.5.5}
A morphism $f:X\to Y$ of $k$-preschemes is said to be \emph{geometrically irreducible} (resp.\ \emph{geometrically connected}), if, for every $y\in Y$, the $\kappa(y)$-prescheme $f^{-1}(y)$ is geometrically irreducible (resp.\ geometrically connected).
\end{definition}

\begin{proposition}[4.5.6]
\label{IV.4.5.6}
\begin{enumerate}[label=(\roman*)]
  \item Let $X$ be a $k$-prescheme, $K$ an extension of $k$.
Then the geometric number of irreducible (resp.\ connected) components of $X_{(\mathrm{K})}$ relative to $K$ is equal to the geometric number of irreducible (resp.\ connected) components of $X$ relative to $k$.
In particular, for the $K$-prescheme $X_{(\mathrm{K})}$ to be geometrically irreducible (resp.\ geometrically connected), it is necessary and sufficient that the $k$-prescheme $X$ be so.
  \item Let $f:X\to Y$, $g:Y'\to Y$ be two morphisms, and set $X'=X_{(Y')}=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$.
If $f$ is irreducible (resp.\ connected), so is $f'$.
The converse is true if $g$ is surjective.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item If $\Omega$ is an algebraically closed extension of $k$, we have $X\otimes_k\Omega=X_{(\mathrm{K})}\otimes_K\Omega$, whence the assertion.
  \item For every $y'\in Y'$, if we set $y=g(y')$, we have $f'^{-1}(y')=f^{-1}(y)\otimes_{\kappa(y)}\kappa(y')$ \sref[I]{I.3.6.4}, hence the two assertions follow from~(i).
\end{enumerate}
\end{proof}

\begin{proposition}[4.5.7]
\label{IV.4.5.7}
Let $f:X\to Y$ be a surjective morphism that is irreducible (resp.\ connected), $g:Y'\to Y$ a morphism, and set $X'=X\times_Y Y'$.
If in addition $Y'$ is irreducible (resp.\ connected), and $f$ is universally open (resp.\ flat and quasi-compact, or universally open, or universally closed), then $X'$ is irreducible (resp.\ connected).
\end{proposition}

\begin{proof}
The properties of $f$ that one considers being all stable under base change, we can restrict to the case where $Y'=Y$; it suffices then to apply $(4.4.2)$ (taking into account \sref{IV.2.3.12}).
\end{proof}

\begin{corollary}[4.5.8]
\label{IV.4.5.8}
Let $X$, $Y$ be two $k$-preschemes.
\begin{enumerate}[label=(\roman*)]
  \item If $X$ is geometrically irreducible (resp.\ geometrically connected) and $Y$ irreducible (resp.\ connected), then $X\times_k Y$ is irreducible (resp.\ connected).
  \item If $X$ and $Y$ are geometrically irreducible (resp.\ geometrically connected), so is $X\times_k Y$.
\end{enumerate}
\end{corollary}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item The structural morphism $X\to\Spec(k)$ is surjective and universally open (resp.\ connected); it suffices therefore to apply $(4.5.7)$.
  \item If $\Omega$ is an algebraically closed extension of $k$, we have $(X\times_k Y)_{(\Omega)}=X_{(\Omega)}\times_\Omega Y_{(\Omega)}$; by hypothesis $X_{(\Omega)}$ and $Y_{(\Omega)}$ are $\Omega$-preschemes geometrically irreducible (resp.\ geometrically connected) \sref[I]{I.3.3.10}; by hypothesis $X_{(\Omega)}$ and $Y_{(\Omega)}$ are $\Omega$-preschemes that are geometrically irreducible (resp.\ geometrically connected) \sref{IV.4.5.6}, and it therefore suffices to apply~(i).
\end{enumerate}
\end{proof}

\begin{proposition}[4.5.9]
\label{IV.4.5.9}
Let $X$ be a $k$-prescheme.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $X$ is geometrically irreducible (in other words, for \emph{every} extension $K$ of $k$, $X\otimes_k K$ is irreducible).
  \item For every finite separable extension $K$ of $k$, $X\otimes_k K$ is irreducible.
  \item $X$ is irreducible, and if $x$ is its generic point, $\kappa(x)$ is a primary extension of $k$.
\end{enumerate}
\end{proposition}

\begin{proof}
It is clear that \emph{a)} implies \emph{b)}.
To see that \emph{b)} implies \emph{c)}, let us consider a finite separable extension $K$ of $k$; if $p:X\otimes_k K\to X$ is the canonical projection, the maximal points of $X\otimes_k K$ are those of the fibre $p^{-1}(x)=\Spec(\kappa(x)\otimes_k K)$ \sref{IV.2.3.4} and $(\mathbf{0}_I,\,2.1.8)$; to say that this fibre is irreducible for every finite separable extension $K$ of $k$ amounts to saying that $\kappa(y)$ is a primary extension of $k$, by virtue of $(4.3.2)$.
Conversely, if \emph{c)} is satisfied, the same reasoning (taking into account $(4.3.2)$) shows that \emph{d)} implies \emph{e)}.
Let us show that \emph{d)} implies \emph{a)}.
It suffices to show that $\kappa(x_\alpha)\otimes_k k'$ is a reduced ring for every finite radicial extension $k'$ of $k$ $(4.3.5)$.
We can restrict to the case where $X=\Spec(A)$ is affine, $A$ being a $k$-algebra that is reduced; then by hypothesis the ring $A\otimes_k k'$ is reduced \sref[I]{I.5.1.4} and $\kappa(x_\alpha)\otimes_k k'$ is a ring of fractions of this ring, hence it is reduced $(\mathbf{0}_I,\,1.2.8)$.
\end{proof}

\begin{corollary}[4.5.10]
\label{IV.4.5.10}
Let $X$ be an irreducible $k$-prescheme, $x$ its generic point, $k'$ the separable algebraic closure of $k$ in $\kappa(x)$, $k''$ a Galois extension of $k$ (of finite degree or not) containing $k'$.
Suppose $k'$ is finite over $k$; then the irreducible components of $X\otimes_k k''$ are geometrically irreducible; their number is equal to $[k':k]$ and is also the geometric number of irreducible components of $X$.
\end{corollary}

\begin{proof}
As in $(4.5.9)$, we are brought to considering the maximal points of the fibre $\Spec(\kappa(x)\otimes_k k'')$; by virtue of $(4.3.4)$, they are $[k':k]$ in number and the residue fields at these points (which are also those of $X\otimes_k k''$) are primary extensions of $k''$, which, taking into account $(4.5.9)$, proves the corollary.
\end{proof}

\begin{corollary}[4.5.11]
\label{IV.4.5.11}
Let $X$ be a $k$-prescheme; suppose that $X$ has only a finite number of maximal points $x_i$ ($1\leq i\leq r$) and that for each $i$, the separable algebraic closure $k'_i$ of $k$ in $\kappa(x_i)$ is of finite degree over $k$.
Then there exists a finite separable extension $L$ of $k$ such that the irreducible components of $X\otimes_k L$ are geometrically irreducible; their number, equal to $\sum_i [k'_i:k]$, is the geometric number of irreducible components of $X$.
\end{corollary}

\begin{proof}
The maximal points of $X\otimes_k L$ are those of the various fibres $\Spec(\kappa(x_i)\otimes_k L)$ applied to the irreducible components of $X$ and to their inverse images in $X\otimes_k L$.
It suffices therefore to take for $L$ a Galois extension of $k$ finite over $k$ containing all the $k'_i$ and to apply $(4.5.10)$.
\end{proof}

We note that $(4.5.11)$ applies in particular to every $k$-prescheme $X$ \emph{of finite type} over $k$, since $X$ is then Noetherian, hence has only a finite number of irreducible components, and each of the $\kappa(x_i)$ is an extension of finite type of $k$, hence it is also the case for the separable algebraic closure of $k$ in $\kappa(x_i)$ (\cite{I}, p.~6-06, lemma~5).

\begin{remarks}[4.5.12]
\label{IV.4.5.12}
\begin{enumerate}[label=(\roman*)]
  \item The notions defined in $(4.5.2)$ depend on the base field $k$.
To abbreviate, and when it is necessary to mention the base field, one may say ``$k$-irreducible'' (resp.\ ``$k$-connected'') instead of ``geometrically irreducible (resp.\ connected) relative to $k$''.
We note that this terminology, in the context of schemes, is exactly the opposite of that of Weil: for Weil, ``$k$-irreducible'', ``$k$-connected'', ``$k$-normal'', etc., designate an \emph{intrinsic} property of a prescheme $X$, independent of the base field $k$ (in other words, of the morphism $X\to\Spec(k)$ chosen); we express these properties by saying simply that $X$ is irreducible (resp.\ connected, normal, etc.).
Weil says on the other hand ``absolutely irreducible'', ``absolutely connected'', ``absolutely normal'', etc., for the corresponding notions relative to the prescheme $X\otimes_k\Omega$, where $\Omega$ is an algebraically closed extension of $k$; the definition of a smaller ``base field'' (i.e., of a scheme over $k$ defining $V$ by extension of the base field from $k$ to $\Omega$) is for him a supplementary structure, and in a certain measure, secondary, so that his qualification of ``absolute'' or ``intrinsic'' on one hand, ``relative to $k$'' on the other, is the opposite of ours.
We will therefore in the sequel avoid using the qualifier ``absolute'', which can lead to confusion, and whenever we use the abbreviated terminology introduced above (in conflict with the accepted terminology, we refer to $(4.5.12)$) to avoid any ambiguity.
\oldpage[IV-2]{64}
  \item The proposition $(4.5.9)$ gives a ``birational'' criterion (in other words, depending only on the residue field at the generic point) for a $k$-prescheme $X$ to be geometrically irreducible.
There is no analogous criterion for $X$ to be geometrically connected.
For example, take $X=\Spec(\mathbf{R}[[\mathrm{T},\mathrm{U}]]/(\mathrm{T}^2+\mathrm{U}^2))$ ($\mathrm{T}$, $\mathrm{U}$ indeterminates); $X$ is an $\mathbf{R}$-scheme that is integral, and its field of rational functions $\kappa(x)$ is $\mathbf{C}((\mathrm{T}))$, as one easily verifies; $X\otimes_\mathbf{R}\mathbf{C}$ decomposes into two irreducible components (the two ``isotropic lines'') that have a common point (the maximal image in $\mathbf{C}[[\mathrm{T},\mathrm{U}]]/(\mathrm{T}^2+\mathrm{U}^2)$); hence $X\otimes_\mathbf{R}\mathbf{C}$ is connected.
But let $X'$ be the complement (open) of the closed point of $X$ corresponding to the maximal ideal $(\mathrm{T})+(\mathrm{U})$ of $\mathbf{R}[[\mathrm{T},\mathrm{U}]]$; then $X'\otimes_\mathbf{R}\mathbf{C}$ is not connected, even though $X$ and $X'$ have the same field of rational functions.
\end{enumerate}
\end{remarks}

\begin{proposition}[4.5.13]
\label{IV.4.5.13}
Let $X$, $Y$ be two $k$-preschemes, $f:Y\to X$ a $k$-morphism.
Suppose $Y$ is geometrically connected and non-empty, and $X$ connected.
Then $X$ is geometrically connected.
\end{proposition}

\begin{proof}
Let $\Omega$ be an algebraic closure of $k$, $X'=X_{(\Omega)}$, $Y'=Y_{(\Omega)}$, $f'=f_{(\Omega)}:Y'\to X'$; it suffices to show that $X'$ is connected.
Let $U'$ be an open and closed non-empty subset of $X'$; note that the morphisms $p:X'\to X$ and $q':Y'\to Y$ are \emph{open} \sref{IV.2.4.10} and \emph{closed} since $\Omega$ is an algebraic extension of $k$ \sref[II]{II.6.1.10}.
Hence $U=p(U')$ is open and closed and non-empty in $X$.
Since $Y$ is non-empty, on concludes that $U'=p(U')$ is non-empty, and we deduce that $U'=X'$, without which the same reasoning applied to the open and closed subset $X'-U'$ would show that $f'^{-1}(X'-U')=Y'$, which is absurd since $Y'$ is connected.
\end{proof}

\begin{proposition}[4.5.15]
\label{IV.4.5.15}
Let $X$ be a $k$-prescheme, $x$ a point of $X$, $k'$ the separable algebraic closure of $k$ in $\kappa(x)$.
Suppose the following conditions are satisfied:
\begin{enumerate}[label=(\roman*)]
  \item $X$ is connected.
  \item $k'$ is a finite extension of $k$.
\end{enumerate}
Then the geometric number of connected components of $X$ is $\leq [k':k]$, and if $k''$ is a finite Galois extension of $k$ containing $k'$, the connected components of $X\otimes_k k''$ are geometrically connected.
\end{proposition}

\begin{proof}
We note that condition~(ii) is satisfied if $X$ is \emph{locally of finite type} over $k$.

There exist finite Galois extensions $k''$ of $k$ containing $k'$ by virtue of condition~(ii).
Set $X''=X\otimes_k k''$; the morphism projection $p:X''\to X$ is faithfully flat and finite, hence closed \sref[II]{II.6.1.10} and by \sref{IV.2.3.6},~(ii)) the image by $p$ of every connected component $X''_\alpha$ of $X''$ is equal to $X$; $X''_\alpha$ therefore contains a point $x''_\alpha\in p^{-1}(x)$.
But $p^{-1}(x)$ has a number of points equal to the number of irreducible components of $\Spec(\kappa(x)\otimes_k k'')$ \sref[I]{I.3.4.9}, that is to say $[k':k]$ $(4.3.4)$, and \emph{a fortiori} the number of connected components of $X''$ is $\leq[k':k]$.
For every $x''_\alpha\in p^{-1}(x)$, $\kappa(x''_\alpha)$ is a primary extension of $k''$ by virtue of $(4.3.5)$ and \sref[I]{I.3.4.9}.
We can therefore apply the corollary $(4.5.14)$ to $x''_\alpha$ and $X''$, which proves that all the connected components of $X''$ are geometrically connected.
\end{proof}

\begin{corollary}[4.5.16]
\label{IV.4.5.16}
Let $X$ be a $k$-prescheme, $(X_\alpha)$ the family of its connected components, and for every $\alpha$, let $x_\alpha\in X_\alpha$.
Suppose the following conditions are satisfied:
\begin{enumerate}[label=(\roman*)]
  \item The family $(X_\alpha)$ is finite.
  \item For every $\alpha$, the separable algebraic closure $k'_\alpha$ of $k$ in $\kappa(x_\alpha)$ is a finite extension of $k$.
\end{enumerate}
Then the geometric number of connected components of $X$ is at most $\sum_\alpha [k'_\alpha:k]$, and there exists a finite separable extension $k''$ of $k$ such that all the connected components of $X\otimes_k k''$ are geometrically connected.
\end{corollary}

\begin{proof}
Noting that the connected components of $X$ are open by virtue of condition~(i), it suffices to apply to each of the sub-preschemes of $X$ induced on the open sets $X_\alpha$ the result of $(4.5.15)$; for having an extension $k''$ answering the question, it suffices to take a Galois extension of $k$ finite over $k$ containing all the $k'_\alpha$.
\end{proof}

\begin{corollary}[4.5.17]
\label{IV.4.5.17}
Suppose that the $k$-prescheme $X$ contains a point $x$ such that the separable algebraic closure of $k$ in $\kappa(x)$ is finite over $k$.
For $X$ to be geometrically connected, it is necessary and sufficient that for every finite separable extension $K$ of $k$, $X\otimes_k K$ be connected.
\end{corollary}

\begin{proposition}[4.5.19]
\label{IV.4.5.19}
Let $X$ be a $k$-prescheme, $Z$ a subset of $X$, $k'$ an algebraically closed extension of $k$, $X'=X\otimes_k k'$, $p:X'\to X$ the canonical projection.
Suppose that $Z'=p^{-1}(Z)$ is contained in a single irreducible component $X'_0$ of $X'$.
Then $X_0=p(X'_0)$ is an irreducible component of $X$ containing $Z$, and furthermore $X_0$ is geometrically irreducible.
If furthermore $X'_0$ is the only irreducible component of $X'$ that meets $Z'$, then $X_0$ is the only irreducible component of $X$ that meets $Z$.
\end{proposition}

To show that $X'_0=p^{-1}(X_0)$, we shall apply the following:

\begin{lemma}[4.5.19.1]
\label{IV.4.5.19.1}
Let $T$, $T'$ be two preschemes, $p:T'\to T$ a morphism; consider the prescheme $T''=T'\times_T T'$, and let $p_1:T''\to T'$, $p_2:T''\to T'$ be the projections.
For a subset $U'$ of $T'$ to be of the form $p^{-1}(U)$, where $U\subset T$, it is necessary and sufficient that $p_1^{-1}(U')=p_2^{-1}(U')$.
\end{lemma}

\begin{proof}
The structural morphism $q:T''\to T$ is equal to $p\circ p_1$ and to $p\circ p_2$ by definition, hence the relation is necessary.
Conversely, suppose it is satisfied; let $u'$ be a point of $U'$, $t'$ a point of $p^{-1}(p(u'))$; since $p(t')=p(u')$, there exists a point $t''\in T''$ such that $p_1(t'')=u'$, $p_2(t'')=t'$ \sref[I]{I.3.4.7}; hence $t''\in p_1^{-1}(U')=p_2^{-1}(U')$ by hypothesis, and by $t'=p_2(t'')\in U'$, which proves the lemma.
\end{proof}

\begin{proof}[Proof of {\rm (4.5.19)}]
With this lemma established, to prove that $X'_0=p^{-1}(X_0)$, we form the product $X''=X'\times_X X'$ (relative to the morphism $p:X'\to X$); if $p_1$ and $p_2$ are the canonical projections of $X''$ to $X'$, it is a matter of showing that $p_1^{-1}(X'_0)=p_2^{-1}(X'_0)$.
Now, if we set $S=\Spec(k)$, $S'=\Spec(k')$, observing that if $S''=S'\times_S S'=\Spec(k'\otimes_k k')$, we can also write $X''=X'\times_{S'}S''$.
If $\varphi'':X''\to S''$ is the structural morphism, it suffices to prove that for every $s''\in S''$, we have $p_1^{-1}(X'_0)\cap\varphi''^{-1}(s'')=p_2^{-1}(X'_0)\cap\varphi''^{-1}(s'')$.
If we set $T=\Spec(\kappa(s''))$, $U=X''\times_{S'}T=X'\times_{S'}T$, and if $v:U\to X''$ is the canonical projection, it is a matter of proving that $U_1=v^{-1}(p_1^{-1}(X'_0))$ and $U_2=v^{-1}(p_2^{-1}(X'_0))$ are identical irreducible components of $U$ $(4.4.5)$, and since $U_1$ and $U_2$ both contain $w^{-1}(Z)$ (where $w=p\circ p_1\circ v=p\circ p_2\circ v$), the result follows from $(4.4.5)$.

\oldpage[IV-2]{67}

There exists a field $K$, common extension of $k'$ and of $\kappa(s'')$, so that if we set $P=\Spec(K)$, the diagram
\[
\xymatrix{
S' & \ar[l] P\\
S \ar[u] & \ar[l] T \ar[u]
}
\]
is commutative; if we set $Q=X\otimes_k K$, the diagram
\[
\xymatrix{
X' & \ar[l]^-{q} Q\\
X \ar[u]^-{p} & \ar[l]_-{w} U \ar[u]_-{r}
}
\]
is therefore also commutative, and the two irreducible components $r^{-1}(U_1)$ and $r^{-1}(U_2)$ of $Q$ $(4.4.5)$ both contain $q^{-1}(Z')$; now, by virtue of $(4.4.5)$, the images $q(r^{-1}(U_1))$ and $q(r^{-1}(U_2))$ are irreducible components of $X'$ containing $Z'$; they are therefore both equal to $X'_0$, hence $r^{-1}(U_1)=r^{-1}(U_2)$ by virtue of $(4.4.5)$, and finally $U_1=U_2$.

Since $p$ is surjective and $X'_0$ dominates an irreducible component of $X$, $X_0=p(X'_0)$ is an irreducible component of $X$ containing $Z$, and since $p^{-1}(X_0)$ is an irreducible component of $X'$, $X_0$ is geometrically irreducible.

To prove the last assertion, let $X_1$ be a second irreducible component of $X$ meeting $Z$.
Taking $z\in Z\cap X_1$ \sref{IV.2.3.5}, one sees that there exists an irreducible component of $X'$ meeting $Z'$ and dominating $X_1$; but since $X'_0$ is by hypothesis the only irreducible component of $X'$ meeting $Z'$, we necessarily have $X_1=X_0$.
\end{proof}

\begin{remark}[4.5.20]
\label{IV.4.5.20}
The fact that $Z'$ is contained in a single irreducible component $X'_0$ of $X'$ does \emph{not} imply that $X_0=p(X'_0)$ is the only irreducible component of $X$ containing $Z$.
Let us indeed take $k=\mathbf{R}$, $k'=\mathbf{C}$, and for $X$ the $k$-scheme obtained by gluing (at a non-generic point) $X_1=\Spec(\mathbf{R}[\mathrm{S},\mathrm{T}]/(\mathrm{S}^2+\mathrm{T}^2+1))$ and $X_2=\Spec(\mathbf{C}[\mathrm{T}])$ (the local rings of these two curves at each of their closed points having residue field isomorphic to $\mathbf{C}$); $X_1$ and $X_2$ are identified with the two irreducible components of $X$; if $x$ is their common point, $p^{-1}(x)$ is formed of two points distinct $y'$, $z'$ of $X'$, $p^{-1}(X_1)$ is irreducible, and $p^{-1}(X_2)$ is the union of two irreducible components $Y'$, $Z'$ of $X'$ such that $y'\in Y'$ and $z'\in Z'$, so that $p^{-1}(x)$ is not contained in a single irreducible component of $X'$.
\end{remark}

\begin{proposition}[4.6.6]
\label{IV.4.6.6}
Let $X$ be a $k$-prescheme of finite type.
There exists a finite radicial extension $k'$ of $k$ such that $(X_{(k')})_{\mathrm{red}}$ is geometrically reduced over $k'$.
\end{proposition}

\begin{proof}
Since $X$ can be covered by a finite number of affine opens $U_i$, we may restrict to the case where $X$ is affine: indeed, if for each $i$, $k_i$ is a finite radicial extension of $k$ such that $(U_i\otimes_k k_i)_{\mathrm{red}}$ is geometrically reduced over $k_i$, we may suppose that the $k_i$ are all contained in the same finite radicial extension $k'$ of $k$; $(U_i\otimes_k k')_{\mathrm{red}}$ then identifies with $(U_i\otimes_k k_i)_{\mathrm{red}}\otimes_{k_i} k'$ by virtue of \sref[I]{I.5.1.8}, and hence is geometrically reduced over $k'$ by hypothesis, and so the same is true of $(X\otimes_k k')_{\mathrm{red}}$.
Suppose therefore that $X=\Spec(A)$, where $A$ is a $k$-algebra of finite type, and let $\Omega$ be the algebraic closure of $k$.
Denote by $\mathfrak{N}$ the nilradical of $A\otimes_k\Omega$; since $A\otimes_k\Omega$ is a Noetherian ring, $\mathfrak{N}$ is an ideal generated by a finite number of elements of the form $y_i = \sum_j x_{ij}\otimes\xi_{ij}$ where $x_{ij}\in A$, $\xi_{ij}\in\Omega$.
Let $K$ be a finite sub-extension of $\Omega$ containing the $\xi_{ij}$, and let $\mathfrak{N}$ be the ideal of $A\otimes_k K$ generated by the $y_i$; then it is clear that the $y_i$ are nilpotent in $A\otimes_k K$; on the other hand, since $\mathfrak{N}\otimes_K\Omega = \mathfrak{N}'$ (and consequently $\mathfrak{N}'\cap(A\otimes_k K) = \mathfrak{N}$), $\mathfrak{N}$ contains the nilradical of $A\otimes_k K$, and hence is equal to it.
Since $(A\otimes_k\Omega)/\mathfrak{N}' = ((A\otimes_k K)/\mathfrak{N})\otimes_K\Omega$, we see that $(X\otimes_k K)_{\mathrm{red}}\otimes_K\Omega$ is reduced, and by \sref{IV.4.6.1}, $(X_{(K)})_{\mathrm{red}}$ is geometrically reduced over $K$.
Replacing $K$ by the quasi-Galois extension over $k$ that it generates in $\Omega$, we see from \sref[I]{I.5.1.8} that we may furthermore suppose $K$ to be quasi-Galois over $k$, and hence a separable extension of a finite radicial extension $k'$ of $k$ (Bourbaki, \emph{Alg.}, chap.~V, \S~10, n\textsuperscript{o}~9, prop.~14).
\oldpage[IV-2]{70}
By virtue of \sref[I]{I.5.1.8}, $(X_{(k')})_{\mathrm{red}}\otimes_{k'} K$ is isomorphic to $(X_{(K)})_{\mathrm{red}}$, and hence is geometrically reduced; it follows that $(X_{(k')})_{\mathrm{red}}$ is geometrically reduced by virtue of \sref{IV.4.6.5},~(i)).
\end{proof}

\begin{definition}[4.6.9]
\label{IV.4.6.9}
Let $k$ be a field, $X$ a $k$-prescheme, $x$ a point of $X$.
We say that $X$ is \emph{geometrically reduced} (or \emph{separable}) (resp.\ \emph{geometrically punctually integral}) at the point $x$ over $k$, if, for every extension $k'$ of $k$ and every point $x'$ of $X'=X\otimes_k k'$ above $x$, $X'$ is reduced (resp.\ integral) at $x'$ (i.e., $\sh{O}_{X',x'}$ is reduced (resp.\ integral)).
We say that $X$ is \emph{geometrically punctually integral} over $k$ in all its points, in other words if, for every extension $k'$ of $k$, all the local rings of $X\otimes_k k'$ are integral (in which case we also say that $X\otimes_k k'$ is \emph{punctually integral}).
\end{definition}

Let us note that, for $X$ to be geometrically reduced over $k$ \sref{IV.4.6.2}, it is necessary and sufficient that it be geometrically reduced over $k$ at every point $x\in X$.

\begin{proposition}[4.6.10]
\label{IV.4.6.10}
Let $k$ be a field, $X$ a $k$-prescheme, $k'$ an extension of $k$, $x'$ a point of $X'=X\otimes_k k'$, $x$ its image in $X$.
For $X$ to be geometrically reduced (resp.\ geometrically punctually integral) at $x$ over $k$, it is necessary and sufficient that $X'$ be so at $x'$ over $k'$.
In particular, for $X$ to be geometrically punctually integral over $k$, it is necessary and sufficient that $X'$ be geometrically punctually integral over $k'$.
\end{proposition}

\begin{proof}
It suffices to prove the first assertion.
The condition being evidently necessary, let us prove that it is sufficient.
Suppose therefore that $X'$ is geometrically reduced (resp.\ geometrically punctually integral) at the point $x'$ over $k'$ and let us prove that $X$ is so at the point $x$ over $k$, in other words that for every extension $k''$ of $k$ and every point $x''$ of $X''=X\otimes_k k''$ above $x$, $\sh{O}_{X'',x''}$ is reduced (resp.\ integral).
For this, let us note that there exists a point $z$ of $X'\times_X X''$ which projects to $x'$ and $x''$ \sref[I]{I.3.4.7}.
Let $s$ be the point of $\Spec(k'\otimes_k k'')$, image of $z$ \sref[I]{I.3.4.9}; set $K=\kappa(s)$, $Z=X\otimes_k K$, so that $K$ may be considered as a composite extension of $k'$ and $k''$, and $z$ as a point of $Z$ whose image in $X'$ (resp.\ $X''$) is $x'$ (resp.\ $x''$).
Since $X'$ is geometrically reduced (resp.\ geometrically punctually integral) at the point $x'$ over $k'$, the ring $\sh{O}_{Z,z}$ is reduced (resp.\ integral), and since $\sh{O}_{Z,z}$ is isomorphic to a local ring of $\Spec(\sh{O}_{X'',x'})\otimes_{k'} K$, which is isomorphic to a flat $\sh{O}_{X'',x'}$-module, it follows that $\sh{O}_{X'',x''}$ is reduced (resp.\ integral) $(\mathbf{0}_I,\,6.5.1)$.
\end{proof}

\begin{corollary}[4.6.11]
\label{IV.4.6.11}
Suppose $k$ is perfect (resp.\ algebraically closed).
For $X$ to be geometrically reduced (resp.\ geometrically punctually integral) at the point $x$ over $k$, it is necessary and sufficient that $\sh{O}_{X,x}$ be reduced (resp.\ integral).
\end{corollary}

\begin{proof}
This is evidently necessary.
Conversely, if this condition is satisfied, then, for every extension $k'$ of $k$, $\Spec(\sh{O}_{X,x})\otimes_k k'$ is reduced \sref{IV.4.6.4} (resp.\ integral \sref{IV.4.4.4});
\oldpage[IV-2]{71}
and hence, for every point $x'$ of $X'=X\otimes_k k'$ above $x$, $\sh{O}_{X',x'}$, which is isomorphic to a local ring of $\Spec(\sh{O}_{X,x})\otimes_k k'$, is reduced (resp.\ integral).
\end{proof}

\begin{proposition}[4.6.12]
\label{IV.4.6.12}
Let $k$ be a field, $X$ a $k$-prescheme, $x$ a point of $X$, $\Omega$ a perfect extension of $k$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $X$ is geometrically reduced at $k$ at the point $x$, in other words, for every extension $k'$ of $k$ and every point $x'$ of $X'=X\otimes_k k'$ above $x$, $\sh{O}_{X',x'}$ is reduced.
  \item The prescheme $X\otimes_k\Omega$ is reduced at a point above $x$.
  \item For every finite radicial extension $k'$ of $k$, $X'=X\otimes_k k'$ is reduced at the unique point $x'$ above $x$.
  \item $\Spec(\sh{O}_{X,x})$ is geometrically reduced over $k$.
  \item $\sh{O}_{X,x}$ is reduced, and for every irreducible component $Z$ of $X$ containing $x$, of generic point $z$, $\kappa(z)$ is a separable extension of $k$.
\end{enumerate}
\end{proposition}

\begin{proof}
The implications $d)\Rightarrow a)\Rightarrow b)\Rightarrow c)$ are immediate, taking into account for the last \sref{IV.4.6.10}, \sref{IV.4.6.11}, and the fact that every radicial extension of $k$ is isomorphic to a sub-extension of $\Omega$.
Let us prove that $c)$ implies $d)$.
It suffices to show that for every finite radicial extension $k'$ of $k$, $\sh{O}_{x,x}\otimes_k k'$ is reduced \sref{IV.4.6.1}; but this ring is the local ring of $X'$ at the unique point $x'$ of $X'$ above $x$ \sref[I]{I.3.5.8} and \sref[I]{I.3.5.7}; it is therefore reduced by virtue of hypothesis $c)$.
Finally, the equivalence of $d)$ and $e)$ follows from the equivalence of $b)$ and $e)$ in \sref{IV.4.6.1}, since the irreducible components of $X$ containing $x$ correspond bijectively to the irreducible components of $\Spec(\sh{O}_{X,x})$ \sref[I]{I.2.4.2}.
\end{proof}

\begin{corollary}[4.6.13]
\label{IV.4.6.13}
Under the hypotheses of \sref{IV.4.6.12}, suppose furthermore that $X$ is locally Noetherian.
Then conditions $a)$ to $e)$ of \sref{IV.4.6.12} are also equivalent to the following:
\begin{enumerate}
  \item[\emph{f)}] There exists an open neighbourhood $U$ of $x$ which is geometrically reduced over $k$.
\end{enumerate}
\end{corollary}

\begin{proof}
Indeed, this condition trivially implies that $X$ is geometrically reduced at the point $x$ over $k$.
Conversely, if this is so, since the ring $\sh{O}_{X,x}$ is reduced, there exists an open neighbourhood $U$ of $x$ in $X$ which is reduced \sref[I]{I.6.1.13}; taking $U$ small enough, we may furthermore suppose that $U$ does not meet any of the irreducible components of $X$ not containing $x$.
The criterion \sref{IV.4.6.12},~$e)$) then shows that $U$ is geometrically reduced over $k$, taking into account the criterion \sref{IV.4.6.1},~$e)$).
\end{proof}

It follows from \sref{IV.4.6.13} that when $X$ is locally Noetherian, the set of $x\in X$ at which $X$ is geometrically reduced over $k$ is \emph{open}, and it is the largest open subset of $X$ which is geometrically reduced over $k$.

Recall that if a prescheme is \emph{locally Noetherian} and punctually integral, it is \emph{locally integral} \sref[I]{I.6.1.13}.
If $X$ is a $k$-prescheme, locally of finite type over $k$, and geometrically punctually integral over $k$, it follows from this remark that, for every extension $k'$ of $k$, $X\otimes_k k'$ is \emph{locally integral} (one also says in this case that $X$ is \emph{geometrically locally integral}).

\begin{proposition}[4.6.15]
\label{IV.4.6.15}
\begin{enumerate}[label=(\roman*)]
  \item Let $k$ be a field, $X$ a $k$-prescheme of finite type.
For $X$ to be geometrically punctually integral, it is necessary and sufficient that it be geometrically reduced and that the geometric number of its irreducible components be equal to the geometric number of its connected components.
  \item Let $X$ be a prescheme locally of finite type over $k$.
If $X$ is geometrically punctually integral at a point $x$, there exists an open neighbourhood $U$ of $x$ which is geometrically punctually integral.
In other words, the set of $x\in X$ at which $X$ is geometrically punctually integral is open in $X$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item Given an algebraically closed extension $\Omega$ of $k$, $X_{(\Omega)}$ is punctually integral (or locally integral, which amounts to the same since $X_{(\Omega)}$ is Noetherian) if and only if it is reduced and the number of its connected components is equal to the number of its irreducible components \sref[I]{I.6.1.10}.
It then suffices to apply \sref{IV.4.5.1}, \sref{IV.4.6.1}, and the first remark of \sref{IV.4.6.14}.
  \item The question being local on $X$, we may restrict to the case where $X$ is a prescheme of finite type over $k$.
Furthermore, we know that there is an open neighbourhood of $x$ in $X$ which is geometrically reduced \sref{IV.4.6.13}, so we may suppose $X$ geometrically reduced.
There exists then a finite extension $k'$ of $k$ such that $X'=X\otimes_k k'$ is reduced and its irreducible components are geometrically irreducible \sref{IV.4.6.8}.
Let $p:X'\to X$ be the canonical projection, which is a surjective finite morphism, and let $x'_j$ ($1\leq j\leq r$) be the points of $p^{-1}(x)$; by hypothesis, $X'$ is integral at each of the $x'_j$, so each $x'_j$ has an open neighbourhood $V'_j$ in $X'$ which is integral.
Since $p$ is closed \sref[II]{II.6.1.10}, each $p(V'_j)$ is a neighbourhood of $x$, and hence there exists an open neighbourhood $U$ of $x$ such that $p^{-1}(U)$ is contained in the union of the $V'_j$, and is therefore integral at each of its points, hence locally integral.
The irreducible components of $p^{-1}(U)$ are open, pairwise disjoint, and since they are geometrically irreducible \sref{IV.4.5.9}, $p^{-1}(U) = U\otimes_k k'$ is geometrically punctually integral by virtue of~(i), and so it is the same for $U$ by definition.
\end{enumerate}
\end{proof}

\begin{proposition}[4.6.16]
\label{IV.4.6.16}
Let $k$ be a field, $X$ a $k$-prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item For every extension of finite type $k'$ of $k$, if we set $X'=X\otimes_k k'$, then $\sh{F}'=\sh{F}\otimes_k k'$ is a reduced $\sh{O}_{X'}$-Module \sref{IV.3.2.2}.
  \item For every finite radicial extension $k'$ of $k$, $\sh{F}'$ is a reduced $\sh{O}_{X'}$-Module.
  \item $\sh{F}$ is reduced, and if $\sh{J}$ is the Ideal of $\sh{O}_X$ annihilating $\sh{F}$, the closed sub-prescheme defined by $\sh{J}$ is geometrically reduced over $k$.
\end{enumerate}
\end{proposition}

\oldpage[IV-2]{73}

\begin{proof}
It is clear that $a)$ implies $b)$.
The condition $b)$ implies first that $\sh{F}$ is reduced: this means \sref{IV.3.2.3} that if $Y$ is the closed sub-prescheme of $X$ defined by the Ideal $\sh{J}$ coherent of $\sh{O}_X$, annihilator of $\sh{F}$, $Y$ is reduced, and there is an $\sh{O}_Y$-Module coherent without torsion $\sh{G}$ of rank $1$ on every irreducible component of $Y$, such that $j_*(\sh{G})=\sh{F}$, $j:Y\to X$ being the canonical injection.
Now, for every extension $k'$ of $k$, the annihilator of $\sh{F}'$ is $\sh{J}'=\sh{J}\otimes_k k'$, the morphism projection $X'\to X$ being flat \sref{IV.2.1.11}; the closed sub-prescheme of $X'$ defined by $\sh{J}'$ is $Y'=Y\otimes_k k'$, and if $j':Y'\to X'$ is the canonical injection, then $\sh{F}'=j'_*(\sh{G}')$.
Note also that, for every extension $k'$ of $k$ such that $X'\otimes_k k'$ is locally Noetherian, $\sh{F}'$ is without associated immersed prime cycle \sref{IV.4.2.7}.
Furthermore, every maximal point $y$ of $Y$ is isomorphic to $\sh{O}_{Y,y}$, $\sh{G}_y$ isomorphic to $\sh{O}_{Y,y}$ at every maximal point $y$ of $Y$; since $\sh{G}_y$ is isomorphic to $\sh{O}_{Y,y}$ at every maximal point $y$ of $Y$, as it amounts to the same to say that $\sh{F}'$ is reduced or that the prescheme $Y'$ is reduced, when $X'$ is locally Noetherian.
The fact that $b)$ implies $c)$ and that $c)$ implies $a)$ is therefore a consequence of \sref{IV.4.6.1},~$d)$ and~$b)$); in~$a)$, the hypothesis that $k'$ is an extension of finite type of $k$ is not made solely to ensure that $X'$ is locally Noetherian.
When $X$ is locally of finite type over $k$, $X$ is locally Noetherian for every extension $k'$ of $k$, which proves in this case the equivalence of~$d)$ and the other conditions.
\end{proof}

\begin{definition}[4.6.17]
\label{IV.4.6.17}
When $\sh{F}$ satisfies the equivalent conditions of \sref{IV.4.6.16}, we say that $\sh{F}$ is \emph{geometrically reduced or separable over $k$}.
\end{definition}

\begin{proposition}[4.6.18]
\label{IV.4.6.18}
Let $k$ be a field, $X$ a $k$-prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item For every extension of finite type $k'$ of $k$, if we set $X'=X\otimes_k k'$, then $\sh{F}'=\sh{F}\otimes_k k'$ is an integral $\sh{O}_{X'}$-Module \sref{IV.3.2.4}.
  \item For every finite radicial extension $k'$ of $k$, $\sh{F}'$ is an integral $\sh{O}_{X'}$-Module.
  \item $\sh{F}$ is reduced (or integral), and if $\sh{J}$ is the Ideal of $\sh{O}_X$ annihilating $\sh{F}$, the closed sub-prescheme defined by $\sh{J}$ is geometrically integral over $k$.
\end{enumerate}
Furthermore, if $X$ is locally of finite type over $k$, these conditions are also equivalent to the following:
\begin{enumerate}
  \item[\emph{d)}] For every extension $k'$ of $k$ (or for an algebraically closed extension $k'$ of $k$), $\sh{F}'$ is an integral $\sh{O}_{X'}$-Module.
\end{enumerate}
\end{proposition}

\begin{proof}
The conditions $a)$, $b)$, and $c)$ imply that $\sh{F}$ is integral, hence $\operatorname{Ass}(\sh{F})$ is reduced to a single point $x$, which is the generic point of $\operatorname{Supp}(\sh{F})$.
For every extension $k'$ of $k$ for which $X'$ is locally Noetherian, the prime cycles associated to $\sh{F}'$ are therefore non-immersed, and are the maximal points of $\operatorname{Supp}(\sh{F}')$, which are all above $x$.
On the other hand, by virtue of \sref{IV.4.6.16}, each of the conditions $a)$, $b)$, $c)$ implies that $\sh{F}$ is geometrically reduced; it is trivial that $a)$ implies $b)$, and $b)$ implies that $\operatorname{Supp}(\sh{F})$ is geometrically irreducible \sref{IV.4.5.9}, hence $c)$, since the closed sub-prescheme $Y$ defined
\oldpage[IV-2]{74}
by $\sh{J}$ is geometrically reduced and geometrically irreducible, hence geometrically integral.
Conversely, if $Y$ is geometrically integral, for \emph{every} extension $k'$ of $k$, $\operatorname{Supp}(\sh{F}')$ is irreducible, and hence $\sh{F}'$ is integral; this shows that $c)$ implies $a)$, and that $c)$ implies $d)$ when $X$ is locally of finite type over $k$.
It is trivial that $a)$ implies $b)$, and $b)$ implies that $\operatorname{Supp}(\sh{F})$ is geometrically irreducible \sref{IV.4.5.9}, hence $c)$.
\end{proof}

\begin{definition}[4.6.19]
\label{IV.4.6.19}
When $\sh{F}$ satisfies the equivalent conditions of \sref{IV.4.6.18}, we say that $\sh{F}$ is \emph{geometrically integral over $k$}.
\end{definition}

\begin{proposition}[4.6.20]
\label{IV.4.6.20}
Let $k$ be a field, $X$ a $k$-prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent.
Let $K$ be an extension of $k$ such that $X\otimes_k K$ is locally Noetherian.
Then, if $\sh{F}$ is geometrically reduced (resp.\ geometrically integral) over $k$, $\sh{F}\otimes_k K$ is geometrically reduced (resp.\ geometrically integral) over $K$.
\end{proposition}

\begin{proof}
Indeed, for every finite extension $K'$ of $K$, $(X\otimes_k K)\otimes_K K' = X\otimes_k K'$ is locally Noetherian, and we have seen in the proofs of \sref{IV.4.6.16} and \sref{IV.4.6.18} that the hypothesis on $\sh{F}$ implies that $\sh{F}\otimes_k K'$ is reduced (resp.\ integral), whence the conclusion.
\end{proof}

\begin{proposition}[4.6.21]
\label{IV.4.6.21}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes locally Noetherian such that $X\times_k Y$ is locally Noetherian.
Let $\sh{F}$ be an $\sh{O}_X$-Module coherent, $\sh{G}$ an $\sh{O}_Y$-Module coherent.
\begin{enumerate}[label=(\roman*)]
  \item If $\sh{F}$ is geometrically reduced (resp.\ geometrically integral) and $\sh{G}$ reduced (resp.\ integral), then $\sh{F}\otimes_k\sh{G}$ is reduced (resp.\ integral).
  \item If $\sh{F}$ and $\sh{G}$ are both geometrically reduced (resp.\ geometrically integral), so is $\sh{F}\otimes_k\sh{G}$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item Using \sref{IV.3.2.3}, \sref{IV.4.6.16}, and \sref{IV.4.6.18}, we may restrict to the case where $\operatorname{Supp}(\sh{F})=X$, $\operatorname{Supp}(\sh{G})=Y$, $X$ being geometrically reduced (resp.\ geometrically integral), $Y$ reduced (resp.\ geometrically integral), $\sh{F}$ and $\sh{G}$ without associated immersed prime cycles, and $\sh{F}_x$ being isomorphic to $\sh{O}_x$ at every maximal point $x$ of $X$, $\sh{G}_y$ isomorphic to $\sh{O}_y$ at every maximal point $y$ of $Y$.
We know then \sref{IV.4.2.3} that $\sh{F}\otimes_k\sh{G}$ is without associated immersed prime cycle, and the prime cycles associated to $\sh{F}\otimes_k\sh{G}$ are exactly the irreducible components of $X\times_k Y$ \sref{IV.4.2.5}.
We may therefore restrict to the case where $X$ and $Y$ are irreducible; then, if $X$ is geometrically reduced and $Y$ reduced, we know \sref{IV.4.2.4},~(ii)) that $X\times_k Y$ is reduced; if $X$ is geometrically integral and $Y$ integral, we know that $X\times_k Y$ is furthermore irreducible \sref{IV.4.5.8},~(i)), hence integral.
Finally, every maximal point $z$ of $X\times_k Y$ is above $x$ and $y$, so the hypotheses on $\sh{F}$ and $\sh{G}$ imply, by virtue of \sref[I]{I.9.1.12}, that $(\sh{F}\otimes_k\sh{G})_z$ is isomorphic to $\sh{O}_z$, which completes the proof in this case.
  \item The argument is analogous, but here (after reduction to the case where $X=\operatorname{Supp}(\sh{F})$, $Y=\operatorname{Supp}(\sh{G})$), $X\times_k Y$ is geometrically reduced (resp.\ geometrically integral) by virtue of \sref{IV.4.6.5},~(ii)).
\end{enumerate}
\end{proof}

The definitions \sref{IV.4.6.18} and \sref{IV.4.6.19} are localised as follows:

\begin{definition}[4.6.22]
\label{IV.4.6.22}
Let $k$ be a field, $X$ a $k$-prescheme locally Noetherian, $\sh{F}$ an $\sh{O}_X$-Module coherent.
We say that $\sh{F}$ is \emph{geometrically reduced or separable} (resp.\ \emph{geometrically punctually integral}) \emph{at a point} $x\in X$ if, for every finite radicial (resp.\ finite) extension $k'$ of $k$, $\sh{F}\otimes_k k'$ is reduced (resp.\ integral) at each of the points $x'$ of $X\otimes_k k'$ above $x$ (cf.\ \sref{IV.3.2.2} and \sref{IV.3.2.4}).
\end{definition}

\oldpage[IV-2]{75}

If $\sh{F}$ is reduced at $x$, there is an open neighbourhood $U$ of $x$ such that $\sh{F}|U$ is reduced \sref{IV.3.2.2}.
Reasoning as in \sref{IV.4.6.16}, we see that saying that $\sh{F}$ is geometrically reduced at the point $x$ is equivalent to saying that $\sh{F}$ is reduced at the point $x$ and that the closed sub-prescheme $Y$ defined by the annihilator $\sh{J}$ of $\sh{F}$ is \emph{geometrically reduced at the point} $x$; it follows that there is an open neighbourhood $U$ of $x$ such that $\sh{F}|U$ is geometrically reduced \sref{IV.4.6.11}.
The same arguments, using \sref{IV.4.6.16} and \sref{IV.4.6.18}, show that saying $\sh{F}$ is geometrically punctually integral at the point $x$ is equivalent to saying that $\sh{F}$ is reduced at the point $x$ and that the closed sub-prescheme $Y$ is \emph{geometrically punctually integral at the point} $x$.


\subsection{Multiplicities in the primary decomposition on an algebraic prescheme}
\label{subsection:IV.4.7}

\begin{lemma}[4.7.1]
\label{IV.4.7.1}
Let $A$, $B$ be two local rings, $\mathfrak{m}$ the maximal ideal of $A$, $\rho:A\to B$ a local homomorphism.
Suppose that $B$ is a flat $A$-module, and $B/\mathfrak{m}B$ a $B$-module of finite length.
Then, for an $A$-module $M$ to be of finite length, it is necessary and sufficient that $M_{(B)}$ be a $B$-module of finite length, and we have
\[
\tag{4.7.1.1}
  \operatorname{length}_B(M_{(B)}) = \operatorname{length}_A(M)\cdot\operatorname{length}_B(B/\mathfrak{m}B).
\]
\end{lemma}

\begin{proof}
This is a special case of \sref{IV.2.5.5.2}.
\end{proof}

\begin{corollary}[4.7.2]
\label{IV.4.7.2}
Let $A$, $B$ be two Noetherian rings, $\rho:A\to B$ a homomorphism making $B$ a flat $A$-module.
Let $\mathfrak{q}$ be a minimal prime ideal of $B$, $\mathfrak{p}=\rho^{-1}(\mathfrak{q})$, $M$ an $A$-module of finite type; then $\mathfrak{p}$ is a minimal prime ideal of $A$, $M_\mathfrak{p}$ is an $A_\mathfrak{p}$-module of finite length, and we have
\[
\tag{4.7.2.1}
  \operatorname{length}_{B_\mathfrak{q}}(M_{(B)})_\mathfrak{q} = (\operatorname{length}_{A_\mathfrak{p}}(M_\mathfrak{p}))(\operatorname{length}_{B_\mathfrak{q}}(B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q})).
\]
\end{corollary}

\begin{proof}
Indeed, $B_\mathfrak{q}$ is a flat $A_\mathfrak{p}$-module $(\mathbf{0}_I,\,6.3.2)$ and the homomorphism $A_\mathfrak{p}\to B_\mathfrak{q}$ is local, hence $B_\mathfrak{q}$ is a faithfully flat $A_\mathfrak{p}$-module $(\mathbf{0}_I,\,6.6.2)$; the fact that $\mathfrak{p}$ is minimal follows from $A$ \sref{IV.2.3.5}.
Furthermore, $A_\mathfrak{p}$ is then an Artinian ring (Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \S~2, n\textsuperscript{o}~5, prop.~9) and $M_\mathfrak{p}$ is therefore an $A_\mathfrak{p}$-module of finite length; the formula \sref{IV.4.7.2.1} is therefore a special case of \sref{IV.4.7.1.1}.
\end{proof}

\begin{proposition}[4.7.3]
\label{IV.4.7.3}
Let $k$ be a field of characteristic exponent $p$, $X$ an integral locally Noetherian prescheme, $K=\mathrm{R}(X)$ the field of rational functions on $X$.
Let $k'$ be an extension of $k$, $(X'_\alpha)$ the family of irreducible components of $X'=X_{(k')}$, $x'_\alpha$ the generic point of $X'_\alpha$.
\begin{enumerate}[label=(\roman*)]
  \item We suppose that $X'$ is locally Noetherian (which holds if $X$ is locally of finite type over $k$, or if $k'$ is an extension of finite type of $k$).
Let $k''$ be any extension of $k'$, $X''=X\otimes_k k''$, $(X''_\beta)$ the family of irreducible components of $X''$, $x''_\beta$ the generic point of $X''_\beta$.
If $x''_\beta$ is above $x'_\alpha$, we have
\[
\tag{4.7.3.1}
  \operatorname{length}(\sh{O}_{x''_\beta}) = \operatorname{length}(\sh{O}_{x'_\alpha})\cdot\operatorname{length}(\sh{O}_{X'',x''_\beta})
\]
where we have set $Z''=(X'_{\mathrm{red}})\otimes_{k'} k''$.
In particular, if $k''$ is a separable extension of $k'$ such that $Z''$ is locally Noetherian, we have $\operatorname{length}(\sh{O}_{x''_\beta}) = \operatorname{length}(\sh{O}_{x'_\alpha})$.
\oldpage[IV-2]{76}
  \item If $X$ is locally of finite type over $k$, or if $k'$ is an extension of finite type of $k$, the numbers $\operatorname{length}(\sh{O}_{x'_\alpha})$ are powers of $p$.
  \item We suppose that $X$ is locally of finite type over $k$.
Then, if $k'$ is perfect, all the numbers $\operatorname{length}(\sh{O}_{x'_\alpha})$ are equal and do not depend on the perfect extension $K$ of $k$ (and not on the extension $k'$ considered).
Furthermore, there exists a finite radicial extension $k_1$ of $k$ such that, if $x_1$ is the generic point of $X_1=X\otimes_k k_1$ (which is irreducible), we have $\operatorname{length}(\sh{O}_{x'_\alpha}) = \operatorname{length}(\sh{O}_{x_1})$ for all $\alpha$.
\end{enumerate}
\end{proposition}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item Recall \sref{IV.4.2.4} that the irreducible components $X'_\alpha$, which are finite in number, correspond bijectively to the minimal prime ideals $\mathfrak{p}_\alpha$ of $K\otimes_k k'$, and that $\sh{O}_{x'_\alpha}$ is an Artinian ring, isomorphic to $(K\otimes_k k')_{\mathfrak{p}_\alpha}$; these rings therefore depend only on the extensions $K$ and $k'$ of $k$ (and are consequently the same for all integral $k$-preschemes having the same field of rational functions).
The formula \sref{IV.4.7.3.1} follows from \sref{IV.4.7.2.1} applied to the Noetherian ring $A$ of an affine neighbourhood of $x'_\alpha$ in $X'$, and the ring $B$ of an affine neighbourhood of $x''_\beta$ in $X''$, with $\mathfrak{p}$ and $\mathfrak{q}$ being the minimal prime ideals corresponding to $x'_\alpha$ and $x''_\beta$ respectively, and taking $M=A$: we will note that $A/\mathfrak{p}$ is the ring of an affine neighbourhood of $x'_\alpha$ in $X'_{\mathrm{red}}$, and that $B=A\otimes_{k'} k''$; if $q'=q/\mathfrak{p}B$, a minimal prime ideal of $B/\mathfrak{p}B$, we have $\sh{O}_{X'',x''_\beta} = (B/\mathfrak{p}B)_{q'} = B_q/\mathfrak{p}B_q$, whence \sref{IV.4.7.3.1}.
The last assertion of~(i) follows from the fact that in this case $Z''$ is reduced \sref{IV.4.2.4}, and hence $\mathfrak{p}B_q = qB_q$.
  \item If $k'$ is an extension of finite type of $k$, there is a transcendental pure extension $k'_0$ of $k$ such that $k'$ is a finite extension of $k'_0$; since $k'_0$ is separable over $k$, it follows from~(i) applied replacing $k'$ and $k''$ by $k$ and $k'_0$ respectively, that we may replace $X$ by $X\otimes_k k'_0$, in other words suppose that $k'$ is a \emph{finite} extension of $k$.
Suppose then that $X$ is locally of finite type over $k$; then, $X''$ is locally Noetherian for every extension $k''$ of $k'$.
Taking for $k''$ an algebraically closed extension of $k'$, it follows from~(i) that we are reduced, by virtue of \sref{IV.4.7.3.1}, to proving the assertion when $k'$ is moreover \emph{radicial}.
We know then that $K\otimes_k k_1$ is an Artinian ring having only a single prime ideal (necessarily maximal), and its quotient by that ideal is necessarily equal to $\overline{K}$; the length of $\overline{K}\otimes_k k_1$ divides that of $K\otimes_k k_1$ by virtue of \sref{IV.4.7.1.1}.
But $\overline{K}\otimes_k k_1$ is a $\overline{K}$-algebra, Artinian, having only a single prime ideal (necessarily maximal), and its quotient by that ideal is an extension finite of $\overline{K}$, hence necessarily equal to $\overline{K}$; the length of $\overline{K}\otimes_k k_1$ is therefore equal to its rank over $\overline{K}$, that is to say $[k_1:k]$, which is a power of $p$.

Suppose then that $X$ is locally of finite type over $k$; then, $X''$ is locally Noetherian for every extension $k''$ of $k'$.
Taking for $k''$ an algebraically closed extension of $k'$, we are reduced by virtue of \sref{IV.4.7.3.1} to proving the assertion when $k'$ is \emph{algebraically closed}.
Since $k'$ is then a separable extension of the algebraic closure $\bar{k}$ of $k$, it follows from~(i) that we are reduced to the case where $k'=\bar{k}$, the algebraic closure.
Now, we know (by replacing $X$ by an affine open of finite type over $k$) that there is a finite radicial extension $k_1$ of $k$ such that $(X\otimes_k k_1)_{\mathrm{red}}$ is separable over $k_1$ \sref{IV.4.6.6}; hence $(X\otimes_k k_1)_{\mathrm{red}}\otimes_{k_1}\bar{k}$ is reduced \sref{IV.4.6.4}.
\oldpage[IV-2]{77}
By virtue of \sref{IV.4.7.3.1}, we are therefore reduced again to the case where $k'$ is a finite radicial extension of $k$, and we conclude as before.
  \item Let us note that two extensions $k'$, $k''$ of $k$ are contained in the same algebraically closed extension of $k$; to show that all the numbers $\operatorname{length}(\sh{O}_{x'_\alpha})$ are equal, for $k'$ perfect, we may therefore restrict to the case where $k'=\bar{k}$, the algebraic closure of $k$.
To show that all the numbers $\operatorname{length}(\sh{O}_{x'_\alpha})$ are equal, we may restrict to the case where $k'=\bar{k}$, the algebraic closure of $k$.
Now, the finite radicial extension $k_1$ of $k$ being then determined as in~(ii), the $x'_\alpha$ are all above the unique generic point of $X_1 = X\otimes_k k_1$, and the relation $\operatorname{length}(\sh{O}_{x'_{k_1}}) = \operatorname{length}(\sh{O}_{x_1})$ for all $\alpha$ follows again from \sref{IV.4.7.3.1}, taking into account the choice of $k_1$.
\end{enumerate}
\end{proof}

\begin{definition}[4.7.4]
\label{IV.4.7.4}
Let $k$ be a field, $X$ a $k$-prescheme integral, $K$ the field of rational functions on $X$.
The \emph{radicial multiplicity} of $K$ (or of $X$) over $k$ is defined as the upper bound of the lengths of the Artinian local rings $K\otimes_k k_1$, where $k_1$ ranges over the set of finite radicial extensions of $k$.
The \emph{separable multiplicity} of $K$ (or of $X$) over $k$ is defined as the number of geometric irreducible components of $X$ if this number is finite, and $+\infty$ otherwise.
Finally, the \emph{total multiplicity} of $K$ (or of $X$) over $k$ is defined as the product of the radicial multiplicity and the separable multiplicity.
\end{definition}

We note that if $p$ is the characteristic exponent of $k$ and if the radicial multiplicity of $K$ over $k$ is finite, it is a power $q=p^f$ of $p$, as follows from \sref{IV.4.7.3},~(ii)); when $p\neq 1$, $f$ is well-defined and called the \emph{exponent of inseparability} of $K$ over $k$; when $p=1$, the radicial multiplicity is always equal to $1$.

To say that the radicial multiplicity (resp.\ separable, resp.\ total) of $X$ over $k$ is $1$ means that $X$ is geometrically reduced (resp.\ geometrically irreducible, resp.\ geometrically integral) over $k$.

When $X$ is locally of finite type, the radicial multiplicity of $X$ over $k$ is also the common length of the local rings $A_i$ at the minimal prime ideals of the total ring of fractions $A$ of $K\otimes_k k'$, for every perfect extension $k'$ of $k$; the separable multiplicity of $X$ over $k$ is the number of minimal prime ideals of $A=K\otimes_k\Omega$ for every algebraically closed extension $\Omega$ of $k$, and finally the total multiplicity is the sum of the lengths of the local rings $A_i$ at the minimal prime ideals of $A=K\otimes_k\Omega$; these numbers are all three finite in this case.

\begin{definition}[4.7.5]
\label{IV.4.7.5}
Let $k$ be a field, $X$ a $k$-prescheme locally of finite type over $k$, $\sh{F}$ an $\sh{O}_X$-Module coherent, $x$ a point of $X$ such that $\sh{F}_x$ is an $\sh{O}_x$-module of finite length.
The \emph{geometric length} of $\sh{F}$ at $x$ (relative to $k$), or \emph{radicial multiplicity of $x$ for} $\sh{F}$, is defined as the product $\lambda_x(\sh{F}) = (\operatorname{length}_{\sh{O}_x}(\sh{F}_x))\mu_i(\kappa(x)\,|\,k)$, where $\mu_i(\kappa(x)\,|\,k)$ denotes the radicial multiplicity of $\kappa(x)$ over $k$.
\end{definition}

When $X$ is integral, and $x$ is the generic point of $X$, we have $\operatorname{length}(\sh{O}_x)=1$, and hence the radicial multiplicity of $x$ for $\sh{O}_X$ is none other than the radicial multiplicity of $X$ over $k$, as defined in \sref{IV.4.7.4}.

\begin{proposition}[4.7.6]
\label{IV.4.7.6}
Let $0\to\sh{F}'\to\sh{F}\to\sh{F}''\to 0$ be an exact sequence of $\sh{O}_X$-Modules coherent.
If at a point $x\in X$ the geometric length $\lambda_x(\sh{F})$ of $\sh{F}$ at this point is defined, the same is true of the geometric lengths $\lambda_x(\sh{F}')$ and $\lambda_x(\sh{F}'')$, of $\sh{F}'$ and $\sh{F}''$ at the point $x$, and conversely; furthermore, we have
\[
\tag{4.7.6.1}
  \lambda_x(\sh{F}) = \lambda_x(\sh{F}') + \lambda_x(\sh{F}'').
\]
\end{proposition}

\begin{proof}
The proposition follows immediately from the definition, since $\operatorname{length}(\sh{F}_x)$ is finite if and only if $\operatorname{length}(\sh{F}'_x)$ and $\operatorname{length}(\sh{F}''_x)$ are finite, and then $\operatorname{length}(\sh{F}_x) = \operatorname{length}(\sh{F}'_x) + \operatorname{length}(\sh{F}''_x)$.
\end{proof}

\begin{env}[4.7.7]
\label{IV.4.7.7}
Under the hypotheses of \sref{IV.4.7.5}, the only points $x\in X$ such that $\sh{F}_x$ is a non-zero $\sh{O}_x$-module of finite length are (by virtue of \sref{IV.3.1.2}) the \emph{maximal points} of $\operatorname{Supp}(\sh{F})$, as follows from Bourbaki, \emph{Alg.\ comm.}, chap.~IV, \S~2, n\textsuperscript{o}~5, cor.~2 of prop.~7, the question being local on $X$.
We know that there exists a closed sub-prescheme $Y$ having $\operatorname{Supp}(\sh{F})$ as underlying space, and an $\sh{O}_Y$-Module coherent $\sh{G}$ such that $\sh{F}=j_*(\sh{G})$, $j:Y\to X$ being the canonical injection.
If $x$ is a maximal point of $Y$, we see that the geometric lengths of $\sh{F}$ and $\sh{G}$ at $x$ are equal, $\sh{O}_{X,x}$ and $\sh{O}_{Y,x}$ having the same residue body.
We deduce from this the following characterisation of the geometric length:
\end{env}

\begin{proposition}[4.7.8]
\label{IV.4.7.8}
Under the hypotheses of \sref{IV.4.7.5}, let $x$ be a maximal point of $\operatorname{Supp}(\sh{F})$.
Let $k'$ be a perfect extension of $k$; set $X'=X\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$; then the geometric length $\lambda_x(\sh{F})$ of $\sh{F}$ at $x$ is equal to the length of the $\sh{O}_{x'}$-module $\sh{F}'_{x'}$ at every maximal point $x'$ of $\operatorname{Supp}(\sh{F}')$ above $x$.
Furthermore, there exists a finite radicial extension $k_1$ of $k$ such that, setting $X_1=X\otimes_k k_1$, $\sh{F}_1=\sh{F}\otimes_k k_1$, $\lambda_x(\sh{F})$ is equal to the length of the $\sh{O}_{x_1}$-module $(\sh{F}_1)_{x_1}$ at every maximal point $x_1$ of $\operatorname{Supp}(\sh{F}_1)$ above $x$.
\end{proposition}

\begin{proof}
The remark of \sref{IV.4.7.7} shows that we may restrict to the case where $x$ is a maximal point of $X$.
Furthermore, the question is local on $X$ and we may suppose $X=\Spec(A)$ affine Noetherian, $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module of finite type; let $\mathfrak{p}$ (resp.\ $\mathfrak{q}$) be the minimal prime ideal of $A$ (resp.\ $B$) corresponding to $x$ (resp.\ $x'$).
We have then, applying the formula \sref{IV.4.7.2.1},
\[
  \operatorname{length}_{\sh{O}_{x'}}(\sh{F}'_{x'}) = \operatorname{length}_{\sh{O}_x}(\sh{F}_x)\cdot\operatorname{length}_{B_\mathfrak{q}}(B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q}).
\]
But since $\kappa(x) = A_\mathfrak{p}/\mathfrak{p}A_\mathfrak{p}$, the same reasoning as in \sref{IV.4.7.3} shows that $\operatorname{length}_{B_\mathfrak{q}}(B_\mathfrak{q}/\mathfrak{p}B_\mathfrak{q})$ is equal to the length of the local ring $(B/\mathfrak{p}B)_{q'}$, where $q'=q/\mathfrak{p}B$; but $B/\mathfrak{p}B$ is a local ring of the ring $(A/\mathfrak{p})\otimes_k k'$ for a minimal prime ideal, hence also a local ring of the total ring of fractions of $\kappa(x)\otimes_k k'$, and the length of such a ring is by definition $\mu_i(\kappa(x)\,|\,k)$, whence the first assertion.
The second is proved in the same way, replacing $k'$ by $k_1$ and using \sref{IV.4.7.3},~(iii)).
\end{proof}

We also say, in the case where $x$ is a maximal point of $\operatorname{Supp}(\sh{F})$, that the geometric length of $\sh{F}$ at $x$ is the \emph{radicial multiplicity of the prime cycle} $\overline{\{x\}}$ associated to $\sh{F}$.

\begin{corollary}[4.7.9]
\label{IV.4.7.9}
Under the hypotheses of \sref{IV.4.7.5}, let $k'$ be an extension of $k$, $X'=X\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$; let $Z$ be an irreducible component of $\operatorname{Supp}(\sh{F})$, and $Z'$ an irreducible component of $\operatorname{Supp}(\sh{F}')$ that dominates $Z$ \sref{IV.4.2.7}.
Then the radicial multiplicities of $Z$ and $Z'$ with respect to $\sh{F}$ and $\sh{F}'$ respectively are the same.
\end{corollary}

\begin{proof}
It suffices to consider a perfect extension $k''$ of $k'$ and, setting $X''=X\otimes_k k''$, $\sh{F}''=\sh{F}\otimes_k k''$, a maximal point $x''$ of $\operatorname{Supp}(\sh{F}'')$ which is above the generic point of $Z$ and $Z'$; it follows from \sref{IV.4.7.8} that $\operatorname{length}(\sh{F}''_{x''})$ is equal to the multiplicities of $Z$ and of $Z'$.
\end{proof}

\oldpage[IV-2]{79}

\begin{proposition}[4.7.10]
\label{IV.4.7.10}
Let $k$ be a field, $X$ a $k$-prescheme locally of finite type over $k$, $\sh{F}$ an $\sh{O}_X$-Module coherent, $x$ a point of $X$, $z_i$ the generic points of the irreducible components of $\operatorname{Supp}(\sh{F})$ that contain $x$.
For $\sh{F}$ to be geometrically reduced at the point $x$ \sref{IV.4.6.22}, it is necessary and sufficient that $x$ does not belong to any associated immersed prime cycle of $\sh{F}$ and that, for all $i$, the geometric length $\lambda_{z_i}(\sh{F})$ of $\sh{F}$ at the point $z_i$ be equal to $1$.
\end{proposition}

\begin{proof}
Let us first show that the conditions of the statement are sufficient.
Indeed, if $k'$ is a finite extension of $k$, $x'$ a point of $X'=X\otimes_k k'$ above $x$, and $\sh{F}'=\sh{F}\otimes_k k'$, $x'$ does not belong to any associated immersed prime cycle of $\sh{F}'$ by virtue of \sref{IV.4.2.7}; furthermore, by virtue of \sref{IV.4.7.8}, the geometric length of $\sh{F}'$ is equal to $1$ at every generic point $z'_j$ of an irreducible component of $\operatorname{Supp}(\sh{F}')$ containing $x'$, a point which is above one of the $z_i$ by virtue of \sref{IV.2.3.4}; \emph{a fortiori} we have $\operatorname{length}(\sh{F}'_{z'_j})=1$, which proves that $\sh{F}'$ is reduced at the point $x'$.

Conversely, it follows first from \sref{IV.4.2.7} that if, for \emph{one} finite extension $k'$ of $k$, $\sh{F}'$ is reduced at all the points $x'$ of $X'$ above $x$, then $x$ does not belong to any associated immersed prime cycle of $\sh{F}$.
Furthermore, taking for $k'$ the field $k_1$ of the statement of \sref{IV.4.7.8} (where $x$ is replaced by one of the $z_i$), the hypothesis that $\sh{F}$ is geometrically reduced at the point $x$ implies that $\lambda_{z_i}(\sh{F})=1$.
\end{proof}

\begin{proposition}[4.7.11]
\label{IV.4.7.11}
Let $k$ be a field, $X$ a $k$-prescheme locally of finite type over $k$, $\sh{F}$ an $\sh{O}_X$-Module coherent, $k'$ an extension of $k$, $X'=X\otimes_k k'$, $\sh{F}'=\sh{F}\otimes_k k'$, $x$ a point of $X$, $x'$ a point of $X'$ above $x$.
For $\sh{F}$ to be geometrically punctually reduced (resp.\ geometrically punctually integral) at the point $x$, it is necessary and sufficient that $\sh{F}'$ be geometrically punctually reduced (resp.\ geometrically punctually integral) at the point $x'$.
\end{proposition}

\begin{proof}
The irreducible components of $\operatorname{Supp}(\sh{F}')$ containing $x'$ each dominate an irreducible component of $\operatorname{Supp}(\sh{F})$ containing $x$, and conversely each of the latter is dominated by an irreducible component of $\operatorname{Supp}(\sh{F}')$ containing $x'$ (\sref{IV.4.2.7},~(i)) and \sref{IV.2.3.4}).
The assertion concerning separability follows therefore from \sref{IV.4.2.7}, \sref{IV.4.7.9}, and \sref{IV.4.7.10}.
Let $Y$ be the closed sub-prescheme of $X$ defined by the Ideal $\sh{J}$ annihilator of $\sh{F}$, so that $Y'=Y\otimes_k k'$ is the closed sub-prescheme of $X'$ defined by the Ideal $\sh{J}'=\sh{J}\otimes_k k'$ annihilator of $\sh{F}'$; saying that $\sh{F}$ (resp.\ $\sh{F}'$) is geometrically punctually reduced at the point $x$ (resp.\ $x'$) is the same as saying that $\sh{F}$ (resp.\ $\sh{F}'$) is geometrically punctually reduced at the point $x$ (resp.\ $x'$) and that $Y$ (resp.\ $Y'$) is geometrically punctually integral at the point $x$ (resp.\ $x'$).
By virtue of the assertion concerning separability, the hypothesis that $\sh{F}$ is geometrically punctually integral at the point $x$, or the hypothesis that $\sh{F}'$ is geometrically punctually integral at the point $x'$, both imply that there exists in $Y$ a neighbourhood of $x$ which is separable, and the conclusion follows from \sref{IV.4.6.10}.
\end{proof}

\begin{definition}[4.7.12]
\label{IV.4.7.12}
Let $k$ be a field, $X$ a $k$-prescheme locally of finite type over $k$, $\sh{F}$ an $\sh{O}_X$-Module coherent, $x$ a maximal point of the support of $\sh{F}$.
\oldpage[IV-2]{80}
The \emph{total multiplicity} of $x$ (or of the prime cycle $\overline{\{x\}}$) for $\sh{F}$ (relative to $k$) is defined as the product of the radicial multiplicity of $x$ for $\sh{F}$ by the separable multiplicity of $\kappa(x)$ over $k$ \sref{IV.4.7.4}.
\end{definition}

It amounts to the same to say that the total multiplicity of $x$ for $\sh{F}$ is the product of the \emph{length} $\operatorname{length}_{\sh{O}_x}(\sh{F}_x)$ by the total multiplicity of $\kappa(x)$ over $k$ \sref{IV.4.7.4}.
With the notation of \sref{IV.4.7.11}, the total multiplicity of $x$ for $\sh{F}$ is equal to the \emph{sum} of the total multiplicities for $\sh{F}'$ of the maximal points $x'_i$ of $\operatorname{Supp}(\sh{F}')$ which are above $x$; furthermore, there exists a \emph{finite} extension $k'$ of $k$, such that the total multiplicity of $x$ for $\sh{F}$ be equal to $\sum_i \operatorname{length}(\sh{F}'_{x'_i})$.

\begin{proposition}[4.7.13]
\label{IV.4.7.13}
The hypotheses and notation being those of \sref{IV.4.7.10}, for $\sh{F}$ to be geometrically punctually integral at the point $x$, it suffices that $x$ does not belong to any associated (necessarily non-immersed) prime cycle of $\sh{F}$ and that the total multiplicity for $\sh{F}$ of the generic point $z$ of this cycle be equal to $1$.
\end{proposition}

\begin{proof}
Indeed, if $k'$ is a finite extension of $k$, $x'$ a point of $X'=X\otimes_k k'$ above $x$ and $\sh{F}'=\sh{F}\otimes_k k'$ above $x$, $x'$ cannot belong to more than a single prime cycle associated to $\sh{F}$, since there is only a single maximal point of $\operatorname{Supp}(\sh{F}')$ above $x$; furthermore, $\sh{F}$ is geometrically reduced at the point $x$ by hypothesis; in addition, by \sref{IV.4.7.10}, $\sh{F}$ is geometrically reduced at the point $x$ by \sref{IV.4.7.10}, whence the conclusion.
\end{proof}


\subsection{Field of definition}
\label{subsection:IV.4.8}

\begin{env}[4.8.1]
\label{IV.4.8.1}
Given a prescheme $X$, we shall denote in this number by $\mathfrak{S}(X)$ the set of sub-preschemes of $X$, and as usual by $\mathfrak{P}(X)$ the set of subsets of the underlying space of $X$; for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, we shall denote by $\Phi(\sh{F})$ the set of quasi-coherent sub-$\sh{O}_X$-Modules of $\sh{F}$.
\end{env}

\begin{env}[4.8.2]
\label{IV.4.8.2}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes, $\sh{F}$, $\sh{G}$ two $\sh{O}_X$-Modules quasi-coherent, $K$, $K'$ two extensions of $k$ such that $K'\subset K$.
There are then, corresponding to the morphism $\Spec(K)\to\Spec(K')$, canonical maps
\begin{align}
\tag{4.8.2.1}
  & \Phi(\sh{F}\otimes_k K') \to \Phi(\sh{F}\otimes_k K) \\
\tag{4.8.2.2}
  & \operatorname{Hom}(\sh{F}\otimes_k K', \sh{G}\otimes_k K') \to \operatorname{Hom}(\sh{F}\otimes_k K, \sh{G}\otimes_k K) \\
\tag{4.8.2.3}
  & \mathfrak{S}(X_{(K')}) \to \mathfrak{S}(X_{(K)}) \\
\tag{4.8.2.4}
  & \operatorname{Hom}_{K'}(X_{(K')}, Y_{(K')}) \to \operatorname{Hom}_K(X_{(K)}, Y_{(K)}) \\
\tag{4.8.2.5}
  & \mathfrak{P}(X_{(K')}) \to \mathfrak{P}(X_{(K)})
\end{align}
\end{env}

If $p_X:X_{(K)}\to X_{(K')}$ is the canonical projection, the map \sref{IV.4.8.2.5} is $M\mapsto p_X^{-1}(M)$, and similarly \sref{IV.4.8.2.3} is $Z\mapsto Z\otimes_{K'} K$ \sref[I]{I.4.4.1}.
The map \sref{IV.4.8.2.1} is $\sh{H}\mapsto\sh{H}\otimes_{\sh{O}_{X_{(K')}}}\sh{O}_{X_{(K)}}$; \sref{IV.4.8.2.2} is $u\mapsto p_X^*(u)$ and finally \sref{IV.4.8.2.4}.

If we denote by $\varepsilon_{K,K'}$ any one of these canonical maps, it is clear that if $K''$, $K'$, $K$ are three extensions of $k$ such that $K''\subset K'\subset K$, we have
\[
\tag{4.8.2.6}
  \varepsilon_{K,K''} = \varepsilon_{K,K'}\circ\varepsilon_{K',K''}.
\]

\begin{proposition}[4.8.3]
\label{IV.4.8.3}
The canonical maps \sref{IV.4.8.2.1} to \sref{IV.4.8.2.5} are injective.
\end{proposition}

\begin{proof}
We know indeed that $p_X$ is faithfully flat and quasi-compact; the fact that \sref{IV.4.8.2.5} is injective follows from the fact that $p_X$ is surjective.
The injectivity of \sref{IV.4.8.2.1} follows from \sref{IV.2.2.2}, that of \sref{IV.4.8.2.2} from \sref{IV.2.2.7}, that of \sref{IV.4.8.2.3} from \sref{IV.2.2.15}, and finally that of \sref{IV.4.8.2.4} from \sref{IV.2.2.16}.
\end{proof}

\oldpage[IV-2]{81}

\begin{definition}[4.8.4]
\label{IV.4.8.4}
With the notation of \sref{IV.4.8.2}, we say that a sub-$\sh{O}_{X_{(K)}}$-Module quasi-coherent of $\sh{F}\otimes_k K$ (resp.\ a homomorphism $\sh{F}\otimes_k K\to\sh{G}\otimes_k K$, resp.\ a sub-prescheme of $X_{(K)}$, resp.\ a $K$-morphism $X_{(K)}\to Y_{(K)}$, resp.\ a subset of $X_{(K)}$) is \emph{defined over} $K'$ if it belongs to the image of the map \sref{IV.4.8.2.1} (resp.\ \sref{IV.4.8.2.2}, resp.\ \sref{IV.4.8.2.3}, resp.\ \sref{IV.4.8.2.4}, resp.\ \sref{IV.4.8.2.5}).
We say that $K'$ is a \emph{field of definition} of the object considered.
\end{definition}

It is clear that $K$ itself is a field of definition of any one of the objects under consideration in any of the five sets of images of maps \sref{IV.4.8.2.1} to \sref{IV.4.8.2.5}.
We will note however that for a $K$-prescheme $Z$, to say that the $\sh{O}_Z$-Module quasi-coherent $\sh{H}$ is ``defined on a sub-field $K'$ of $K$'' has no meaning unless $Z$ has been \emph{given} in the form $X_{(K)}$ for a prescheme $X$ over a sub-field $k$ of $K$, and if $\sh{H}$ is a sub-$\sh{O}_Z$-Module of an $\sh{O}_Z$-Module which has been \emph{given} in the form $\sh{F}\otimes_k K$, where $\sh{F}$ is an $\sh{O}_X$-Module quasi-coherent; there are analogous remarks for the other notions.
It follows from \sref{IV.4.8.2.6} that if an element of one of the five sets of images of the maps \sref{IV.4.8.2.1} to \sref{IV.4.8.2.5} is \emph{defined} on $K'$, it is also defined on every sub-field $K''$ such that $K'\subset K''\subset K$.
Finally, with the notation of \sref{IV.4.8.2.6}, if $K_1$ is an extension of $K$, and an element of one of the sets of images of $\varepsilon_{K_1,K'}$ is defined on $K'$, it suffices that its image under $\varepsilon_{K_1,K}$ be defined on $K'$, by virtue of the relation $\varepsilon_{K_1,K'} = \varepsilon_{K_1,K}\circ\varepsilon_{K,K'}$.

\begin{proposition}[4.8.5]
\label{IV.4.8.5}
With the notation of \sref{IV.4.8.2}, let $(X_\alpha)$ be an open covering of $X$.
For an element $\sh{H}\in\Phi(\sh{F}\otimes_k K)$ (resp.\ $u\in\operatorname{Hom}(\sh{F}\otimes_k K, \sh{G}\otimes_k K)$, resp.\ $Z\in\mathfrak{S}(X_{(K)})$, resp.\ $M\in\mathfrak{P}(X_{(K)})$) to be defined over $K'$, it is necessary and sufficient that for every $\alpha$, $\sh{H}\,|\,(X_\alpha)_{(K)}$ (resp.\ $Z\cap(X_\alpha)_{(K)}$, resp.\ $M\cap(X_\alpha)_{(K)}$) be defined over $K'$.
\end{proposition}

\begin{proof}
This follows immediately from the injectivity of the maps \sref{IV.4.8.2.1}, \sref{IV.4.8.2.2}, \sref{IV.4.8.2.3}, and \sref{IV.4.8.2.5}: for example, if for every $\alpha$, there is a sub-$\sh{O}_{(X_\alpha)_{(K')}}$-Module quasi-coherent $\sh{H}'_\alpha$ such that $\sh{H}\,|\,(X_\alpha)_{(K)} = \sh{H}'_\alpha\otimes_{K'} K$, for two indices $\alpha$, $\beta$ arbitrary, we will have $\sh{H}'_\alpha\,|\,(X_\alpha\cap X_\beta)_{(K')} = \sh{H}'_\beta\,|\,(X_\alpha\cap X_\beta)_{(K')}$, hence $\sh{H}'_\alpha\,|\,(X_\alpha\cap X_\beta)_{(K')} = \sh{H}'_\beta\,|\,(X_\alpha\cap X_\beta)_{(K')}$ for any two indices $\alpha$ and $\beta$, and so there is a sub-$\sh{O}_{X_{(K')}}$-Module quasi-coherent $\sh{H}'$ of $\sh{F}\otimes_k K'$ such that $\sh{H} = \sh{H}'\otimes_{K'} K$.
We argue in the same way for the other cases.
\end{proof}

\begin{lemma}[4.8.6]
\label{IV.4.8.6}
Let $k$ be a field, $K$ an extension of $k$, $A$ a $k$-algebra, $M$ an $A$-module, $\overline{N}$ a sub-$A_{(K)}$-module of $M_{(K)}$, $K'$ a sub-extension of $K$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\overline{N}$ is of the form $N'\otimes_{K'} K$, where $N'$ is a sub-$A_{(K')}$-module of $M_{(K')}$.
  \item If $X=\Spec(A)$, the $\sh{O}_{X_{(K)}}$-Module quasi-coherent $(\overline{N})^\sim$ on $X_{(K)}=\Spec(A_{(K)})$ is defined over $K'$.
  \item $\overline{N}$ is of the form $N'\otimes_{K'} K$, where $N'$ is a sub-$K'$-vector subspace of $M_{(K')}$.
\end{enumerate}
\end{lemma}

\begin{proof}
\oldpage[IV-2]{82}
The equivalence of $a)$ and $b)$ follows trivially from \sref[I]{I.1.6.5}; it is clear that $a)$ implies $c)$.
Conversely, if $c)$ is satisfied, we may write (up to canonical identification) $N'=\overline{N}\cap M_{(K')}$.
Since $\overline{N}$ and $M_{(K')}$ are by hypothesis sub-$A_{(K')}$-modules, it follows that $N'$ is also an $A_{(K')}$-module.
\end{proof}

\begin{corollary}[4.8.7]
\label{IV.4.8.7}
Under the hypotheses of \sref{IV.4.8.6}, there exists a smallest sub-field $K'$ of $K$ containing $k$ such that the conditions of \sref{IV.4.8.6} are satisfied.
\end{corollary}

\begin{proof}
It suffices to see this for condition $c)$, where this follows from Bourbaki, \emph{Alg.}, chap.~II, 3\textsuperscript{e}~\'{e}d., \S~8, n\textsuperscript{o}~6, prop.~6.
\end{proof}

\begin{lemma}[4.8.8]
\label{IV.4.8.8}
With the notation of \sref{IV.4.8.2}:
\begin{enumerate}[label=(\roman*)]
  \item Let $\overline{u}:\sh{F}\otimes_k K\to\sh{G}\otimes_k K$ be an $\sh{O}_{X_{(K)}}$-homomorphism, and let $\sh{H}\subset(\sh{F}\otimes_k K)\oplus(\sh{G}\otimes_k K)$ be its graph.
For $\overline{u}$ to be defined over $K'$, it is necessary and sufficient that $\sh{H}$ be so.
  \item Let $\overline{Z}$ be a closed sub-prescheme of $X_{(K)}$, and let $\overline{\sh{J}}$ be the quasi-coherent Ideal of $\sh{O}_{X_{(K)}}$ defining $\overline{Z}$.
For $\overline{Z}$ to be defined over $K'$, it is necessary and sufficient that $\overline{\sh{J}}$ be so.
  \item Let $\overline{f}$ be a $K$-morphism $\overline{f}:X_{(K)}\to Y_{(K)}$ and let $\overline{Z}$ be the sub-prescheme of $X_{(K)}\times_K Y_{(K)}$, graph of $\overline{f}$ \sref[I]{I.5.3.11}; for $\overline{f}$ to be defined over $K'$, it is necessary and sufficient that $\overline{Z}$ be so.
\end{enumerate}
\end{lemma}

\begin{proof}
Assertion~(ii) follows immediately from \sref[I]{I.4.4.5}.
The necessity in assertion~(iii) is evident; suppose conversely that $\overline{Z}=Z'_{(K')}$, where $Z'$ is a sub-prescheme of $(X\times_k Y)_{(K')}$, and if $p':Z'\to X_{(K')}$ is the restriction to $Z'$ of the first projection, we know that $p'_{(K)}$ is an isomorphism of preschemes, hence so is $p'$ (\sref{IV.2.7.1},~(viii)); but this means \sref[I]{I.5.3.11} that $Z'$ is the graph of a $K'$-morphism $f':X_{(K')}\to Y_{(K')}$ \sref[I]{I.5.3.12}.

Finally, to prove~(i), by virtue of \sref{IV.4.8.5}, we may suppose that $X=\Spec(A)$ is affine, $\sh{F}=\widetilde{M}$, $\sh{G}=\widetilde{N}$, where $M$ and $N$ are $A$-modules, and $\overline{u}=(\overline{v})^\sim$, where $\overline{v}:M_{(K)}\to N_{(K)}$ is an $A_{(K)}$-homomorphism.
Let us suppose that the graph of $\overline{v}$ is of the form $P'_{(K)}$, where $P'$ is a sub-$A_{(K')}$-module of $M_{(K')}\oplus N_{(K')}$; if $p':P'\to M_{(K')}$ is the restriction to $P'$ of the first projection, we know that $p'\otimes 1_K$ is an isomorphism of $A_{(K)}$-modules, hence, by faithful flatness $(\mathbf{0}_I,\,6.4.1)$, $p'$ is an isomorphism of $A_{(K')}$-modules; hence $P'$ is the graph of an $A_{(K')}$-homomorphism $v':M_{(K')}\to N_{(K')}$ and we have evidently $\overline{v}=v'\otimes 1_K$.
\end{proof}

\begin{proposition}[4.8.9]
\label{IV.4.8.9}
Let $k$ be a field, $K$ an extension of $k$, $X$ a $k$-prescheme, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent.
Let $\sh{H}$ be a sub-$\sh{O}_{X_{(K)}}$-Module quasi-coherent of $\sh{F}\otimes_k K$.
Then there exists a smallest field of definition of $\sh{H}$.
\end{proposition}

\begin{proof}
When $X$ is affine, the assertion is simply \sref{IV.4.8.7}.
In the general case, we consider a covering $(X_\alpha)$ of $X$ by affine opens; by virtue of the above, for every $\alpha$ there is a smallest sub-field $K'_\alpha$ of $K$ containing $k$ and such that $\sh{H}\,|\,(X_\alpha)_{(K)}$ is defined on $K'_\alpha$.
By virtue of \sref{IV.4.8.5}, the sub-field of $K$ generated by the $K'_\alpha$ is the smallest field of definition of $\sh{H}$.
\end{proof}

\begin{corollary}[4.8.10]
\label{IV.4.8.10}
With the notation of \sref{IV.4.8.9}, let $\sh{G}$ be a second $\sh{O}_X$-Module quasi-coherent, and let $\overline{u}:\sh{F}\otimes_k K\to\sh{G}\otimes_k K$ be an $\sh{O}_{X_{(K)}}$-homomorphism; then there exists a smallest field of definition of $\overline{u}$.
\end{corollary}

\begin{proof}
\oldpage[IV-2]{83}
Indeed, it follows from \sref{IV.4.8.8} that such a field is also the smallest field of definition of the graph of $\overline{u}$, and the existence of the latter follows from \sref{IV.4.8.9}.
\end{proof}

\begin{corollary}[4.8.11]
\label{IV.4.8.11}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$, $\overline{Z}$ a closed sub-prescheme of $X_{(K)}$; then there exists a smallest field of definition of $\overline{Z}$.
\end{corollary}

\begin{proof}
Indeed, if $\overline{\sh{J}}$ is the quasi-coherent Ideal of $\sh{O}_{X_{(K)}}$ defining $\overline{Z}$, it follows from \sref{IV.4.8.8} that this is equivalent to the existence of a smallest field of definition of $\overline{\sh{J}}$, which follows from \sref{IV.4.8.9}.
\end{proof}

\begin{corollary}[4.8.12]
\label{IV.4.8.12}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes, $Y$ being supposed separated, $K$ an extension of $k$, $\overline{f}:X_{(K)}\to Y_{(K)}$ a $K$-morphism.
Then there exists a smallest field of definition of $\overline{f}$.
\end{corollary}

\begin{proof}
Indeed, the existence of such a field, by virtue of \sref{IV.4.8.8}, is equivalent to that of the smallest field of definition of the graph $\overline{Z}$ of $\overline{f}$; but as $\overline{Z}$ is a sub-prescheme \emph{closed} of $(X\times_k Y)_{(K)}$ \sref[I]{I.5.4.3}, the existence of a smallest field of definition of $\overline{Z}$ follows from \sref{IV.4.8.11}.
\end{proof}

\begin{proposition}[4.8.13]
\label{IV.4.8.13}
Under the hypotheses of \sref{IV.4.8.11} (resp.\ \sref{IV.4.8.9}, \sref{IV.4.8.10}, \sref{IV.4.8.12}), suppose furthermore that $X$ is of finite type over $k$ (resp.\ that $X$ is of finite type over $k$ and $\sh{F}$ and $\sh{G}$ coherent, resp.\ that $X$ and $Y$ are of finite type over $k$).
Then the smallest field of definition of $\overline{Z}$ (resp.\ $\sh{H}$, $\overline{u}$, $\overline{f}$) is an extension of finite type of $k$.
\end{proposition}

\begin{proof}
We may restrict to the case of \sref{IV.4.8.9}, and suppose $X$ affine since $X$ is a finite union of affine opens.
The whole thing reduces then, with the notation of \sref{IV.4.8.6}, to showing that if $A$ is an algebra of finite type over $k$ and $M$ an $A$-module of finite type, there is a sub-extension $K'$ of $K$ satisfying conditions $a)$, $b)$, or $c)$ and which is of finite type over $k$ (every sub-extension of an extension of finite type being of finite type).
Now, let $(m_i)_{1\leq i\leq n}$ be a system of generators of the $A$-module $M$; then $\overline{N}$ is an $A_{(K)}$-module of finite type, admitting a finite system of generators $n_j$ ($1\leq j\leq r$) such that $n_j = \sum_i (m_i\otimes 1)b_{ij}$, where $b_{ij}\in A_{(K)}$; each $b_{ij}$ may be written as $b_{ij} = \sum_h a_{ijh}\otimes c_{ijh}$, where $a_{ijh}\in A$ and $c_{ijh}\in K$; it is clear that the extension $K'$ of $k$ generated by the $c_{ijh}$ answers the question.
\end{proof}

\begin{proposition}[4.8.14]
\label{IV.4.8.14}
Let $k$ be a field, $X$ a $k$-prescheme of finite type, $K$ an extension of $k$ containing a perfect extension of $k$.
Then the smallest field of definition of the closed sub-prescheme $(X_{(K)})_{\mathrm{red}}$ of $X_{(K)}$ is a finite radicial extension of $k$.
\end{proposition}

\begin{proof}
If $p$ is the characteristic exponent of $k$, the hypothesis implies that $k'=k^{p^{-\infty}}$ is contained in $K$; we know that $(X_{(k')})_{\mathrm{red}}$ is geometrically reduced over $k'$ \sref{IV.4.6.1}, and hence $(X_{(k')})_{\mathrm{red}}\otimes_{k'} K$ is reduced, and is consequently equal to $(X_{(K)})_{\mathrm{red}}$ \sref[I]{I.5.1.8}; in other words, $k'$ is a field of definition of $(X_{(K)})_{\mathrm{red}}$.
Since $k'$ is an algebraic extension of $k$, the conclusion follows from \sref{IV.4.8.13}.
\end{proof}

\begin{remarks}[4.8.15]
\label{IV.4.8.15}
\begin{enumerate}[label=(\roman*)]
  \item The conclusion of \sref{IV.4.8.14} does not necessarily hold if one does not assume that $K$ contains a perfect extension of $k$.
\oldpage[IV-2]{84}
For example, let $k_0$ be a field of characteristic $p>0$, $k$ the field of rational functions $k_0(s,t)$ in two indeterminates, $L$ the field $k(s^{1/p},t^{1/p})$ and $X=\Spec(L)$.
Let $u$ be a third indeterminate and $K$ be the field $k(u,s^{1/p}+ut^{1/p})$; one easily verifies that $k$ is algebraically closed in $K$ and that $L\otimes_k K$ has a non-zero nilradical, and hence $X_{(K)}$ is not reduced; but as $(X_{(K)})_{\mathrm{red}}$ is not defined on $k$, and $K$ does not contain any sub-extension of finite degree over $k$ other than $k$, the smallest field of definition of $(X_{(K)})_{\mathrm{red}}$ cannot be finite over $k$.
  \item Given an element of one of the sets of images of the maps \sref{IV.4.8.2}, and an extension $K_1$ of $K$, it is equivalent to say that the object considered admits a smallest field of definition, or that its image under $\varepsilon_{K_1,K}$ admits one (the two sub-fields of definition being necessarily the same), as one has seen in \sref{IV.4.8.4}.
\end{enumerate}
\end{remarks}


\subsection{Field of definition of a part of a prescheme}
\label{subsection:IV.4.9}

\begin{proposition}[4.9.1]
\label{IV.4.9.1}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$, $\overline{T}$ a closed subset of $X_{(K)}$, $K'$ a sub-extension of $K$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\overline{T}$ is defined over $K'$.
  \item The open subset $\overline{U}=X_{(K)}-\overline{T}$ of $X_{(K)}$ is defined over $K'$.
  \item[\emph{b$'$)}] The sub-prescheme of $X_{(K)}$ induced on the open $\overline{U}$ is defined over $K'$.
  \item[\emph{c)}] There exists a closed sub-prescheme $\overline{Z}$ of $X_{(K)}$ having $\overline{T}$ as underlying space, and defined over $K'$.
\end{enumerate}
\end{proposition}

\begin{proof}
If $g:X_{(K)}\to X_{(K')}$ is the canonical projection, saying that $\overline{T}$ is defined over $K'$ means that there exists a closed subset $T'$ of $X_{(K')}$ such that $\overline{T}=g^{-1}(T')$; since $g^{-1}(X_{(K')}-T')=X_{(K)}-g^{-1}(T')$ for every subset $T'$ of $X_{(K')}$, $a)$, $b)$, and $b')$ are equivalent.
If $\overline{Z}$ is a closed sub-prescheme having $\overline{T}$ as underlying space, defined over $K'$, then $\overline{Z}=Z'\otimes_{K'} K$, where $Z'$ is a closed sub-prescheme of $X_{(K')}$ having $T'$ as underlying space; hence $a)$ implies $c)$.
Conversely, to see that $a)$ implies $c)$, it suffices to consider a closed sub-prescheme $Z'$ of $X_{(K')}$ having $T'$ as underlying space and to take $\overline{Z}=Z'\otimes_{K'} K$ \sref[I]{I.5.2.1}.
\end{proof}

\begin{corollary}[4.9.2]
\label{IV.4.9.2}
With the notation of \sref{IV.4.9.1}, suppose that $K'$ be a field of definition of $\overline{T}$; every extension $K''\subset K$ of $k$, such that $K'$ be a radicial extension of $K''$, is also a field of definition of $\overline{T}$.
\end{corollary}

\begin{proof}
It suffices to observe that the canonical projection $g:X_{(K')}\to X_{(K'')}$ is a surjective, integral, and radicial morphism, hence a homeomorphism \sref{IV.2.4.5}.
\end{proof}

\begin{remark}[4.9.3]
\label{IV.4.9.3}
It follows from \sref{IV.4.9.2} that a closed subset $\overline{T}$ of $X_{(K)}$ does not necessarily admit a \emph{smallest} field of definition.
For example, if $K=k(t)$ where $t$ is an indeterminate and $k$ is a perfect field of characteristic $p>0$, a closed subset $\overline{F}$ of $X_{(K)}$ can have no smallest field of definition: indeed, for every sub-field $K'$ of $K$ containing $k$ and $\neq k$, we have $K'^p\neq K'$, $K'^p$ contains $k$ and $K'$,
\oldpage[IV-2]{85}
and hence if $K'$ is a field of definition of $\overline{F}$, so is $K'^p$.
However:
\end{remark}

\begin{proposition}[4.9.4]
\label{IV.4.9.4}
With the notation of \sref{IV.4.9.1}, suppose that $K$ is a separable extension of $K'$; for $K'$ to be a field of definition of $\overline{T}$, it is necessary and sufficient that $K'$ be a field of definition of the reduced sub-prescheme of $X_{(K)}$ having $\overline{T}$ as underlying space.
\end{proposition}

\begin{proof}
The condition is evidently sufficient by virtue of \sref{IV.4.9.1}; to see that it is necessary, let us note that if $\overline{T}=g^{-1}(T')$, where $T'$ is a closed subset of $X_{(K')}$, and if $Z'$ is the reduced sub-prescheme of $X_{(K')}$ having $T'$ as underlying space, then $\overline{Z}=Z'\otimes_{K'} K=\Spec(K)\times_{K'} Z'$ is reduced by virtue of the criterion \sref{IV.4.6.1},~$e)$) and of the hypothesis on $K$, and has $\overline{T}$ as underlying space.
\end{proof}

\begin{corollary}[4.9.5]
\label{IV.4.9.5}
Suppose one of the following hypotheses is satisfied:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $k$ is of characteristic $0$.
  \item $K$ is an algebraic extension of $k$ and $X$ is of finite type.
\end{enumerate}
Then every closed subset $\overline{T}$ of $X_{(K)}$ has a smallest field of definition, which is a separable extension of $k$ (and finite over $k$ under hypothesis $b)$).
\end{corollary}

\begin{proof}
Under hypothesis $a)$, $K$ is separable on each of its sub-fields, and it follows from \sref{IV.4.9.4} that the fields of definition of $\overline{T}$ are the same as those of the reduced sub-prescheme of $X_{(K)}$ having $\overline{T}$ as underlying space.
The corollary follows then from \sref{IV.4.8.11}.
Under hypothesis $b)$, we know that $K$ is a radicial extension of a separable extension $K_1$ of $k$.
Then, for every sub-extension $K'$ of $K$, $K'$ is radicial on $K'\cap K_1$, and hence, by virtue of \sref{IV.4.9.2}, $K'$ is a field of definition of $\overline{T}$ if and only if $K'\cap K_1$ is.
This brings us back to the case where $K'=K_1$, and hence to the case where $K'$ is a separable algebraic extension of $k$, and one completes the argument as in the case $a)$.
\end{proof}

\begin{proposition}[4.9.6]
\label{IV.4.9.6}
Let $k$ be a field, $X$ a $k$-prescheme, $K$ an extension of $k$, $\overline{U}$ a subset which is both open and closed in $X_{(K)}$, $\overline{V}$ its complement.
Let $K'$ be a sub-extension of $K$; the following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $K'$ is a field of definition of the subset $\overline{U}$ of $X_{(K)}$.
  \item[\emph{a$'$)}] $K'$ is a field of definition of the closed sub-prescheme of $X_{(K)}$ induced on the open $\overline{U}$.
  \item[\emph{b)}] $K'$ is a field of definition of the subset $\overline{V}$ of $X_{(K)}$.
  \item[\emph{b$'$)}] $K'$ is a field of definition of the closed sub-prescheme of $X_{(K)}$ induced on the open $\overline{V}$.
\end{enumerate}
\end{proposition}

\begin{proof}
It is clear that $a')$ implies $a)$, and $a)$ implies $b')$ by virtue of \sref{IV.4.9.1}.
Since $\overline{U}$ and $\overline{V}$ play symmetric roles, $b')$ implies $b)$ and $b)$ implies $a')$, which completes the proof.
\end{proof}

\begin{corollary}[4.9.7]
\label{IV.4.9.7}
Under the hypotheses of \sref{IV.4.9.6}, there exists a smallest field of definition $K'$ for the subset $\overline{U}$ of $X_{(K)}$, and $K'$ is also the smallest field of definition of the subset $\overline{V}$ of $X_{(K)}$ and of the closed sub-preschemes of $X_{(K)}$ induced on the opens $\overline{U}$ and $\overline{V}$.
If furthermore $X$ is of finite type over $k$, $K'$ is a finite separable extension of $k$.
\end{corollary}

\begin{proof}
Taking into account \sref{IV.4.9.6} and \sref{IV.4.8.11}, the only thing to prove is the last assertion.
By virtue of \sref{IV.4.8.15},~(ii)), we may restrict to the case where $K$ is algebraically closed.
By virtue of \sref{IV.4.8.13}, we may restrict to the case where $K$ is algebraically closed.
Let $\bar{k}$ be the algebraic closure of $k$ contained in $K$, and let us denote by $W_i$ the connected components
\oldpage[IV-2]{86}
of $X_{(\bar{k})}$ ($1\leq i\leq m$); if $p:X_{(K)}\to X_{(\bar{k})}$ is the canonical projection, we know that the $p^{-1}(W_i)$ are the connected components of $X_{(K)}$ \sref{IV.4.4.6}, and hence $U$ is a union of a certain number of these components, and is therefore defined on $\bar{k}$; the conclusion follows by virtue of \sref{IV.4.9.5},~$b)$).
\end{proof}
