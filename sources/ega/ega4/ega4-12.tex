\section{Study of fibres of finitely presented flat morphisms}
\label{section:IV.12}

\subsection{Introduction}
\label{subsection:IV.12.0}

\oldpage[IV-3]{173}

\emph{We will use throughout this paragraph the general notation of} \sref{IV.9.4.1}.

\begin{env}[12.0.1]
\label{IV.12.0.1}
Given a morphism \emph{locally of finite presentation} $f:X\to Y$, we have seen \sref{IV.9.9} that for certain \emph{local} properties $\mathbf{P}$ of preschemes over fields or of Modules over these preschemes, the set $\mathrm{E}$ of $x\in\mathrm{X}$ such that the property $\mathbf{P}$ holds, for the \emph{fibre} $\mathrm{X}_{f(x)}$, at the point $x$ of this fibre, is \emph{locally constructible in} $\mathrm{X}$.
We now propose to show that, for the majority of these properties, if one further supposes that the morphism $f$ is \emph{flat}, then the set $\mathrm{E}$ is in fact \emph{open in} $\mathrm{X}$.
Similarly (\sref{IV.9.2} to \sref{IV.9.8}) we have shown that if $f$ is \emph{of finite presentation}, and if this time $\mathbf{P}$ denotes certain \emph{global} properties of preschemes over fields or of Modules over these preschemes, the set $\mathrm{F}$ of $y\in\mathrm{Y}$ such that the property $\mathbf{P}$ holds is \emph{locally constructible in} $\mathrm{Y}$.
We will show that if one further supposes that the morphism $f$ is \emph{proper} and \emph{flat}, $\mathrm{F}$ is in fact \emph{open in} $\mathrm{Y}$.
\end{env}

\begin{env}[12.0.2]
\label{IV.12.0.2}
The general method of proof of the properties in question involves three stages.
One first reduces to the case where $\mathrm{Y}$ is \emph{affine} and $\mathrm{X}$ of finite presentation; then:

\medskip
\begin{enumerate}[label=\emph{\Alph*)}]
  \item With the help of \sref{IV.8.9.1} (and possibly other results of \textsection\,8) and of \sref{IV.11.2.6}, one reduces to the case where $\mathrm{X}$ and $\mathrm{Y}$ are \emph{Noetherian}.

  \item One applies the results of \textsection\,9 recalled in \sref{IV.12.0.1} to show that $\mathrm{E}$ (resp.\ $\mathrm{F}$) is \emph{constructible}.

  \item To see that $\mathrm{E}$ is open, it suffices, by virtue of \sref[0]{0.9.2.5}, to show that if $x\in\mathrm{E}$, then every \emph{generization} $x'$ of $x$ also belongs to $\mathrm{E}$.
Using \sref[II]{II.7.1.7}, one sees that, since $\mathrm{Y}$ is \emph{Noetherian}, there exists a spectrum of a \emph{discrete valuation ring} $\mathrm{Y}_1$ and a morphism $h:\mathrm{Y}_1\to\mathrm{X}$ such that, if $y_1$ (resp.\ $y_1'$) is the closed point (resp.\ the generic point) of $\mathrm{Y}_1$, one has $h(y_1)=x$, $h(y_1')=x'$.
One then performs the base change $g=f\circ h:\mathrm{Y}_1\to\mathrm{Y}$; taking into account the results of \textsection\textsection\,4 and 6 on locally Noetherian preschemes over fields and the changes of base field, one is reduced to proving the assertion in question for a point $x_1$ of $\mathrm{X}_1=\mathrm{X}\times_{\mathrm{Y}}\mathrm{Y}_1$ above $y_1$ and for a generization $x_1'$ of $x_1$ above $y_1'$.
Since $\mathrm{Y}_1=\Spec(\mathrm{A})$, where $\mathrm{A}$ is a discrete valuation ring, with uniformiser $t$, one must in the end, for an $\mathrm{A}$-module $\mathrm{M}$, prove a property of $\mathrm{M}$, knowing that $\mathrm{M}/t\mathrm{M}$ has the same property and that $t$ is $\mathrm{M}$-\emph{regular} (which follows from the flatness hypothesis); one uses for this the results of \sref{IV.3.4} and of \sref{IV.5.12}.
One proceeds similarly for the set $\mathrm{F}\subset\mathrm{Y}$.
\end{enumerate}
\end{env}

\subsection{Local properties of fibres of a flat morphism locally of finite presentation}
\label{subsection:IV.12.1}

\oldpage[IV-3]{174}

\begin{theorem}[12.1.1]
\label{IV.12.1.1}
Let $f:X\to Y$ be a morphism locally of finite presentation, $\sh{F}$ an $\sh{O}_X$-Module $f$-flat and of finite presentation, $\Phi$ a finite subset of $\mathbf{Z}\cup\{\pm\infty\}$, $k$ an integer.
The following subsets of $X$ are open:
\begin{enumerate}[label=(\roman*)]
  \item The set of $x\in X$ such that the dimensions of the prime cycles associated to $\sh{F}_{f(x)}$ and containing $x$ are elements of $\Phi$.
  \item The set of $x\in X$ such that all the prime cycles associated to $\sh{F}_{f(x)}$ containing $x$ have dimension $k$.
  \item The set of $x\in X$ such that $x$ does not belong to any associated immersed prime cycle of $\sh{F}_{f(x)}$.
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is equidimensional at the point $x$ and possesses the property $(\mathrm{S}_k)$ at the point $x$ \sref{IV.5.7.2}.
  \item The set of $x\in X$ such that $\operatorname{coprof}((\sh{F}_{f(x)})_x)\leqslant k$ \sref[0]{0.16.4.9}.
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is a Cohen--Macaulay $\sh{O}_{X_{f(x)}}$-Module at the point $x$ \sref{IV.5.7.1}.
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is geometrically reduced at the point $x$ \sref{IV.4.6.22}.
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is geometrically punctually integral at the point $x$ \sref{IV.4.6.22}.
\end{enumerate}

The questions being local on $X$, one first reduces to the case where $X=\Spec(B)$ and $Y=\Spec(A)$ are affine, $f$ a morphism of finite presentation.
There is then a Noetherian subring $A_0$ of $A$, an $A_0$-prescheme of finite type $X_0$, a coherent $\sh{O}_{X_0}$-Module $\sh{F}_0$ and $f$-flat; in addition, by virtue of \sref{IV.11.2.6}, one can suppose that $\sh{F}_0$ is $Y_0$-flat (with $Y_0=\Spec(A_0)$).
If $h:X\to X_0$ is the canonical projection, the set $E$ of points $x\in X$ where one of the properties (i) to (viii) is satisfied is equal to $h^{-1}(E_0)$, where $E_0$ is the set of $x_0\in X_0$ where the corresponding property is satisfied: this follows, for properties (i) to (iii), from \sref{IV.4.2.7}; for properties (iv) to (vi), from \sref{IV.6.7.1}; for properties (vii) and (viii), from \sref{IV.4.7.11}.
\end{theorem}

One is thus reduced to the case where $A$ and $B$ are Noetherian, $f$ of finite type, and $\sh{F}=\widetilde{M}$, where $M$ is a $B$-module of finite type.
In virtue of \sref{IV.9.9.2}, one knows that the set $E$ is \emph{constructible} for all the properties considered, and it remains in each case the step~C) of \sref{IV.12.0.2}, where one must prove that $E$ is \emph{stable under generization}.

\begin{env}[12.1.1.1]
\label{IV.12.1.1.1}
Let us begin with case~(iii), which is the simplest.
Let $x'\neq x$ be a generization of $x$ in $X$.
Set $y=f(x)$, $y'=f(x')$; there exists a spectrum of a discrete valuation ring $Y_1$ and a morphism $u:Y_1\to X$ such that, if $y_1$ and $y_1'$ are the closed point and the generic point of $Y_1$ respectively, one has $u(y_1)=x$ and $u(y_1')=x'$ \sref[II]{II.7.1.7}; set $g=f\circ u:Y_1\to Y$; if $X_1=X\times_Y Y_1$, there exists a $Y_1$-section $u_1:Y_1\to X_1$ such that $u=g_1\circ u_1$, where $g_1:X_1\to X$ is the canonical projection.
If $x_1=u_1(y_1)$, $x_1'=u_1(y_1')$, one has $g_1(x_1)=x$ and $g_1(x_1')=x'$, and $x_1'$ is a generization of $x_1$.
Applying again \sref{IV.4.2.7}, one sees
\oldpage[IV-3]{175}
that one is reduced to the case where $Y=\Spec(A)$ is the spectrum of a discrete valuation ring, $y$ being the closed point and $y'$ the generic point of $Y$.
If $t$ is a uniformiser of $A$, the hypothesis that $\sh{F}$ is $f$-flat implies that $t$ is $\sh{F}$-regular \sref[0]{0.6.3.4}.
By hypothesis, none of the associated immersed prime cycles of $\sh{F}/t\sh{F}$ contains $x$.
Then, it follows from \sref{IV.3.4} that $x$ does not belong to any associated immersed prime cycle of $\sh{F}$, and the same is true of every generization of $x$.
In particular, $x'$ does not belong to any associated immersed prime cycle of $\sh{F}$, nor \emph{a fortiori} to any of those associated to $\sh{F}_{y'}$ ($X_{y'}$ being a sub-prescheme inducing an open subset of $X$).
\end{env}

\begin{env}[12.1.1.2]
\label{IV.12.1.1.2}
Let us now consider cases~(iv) and~(v) ((vi) is only a particular case of~(v)).
One proceeds as above (using \sref{IV.6.7.1} in place of \sref{IV.4.2.7}) and one is reduced to the case where $Y$ is the spectrum of a discrete valuation ring~$A$.

The ring $C=\sh{O}_{X,x}$ is then the localisation of an $A$-algebra of finite type, hence is catenary \sref{IV.5.6.4}, and by hypothesis the $C$-module $M_x/tM_x$ satisfies $(\mathrm{S}_k)$ and is equidimensional (resp.\ on a $\operatorname{coprof}(M_x/tM_x)\leqslant k$); one deduces from \sref{IV.5.12.2} (resp.\ from \sref[0]{0.16.4.10}) that $M_x$ satisfies $(\mathrm{S}_k)$ and is equidimensional (resp.\ that $\operatorname{coprof}(M_x)\leqslant k$) since $t$ is $M_x$-regular and belongs to the maximal ideal of $C$.
Whence the conclusions since $x'$ is a generization of $x$ and $(\sh{F}_{y'})_{x'}=\sh{F}_{x'}$.
\end{env}

\begin{env}[12.1.1.3]
\label{IV.12.1.1.3}
Let us pass to cases~(vii) and~(viii).
One can evidently replace $Y$ by the integral sub-prescheme having $\{y'\}$ as underlying space, and $X$ by $f^{-1}(\{y'\})$, without changing the fibres in $y$ and $y'$, so one can suppose $A$ integral, with field of fractions $K=k(y')$.
One knows (\sref{IV.4.5.11} and \sref{IV.4.7.8}) that there exists a finite extension $K'$ of $K$ such that, for the $(\sh{O}_X\otimes_A K')$-Module $\sh{F}'=\sh{F}\otimes_A K'$, the associated prime cycles are geometrically irreducible and the geometric lengths of $\sh{F}'$ at its maximal points $z_i'$ of its support are respectively equal to the lengths of $\sh{F}_{y_1'}'$ at these points; it amounts therefore to the same to say that $\sh{F}$ is geometrically reduced (resp.\ geometrically punctually integral) at a point $x'$ of $X_{y'}=X\otimes_A K$, or to say that $\sh{F}'$ is \emph{reduced} (resp.\ \emph{integral}), taking into account \sref{IV.4.2.7}.
Let $A'$ be a finite $A$-algebra whose field of fractions is $K'$; it follows from \sref{IV.4.7.11} that at every point of $X\otimes_A A'$ above $x$, $\sh{F}\otimes_A A'$ has property~(vii) (resp.~(viii)).
Replacing $A$ by $A'$ and $X$ by $X\otimes_A A'$, one sees that one can reduce to proving that $\sh{F}_{f(x')}$ is \emph{reduced} (resp.\ \emph{integral}) at every generization $x'$ of $x$.
One proceeds then as in \sref{IV.12.1.1.1}, and using this time \sref{IV.4.7.11}, in the case where $A$ is a discrete valuation ring, $y$ being the closed point, $y'$ the generic point of $Y$, and the uniformiser $t$ of $A$ being an $M$-\emph{regular} element.
Since by hypothesis $\sh{F}/t\sh{F}$ is reduced (resp.\ integral) at the point $x$, it follows from \sref{IV.3.4.6} (resp.\ \sref{IV.3.4.5}) that $\sh{F}$ is reduced (resp.\ integral) at the point $x$, hence also in a neighbourhood of $x$, and in particular at the point $x'$, which finishes the proof in cases~(vii) and~(viii).
\end{env}

\begin{env}[12.1.1.4]
\label{IV.12.1.1.4}
It remains to examine cases~(i) and~(ii).
Replacing $Y$ by the integral sub-prescheme having $\{y'\}$ as underlying space, one can reduce to the case where $Y$ is irreducible
\oldpage[IV-3]{176}
containing $x'$, and let $Z'$ be the closure of $z'$ in $X$, so that one has $x\in Z'$.
To treat case~(i), it will suffice to prove the

\begin{proposition}[12.1.1.5]
\label{IV.12.1.1.5}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $\sh{F}$ a coherent $\sh{O}_X$-Module and $f$-flat; for every $y\in Y$, set $\sh{F}_y=\sh{F}\otimes_{\sh{O}_Y}k(y)$.
Let $z'$ be a point of $X$, $y'=f(z')$, and suppose that $z'\in\operatorname{Ass}(\sh{F}_{y'})$.
Let $Z'$ \textnormal{(}resp.\ $Y'$\textnormal{)} be the reduced sub-prescheme of $X$ \textnormal{(}resp.\ $Y$\textnormal{)} having the closure of $\{z'\}$ \textnormal{(}resp.\ of $\{y'\}$\textnormal{)} in $Y$ as underlying space.
Then, for every $y\in f(Z')$, the dimensions of all the irreducible components of $Z'_y$ are equal to $\dim(Z'_{y'})$ \textnormal{(}dimension of the closure of $z'$ in $X_{y'}$\textnormal{)} \textnormal{(}which we will express more precisely later \sref{IV.13.2.2} by saying that the restriction $Z'\to Y'$ of $f$ is an equidimensional morphism\textnormal{)}; furthermore, at every maximal point $z$ of $Z'_y$, one has $z\in\operatorname{Ass}(\sh{F}_y)$.
\end{proposition}

For this, we will now reduce to the case where $Y$ is the spectrum of a discrete valuation ring.
Let us take a spectrum of a discrete valuation ring $Y_1$ and a morphism $h:Y_1\to X$ such that $h(y_1)=z$, $h(y_1')=z'$, $y_1$ and $y_1'$ being the closed point and the generic point of $Y_1$ respectively \sref[II]{II.7.1.7}.
Set $g=f\circ h:Y_1\to Y$, $X_1=X\times_Y Y_1$, and let $f_1:X_1\to Y_1$, $g_1:X_1\to X$ be the canonical projections; there is a $Y_1$-section $h_1:Y_1\to X_1$ such that $h=g_1\circ h_1$ \sref[I]{I.3.3.14}.
If $Z_i'=Z'_y\otimes_{k(y')}k(y_1')$, one knows \sref{IV.4.2.6} that the irreducible components of $Z_i'$ dominate $Z'_y$; let $z_i'$ be the generic point of one of these components which contains $h_1(y_1')$, so that $f_1(z_i')=y_1'$ and $g_1(z_i')=z'$.
Since $h_1(y_1')$ is a specialisation of $h_1(y_1)$, it is \emph{a fortiori} a specialisation of $z_i'$; let $z_1$ be a generic point of one of the irreducible components of $\overline{\{z_i'\}}\cap(X_1)_{y_1}$ containing $h_1(y_1)$.
Then $g_1(z_1)$ is a specialisation of $g_1(z_i')=z'$ in $X$, and $z=h(y_1)=g_1(h_1(y_1))$ is a specialisation of $g_1(z_1)$; but since $f(g_1(z_1))=y$ and $z$ is a generic point of $Z'_y$, one necessarily has $z=g_1(z_1)$.

Suppose now that, if one sets $\sh{F}_1=\sh{F}\otimes_{\sh{O}_Y}\sh{O}_{Y_1}$, it will follow from \sref{IV.4.2.7} that $z\in\operatorname{Ass}(\sh{F}_y)$; furthermore, the dimensions of $\{z_1\}\cap(X_1)_{y_1}$ and of $\{z\}\cap X_y$ are equal, and of the same dimensions as $\{z_i'\}\cap(X_1)_{y_1'}$ and of $\{z'\}\cap X_{y'}$ \sref{IV.4.2.7}; one has therefore indeed reduced, as announced, to the case where $Y$ is the spectrum of a discrete valuation ring $A$, $y$ and $y'$ being its closed point and its generic point respectively.

Since $z'\in\operatorname{Ass}(\sh{F}_y)$, on a \emph{a fortiori} $z'\in\operatorname{Ass}(\sh{F})$.
If $t$ is a uniformiser of $A$, $t$ is $\sh{F}$-regular by flatness, so \sref{IV.3.4.3} on a $z\in\operatorname{Ass}(\sh{F}/t\sh{F})=\operatorname{Ass}(\sh{F}_y)$.
As for the assertion relative to dimensions, it follows from \sref{IV.7.1.13}, applied to a closed sub-prescheme of $X$ having $Z'$ as underlying space.

Let us finally tackle case~(ii) of \sref{IV.12.1.1}.
With the same notation, it is necessary to prove that if all the prime cycles associated to $\sh{F}_y$ and containing $x$ have dimension $k$, the same is true of all the prime cycles associated to $\sh{F}_{y'}$ and containing $x'$.
Now, one has just seen that \emph{every} prime cycle associated to $\sh{F}_{y'}$ and containing $x'$ has the same dimension as \emph{one} of the prime cycles associated to $\sh{F}_y$ and containing $x$; this shows~(ii).
\end{env}

\begin{remarks}[12.1.2]
\label{IV.12.1.2}
\begin{enumerate}[label=(\roman*)]
  \item Under the conditions of \sref{IV.12.1.1.5} with $\sh{F}=\sh{O}_X$ (so that $f$ is \emph{flat}), one cannot in general assert that the restriction $Z'\to Y'$ of $f$ is a flat morphism nor even an \emph{open} morphism.
This is what the example \sref{IV.6.5.5},
\oldpage[IV-3]{177}
(ii) shows, where one takes for $Z'$ one of the two irreducible components of $X=\Spec(B)$.
It is immediate that this morphism restriction is not open at the points of $Z'$ above the ``double point'' of $Y'=Y$.

  \item Under the hypotheses of \sref{IV.12.1.1.5}, with $\sh{F}=\sh{O}_X$, one cannot weaken the condition ``$f$ is flat'' to ``$f$ is universally open'' (cf.\ \sref{IV.2.4.6}), as we will see further on with an example \sref{IV.14.4.10},~(i)).
\end{enumerate}
\end{remarks}

\begin{corollary}[12.1.3]
\label{IV.12.1.3}
Under the hypotheses of \sref{IV.12.1.1}, the function $x\mapsto\operatorname{coprof}((\sh{F}_{f(x)})_x)$ is upper semi-continuous in $X$ and the function $x\mapsto\operatorname{prof}_x^*(\sh{F}_{f(x)})$ \sref{IV.10.8.1} is lower semi-continuous in $X$.
\end{corollary}

The first assertion is nothing other than an equivalent formulation of \sref{IV.12.1.1},~(v).
To prove the second, one can first, by taking into account \sref{IV.10.8.7} and \sref{IV.10.8.8}, reduce as in \sref{IV.12.1.1} to the case where $Y$ is the spectrum of a discrete valuation ring $A$ of closed point $y$ and of generic point $y'$, $X=\Spec(B)$ being affine, $\sh{F}=\widetilde{M}$, where $M$ is a $B$-module of finite type.
Since $\sh{F}$ is $f$-flat, every irreducible component of $\operatorname{Supp}(\sh{F})$ which contains $x\in X_y$ dominates $Y$ \sref{IV.2.3.4}, in other words its generic point belongs to $X_{y'}$; the irreducible components $Z$ of $\operatorname{Supp}(\sh{F})$ which contain a generization $x'$ of $x$ belonging to $X_{y'}$ are therefore exactly those which contain $x$ and which, furthermore, by \sref{IV.12.1.1.5}, $\dim(Z_y)=\dim(Z_{y'})$, whence, if $T=\operatorname{Supp}(\sh{F})$, $\dim_x(T_y)=\dim_{x'}(T_{y'})$.
Furthermore, for a uniformiser $t$ of $A$, one has seen in \sref{IV.12.1.1},~(v) that $\operatorname{coprof}(M_x/tM_x)=\operatorname{coprof}(M_x)$, and since on the other hand $\operatorname{coprof}(M_{x'})\leqslant\operatorname{coprof}(M_x)$ \sref{IV.6.11.5}, one sees that one has $\operatorname{coprof}((\sh{F}_y)_x)\leqslant\operatorname{coprof}((\sh{F}_{y'})_z)$, the relation $\operatorname{prof}_x^*(\sh{F}_y)\leqslant\operatorname{prof}_{x'}^*(\sh{F}_{y'})$ follows from \sref{IV.10.8.7}.

\begin{remark}[12.1.4]
\label{IV.12.1.4}
One also deduces from \sref{IV.12.1.1},~(i) that the function $x\mapsto\dim_x(X_{f(x)})$ is upper semi-continuous in $X$, but we will see further on \sref{IV.13.1.3} that this property is true even without supposing $f$ flat.
\end{remark}

\begin{corollary}[12.1.5]
\label{IV.12.1.5}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $\sh{F}$ an $\sh{O}_X$-Module coherent and $f$-flat, $\sh{G}$ an $\sh{O}_X$-Module coherent, $x$ a point of $X$, $y=f(x)$.
Suppose that $\sh{G}$ has the property $(\mathrm{S}_k)$ at the point $y$, and that $\sh{F}_y$ has the property $(\mathrm{S}_k)$ at the point $x$ and is equidimensional at the point $x$.
Then:
\begin{enumerate}[label=(\roman*)]
  \item $\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ has the property $(\mathrm{S}_k)$ at the point $x$.
  \item If furthermore $Y$ is locally immersible in a regular scheme, or is a prescheme excellent \sref{IV.7.8.5}, there exists a neighbourhood $V$ of $x$ in $X$ such that $\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ has the property $(\mathrm{S}_k)$ in this neighbourhood.
\end{enumerate}
\end{corollary}

\begin{proof}
Indeed, by \sref{IV.12.1.1},~(iv), there exists an open neighbourhood $V$ of $x$ such that for every $x'\in V$, $\sh{F}_{f(x')}$ satisfies $(\mathrm{S}_k)$ at the point $x'$.
On the other hand, $\sh{G}$ satisfies $(\mathrm{S}_k)$ at all points of $\Spec(\sh{O}_y)$ \sref{IV.5.7.2}.
Replacing $Y$ by $Y'=\Spec(\sh{O}_y)$, one is reduced (taking into account \sref[I]{I.3.6.5}) to the case where $\sh{G}$ satisfies $(\mathrm{S}_k)$ in $Y$ entirely and where, for every $y\in f(X)$, $\sh{F}_y$ has the property $(\mathrm{S}_k)$ at all points of $f^{-1}(y)$; since $\sh{F}$ is $f$-flat, one knows then \sref{IV.6.4.1} that $\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ has the property $(\mathrm{S}_k)$ in $X$, which proves~(i).
To prove~(ii), it suffices to observe that, under the hypotheses made, $\sh{G}$ has the
\oldpage[IV-3]{178}
property $(\mathrm{S}_k)$ in a neighbourhood $U$ of $y$ in $Y$ \sref{IV.6.11.2} and \sref{IV.7.8.3},~(iv)); one concludes then in the same manner by replacing $Y$ this time by $U$ and $X$ by $V\times_Y U$.
\end{proof}

\begin{theorem}[12.1.6]
\label{IV.12.1.6}
Let $f:X\to Y$ be a morphism flat and locally of finite presentation, $k$ an integer.
The following subsets of $X$ are open:
\begin{enumerate}[label=(\roman*)]
  \item The set of points $x\in X$ such that $X_{f(x)}$ has the property $(\mathrm{S}_k)$ at the point $x$.
  \item The set of $x\in X$ such that $X_{f(x)}$ satisfies the property $(\mathrm{R}_k)$ geometrically at the point $x$, is equidimensional and also such that $x$ does not belong to any associated immersed prime cycle of $X_{f(x)}$.
  \item The set of $x\in X$ such that $X_{f(x)}$ is geometrically regular (i.e.\ smooth) at the point $x$.
  \item The set of $x\in X$ such that $X_{f(x)}$ is geometrically normal at the point $x$.
\end{enumerate}

Steps~A) and~B) of \sref{IV.12.0.2} are done here as in \sref{IV.12.1.1}; for step~A), one uses \sref{IV.6.7.8}, and also \sref{IV.4.2.7} for~(ii); for step~B), on uses \sref{IV.9.9.2} and \sref{IV.9.9.4}.
It remains to examine step~C) in each case.
\end{theorem}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item As in \sref{IV.12.1.1.2}, one reduces (using \sref{IV.6.7.8}) to the case where $Y$ is the spectrum of a discrete valuation ring $A$, $X=\Spec(B)$, where $B$ is an $A$-algebra of finite type, $x$, $x'$ two points of $X$ such that $f(x)=y$ is the closed point and $y'=f(x')$ the generic point of $Y$, $x'$ being in addition a generization of $x$.
As one wishes to prove that $X$ satisfies $(\mathrm{S}_k)$ at the point $x'$, that is to say suppose the ring $B$ local, the homomorphism $A\to B$ being local and $B$ being an $A$-algebra essentially of finite type \sref{IV.1.3.8}.
Then, by hypothesis, if $t$ is a uniformiser of $A$, $t$ is a $B$-regular element by flatness, and $B/tB$ satisfies $(\mathrm{S}_k)$.
But since $A$ is a universally catenary ring \sref{IV.5.6.4}, $B$ is catenary, so, by \sref{IV.5.12.4}, it follows that $B$ satisfies $(\mathrm{S}_k)$, which finishes the proof in case~(i).

  \item Here step~C) is unnecessary; since $f$ is flat, one knows indeed \sref{IV.6.8.7} that when $Y$ is locally Noetherian and $f$ locally of finite type, the set of $x\in X$ such that $X_{f(x)}$ is geometrically regular at the point $x$ is open in $X$.

  \item By reasoning as in \sref{IV.12.1.1.3}, to prove that the property considered is stable under generization, one can first, by considering a finite extension $K'$ of $k(y')$, and taking into account the definition \sref{IV.6.7.6} of the geometric property $(\mathrm{R}_k)$, as well as the invariance by change of base field of the two other properties appearing in~(ii), replace in~(ii) the property $(\mathrm{R}_k)$ geometric by the property $(\mathrm{R}_k)$.
Proceeding then as in~(i), one reduces to the case where $Y$ is the spectrum of a discrete valuation ring $A$, and since $A$ is catenary, it suffices to apply \sref{IV.5.12.5} to conclude.

  \item The set of points $x\in X$ such that $X_{f(x)}$ is geometrically normal at the point $x$ is contained in that of $x\in X$ such that $X_{f(x)}$ is geometrically punctually integral and satisfies $(\mathrm{S}_2)$ at the point $x$, and this latter set is open in $X$ by virtue of \sref{IV.12.1.1},~(viii).
One can therefore already suppose that, for every $x\in X$, $X_{f(x)}$ is geometrically punctually integral and satisfies $(\mathrm{S}_2)$ at the point $x$, and \emph{a fortiori} it is equidimensional.
On the other hand, by the Serre criterion \sref{IV.5.8.6}, to say that $X_{f(x)}$ is geometrically normal at the point $x$ means that $X_{f(x)}$ satisfies $(\mathrm{S}_2)$ and the property $(\mathrm{R}_1)$ geometrically at $x$;
\oldpage[IV-3]{179}
but by virtue of~(ii) and the preceding remarks, this set is the intersection of two open subsets of $X$, hence is open in $X$.
\end{enumerate}
\end{proof}

\begin{corollary}[12.1.7]
\label{IV.12.1.7}
Let $f:X\to Y$ be a morphism locally of finite presentation.
Then the set of points $x\in X$ where $f$ possesses one of the following properties \sref{IV.6.8.1}:
\begin{enumerate}[label=(\roman*)]
  \item satisfy the property $(\mathrm{S}_k)$;
  \item be of coprofundity $\leqslant n$;
  \item be Cohen--Macaulay;
  \item be regular (or smooth, which amounts to the same);
  \item be normal;
  \item be reduced;
\end{enumerate}
is open in $X$.
\end{corollary}

\begin{proof}
Indeed, it follows from \sref{IV.11.3.1} that the set of $x\in X$ where $f$ is \emph{flat} is \emph{open}.
One can therefore restrict to the case where $f$ is flat, and then the corollary follows from \sref{IV.12.1.1},~(iv), (vi) and~(vii)) and from \sref{IV.12.1.6},~(i),~(ii) and~(iv)).
\end{proof}

\subsection{Local and global properties of fibres of a proper, flat morphism of finite presentation}
\label{subsection:IV.12.2}

\oldpage[IV-3]{179}

\begin{theorem}[12.2.1]
\label{IV.12.2.1}
Let $f:X\to Y$ be a morphism \emph{proper} of finite presentation, $\sh{F}$ an $\sh{O}_X$-Module $f$-flat and of finite presentation, $\Phi$ a finite subset of $\mathbf{Z}\cup\{\pm\infty\}$, $k$ an integer.
The following subsets of $Y$ are open:
\begin{enumerate}[label=(\roman*)]
  \item The set of $y\in Y$ such that the dimensions of the prime cycles associated to $\sh{F}_y$ are contained in $\Phi$.
  \item The set of $y\in Y$ such that the dimensions of the irreducible components of $\operatorname{Supp}(\sh{F}_y)$ are contained in $\Phi$.
  \item The set of $y\in Y$ such that all the prime cycles associated to $\sh{F}_y$ have the same dimension equal to $k$.
  \item The set of $y\in Y$ such that $\sh{F}_y$ has no associated immersed prime cycle and that the set of dimensions of the irreducible components of $\operatorname{Supp}(\sh{F}_y)$ is equal to $\Phi$.
\oldpage[IV-3]{180}
  \item The set of $y\in Y$ such that $\sh{F}_y$ has the property $(\mathrm{S}_k)$ and is equidimensional at every point of $X_y$.
  \item The set of $y\in Y$ such that $\operatorname{coprof}(\sh{F}_y)\leqslant k$.
  \item The set of $y\in Y$ such that $\sh{F}_y$ is a Cohen--Macaulay $\sh{O}_{X_y}$-Module.
  \item The set of $y\in Y$ such that $\sh{F}_y$ is geometrically reduced.
  \item The set of $y\in Y$ such that $\sh{F}_y$ is geometrically punctually integral at every point of $X_y$.
  \item The set of $y\in Y$ such that $\sh{F}_y$ is geometrically integral.
  \item The set of $y\in Y$ such that $\sh{F}_y$ has no associated immersed prime cycle and that the sum $l(y)$ of the total multiplicities \sref{IV.4.7.12} of $\sh{F}_y$ at the maximal points of $\operatorname{Supp}(\sh{F}_y)$ is $\leqslant k$.
\end{enumerate}

With the exception of~(ii), (iii), (iv), (x) and~(xi), the properties $\mathbf{P}(y)$ considered are of the form: ``for every $x\in X_y$, $\sh{F}_y$ has the property $\mathbf{Q}(x)$'', where $\mathbf{Q}(x)$ is one of the properties~(i) to~(viii) of \sref{IV.12.1.1}.
It follows from \sref{IV.12.1.1} that the set $U$ of $x\in X$ such that $\mathbf{Q}(x)$ is true is open, and the set of $y\in Y$ such that $\mathbf{P}(y)$ is true is nothing other than $Y-f(X-U)$; in all these cases, the theorem is therefore already true by only supposing the morphism $f$ \emph{closed}.
It remains to examine cases~(ii), (iv) and~(xi) separately ((x) follows by deduction immediately from~(xi) and~(iv)) by always applying the method described in \sref{IV.12.0.2}.
\end{theorem}

\begin{proof}
For~(iii), one applies~(i) with $\Phi$ equal to the smallest interval of $\mathbf{N}$ containing the dimensions of the prime cycles associated to $\sh{F}_y$, and reduces to a single element.
For~(ii) and~(iv), it remains to examine cases~(ii), (iv) and~(xi) separately.
\end{proof}

\begin{env}[12.2.1.1]
\label{IV.12.2.1.1}
Case~(ii): Steps~A) and~B) of the proof proceed as at the start of \sref{IV.12.1.1}; for step~A), one uses \sref{IV.8.9.1}, \sref{IV.11.2.6} as well as \sref{IV.4.2.7}; for step~B), one uses \sref{IV.9.5.5} applied to $\operatorname{Supp}(\sh{F})$.
It remains to show that if (supposing $Y$ Noetherian) the set of dimensions of the irreducible components of $\operatorname{Supp}(\sh{F}_y)$ is contained in $\Phi$, the same is true of the set of dimensions of the irreducible components of $\operatorname{Supp}(\sh{F}_{y'})$, for every \emph{generization} $y'$ of $y$ in $Y$.
Let $Y_1$ be a spectrum of a discrete valuation ring such that, if $y_1$ and $y_1'$ are the closed point and the generic point of $Y_1$, there exists a morphism $h:Y_1\to Y$ such that $h(y_1)=y$, $h(y_1')=y'$ \sref[II]{II.7.1.7}.
Applying \sref{IV.4.2.7}, one sees that one can replace $Y$ by $Y_1$ and $X$ by $X_1=X\times_Y Y_1$, in other words, one can restrict to the case where $Y$ is the spectrum of a discrete valuation ring, $y$ its closed point and $y'$ its generic point.

Using \sref[III]{III.4.7.9}, one can restrict to the case where $\operatorname{Supp}(\sh{F})=X$, so that $f$ is quasi-flat \sref{IV.2.3.3}; the irreducible components $Z_i$ of $X$ dominate $Y$ then \sref{IV.2.3.4}, in other words their generic points belong to $X_{y'}$, and the $Z_i\cap X_{y'}$ are the irreducible components of $X_{y'}$.
But every irreducible component of $X_y$ is contained in one of the $Z_i$; now, one knows \sref{IV.7.1.13} that the dimensions of all the irreducible components of $(Z_i)_y$ are equal to $\dim((Z_i)_{y'})$, which finishes the proof of~(ii).
\end{env}

\begin{env}[12.2.1.2]
\label{IV.12.2.1.2}
Case~(iv): The same reasoning as at the start of \sref{IV.12.2.1.1}, using \sref{IV.12.1.1},~(iii), shows that one can already suppose (replacing $Y$ by an open subset of $Y$) that for every $y\in Y$, $\sh{F}_y$ has no associated immersed prime cycle.
The assertion of case~(iv) is then an immediate consequence of the assertions of cases~(i) and~(ii).
\end{env}

\begin{env}[12.2.1.3]
\label{IV.12.2.1.3}
\oldpage[IV-3]{181}
Case~(xi): For step~A) of the reasoning, one uses \sref{IV.8.9.1}, \sref{IV.11.2.6}, \sref{IV.8.10.5},~(xii) (to keep the hypothesis that $f$ is proper), as well as \sref{IV.4.2.7}, \sref{IV.4.5.6} and \sref{IV.4.7.9}.
For step~B), one sees, as in case~(iv), that one can suppose that for every $y\in Y$, $\sh{F}_y$ has no associated immersed prime cycle, and one applies \sref{IV.9.8.8} which proves that the function $z\mapsto l(z)$ is constructible.
It remains therefore to see (supposing $Y$ Noetherian and $f$ proper) that for every generization $y'$ of a point $y\in Y$, $l(y')\leqslant l(y)$.
By reasoning as in \sref{IV.12.1.1.3}, one sees (with the help of \sref{IV.4.7.8} and \sref{IV.4.5.11}) that one can suppose the irreducible components of $\operatorname{Supp}(\sh{F}_{y'})$ are geometrically irreducible and that $l(y')$ is the \emph{sum of the lengths} of $(\sh{F}_{y'})_{z_j'}$ at the maximal points $z_j'$ of $\operatorname{Supp}(\sh{F}_{y'})$.
Applying again \sref{IV.4.2.7}, \sref{IV.4.5.6} and \sref{IV.4.7.9}, one reduces, as in \sref{IV.12.2.1.1}, to the case where $Y$ is a spectrum of a discrete valuation ring, $y$ its closed point and $y'$ its generic point.
The fact that $l(y')\leqslant l(y)$ will then be a consequence of the

\begin{lemma}[12.2.1.4]
\label{IV.12.2.1.4}
Let $Y$ be a spectrum of a discrete valuation ring, $y$ its closed point, $y'$ its generic point, $f:X\to Y$ a proper morphism, $\sh{F}$ a coherent $\sh{O}_X$-Module and $f$-flat, $z_i$ \textnormal{(}resp.\ $z_j'$\textnormal{)} the maximal points of $\operatorname{Supp}(\sh{F}_y)$ \textnormal{(}resp.\ $\operatorname{Supp}(\sh{F}_{y'})$\textnormal{)}.
Suppose that $\sh{F}_y$ has no associated immersed prime cycle.
Then one has
\[
  \sum_j \operatorname{long}((\sh{F}_{y'})_{z_j'}) \leqslant \sum_i \operatorname{long}((\sh{F}_y)_{z_i}).
\]
\end{lemma}

One has $Y=\Spec(A)$, where $A$ is a discrete valuation ring, whose uniformiser we will denote by $t$, so that $\sh{F}_y=\sh{F}/t\sh{F}$.
Since $\sh{F}$ is $f$-flat, the $z_j'$ are also the maximal points of $\operatorname{Supp}(\sh{F})$ \sref{IV.2.3.4}; for every $z_i$, denote by $z_{ik}'$ those of the $z_j'$ which are generizations of $z_i$; it follows from \sref{IV.3.4.1.1} that one has
\[
  \operatorname{long}((\sh{F}_y)_{z_i}) \geqslant \sum_k \operatorname{long}((\sh{F}_{y'})_{z_{ik}'})
\]
whence by summing
\[
  \sum_i \operatorname{long}((\sh{F}_y)_{z_i}) \geqslant \sum_i \sum_k \operatorname{long}((\sh{F}_{y'})_{z_{ik}'}).
\]

The lemma will therefore be proved if we establish that for every $z_j'$, there is at least one index $i$ such that $z_j'$ is one of the $z_{ik}'$.
Now, since $f$ is proper (hence closed) and $f(z_j')=y'$, there exists $x\in X$ which is a specialisation of $z_j'$ and is such that $f(x)=y$, in other words $\{z_j'\}\cap X_y$ is non-empty.
Since $t$ is $\sh{F}$-regular by flatness, one deduces from \sref{IV.3.4.3} that there is at least one point of $\operatorname{Ass}(\sh{F}_y)$ of which $z_j'$ is a generization; but since the points of $\operatorname{Ass}(\sh{F}_y)$ are the $z_i$, this finishes the proof of the lemma.
\end{env}

\begin{corollary}[12.2.2]
\label{IV.12.2.2}
Under the hypotheses of \sref{IV.12.2.1}:
\begin{enumerate}[label=(\roman*)]
  \item The function $y\mapsto\dim(\operatorname{Supp}(\sh{F}_y))$ is continuous (hence locally constant) in $Y$.
  \item The function $y\mapsto\operatorname{coprof}(\sh{F}_y)$ \sref{IV.5.7.1} is upper semi-continuous in $Y$.
  \item The function $y\mapsto l(y)$ \sref{IV.12.2.1},~(xi) is upper semi-continuous in $Y$ when the $\sh{F}_y$ have no associated immersed prime cycle.
\oldpage[IV-3]{182}
  \item The function $y\mapsto\operatorname{prof}^*(\sh{F}_y)$ \sref{IV.10.8.1} is lower semi-continuous in $Y$.
\end{enumerate}
\end{corollary}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item It follows from \sref{IV.12.2.1},~(i) applied to $\Phi$ equal to the smallest interval of $\mathbf{N}$ containing the dimensions of the prime cycles associated to $\sh{F}_y$, that $z\mapsto\dim(\operatorname{Supp}(\sh{F}_z))$ is upper semi-continuous at the point $y$; it follows on the other hand from \sref{IV.12.2.1},~(ii) applied to $\Phi$ equal to the set of dimensions of the irreducible components of $\operatorname{Supp}(\sh{F}_y)$, that this same function is lower semi-continuous at the point $y$, whence the conclusion.
The assertions~(ii) and~(iii) are nothing but other formulations of \sref{IV.12.2.1},~(vi) and~(xi).
Finally, by \sref{IV.12.1.3}, the set of $x\in X$ such that $\operatorname{prof}_{f(x)}^*(\sh{F}_{f(x)})\leqslant k$ is open, and the conclusion of~(iv) follows by the same reasoning as at the beginning of \sref{IV.12.2.1}.
\end{enumerate}
\end{proof}

\begin{remarks}[12.2.3]
\label{IV.12.2.3}
\begin{enumerate}[label=(\roman*)]
  \item One will observe that for \sref{IV.12.2.1},~(iii), one can dispense with the hypothesis that $f$ is \emph{proper}.

  \item In \sref{IV.12.2.1},~(ii), one cannot replace the condition that the set $E(y)$ of dimensions of the irreducible components of $\operatorname{Supp}(\sh{F}_y)$ contains $\Phi$, by the condition that $E(y)$ be \emph{equal to} $\Phi$.
Indeed, let $k$ be a field, and consider the projective space $P=\mathbf{P}_k^2=\operatorname{Proj}(C)$ of $3$ dimensions, where $C=k[t_0,t_1,t_2,t_3]$ ($t_i$ indeterminates); in $P$, let $X_0$ be the closed sub-prescheme ``union of the line $X_1$ defined by $t_1=t_3=0$ and of the plane $X_2$ defined by $t_2-t_3=0$'', which one describes in geometric terms by saying that $X_0$ is equal to $\operatorname{Proj}(C/\mathfrak{a})$, where the graded ideal $\mathfrak{a}$ is equal to $\mathfrak{b}\cdot\mathfrak{c}$, with $\mathfrak{b}=Ct_1+Ct_3$, $\mathfrak{c}=C(t_2-t_3)$; consider the morphism $f_0:X_0\to X_1$ which, in geometric terms, is the ``projection of $X_0$ onto $X_1$ from the line defined by $t_0=t_2=0$''; in algebraic terms, $f_0$ corresponds to the homomorphism of graded algebras $k[t_0,t_3]\to k[t_0,t_1,t_2,t_3]/\mathfrak{a}$ obtained by passage to the quotient from the canonical injection $k[t_0,t_3]\to k[t_0,t_1,t_2,t_3]$.
It is clear that $f_0$ is a projective morphism, hence \emph{proper}; furthermore $f$ is \emph{flat}, it suffices to note that $k[t_3]$ is a principal ring and that the ring $C'=k[t_1,t_2,t_3]/(t_2-t_3)((t_2)+( t_3))$ is a $k[t_3]$-module without torsion \sref[0]{0.6.3.4}.

  \item In \sref{IV.12.2.1},~(i), one cannot also replace the condition that the set $E'(y)\supset E(y)$ of dimensions of the prime cycles associated to $\sh{F}_y$ be contained in $\Phi$ by the condition that it be \emph{equal to} $\Phi$.
Indeed, let $k$ be a field, $A$ the polynomial ring $k[t]$, $B$ the subring $k[t^4,t^3u,tu^3,u^4]$ of the polynomial ring $k[t,u]$ ($t$, $u$ indeterminates); $B$ is not integrally closed, the element $t^2u^2$ belonging to the integral closure but not being in $B$; if $\mathfrak{m}$ is the ideal maximal of $B$, generated by $t^4$, $t^3u$, $tu^3$ and $u^4$, $\mathfrak{m}$ is an associated \emph{immersed} prime ideal of $A/A^{t4}$.
Set $Y=\Spec(A)$, $X=\Spec(B)$, and let $f:X\to Y$ be the morphism corresponding to the homomorphism $A\to B$ which sends $t$ to $t^4$.
Since this homomorphism makes $B$ into an $A$-module without torsion, $f$ is flat \sref[0]{0.6.3.4}.
If $y$ is the point of $Y$ corresponding to the ideal $tA$, the prescheme $f^{-1}(y)$ is irreducible and of dimension~1, but admits a prime cycle associated immersed of dimension~0 and does not satisfy the condition $(\mathrm{R}_0)$ geometrically since the local ring at its generic point is a field.
In contrast, at the generic point $y'$ of $Y$, $f^{-1}(y')$ has no prime cycle associated immersed, being the spectrum of a $k(y')$-algebra integral of dimension~1.

  \item In \sref{IV.12.2.1},~(xi), one cannot suppress the hypothesis that $f$ be proper.
Indeed, let $k$ be a field, and the ``affine line'', spectrum of the polynomial ring $k[t]$ ($t$ indeterminate), $X$ the scheme \emph{somme} of the closed point $y$ of $Y$ and of the open complement of the closed point $y$ in $Y$, and the morphism $f:X\to Y$ equal to the canonical injections on each of the components of $X$, which is flat.
But one has $l(y_0)=1$ and $l(y)=2$ for $y\neq y_0$ in $Y$.

  \item In \sref{IV.12.2.1},~(xi), one cannot also suppress the hypothesis that $\sh{F}_y$ has no associated immersed prime cycle.
This is shown by the example of Remark~(ii): one also sees immediately that, with $\sh{F}=\sh{O}_X$, $l(y_0)=1$ and $l(y)=2$ for $y\neq y_0$ in $Y$.

  \item We do not know if, in \sref{IV.12.2.1},~(xi), one can replace the hypothesis that $\sh{F}$ is $f$-flat by the hypothesis that $\operatorname{Supp}(\sh{F})=X$ and that $f$ is universally open.
By taking up the proof of \sref{IV.12.2.1.4}, one sees (using also \sref{IV.14.3.6}) that one would have to solve the following question: let $A$ be a local Noetherian ring, $M$ an $A$-module of finite type,
\oldpage[IV-3]{183}
$t$ an element of the maximal ideal $\mathfrak{m}$ of $A$, which is not contained in any minimal prime ideal of $A$; if there exists one of these minimal prime ideals $\mathfrak{p}$ such that $\dim(A/\mathfrak{p})=1$, is it true that $\mathfrak{m}$ is an associated prime ideal of $M/tM$ ($t$ not being supposed $M$-regular)?

One will note moreover that one cannot, in \sref{IV.12.2.1},~(xi), purely and simply suppress the hypothesis that $\sh{F}$ is $f$-flat.
One will take here the $P$ of the Remark~(ii), then in $P$ the closed sub-prescheme $X_0$ union of the three lines $X_1$, $X_2$, $X_3$ defined respectively by $t_1=t_3=0$, $t_1-t_0=t_3=0$, $t_2=t_3=0$; one defines the projection $f_0$ of $X_0$ onto $X_1$ as above, and its restriction $f:X\to Y$, where $Y=D_+(t_0)$ is the affine line and $X=f_0^{-1}(Y)$; to see that $f$ is flat, it suffices to note that $k[t_3]$ is a principal ring and the ring $C'=k[t_1,t_2,t_3]/(t_2-t_3)((t_2)+(t_3))$ is a $k[t_3]$-module without torsion \sref[0]{0.6.3.4}.

But one has $l(y_0)=1$ and $l(y)=2$ for $y\neq y_0$ in $Y$.
\end{enumerate}
\end{remarks}

\begin{theorem}[12.2.4]
\label{IV.12.2.4}
Let $f:X\to Y$ be a morphism proper, flat and of finite presentation, $k$ an integer $\geqslant 1$.
The following subsets of $Y$ are open:
\begin{enumerate}[label=(\roman*)]
  \item The set of $y\in Y$ such that $X_y$ possesses the property $(\mathrm{S}_k)$.
  \item The set of $y\in Y$ such that $X_y$ satisfies the property $(\mathrm{R}_k)$ geometrically, is equidimensional at every point, and has no associated immersed prime cycle.
  \item The set of $y\in Y$ such that $X_y$ is geometrically regular (i.e.\ smooth over $k(y)$).
  \item The set of $y\in Y$ such that $X_y$ is geometrically normal.
  \item The set of $y\in Y$ such that $X_y$ is geometrically reduced.
  \item The set of $y\in Y$ such that $X_y$ is geometrically reduced and that the number of geometric connected components of $X_y$ is equal to $k$.
  \item The set of $y\in Y$ such that $X_y$ is geometrically punctually integral \sref{IV.4.6.9}.
  \item The set of $y\in Y$ such that $X_y$ is geometrically integral.
  \item The set of $y\in Y$ such that $X_y$ has no associated immersed prime cycle and that the total multiplicity \sref{IV.4.7.4} of $X_y$ over $k(y)$ is $\leqslant k$.
\end{enumerate}
\end{theorem}

\begin{proof}
Except for~(vi), these assertions are special cases of assertions of \sref{IV.12.2.1}, or are deduced from assertions of \sref{IV.12.1.6} as at the start of the proof of \sref{IV.12.2.1}.
For~(vi), one reduces as always (taking into account the invariance of the geometric number of connected components by base change \sref{IV.4.5.6}) to the case where $Y$ is Noetherian.
The set $U$ of $y\in Y$ such that $X_y$ is geometrically reduced is open in $Y$ by virtue of~(v); it follows then from \sref[III]{III.7.8.7} and~\sref[III]{III.7.8.6} that for every $y\in U$, there is a neighbourhood $V\subset U$ of $y$ and an integer $m$ such that, for \emph{every} $z\in V$, $\Gamma(X_z,\sh{O}_{X_z})$ is isomorphic to $(k(z))^m$; but by virtue of \sref[III]{III.4.3.4}, $m$ is then the geometric number of connected components of $X_z$, whence the conclusion.
We will give another proof of~(vi) in \sref{IV.15.5.9}.
\end{proof}


\subsection{Local cohomological properties of fibres of a flat morphism, locally of finite presentation}
\label{subsection:IV.12.3}

\oldpage[IV-3]{183}

\begin{lemma}[12.3.1]
\label{IV.12.3.1}
Let $A$ be a ring, $B$ an $A$-algebra of finite presentation which is a flat $A$-module, $M$ a $B$-module of finite presentation, which is a flat $A$-module.
Then the following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $M$ is a projective $B$-module.
  \item $M$ is a flat $B$-module.
  \item For every $y\in\Spec(A)$, $M\otimes_A k(y)$ is a projective $(B\otimes_A k(y))$-module.
\end{enumerate}
\end{lemma}

\oldpage[IV-3]{184}

The equivalence of \emph{a)} and \emph{b)} follows from Bourbaki, \emph{Alg.\ comm.}, chap.~II, \textsection\,5, n\textsuperscript{o}\,2, cor.~2 of th.~1.
Since \emph{a)} implies \emph{c)} trivially, it remains to prove that \emph{c)} implies \emph{b)}, which follows from the criterion of flatness by fibres \sref{IV.11.3.10}, applied with $g=h$, $f=1_X$.

\begin{proposition}[12.3.2]
\label{IV.12.3.2}
Let $A$ be a ring, $B$ an $A$-algebra of finite presentation which is a flat $A$-module, $M$ a $B$-module of finite presentation which is a flat $A$-module.
Then:
\begin{enumerate}[label=(\roman*)]
  \item There exists a left resolution of $M$ by $B$-modules free of finite type.
  \item One has
\end{enumerate}
\begin{equation}
\label{IV.12.3.2.1}
\operatorname{dim.proj}_B(M) = \sup_{y\in\Spec(A)} \operatorname{dim.proj}_{B\otimes_A k(y)}(M\otimes_A k(y)) = \operatorname{Tor.dim}_B(M)
\end{equation}
where $\operatorname{Tor.dim}_B(M)$ is the smallest integer $i$ such that $\operatorname{Tor}_j^B(M,N)=0$ for every $j\geqslant i$ and every $B$-module $N$ (and $+\infty$ if no such integer $i$ exists).
\end{proposition}

\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item By virtue of \sref{IV.8.9.1}, there exist a Noetherian subring $A_0$ of $A$, an $A_0$-algebra of finite type $B_0$ and a $B_0$-module of finite type $M_0$ such that $B$ is isomorphic to $B_0\otimes_{A_0}A$ and $M$ to $M_0\otimes_{A_0}A$.
Furthermore \sref{IV.11.2.7} one can suppose that $B_0$ and $M_0$ are flat $A_0$-modules.
There is a left resolution $(L_0)$ of $M_0$ formed of $B_0$-modules free of finite type, and since $M_0$ and the $L_{0i}$ are flat $A_0$-modules, $(L_0)\otimes_{A_0}A$ is a resolution of $M$ formed of $B$-modules free of finite type \sref{IV.2.1.10}.

  \item By virtue of \sref[0]{0.17.2.2},~(ii), on a $\operatorname{Tor.dim}_B(M)\leqslant\operatorname{dim.proj}_B(M)$, and the definition of projective dimension proves immediately that $\operatorname{dim.proj}_B(M)\leqslant n$ (resp.\ $\operatorname{dim.proj}_{B\otimes_A k(y)}(M\otimes_A k(y))\leqslant n$) for every $y\in\Spec(A)$.
To prove the reverse inequalities, consider a left resolution of $M$ by $B$-modules free of finite type, and suppose that $\operatorname{Tor.dim}_B(M)=n$ (resp.\ $\operatorname{dim.proj}_{B\otimes_A k(y)}(M\otimes_A k(y))\leqslant n$ for every $y\in\Spec(A)$).
Then $R=\operatorname{Im}(L_n\to L_{n-1})=\operatorname{Ker}(L_{n-1}\to L_{n-2})$ is a $B$-module of finite type which is also a flat $A$-module, by virtue of the hypothesis on $M$ and $B$ and of \sref{IV.2.1.10}.
Furthermore, on a $\operatorname{Tor}_1^B(R,N)=\operatorname{Tor}_{n+1}^B(M,V,7)$.
The hypothesis $\operatorname{Tor.dim}_B(M)=n$ implies therefore $\operatorname{Tor}_1^B(R,N)=0$ for every $B$-module $N$, that is to say $R$ is a flat $B$-module, hence projective by virtue of \sref{IV.12.3.1}; this establishes that $\operatorname{dim.proj}_B(M)\leqslant n$.
The hypothesis $\operatorname{dim.proj}_{B\otimes_A k(y)}(M\otimes_A k(y))\leqslant n$ for every $y\in\Spec(A)$ implies on the other hand, by tensorisation with $k(y)$, that in each of the sequences (exact by the flatness of $M$ over $A$, of the $L_i$ and of $R$ \sref{IV.2.1.10})
\[
  0\to R\otimes_A k(y)\to L_{n-1}\otimes_A k(y)\to\cdots\to L_0\otimes_A k(y)\to M\otimes_A k(y)\to 0
\]
$R\otimes_A k(y)$ is a $(B\otimes_A k(y))$-module projective (for every $y\in\Spec(A)$).
One concludes then again from \sref{IV.12.3.1} that $R$ is a projective $B$-module, hence $\operatorname{dim.proj}_B(M)\leqslant n$.
\end{enumerate}
\end{proof}

\begin{proposition}[12.3.3]
\label{IV.12.3.3}
Let $f:X\to S$ be a morphism locally of finite presentation, $\sh{L}_\bullet$ a complex formed of $\sh{O}_X$-Modules quasi-coherent of finite presentation; for every $s\in S$, let $(\sh{L}_\bullet)_s$ be the complex $\sh{L}_\bullet\otimes_{\sh{O}_S}k(s)$ of $\sh{O}_{X_s}$-Modules of finite type.
Suppose that $\sh{L}_{n-1}$ is $f$-flat.
Then the set $U$ of $x\in X$ such that $(\sh{H}_n((\sh{L}_\bullet)_{f(x)}))_x=0$ is open in $X$.
If furthermore $\sh{L}_n$ is $f$-flat, then $\sh{H}_n(\sh{L}_\bullet)|U=0$.
\end{proposition}

\begin{proof}
One can evidently limit to the case where $n=1$ and where the complex $\sh{L}_\bullet$ is reduced to $0\to\sh{L}_2\xrightarrow{u}\sh{L}_1\xrightarrow{v}\sh{L}_0\to 0$.
One can first reduce to the case where $S=\Spec(A)$ and $X$
\oldpage[IV-3]{185}
are affine, then to the case where $S$ is \emph{Noetherian}; indeed, by \sref{IV.8.9.1} and \sref{IV.8.5.2}, one knows that there exists a Noetherian prescheme $S'=\Spec(A')$, where $A'$ is a subring of $A$, a morphism of finite type $f':X'\to S'$ and three coherent $\sh{O}_{X'}$-Modules $\sh{L}_i'$ ($0\leqslant i\leqslant 2$) such that $X=X'\times_{S'}S$, $f=f'_{(S)}$, $\sh{L}_i=\sh{L}_i'\otimes_{S'}S$, as well as two homomorphisms $u'$, $v'$ such that $u=u'\otimes 1$, $v=v'\otimes 1$ and $v'\circ u'=0$.
One can furthermore suppose that $\sh{L}_0$ is $f'$-flat, and that $\sh{L}_1$ is $f'$-flat \sref{IV.11.2.7}, and if $U'$ is the set of $x'\in X'$ such that $(\sh{H}_n((\sh{L}_\bullet')_{f'(x')}))_{x'}=0$, $U$ is the inverse image of $U'$ by the projection $X\to X'$, which reduces us, for the first assertion, to the case where $S$ is Noetherian.
For the second assertion, one notes furthermore that $A$ is an inductive limit of its subrings which are $A'$-algebras of finite type; setting $X_\lambda=X'\times_{S'}S_\lambda$, where $S_\lambda=\Spec(A_\lambda)$, and letting $u_\lambda=u'\otimes 1:\sh{L}_2^{(\lambda)}\to\sh{L}_1^{(\lambda)}$, $v_\lambda=v'\otimes 1:\sh{L}_1^{(\lambda)}\to\sh{L}_0^{(\lambda)}$, so that $\sh{H}_1(\sh{L}_\bullet^{(\lambda)})=\operatorname{Ker}(v_\lambda)/\operatorname{Im}(u_\lambda)$.
Or, since the functor $\varinjlim$ is exact in the category of commutative groups, on a $\operatorname{Ker}(v)=\varinjlim(\operatorname{Ker}(v_\lambda))$, $\operatorname{Im}(u)=\varinjlim(\operatorname{Im}(u_\lambda))$, and $\sh{H}_1(\sh{L}_\bullet)=\varinjlim\sh{H}_1(\sh{L}_\bullet^{(\lambda)})$.
If one has supposed that $\sh{L}_1$ is $f$-flat and (reducing to the case where $X=U$) that $\sh{H}_1(\sh{L}_\bullet^{(\lambda)})=0$ for every $\lambda$, one deduces indeed $\sh{H}_1(\sh{L}_\bullet)=0$.

\medskip
I) Suppose now $S$ Noetherian.
One knows (without the flatness hypothesis on $\sh{L}_0$) that the set $U$ is \emph{constructible} in $X$ \sref{IV.9.9.6}.
Using now \sref[0]{0.9.2.5}, it remains to show that for every \emph{generization} $x'$ of $x\in U$ in $X$, one has also $x'\in U$.
The method exposed in \sref{IV.12.0.2} applies without change (taking into account the fact that for a prescheme $Z$ over a field $k$ and an extension $k'$ of $k$, the projection $Z\otimes_k k'\to Z$ is faithfully flat).
One can therefore suppose that $S$ is the spectrum of a discrete valuation ring $A$, with uniformiser $t$, $x$ being above the closed point of $S$ and $x'$ above the generic point of $S$.
The hypothesis that $\sh{L}_0$ is $f$-flat implies that $t$ is $\sh{L}_0$-regular \sref[0]{0.6.3.4}.
One is then reduced to proving the following lemma (where $B$, $M$, $N$, $P$ will be replaced by $\sh{O}_x$, $(\sh{L}_2)_x$, $(\sh{L}_1)_x$ and $(\sh{L}_0)_x$ respectively):

\begin{lemma}[12.3.3.1]
\label{IV.12.3.3.1}
Let $B$ be a local Noetherian ring, $M$, $N$, $P$ three $B$-modules, $N$ being supposed of finite type, $M\xrightarrow{u} N\xrightarrow{v} P$ two homomorphisms such that $v\circ u=0$.
Let $t$ be an element of $B$ such that $t$ be $P$-regular.
Then the relation
\begin{equation}
\label{IV.12.3.3.2}
M/tM \xrightarrow{u\otimes 1_{B/tB}} N/tN \xrightarrow{v\otimes 1_{B/tB}} P/tP
\end{equation}
is exact implies that the sequence $M\xrightarrow{u} N\xrightarrow{v} P$ is exact.
\end{lemma}

Note first that if one replaces $M$ by its image $Q$ in $N$ and $u$ by the injection $j:Q\to N$, the image of $j\otimes 1$ is the same as that of $u\otimes 1$, so one can, without changing the hypothesis nor the conclusion, suppose $u$ injective.
On the other hand, if $R$ is the image of $v$ and $\rho:N\to R$ the canonical surjection, $t$ is evidently $R$-regular, on a $\rho\circ u=0$, and the kernel of $\rho\otimes 1$ is contained in that of $v\otimes 1$; it is therefore equal to this, if \sref{IV.12.3.3.2} is exact; since on the other hand, $\operatorname{Ker}(\rho)=\operatorname{Ker}(v)$, we see from this that
\begin{equation}
\label{IV.12.3.3.4}
\operatorname{Ker}(v\otimes 1)/\operatorname{Im}(u\otimes 1) = (\operatorname{Ker}(v)/\operatorname{Im}(u))\otimes_B(B/tB)
\end{equation}
up to canonical isomorphism.

Indeed, the hypothesis that the sequence \sref{IV.12.3.3.2} is exact implies then that $(\operatorname{Ker}\,v/\operatorname{Im}\,u)\otimes_B(B/tB)=0$, and since $t$ belongs to the maximal ideal of $B$ and $\operatorname{Ker}(v)$ is a $B$-module of finite type (since $B$ is supposed Noetherian in \sref{IV.12.3.3.1}), the Nakayama lemma proves that $\operatorname{Ker}(v)=\operatorname{Im}(u)$.

It remains therefore to show \sref{IV.12.3.3.4}.
Set $u'=u\otimes 1$, $v'=v\otimes 1$, $Z=\operatorname{Ker}(v)$, $H=Z/\operatorname{Im}(u)$, so that the sequences
\begin{gather*}
  0\to M\to Z\to H\to 0\\
  0\to Z\to N\xrightarrow{v} P\to 0
\end{gather*}
\oldpage[IV-3]{186}
whence, by tensorising by $B/tB$ and using the lemma \sref{IV.3.4.1.4} and the fact that $t$ is $P$-regular, the exact sequences
\begin{gather*}
  M/tM \xrightarrow{w'} Z/tZ \to H/tH \to 0\\
  0\to Z/tZ \to N/tN \xrightarrow{v'} P/tP \to 0
\end{gather*}
whence $\operatorname{Ker}(v')=Z/tZ$, and since $\operatorname{Im}(w')=\operatorname{Im}(u')$, one obtains \sref{IV.12.3.3.4}.

\medskip
II) Suppose now furthermore that $\sh{L}_1$ is $f$-flat; replacing in addition $X$ by $U$, one can suppose that $X=U$.
It is a question of seeing that for every $x\in X$, $(\sh{H}_1(\sh{L}_\bullet))_x=0$.
Set $s=f(x)$, and let $\mathfrak{n}$ be the ideal $\mathfrak{m}_s\sh{O}_{X,x}$, which is contained in the maximal ideal of $\sh{O}_{X,x}$; since $(\sh{H}_1(\sh{L}_\bullet))_x$ is an $\sh{O}_{X,x}$-module of finite type ($X$ being locally Noetherian), it suffices to show that its $\mathfrak{n}$-adic separated completion is $0$.
Now, by virtue of \sref[III]{III.7.4.7.2}, this separated completion is equal to $\varprojlim\sh{H}_1(\sh{L}_\bullet\otimes_{\sh{O}_S}\sh{O}_{S,s}/\mathfrak{m}_s^{k+1}))$, and it will therefore suffice to prove that each of the terms of this projective system is zero; we will therefore reason by induction on $k$.
The conclusion will follow from the following more general lemma (which we will apply for $A=\sh{O}_{S,s}/\mathfrak{m}_s^{k+1}$ and $\mathfrak{J}$ equal to the maximal ideal $\mathfrak{m}_s/\mathfrak{m}_s^{k+1}$ of $A$):

\begin{lemma}[12.3.3.5]
\label{IV.12.3.3.5}
Let $A$ be a ring, $\mathfrak{J}$ a nilpotent ideal of $A$, $L_\bullet:P\xrightarrow{u_0}Q\xrightarrow{v_0}R$ a complex of $A$-modules.
Set $A_0=A/\mathfrak{J}$, $L_\bullet^{(0)}=L_\bullet\otimes_A A_0:P_0\xrightarrow{u_0}Q_0\xrightarrow{v_0}R_0$, and suppose that $\operatorname{gr}_{\mathfrak{J}}^*(A)$ is a flat $A_0$-module, and that $Q$ and $R$ are flat $A$-modules.
Then the relation $\mathrm{H}_1(L_\bullet^{(0)})=0$ implies $\mathrm{H}_1(L_\bullet)=0$.
\end{lemma}

\begin{proof}
Set $L_\bullet^{(k)}=L_\bullet\otimes_A(A/\mathfrak{J}^{k+1}):P_k\xrightarrow{u_k}Q_k\xrightarrow{v_k}R_k$; since there exists an integer $k\geqslant 1$ such that $L_\bullet^{(k)}=L_\bullet$, it will suffice to prove, by induction on $k$, that the sequence $P_k\xrightarrow{u_k}Q_k\xrightarrow{v_k}R_k$ is
exact.
This is true by hypothesis for $k=0$.
The conclusion will follow from the following lemma:

\begin{lemma}[12.3.3.3]
\label{IV.12.3.3.3}
Let $B$ be a ring, $M$, $N$, $P$ three $B$-modules, $u:M\to N$, $v:N\to P$ two homomorphisms such that $u$ is injective, $v$ surjective and $v\circ u=0$.
Let $t$ be an element of $B$ such that $t$ be $P$-regular.
Then one has
\begin{equation}
\label{IV.12.3.3.4b}
\operatorname{Ker}(v\otimes 1)/\operatorname{Im}(u\otimes 1) = (\operatorname{Ker}(v)/\operatorname{Im}(u))\otimes_B(B/tB)
\end{equation}
up to a canonical isomorphism.
\end{lemma}

Indeed, the hypothesis that the sequence \sref{IV.12.3.3.2} is exact implies then that it becomes exact, and since $t$ belongs to the maximal ideal of $B$ and $\operatorname{Ker}(v)$ is a $B$-module of finite type (since $B$ is supposed Noetherian in \sref{IV.12.3.3.1}), the Nakayama lemma proves that $\operatorname{Ker}(v)=\operatorname{Im}(u)$.

It remains therefore to show \sref{IV.12.3.3.3}.
Set $u'=u\otimes 1_{B/tB}$, $v'=v\otimes 1_{B/tB}$, $Z=\operatorname{Ker}(v)$, $H=Z/\operatorname{Im}(u)$, so that one has the exact sequences
\begin{gather*}
  0\to M\to Z\to H\to 0\\
  0\to Z\to N\xrightarrow{v} P\to 0
\end{gather*}
whence, by tensorising by $B/tB$ and using the lemma \sref{IV.3.4.1.4} and the fact that $t$ is $P$-regular, the exact sequences
\begin{gather*}
  M/tM\xrightarrow{w'} Z/tZ\to H/tH\to 0\\
  0\to Z/tZ\to N/tN\xrightarrow{v'} P/tP\to 0
\end{gather*}
whence $\operatorname{Ker}(v')=Z/tZ$, and since $\operatorname{Im}(w')=\operatorname{Im}(u')$, one obtains \sref{IV.12.3.3.4b}.
\end{proof}
\end{proof}
