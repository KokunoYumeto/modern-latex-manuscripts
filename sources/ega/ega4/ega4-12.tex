\section{Study of fibres of finitely presented flat morphisms}
\label{section:IV.12}

\subsection{Introduction}
\label{subsection:IV.12.0}

\oldpage[IV-3]{173}

We shall use throughout this paragraph the general notations of (9.4.1).

\begin{env}[12.0.1]
\label{IV.12.0.1}
Given a morphism \emph{locally of finite presentation} $f:X\to Y$, we have seen (9.9) that for certain \emph{local} properties $\mathbf{P}$ of preschemes on corps or of Modules on these preschemes, the set $E$ of $x\in X$ such that the property $\mathbf{P}$ holds, for the fibre $X_{f(x)}$, at the point $x$ of this fibre, is \emph{locally constructible in $X$}.
We propose to show that, for most of these properties, if one furthermore supposes that the morphism $f$ is \emph{flat}, then $E$ is even \emph{open} in $X$.
Similarly ((9.2) to (9.8)) we have shown that if $f$ is of \emph{finite presentation}, and if this time $\mathbf{P}$ denotes certain \emph{global} properties of preschemes on corps or of Modules on these preschemes, the set $F$ of $y\in Y$ such that the property $\mathbf{P}$ holds for the fibre $X_y$ is \emph{locally constructible in $Y$}.
We shall show that if one furthermore supposes the morphism $f$ to be \emph{proper} and \emph{flat}, $F$ is even \emph{open} in $Y$.
\end{env}

\begin{env}[12.0.2]
\label{IV.12.0.2}
The general method of proof of the properties in question comprises three steps.
One first reduces to the case where $Y$ is \emph{affine} and $X$ of finite presentation; then:
\begin{enumerate}[label=\emph{\Alph*)}]
  \item Using (8.9.1) (and possibly other results of \textsection~8) and of (11.2.6), one reduces to the case where $X$ and $Y$ are \emph{Noetherian}.
  \item One applies the results of \textsection~9 recalled in (12.0.1) to prove that $E$ is \emph{constructible}.
  \item To see that $E$ is open, it suffices, by $(\mathbf{0_{III}}$, 9.2.5) to show that if $x\in E$, then every \emph{generisation} $x'$ of $x$ also belongs to $E$.
  Using (\textbf{II}, 7.1.7), one sees, since $Y$ is Noetherian, that there exists a spectrum of a \emph{discrete valuation ring} $Y_1$ and a morphism $h:Y_1\to X$ such that, if $y_1$ (resp.\ $y'_1$) is the closed point (resp.\ the generic point) of $Y_1$, one has $h(y_1)=x$ and $h(y'_1)=x'$.
  One then performs the base change $g=f\circ h:Y_1\to Y$; taking into account the results of \textsection\textsection~4 and~6 on preschemes locally Noetherian on corps and on base changes of the base field, one is reduced to proving the assertion in question for a point $x_1$ of $X_1=X\times_Y Y_1$ above $y_1$ and for a generisation $x'_1$ of $x_1$ above $y'_1$.
  As $Y_1=\Spec(A)$, where $A$ is a discrete valuation ring, $y$ the closed point and $y'$ the generic point of $Y$, the hypothesis that $\sh{F}$ is $f$-flat implies that $t$ is $\sh{F}$-regular $(\mathbf{0_I}$, 6.3.4).
  One uses the results of (3.4) and (5.12) for this.
\end{enumerate}
One proceeds similarly for the set $F\subset Y$.
\end{env}

\subsection{Local properties of fibres of a morphism that is flat and locally of finite presentation}
\label{subsection:IV.12.1}

\begin{theorem}[12.1.1]
\label{IV.12.1.1}
\oldpage[IV-3]{174}
Let $f:X\to Y$ be a morphism locally of finite presentation, $\sh{F}$ an $\sh{O}_X$-Module $f$-flat and of finite presentation, $\Phi$ a finite subset of $\bb{Z}\cup\{\pm\infty\}$, $k$ an integer.
The following subsets of $X$ are open:
\begin{enumerate}[label=(\roman*)]
  \item The set of $x\in X$ such that the dimensions of the prime cycles associated to $\sh{F}_{f(x)}$ and containing $x$ are elements of $\Phi$.
  \item The set of $x\in X$ such that the dimensions of the prime cycles associated to $\sh{F}_{f(x)}$ containing $x$ all have dimension $k$.
  \item The set of $x\in X$ such that $x$ does not belong to any immersed prime cycle associated to $\sh{F}_{f(x)}$.
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is equidimensional at the point $x$ and possesses the property $(\mathrm{S}_k)$ at $x$ (5.7.2).
  \item The set of $x\in X$ such that $\operatorname{coprof}((\sh{F}_{f(x)})_x)\leq k$ ($\mathbf{0}$, 16.4.9).
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is a Cohen-Macaulay $\sh{O}_{X_{f(x)}}$-Module at the point $x$ (5.7.1).
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is geometrically reduced at the point $x$ (4.6.22).
  \item The set of $x\in X$ such that $\sh{F}_{f(x)}$ is geometrically punctually integral at the point $x$ (4.6.22).
\end{enumerate}
\end{theorem}

\begin{proof}
The questions being local on $X$, one first reduces to the case where $X=\Spec(A)$ and $Y=\Spec(B)$ are affine, $f$ a morphism of finite presentation.
There is then a Noetherian sub-ring $A_0$ of $A$, a $A_0$-prescheme of finite type $X_0$, an $\sh{O}_{X_0}$-Module coherent $\sh{F}_0$ such that $\sh{F}_0\otimes_{A_0} A$ is isomorphic to $\sh{F}$ (8.9.1); in addition, by (11.2.6), one can suppose $\sh{F}_0$ is $Y_0$-flat (with $Y_0=\Spec(A_0)$).
If $h:X\to X_0$ is the canonical projection, the set $E$ of points $x\in X$ where one of the properties (i) to (viii) is verified corresponds to the set $E_0$ where the corresponding property holds for $\sh{F}_0$: this follows for properties (i) to (iii) from (4.2.7); for properties (iv) to (vi) from (6.7.1); for properties (vii) and (viii) from (4.7.11).

One is thus reduced to the case where $X$ and $Y$ are \emph{Noetherian}, and $\sh{F}=\widetilde{M}$, where $M$ is a $B$-module of finite type and the $N_\lambda$ are $A$-modules flat.
By (9.9.2), the set $E$ is \emph{constructible} for all the properties considered, and it remains in each case to verify step C) of (12.0.2).

\medskip\noindent\textbf{(12.1.1.1)}
Let us begin with case (iii), the simplest.
Replacing $Y$ by the integral sub-scheme having $\{y'\}$ for underlying space, one can restrict to the case where $Y$ is the spectrum of a discrete valuation ring, $y$ the closed point and $y'$ the generic point of $Y$.
The hypothesis that $\sh{F}$ is $f$-flat implies that $t$ is $\sh{F}$-regular $(\mathbf{0_I}$, 6.3.4).
By hypothesis, none of the immersed prime cycles of $\sh{F}/t\sh{F}$ contains $x$.
Then, it follows from (3.4.4) that $x$ does not belong to any immersed prime cycle associated to $\sh{F}$, and it follows likewise for every generisation $x'$ of $x$.
In particular, $x'$ does not belong to any of the immersed cycles associated to $\sh{F}_{y'}$ ($X_{y'}$ being a sub-prescheme induced on an open of $X$).

\medskip\noindent\textbf{(12.1.1.2)}
Let us consider the cases (iv) and (v) ((vi) is a particular case of (v)).
One proceeds as above (using (6.7.1) instead of (4.2.7)) and is again reduced to the case where $Y$ is the spectrum of a discrete valuation ring.
The ring $C=\sh{O}_{X,x}$ is then a localisation of an $A$-algebra of finite type, hence catenary (5.6.4), and by hypothesis the $C$-module $M_x/tM_x$ verifies $(\mathrm{S}_k)$; one deduces from (5.12.2) (resp.\ from ($\mathbf{0}$, 16.4.10)) that $M_x$ verifies $(\mathrm{S}_k)$ and is equidimensional (resp.\ $\operatorname{coprof}(M_x)\leq k$) since $t$ is $M_x$-regular and belongs to the maximal ideal of $C$.
Hence the conclusions since $x'$ is a generisation of $x$.

\medskip\noindent\textbf{(12.1.1.3)}
Let us pass to cases (vii) and (viii).
One can evidently replace $Y$ by the integral sub-scheme having $\{y'\}$ for underlying space, and $X$ by $f^{-1}(\{y'\})$, without changing the fibres in $y$ and $y'$, hence one can suppose $A$ integral, of field of fractions $K=\kappa(y')$.
There exists a finite extension $K'$ of $K$ such that, for the $(\sh{O}_X\otimes_A K')$-Module $\sh{F}'=\sh{F}\otimes_A K'$, the prime cycles associated are geometrically irreducible and the geometric lengths of its support are respectively equal to the lengths of $\sh{F}'_{z'_i}$ at its maximal points $z'_i$ of $\operatorname{Supp}(\sh{F}')$ which are closed.
It follows from (4.7.11) that $\sh{F}\otimes_A A'$ has property (vii) (resp.\ (viii)).
Replacing $A$ by $A'$ and $X$ by $X\otimes_A A'$, one sees that one can restrict to the case where $A$ is a discrete valuation ring, the uniformiser $t$ of $A$ being $M$-regular.
By hypothesis $\sh{F}/t\sh{F}$ is reduced (resp.\ integral) at the point $x$, it follows from (3.4.6) (resp.\ (3.4.5)) that $\sh{F}$ is reduced (resp.\ integral) at the point $x$, hence also in a neighbourhood of the point $x$, and in particular at the point $x'$, which completes the proof in cases (vii) and (viii).

\medskip\noindent\textbf{(12.1.1.4)}
It remains to examine cases (i) and (ii).
Replacing $Y$ by the integral sub-scheme having $\{y'\}$ for underlying space, one can restrict to the case where $Y$ is irreducible and $y'$ its generic point.
The detailed proof proceeds by reducing via (4.2.7) to the case where $Y$ is a discrete valuation ring, and then applying the dimension theory from (5.1.2), (4.5.6), (4.7.9), and the flatness hypothesis on $\sh{F}$.
We refer to the original for the complete argument.
\end{proof}

\begin{proposition}[12.1.5]
\label{IV.12.1.5}
\oldpage[IV-3]{176}
Let $Y$ be a prescheme locally Noetherian, $f:X\to Y$ a morphism locally of finite type, $\sh{F}$ an $\sh{O}_X$-Module coherent and $f$-flat; for every $y\in Y$, set $\sh{F}_y=\sh{F}\otimes_{\sh{O}_Y}\kappa(y)$.
Let $z'$ be a point of $X$, $y'=f(z')$, and suppose that $z'\in\operatorname{Ass}(\sh{F}_{y'})$.
Let $Z'$ (resp.\ $Y'$) be the sub-prescheme reduced of $X$ (resp.\ $Y$) having for underlying space the adherence of $\{z'\}$ (resp.\ $\{y'\}$) in $X$ (resp.\ $Y$).
Then, for every $y\in f(Z')$, the dimensions of all the irreducible components of $Z'_y$ are equal to $\dim(Z'_{y'})$ (dimension of the adherence of $z'$ in $X_{y'}$) (which we shall express later (13.2.2) by saying that the restriction $Z'\to Y'$ of $f$ is an \emph{equidimensional morphism}); moreover, at every maximal point $z$ of $Z'_y$, one has $z\in\operatorname{Ass}(\sh{F}_y)$.
\end{proposition}

\begin{proof}
We shall reduce to the case where $Y$ is the spectrum of a discrete valuation ring.
Take $Y_1$ a spectrum of discrete valuation ring $h:Y_1\to X$ a morphism such that, if $y_1$ and $y'_1$ are the closed point and the generic point of $Y_1$ respectively (\textbf{II}, 7.1.7), one has $h(y_1)=z'$, $h(y'_1)=y'$.
Set $g=f\circ h:Y_1\to Y$, $X_1=X\times_Y Y_1$; let $f_1:X_1\to Y_1$, $g_1:X_1\to X$ be the projections canoniques; there is a $Y_1$-section $h_1:Y_1\to X_1$ such that $u=g_1\circ h_1$ (\textbf{I}, 3.3.14).
If $Z'_1=Z'_y\otimes_{\kappa(y')}\kappa(y'_1)$, one says (4.2.6) that the irreducible components of $Z'_1$ dominate $Z'_{y'}$; let $z'_1$ be the generic point of one of these components which contains $h_1(y'_1)$.
As $h_1(y'_1)$ is a specialisation of $z'_1$, it is \emph{a fortiori} a specialisation of $z'_1$; let $z_1$ be a generic point of one of the irreducible components of $\{z'_1\}\cap(X_1)_{y_1}$ containing $h_1(y_1)$.
Then $g_1(z_1)$ is a specialisation of $g_1(z'_1)=z'$ in $X$, and $g_1(z_1)\in X_y$; but since $f(g_1(z_1))=y$ and $z$ is a generic point of $Z'_y$, we have necessarily $z=g_1(z_1)$.

Now suppose that $z_1\in\operatorname{Ass}((\sh{F}_1)_{y_1})$, where $\sh{F}_1=\sh{F}\otimes_{\sh{O}_Y}\sh{O}_{Y_1}$.
Since $\sh{F}$ is $f$-flat, $t$ is $\sh{F}_1$-regular by ($\mathbf{0_I}$, 6.3.4).
By (3.4.3) we deduce that $z'_1\in\operatorname{Ass}(\sh{F}_{y'})$.
As the points of $\operatorname{Ass}(\sh{F}_y)$ are the $z_i$, this concludes the proof.
\end{proof}

\begin{theorem}[12.1.6]
\label{IV.12.1.6}
\oldpage[IV-3]{178}
Let $f:X\to Y$ be a morphism flat and locally of finite presentation, $k$ an integer.
The following subsets of $X$ are open:
\begin{enumerate}[label=(\roman*)]
  \item The set of $x\in X$ such that $X_{f(x)}$ verifies the property $(\mathrm{S}_k)$ at the point $x$.
  \item The set of $x\in X$ such that $X_{f(x)}$ verifies the property $(\mathrm{R}_k)$ geometric at the point $x$, is equidimensional and such that $x$ does not belong to any immersed prime cycle associated to $X_{f(x)}$.
  \item The set of $x\in X$ such that $X_{f(x)}$ is geometrically regular (i.e.\ lisse over $\kappa(y)$) at the point $x$.
  \item The set of $x\in X$ such that $X_{f(x)}$ is geometrically normal at the point $x$.
\end{enumerate}
\end{theorem}

\begin{proof}
Steps A) and B) of (12.0.2) are performed as in (12.1.1); for step A), one uses (6.7.8), as well as (4.2.7); for step B), one uses (9.9.2) and (9.9.4).
It remains to examine step C) in each case.

For (iii), step C is unnecessary; as $f$ is flat, one knows that (6.8.7) when $Y$ is locally Noetherian and $f$ locally of finite type, the set of $x\in X$ such that $X_{f(x)}$ is geometrically regular at the point $x$ is open in $X$.

For (i), as in (12.1.1.2), one reduces to the case where $Y$ is the spectrum of a discrete valuation ring $A$, where $B$ is an $A$-algebra of finite type, $x$, $x'$ two points of $X$ such that $f(x)=y$ the closed point, $f(x')=y'$ the generic point of $Y$.
The hypothesis that $\sh{F}$ is $f$-flat implies $t$ is $B$-regular by flatness, and $B/tB$ verifies $(\mathrm{S}_k)$, hence by (5.12.4) $B$ verifies $(\mathrm{S}_k)$.
As $A$ is a universally catenary ring, this suffices to conclude (5.12.5).

For (ii), as in (12.1.1.3), to prove that the property considered is stable by generisation, one can, by considering a finite extension $K'$ of $\kappa(y')$, and the invariance by change of the base field of the two other properties in (ii), replace in (ii) the property $(\mathrm{R}_k)$ geometric by the property $(\mathrm{R}_k)$.
Proceeding then as in (i), one reduces to the case where $Y$ is the spectrum of a discrete valuation ring, and since $A$ is catenary, it suffices to apply (5.12.5) to conclude.

For (iv), the set of points $x\in X$ such that $X_{f(x)}$ is geometrically normal at the point $x$ is contained in the set of $x\in X$ such that $X_{f(x)}$ is geometrically punctually integral and verifies $(\mathrm{S}_2)$ at $x$, and this latter is open in $X$ by (12.1.1, (viii)).
One can thus already suppose that, for every $x\in X$, $X_{f(x)}$ is geometrically punctually integral and verifies $(\mathrm{S}_2)$ at $x$.
By the Serre criterion (5.8.6), saying that $X_{f(x)}$ is geometrically normal at $x$ amounts to $X_{f(x)}$ verifying $(\mathrm{S}_2)$ and the property $(\mathrm{R}_1)$ geometric at $x$;
by (ii) and the remarks preceding, this ensemble is intersection of two opens of $X$, hence is open in $X$. C.Q.F.D.
\end{proof}

\begin{corollary}[12.1.7]
\label{IV.12.1.7}
\oldpage[IV-3]{179}
Let $f:X\to Y$ be a morphism locally of finite presentation.
Then the set of points $x\in X$ where $f$ possesses one of the following properties (6.8.1):
\begin{enumerate}[label=(\roman*)]
  \item to verify the property $(\mathrm{S}_k)$;
  \item to be of coprofondeur $\leq n$;
  \item to be Cohen-Macaulay;
  \item to be regular (or lisse, which amounts to the same);
  \item to be normal;
  \item to be reduced;
\end{enumerate}
is open in $X$.
\end{corollary}

\begin{proof}
Indeed, it follows from (11.3.1) that the set of $x\in X$ where $f$ is \emph{flat} is open.
One can thus restrict to the case where $f$ is flat, and then the corollary results from (12.1.1, (iv), (vi) and (vii)) and from (12.1.6, (i), (ii) and (iv)).
\end{proof}

\begin{remarks}[12.1.8]
\label{IV.12.1.8}
\begin{enumerate}[label=(\roman*)]
  \item In (12.1.6, (ii)), one cannot suppress the hypothesis that $x$ does not belong to any immersed prime cycle of $X_{f(x)}$.
  This is shown by the example (5.12.3), where one takes $X=\Spec(A)$, $Y$ the spectrum of the local ring $A_0$ of $k[T]$ at the prime ideal $(T)$ and the morphism $X\to Y$ corresponding to the injection $A_0\to A$, deduced by localisation and passage to the quotient of the injection $k[t_0,t_2]\to k[t_0,t_1,t_2,t_3]$.
  \item In the hypotheses of (12.1.1.5), with $\sh{F}=\sh{O}_X$ (so that $f$ is flat), one cannot in general weaken the condition ``$f$ is flat'' to ``$f$ is universally open'' (cf.\ (2.4.6)), as we shall see later in an example (14.4.10, (i)).
\end{enumerate}
\end{remarks}

\subsection{Local and global properties of fibres of a proper, flat morphism of finite presentation}
\label{subsection:IV.12.2}

\begin{theorem}[12.2.1]
\label{IV.12.2.1}
\oldpage[IV-3]{179}
Let $f:X\to Y$ be a morphism proper, flat and of finite presentation, $\sh{F}$ an $\sh{O}_X$-Module $f$-flat and of finite presentation, $\Phi$ a finite subset of $\bb{Z}\cup\{\pm\infty\}$, $k$ an integer.
The following subsets of $Y$ are open:
\begin{enumerate}[label=(\roman*)]
  \item The set of $y\in Y$ such that the dimensions of the prime cycles associated to $\sh{F}_y$ and containing $x$ are elements of $\Phi$.
  \item The set of $y\in Y$ such that the dimensions of the prime cycles associated to $\sh{F}_y$ containing $x$ all have the same dimension $k$.
  \item The set of $y\in Y$ such that $\sh{F}_y$ has no immersed prime cycle.
  \item The set of $y\in Y$ such that $\sh{F}_y$ is equidimensional at every point $x$ of $X_y$ and possesses property $(\mathrm{S}_k)$ at $x$ (5.7.2).
  \item The set of $y\in Y$ such that $\sh{F}_y$ possesses the property $(\mathrm{S}_k)$ and is equidimensional at every point of $X_y$.
  \item The set of $y\in Y$ such that $\operatorname{coprof}(\sh{F}_y)\leq k$.
  \item The set of $y\in Y$ such that $\sh{F}_y$ is a Cohen-Macaulay Module.
  \item The set of $y\in Y$ such that $\sh{F}_y$ is geometrically reduced.
  \item The set of $y\in Y$ such that $\sh{F}_y$ is geometrically punctually integral at every point of $X_y$.
  \item The set of $y\in Y$ such that $\sh{F}_y$ is geometrically integral.
  \item The set of $y\in Y$ such that $\sh{F}_y$ has no immersed prime cycle associated and that the sum $l(y)$ of the total multiplicities (4.7.12) of $\sh{F}_y$ at the maximal points of $\Supp(\sh{F}_y)$ is $\leq k$.
\end{enumerate}
\end{theorem}

\begin{proof}
With the exception of (ii), (iii), (iv), (x) and (xi), the properties $\mathbf{P}(y)$ considered are of the form: ``for every $x\in X_y$, $\sh{F}_y$ has the property $\mathbf{Q}(x)$'', where $\mathbf{Q}(x)$ is one of the properties (i) to (viii) of (12.1.1).
It follows from (12.1.1) that the set $U$ of $x\in X$ such that $\mathbf{Q}(x)$ holds is open, and the set $V$ of $y\in Y$ such that $\mathbf{P}(y)$ holds is none other than $Y-f(X-U)$; in all these cases, the theorem is thus already true by only supposing the morphism $f$ \emph{closed} (in particular proper).

For (iii), (i) with $\Phi$ reduced to a single element, it follows from (12.1.1) that the set $U$ of $x\in X$ such that $\mathbf{Q}(x)$ holds is open, and by (10.8.1) the function $x\mapsto\operatorname{dim}(\Supp(\sh{F}_{f(x)}))$ is semi-continuous inferiorly in $X$.
The proof for the remaining cases (ii), (iv), (x), (xi) proceeds by reducing to the case where $Y$ is Noetherian using steps A), B), C) of (12.0.2).

\medskip\noindent\textbf{(12.2.1.1)}
Case (ii): The same reasoning as at the beginning shows, using (12.1.1, (iii)) that one can already suppose that for every $y\in Y$, $\sh{F}_y$ has no immersed associated prime cycle.
It follows from the assertions in cases (i) and (ii).

\medskip\noindent\textbf{(12.2.1.2)}
Case (iv): The same reasoning as at the beginning shows, using (12.1.1, (iii)) that one can already suppose (replacing $Y$ by an open of $Y$) that for every $y\in Y$, $\sh{F}_y$ has no immersed associated prime cycle.
The assertion of case (iv) is then an immediate consequence of cases (i) and (ii).

\medskip\noindent\textbf{(12.2.1.3)}
Case (xi): For step A) of the reasoning, one uses (8.9.1), (11.2.6), (8.10.5, (xii)) (to preserve the hypothesis that $f$ is proper), as well as (4.2.7), (4.5.6) and (4.7.9).
For step B), one sees, as in case (iv), that one can suppose that for every $y\in Y$, $\sh{F}_y$ has no immersed associated prime cycle, and one applies (9.8.8) which proves that the function $z\mapsto l(z)$ is constructible.
It remains to verify (assuming $Y$ Noetherian and $f$ proper) that for every generisation $y'$ of a point $y\in Y$, one has $l(y')\leq l(y)$.
The proof uses a key lemma:
\end{proof}

\begin{lemma}[12.2.1.4]
\label{IV.12.2.1.4}
Let $Y$ be the spectrum of a discrete valuation ring, $y$ its closed point, $y'$ its generic point, $f:X\to Y$ a proper morphism, $\sh{F}$ an $\sh{O}_X$-Module coherent and $f$-flat, $z_i$ (resp.\ $z'_j$) the maximal points of $\Supp(\sh{F}_y)$ (resp.\ $\Supp(\sh{F}_{y'})$).
Suppose that $\sh{F}_{y'}$ has no immersed prime cycle associated.
Then one has
\[
  \sum_j \operatorname{long}((\sh{F}_{y'})_{z'_j})\leq\sum_i \operatorname{long}((\sh{F}_y)_{z_i}).
\]
\end{lemma}

\begin{corollary}[12.2.2]
\label{IV.12.2.2}
\oldpage[IV-3]{181}
Under the hypotheses of \sref{IV.12.2.1}:
\begin{enumerate}[label=(\roman*)]
  \item The function $y\mapsto\dim(\Supp(\sh{F}_y))$ is continuous (hence locally constant) in $Y$.
  \item The function $y\mapsto\operatorname{coprof}(\sh{F}_y)$ (5.7.1) is semi-continuous superiorly in $Y$.
  \item The function $y\mapsto l(y)$ \sref{IV.12.2.1}, (xi)) is semi-continuous superiorly in $Y$ when the $\sh{F}_y$ have no immersed prime cycle associated.
  \item The function $y\mapsto\operatorname{prof}^*(\sh{F}_y)$ (10.8.1) is semi-continuous inferiorly in $Y$.
\end{enumerate}
\end{corollary}

\begin{proof}
(i) follows from \sref{IV.12.2.1} applied to $\Phi$ equal to the smallest interval of $\bb{N}$ containing the dimensions of the prime cycles associated to $\sh{F}_y$, and noting that $z\mapsto\dim(\Supp(\sh{F}_z))$ is semi-continuous superiorly at the point $y$; it also follows from \sref{IV.12.2.1}, (ii) applied to the set of dimensions of the irreducible components of $\Supp(\sh{F}_y)$, that this function is also semi-continuous inferiorly at the point $y$, hence the conclusion.
Assertions (ii) and (iii) are other formulations of \sref{IV.12.2.1}, (vi) and (xi).
Finally, from \sref{IV.12.1.5}, the set of $x\in X$ such that $\operatorname{prof}^*_x(\sh{F}_{f(x)})\leq k$ is open, and the conclusion of (iv) follows by the same reasoning as at the beginning of \sref{IV.12.2.1}.
\end{proof}

\begin{remarks}[12.2.3]
\label{IV.12.2.3}
\oldpage[IV-3]{182}
\begin{enumerate}[label=(\roman*)]
  \item For \sref{IV.12.2.1}, (ii), one can dispense from the hypothesis that $f$ is \emph{proper}.
  \item In \sref{IV.12.2.1}, (ii), one cannot replace the condition that the set $E(y)$ of dimensions of irreducible components of $\Supp(\sh{F}_y)$ be contained in $\Phi$ by the condition that it be \emph{equal} to $\Phi$.
  \item The Remarks (ii) and (iii) show the necessity of including in \sref{IV.12.2.1}, (iv)) the condition that $\sh{F}_y$ has no immersed prime cycle associated.
  \item In \sref{IV.12.2.1}, (xi), one cannot suppress the hypothesis that $f$ be proper.
  \item In \sref{IV.12.2.1}, (xi), one cannot suppress the hypothesis that $\sh{F}_y$ has no immersed associated cycle.
  \item We do not know if, in \sref{IV.12.2.1}, (xi)), one can replace the hypothesis that $\sh{F}$ is $f$-flat by that $\Supp(\sh{F})=X$ and that $f$ is universally open.
\end{enumerate}
\end{remarks}

\begin{theorem}[12.2.4]
\label{IV.12.2.4}
\oldpage[IV-3]{183}
Let $f:X\to Y$ be a morphism proper, flat and of finite presentation, $k$ an integer $\geq 1$.
The following subsets of $Y$ are open:
\begin{enumerate}[label=(\roman*)]
  \item The set of $y\in Y$ such that $X_y$ possesses the property $(\mathrm{S}_k)$.
  \item The set of $y\in Y$ such that $X_y$ verifies the property $(\mathrm{R}_k)$ geometric, is equidimensional at every point and has no immersed prime cycle associated to $X_y$.
  \item The set of $y\in Y$ such that $X_y$ is geometrically regular (i.e.\ lisse over $\kappa(y)$).
  \item The set of $y\in Y$ such that $X_y$ is geometrically normal.
  \item The set of $y\in Y$ such that $X_y$ is geometrically reduced.
  \item The set of $y\in Y$ such that $X_y$ is geometrically reduced and the number of geometric connected components of $X_y$ over $\kappa(y)$ is equal to $k$.
  \item The set of $y\in Y$ such that $X_y$ is geometrically punctually integral (4.6.9).
  \item The set of $y\in Y$ such that $X_y$ is geometrically integral.
  \item The set of $y\in Y$ such that $X_y$ has no immersed prime cycle associated and that the total multiplicity (4.7.4) of $X_y$ over $\kappa(y)$ is $\leq k$.
\end{enumerate}
\end{theorem}

\begin{proof}
Except for (vi), these assertions are particular cases of assertions of \sref{IV.12.2.1}, or are deduced from assertions of (12.1.6) as at the beginning of the proof of \sref{IV.12.2.1}.
For (vi), one reduces as always (taking into account the invariance of the geometric number of connected components by base change (4.5.6)) to the case where $Y$ is Noetherian.
The set $U$ of $y\in Y$ such that $X_y$ is geometrically reduced is open in $Y$ by (v); it follows then from (\textbf{III}, 7.8.7 and 7.8.6) that there is a neighbourhood $V\subset U$ of $y$ and an integer $m$ such that, for every $z\in V$, $\Gamma(X_z,\sh{O}_{X_z})$ is isomorphic to $(\kappa(z))^m$; but by (\textbf{III}, 4.3.4), $m$ is then the number of geometric connected components of $X_z$, hence the conclusion.
One will give another proof of (vi) in (15.5.9).
\end{proof}

\subsection{Local cohomological properties of fibres of a flat morphism locally of finite presentation}
\label{subsection:IV.12.3}

\begin{lemma}[12.3.1]
\label{IV.12.3.1}
\oldpage[IV-3]{183}
Let $A$ be a ring, $B$ a $A$-algebra of finite presentation which is an $A$-module flat, $M$ a $B$-module of finite presentation, which is an $A$-module flat.
Then the following conditions are equivalent:
\begin{enumerate}[label=\alph*)]
  \item $M$ is a $B$-module \emph{projective}.
  \item $M$ is a $B$-module \emph{flat}.
  \item For every $y\in\Spec(A)$, $M\otimes_A\kappa(y)$ is a $(B\otimes_A\kappa(y))$-module \emph{projective}.
\end{enumerate}
\end{lemma}

\begin{proof}
The equivalence of $a)$ and $b)$ follows from Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection~5, n\textsuperscript{o}~2, cor.~2 du th.~1.
As $a)$ implies $c)$ trivially, it remains to prove that $c)$ implies $b)$, which follows from the criterion of flatness by fibres (11.3.10), applied with $g=h$, $f=1_X$.
\end{proof}

\begin{proposition}[12.3.2]
\label{IV.12.3.2}
\oldpage[IV-3]{184}
Let $A$ be a ring, $B$ a $A$-algebra of finite presentation which is an $A$-module flat, $M$ a $B$-module of finite presentation which is an $A$-module flat.
Then:
\begin{enumerate}[label=(\roman*)]
  \item There exists a left resolution of $M$ by $B$-modules free of finite type.
  \item One has
  \[
  \label{IV.12.3.2.1}\tag{12.3.2.1}
    \operatorname{dim.proj}_B(M)=\sup_{y\in\Spec(A)} \operatorname{dim.proj}_{B\otimes_A\kappa(y)}(M\otimes_A\kappa(y))=\operatorname{Tor.dim}_B(M)
  \]
  where $\operatorname{Tor.dim}_B(M)$ is the smallest integer $i$ such that $\operatorname{Tor}^B_j(M,N)=0$ for every $j\geq i$ and every $B$-module $N$ (and $+\infty$ if no such integer $i$ exists).
\end{enumerate}
\end{proposition}

\begin{proof}
(i) By (8.9.1), there exist a Noetherian sub-ring $A_0$ of $A$, an $A_0$-algebra of finite type $B_0$ and a $B_0$-module of finite type $M_0$ such that $B$ is isomorphic to $B_0\otimes_{A_0} A$ and $M$ to $M_0\otimes_{A_0} A$.
Moreover by (11.2.7) one can suppose that $B_0$ and $M_0$ are $A_0$-modules flat.
There is then a left resolution $(L_0)_*$ of $M_0$ by $B_0$-modules free of finite type, $(L_0)_*\otimes_{A_0} A$ is also a resolution of $M$ formed by $B$-modules free of finite type (2.1.10).

(ii) By ($\mathbf{0}$, 17.2.2, (ii)), on a $\operatorname{Tor.dim}_B(M)\leq\operatorname{dim.proj}_B(M)$, and the definition of projective dimension proves immediately that, for every $y\in\Spec(A)$, one has $\operatorname{dim.proj}_{B\otimes_A\kappa(y)}(M\otimes_A\kappa(y))\leq\operatorname{dim.proj}_B(M)$.
To prove the inverse inequalities, consider a left resolution $(L_i)$ of $M$ by $B$-modules free of finite type, and suppose that $\operatorname{Tor.dim}_B(M)=n$ (resp.\ $\operatorname{dim.proj}_{B\otimes_A\kappa(y)}(M\otimes_A\kappa(y))\leq n$ for every $y\in\Spec(A)$).
Then $R=\operatorname{Im}(L_n\to L_{n-1})=\Ker(L_{n-1}\to L_{n-2})$ is also a $B$-module of finite type which is also an $A$-module flat, by the hypothesis on $M$ and by (2.1.10).
Moreover, on has $\operatorname{Tor}^B_{n+1}(M,N)=\operatorname{Tor}^B_1(R,N)$ for every $B$-module $N$ (M, V, 7).
The hypothesis $\operatorname{Tor.dim}_B(M)=n$ implies $\operatorname{Tor}^B_1(R,N)=0$ for every $B$-module $N$, i.e.\ $R$ is a $B$-module flat, hence projective by \sref{IV.12.3.1}; this establishes that $\operatorname{dim.proj}_B(M)\leq n$.
Similarly, the hypothesis $\operatorname{dim.proj}_{B\otimes_A\kappa(y)}(M\otimes_A\kappa(y))\leq n$ for every $y\in\Spec(A)$ implies, by (2.1.10), that in each of the suites (exact by the flatness of $A$ on $M$, of the $L_i$ and of $R$)
\[
  0\to R\otimes_A\kappa(y)\to L_{n-1}\otimes_A\kappa(y)\to\ldots\to L_0\otimes_A\kappa(y)\to M\otimes_A\kappa(y)\to 0
\]
$R\otimes_A\kappa(y)$ is a $(B\otimes_A\kappa(y))$-module projective (for every $y\in\Spec(A)$).
One concludes then also that $R$ is a $B$-module projective, hence $\operatorname{dim.proj}_B(M)\leq n$.
\end{proof}

\begin{proposition}[12.3.3]
\label{IV.12.3.3}
Let $f:X\to S$ be a morphism locally of finite presentation, $\sh{L}_\bullet$ a complex of $\sh{O}_X$-Modules quasi-coherent of finite presentation; for every $s\in S$, let $(\sh{L}_\bullet)_s$ the complex $\sh{L}_\bullet\otimes_{\sh{O}_S}\kappa(s)$ of $\sh{O}_{X_s}$-Modules of finite type.
Suppose that $\sh{L}_{n-1}$ is $f$-flat.
Then the set $U$ of $x\in X$ such that $(\sh{H}_n((\sh{L}_\bullet)_{f(x)}))_x=0$ is open in $X$.
If moreover $\sh{L}_n$ is $f$-flat, then on a $\sh{H}_n(\sh{L}_\bullet)|U=0$.
\end{proposition}

\begin{proof}
One can evidently restrict to the case where $n=1$ and where the complex $\sh{L}_\bullet$ reduces to $0\to\sh{L}_2\xrightarrow{u}\sh{L}_1\xrightarrow{v}\sh{L}_0\to 0$.
One can first reduce to the case where $S=\Spec(A)$ and $X$ are affine, then to the case where $S$ is \emph{Noetherian}; indeed, by (8.9.1) and (8.5.2), one knows that there exists a Noetherian prescheme $S'=\Spec(A')$, where $A'$ is a sub-ring of $A$, a morphism of finite type $f':X'\to S'$ and three coherent $\sh{O}_{X'}$-Modules $\sh{L}'_i$ ($0\leq i\leq 2$), $f=f'_{(S)}$, $\sh{L}_i=\sh{L}'_i\otimes_{S'} S$, and two homomorphisms $u'$, $v'$ such that $u=u'\otimes 1$, $v=v'\otimes 1$ and $v'\circ u'=0$.
One can also suppose that $\sh{L}'_0$ is $f'$-flat (11.2.7), and that $\sh{L}'_1$ is $f'$-flat.
By faithful flatness, the hypothesis $(\sh{H}_n((\sh{L}_\bullet)_{f(x)}))_x=0$ is equivalent to $(\sh{H}_n((\sh{L}'_\bullet)_{f'(x')}))_{x'}=0$, where $x'$ is the projection of $x$ into $X'$; if $U'$ is the set of the $x'\in X'$ such that $(\sh{H}_n((\sh{L}'_\bullet)_{f'(x')}))_{x'}=0$, $U$ is thus the reciprocal image of $U'$ by the projection $X\to X'$, which reduces us to the first assertion for the case where $S$ is Noetherian.

For the second assertion, one notes also that $A$ is the inductive limit of its sub-$A'$-algebras of finite type; setting $X_\lambda=X'\times_{S'} S_\lambda$, $\sh{L}^{(\lambda)}_i=\sh{L}'_i\otimes_{S'} S_\lambda$, and $u_\lambda$, $v_\lambda$ the homomorphisms obtained from $u_\lambda=u'\otimes 1:\sh{L}^{(\lambda)}_2\to\sh{L}^{(\lambda)}_1$, $v_\lambda=v'\otimes 1:\sh{L}^{(\lambda)}_1\to\sh{L}^{(\lambda)}_0$, so that one has $\sh{H}_1(\sh{L}^{(\lambda)}_\bullet)=\Ker(v_\lambda)/\operatorname{Im}(u_\lambda)$.
Since in the category of commutative groups, the $\varinjlim$ functor is exact, one has $\Ker(v)=\varinjlim(\Ker(v_\lambda))$, $\operatorname{Im}(u)=\varinjlim(\operatorname{Im}(u_\lambda))$, and $\sh{H}_1(\sh{L}_\bullet)=\varinjlim\sh{H}_1(\sh{L}^{(\lambda)}_\bullet)$.
If one has supposed that $\sh{L}_1$ is $f$-flat and (reducing to the case $X=U$) that $\sh{H}_1(\sh{L}^{(\lambda)}_\bullet)=0$ for every $\lambda$, one deduces $\sh{H}_1(\sh{L}_\bullet)=0$.

I) Now suppose $S$ Noetherian.
One knows (without flatness hypothesis on $\sh{L}_0$) that the set $U$ is \emph{constructible} in $X$ (9.9.6).
Using now (12.0.2), the method described in (12.0.2) also applies without change (taking into account the fact that for a prescheme $Z$ over a corps $k$ and an extension $k'$ of $k$, the projection $Z\otimes_k k'\to Z$ is faithfully flat).
One can thus suppose that $S$ is the spectrum of a discrete valuation ring $A$, of which we designate by $t$ a uniformiser, $x$ being above the closed point of $S$ and $x'$ above the generic point of $S$.
The hypothesis that $\sh{L}_0$ is $f$-flat implies that $t$ is $\sh{L}_0$-regular $(\mathbf{0_I}$, 6.3.4).
One is then reduced to proving the following lemma (where $B$, $M$, $N$, $P$ are replaced by $\sh{O}_x$, $(\sh{L}_2)_x$, $(\sh{L}_1)_x$ and $(\sh{L}_0)_x$ respectively):
\end{proof}

\begin{lemma}[12.3.3.1]
\label{IV.12.3.3.1}
\oldpage[IV-3]{185}
Let $B$ be a local Noetherian ring, $M$, $N$, $P$ three $B$-modules, $N$ being supposed of finite type, $u:M\to N$, $v:N\to P$ two homomorphisms such that $v\circ u=0$.
Let $t$ be an element of the maximal ideal of $B$ such that $t$ is $P$-regular and such that the sequence
\[
\label{IV.12.3.3.2}\tag{12.3.3.2}
  M/tM\xrightarrow{u\otimes 1_{B/tB}} N/tN\xrightarrow{v\otimes 1_{B/tB}} P/tP
\]
is exact.
Then the sequence $M\xrightarrow{u}N\xrightarrow{v}P$ is exact.
\end{lemma}

\begin{proof}
Note first that if one replaces $M$ by its image $Q$ in $N$ and $u$ by the injection $j:Q\to N$, the image of $j\otimes 1$ is the same as that of $u\otimes 1$, so one can, without changing the hypothesis or the conclusion, suppose $u$ injective.
On the other hand, if $R$ is the image of $v$ and $p:N\to R$ the canonical surjection, $t$ is evidently $R$-regular, and the kernel of $p\otimes 1$ is contained in that of $v\otimes 1$; it is therefore equal to it if the sequence \sref{IV.12.3.3.2} is exact; as on the other hand one evidently has $\Ker(p)=\Ker(v)$, one sees that one can, for proving the lemma, suppose in addition $v$ surjective.
The lemma will then be a consequence of the following:
\end{proof}

\begin{lemma}[12.3.3.3]
\label{IV.12.3.3.3}
Let $B$ be a ring, $M$, $N$, $P$ three $B$-modules, $u:M\to N$, $v:N\to P$ two homomorphisms such that $u$ is injective, $v$ surjective and $v\circ u=0$.
Let $t$ be an element of $B$ such that $t$ is $P$-regular.
Then on a
\[
\label{IV.12.3.3.4}\tag{12.3.3.4}
  \Ker(v\otimes 1)/\operatorname{Im}(u\otimes 1)=(\Ker(v)/\operatorname{Im}(u))\otimes_B(B/tB)
\]
as a canonical isomorphism.
\end{lemma}

\begin{proof}
Indeed, the hypothesis that the sequence \sref{IV.12.3.3.2} is exact will then imply that $(\Ker v/\operatorname{Im} u)\otimes_B(B/tB)=0$, and as $t$ belongs to the maximal ideal of $B$ and $\Ker(v)/\operatorname{Im}(u)$ is a $B$-module of finite type (since $B$ is supposed Noetherian in \sref{IV.12.3.3.1}), Nakayama's lemma proves that $\Ker(v)=\operatorname{Im}(u)$.
It remains therefore to prove \sref{IV.12.3.3.3}.
Set $u'=u\otimes 1$, $v'=v\otimes 1$, $H=Z/\operatorname{Im}(u)$, $Z=\Ker(v)$, so that one has the exact sequences
\begin{gather*}
  0\to M\to Z\to H\to 0\\
  0\to Z\to N\xrightarrow{v} P\to 0
\end{gather*}
hence, tensoring by $B/tB$ and using the fact that $t$ is $P$-regular, the exact sequences
\begin{gather*}
  M/tM\xrightarrow{w'} Z/tZ\to H/tH\to 0\\
  0\to Z/tZ\to N/tN\xrightarrow{v'} P/tP\to 0
\end{gather*}
hence $\Ker(v')=Z/tZ$, and as $\operatorname{Im}(w')=\operatorname{Im}(u')$, one obtains \sref{IV.12.3.3.4}.
\end{proof}

\begin{env}[12.3.3.5]
\label{IV.12.3.3.5}
II) Suppose now also that $\sh{L}_1$ is $f$-flat; replacing eventually $X$ by $U$, one can suppose $X=U$.
It suffices to show that for every $x\in X$, one has $(\sh{H}_1(\sh{L}_\bullet))_x=0$.
Let $s=f(x)$, and let $\mathfrak{n}$ be the ideal $\mathfrak{m}_s\sh{O}_{X,x}$, which is contained in the maximal ideal of $\sh{O}_{X,x}$; as $(\sh{H}_1(\sh{L}_\bullet))_x$ is an $\sh{O}_{X,x}$-module of finite type ($X$ being locally Noetherian), it is separated for the $\mathfrak{n}$-adic topology $(\mathbf{0_I}$, 7.3.5), hence it suffices to show that its separated completion for this topology is $0$.
Now, by (\textbf{III}, 7.4.7.2), this separated completion is equal to $\varprojlim\sh{H}_1(\sh{L}_\bullet\otimes_{\sh{O}_S}(\sh{O}_{S,s}/\mathfrak{m}_s^{k+1}))$, and it suffices hence to prove that each of the terms of this projective system is zero.
But this is true by hypothesis for $k=0$; the conclusion follows by induction on $k$.
The conclusion will follow from the following more general lemma (which one will apply for $A=\sh{O}_{S,s}/\mathfrak{m}_s^{k+1}$ and $\mathfrak{J}$ equal to the maximal ideal $\mathfrak{m}_s/\mathfrak{m}_s^{k+1}$ of $A$):
\end{env}

\begin{lemma}[12.3.3.5]
\label{IV.12.3.3.5b}
Let $A$ be a ring, $\mathfrak{J}$ a nilpotent ideal of $A$, $L_\bullet:P\xrightarrow{u}Q\xrightarrow{v}R$ a complex of $A$-modules.
Set $A_0=A/\mathfrak{J}$, $L_\bullet^{(0)}=L_\bullet\otimes_A A_0:P_0\xrightarrow{u_0}Q_0\xrightarrow{v_0}R_0$, and suppose that $\operatorname{gr}^*_\mathfrak{J}(A)$ is an $A_0$-module flat, and that $Q$ and $R$ are $A$-modules flat.
Then the relation $H_1(L_\bullet^{(0)})=0$ implies $H_1(L_\bullet)=0$.
\end{lemma}

\begin{proof}
Set $L^{(k)}_\bullet=L_\bullet\otimes_A(A/\mathfrak{J}^{k+1}):P_k\xrightarrow{u_k}Q_k\xrightarrow{v_k}R_k$; as there exists an integer $k\geq 1$ such that $L^{(k)}_\bullet=L_\bullet$, it will suffice to prove, by induction on $k$, that the sequence $P_k\xrightarrow{u_k}Q_k\xrightarrow{v_k}R_k$ is exact (which one applies for $A=\sh{O}_{S,s}/\mathfrak{m}_s^{k+1}$ and $\mathfrak{J}$ equal to the maximal ideal $\mathfrak{m}_s/\mathfrak{m}_s^{k+1}$ of $A$).
Let $x\in Q_k$ be an element such that $v_k(x)=0$.
By the induction hypothesis, the canonical image of $x$ in $Q_{k-1}=Q_k/(\mathfrak{J}^k Q/\mathfrak{J}^{k+1}Q)$ belongs to $\operatorname{Im}(u_{k-1})$, hence there exists $z\in P_k$ such that $x=u_k(z)+x'$ with $x'\in\mathfrak{J}^k Q/\mathfrak{J}^{k+1}Q$.
Since $v_k\circ u_k=0$, the relation $v_k(x)=0$ implies $v_k(x')=0$, and the $v_k(x')\in\mathfrak{J}^k R/\mathfrak{J}^{k+1}R$; but by hypothesis, $\mathfrak{J}^k/\mathfrak{J}^{k+1}$ is an $(A/\mathfrak{J})$-module flat, hence the suite
\[
  \mathfrak{J}^k P/\mathfrak{J}^{k+1}P\xrightarrow{u''}\mathfrak{J}^k Q/\mathfrak{J}^{k+1}Q\xrightarrow{v''}\mathfrak{J}^k R/\mathfrak{J}^{k+1}R
\]
(where $u'$ and $v'$ come from $u$ and $v$ by restriction and passage to quotients) is exact.
Now, by hypothesis, $\mathfrak{J}^k/\mathfrak{J}^{k+1}$ is an $(A/\mathfrak{J})$-module flat, hence the suite $(Q/\mathfrak{J} Q)\otimes_{A/\mathfrak{J}}(\mathfrak{J}^k/\mathfrak{J}^{k+1})$ identifies with $\mathfrak{J}^k Q/\mathfrak{J}^{k+1}Q$ by the flatness of $Q$ on $A$, and similarly $(R/\mathfrak{J} R)\otimes_{A/\mathfrak{J}}(\mathfrak{J}^k/\mathfrak{J}^{k+1})$ identifies with $\mathfrak{J}^k R/\mathfrak{J}^{k+1}R$. The image of $u'$ then identifies with that of $u''$, and this concludes the proof.
\end{proof}

\begin{corollary}[12.3.4]
\label{IV.12.3.4}
\oldpage[IV-3]{187}
Let $f:X\to S$ be a morphism flat locally of finite presentation, $\sh{F}$, $\sh{G}$ two $\sh{O}_X$-Modules of finite presentation and $f$-flat.
Let $n$ be an integer $\geq 0$, $U$ the set of $x\in X$ such that $(\mathscr{E}\!xt^n_{\sh{O}_{X_{f(x)}}}(\sh{F}_{f(x)},\sh{G}_{f(x)}))_x=0$ (resp.\ $(\mathscr{T}\!or_n^{\sh{O}_{X_{f(x)}}}(\sh{F}_{f(x)},\sh{G}_{f(x)}))_x=0$).
Then $U$ is open, and one has $\mathscr{E}\!xt^n_{\sh{O}_X}(\sh{F},\sh{G})|U=0$ (resp.\ $\mathscr{T}\!or_n^{\sh{O}_X}(\sh{F},\sh{G})|U=0$).
\end{corollary}

\begin{proof}
One can evidently restrict to the case where $S$ and $X$ are affine.
Then \sref{IV.12.3.2} gives a left resolution $\sh{L}_\bullet=(\sh{L}_i)$ of $\sh{F}$ formed by $\sh{O}_X$-Modules \emph{free} of finite type.
If one sets $\sh{M}^\bullet=\mathscr{H}\!om(\sh{L}_\bullet,\sh{G})$ (resp.\ $\sh{N}_\bullet=\sh{L}_\bullet\otimes_{\sh{O}_X}\sh{G}$), one deduces that each $\sh{M}^i$ (resp.\ $\sh{N}_i$) is isomorphic to an $\sh{O}_X$-Module of the form $\sh{G}^m$, hence the $\sh{M}^i$ and $\sh{N}_i$ are $\sh{O}_X$-Modules of finite presentation and $f$-flat.
Moreover, for every base change $S'\to S$, if one sets $X'=X\times_S S'$, $\sh{F}'=\sh{F}\otimes_S S'$, $\sh{G}'=\sh{G}\otimes_S S'$, $\sh{L}'_\bullet=\sh{L}_\bullet\otimes_S S'$ is still a left resolution of $\sh{F}'$ by $\sh{O}_{X'}$-Modules free of finite type (2.1.10), and $\sh{M}'^\bullet=\sh{M}^\bullet\otimes_S S'$ is equal to $\mathscr{H}\!om(\sh{L}'_\bullet,\sh{G}')$ (resp.\ $\sh{N}'_\bullet=\sh{N}_\bullet\otimes_S S'$ is equal to $\sh{L}'_\bullet\otimes_{\sh{O}_{X'}}\sh{G}'$) from what precedes.
In particular, for every $s\in S$, $\mathscr{E}\!xt^n_{X_s}(\sh{F}_s,\sh{G}_s)=\sh{H}^n((\sh{M}^\bullet)_s)$ (resp.\ $\mathscr{T}\!or_n^{X_s}(\sh{F}_s,\sh{G}_s)=\sh{H}_n((\sh{N}_\bullet)_s)$).
Applying \sref{IV.12.3.3} to the $f$-flat complexes $\sh{M}^\bullet$ and $\sh{N}_\bullet$, one deduces immediately the corollary.
\end{proof}
