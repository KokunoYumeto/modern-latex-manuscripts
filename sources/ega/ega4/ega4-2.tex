\section{Base change and flatness}
\label{section:IV.2}

\oldpage[IV-2]{5}

This section (unlike \textsection~6) only exceptionally makes use of Noetherian techniques.
The n\textsuperscript{os}~1 and~2 are essentially nothing more than translations of elementary properties of flatness in commutative Algebra (cf.~Bourbaki, \emph{Alg.~comm.}, chap.~I\textsuperscript{er}), and are included here for the convenience of references.
The following numbers are primarily devoted to properties of ``descent'' by flat or faithfully flat morphisms: if $g:Y'\to Y$ is such a morphism, we want to be able to assert that a part of~$Y$, or an $\sh{O}_Y$-Module, or a morphism $X\to Y$, has a certain property, when we know that its \emph{inverse image} by $g$ possesses this property.
We limit ourselves here to properties that do not make use of the general technique of ``descent'', which will be developed in Chapter~V.

\subsection{Flat Modules on preschemes}
\label{subsection:IV.2.1}

\begin{env}[2.1.1]
\label{IV.2.1.1}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ an $\sh{O}_X$-Module;
recall~\sref[0]{0.6.7.1} that $\sh{F}$ is said to be \emph{$f$-flat} (or \emph{$Y$-flat}) \emph{at a point} $x\in X$ if $\sh{F}_x$ is an $\sh{O}_{f(x)}$-module flat, \emph{$f$-flat} (or \emph{$Y$-flat}) if it is $f$-flat at every point $x\in X$, and finally that the morphism $f$ is said to be \emph{flat at the point} $x\in X$ (resp.\ \emph{flat}) if $\sh{O}_X$ is $f$-flat at the point $x$ (resp.\ $f$-flat).
When $f=1_X$, we will simply say that an $\sh{O}_X$-Module $\sh{F}$ is \emph{flat at the point} $x$ (resp.\ \emph{flat}) if it is $X$-flat at this point (resp.\ at every point $x\in X$), that is to say if $\sh{F}_x$ is a flat $\sh{O}_x$-module (resp.\ if this is the case for every $x\in X$).
Recall that we have shown~\sref[III]{III.1.4.15.1} the following property:
\end{env}

\begin{proposition}[2.1.2]
\label{IV.2.1.2}
Let $A$, $B$ be two rings, $\varphi:A\to B$ a ring homomorphism, $X=\Spec(B)$, $Y=\Spec(A)$, $f:X\to Y$ the corresponding morphism.
For $\sh{F}=\widetilde{M}$ to be $f$-flat, it is necessary and sufficient that $M$ be a flat $A$-module.
\end{proposition}

\begin{proposition}[2.1.3]
\label{IV.2.1.3}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module. The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item For every base change $g:Y'\to Y$, if we set $X'=X\times_Y Y'$, the functor $\sh{G}'\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ from the category of quasi-coherent $\sh{O}_{Y'}$-Modules to that of quasi-coherent $\sh{O}_{X'}$-Modules is exact.
  \item[\emph{a$'$)}] Condition~\emph{a)} is satisfied for all canonical morphisms $g:\Spec(\sh{O}_y)\to Y$ where $y$ ranges over $Y$.
  \item $\sh{F}$ is $f$-flat.
\end{enumerate}
\end{proposition}

\begin{proof}
The questions being local on $X$ and $Y$, we can restrict to the case where $Y=\Spec(B)$, $X=\Spec(A)$, $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module.
It is clear that \emph{a)} implies \emph{a$'$)};
condition~\emph{a$'$)} implies that for every $x\in X$, the functor $N\mapsto M_\mathfrak{n}\otimes_{B_\mathfrak{n}}N$ is exact in the category of $B_\mathfrak{n}$-modules, $\mathfrak{n}$ being the ideal $\mathfrak{j}_{f(x)}$ of $B$; this means that $M_\mathfrak{n}$ is a flat $B_\mathfrak{n}$-module, and it follows from~\sref[0]{0.6.3.3} and from~\sref{IV.2.1.2} that $\sh{F}$ is $f$-flat.
Finally, to see that \emph{b)} implies \emph{a)}, we can also restrict to the case where $Y'=\Spec(A')$ is affine and $\sh{G}'=\widetilde{N'}$, where $N'$ is an $A'$-module; the conclusion then follows from~\sref{IV.2.1.2} and from the definition of flatness, since $(M\otimes_A N')^{\sim}=\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$.
\end{proof}

\begin{proposition}[2.1.4]
\label{IV.2.1.4}
\oldpage[IV-2]{6}
Let $f:X\to Y$, $g:Y'\to Y$ be two morphisms of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module; set $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$, $\sh{F}'=\sh{F}\otimes_{\sh{O}_Y}\sh{O}_{Y'}$, $y=g(y')$, $y'=f'(x')$, $x=g'(x')$.
If $\sh{F}$ is $f$-flat at the point $x$, $\sh{F}'$ is $f'$-flat at the point $x'$; in particular if $\sh{F}$ is $f$-flat, $f'$ is flat, $f$ is flat.
\end{proposition}

\begin{proof}
It suffices to prove the first assertion; applying three times~\sref[I]{I.3.6.5}, as well as~\sref[I]{I.2.4.4}, we can reduce to the case where $Y=\Spec(\sh{O}_y)$, $X=\Spec(\sh{O}_x)$, $Y'=\Spec(\sh{O}_{y'})$, $\sh{F}=\widetilde{M}$, where $M$ is an $A$-module.
The hypothesis and~\sref{IV.2.1.2} imply that $\sh{F}$ is $f$-flat, and the conclusion then also follows from~\sref{IV.2.1.2} and from the definition of flatness.
\end{proof}

\begin{proposition}[2.1.5]
\label{IV.2.1.5}
Consider a commutative diagram of morphisms of preschemes
\[
\xymatrix{
  X \ar[d]_t & X' \ar[l]_{g'} \ar[d]^{f'} \\
  Y & Y' \ar[l]^g \\
  & Z \ar[u]^h
}
\]
where $X'=X\times_Y Y'$ and $f'=f_{(Y')}$.
Let $x'$ be a point of $X'$, and set $x=g'(x')$, $y'=f'(x')$, $y=f(x)=g(y')$, $z=h(y')$.
Let $\sh{F}$ be a quasi-coherent $\sh{O}_X$-Module $f$-flat at the point $x$ \textnormal{(}resp.\ $f$-flat\textnormal{)}, and $\sh{G}'$ a quasi-coherent $\sh{O}_{Y'}$-Module $h$-flat at the point $y'$ \textnormal{(}resp.\ $h$-flat\textnormal{)}; then $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ is a quasi-coherent $\sh{O}_{X'}$-Module $(h\circ f')$-flat at the point $x'$ \textnormal{(}resp.\ $(h\circ f')$-flat\textnormal{)}.
\end{proposition}

\begin{proof}
As in~\sref{IV.2.1.4}, we reduce to the case where $X=\Spec(\sh{O}_x)$, $Y=\Spec(\sh{O}_y)$,
\oldpage[IV-2]{7}
$Y'=\Spec(\sh{O}_{y'})$ and $Z=\Spec(\sh{O}_z)$, and it suffices then to prove that $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ is $(h\circ f')$-flat.
Taking into account~\sref{IV.2.1.2}, the proposition follows from Bourbaki, \emph{Alg.~comm.}, chap.~I\textsuperscript{er}, \textsection~2, n\textsuperscript{o}~7, prop.~8.
\end{proof}

\begin{corollary}[2.1.6]
\label{IV.2.1.6}
Let $f:X\to Y$, $g:Y\to Z$ be two morphisms of preschemes, $\sh{F}$ an $\sh{O}_X$-Module. If $\sh{F}$ is $f$-flat at the point $x\in X$ and if $g$ is flat at the point $f(x)$, $\sh{F}$ is $(g\circ f)$-flat at the point $x$.
In particular, if $f$ and $g$ are flat morphisms, so is $g\circ f$.
\end{corollary}

\begin{proof}
This is the special case of~\sref{IV.2.1.5} with $Y'=Y$, $\sh{G}'=\sh{O}_Y$.
\end{proof}

\begin{corollary}[2.1.7]
\label{IV.2.1.7}
If $f:X\to X'$, $g:Y\to Y'$ are two flat $S$-morphisms, $f\times_S g:X\times_S Y\to X'\times_S Y'$ is a flat morphism.
\end{corollary}

\begin{proof}
This follows from~\sref{IV.2.1.4} and~\sref{IV.2.1.6} (cf.~\sref[I]{I.3.5.1}).
\end{proof}

\begin{proposition}[2.1.8]
\label{IV.2.1.8}
Let $f:X\to Y$ be a morphism of preschemes,
\[
  0 \to \sh{F}' \to \sh{F} \to \sh{F}'' \to 0
\]
an exact sequence of quasi-coherent $\sh{O}_X$-Modules, such that $\sh{F}''$ is $Y$-flat.
\begin{enumerate}[label=(\roman*)]
  \item For every morphism $g:Y'\to Y$ and every quasi-coherent $\sh{O}_{Y'}$-Module $\sh{G}'$, the sequence
    $0\to\sh{F}'\otimes_{\sh{O}_Y}\sh{G}'\to\sh{F}\otimes_{\sh{O}_Y}\sh{G}'\to\sh{F}''\otimes_{\sh{O}_Y}\sh{G}'\to 0$ of $\sh{O}_{X'}$-Modules \textnormal{(}where $X'=X\times_Y Y'$\textnormal{)} is exact.
  \item For $\sh{F}$ to be $Y$-flat, it is necessary and sufficient that $\sh{F}'$ be so.
\end{enumerate}
\end{proposition}

\begin{proof}
We can evidently suppose $X$, $Y$, $Y'$ affine; the conclusion then follows from~\sref{IV.2.1.2} and from~\sref[0]{0.6.1.2}.
\end{proof}

\begin{corollary}[2.1.9]
\label{IV.2.1.9}
Let $\sh{L}^\bullet$ be a complex of quasi-coherent $\sh{O}_X$-Modules, $i$ an index such that if $d^i:\sh{L}^i\to\sh{L}^{i+1}$ is the derivation operator, $\sh{B}^{i+1}(\sh{L}^\bullet)=\operatorname{Im}(d^i)$ and $\sh{Z}^{i+1}(\sh{L}^\bullet)=\operatorname{Coker}(d^i)$ are $Y$-flat.
Then, with the notations of~\sref{IV.2.1.8}, the canonical homomorphism
\[
  \sh{H}^i(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}' \to \sh{H}^i(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')
\]
is bijective.
\end{corollary}

\begin{proof}
As the tensor product is right exact, we have
\[
  \sh{Z}^{i+1}(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}' = \operatorname{Coker}(d^i\otimes 1) = \sh{Z}^{i+1}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')
\]
and $\sh{Z}^{i}(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}' = \sh{Z}^{i}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')$.
Moreover, in the exact sequence
\[
  0\to\sh{B}^{i+1}(\sh{L}^\bullet)\to\sh{L}^{i+1}\to\sh{Z}^{i+1}(\sh{L}^\bullet)\to 0
\]
$\sh{Z}^{i+1}(\sh{L}^\bullet)$ is $Y$-flat, so it follows from~\sref{IV.2.1.8},~(i) that the exact sequence
\[
  0\to\sh{B}^{i+1}(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}'\to\sh{L}^{i+1}\otimes_{\sh{O}_Y}\sh{G}'\to\sh{Z}^{i+1}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')\to 0
\]
whence $\sh{B}^{i+1}(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}'=\operatorname{Im}(d^i\otimes 1)=\sh{B}^{i+1}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')$.
Then, as in the exact sequence
\[
  0\to\sh{H}^i(\sh{L}^\bullet)\to\sh{Z}^i(\sh{L}^\bullet)\to\sh{B}^{i+1}(\sh{L}^\bullet)\to 0
\]
$\sh{B}^{i+1}(\sh{L}^\bullet)$ is $Y$-flat, it follows from~\sref{IV.2.1.8},~(i) and from what precedes that the sequence
\[
  0\to\sh{H}^i(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}'\to\sh{Z}^i(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')\to\sh{B}^{i+1}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')\to 0
\]
is exact, which proves the corollary.
\end{proof}

\begin{corollary}[2.1.10]
\label{IV.2.1.10}
\oldpage[IV-2]{8}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module which is $Y$-flat, $\sh{L}_\bullet=(\sh{L}_i)$ a left resolution of $\sh{F}$ formed of quasi-coherent $\sh{O}_X$-Modules which are $Y$-flat.
Then, for every morphism $g:Y'\to Y$ and every quasi-coherent $\sh{O}_{Y'}$-Module $\sh{G}'$, the complex $\sh{L}_\bullet\otimes_{\sh{O}_Y}\sh{G}'=(\sh{L}_i\otimes_{\sh{O}_Y}\sh{G}')_{i\geq 0}$ is a left resolution of $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$.
Moreover, if $\sh{Z}_i(\sh{L}_\bullet)=\operatorname{Ker}(\sh{L}_i\to\sh{L}_{i-1})$, the $\sh{Z}_i(\sh{L}_\bullet)$ are $Y$-flat, and we have $\sh{Z}_i(\sh{L}_\bullet)\otimes_{\sh{O}_Y}\sh{G}'=\sh{Z}_i(\sh{L}_\bullet\otimes_{\sh{O}_Y}\sh{G}')=\operatorname{Ker}(\sh{L}_i\otimes_{\sh{O}_Y}\sh{G}'\to\sh{L}_{i-1}\otimes_{\sh{O}_Y}\sh{G}')$.
Setting $\sh{R}_i=\operatorname{Im}(\sh{L}_{i+1}\to\sh{L}_i)=\sh{Z}_i(\sh{L}_\bullet)$; we then have the exact sequences
\[
  0\leftarrow\sh{F}\leftarrow\sh{L}_0\leftarrow\sh{R}_0\leftarrow 0
\]
\[
  0\leftarrow\sh{R}_i\leftarrow\sh{L}_{i+1}\leftarrow\sh{R}_{i+1}\leftarrow 0 \qquad (i\geq 0)
\]
and since $\sh{F}$ and $\sh{L}_i$ are $Y$-flat, we deduce from~\sref{IV.2.1.8},~(ii) by induction that all the $\sh{R}_i$ are also $Y$-flat; using~\sref{IV.2.1.8},~(i), we thus have the exact sequences
\[
  0\leftarrow\sh{F}\otimes_{\sh{O}_Y}\sh{G}'\leftarrow\sh{L}_0\otimes_{\sh{O}_Y}\sh{G}'\leftarrow\sh{R}_0\otimes_{\sh{O}_Y}\sh{G}'\leftarrow 0
\]
\[
  0\leftarrow\sh{R}_i\otimes_{\sh{O}_Y}\sh{G}'\leftarrow\sh{L}_{i+1}\otimes_{\sh{O}_Y}\sh{G}'\leftarrow\sh{R}_{i+1}\otimes_{\sh{O}_Y}\sh{G}'\leftarrow 0 \qquad (i\geq 0)
\]
which prove the corollary.
\end{corollary}

\begin{proposition}[2.1.11]
\label{IV.2.1.11}
Let $f:X\to Y$ be a flat morphism, $\sh{F}$ a quasi-coherent $\sh{O}_Y$-Module of finite presentation.
If $\sh{J}$ is the Ideal of $\sh{O}_Y$ annihilator of $\sh{F}$, $f^*(\sh{J})$ is the Ideal of $\sh{O}_X$ annihilator of $f^*(\sh{F})$.
\end{proposition}

\begin{proof}
We have indeed by definition an exact sequence~\sref[0]{0.5.3.7}
\[
  0\to\sh{J}\to\sh{O}_Y\to\shHom_{\sh{O}_Y}(\sh{F},\,\sh{F})
\]
whence, since $f$ is flat, an exact sequence
\[
  0\to f^*(\sh{J})\to\sh{O}_X\to f^*(\shHom_{\sh{O}_Y}(\sh{F},\,\sh{F}))
\]
and since $\sh{F}$ is an $\sh{O}_Y$-Module of finite presentation, $f^*(\shHom_{\sh{O}_Y}(\sh{F},\,\sh{F}))$ canonically identifies with $\shHom_{\sh{O}_X}(f^*(\sh{F}),f^*(\sh{F}))$~\sref[0]{0.6.7.6}, whence the conclusion.
\end{proof}

\begin{proposition}[2.1.12]
\label{IV.2.1.12}
Let $X$ be a prescheme, $\sh{F}$ an $\sh{O}_X$-Module of finite presentation, $x$ a point of $X$.
The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item $\sh{F}_x$ is a flat $\sh{O}_x$-module.
  \item There exists an open neighbourhood $U$ of $x$ such that $\sh{F}|U$ is a locally free $(\sh{O}_X|U)$-Module.
\end{enumerate}
\end{proposition}

\begin{proof}
Indeed, $\sh{F}_x$ is a module of finite presentation over $\sh{O}_x$ a local ring; it amounts to the same thing to say that $\sh{F}_x$ is a flat $\sh{O}_x$-module or a free $\sh{O}_x$-module (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection~3, n\textsuperscript{o}~2, cor.~2 of prop.~5); whence the conclusion, taking into account~\sref[0]{0.5.2.7}.
We note that the proposition is valid for any \emph{ringed space with local rings}.
\end{proof}

\begin{proposition}[2.1.13]
\label{IV.2.1.13}
Let $f:X\to Y$ be a morphism of preschemes. If $f$ is flat at a point $x\in X$ and if the ring $\sh{O}_x$ is reduced \textnormal{(}resp.\ integral and integrally closed\textnormal{)}, then the ring $\sh{O}_{f(x)}$
\oldpage[IV-2]{9}
is reduced \textnormal{(}resp.\ integral and integrally closed\textnormal{)}.
If $f$ is faithfully flat and if $X$ is reduced \textnormal{(}resp.\ normal\textnormal{)}, then $Y$ is reduced \textnormal{(}resp.\ normal\textnormal{)}.
\end{proposition}

\begin{proof}
Set $\sh{O}_{f(x)}=A$, $\sh{O}_x=B$.
If $B$ is a flat $A$-module, it is also a faithfully flat $A$-module~\sref[0]{0.6.6.2}, so $A$ identifies with a subring of $B$; if $B$ is reduced, so is $A$.
Suppose now $B$ integral and integrally closed, then $A\subset B$ is integral; let $L$ be the field of fractions of $A$ and let $K\subset L$ be its field of fractions; denote by $K$ the field of fractions.
The hypothesis implies that $B\cap K=A$ (Bourbaki, \emph{Alg.~comm.}, chap.~I, \textsection~3, n\textsuperscript{o}~5, prop.~10).
If then $t\in K$ is integral over $A$, it is also integral over $B$, so belongs to $B$ by hypothesis, and thus $t\in A$, which proves that $A$ is integrally closed.
\end{proof}

\begin{proposition}[2.1.14]
\label{IV.2.1.14}
Let $f:X\to Y$ be a faithfully flat morphism. If $X$ is locally integral and if the underlying space is locally Noetherian, then $Y$ is locally integral.
\end{proposition}

\begin{proof}
Indeed, every $y\in Y$ is of the form $f(x)$ for some $x\in X$ and by hypothesis $\sh{O}_y$ identifies with a subring of $\sh{O}_x$~\sref[0]{0.6.6.1}; as $\sh{O}_x$ is integral, so is $\sh{O}_y$ and this proves the proposition~\sref[I]{I.5.1.4}.
\end{proof}

\subsection{Faithfully flat Modules on preschemes}
\label{subsection:IV.2.2}

\begin{proposition}[2.2.1]
\label{IV.2.2.1}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module. The following conditions are equivalent:
\begin{enumerate}[label=\emph{\alph*)}]
  \item For every base change $g:Y'\to Y$, if we set $X'=X\times_Y Y'$, the functor $\sh{G}'\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$, from the category of quasi-coherent $\sh{O}_{Y'}$-Modules to that of quasi-coherent $\sh{O}_{X'}$-Modules, is exact and faithful.
  \item[\emph{a$'$)}] Condition~\emph{a)} is satisfied for all canonical morphisms $g:\Spec(\sh{O}_y)\to Y$, where $y$ ranges over $Y$.
  \item[\emph{a$''$)}] Condition~\emph{a)} is satisfied for all canonical immersions $Y'\to Y$, where $Y'$ ranges over the set of affine open sub-preschemes of $Y$.
  \item $\sh{F}$ is $Y$-flat and, for every $y\in Y$, if we denote by $X_y$ the fibre $f^{-1}(y)$, the $\sh{O}_{X_y}$-Module $\sh{F}_y=\sh{F}\otimes_{\sh{O}_Y}k(y)$ is $\neq 0$.
\end{enumerate}
\end{proposition}

\begin{proof}
It is clear that \emph{a)} implies \emph{a$'$)} and \emph{a$''$)}; condition~\emph{a$'$)} implies first that $\sh{F}$ is $Y$-flat~\sref{IV.2.1.3}; on the other hand it implies that for every $y\in Y$, the functor $N\mapsto\sh{F}\otimes_{\sh{O}_y}\widetilde{N}$ is faithful in the category of $\sh{O}_y$-modules; taking in particular $N=k(y)$, we obtain the second assertion of~\emph{b)}.
To show that \emph{b)} implies \emph{a)}, we can restrict to the case where $Y$ is affine.
Likewise, to prove that \emph{a$''$)} implies \emph{a)}, the question being local on $Y$.
In other words, we are reduced to proving the following more precise proposition:
\end{proof}

\begin{proposition}[2.2.2]
\label{IV.2.2.2}
\oldpage[IV-2]{10}
Let $Y=\Spec(A)$ be an affine scheme, $f:X\to Y$ a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
Condition~\emph{a)} of~\sref{IV.2.2.1} is equivalent to each of the following:
\begin{enumerate}
  \item[\emph{b$'$)}] $\sh{F}$ is $Y$-flat and, for every closed point $y$ of $Y$, we have $\sh{F}_y\neq 0$.
  \item[\emph{c)}] The functor $\sh{G}\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ from the category of quasi-coherent $\sh{O}_Y$-Modules to that of quasi-coherent $\sh{O}_X$-Modules is exact and faithful.
\end{enumerate}
\end{proposition}

\begin{proof}
If \emph{b$'$)} is satisfied, there is at least one $x\in f^{-1}(y)$ such that $(\sh{F}_y)_x\neq 0$; let $U=\Spec(B)$, where $M$ is a $B$-module.
Then \emph{b$'$)} implies (since $M$ is an $A$-module flat~\sref{IV.2.1.2}) that $M\otimes_A A_y$ is an $A_y$-module faithfully flat~\sref[0]{0.6.4.5}.
The relation $(\sh{F}\otimes_{\sh{O}_Y}\sh{G})\otimes_{\sh{O}_y}\sh{G}_y=0$ for a closed point $y$ of $Y$ implies $(\sh{F}\otimes_{\sh{O}_Y}\sh{G})\otimes_{\sh{O}_y}\sh{G}_y=0$, so $\sh{G}_y=0$; but if we have $\sh{G}_y=0$ for every closed point $y$ of $Y$, we conclude $\sh{G}=0$, for if $\sh{G}=\widetilde{N}$, the annihilator of $N$ is not contained in any maximal ideal of $A$, so $A$ is all of it.
The relation $\sh{F}\otimes_{\sh{O}_Y}\sh{G}=0$ implies therefore $\sh{G}=0$, in other words the functor $\sh{G}\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ is faithful; we know moreover that this functor is exact~\sref{IV.2.1.3}, which shows that \emph{b$'$)} implies \emph{c)}.

Finally, to see that \emph{c)} implies \emph{a)}, we can restrict to the case where $Y'$ is also affine;
as $g:Y'\to X$ is a flat morphism, so is it the same, by the more~\sref[II]{II.1.5.5}; de plus, the functor $\sh{H}'\mapsto g'_*(\sh{H}')$ is then exact in the category of quasi-coherent $\sh{O}_{X'}$-Modules~\sref[I]{I.1.6.4}, and if $g'_*(\sh{H}')=\sh{H}$, on a $\sh{H}'=\widetilde{\sh{F}}$, so the preceding functor is also faithful; to see that $\sh{G}'\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ is exact and faithful, it suffices therefore to see that the functor $\sh{G}'\mapsto g'_*(\sh{F}\otimes_{\sh{O}_Y}\sh{G}')=\sh{F}\otimes_{\sh{O}_X}f^*(\sh{G}')$; the fact that $g$ is affine implies that we have a canonical isomorphism

\begin{equation}
\tag{2.2.2.1}
\label{IV.2.2.2.1}
  \sh{F}\otimes_{\sh{O}_X}f^*(g_*(\sh{G}'))\xrightarrow{\sim} g'_*(g^*(\sh{F})\otimes_{\sh{O}_{X'}}f'^*(\sh{G}')).
\end{equation}

Indeed, we know~\sref[II]{II.1.5.2} that we have a canonical isomorphism
\[
  f^*(g_*(\sh{G}'))\xrightarrow{\sim} g'_*(f'^*(\sh{G}')),
\]
and on the other hand a canonical homomorphism $\sh{F}\to g'_*(g'^*(\sh{F}))$~\sref[0]{0.4.4.3.2}; composing the homomorphism $\sh{F}\otimes_{\sh{O}_X}f^*(g_*(\sh{G}'))\to g'_*(g'^*(\sh{F}))\otimes_{\sh{O}_X}g'_*(f'^*(\sh{G}'))$ with the canonical homomorphism~\sref[0]{0.4.2.2.1}, we deduce the homomorphism~\sref{IV.2.2.2.1}; the verification that it is an isomorphism is immediate by reducing to the case where $X$ is affine.
This being so, the functor $\sh{G}'\mapsto g_*(\sh{G}')$ is exact and faithful and by hypothesis it is the same of $\sh{M}\mapsto\sh{M}'\otimes_{\sh{O}_Y}\sh{F}$ (from the category of $\sh{O}_Y$-Modules to that of quasi-coherent $\sh{O}_X$-Modules); the composition is exact and faithful, which completes the proof of~\sref{IV.2.2.1} and~\sref{IV.2.2.2}.
\end{proof}

\begin{corollary}[2.2.3]
\label{IV.2.2.3}
Let $X=\Spec(B)$, $Y=\Spec(A)$ be two affine schemes, $f:X\to Y$ a morphism, $\sh{F}=\widetilde{M}$ a quasi-coherent $\sh{O}_X$-Module.
For $\sh{F}$ to satisfy the equivalent conditions of~\sref{IV.2.2.1} \textnormal{(}or~\sref{IV.2.2.2}\textnormal{)}, it is necessary and sufficient that the $A$-module $M$ be faithfully flat.
\end{corollary}

\begin{proof}
Indeed, condition~\emph{c)} of~\sref{IV.2.2.2} means that the functor $N\mapsto M\otimes_A N$ from the category of $A$-modules to that of $B$-modules is exact and faithful, and the conclusion follows from~\sref[0]{0.6.4.1}.
\end{proof}

\begin{definition}[2.2.4]
\label{IV.2.2.4}
When the equivalent conditions of~\sref{IV.2.2.1} are satisfied, we say that the quasi-coherent $\sh{O}_X$-Module $\sh{F}$ is \emph{faithfully flat relative to} $f$ (or to~$Y$).
\end{definition}

\oldpage[IV-2]{11}
We note that this notion is local over $Y$, but \emph{not over} $X$; in particular we can have $\sh{F}_x=0$ for certain $x\in X$, in other words $\operatorname{Supp}(\sh{F})$ is not necessarily equal to~$X$.
However, it follows immediately from the criterion~\sref{IV.2.2.1},~\emph{b)} that for every $y\in Y$, there exists at least one $x\in f^{-1}(y)$ such that $(\sh{F}_y)_x=\sh{F}_x\otimes_{\sh{O}_y}k(y)\neq 0$, and \emph{a fortiori} $\sh{F}_x\neq 0$; in other terms:

\begin{corollary}[2.2.5]
\label{IV.2.2.5}
If $\sh{F}$ is a quasi-coherent $\sh{O}_X$-Module, faithfully flat relative to $f$, we have $f(\operatorname{Supp}(\sh{F}))=Y$, and a fortiori $f$ is a surjective morphism.
\end{corollary}

This result admits a partial converse:

\begin{corollary}[2.2.6]
\label{IV.2.2.6}
Let $\sh{F}$ be a quasi-coherent $\sh{O}_X$-Module of finite type. For $\sh{F}$ to be faithfully flat relative to $f$, it is necessary and sufficient that $\sh{F}$ be $f$-flat and that $f(\operatorname{Supp}(\sh{F}))=Y$.
\end{corollary}

\begin{proof}
Indeed~\sref[I]{I.9.1.13} and \sref{IV.2.3.6.1} we have $\operatorname{Supp}(\sh{F}_y)=f^{-1}(y)\cap\operatorname{Supp}(\sh{F})$, so in this case the criterion~\sref{IV.2.2.1},~\emph{b)} is nothing else than the condition of the corollary.
In particular, the $\sh{O}_X$-Module $\sh{O}_X$ is faithfully flat relative to $f$ if and only if $f$ is flat and $f$ is surjective, that is to say if and only if the \emph{morphism $f$ is faithfully flat}~\sref[0]{0.6.7.8}.
\end{proof}

Let us spell out the properties intervening in definition~\sref{IV.2.2.4}:

\begin{proposition}[2.2.7]
\label{IV.2.2.7}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module faithfully flat relative to $f$.
Then, for a sequence $\sh{G}'\to\sh{G}\to\sh{G}''$ of quasi-coherent $\sh{O}_Y$-Modules to be exact, it is necessary and sufficient that the corresponding sequence $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'\to\sh{F}\otimes_{\sh{O}_Y}\sh{G}\to\sh{F}\otimes_{\sh{O}_Y}\sh{G}''$ be so.
In particular, for a homomorphism $u:\sh{G}\to\sh{G}'$ of quasi-coherent $\sh{O}_Y$-Modules to be injective \textnormal{(}resp.\ surjective, bijective, zero\textnormal{)}, it is necessary and sufficient that $1_{\sh{F}}\otimes f^*(u):\sh{F}\otimes_{\sh{O}_Y}\sh{G}\to\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ be so.
For a quasi-coherent $\sh{O}_Y$-Module $\sh{G}$ to be zero, it is necessary and sufficient that $\sh{F}\otimes_{\sh{O}_Y}\sh{G}=0$.
For an $\sh{O}_Y$-Module quasi-coherent $\sh{G}$, the map $\sh{G}'\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ from the set of quasi-coherent sub-$\sh{O}_Y$-Modules of $\sh{G}$ to the set of sub-$\sh{O}_X$-Modules quasi-coherent of $\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ is injective.
\end{proposition}

\begin{proof}
To prove the last assertion, that is to say that for two quasi-coherent sub-$\sh{O}_Y$-Modules $\sh{G}'$, $\sh{G}''$ of $\sh{G}$, the relation $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'=\sh{F}\otimes_{\sh{O}_Y}\sh{G}''$ implies $\sh{G}'=\sh{G}''$, we can reduce to the case where $\sh{G}'\subset\sh{G}''$ (by replacing $\sh{G}''$ by $\sh{G}'+\sh{G}''$), and it then suffices to apply the second assertion of the statement to the injection $u:\sh{G}'\to\sh{G}''$.
\end{proof}

\begin{corollary}[2.2.8]
\label{IV.2.2.8}
Let $f:X\to Y$ be a faithfully flat morphism, $\sh{G}$ a quasi-coherent $\sh{O}_Y$-Module.
The canonical map
\begin{equation}
\tag{2.2.8.1}
\label{IV.2.2.8.1}
  \Gamma(Y,\,\sh{G}) \to \Gamma(X,\,f^*(\sh{G}))
\end{equation}
is injective.
\end{corollary}

\begin{proof}
Indeed, $\Gamma(Y,\,\sh{G})$ canonically identifies with $\operatorname{Hom}_{\sh{O}_Y}(\sh{O}_Y,\,\sh{G})$~\sref[0]{0.5.1.1} and similarly $\Gamma(X,\,f^*(\sh{G}))$ with $\operatorname{Hom}_{\sh{O}_X}(\sh{O}_X,\,f^*(\sh{G}))$.
By virtue of~\sref{IV.2.2.1} and~\sref{IV.2.2.4}, the hypothesis implies that the functor $\sh{G}\mapsto f^*(\sh{G})$ is exact and faithful in the category of $\sh{O}_Y$-Modules quasi-coherent, and thus a homomorphism $u:\sh{O}_Y\to\sh{G}$ is zero if and only if $f^*(u):f^*(\sh{O}_Y)=\sh{O}_X\to f^*(\sh{G})$ is zero.
\end{proof}

\begin{remark}[2.2.9]
\label{IV.2.2.9}
The assertions of~\sref{IV.2.2.7} and~\sref{IV.2.2.8} are still true when the $\sh{O}_Y$-Modules $\sh{G}'$, $\sh{G}$, $\sh{G}''$ appearing there are arbitrary (not necessarily quasi-coherent).
Indeed, for every $y\in Y$, there exists $x\in f^{-1}(y)$ such that $\sh{F}_x$ is a flat $\sh{O}_y$-module
\oldpage[IV-2]{12}
faithfully flat, and thus the functor $\sh{G}_y\mapsto\sh{G}_y\otimes_{\sh{O}_y}\sh{F}_x$ is faithful; since moreover for \emph{every} $x\in f^{-1}(y)$ the functor $\sh{G}_y\mapsto\sh{G}_y\otimes_{\sh{O}_y}\sh{F}_x$ is exact, we immediately deduce the conclusion.
\end{remark}

\begin{proposition}[2.2.10]
\label{IV.2.2.10}
Let $f:X\to Y$, $g:Y'\to Y$, $h:Y'\to Z$ be three morphisms of preschemes, $X'=X\times_Y Y'$, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module faithfully flat relative to $Y$, $\sh{G}'$ a quasi-coherent $\sh{O}_{Y'}$-Module.
For $\sh{G}'$ to be flat \textnormal{(}resp.\ faithfully flat\textnormal{)} relative to $Z$, it is necessary and sufficient that $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ be a flat \textnormal{(}resp.\ faithfully flat\textnormal{)} $\sh{O}_{X'}$-Module relative to $Z$.
\end{proposition}

\begin{proof}
We know already that if $\sh{G}'$ is $Z$-flat, so is $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$~\sref{IV.2.1.5}.
Consider a base change $Z''\to Z$ and set $X''=X'\times_Z Z''$; if $\sh{G}'$ is faithfully flat relative to $Z$, the functor
\begin{equation}
\tag{2.2.10.1}
\label{IV.2.2.10.1}
\sh{H}''\mapsto\sh{H}''\otimes_Z\sh{G}'\to(\sh{H}''\otimes_Z\sh{G}')\otimes_{\sh{O}_Y}\sh{F}=\sh{H}''\otimes_Z(\sh{G}'\otimes_{\sh{O}_Y}\sh{F})
\end{equation}
from the category of $\sh{O}_{Z''}$-Modules to that of $\sh{O}_{X''}$-Modules is composed of two exact and faithful functors, so this functor is also exact and faithful.
Conversely, if ce foncteur $\sh{H}''\mapsto\sh{H}''\otimes_Z\sh{G}'$ is exact (resp.\ exact and faithful), since the functor $\sh{M}\mapsto\sh{M}'\otimes_{\sh{O}_Y}\sh{F}$ (from the category of $\sh{O}_{Y'}$-Modules to that of quasi-coherent $\sh{O}_{X'}$-Modules) is exact and faithful by hypothesis.
\end{proof}

\begin{corollary}[2.2.11]
\label{IV.2.2.11}
\medskip\noindent
\begin{enumerate}[label=(\roman*)]
  \item Let $f:X\to Y$, $g:Y'\to Y$ be two morphisms, $X'=X\times_Y Y'$, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module. If $\sh{F}$ is faithfully flat relative to $Y$, $\sh{F}'=\sh{F}\otimes_{\sh{O}_Y}\sh{O}_{Y'}$ is faithfully flat relative to $Y'$.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module faithfully flat relative to $Y$.
    For $g$ to be a faithfully flat morphism, it is necessary and sufficient that $\sh{F}$ be faithfully flat relative to $Z$.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms, $\sh{G}$ a quasi-coherent $\sh{O}_Y$-Module.
    Suppose the morphism $f$ is faithfully flat.
    For $\sh{G}$ to be flat \textnormal{(}resp.\ faithfully flat\textnormal{)} relative to $Z$, it is necessary and sufficient that $f^*(\sh{G})$ be flat \textnormal{(}resp.\ faithfully flat\textnormal{)} relative to $Z$.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms, $x$ a point of $X$.
    Suppose $f$ is flat at the point $x$. For $g\circ f$ to be flat at the point $x$, it is necessary and sufficient that $g$ be flat at the point $f(x)$.
\end{enumerate}
\end{corollary}

\begin{proof}
To prove~(i), we apply~\sref{IV.2.2.10} replacing $Z$ par $Y'$ and $\sh{G}'$ by $\sh{O}_{Y'}$.
To prove~(ii), we apply~\sref{IV.2.2.10} taking $Y'\to Y$ the identity and replacing $\sh{G}'$ by $\sh{O}_Y$.
To prove~(iii), we apply~\sref{IV.2.2.10} taking $Y'\to Y$ the identity, and replacing $\sh{F}$ par $\sh{O}_X$ and $\sh{G}'$ by $\sh{G}$.
Finally~(iv) is deduced from~(ii) by replacing $X$ by $\Spec(\sh{O}_x)$, $\sh{F}$ by $\sh{O}_X$, $Y$ by $\Spec(\sh{O}_{f(x)})$ and $Z$ by $\Spec(\sh{O}_{g(f(x))})$, and taking into account~\sref[0]{0.6.6.2}.
\end{proof}

\begin{corollary}[2.2.12]
\label{IV.2.2.12}
Let $Y$ be an affine scheme, $f:X\to Y$ a quasi-compact morphism,
\oldpage[IV-2]{13}
$\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
If $\sh{F}$ is faithfully flat relative to $f$, there exists an affine scheme $X'$ and a faithfully flat morphism $g:X'\to X$ such that $g^*(\sh{F})$ is faithfully flat relative to $f\circ g$.
\end{corollary}

\begin{proof}
Indeed, $X$ is quasi-compact, so a finite union of open affine sets $X_i$; it suffices to take for $X'$ the affine scheme \emph{direct sum} of the preschemes induced on the open sets $X_i$ and for $g:X'\to X$ the canonical morphism; it is clear that $g$ is faithfully flat, and the hypothesis implies that $g^*(\sh{F})$ is faithfully flat relative to $f\circ g$, by virtue of~\sref{IV.2.2.11},~(iii).
\end{proof}

\begin{corollary}[2.2.13]
\label{IV.2.2.13}
\medskip\noindent
\begin{enumerate}[label=(\roman*)]
  \item Let $f:X\to Y$, $g:Y'\to Y$ be two morphisms, $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$.
    If $f$ is a faithfully flat morphism, so is $f'$.
  \item If $f:X\to X'$, $g:Y\to Y'$ are two flat $S$-morphisms,
    \[
      f\times_S g: X\times_S Y \to X'\times_S Y'
    \]
    is faithfully flat.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms such that $f$ is faithfully flat.
    For $g$ to be a flat \textnormal{(}resp.\ faithfully flat\textnormal{)} morphism, it is necessary and sufficient that $g\circ f$ be a flat \textnormal{(}resp.\ faithfully flat\textnormal{)} morphism.
\end{enumerate}
\end{corollary}

\begin{proposition}[2.2.14]
\label{IV.2.2.14}
Let $f:X\to Y$ be a faithfully flat morphism and quasi-compact.
If $X$ is locally Noetherian, so is $Y$.
\end{proposition}

\begin{proof}
The question being local on $Y$, we can suppose $Y=\Spec(A)$ affine; as $f$ is quasi-compact, it follows from~\sref{IV.2.2.12} that we can also restrict to the case where $X=\Spec(B)$ is affine.
Then $B$ is a Noetherian ring by hypothesis, and a faithfully flat $A$-module~\sref{IV.2.2.3}; so $A$ is Noetherian~\sref[0]{0.6.5.2}.
\end{proof}

\begin{proposition}[2.2.15]
\label{IV.2.2.15}
Let $f:X\to Y$ be a faithfully flat morphism.
If $\mathfrak{S}(X)$ and $\mathfrak{S}(Y)$ denote respectively the sets of sub-preschemes of $X$ and of $Y$, the map $Z\mapsto f^{-1}(Z)$ of $\mathfrak{S}(Y)$ to $\mathfrak{S}(X)$ is injective.
\end{proposition}

\begin{proof}
Since $f$ is surjective, for a sub-prescheme $Z$ of $Y$, $f(f^{-1}(Z))=Z$.
On the other hand, if $U$ is an open subset of $Y$ containing $Z$ and in which $Z$ is closed, $f^{-1}(Z)$ is closed in $f^{-1}(U)$, and the restriction $f^{-1}(U)\to U$ of $f$ is a faithfully flat morphism.
We can therefore restrict to considering only the \emph{closed} sub-preschemes of $Y$.
Now, if $Z$ is a closed sub-prescheme of $Y$ corresponding to a quasi-coherent Ideal $\sh{J}$ of $\sh{O}_Y$, we know that $f^{-1}(Z)$ corresponds to the quasi-coherent Ideal $f^*(\sh{J})\sh{O}_X$ of $\sh{O}_X$~\sref[I]{I.4.4.5}, and since $f$ is flat, $f^*(\sh{J})\sh{O}_X$ identifies with $f^*(\sh{J})$.
But the map $\sh{J}\mapsto f^*(\sh{J})$ from the set of quasi-coherent Ideals of $\sh{O}_X$ to the set of quasi-coherent Ideals of $\sh{O}_X$ is injective~\sref{IV.2.2.7}, whence the conclusion.
\end{proof}

\begin{corollary}[2.2.16]
\label{IV.2.2.16}
Let $X$, $Y$ be two $S$-preschemes; if $S'\to S$ is a faithfully flat morphism, the map $f\mapsto f_{(S')}$ from $\operatorname{Hom}_S(X,Y)$ to $\operatorname{Hom}_{S'}(X_{(S')},Y_{(S')})$ is injective.
\end{corollary}

\begin{proof}
We have $X_{(S')}\times_{S'}Y_{(S')}=(X\times_S Y)_{(S')}$~\sref[I]{I.3.3.10}, so the morphism projection $X_{(S')}\times_{S'}Y_{(S')}\to X\times_S Y$ is faithfully flat~\sref{IV.2.2.13}.
The elements of $\operatorname{Hom}_S(X,Y)$ correspond bijectively to sub-preschemes of $X\times_S Y$ that are \emph{graphs} of $f\in\operatorname{Hom}_S(X,Y)$, and if $Z_f$ is the graph of $f$, we have $Z_{f_{(S')}}=g^{-1}(Z_f)$~\sref[I]{I.5.3.12}.
It suffices therefore to apply~\sref{IV.2.2.15}.
\end{proof}

\begin{proposition}[2.2.17]
\label{IV.2.2.17}
Let $A$ be a ring, $B$ an $A$-algebra such that $B$ is a faithfully flat $A$-module and of finite presentation.
Then the structural homomorphism $\varphi:A\to B$ is an isomorphism of the $A$-module $A$ onto a direct summand of the $A$-module $B$.
If $A$ is a local ring, $B$ is a free $A$-module and there exists a basis of this module containing the identity element of $B$.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{14}
By virtue of Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection~3, n\textsuperscript{o}~3, prop.~12, it suffices to prove the proposition when $A$ is a local ring; we know then (\emph{loc.~cit.}, n\textsuperscript{o}~2, cor.~2 of prop.~5) that $B$ is an $A$-module free of finite type, and the conclusion follows from \emph{loc.~cit.}, prop.~5.
\end{proof}

\subsection{Topological properties of flat morphisms}
\label{subsection:IV.2.3}

\begin{lemma}[2.3.1]
\label{IV.2.3.1}
Let $f:X\to Y$ be a quasi-compact and quasi-separated morphism, $g:Y'\to Y$ a flat morphism; set $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$.
Then, for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, the canonical homomorphism
\begin{equation}
\tag{2.3.1.1}
\label{IV.2.3.1.1}
  g^*(f_*(\sh{F})) \to f'_*(\sh{F}\otimes_{\sh{O}_Y}\sh{O}_{Y'})
\end{equation}
is bijective.
\end{lemma}

\begin{proof}
This is a special case of~\sref[III]{III.1.4.15} (improved in~(1.7.21)).
\end{proof}

\begin{proposition}[2.3.2]
\label{IV.2.3.2}
Let $S$ be a prescheme, $f:X\to Y$ an $S$-morphism quasi-compact and quasi-separated, $Z$ the sub-prescheme which is the closed image of $X$ by $f$~\textnormal{(}\sref[I]{I.9.5.3} and~(1.7.8)\textnormal{)}, and let $j:Z\to Y$ be the injection canonical, so that $f=j\circ g$, where $g:X\to Z$ is a morphism \textnormal{(}\emph{loc.~cit.}\textnormal{)}.
Let $h:S'\to S$ be a morphism, and set $f'=f_{(S')}:X_{(S')}\to Y_{(S')}$; then $j'=j_{(S')}:Z_{(S')}\to Y_{(S')}$ identifies with the canonical injection of the sub-prescheme of $Y_{(S')}$, closed image of $X_{(S')}$ by $f'$.
\end{proposition}

\begin{proof}
Since the morphism $Y_{(S')}\to Y$ is flat~\sref{IV.2.1.4}, we can restrict to the case where $S=Y$~\sref[I]{I.3.3.11}; if $f=(\psi,\theta)$, the canonical injection of $Z$ is the sub-prescheme of $Y$ defined by the Ideal $\sh{J}$ of $\sh{O}_S$ kernel of $\theta:{\sh{O}_S}\to f_*(\sh{O}_X)$.
As $h$ is a flat morphism, the Ideal $\sh{J}\sh{O}_{S'}$ identifies with the kernel of $h^*(\theta):h^*(\sh{O}_S)\to h^*(f_*(\sh{O}_X))$~\sref[0]{0.6.3.1}); as $f$ is quasi-compact and quasi-separated, the canonical homomorphism~\sref{IV.2.3.1.1} $h^*(f_*(\sh{O}_X))\to f'_*(\sh{O}_{X'})$ is an isomorphism, the conclusion follows from~\sref{IV.2.3.1} and from~\sref[I]{I.9.5.2}, since $f'$ is quasi-compact~(1.1.2 and 1.2.2).
\end{proof}

\begin{env}[2.3.3]
\label{IV.2.3.3}
We will say that a morphism $f:X\to Y$ is \emph{quasi-flat} if there exists a quasi-coherent $\sh{O}_X$-Module $\sh{F}$ \emph{of finite type} which is $f$-flat and whose support is equal to $X$.
We will say that $f$ is \emph{quasi-faithfully flat} if it is quasi-flat and surjective.
Every morphism flat (resp.\ faithfully flat) is quasi-flat (resp.\ quasi-faithfully flat), for then $\sh{F}=\sh{O}_X$ satisfies the preceding conditions.
\end{env}

It follows immediately from~\sref{IV.2.1.4} and from~\sref[I]{I.9.1.13}, that if $f$ is quasi-flat, then, for every morphism $Y'\to Y$, the morphism $f_{(Y')}:X\times_Y Y'\to Y'$ is quasi-flat.
Similarly (taking into account~\sref[I]{I.3.5.2}), if $f$ is quasi-faithfully flat, so is $f_{(Y')}$.

\begin{proposition}[2.3.4]
\label{IV.2.3.4}
Let $f:X\to Y$ be a quasi-flat morphism~\sref{IV.2.3.3}. Then $f$ possesses the following properties \textnormal{(}which are equivalent by virtue of~(1.10.4)\textnormal{)}:
\begin{enumerate}[label=(\roman*)]
  \item For every $x\in X$ and every generization $y'$ of $y=f(x)$, there exists a generization $x'$ of $x$ such that $f(x')=y'$.
  \item For every $x\in X$, the image by $f$ of $\Spec(\sh{O}_{X,x})$ is $\Spec(\sh{O}_{Y,y})$.
\oldpage[IV-2]{15}
  \item[\emph{(iii)}] For every closed irreducible part $Y'$ of $Y$, every irreducible component of $f^{-1}(Y')$ dominates $Y'$.
\end{enumerate}
\end{proposition}

\begin{proof}
It suffices, for example, to prove~(ii).
By hypothesis, there is an $\sh{O}_X$-Module quasi-coherent of finite type $\sh{F}$, which is $f$-flat and such that $\operatorname{Supp}(\sh{F})=X$.
For every $x\in X$, $\sh{F}_x$ is then an $\sh{O}_x$-module of finite type, nonzero, and moreover $\sh{F}_x$ is an $\sh{O}_y$-module \emph{flat}, for the homomorphism $\rho:\sh{O}_y\to\sh{O}_x$.
As the latter is local and $\sh{F}_x\neq 0$, the Nakayama lemma shows that $\sh{F}_x\otimes_{\sh{O}_y}k(y)\neq 0$, so $\sh{F}_x$ is a faithfully flat $\sh{O}_y$-module~\sref[0]{0.6.4.1}.
It follows that for every prime ideal $\mathfrak{q}$ of $\sh{O}_y$, there exists a prime ideal $\mathfrak{p}$ of $\sh{O}_x$ such that $\mathfrak{q}=\rho^{-1}(\mathfrak{p})$~\sref[0]{0.6.5.1}, which proves~(ii).
\end{proof}

\begin{corollary}[2.3.5]
\label{IV.2.3.5}
Let $f:X\to Y$ be a morphism satisfying the equivalent conditions~(i), (ii), (iii) of~\sref{IV.2.3.4} \textnormal{(}in particular a quasi-flat or open morphism~(1.10.4)\textnormal{)}.
\begin{enumerate}[label=(\roman*)]
  \item Let $Z$, $Z'$ be two closed irreducible subsets of $Y$ such that $Z\subset Z'$, and let $\mathrm{T}$ be an irreducible component of $f^{-1}(Z)$; then there exists an irreducible component $\mathrm{T}'$ of $f^{-1}(Z')$ containing $\mathrm{T}$ \textnormal{(}and dominating $Z'$\textnormal{)}.
  \item For every irreducible component $\mathrm{T}$ of $X$, $\overline{f(\mathrm{T})}$ is an irreducible component of $Y$.
  \item Suppose $Y$ irreducible, and, if $y$ is its generic point, suppose that $f^{-1}(y)$ is irreducible. Then $X$ is irreducible.
\end{enumerate}
\end{corollary}

\begin{proof}
(i) It suffices to apply~\sref{IV.2.3.4},~(i) by taking for $x$ the generic point of $\mathrm{T}$ ($y=f(x)$ is then the generic point of $Z$) and for $y'$ the generic point of $Z'$.

(ii) It is clear that $\overline{f(\mathrm{T})}=Z$ is irreducible, and in virtue of~(i), $Z$ cannot be strictly contained in a closed irreducible part of $Y$.

(iii) By virtue of~(ii), every irreducible component of $X$ dominates $Y$, and if $y$ is the generic point of $Y$, its inverse image $f^{-1}(y)$ meets each of these sets $f^{-1}(y_i)$~\sref{IV.2.3.4}; the conclusion then follows from~\sref[0]{0.2.1.8}.
\end{proof}

\begin{proposition}[2.3.6]
\label{IV.2.3.6}
Let $Y$ be a prescheme whose set of irreducible components is locally finite \textnormal{(}which is the case if the underlying space is locally Noetherian~(cf.~\sref[I]{I.6.1.9})\textnormal{)}.
\begin{enumerate}[label=(\roman*)]
  \item Every closed subset $W$ of $Y$ stable by generization~\sref[0]{0.2.1.2} is open. In particular, every connected component of $Y$ is open.
  \item Let $f:X\to Y$ be a closed morphism satisfying in addition the equivalent conditions~(i), (ii), (iii) of~\sref{IV.2.3.4} \textnormal{(}which will be the case if $f$ is quasi-flat or open\textnormal{)}. Then the image by $f$ of every connected component $C$ of $X$ is a connected component of $Y$.
  \item Let $f:X\to Y$ be a morphism satisfying the equivalent conditions~(i), (ii), (iii) of~\sref{IV.2.3.4}, and such that moreover for every $y\in Y$, the set $f^{-1}(y)$ is finite \textnormal{(}which will be the case if $f$ is quasi-finite or radicial\textnormal{)}. Then the set of irreducible components of $X$ is locally finite.
\end{enumerate}
\end{proposition}

\begin{proof}
\oldpage[IV-2]{16}
(i) If $y\in W$, the generic points $\eta_i$ ($1\leq i\leq m$) of the irreducible components $Y_i$ of $Y$ containing $y$ belong to $W$ by hypothesis, so these components themselves are contained in $W$; since there is by hypothesis a neighbourhood $U$ of $y$ such that $U$ is a union of the $U\cap Y_i$~\sref[0]{0.2.1.6}, we have $U\subset W$, so $W$ is open.
Since for every $y\in Y$, $\overline{\{y\}}$ is connected, a connected component of $Y$ is stable by generization, so the second assertion follows immediately from the first.

(ii) Since $C$ is closed in $X$, $f(C)$ is closed in $Y$ by hypothesis.
Moreover, since $C$ is stable by generization, the hypothesis on $f$ implies that $f(C)$ is stable by generization; so, by virtue of~(i), $f(C)$ is open and closed, and since it is connected, it is a connected component of $Y$.

(iii) Let $x\in X$; by hypothesis, there is an open neighbourhood $U$ such that $y=f(x)$ meets only a finite number of irreducible components $Y_i$ of $Y$ ($1\leq i\leq n$); let $y_i$ be the generic point of $Y_i$.
For every irreducible component $Z$ of $X$ meeting $f^{-1}(U)$, the generic point $z$ of $Z$ is necessarily in one of the sets $f^{-1}(y_i)$~\sref{IV.2.3.4}.
Since each of these sets is finite by hypothesis, this proves our assertion.
\end{proof}

\begin{proposition}[2.3.7]
\label{IV.2.3.7}
Let $f:X\to Y$ be a morphism, $g:Y'\to Y$ a quasi-flat morphism, $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$.
\begin{enumerate}[label=(\roman*)]
  \item If $f$ is quasi-compact and dominant, so is $f'$.
  \item If every irreducible component of $X$ dominates an irreducible component of $Y$, then every irreducible component of $X'$ dominates an irreducible component of $Y'$.
\end{enumerate}
\end{proposition}

\begin{proof}
Denote by $g':X'\to X$ the canonical projection, which is a quasi-flat morphism~\sref{IV.2.3.3}.

(i) We know already~\sref{IV.1.1.2} that $f'$ is quasi-compact; moreover, if $y'$ is a maximal point of $Y'$, $y=g(y')$ is a maximal point of $Y$, as it follows from~\sref{IV.2.3.4},~(ii).
By hypothesis~\sref{IV.1.1.5}, there exists $x\in X$ such that $f(x)=y$; so~\sref[I]{I.3.4.7}, there exists $x'\in X'$ such that $f'(x')=y'$.

(ii) Let $x'$ be a maximal point of $X'$, and let $x=g'(x')$; it follows from~\sref{IV.2.3.4},~(ii) that $x$ is a maximal point of $X$, and by hypothesis $y=f(x)$ is a maximal point of $Y$.
Set $y'=f'(x')$, so that $g(y')=y$; it remains to show that $y'$ is a maximal point of $Y'$.
By virtue of~\sref[I]{I.5.1.7} and~\sref{IV.2.3.3}, we can restrict to the case where $\sh{O}_x$ and $\sh{O}_y$ are \emph{fields}, so $\sh{O}_x$ and $\sh{O}_y$ are reduced, and by virtue of~\sref[I]{I.3.6.5} applied twice and of~\sref[I]{I.2.4.4}, we can restrict to the case where $Y=\Spec(\sh{O}_y)$ and $Y'=\Spec(\sh{O}_{y'})$.
Then $f$ is a morphism \emph{flat} since $\sh{O}_y$ is a field~\sref{IV.2.1.2}, and it is of the same of $f'$~\sref{IV.2.1.4}; so it follows from~\sref{IV.2.3.4},~(ii) that $y'=f'(x')$ is a maximal point of $Y'$.
\end{proof}

\begin{corollary}[2.3.8]
\label{IV.2.3.8}
\emph{(Zariski).}
Let $A$, $B$ be two Noetherian local rings, $\mathfrak{m}$, $\mathfrak{n}$ their maximal ideals respectively, $\varphi:A\to B$ a local homomorphism. Suppose the following hypotheses satisfied:
\begin{enumerate}
  \item[$1^\circ$] $B$ is an $A$-algebra essentially of finite type~(1.3.6).
  \item[$2^\circ$] The $\mathfrak{m}$-adic completion $\widehat{A}$ of $A$ for the $\mathfrak{m}$-adic topology is integral.
  \item[$3^\circ$] $\varphi$ is injective.
\end{enumerate}
Then the $\mathfrak{m}$-adic topology of $A$ is induced by the $\mathfrak{n}$-adic topology of $B$.
\end{corollary}

\begin{proof}
\oldpage[IV-2]{17}
Set $B'=B\otimes_A\widehat{A}$; by hypothesis $1^\circ$, $B'$ is of the form $S^{-1}(C\otimes_A\widehat{A})$, where $C$ is an $A$-algebra of finite type and $S$ is a multiplicative subset of $C$.
As $A$ identifies with a subring of $\widehat{A}$~\sref[0]{0.7.3.5}, $A$ is integral by hypothesis $2^\circ$.
Hypothesis $3^\circ$ implies that there exists a prime ideal $\mathfrak{q}$ of $B$ inducing the prime ideal $\mathfrak{q}$ of $A$ of the homomorphism local $A\to B/\mathfrak{q}$ injective.
We can therefore restrict to proving the conclusion of~\sref{IV.2.3.8} under the added hypothesis that $B$ is a local ring which is \emph{integral}.
We apply~\sref{IV.2.3.7},~(ii) to $Y=\Spec(A)$, $X=\Spec(B)$, $Y'=\Spec(\widehat{A})$ and $X'=\Spec(B')$; since the morphism $Y'\to Y$ is flat and $X$ dominates the integral scheme $Y$ (integral by hypothesis), every irreducible component of $X'$ dominates $Y'$.
If $y$, $x$, $y'$ are the closed points of $Y$, $X$ and $Y'$ respectively, and designating by $\mathfrak{r}$ the ideal maximal of $\sh{O}_{x'}$, the intersection of ideals $\mathfrak{r}^k\cap\widehat{A}$ is null~\sref[0]{0.7.3.5}; since $\widehat{A}$ is \emph{complete}, this implies (Bourbaki, \emph{Alg.~comm.}, chap.~III, \textsection~2, n\textsuperscript{o}~7, prop.~8) that the topology of $\widehat{A}$ is \emph{induced} by the $\mathfrak{r}$-preadic topology of $\sh{O}_{x'}$; \emph{a fortiori} it is the same for the topology of $A$~\sref[0]{0.7.3.5}.
Moreover, on a $\mathfrak{m}^k\cap A\subset\mathfrak{r}^k\cap A$, so $\mathfrak{m}^k\subset\mathfrak{r}^k\cap A$; but since $\mathfrak{m}^k\supset\mathfrak{r}^k\cap A$, these two topologies are \emph{identical}.
\end{proof}

\begin{remark}[2.3.9]
\label{IV.2.3.9}
We will see further on~(7.8.3, (vii)) that for the Noetherian local rings $A$ most commonly used in Algebraic Geometry, the hypothesis that $A$ is integral and integrally closed implies that it has the same of $\widehat{A}$.
This is why, in Algebraic Geometry over a base field, one generally states~\sref{IV.2.3.8} under the hypothesis that $A$ is \emph{integral and integrally closed}.
\end{remark}

\begin{theorem}[2.3.10]
\label{IV.2.3.10}
Let $f:X\to Y$ be a quasi-flat morphism~\sref{IV.2.3.3}.
Then, for every pro-constructible part~(1.9.4) $Z$ of $Y$, we have $\overline{f^{-1}(Z)}=f^{-1}(\overline{Z})$.
\end{theorem}

\begin{proof}
Since $f$ is continuous, we have $\overline{f^{-1}(Z)}\subset f^{-1}(\overline{Z})$ and it amounts to proving that for every $x\in X$ such that $f(x)\in\overline{Z}$, $x$ is adherent to $f^{-1}(Z)$;
it is clear that the question is local on $Y$, so we can suppose $Y$ affine.
By virtue of the hypothesis, there exists an affine scheme $Y'$ and a morphism $g:Y'\to Y$ such that $g(Y')=Z$~(1.9.5,~(ix)).
Let $Y_1$ be the closed image of $Y'$ by $g$~\sref[I]{I.9.5.3}, and let $X_1$ be the closed sub-prescheme $f^{-1}(Y_1)$ of $X$; if $f_1:X_1\to Y_1$ is the restriction of $f$ to $X_1$, we know that $f_1$ is quasi-flat~\sref{IV.2.3.3}; we can therefore replace $X$, $Y$ by $X_1$, $Y_1$ respectively, in other words suppose that $g$ is \emph{dominant}.
Set then $X'=X\times_Y Y'$, and let $f'$ and $g'$ be the projections of $X'$ into $Y'$ and $X$ respectively, so that we have the commutative diagram
\[
\xymatrix{
  X \ar[d]_t & X' \ar[l]_{g'} \ar[d]^{f'} \\
  Y & Y' \ar[l]^g
}
\]
Since $f$ is quasi-flat, $g$ quasi-compact and dominant, it follows from~\sref{IV.2.3.7} (with the roles of $f$ and $g$ interchanged) that $g'$ is a dominant morphism, which proves the theorem.
\end{proof}

\begin{corollary}[2.3.11]
\label{IV.2.3.11}
Let $f$ be a quasi-flat and quasi-compact morphism, $F$ a closed part of $X$ such that $F=f^{-1}(f(F))$; then $F=f^{-1}(\overline{f(F)})$.
\end{corollary}

\begin{proof}
\oldpage[IV-2]{18}
Let $Y'$ be the reduced sub-prescheme of $X$ having $F$ as underlying space, and let $X_1$ be its closed image, so that $Z=f(F)$ is pro-constructible in $Y$~(1.9.5,~(vii)), and the corollary follows from the fact that $F$ is closed.

We can further write the result of~\sref{IV.2.3.11} in the form $F=f^{-1}(f(X)\cap\overline{f(F)})$, in other words, the closed subsets of the subspace $f(X)$ of $Y$ are the parts $Z\subset f(X)$ such that $f^{-1}(Z)$ is closed (resp.\ \emph{open}) in $X$; this also means that \emph{the topology induced on $Y$ by $f$ on $f(X)$ is the quotient of that of $X$ by the equivalence relation defined by $f$}.
In particular:
\end{proof}

\begin{corollary}[2.3.12]
\label{IV.2.3.12}
Let $f:X\to Y$ be a quasi-faithfully flat morphism~\sref{IV.2.3.3} and quasi-compact.
Then the topology of $Y$ is the quotient of that of $X$ by the equivalence relation defined by $f$, in other words, for $Z\subset Y$ to be open \textnormal{(}resp.\ closed\textnormal{)} in $Y$, it is necessary and sufficient that $f^{-1}(Z)$ be open \textnormal{(}resp.\ closed\textnormal{)} in $X$.
\end{corollary}

\begin{proof}
Indeed, we then have $f(X)=Y$.
\end{proof}

\begin{corollary}[2.3.13]
\label{IV.2.3.13}
Let $X$, $Y$ be two $S$-preschemes, $f:X\to Y$ an $S$-morphism faithfully flat and quasi-compact.
For $Y$ to be separated over $S$, it is necessary and sufficient that the canonical immersion $j:X\times_Y X\to X\times_S X$~\sref[I]{I.5.3.10} be \emph{closed}.
\end{corollary}

\begin{proof}
Let us note for this that we have the commutative diagram~\sref[I]{I.5.3.5}
\[
\xymatrix{
  X\times_Y X \ar[r]^j \ar[d] & X\times_S X \ar[d]^{f\times_S f}\\
  Y \ar[r]^{\Delta_Y} & Y\times_S Y
}
\]
identifying $X\times_Y X$ with the product of the $(Y\times_S Y)$-preschemes $Y$ and $X\times_S X$.
Since $f$ is surjective, so is $f\times_S f$, so the diagonal $\Delta_Y(Y)$ has for inverse image by $f\times_S f$ the image $j(X\times_Y X)$~\sref[I]{I.3.4.8}.
Since $f\times_S f$ is faithfully flat and quasi-compact~(1.1.2 and 2.2.13), it suffices to apply~\sref{IV.2.3.12} to this morphism.
\end{proof}

\begin{corollary}[2.3.14]
\label{IV.2.3.14}
Let $f:X\to Y$ be a quasi-faithfully flat morphism, and quasi-compact, and let $Z$ be a part of $Y$.
For $Z$ to be a locally closed part pro-constructible of $Y$, it is necessary and sufficient that $f^{-1}(Z)$ be a locally closed part pro-constructible of $X$.
\end{corollary}

\begin{proof}
We already know~(1.9.12) that, for $Z$ to be a pro-constructible part of $Y$, it is necessary and sufficient that $f^{-1}(Z)$ be a pro-constructible part of $X$.
The condition is evidently necessary.
To prove that it is sufficient, consider the closed reduced sub-prescheme $Y_1$ of $Y$ having $\overline{Z}$ as underlying space and let $X_1$ be its inverse image, so that $f^{-1}(\overline{Z})=\overline{f^{-1}(Z)}$ by virtue of~\sref{IV.2.3.10}.
As $f^{-1}(Z)$ is locally closed in $X$, it is open in $\overline{f^{-1}(Z)}$, so in $X_1$; now, the morphism $f_1:X_1\to Y_1$ deduced from $f$ by restriction is quasi-faithfully flat and quasi-compact~(1.1.2 and 2.3.3); we deduce from~\sref{IV.2.3.12} that $Z$ is open in $\overline{Z}$, which shows that $Z$ is locally closed in $Y$.
\end{proof}

\begin{remark}[2.3.15]
\label{IV.2.3.15}
It does not suffice, in~\sref{IV.2.3.12}, to suppose only that $f$ is \emph{faithfully flat}.
For example, take for $Y$ a reduced irreducible algebraic curve~\sref[II]{II.7.4.2} of generic point
\oldpage[IV-2]{19}
$y$, for $X$ the prescheme direct sum $\coprod\Spec(\sh{O}_\eta)$, where $\eta$ ranges over $Y$~\sref[I]{I.3.1}, for $f$ the canonical morphism, which is faithfully flat~\sref[I]{I.2.4.2};
if $\eta$ denotes the generic point of $Y$, $Z=\{\eta\}$ is not open in $Y$~\sref[II]{II.7.4.3}, but $f^{-1}(Z)=\Spec(\sh{O}_\eta)$ is open in $X$, since $\sh{O}_\eta$ is a field, and thus $f^{-1}(Z)$ is open in $X$.
\end{remark}

\subsection{Universally open morphisms and flat morphisms}
\label{subsection:IV.2.4}

\begin{env}[2.4.1]
\label{IV.2.4.1}
We have already defined~\sref[II]{II.5.4.9} the notion of \emph{universally closed} morphism; in the same manner, we give the following definitions:
\end{env}

\begin{definition}[2.4.2]
\label{IV.2.4.2}
We say that a morphism of preschemes $f:X\to Y$ is \emph{universally open} \textnormal{(}resp.\ \emph{universally bicontinuous}, resp.\ a \emph{universal homeomorphism}\textnormal{)} if, for every morphism $g:Y'\to Y$, the morphism $f_{(Y')}:X\times_Y Y'\to Y'$ is open \textnormal{(}resp.\ a homeomorphism onto its image, resp.\ a homeomorphism onto $Y'$\textnormal{)}.
\end{definition}

We will see further on~(14.3.2) that when $Y$ is locally Noetherian, the definition of universally open morphism given here is equivalent to the definition~\sref[III]{III.4.3.9} for morphisms of finite type; the reader can verify that we do not use the latter definition before \textsection~14.

\begin{proposition}[2.4.3]
\label{IV.2.4.3}
\medskip\noindent
\begin{enumerate}[label=(\roman*)]
  \item An immersion \textnormal{(}resp.\ an open immersion, resp.\ a closed immersion\textnormal{)} is universally bicontinuous \textnormal{(}resp.\ universally open, resp.\ universally closed\textnormal{)}.
  \item The composition of two universally open morphisms \textnormal{(}resp.\ universally closed, resp.\ universal homeomorphisms\textnormal{)} is also so.
  \item If $f:X\to Y$ is an $S$-morphism universally open \textnormal{(}resp.\ universally closed, resp.\ a universal homeomorphism\textnormal{)}, so is $f_{(S')}:X_{(S')}\to Y_{(S')}$ for every base change $S'\to S$.
  \item If $f:X\to X'$, $g:Y\to Y'$ are two $S$-morphisms universally open \textnormal{(}resp.\ universally closed, resp.\ universal homeomorphisms\textnormal{)}, so is $f\times_S g:X\times_S Y\to X'\times_S Y'$.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms such that $f$ is surjective; if $g\circ f$ is universally open \textnormal{(}resp.\ universally closed, resp.\ a universal homeomorphism\textnormal{)}, so is $g$.
  \item For $f$ to be a universally open morphism \textnormal{(}resp.\ universally closed, resp.\ a universal homeomorphism\textnormal{)}, it is necessary and sufficient that $f_{\mathrm{red}}$ be so.
  \item Let $(U_\alpha)$ be an open covering of $Y$. For $f:X\to Y$ to be universally open \textnormal{(}resp.\ universally closed, resp.\ a universal homeomorphism\textnormal{)}, it is necessary and sufficient that for every $\alpha$, its restriction $f^{-1}(U_\alpha)\to U_\alpha$ be so.
\end{enumerate}
\end{proposition}

\begin{proof}
Assertion~(i) follows from~\sref[I]{I.4.3.2}.
Assertion~(ii) also follows immediately from the definitions, and it is the same of~(iii) by reducing to the case where $Y=S$, $Y'=S'$, which one can do by~\sref[I]{I.3.3.11}; we know that~(iv) follows from~(ii) and~(iii)~\sref[I]{I.3.5.1}.
To prove~(v), let us note that for every morphism $Z'\to Z$, $f_{(Z)}:X_{(Z)}\to Y_{(Z)}$ is surjective~\sref[I]{I.3.5.2}; on the other hand, to note that a morphism $g$ is open (resp.\ closed, resp.\ a homeomorphism onto its image, resp.\ a bijective homeomorphism), it is the same of $g$, one can then restrict to the case where $g\circ f$ is open (resp.\ closed), the fact that $g$ is then open (resp.\ closed) follows from Bourbaki, \emph{Top.~g\'{e}n.}, chap.~I\textsuperscript{er}, 3\textsuperscript{e}~\'{e}d., \textsection~5, n\textsuperscript{o}~1, prop.~1; for the two other cases, we can restrict to assuming that $g(f(X))=g(Y)=Z$, in other words that $g\circ f$ is a homeomorphism of $X$ onto $Z$; since $f$ is surjective, $g$ is necessarily bijective, and since $g$ is a continuous open map by what precedes, $g$ is indeed a homeomorphism of $Y$ onto $Z$.

\oldpage[IV-2]{20}
To prove~(vi), let us note that to say that a morphism $g$ is open (resp.\ closed, resp.\ a bijective homeomorphism) is equivalent to saying that $g_{\mathrm{red}}$ has the same property.
On the other hand~\sref[I]{I.5.1.8}, for every morphism $Y''\to Y$, on a $(X_{\mathrm{red}}\times_{Y_{\mathrm{red}}}Y''_{\mathrm{red}})_{\mathrm{red}}=(X\times_Y Y'')_{\mathrm{red}}$, so the preceding remark shows that if $f_{\mathrm{red}}$ is universally open (resp.\ universally closed, resp.\ universally bicontinuous, resp.\ a universal homeomorphism), so is $f$.
The converse is proved in the same manner, noting here that for every morphism $Y''\to Y_{\mathrm{red}}$, we have $(X_{\mathrm{red}}\times_{Y_{\mathrm{red}}}Y'')_{\mathrm{red}}=(X\times_Y Y'')_{\mathrm{red}}$~\sref[I]{I.5.1.8}.

Finally, the necessity of condition~(vii) follows immediately from~(iii).
Conversely, suppose the condition satisfied, and let $g:Y'\to Y$ be a morphism; then the $g^{-1}(U_\alpha)=U'_\alpha$ form an open covering of $Y'$ and if one notes $f_\alpha$ the restriction $f^{-1}(U_\alpha)\to U_\alpha$ of $f$, and $f'$ the morphism $f_{(Y')}$, the restriction $f'^{-1}(U'_\alpha)\to U'_\alpha$ is nothing other than $(f_\alpha)_{(U'_\alpha)}$.
We can therefore restrict to proving that $f$ is open (resp.\ closed, resp.\ a homeomorphism onto its image, resp.\ a homeomorphism onto $Y'$), which is immediate.
\end{proof}

\begin{proposition}[2.4.4]
\label{IV.2.4.4}
A universally bicontinuous morphism $f:X\to Y$ is radicial \textnormal{(}and hence separated~(1.7.7.1)\textnormal{)}.
\end{proposition}

\begin{proof}
Indeed, $f$ is universally injective by hypothesis~\sref[I]{I.3.5.11}.
\end{proof}

\begin{proposition}[2.4.5]
\label{IV.2.4.5}
\medskip\noindent
\begin{enumerate}[label=(\roman*)]
  \item A morphism $f:X\to Y$ which is entire, surjective, and radicial is a universal homeomorphism.
  \item Conversely, suppose $Y$ locally Noetherian. Then, if a morphism of finite type $f:X\to Y$ is a universal homeomorphism, it is surjective, finite and radicial.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) It suffices to observe that the three properties of $f$ are conserved by base change~\sref[I]{I.3.5.2},~\sref[I]{I.3.5.7} and~\sref[II]{II.6.1.5}, and since an entire morphism is closed~\sref[II]{II.6.1.10}, it is clear that $f$ is a homeomorphism of $X$ onto $Y$.

(ii) Since $f$ is of finite type and universally closed by hypothesis and since it is separated by~\sref{IV.2.4.4}, and for every $y\in Y$, $f^{-1}(y)$ is reduced to a single element; so it is clear that $f$ is surjective, and it is radicial since it is universally injective~\sref[I]{I.3.5.11}.
Since $f$ is of finite type and universally closed, it is proper by hypothesis~\sref[II]{II.5.4.4}, and for every $y\in Y$, $f^{-1}(y)$ is finite; so~\sref[III]{III.4.4.2} $f$ is finite, and it is radicial since it is universally injective~\sref[I]{I.3.5.11}.
\end{proof}

\begin{theorem}[2.4.6]
\label{IV.2.4.6}
Let $f:X\to Y$ be a quasi-flat morphism~\sref{IV.2.3.3} and locally of finite presentation.
Then $f$ is universally open.
In particular, a flat morphism locally of finite presentation is universally open.
\end{theorem}

\begin{proof}
We know that for every base change $Y'\to Y$, $f_{(Y')}$ is quasi-flat~\sref{IV.2.3.3} and locally of finite presentation~(1.4.3,~(iii)).
It suffices therefore to prove that $f$ is an
\oldpage[IV-2]{21}
open morphism.
But this follows from the criterion~(1.10.4) for morphisms locally of finite presentation, conditions~\emph{b)}, \emph{b$'$)}, and \emph{c)} of~(1.10.4) being nothing other than conditions~(i), (ii), (iii) of~\sref{IV.2.3.4}.
\end{proof}

\begin{corollary}[2.4.7]
\label{IV.2.4.7}
For every prescheme $Y$, the structural morphism $\mathbf{V}_Y^n=Y\otimes_\mathbf{Z}\Spec(\mathbf{Z}[T_1,\ldots,T_n])$, also denoted $Y[T_1,\ldots,T_n]$\textnormal{)} is universally open.
\end{corollary}

\begin{proof}
Indeed, for $Y=\Spec(A)$, $Y[T_1,\ldots,T_n]=\Spec(A[T_1,\ldots,T_n])$, and $A[T_1,\ldots,T_n]$ is a free $A$-algebra of finite presentation.
\end{proof}

\begin{remark}[2.4.8]
\label{IV.2.4.8}
\medskip\noindent
\begin{enumerate}[label=(\roman*)]
  \item We note that a morphism $f:X\to Y$ faithfully flat and quasi-compact is \emph{not necessarily open}, even when $X$ and $Y$ are Noetherian.
    Take for example for $Y$ a reduced irreducible algebraic curve~\sref[II]{II.7.4.2} of generic point $y$, and let $X$ be the prescheme direct sum $Y\coprod\Spec(k(y))$, $f$ being the canonical morphism; it is clear that $f$ is flat and surjective, hence quasi-compact, but the image by $f$ of the open part $\Spec(k(y))$ of $X$ is the set $\{y\}$, which is not open in $Y$~\sref[II]{II.7.4.3}.
  \item For every prescheme $X$, the canonical morphism $f:X_{\mathrm{red}}\to X$ is a closed immersion and a \emph{universal homeomorphism}~\sref{IV.2.4.5},~(vi); but when $X$ is locally Noetherian, $f$ is not \emph{flat} unless $X$ is reduced, so $f=1_X$~\sref{IV.2.2.17}.
\end{enumerate}
\end{remark}

\begin{proposition}[2.4.9]
\label{IV.2.4.9}
Let $Y$ be a discrete prescheme. Then every morphism $f:X\to Y$ is universally open.
\end{proposition}

\begin{proof}
The question being local on $Y$~\sref{IV.2.4.3},~(vii), we can restrict to the case where the underlying space of $Y$ is reduced to a point; replacing $f$ by $f_{\mathrm{red}}$~\sref{IV.2.4.3},~(vi), we can moreover suppose that $Y$ is the spectrum of a \emph{field} $k$; on the other hand, for every base change $Y'\to Y$, the inverse images of affine open sets of $X$ cover $X'$, so we can suppose $X=\Spec(B)$ affine, $B$ being a \emph{$k$-algebra}.
It suffices then to prove that for every $k$-algebra $A'$, if we set $B'=B\otimes_k A'$, the image of every open subset of the form $U=D(t)$ ($t\in B'$) is open in $\Spec(A')$.
Now, $B$ is the inductive limit of the filtered increasing family $(B_\alpha)$ of its sub-$k$-algebras \emph{of finite type} (since the functor $\varinjlim$ commuting with the tensor product), $B'$ is the inductive limit of the $A'$-algebras $B_\alpha\otimes_k A'=B'_\alpha$;
there exists $\alpha$ such that $t$ is the image in $B'$ of an element $t_\alpha\in B'_\alpha$, so $D(t)$ is the inverse image by the canonical morphism $u_\alpha:\Spec(B')\to\Spec(B'_\alpha)$ of the open $U_\alpha=D(t_\alpha)$~\sref[I]{I.1.2.2}.
But since $k$ is a field, $B_\alpha$ is a $k$-algebra of finite presentation and a $k$-module flat, hence $B'_\alpha$ is an $A'$-algebra of finite presentation and a flat $A'$-module; we thus conclude from~\sref{IV.2.4.6} that the image of $U_\alpha$ by $f'_\alpha:\Spec(B'_\alpha)\to\Spec(A')$ is \emph{open} in $\Spec(A')$.
It therefore remains to see that $f'(U)=f'_\alpha(U_\alpha)$ and it remains to see that for every point $y'\in f'_\alpha(U_\alpha)$, the intersection $V=U\cap f'^{-1}(y')$ is not empty.
Now $V=u_\alpha^{-1}(V_\alpha)$, where $V_\alpha=U_\alpha\cap f_\alpha'^{-1}(y')$; in other words, $V_\alpha$ is a non-empty open subset of the prescheme $\Spec(B'_\alpha\otimes_{A'}k(y'))=\Spec(B_\alpha\otimes_k k(y'))$ and $V$ is its inverse image by the morphism $v:\Spec(B\otimes_k k(y'))\to\Spec(B_\alpha\otimes_k k(y'))$.

\end{proof}

where $r$ is the homomorphism \sref{IV.2.5.8.4}, $s$ the homomorphism \sref{IV.2.5.8.2} and $t$ the composite homomorphism
\[
  (\operatorname{Hom}_{\mathrm{A}}(\mathrm{V},\mathrm{W}))^{\mathrm{I}}\otimes_{\mathrm{A}}\mathrm{A}' \xrightarrow{w} (\operatorname{Hom}_{\mathrm{A}}(\mathrm{V},\mathrm{W})\otimes_{\mathrm{A}}\mathrm{A}')^{\mathrm{I}} \xrightarrow{s^{\mathrm{I}}} (\operatorname{Hom}_{\mathrm{A}'}(\mathrm{V}',\mathrm{W}'))^{\mathrm{I}}
\]

$w$ being the canonical homomorphism (Bourbaki, \emph{Alg.}, chap.~II, 3\textsuperscript{e} \'ed., \S~3, no.~7). One verifies immediately that the
diagram \sref{IV.2.5.8.5} is commutative, and as $\mathrm{A}'$ is a flat $\mathrm{A}$-module, its rows are exact. Finally, we have seen

that $s$ is an isomorphism, and it is likewise of $s^{\mathrm{I}}$; in case~(ii), we can take $\mathrm{I}$ finite, and we know then that
$w$ is bijective (Bourbaki, \emph{loc.\ cit.}, prop.~7); in case~(i), we note that if $\mathrm{B}'$ (resp.\ $\mathrm{B}''$) is the image of $\mathrm{B}$ by $h$

It follows therefore from \sref{IV.2.5.8.4} that if one sets

\begin{env}[2.5.8.7]

\emph{Reduction to the case where} $\mathrm{V}=\mathrm{B}_s$. The hypothesis that $\mathrm{V}'$ and $\mathrm{W}'$ are $\mathrm{B}'$-modules isomorphic implies

generator of $\mathrm{M}'$ in its capacity as $\mathrm{C}'$-module to the right; or, in case~(i), $\mathrm{V}'$ and $\mathrm{W}'$ are $\mathrm{A}'$-modules of finite type,
therefore Noetherian since $\mathrm{A}'$ is Noetherian; in case~(ii), the torsion
modules of $\mathrm{V}'$ are of finite presentation. One can therefore apply \sref{IV.2.5.7} to $\mathrm{A}'$, $\mathrm{B}'$, $\mathrm{V}'$, $\mathrm{W}'$ and to conclude
that $u'$ is bijective \sref[0]{0.6.4.1}, that is to say the
conclusion of \sref{IV.2.5.8}. But as $\mathrm{A}'$ is faithfully flat over $\mathrm{A}$, this implies that $u$ is a $\mathrm{B}'$-isomorphism
of $\mathrm{V}'$ onto $\mathrm{W}'$. But as $\mathrm{A}'$ is faithfully flat over $\mathrm{A}$, and one concludes that $u'$ is a $\mathrm{B}'$-isomorphism
is reduced (by changing the notations) to proving \sref{IV.2.5.8} in the particular case where $\mathrm{V}=\mathrm{B}_s$, and it suffices to
prove the existence of a \emph{generator} of the $\mathrm{B}$-module $\mathrm{W}$. We note that $\mathrm{B}$ is an $\mathrm{A}$-module \emph{of finite type}.
\end{env}

\begin{env}[2.5.8.8]
\label{IV.2.5.8.8}
\emph{Reduction to the case where $\mathrm{A}$ and $\mathrm{A}'$ are fields}, $\mathrm{B}$ an \emph{Artinian simple algebra over the centre} $\mathrm{A}$.

Suppose now that $\mathrm{B}$ is a \emph{finite product of fields} $k_i$ ($1\leq i\leq n$), $\mathrm{A}'$ is composed of direct
factors of $k_i$'s ($1\leq i\leq n$), each of the $\mathrm{A}'_i$ being cancelled by the $k_j$ of index $j\neq i$; the hypothesis that $\mathrm{A}'$ is a
faithfully flat $\mathrm{A}$-module implies that the $\mathrm{A}_i'$ are $\neq 0$; thus there exists for each an extension $\mathrm{A}_i''$ of $\mathrm{A}_i'$ which is a field,
and $\mathrm{A}''$, composed of direct factors of $\mathrm{A}_i''$, is a faithfully flat $\mathrm{A}$-module and a quotient of $\mathrm{A}'$. If one considers then
the ring $\mathrm{B}''=\mathrm{B}\otimes_{\mathrm{A}}\mathrm{A}''$, $\mathrm{W}''=\mathrm{W}\otimes_{\mathrm{A}}\mathrm{A}''$ and a $\mathrm{B}''$-module isomorphic to $\mathrm{B}_s''$; one can then
restrict ourselves to showing that \sref{IV.2.5.8} by replacing $\mathrm{A}'$ by $\mathrm{A}''$, that is to say one can also suppose that $\mathrm{A}'$ is a \emph{product
of fields}.

Let $Z$ be the centre of $\mathrm{B}$, which is a finite product of fields and an $\mathrm{A}$-module of finite type; we note that $\mathrm{B}$ and $\mathrm{W}$ are
$Z$-modules of finite type, projective as $Z$-modules; moreover, on a $\mathrm{B}'=\mathrm{B}\otimes_Z Z'$ and $\mathrm{W}'=\mathrm{W}\otimes_Z Z'$ by setting
$Z'=Z\otimes_{\mathrm{A}}\mathrm{A}'$, and $Z'$ is an $\mathrm{A}$-module faithfully flat. Thus one can reduce showing \sref{IV.2.5.8} by replacing $\mathrm{A}'$ by $Z'$; one can then
restrict ourselves to proving \sref{IV.2.5.8} in the case where $\mathrm{A}$ and $\mathrm{A}'$ are fields and $\mathrm{B}$ a simple algebra over its centre $\mathrm{A}$.

Let now $\mathrm{A}$ and $\mathrm{A}'$ be fields and suppose for the rest of the argument that $\mathrm{B}$ is a finite product of simple $k_i$-algebras ($1\leq i\leq n$), $k_i$ being an extension of $k_i$ and being
cancelled by the $k_j$ of index $j\neq i$; moreover, the reasoning made above shows
that one can suppose that $\mathrm{A}'$ is a product of fields $k_i'$ ($1\leq i\leq n$), $k_i'$ being an extension of $k_i$ and being
cancelled by the $k_j$ of index $j\neq i$; in this case, the $\mathrm{B}_i\otimes_{k_i} k_i'$ and $\mathrm{W}_i\otimes_{k_i} k_i'$ are $(\mathrm{B}_i\otimes_{k_i} k_i')$-modules isomorphic; it therefore suffices to prove \sref{IV.2.5.8} when
$n=1$, that is to say in the case where $\mathrm{A}$ and $\mathrm{A}'$ are fields and $\mathrm{B}$ a \emph{simple algebra over its centre} $\mathrm{A}$.
\end{env}

\begin{env}[2.5.8.9]
\label{IV.2.5.8.9}
\emph{End of the proof.} One knows then (Bourbaki, \emph{Alg.}, chap.~VIII, \S~5, prop.~6 and~8) that every
$\mathrm{B}$-module is a direct sum of modules isomorphic to a minimal ideal of $\mathrm{B}$, and that two $\mathrm{B}$-modules of finite rank over
$\mathrm{A}$ are therefore isomorphic if and only if they have the same \emph{rank} over $\mathrm{A}$. By hypothesis, on a $[\mathrm{W}':\mathrm{A}']=[\mathrm{B}'_s:\mathrm{A}']$. But
on a $[\mathrm{W}':\mathrm{A}']=[\mathrm{W}:\mathrm{A}]$ and $[\mathrm{B}'_s:\mathrm{A}']= [\mathrm{B}_s:\mathrm{A}]$; therefore $[\mathrm{W}:\mathrm{A}]=[\mathrm{B}_s:\mathrm{A}]$, which concludes the proof.
\end{env}
\end{proof}

\subsection{Permanence of set-theoretic and topological properties of morphisms
by faithfully flat descent}
\label{subsection:IV.2.6}

\begin{proposition}[2.6.1]
\label{IV.2.6.1}
Let $f:X\to Y$ be an $\mathrm{S}$-morphism of $\mathrm{S}$-preschemes, $g:\mathrm{S}'\to\mathrm{S}$
a surjective morphism, $X'=X_{(\mathrm{S}')}$, $Y'=Y_{(\mathrm{S}')}$, $f'=f_{(\mathrm{S}')}: X'\to Y'$. Consider, for a
morphism, the property of being:
\begin{enumerate}[label=(\roman*)]
  \item surjective;
  \item injective;
  \item with finite fibres (as sets);
  \item bijective;
  \item radicial.
\end{enumerate}
Then, if $\mathbf{P}$ denotes one of the preceding properties and if $f'$ possesses the property $\mathbf{P}$, the same is true
of $f$.
\end{proposition}

\begin{proof}
As the projection morphism $Y'\to Y$ is itself surjective \sref[I]{I.3.5.2}, one can,
by virtue of \sref[I]{I.3.3.11}, restrict to the case where $Y=\mathrm{S}$, $Y'=\mathrm{S}'$. For all $y\in Y$ (resp.\ $y'\in Y'$)
denote by $X_y$ (resp.\ $X'_{y'}$) the fibre prescheme $f^{-1}(y)$ (resp.\ $f'^{-1}(y)$) \sref[I]{I.3.6.2}; we know
that for $y'\in Y'$, $y=g(y')$, we have an isomorphism $X'_{y'}\xrightarrow{\;\sim\;}X_y\otimes_{k(y)}k(y')$ \sref[I]{I.3.6.4};
as the morphism $\Spec(k(y'))\to\Spec(k(y))$ is surjective, it is the same of the
projection $X'_{y'}\to X_y$ \sref[I]{I.3.5.2}; therefore if $X'_{y'}$ is non-empty (resp.\ has at most one point, resp.\
is a finite set), the same is true of $X_y$; as $g:Y'\to Y$ is surjective, this
proves the proposition in cases~(i) and~(iii), and~(iv) follows from~(i) and~(ii);
finally, to prove~(v), it suffices to show that if $f'$ is universally injective
\sref[I]{I.3.5.11}, the same is true of $f$; or, let $Y_1\to Y$ be any morphism; set
$X_1=X\times_Y Y_1$ and $f_1=f_{(Y_1)}$. On the other hand, set $Y_1'=Y_1\times_Y Y'$, $X_1'=X'\times_{Y'}Y_1'=X_1\times_{Y_1}Y_1'$,
$f_1'=f'_{(Y_1')}=(f_1)_{(Y_1')}:X_1'\to Y_1'$;
as $g'=g_{(Y_1)}:Y_1'\to Y'$ is surjective \sref[I]{I.3.5.2} and $f'$
is universally injective, $f_1'$ is injective, and it follows from~(ii) that $f_1$ is injective, whence
our assertion.
\end{proof}

\begin{proposition}[2.6.2]
\label{IV.2.6.2}
With the notations being those of \sref{IV.2.6.1}, suppose the morphism
$g:\mathrm{S}'\to\mathrm{S}$ faithfully flat and quasi-compact. Consider, for a morphism, the property of being:
\begin{enumerate}[label=(\roman*)]
  \item open;
  \item closed;
  \item quasi-compact and universally bicontinuous;
  \item a universal homeomorphism.
\end{enumerate}
Then, if $\mathbf{P}$ denotes one of the preceding properties and if $f'$ possesses the property $\mathbf{P}$, the same is true
of $f$.
\end{proposition}

\oldpage[IV-2]{28}

\begin{proof}
As the morphism $Y'\to Y$ is faithfully flat and quasi-compact \sref{IV.2.2.13} and
\sref{IV.1.1.2}, one can, by virtue of \sref[I]{I.3.3.11}, restrict to the case where $Y=\mathrm{S}$, $Y'=\mathrm{S}'$. If $g'$ is the
projection $X'\to X$, we know that we have, for every subset $\mathrm{M}$ of $X$, $g^{-1}(f(\mathrm{M}))=f'(g'^{-1}(\mathrm{M}))$
\sref[I]{I.3.4.8}. If $f'$ is an open morphism (resp.\ closed) then, for every open subset
(resp.\ closed subset) $\mathrm{M}$ of $X$, $f'(g'^{-1}(\mathrm{M}))$ is open (resp.\ closed) in $Y'$, and as $g$
is faithfully flat and quasi-compact, we conclude that $f(\mathrm{M})$ is open (resp.\ closed)
in $Y$ by virtue of \sref{IV.2.3.12}. This proves the proposition in cases~(i) and~(ii).
Let us prove it in case~(iii) (since this implies in case~(iv), taking into account
\sref{IV.2.6.1},~(iv)). By virtue of \sref{IV.2.6.1},~(ii) $f$ is injective, and it remains to prove that $f$, considered
as a map from $X$ to $f(X)$, is a quasi-compact and \emph{open} map. As $f'$
is quasi-compact, it is the same of $f$ \sref{IV.1.1.4}. It therefore suffices to show that for
every closed subset $Z$ of $X$, $Z=f^{-1}(\overline{f(Z)})$; as $g'$ is surjective, this relation
is equivalent to $g'^{-1}(Z)=g'^{-1}(f^{-1}(\overline{f(Z)}))$ or equivalently $g'^{-1}(Z)=f'^{-1}(g^{-1}(\overline{f(Z)}))$. Or,
as $f$ is quasi-compact, it is the same of its composite with the canonical injection
$Z\to X$ ($Z$ being here the sub-prescheme reduced of $X$ having $Z$ as underlying space);
applying \sref{IV.2.3.10} to the part $f(Z)$ of $Y$ (image of the morphism $f|Z$),
we get $g^{-1}(\overline{f(Z)})=\overline{g^{-1}(f(Z))}$; the formula to prove is therefore equivalent to the following:
$Z'=f'^{-1}(f'(Z'))$, where we have set $Z'=g'^{-1}(Z)$; but this formula results from the hypothesis
that $f'$ is a homeomorphism onto the sub-space image of $X'$ over $f'(X')$.
\end{proof}

\begin{remark}[2.6.3]
\label{IV.2.6.3}
In cases~(i) and~(ii), the conclusions of \sref{IV.2.6.2} are still
valid when one only supposes $g$ \emph{quasi-faithfully flat} \sref{IV.2.3.3} and \emph{quasi-compact};
indeed, by virtue of \sref{IV.2.1.4}, \sref[I]{I.3.5.2} and \sref[I]{I.9.1.13.1}, one can reduce to the case
where $Y=\mathrm{S}$, $Y'=\mathrm{S}'$; the conclusion results then from \sref{IV.2.3.12}. In cases~(iii) and~(iv),
the conclusions remain valid when one only supposes $g$ \emph{quasi-faithfully flat},
provided one already supposes $f$ quasi-compact; indeed, we then only use \sref{IV.2.3.10} and
the fact that $g$ is surjective. Finally, if $g$ is faithfully flat and locally of finite presentation,
or if $g$ is surjective and $\mathrm{S}$ discrete, the conclusion of \sref{IV.2.6.2} is valid when $\mathbf{P}$ is
the property:
\begin{enumerate}
  \item[(iii \emph{bis})] to be a homeomorphism onto the sub-space image;
\end{enumerate}
this results indeed from the proof given in \sref{IV.2.6.2} and Remark \sref{IV.2.4.11}.
\end{remark}

\begin{corollary}[2.6.4]
\label{IV.2.6.4}
With the notations being those of \sref{IV.2.6.1}, suppose the morphism
$g:\mathrm{S}'\to\mathrm{S}$ faithfully flat and quasi-compact. Consider, for a morphism, the property of being:
\begin{enumerate}[label=(\roman*)]
  \item universally open;
  \item universally closed;
  \item quasi-compact and universally bicontinuous;
  \item a universal homeomorphism;
  \item quasi-compact;
  \item quasi-compact and dominant.
\end{enumerate}
Then, if $\mathbf{P}$ denotes one of the preceding properties, for $f$ to possess the property $\mathbf{P}$, it is necessary
and it suffices that $f'$ possess it.
\end{corollary}

\begin{proof}
Properties~(v) and~(vi) are only given for the record, being consequences
of \sref{IV.1.1.4}, \sref{IV.1.1.6} and \sref{IV.2.3.7}. As far as the others are concerned, the condition is necessary
by virtue of \sref{IV.2.4.3}. Conversely, suppose for example that $f'$ is universally
open, and let $Y_1\to Y$ be any morphism; set $X_1=X\times_Y Y_1$ and $f_1=f_{(Y_1)}$.
On the other hand, set $Y_1'=Y_1\times_Y Y'$, $X_1'=X'\times_{Y'}Y_1'=X_1\times_{Y_1}Y_1'$, $f_1'=f'_{(Y_1')}=f_{1(Y_1')}:X_1'\to Y_1'$;
as $g'=g_{(Y_1)}:Y_1'\to Y'$ is faithfully flat and quasi-compact \sref{IV.2.2.13} and \sref{IV.1.1.2},
and $f'$ is universally open, $f_1'$ is open and it follows from \sref{IV.2.6.2} that $f_1$ is
open; therefore $f$ is universally open. The same reasoning applies for the other cases.

We note again here that one can replace ``faithfully flat'' by ``quasi-faithfully
flat'' and, when $g$ is moreover locally of finite presentation, or $g$
is surjective and $\mathrm{S}$ discrete, one can replace the property~(iii) by:
\begin{enumerate}
  \item[(iii \emph{bis})] universally bicontinuous.
\end{enumerate}
\end{proof}

\subsection{Permanence of various properties of morphisms by faithfully flat descent}
\label{subsection:IV.2.7}

\begin{proposition}[2.7.1]
\label{IV.2.7.1}
Let $f:X\to Y$ be an $\mathrm{S}$-morphism of $\mathrm{S}$-preschemes, $g:\mathrm{S}'\to\mathrm{S}$
a faithfully flat and quasi-compact morphism, $X'=X_{(\mathrm{S}')}$, $Y'=Y_{(\mathrm{S}')}$, $f'=f_{(\mathrm{S}')}:X'\to Y'$. Consider, for a morphism, the property of being:
\begin{enumerate}[label=(\roman*)]
  \item separated;
  \item quasi-separated;
  \item locally of finite type;
  \item locally of finite presentation;
  \item of finite type;
  \item of finite presentation;
  \item proper;
  \item an isomorphism;
  \item a monomorphism;
  \item an open immersion;
  \item a quasi-compact immersion;
  \item a closed immersion;
  \item affine;
  \item quasi-affine;
  \item finite;
  \item quasi-finite;
  \item integral.
\end{enumerate}
Then, if $\mathbf{P}$ denotes one of the preceding properties, for $f$ to possess the property $\mathbf{P}$, it is necessary
and it suffices that $f'$ possess it.
\end{proposition}

\begin{proof}
It has been proved in chapters~I\textsuperscript{er}, II and in chapter~IV, \S~1, that if $f$ possesses
one of the properties $\mathbf{P}$ above, the same is true of $f'$ (without hypothesis on the morphism
$g:\mathrm{S}'\to\mathrm{S}$). It therefore remains to prove the converse; as the projection $Y'\to Y$ is

\oldpage[IV-2]{30}

a faithfully flat and quasi-compact morphism \sref{IV.2.2.13} and \sref{IV.1.1.2}, one can restrict to
the case where $\mathrm{S}=Y$, $\mathrm{S}'=Y'$ by virtue of \sref[I]{I.3.3.11}.

(i) To say that $f$ is separated means that the diagonal morphism $\Delta_f:X\to X\times_Y X$ is
\emph{closed}; as $\Delta_{f'}=(\Delta_f)_{(Y')}$ \sref[I]{I.5.3.4}, if $\Delta_{f'}$ is closed, it is the same of $\Delta_f$ by virtue
of \sref{IV.2.6.2}, therefore $f$ is separated.

(ii) has already been proved under weaker hypotheses \sref{IV.1.2.5}.

(iii) and (iv): The question is evidently local over $X$ and $Y$, and, taking into account
\sref{IV.2.2.12}, it suffices therefore to prove the

\begin{lemma}[2.7.1.1]
\label{IV.2.7.1.1}
Let $\mathrm{A}$ be a ring, $\mathrm{B}$ an $\mathrm{A}$-algebra, $\mathrm{A}'$ an $\mathrm{A}$-algebra which is
a faithfully flat $\mathrm{A}$-module, $\mathrm{B}'=\mathrm{B}\otimes_{\mathrm{A}}\mathrm{A}'$. For $\mathrm{B}$ to be an $\mathrm{A}'$-algebra of finite type \textnormal{(}resp.\ of finite
presentation\textnormal{)}, it is necessary and it suffices that $\mathrm{B}'$ be an $\mathrm{A}'$-algebra of finite type \textnormal{(}resp.\ of finite
presentation\textnormal{)}.
\end{lemma}

We already know that the condition is necessary without hypothesis on $\mathrm{A}'$ \sref{IV.1.3.4},~\sref{IV.1.3.6},
\sref{IV.1.4.3} and also $\mathrm{B}'=\varinjlim(\mathrm{B}_\alpha\otimes_{\mathrm{A}}\mathrm{A}')$, the tensor product commuting with inductive
limits; if $(x_i')$ is a finite system of generators of the $\mathrm{A}'$-algebra $\mathrm{B}'$, there exists an index $\alpha$
such that all the $x_i'$ belong to the sub-algebra $\mathrm{B}_\alpha\otimes_{\mathrm{A}}\mathrm{A}'$ of $\mathrm{B}'$, whence $\mathrm{B}'=\mathrm{B}_\alpha\otimes_{\mathrm{A}}\mathrm{A}'$,
and as $\mathrm{A}'$ is faithfully flat, $\mathrm{B}=\mathrm{B}_\alpha\otimes_{\mathrm{A}}\mathrm{A}'$.

Suppose now that $\mathrm{B}'$ is an $\mathrm{A}'$-algebra of finite presentation; we already know
that $\mathrm{B}$ is an $\mathrm{A}'$-algebra of finite type, therefore there exists an $\mathrm{A}$-algebra
of polynomials $\mathrm{C}=\mathrm{A}[\mathrm{T}_1,\ldots,\mathrm{T}_m]$ and an $\mathrm{A}$-homomorphism surjective of algebras $\mathrm{C}\to\mathrm{B}$;
let $\mathfrak{J}$ be the kernel of this homomorphism, so that we have an exact sequence
$0\to\mathfrak{J}\to\mathrm{C}\to\mathrm{B}\to 0$, and therefore also an exact sequence $0\to\mathfrak{J}'\to\mathrm{C}'\to\mathrm{B}'\to 0$ (since $\mathrm{A}'$
is $\mathrm{A}$-flat), by setting $\mathrm{C}'=\mathrm{C}\otimes_{\mathrm{A}}\mathrm{A}'=\mathrm{A}'[\mathrm{T}_1,\ldots,\mathrm{T}_m]$, and $\mathfrak{J}'=\mathfrak{J}\otimes_{\mathrm{A}}\mathrm{A}'$ (identified to an
ideal of $\mathrm{C}'$). As $\mathrm{B}'$ is an $\mathrm{A}'$-algebra of finite presentation, $\mathfrak{J}'$ is a $\mathrm{C}'$-module
of finite type \sref{IV.1.4.4}; but on a $\mathfrak{J}'=\mathfrak{J}\otimes_{\mathrm{C}}\mathrm{C}'$ and $\mathrm{C}'$ is a $\mathrm{C}$-module faithfully flat
\sref{IV.2.2.13} and \sref{IV.2.2.3}; one knows then that the hypothesis that $\mathfrak{J}'$ is a $\mathrm{C}'$-module of finite type
implies that $\mathfrak{J}$ is a $\mathrm{C}$-module of finite type (Bourbaki, \emph{Alg.\ comm.}, chap.~I\textsuperscript{er}, \S~3, no.~6,
prop.~11), therefore $\mathrm{B}$ is an $\mathrm{A}$-algebra of finite presentation.

(v) results from~(iii) and from \sref{IV.2.6.2},~(v) by virtue of \sref{IV.1.5.2}.

(vi) results similarly from~(iv), (v) and~(ii) by virtue of \sref{IV.1.6.1}.

(vii) results from~(i), (v) and \sref{IV.2.6.4},~(ii) \sref[II]{II.5.4.1}.

(viii) We note first that as $f'$ is an isomorphism, it is a homeomorphism
universal, therefore the same is true of $f$ \sref{IV.2.6.4}; one concludes already that $f$ is
quasi-compact and separated \sref{IV.2.4.4}. Set $f=(\psi,\theta)$, where $\psi$ is therefore a homeomorphism;
it is a question of proving that $\theta:\sh{O}_Y\to f_*(\sh{O}_X)$ is an isomorphism of $\sh{O}_Y$-Modules. Or, if one
sets $f'=(\psi',\theta')$, the homomorphism $\theta':\sh{O}_{Y'}\to f'_*(\sh{O}_{X'})$ is composed of the
canonical homomorphism $g^*(f_*(\sh{O}_X))\to f'_*(\sh{O}_{X'})$ and of $g^*(\theta)$ \sref{IV.2.3.2}; but the first of these two homo-
morphisms is bijective by virtue of the hypothesis on $g$ \sref{IV.2.3.1}, therefore if $\theta'$ is bijective,
the same is true of $g^*(\theta)$, and as $g$ is faithfully flat, $\theta$ is bijective \sref{IV.2.2.7}, which
proves~(viii).

(ix) The proposition results from~(viii), from \sref[I]{I.5.3.4} and from \sref[I]{I.5.3.8} which reduces
monomorphisms to isomorphisms.

(x) If $f'$ is an open immersion, $f'(X')$ is open in $Y'$, and we have
$f'(X')=g^{-1}(f(X))$ \sref[I]{I.3.4.8}; it follows from \sref{IV.2.3.12} that $f(X)$ is open. One can then
replace $Y$ (resp.\ $Y'$) by the sub-prescheme induced on the open set $f(X)$ (resp.\ $f'(X')$);
taking into account \sref{IV.1.1.2} and \sref{IV.2.2.13}; then $f'$ becomes an isomorphism, therefore it is the same
of $f$ by~(viii), and this establishes~(x).

\oldpage[IV-2]{31}

(xi) If $f'$ is a quasi-compact immersion, $f'$ is a quasi-compact
morphism and quasi-separated \sref{IV.1.2.2}, therefore the same is true of $f$ by~(ii) and \sref{IV.2.6.2},~(v). Let $Z$ be the
closed sub-prescheme of $Y$, closed image of $X$ by $f$ \sref{IV.1.7.8}, and set $f=j\circ g$ with $j'=j_{(Y')}$, $g'=g_{(Y')}$,
and we know that $j'$
identifies with the canonical injection $Z'\to Y'$ of the sub-prescheme $Z'$ of $Y'$, closed image
of $X'$ by $f'$ \sref{IV.2.3.2}. The hypothesis on $f'$ means then that $g'$ is an open immersion
ouverte \sref[I]{I.9.5.10}, therefore the same is true of $g$ by~(x), and this shows that $f$ is an
immersion.

(xii) To say that $f$ (resp.\ $f'$) is a closed immersion means that $f$ (resp.\ $f'$) is a quasi-compact
immersion and a closed morphism; we see therefore that~(xii) results from~(xi)
and from \sref{IV.2.6.2},~(ii).

(xiii) and (xiv) Suppose $f'$ affine (resp.\ quasi-affine); note then that $f'$ is quasi-
compact and quasi-separated \sref[II]{II.5.1.1}, therefore the same is true of $f$ by~(ii) and \sref{IV.2.6.2},~(v).
Set $\sh{A}=f_*(\sh{O}_X)$, $\sh{A}'=f'_*(\sh{O}_{X'})$; by virtue of \sref{IV.2.3.1}, the canonical homomorphism
of $\sh{O}_{Y'}$-Algebras $g^*(\sh{A})\to\sh{A}'$ is bijective; consequently, if $h:Z=\Spec(\sh{A})\to Y$ is
the structural morphism, the structural morphism $h':Z'=\Spec(\sh{A}')\to Y'$ identifies
to $h_{(Y')}$ \sref[II]{II.1.5.2}. Let then $u:X\to Z$ (resp.\ $u':X'\to Z'$) be the $Y$-morphism (resp.\
$Y'$-morphism) canonical corresponding to the identical homomorphism of $\sh{A}$ (resp.\ $\sh{A}'$)
\sref[II]{II.1.2.7}; as on the commutative diagram
\[
\xymatrix{
  X \ar[d]_u & X' \ar[l] \ar[d]^{u'} \\
  Z \ar[d]_h & Z' \ar[l] \ar[d]^{h'} \\
  Y & Y' \ar[l]_g
}
\]

and $h'\circ u'=f'$, it follows from \sref[II]{II.1.2.7} that $u'=u_{(Z')}$. Moreover, $g'$
is faithfully flat and quasi-compact \sref{IV.1.1.2} and \sref{IV.2.2.13}. The hypothesis on $f'$ signifies that $u'$ is
an isomorphism (resp.\ an open immersion) \sref[II]{II.5.1.6}; it follows then from~(viii)
(resp.\ (x)) that $u$ is an isomorphism (resp.\ an open immersion), whence~(xiii)
(resp.\ (xiv)).

\oldpage[IV-2]{32}

(xv) If $f'$ is finite, it is affine, therefore $f$ is also affine by~(xiii); moreover, with the notations
of the proof of~(xiii), $\sh{A}'$ is an $\sh{O}_{Y'}$-Module of finite type, and $\sh{A}'$ is isomorphic
to $g^*(\sh{A})$; it follows from \sref{IV.2.5.2} that $\sh{A}$ is an $\sh{O}_Y$-Module of finite type, therefore $f$ is a finite
morphism.

(xvi) To say that $f$ is quasi-finite means that $f$ is a morphism of finite type and that
for all $y\in Y$, $f^{-1}(y)$ is finite \sref[II]{II.6.2.2} and \textbf{I}, \sref{IV.6.4.4}; the conclusion results therefore from~(v)
and from~(xv).

(xvii) One argues as in~(xv) that $f$ is affine. One can restrict to the case
where $Y=\Spec(\mathrm{A})$, $Y'=\Spec(\mathrm{A}')$, and then $X=\Spec(\mathrm{B})$, $X'=\Spec(\mathrm{B}')$, where $\mathrm{B}'=\mathrm{B}\otimes_{\mathrm{A}}\mathrm{A}'$;
$\mathrm{B}$ is equal to the inductive limit of its sub-$\mathrm{A}$-algebras of finite type $\mathrm{B}_\alpha$, therefore
$\mathrm{B}'=\varinjlim\mathrm{B}'_\alpha$ where $\mathrm{B}'_\alpha=\mathrm{B}_\alpha\otimes_{\mathrm{A}}\mathrm{A}'$, and $\mathrm{B}'_\alpha$ is an $\mathrm{A}'$-algebra of finite type. But by
hypothesis $\mathrm{B}'$ is integral over $\mathrm{A}'$, therefore $\mathrm{B}'_\alpha$ is an $\mathrm{A}'$-\emph{module} of finite type, and $\mathrm{B}_\alpha$ is therefore also an
$\mathrm{A}$-module of finite type \sref{IV.2.5.2}.
\end{proof}

\begin{corollary}[2.7.2]
\label{IV.2.7.2}
With the hypotheses and notations being those of \sref{IV.2.7.1}, suppose $f$ quasi-
compact; let $\sh{L}$ be an $\sh{O}_X$-Module invertible, $\sh{L}'=\sh{L}\otimes_{\sh{O}_Y}\sh{O}_{Y'}$ its inverse image. For $\sh{L}$
to be ample \textnormal{(}resp.\ very ample\textnormal{)} for $f$, it is necessary and it suffices that $\sh{L}'$ be ample \textnormal{(}resp.\ very ample\textnormal{)} for $f'$.
\end{corollary}

\begin{proof}
The condition is necessary without hypothesis on $g:\mathrm{S}'\to\mathrm{S}$ \sref[II]{II.4.4.10} and \sref{IV.4.6.13};
to see that it is sufficient, one can, as in \sref{IV.2.7.1} limit to the case where $\mathrm{S}=Y$,
$\mathrm{S}'=Y'$. The hypothesis on $\sh{L}'$ implies that $f'$ is quasi-compact and separated \sref[II]{II.4.6.1},
therefore it is the same of $f$ by~((2.6.2,~(v)) and \sref{IV.2.7.1},~(i)). Setting $\sh{E}=f_*(\sh{L})$, $\sh{E}'=f'_*(\sh{L}')$;
it follows from \sref{IV.2.3.1} that the canonical homomorphism $u:g^*(\sh{E})\to\sh{E}'$ is bijective. If $\sh{L}'$
is very ample for $f$, the canonical homomorphism $\sigma':f'^*(\sh{E}')\to\sh{L}'$ is surjective, and the
morphism $r'=r_{\sh{L}',\sigma'}:X'\to\mathbf{P}(\sh{E}')$ is an immersion
\sref[II]{II.4.4.4},~$b$)) necessarily
quasi-compact \sref{IV.1.1.2},~(v)). The fact that $u:g^*(\sh{E})\to\sh{E}'$ is bijective implies that if
$h:\mathbf{P}(\sh{E})\to Y$, $h':\mathbf{P}(\sh{E}')\to Y'$ are the structural morphisms, $h'$ identifies to $h_{Y'}$
\sref[II]{II.4.1.3}; on the other hand, by designating by $g'$ the projection $X'\to X$, $g'$ is faithfully
flat \sref{IV.2.2.13}, and on a $f\circ g'=g\circ f'$, and one verifies easily that the homomorphism
$g^*(\sigma):g^*(f^*(f_*(\sh{L})))\to g^*(\sh{L})$ is identical to the composite homomorphism
\[
  f'^*(g^*(f_*(\sh{L}))) \xrightarrow{f'^*(u)} f'^*(f'_*(\sh{L}')) \xrightarrow{\sigma'} \sh{L}'
\]

(for example by reducing to the case where $Y$ and $Y'$ are affine). As $\sigma'$ is surjective
and $f'^*(u)$ bijective, it is the same of $\sigma'$ therefore the same of $\sigma$ \sref{IV.2.2.7}.
One concludes therefore that the morphism $r=r_{\sh{L},\sigma}:X\to\mathbf{P}(\sh{E})$ is defined everywhere \sref[II]{II.3.7.4};
moreover, if $l'$ is the projection $\mathrm{P}'\to\mathrm{P}$, and if $g''$ is the
projection $r'$ identifies to $r_{(P')}$ \sref[II]{II.4.2.10} and $g''$ is faithfully flat
and quasi-compact \sref{IV.1.1.2} and \sref{IV.2.2.13}. On
concludes therefore from \sref{IV.2.7.1},~(xi)) that $r$ is an immersion, and therefore $\sh{L}$ is \emph{very ample}
\sref[II]{II.4.4.4},~$b$)).

Suppose now that $\sh{L}'$ is ample for $f'$; to prove that $\sh{L}$ is ample
for $f$, one can restrict to the case where $Y$ is affine \sref[II]{II.4.6.4}, and by virtue of \sref{IV.2.2.12} and
of \sref[II]{II.4.6.13}, one can also suppose that $Y'$ is affine. Then $X$ and $X'$ are
quasi-compact schemes, and to prove that $\sh{L}$ is $f$-ample, one can apply the
criterion of \sref[II]{II.4.6.8},~$c$)). Let therefore $\sh{F}$ be an $\sh{O}_X$-Module quasi-coherent of finite type;
if $\sigma:f^*(f_*(\sh{F}\otimes\sh{L}^{\otimes n}))\to\sh{F}\otimes\sh{L}^{\otimes n}$ is the canonical homomorphism, we see as
above that $g^*(\sigma)$ is the composite homomorphism
\[
  f'^*(g^*(f_*(\sh{F}\otimes\sh{L}^{\otimes n}))) \xrightarrow{f'^*(u)} f'^*(f'_*(\sh{F}'\otimes\sh{L}'^{\otimes n})) \xrightarrow{\sigma'} \sh{F}'\otimes\sh{L}'^{\otimes n}
\]

by setting $\sh{F}'=g^*(\sh{F})$, taking into account \sref[0]{0.4.3.3.1} and designating by $u$ the
canonical homomorphism $g^*(f_*(\sh{F}\otimes\sh{L}^{\otimes n}))\to f'_*(\sh{F}'\otimes\sh{L}'^{\otimes n})$. Or, we know that $u$ is bijective
for all $n$ \sref{IV.2.3.1}; on the other hand, as $\sh{F}'$ is quasi-compact and of finite type,
the hypothesis that $\sh{L}'$ is ample for $f'$ implies the existence of an $n_0$ such that for $n\geq n_0$,
$\sigma'$ is surjective; we see therefore that $g^*(\sigma)$ is surjective for $n\geq n_0$, and as $g'$ is
faithfully flat, $\sigma$ is surjective for these values of $n$ \sref{IV.2.2.7}, which completes the
proof.
\end{proof}

\begin{remarks}[2.7.3]
\label{IV.2.7.3}
\begin{enumerate}[label=(\roman*)]
  \item It follows from \sref{IV.2.6.1}, \sref{IV.2.6.4} and \sref{IV.2.5.4.1} that the
conclusions of \sref{IV.2.7.1} are still valid in cases~(i), (iii), (v), (vii) and~(xvi)
when one only supposes $g$ quasi-compact and \emph{quasi-faithfully flat} \sref{IV.2.3.3};
on has already remarked that \sref{IV.2.7.1} is valid in case~(ii) assuming only $g$
surjective and quasi-compact.

  \item With the hypotheses and notations of \sref{IV.2.7.1}, one can assert that $f$ is \emph{proper}
and $f'$ is \emph{projective} without $f$ being a \emph{quasi-projectif}. Indeed, Hironaka~\cite{IV-34} gave an example
of a proper non-projective morphism $f:X\to Y$, where $X$ and $Y$ are two algebraic schemes
regular \sref[0]{0.4.1.4} over the same field $k$, $Y$ being projective over $k$; moreover $Y$ is the union
of two affine open sets $Y_i$ ($i=1$, 2) such that $f_i:X\times_Y Y_i\to Y_i$ is projective for $i=1$, 2.
Let then $Y'=Y_1\amalg Y_2$ the prescheme coproduct; it is clear that the canonical morphism
$g:Y'\to Y$ coinciding with the canonical injections from $Y_1$ and $Y_2$, is faithfully flat,
and it is quasi-compact by virtue of \sref[I]{I.5.5.10}; however, $f':X\times_Y Y'\to Y'$
(coinciding with $f_i$ in each of the $Y_i$) is projective \sref[II]{II.5.5.6}, it is nonetheless not true of
$f$ itself. There exists therefore an $\sh{O}_X$-Module invertible $\sh{L}'$ which is $f'$-ample but which
\emph{is not of the form $g^*(\sh{L})$ for an $\sh{O}_X$-Module invertible $\sh{L}$}, by virtue of \sref{IV.2.7.2}.

  \item Under the hypotheses of \sref{IV.2.7.1}, one can assert that $f'$ is a \emph{local}
isomorphism without $f$ being a local immersion. Indeed, let $k$ be a field, $\bar{k}$ an algebraic closure
of $k$, $\mathrm{K}$ a separable extension of finite degree of $k$, distinct from $k$; then the
structural morphism $f:X\to Y$, where $X=\Spec(\mathrm{K})$ and $Y=\Spec(k)$, is not a
local immersion, but if one takes $Y'=\Spec(\bar{k})$, the morphism $Y'\to Y$ is faithfully
flat and quasi-compact, and $f'$ is a local isomorphism, since $X'=X\times_Y Y'$ is a coproduct
of a finite number of schemes isomorphic to $Y'$.
\end{enumerate}
\end{remarks}

\subsection{Preschemes over a regular base of dimension \texorpdfstring{$1$}{1}; closure of a closed
sub-prescheme of the generic fibre}
\label{subsection:IV.2.8}

\begin{proposition}[2.8.1]
\label{IV.2.8.1}
Let $Y$ be a prescheme locally Noetherian, regular, irreducible,
and of dimension $1$, with generic point $\eta$, $f:X\to Y$ a morphism, $\mathrm{X}_\eta=f^{-1}(\eta)$ the fibre at the point

\oldpage[IV-2]{34}

generic, $i:\mathrm{X}_\eta\to X$ the canonical morphism. Let $\sh{F}$ be an $\sh{O}_X$-Module quasi-coherent, $\sh{F}_\eta=i^*(\sh{F})$,
$\sh{G}'$ an $\sh{O}_{\mathrm{X}_\eta}$-Module quotient of $\sh{F}_\eta$, and let $\overline{\sh{G}'}$ be the $\sh{O}_X$-Module, image of $\sh{F}$ by the composite homomorphism
\sref[0]{0.4.4.3.2}
\[
  \sh{F} \xrightarrow{p_{\sh{F}}} i_*(i^*(\sh{F})) \to i_*(\sh{G}').
\]

Then $\overline{\sh{G}'}$ is an $\sh{O}_X$-Module quotient of $\sh{F}$, quasi-coherent and $f$-flat, such that $i^*(\overline{\sh{G}'})=\sh{G}'$, and it is
the unique $\sh{O}_X$-Module quotient of $\sh{F}$ having these properties.
\end{proposition}

\begin{proof}
As $i_*$ is quasi-compact and quasi-separated \sref{IV.1.1.2} and \sref{IV.1.2.2}, it follows from \sref{IV.1.7.4}
that for all $\sh{O}_{\mathrm{X}_\eta}$-Module quasi-coherent $\sh{H}'$, $i_*(\sh{H}')$ is an $\sh{O}_X$-Module quasi-coherent;
moreover, for every open subset $U$ of $X$, we have $(i_*(\sh{H}'))|U=(i|U\cap\mathrm{X}_\eta)_*(\sh{H}'|U\cap\mathrm{X}_\eta)$ by
definition \sref[0]{0.3.4.1}. If one proves the proposition when $X$ and $Y$ are affine, it
follows by gluing in the general case, in view of the assertion of uniqueness valid in
the affine case. In other words, one is reduced to proving the

\begin{lemma}[2.8.1.1]
\label{IV.2.8.1.1}
Let $\mathrm{A}$ be a Noetherian regular ring \sref[0]{0.17.3.6}, integral of
dimension $1$, $\mathrm{K}$ its field of fractions, $\mathrm{M}$ an $\mathrm{A}$-module, $\mathrm{N}'$ a $\mathrm{K}$-module quotient of $\mathrm{M}_{(\mathrm{K})}=\mathrm{M}\otimes_{\mathrm{A}}\mathrm{K}$
by a sub-$\mathrm{K}$-module $\mathrm{P}'$, $\overline{\mathrm{N}}'$ the image of $\mathrm{M}$ by the composite homomorphism $\mathrm{M}\to\mathrm{M}_{(\mathrm{K})}\to\mathrm{N}'$.
Then $\overline{\mathrm{N}}'$ is a flat $\mathrm{A}$-module, and is the unique quotient module $\mathrm{N}$ of $\mathrm{M}$ which is a flat $\mathrm{A}$-module
and such that the kernel of the surjective homomorphism $\mathrm{M}_{(\mathrm{K})}\to\mathrm{N}_{(\mathrm{K})}$ is equal to $\mathrm{P}'$.
\end{lemma}

\begin{proof}
As for every maximal ideal $\mathfrak{m}$ of $\mathrm{A}$, $\mathrm{A}_\mathfrak{m}$ is a local regular ring of
dimension $1$, and therefore a discrete valuation ring, it amounts to the same to say that an
$\mathrm{A}$-module $\mathrm{N}$ is flat or that it is \emph{without torsion} \sref[0]{0.6.3.4}. As $\mathrm{N}'$ is a $\mathrm{K}$-vector space,
therefore without torsion, $\overline{\mathrm{N}'}$ is the same as $\overline{\mathrm{N}'}$, a sub-module
of $\mathrm{N}'$; moreover, it is immediately verified that $\overline{\mathrm{N}'}_{(\mathrm{K})}$ is identified with $\mathrm{N}'$. Conversely, if $\mathrm{N}$
is a flat $\mathrm{A}$-module quotient of $\mathrm{M}$, the fact that $\mathrm{N}$ is an
$\mathrm{A}$-module flat implies that the canonical homomorphism $\mathrm{N}\to\mathrm{N}_{(\mathrm{K})}=\mathrm{N}\otimes_{\mathrm{A}}\mathrm{K}$ is \emph{injective}.
As $\mathrm{N}_{(\mathrm{K})}$ is identified with $\mathrm{N}'$, the conclusion results from the commutativity of the diagram
\[
\xymatrix{
  \mathrm{M} \ar[r] & \mathrm{N} \\
  \mathrm{M}_{(\mathrm{K})} \ar[u] \ar[r] & \mathrm{N}_{(\mathrm{K})} \ar[u]
}
\]
\end{proof}
\end{proof}

\begin{corollary}[2.8.2]
\label{IV.2.8.2}
Under the conditions of \sref{IV.2.8.1}, for $\sh{F}$ to be $f$-flat, it is necessary and
it suffices that the canonical homomorphism $\sh{F}\to i_*(\sh{F}_\eta)=i_*(i^*(\sh{F}))$ be injective.
\end{corollary}

\begin{env}[2.8.3]
\label{IV.2.8.3}
The formation of the $\sh{O}_X$-Module $\overline{\sh{G}'}$ is \emph{functorial} in $\sh{F}$ and $\sh{G}'$: to be precise,
if $\sh{F}_1$, $\sh{F}_2$ are two $\sh{O}_X$-Modules quasi-coherent, $u:\sh{F}_1\to\sh{F}_2$ an $\sh{O}_X$-homomorphism,
$\sh{G}'_i$ an $\sh{O}_{\mathrm{X}_\eta}$-Module quotient of $(\sh{F}_i)_\eta$ ($i=1$, 2) and $v:\sh{G}'_1\to\sh{G}'_2$ a homomorphism
making the diagram
\[
\xymatrix{
  (\sh{F}_1)_\eta \ar[r]^{i^*(u)} & (\sh{F}_2)_\eta \\
  \sh{G}'_1 \ar[u] \ar[r]_v & \sh{G}'_2 \ar[u]
}
\]

commutative, then there is a unique homomorphism $w:\overline{\sh{G}'_1}\to\overline{\sh{G}'_2}$ making the diagram
\[
\xymatrix{
  \sh{F}_1 \ar[r]^u & \sh{F}_2 \\
  \overline{\sh{G}'_1} \ar[u] \ar[r]_w & \overline{\sh{G}'_2} \ar[u]
}
\]

commutative (determined uniquely, when it exists, by this property),

\oldpage[IV-2]{35}

and this is a homomorphism determined uniquely (when it exists) by this property.
\end{env}

\begin{proposition}[2.8.4]
\label{IV.2.8.4}
The hypotheses on $Y$ being those of \sref{IV.2.8.1}, let $\mathrm{X}_1$, $\mathrm{X}_2$ be
two $Y$-preschemes, $\sh{F}_i$ an $\sh{O}_{\mathrm{X}_i}$-Module quasi-coherent, $\sh{G}'_i$ an $\sh{O}_{(\mathrm{X}_i)_\eta}$-Module quotient of $(\sh{F}_i)_\eta$
($i=1$, 2). Then we have
\end{proposition}

\begin{env}[2.8.4.1]
\label{IV.2.8.4.1}
\[
  \overline{\sh{G}'_1\otimes_{\sh{O}_Y}\sh{G}'_2} = \overline{\sh{G}'_1}\otimes_{k(\eta)}\overline{\sh{G}'_2}.
\]
\end{env}

\begin{proof}
Indeed, setting $X=\mathrm{X}_1\times_Y\mathrm{X}_2$; the first member of \sref{IV.2.8.3.1} is an $\sh{O}_X$-Module
quasi-coherent which is $Y$-flat \sref[0]{0.6.2.1}, whose inverse image in $\mathrm{X}_\eta$ is $\sh{G}'_1\otimes_{k(\eta)}\sh{G}'_2$
\sref[I]{I.9.1.5} and which is a quotient of $\sh{F}_1\otimes_{\sh{O}_Y}\sh{F}_2$; the conclusion results therefore from the property
of uniqueness of \sref{IV.2.8.1}.
\end{proof}

\begin{proposition}[2.8.5]
\label{IV.2.8.5}
The hypotheses on $X$ and $Y$ being those of \sref{IV.2.8.1}, let $Z'$ be a
closed sub-prescheme of $\mathrm{X}_\eta$. There exists then a unique closed sub-prescheme $\overline{Z'}$ of $X$ which is $Y$-flat
and such that $i^{-1}(\overline{Z'})=Z'$.
\end{proposition}

\begin{proof}
If $\sh{J}'$ is the quasi-coherent Ideal of $\sh{O}_{\mathrm{X}_\eta}$ defining $Z'$, it suffices to apply \sref{IV.2.8.1}
to the case where $\sh{F}=\sh{O}_X$ and $\sh{G}'=\sh{O}_{\mathrm{X}_\eta}/\sh{J}'$; if $\overline{\sh{G}'}=\sh{O}_X/\sh{J}$, we have indeed $\sh{J}'=i^*(\sh{J})\sh{O}_{\mathrm{X}_\eta}$, therefore
$i^{-1}(\overline{Z'})=Z'$ \sref[I]{I.4.4.5}.
\end{proof}

We note that the prescheme $\overline{Z'}$ is the \emph{closed image} of $Z'$ by the composite morphism
$Z'\to\mathrm{X}_\eta\xrightarrow{i} X$, where the first arrow is the canonical injection \sref[I]{I.9.5.3}; its underlying
space is the closure \emph{in} $X$ of $Z'$ \sref[I]{I.9.5.4}, which justifies the adopted notation.
We say moreover that $\overline{Z'}$ is the sub-prescheme of $X$ \emph{closure} of $Z'$.

\begin{corollary}[2.8.6]
\label{IV.2.8.6}
Let $\mathrm{X}_1$, $\mathrm{X}_2$ be two $Y$-preschemes, $Z'_i$ a closed sub-prescheme
of $(\mathrm{X}_i)_\eta$ ($i=1$, 2). Then we have
\end{corollary}

\begin{env}[2.8.6.1]
\label{IV.2.8.6.1}
\[
  \overline{Z'_1\times_Y Z'_2} = \overline{Z'_1}\times_{k(\eta)}\overline{Z'_2}.
\]
\end{env}

\begin{proof}
This results from \sref{IV.2.8.4} and \sref{IV.2.8.5}.
\end{proof}
