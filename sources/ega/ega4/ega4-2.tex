\section{Base change and flatness}
\label{section:IV.2}

\oldpage[IV-2]{5}

This section (contrary to \textsection\hyperref[section:IV.6]{6}) only exceptionally uses Noetherian techniques.
Nos.~1 and~2 are scarcely more than translations of elementary properties of flatness in commutative algebra (cf.~Bourbaki, \emph{Alg.~comm.}, chap.~I), and are included here for convenience of reference.
The following numbers are primarily devoted to properties of ``descent'' by flat or faithfully flat morphisms: if $g:Y'\to Y$ is such a morphism, the goal is to be able to assert that a part of $Y$, or an $\sh{O}_Y$-Module, or a morphism $X\to Y$, has a certain property, when one knows that its inverse image by $g$ possesses this property.
We restrict ourselves here to properties that do not make use of the general technique of ``descent'', which will be developed in Chapter~V.

\subsection{Flat modules on preschemes}
\label{subsection:IV.2.1}

\begin{env}[2.1.1]
\label{IV.2.1.1}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ an $\sh{O}_X$-Module;
recall $(\mathbf{0_I}$, 6.7.1) that $\sh{F}$ is said to be \emph{$f$-flat} (or \emph{$Y$-flat}) \emph{at a point} $x\in X$ if $\sh{F}_x$ is a flat $\sh{O}_{f(x)}$-module, \emph{$f$-flat} (or \emph{$Y$-flat}) if it is $f$-flat at every $x\in X$, and finally that the morphism $f$ is said to be \emph{flat at the point} $x\in X$ (resp. \emph{flat}) if $\sh{O}_X$ is $f$-flat at the point $x$ (resp. $f$-flat).
When $f=1_X$, we will simply say that an $\sh{O}_X$-Module $\sh{F}$ is \emph{flat at the point} $x$ (resp. \emph{flat}) if it is $X$-flat at this point (resp. at every point $x\in X$), that is to say if $\sh{F}_x$ is a flat $\sh{O}_x$-module (resp. if this holds for every $x\in X$).
Recall that we have shown \sref[III]{III.1.4.15.1} the following property:
\end{env}

\begin{proposition}[2.1.2]
\label{IV.2.1.2}
Let $A$, $B$ be two rings, $\varphi:A\to B$ a ring homomorphism, $X=\Spec(B)$, $Y=\Spec(A)$, $f:X\to Y$ the morphism corresponding to $\varphi$, and $M$ a $B$-module.
For $\sh{F}=\widetilde{M}$ to be $f$-flat, it is necessary and sufficient that $M$ be a flat $A$-module.
\end{proposition}

\begin{proposition}[2.1.3]
\label{IV.2.1.3}
\oldpage[IV-2]{6}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
The following conditions are equivalent:
\begin{enumerate}[a)]
  \item For every base change $g:Y'\to Y$, if we set $X'=X\times_Y Y'$, the functor $\sh{G}'\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ from the category of quasi-coherent $\sh{O}_{Y'}$-Modules to that of quasi-coherent $\sh{O}_{X'}$-Modules, is exact.
  \item[\emph{a')}] Condition \emph{a)} is satisfied for all the canonical morphisms $g:\Spec(\sh{O}_y)\to Y$ \sref[I]{I.2.4.1}, where $y$ ranges over $Y$.
  \item $\sh{F}$ is $f$-flat.
\end{enumerate}
\end{proposition}

\begin{proof}
The questions being local on $X$ and $Y$, we can reduce to the case where $Y=\Spec(A)$, $\sh{F}=\widetilde{M}$, where $M$ is a $B$-module.
It is clear that a) implies a'); condition a') implies that for every $x\in X$, the functor $N\mapsto M_\mathfrak{n}\otimes_{B_\mathfrak{n}}N$ is exact in the category of $B_\mathfrak{n}$-modules, $\mathfrak{n}$ being the ideal $\mathfrak{j}_{f(x)}$ of $B$; this means that $M_\mathfrak{n}$ is a flat $B_\mathfrak{n}$-module, and it follows from $(\mathbf{0_I}$, 6.3.3) and from \sref{IV.2.1.2} that $\sh{F}$ is $f$-flat.
Finally, to see that b) implies a), we can also reduce to the case where $Y'=\Spec(A')$ and $\sh{G}'=\widetilde{N'}$, where $N'$ is an $A'$-module; the conclusion then follows again from \sref{IV.2.1.2} and the definition of flatness, since $(M\otimes_A N')^\sim = \sh{F}\otimes_{\sh{O}_Y}\sh{G}'$.
\end{proof}

\begin{proposition}[2.1.4]
\label{IV.2.1.4}
Let $f:X\to Y$, $g:Y'\to Y$ be two morphisms of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module; set $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$, $\sh{F}'=\sh{F}\otimes_{\sh{O}_Y}\sh{O}_{Y'}$, and let $g'$ be the canonical projection $X'\to X$.
Let $x'$ be a point of $X'$, $x=g'(x')$, $y'=f'(x')$, $y=g(y')=f(x)$.
If $\sh{F}$ is $f$-flat at the point $x$, then $\sh{F}'$ is $f'$-flat at the point $x'$; in particular if $\sh{F}$ is $f$-flat, $\sh{F}'$ is $f'$-flat; if $f$ is flat, $f'$ is flat.
\end{proposition}

\begin{proof}
It suffices to prove the first assertion; applying \sref[I]{I.3.6.5} three times, as well as \sref[I]{I.2.4.4}, we can reduce to the case where $Y=\Spec(\sh{O}_y)$, $X=\Spec(\sh{O}_x)$, $Y'=\Spec(\sh{O}_{y'})$, $\sh{F}=\widetilde{M}$, where $M=\sh{F}_x$; the hypothesis and \sref{IV.2.1.2} then imply that $\sh{F}$ is $f$-flat; we are thus reduced to proving a particular case of the second assertion, which follows immediately from \sref{IV.2.1.3}.
\end{proof}

\begin{proposition}[2.1.5]
\label{IV.2.1.5}
Consider a commutative diagram of morphisms of preschemes
\[
\xymatrix{
  X \ar[d]_f & X' \ar[l]_{g'} \ar[d]^{f'} \\
  Y & Y' \ar[l]^{g} \ar[d]^{h} \\
  & Z
}
\]
where $X'=X\times_Y Y'$ and $f'=f_{(Y')}$.
Let $x'$ be a point of $X'$, and set $x=g'(x')$, $y'=f'(x')$, $y=f(x)=g(y')$, $z=h(y')$.
Let $\sh{F}$ be a quasi-coherent $\sh{O}_X$-Module $f$-flat at the point $x$ \emph{(}resp. $f$-flat\emph{)}, and $\sh{G}$ a quasi-coherent $\sh{O}_{Y'}$-Module $h$-flat at the point $y'$ \emph{(}resp. $h$-flat\emph{)}; then $\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ is a quasi-coherent $\sh{O}_{X'}$-Module $(h\circ f')$-flat at the point $x'$ \emph{(}resp. $(h\circ f')$-flat\emph{)}.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{7}
As in \sref{IV.2.1.4}, we can reduce to the case where $Y'=\Spec(\sh{O}_{y'})$ and $Z=\Spec(\sh{O}_z)$, and it then suffices to prove that $\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ is $(h\circ f')$-flat.
Taking into account \sref{IV.2.1.2}, the proposition follows from Bourbaki, \emph{Alg.~comm.}, chap.~I, \textsection2, no.~7, prop.~8.
\end{proof}

\begin{corollary}[2.1.6]
\label{IV.2.1.6}
Let $f:X\to Y$, $g:Y\to Z$ be two morphisms of preschemes, $\sh{F}$ an $\sh{O}_X$-Module.
If $\sh{F}$ is $f$-flat at the point $x\in X$ and if $g$ is flat at the point $f(x)$, then $\sh{F}$ is $(g\circ f)$-flat at the point $x$.
In particular, if $f$ and $g$ are flat morphisms, the same is true of $g\circ f$.
\end{corollary}

\begin{proof}
This is the particular case of \sref{IV.2.1.5} with $Y'=Y$, $\sh{G}=\sh{O}_Y$.
\end{proof}

\begin{corollary}[2.1.7]
\label{IV.2.1.7}
If $f:X\to X'$, $g:Y\to Y'$ are two flat $S$-morphisms, then
\[
  f\times_S g : X\times_S Y \to X'\times_S Y'
\]
is a flat morphism.
\end{corollary}

\begin{proof}
This follows from \sref{IV.2.1.4} and \sref{IV.2.1.6} (cf.~\sref[I]{I.3.5.1}).
\end{proof}

\begin{proposition}[2.1.8]
\label{IV.2.1.8}
Let $f:X\to Y$ be a morphism of preschemes,
\[
  0 \to \sh{F}' \to \sh{F} \to \sh{F}'' \to 0
\]
an exact sequence of quasi-coherent $\sh{O}_X$-Modules, such that $\sh{F}''$ is $Y$-flat.
\begin{enumerate}[\emph{(i)}]
  \item For every $g:Y'\to Y$ and every quasi-coherent $\sh{O}_{Y'}$-Module $\sh{G}'$, the sequence
  \[
    0 \to \sh{F}'\otimes_{\sh{O}_Y}\sh{G}' \to \sh{F}\otimes_{\sh{O}_Y}\sh{G}' \to \sh{F}''\otimes_{\sh{O}_Y}\sh{G}' \to 0
  \]
  of $\sh{O}_{X'}$-Modules \emph{(}where $X'=X\times_Y Y'$\emph{)} is exact.
  \item For $\sh{F}$ to be $Y$-flat, it is necessary and sufficient that $\sh{F}'$ be so.
\end{enumerate}
\end{proposition}

\begin{proof}
We can evidently suppose $X$, $Y$, $Y'$ affine; the conclusion then follows from \sref{IV.2.1.2} and $(\mathbf{0_I}$, 6.1.2).
\end{proof}

\begin{corollary}[2.1.9]
\label{IV.2.1.9}
Let $\sh{L}^\bullet$ be a complex of quasi-coherent $\sh{O}_X$-Modules, $i$ an index such that, denoting by $d^i:\sh{L}^i\to\sh{L}^{i+1}$ the derivation operator, $\sh{B}^{i+1}(\sh{L}^\bullet)=\Im(d^i)$ and $\mathscr{Z}^{i+1}(\sh{L}^\bullet)=\Coker(d^i)$ are $Y$-flat.
Then, with the notations of \sref{IV.2.1.8}, the canonical homomorphism
\[
  \sh{H}^i(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}' \to \sh{H}^i(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')
\]
is bijective.
\end{corollary}

\begin{proof}
Since the tensor product is right exact, we have
\[
  \mathscr{Z}^{i+1}(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}' = \Coker(d^i\otimes 1) = \mathscr{Z}^{i+1}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')
\]
and $\mathscr{Z}^{i}(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}' = \mathscr{Z}^{i}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')$.
Furthermore, in the exact sequence
\[
  0 \to \sh{B}^{i+1}(\sh{L}^\bullet) \to \sh{L}^{i+1} \to \mathscr{Z}^{i+1}(\sh{L}^\bullet) \to 0
\]
$\mathscr{Z}^{i+1}(\sh{L}^\bullet)$ is $Y$-flat, so it follows from \sref{IV.2.1.8}~(i) that we have the exact sequence
\[
  0 \to \sh{B}^{i+1}(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}' \to \sh{L}^{i+1}\otimes_{\sh{O}_Y}\sh{G}' \to \mathscr{Z}^{i+1}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}') \to 0
\]
whence $\sh{B}^{i+1}(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}' = \Im(d^i\otimes 1) = \sh{B}^{i+1}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}')$.
Then, since in the exact sequence
\[
  0 \to \sh{H}^i(\sh{L}^\bullet) \to \mathscr{Z}^{i}(\sh{L}^\bullet) \to \sh{B}^{i+1}(\sh{L}^\bullet) \to 0
\]
$\sh{B}^{i+1}(\sh{L}^\bullet)$ is $Y$-flat, it follows from \sref{IV.2.1.8}~(i) and from what precedes that the sequence
\[
  0 \to \sh{H}^i(\sh{L}^\bullet)\otimes_{\sh{O}_Y}\sh{G}' \to \mathscr{Z}^{i}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}') \to \sh{B}^{i+1}(\sh{L}^\bullet\otimes_{\sh{O}_Y}\sh{G}') \to 0
\]
is exact, which proves the corollary.
\end{proof}

\begin{corollary}[2.1.10]
\label{IV.2.1.10}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module that is $Y$-flat, $\sh{L}_\bullet=(\sh{L}_i)$ a left resolution of $\sh{F}$ by quasi-coherent $\sh{O}_X$-Modules that are $Y$-flat.
Then, for every morphism $g:Y'\to Y$ and every quasi-coherent $\sh{O}_{Y'}$-Module $\sh{G}'$, the complex $\sh{L}_\bullet\otimes_{\sh{O}_Y}\sh{G}'=(\sh{L}_i\otimes_{\sh{O}_Y}\sh{G}')_{i\geq 0}$ is a left resolution of $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$.

Furthermore, if $\mathscr{Z}_i(\sh{L}_\bullet)=\Ker(\sh{L}_i\to\sh{L}_{i-1})$, the $\mathscr{Z}_i(\sh{L}_\bullet)$ are $Y$-flat, and we have $\mathscr{Z}_i(\sh{L}_\bullet)\otimes_{\sh{O}_Y}\sh{G}'=\mathscr{Z}_i(\sh{L}_\bullet\otimes_{\sh{O}_Y}\sh{G}')=\Ker(\sh{L}_i\otimes_{\sh{O}_Y}\sh{G}'\to\sh{L}_{i-1}\otimes_{\sh{O}_Y}\sh{G}')$.
\end{corollary}

\begin{proof}
Set $\sh{R}_i=\Im(\sh{L}_{i+1}\to\sh{L}_i)=\mathscr{Z}_i(\sh{L}_\bullet)$; we then have the exact sequences
\begin{align*}
  &0 \leftarrow \sh{F} \leftarrow \sh{L}_0 \leftarrow \sh{R}_0 \leftarrow 0 \\
  &0 \leftarrow \sh{R}_i \leftarrow \sh{L}_{i+1} \leftarrow \sh{R}_{i+1} \leftarrow 0
\end{align*}
and since $\sh{F}$ and the $\sh{L}_i$ are $Y$-flat, we deduce from \sref{IV.2.1.8}~(ii) by induction that all the $\sh{R}_i$ are also $Y$-flat; using \sref{IV.2.1.8}~(i), we therefore have the exact sequences
\begin{align*}
  &0 \leftarrow \sh{F}\otimes_{\sh{O}_Y}\sh{G}' \leftarrow \sh{L}_0\otimes_{\sh{O}_Y}\sh{G}' \leftarrow \sh{R}_0\otimes_{\sh{O}_Y}\sh{G}' \leftarrow 0 \\
  &0 \leftarrow \sh{R}_i\otimes_{\sh{O}_Y}\sh{G}' \leftarrow \sh{L}_{i+1}\otimes_{\sh{O}_Y}\sh{G}' \leftarrow \sh{R}_{i+1}\otimes_{\sh{O}_Y}\sh{G}' \leftarrow 0 \quad (i\geq 0)
\end{align*}
which prove the corollary.
\end{proof}

\begin{proposition}[2.1.11]
\label{IV.2.1.11}
Let $f:X\to Y$ be a flat morphism, $\sh{F}$ a quasi-coherent $\sh{O}_Y$-Module of finite presentation.
If $\sh{J}$ is the ideal of $\sh{O}_Y$ that annihilates $\sh{F}$, then $f^*(\sh{J})$ is the ideal of $\sh{O}_X$ that annihilates $f^*(\sh{F})$.
\end{proposition}

\begin{proof}
Indeed, by definition we have an exact sequence $(\mathbf{0_I}$, 5.3.7)
\[
  0 \to \sh{J} \to \sh{O}_Y \to \shHom_{\sh{O}_Y}(\sh{F},\sh{F})
\]
whence, since $f$ is flat, an exact sequence
\[
  0 \to f^*(\sh{J}) \to \sh{O}_X \to f^*(\shHom_{\sh{O}_Y}(\sh{F},\sh{F}))
\]
and since by hypothesis $\sh{F}$ is a finitely presented $\sh{O}_Y$-Module, $f^*(\shHom_{\sh{O}_Y}(\sh{F},\sh{F}))$ is canonically identified with $\shHom_{\sh{O}_X}(f^*(\sh{F}),f^*(\sh{F}))$ $(\mathbf{0_I}$, 6.7.6), whence the conclusion.
\end{proof}

\begin{proposition}[2.1.12]
\label{IV.2.1.12}
Let $X$ be a prescheme, $\sh{F}$ an $\sh{O}_X$-Module of finite presentation, $x$ a point of $X$.
The following conditions are equivalent:
\begin{enumerate}[\emph{a)}]
  \item $\sh{F}_x$ is a flat $\sh{O}_x$-module.
  \item There exists an open neighbourhood $U$ of $x$ such that $\sh{F}|U$ is a locally free $(\sh{O}_X|U)$-Module.
\end{enumerate}
\end{proposition}

\begin{proof}
Indeed, $\sh{F}_x$ is a finitely presented $\sh{O}_x$-module and $\sh{O}_x$ a local ring; it therefore amounts to the same to say that $\sh{F}_x$ is a flat $\sh{O}_x$-module or a free $\sh{O}_x$-module (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection3, no.~2, cor.~2 of prop.~5); whence the conclusion, taking into account $(\mathbf{0_I}$, 5.2.7).
We note that the proposition is valid for an arbitrary locally ringed space.
\end{proof}

\begin{proposition}[2.1.13]
\label{IV.2.1.13}
Let $f:X\to Y$ be a morphism of preschemes.
If $f$ is flat at a point $x\in X$ and if the ring $\sh{O}_x$ is reduced \emph{(}resp. integral and integrally closed\emph{)}, then the ring $\sh{O}_{f(x)}$
\oldpage[IV-2]{9}
is reduced \emph{(}resp. integral and integrally closed\emph{)}.
If $f$ is faithfully flat and if $X$ is reduced \emph{(}resp. normal\emph{)}, then $Y$ is reduced \emph{(}resp. normal\emph{)}.
\end{proposition}

\begin{proof}
Set $\sh{O}_{f(x)}=A$, $\sh{O}_x=B$.
If $B$ is a flat $A$-module, it is also a faithfully flat $A$-module $(\mathbf{0_I}$, 6.6.2), so $A$ is identified with a subring of $B$; if $B$ is reduced, the same is therefore true of $A$.
Suppose now that $B$ is integral and integrally closed, and let $L$ be its field of fractions; denote by $K\subset L$ the field of fractions of $A$.
The hypothesis implies that $B\cap K=A$ (Bourbaki, \emph{Alg.~comm.}, chap.~I, \textsection3, no.~5, prop.~10).
If then $t\in K$ is integral over $A$, it is also integral over $B$, and so belongs to $B$ by hypothesis, and therefore $t\in A$, which proves that $A$ is integrally closed.
\end{proof}

\begin{proposition}[2.1.14]
\label{IV.2.1.14}
Let $f:X\to Y$ be a faithfully flat morphism.
If $X$ is integral and if the underlying space of $Y$ is locally noetherian, then $Y$ is locally integral.
\end{proposition}

\begin{proof}
Indeed, every $y\in Y$ is of the form $f(x)$ for some $x\in X$ and by hypothesis $\sh{O}_y$ is identified with a subring of $\sh{O}_x$ $(\mathbf{0_I}$, 6.6.1); since $\sh{O}_x$ is integral, the same is true of $\sh{O}_y$, and this proves the proposition \sref[I]{I.5.1.4}.
\end{proof}

\subsection{Faithfully flat modules on preschemes}
\label{subsection:IV.2.2}

\begin{proposition}[2.2.1]
\label{IV.2.2.1}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
The following conditions are equivalent:
\begin{enumerate}[\emph{a)}]
  \item For every base change $g:Y'\to Y$, if we set $X'=X\times_Y Y'$, the functor $\sh{G}'\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$, from the category of quasi-coherent $\sh{O}_{Y'}$-Modules to that of quasi-coherent $\sh{O}_{X'}$-Modules, is exact and faithful.
  \item[\emph{a')}] Condition \emph{a)} is satisfied for all the canonical morphisms $g:\Spec(\sh{O}_y)\to Y$ \sref[I]{I.2.4.1}, where $y$ ranges over $Y$.
  \item[\emph{a'')}] Condition \emph{a)} is satisfied for all the canonical immersions $Y'\to Y$, where $Y'$ ranges over the set of affine open subschemes of $Y$.
  \item $\sh{F}$ is $Y$-flat and, for every $y\in Y$, denoting by $X_y$ the fibre $f^{-1}(y)$, the $\sh{O}_{X_y}$-Module $\sh{F}_y=\sh{F}\otimes_{\sh{O}_Y}k(y)$ is $\neq 0$.
\end{enumerate}
\end{proposition}

\begin{proof}
It is clear that a) implies a') and a''); condition a') implies first that $\sh{F}$
\oldpage[IV-2]{10}
is $Y$-flat \sref{IV.2.1.3}; on the other hand, it implies that for every $y\in Y$, the functor $N\mapsto\sh{F}\otimes_{\sh{O}_y}\widetilde{N}$ is faithful in the category of $\sh{O}_y$-modules; taking in particular $N=k(y)$, we obtain the second assertion of b).
For the second assertion of b) to imply a), on the other hand, it suffices to restrict to the case where $Y$ is affine, whence the question is local on $Y$, and one can also assume $X$ affine, and the condition then reduces to showing that the functor $\sh{G}'\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ is exact and faithful, which proves that b') implies a).
For showing that b) implies a''), we can restrict to the case where $Y$ is affine; then $f$ is faithfully flat in the category of $\sh{O}_Y$-modules, and so the functor $\sh{G}'\hookrightarrow\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ is exact and faithful, which shows that b') implies c) and concludes the proof.
\end{proof}

\begin{proposition}[2.2.2]
\label{IV.2.2.2}
Let $Y=\Spec(A)$ be an affine scheme, $f:X\to Y$ a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
Condition \emph{a)} of \sref{IV.2.2.1} is equivalent to each of the following:
\begin{enumerate}
  \item[\emph{b')}] $\sh{F}$ is $Y$-flat and, for every closed point $y$ of $Y$, $\sh{F}_y\neq 0$.
  \item[\emph{c)}] The functor $\sh{G}\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ from the category of quasi-coherent $\sh{O}_Y$-Modules to that of quasi-coherent $\sh{O}_X$-Modules, is exact and faithful.
\end{enumerate}
\end{proposition}

\begin{proof}
If b') is satisfied, there is at least one $x\in f^{-1}(y)$ such that $(\sh{F}_y)_x\neq 0$; let $U=\Spec(B)$ be an affine neighbourhood of $x$, so that $\sh{F}|U=\widetilde{M}$, where $M$ is a $B$-module.
Then b') implies that $M/\mathfrak{j}_y M\neq 0$ (since $M$ is an $A$-module that is flat \sref{IV.2.1.2}), which implies that $M\otimes_A A_y\neq 0$ for every closed point $y\in Y$, that $M\otimes_A A_y\widetilde{N}$ is exact and faithful $(\mathbf{0_I}$, 6.4.5).
The relation $\sh{F}\otimes_{\sh{O}_Y}\sh{G}=0$ implies $\sh{G}=0$, otherwise said the functor $\sh{G}\hookrightarrow\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ is faithful; since it is also exact \sref{IV.2.1.3}, this shows that b') implies c), and it is clear that c) implies a).
\end{proof}

Suppose now that c) holds and that $g:Y'\to Y$ is also an affine morphism; moreover, the functor $\sh{H}'\mapsto g'_*(\sh{H}')$ is then exact in the category of quasi-coherent $\sh{O}_{X'}$-Modules \sref[I]{I.1.6.4}, and if $g'_*(\sh{H}')=\sh{H}'$, on has $\sh{H}'=\widetilde{\sh{H}}$; the preceding functor is also faithful; to see that $\sh{G}'\hookrightarrow\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ is exact and faithful, it suffices therefore to see that the functor $\sh{G}'\mapsto g'_*(\sh{F}\otimes_{\sh{O}_Y}\sh{G}')$ is so.
Or, if $f'=f_{(Y')}:X'\to Y'$, the fact that $g$ is affine implies that we have a canonical isomorphism
\begin{equation}
\label{IV.2.2.2.1}
  \sh{F}\otimes_{\sh{O}_X}f^*(g_*(\sh{G}')) \xrightarrow{\sim} g'_*(g'^*(\sh{F})\otimes_{\sh{O}_{X'}}f'^*(\sh{G}')).
\end{equation}

Indeed, we know \sref[II]{II.1.5.2} that there is a canonical isomorphism
$f^*(g_*(\sh{G}'))\xrightarrow{\sim} g'_*(f'^*(\sh{G}'))$,
and on the other hand a canonical homomorphism $\sh{F}\to g'_*(g'^*(\sh{F}))$ $(\mathbf{0_I}$, 4.4.3.2);
composing the homomorphism $\sh{F}\otimes_{\sh{O}_X}f^*(g_*(\sh{G}'))\to g'_*(g'^*(\sh{F}))\otimes_{\sh{O}_X}g'_*(f'^*(\sh{G}'))$ with the canonical morphism $(\mathbf{0_I}$, 4.2.2.1), we deduce the homomorphism \sref{IV.2.2.2.1}; the verification that it is an isomorphism is immediate, reducing to the case where $X$ is affine.
This being the case, the functor $\sh{G}'\mapsto g_*(\sh{G}')$ is exact and faithful and by hypothesis the same is true of $\sh{G}\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}$; their composite is therefore exact and faithful \sref{IV.2.2.1} and \sref{IV.2.2.2}.

\begin{corollary}[2.2.3]
\label{IV.2.2.3}
Let $X=\Spec(B)$, $Y=\Spec(A)$ be two affine schemes, $f:X\to Y$ a morphism, $\sh{F}=\widetilde{M}$ a quasi-coherent $\sh{O}_X$-Module.
For $\sh{F}$ to satisfy the equivalent conditions of \sref{IV.2.2.1} \emph{(}or \sref{IV.2.2.2}\emph{)}, it is necessary and sufficient that the $A$-module $M$ be faithfully flat.
\end{corollary}

\begin{proof}
Indeed, condition c) of \sref{IV.2.2.2} means that the functor $N\mapsto M\otimes_A N$ from the category of $A$-modules to that of $B$-modules is exact and faithful, and the conclusion follows from $(\mathbf{0_I}$, 6.4.1).
\end{proof}

\begin{definition}[2.2.4]
\label{IV.2.2.4}
When the equivalent conditions of \sref{IV.2.2.1} are satisfied, we say that the quasi-coherent $\sh{O}_X$-Module $\sh{F}$ is \emph{faithfully flat relative to $f$} (or relative to $Y$).
\end{definition}

\oldpage[IV-2]{11}

We note that this notion is local on $Y$, but \emph{not on} $X$; in particular one can have $\sh{F}_x=0$ for certain $x\in X$, in other words $\Supp(\sh{F})$ is not necessarily equal to $X$.
However, it follows immediately from the criterion \sref{IV.2.2.1}~b) that for every $y\in Y$, there exists at least one $x\in f^{-1}(y)$ such that $(\sh{F}_y)_x=\sh{F}_x\otimes_{\sh{O}_y}k(y)\neq 0$, and \emph{a fortiori} $\sh{F}_x\neq 0$; in other words:

\begin{corollary}[2.2.5]
\label{IV.2.2.5}
If $\sh{F}$ is a quasi-coherent $\sh{O}_X$-Module, faithfully flat relative to $f$, then $f(\Supp(\sh{F}))=Y$, and \emph{a fortiori} $f$ is a surjective morphism.
\end{corollary}

This result admits a partial converse:

\begin{corollary}[2.2.6]
\label{IV.2.2.6}
Let $\sh{F}$ be a quasi-coherent $\sh{O}_X$-Module of finite type.
For $\sh{F}$ to be faithfully flat relative to $f$, it is necessary and sufficient that $\sh{F}$ be $f$-flat and that $f(\Supp(\sh{F}))=Y$.
\end{corollary}

\begin{proof}
Indeed \sref[I]{I.9.1.13} and \sref[I]{I.3.6.1}, we have $\Supp(\sh{F}_y)=f^{-1}(y)\cap\Supp(\sh{F})$, so in this case the criterion \sref{IV.2.2.1}~b) is none other than the condition of the corollary.
\end{proof}

In particular, the $\sh{O}_X$-Module $\sh{O}_X$ is faithfully flat relative to $f$ if and only if it is $f$-flat and $f$ is surjective, in other words if and only if the morphism $f$ is \emph{faithfully flat} $(\mathbf{0_I}$, 6.7.8).

Let us make explicit the properties intervening in Definition~\sref{IV.2.2.4}:

\begin{proposition}[2.2.7]
\label{IV.2.2.7}
Let $f:X\to Y$ be a morphism of preschemes, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module faithfully flat relative to $f$.
Then, for a sequence $\sh{G}'\to\sh{G}\to\sh{G}''$ of quasi-coherent $\sh{O}_Y$-Modules to be exact, it is necessary and sufficient that the corresponding sequence $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'\to\sh{F}\otimes_{\sh{O}_Y}\sh{G}\to\sh{F}\otimes_{\sh{O}_Y}\sh{G}''$ be so.
In particular, for a homomorphism $u:\sh{G}\to\sh{G}'$ of quasi-coherent $\sh{O}_Y$-Modules to be injective \emph{(}resp. surjective, bijective, zero\emph{)}, it is necessary and sufficient that $1_\sh{F}\otimes f^*(u):\sh{F}\otimes_{\sh{O}_Y}\sh{G}\to\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ be so.
For a quasi-coherent $\sh{O}_Y$-Module $\sh{G}$ to be zero, it is necessary and sufficient that $\sh{F}\otimes_{\sh{O}_Y}\sh{G}=0$.
For every quasi-coherent $\sh{O}_Y$-Module $\sh{G}$, the map $\sh{G}'\mapsto\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ from the set of quasi-coherent sub-$\sh{O}_Y$-Modules of $\sh{G}$ to the set of quasi-coherent sub-$\sh{O}_X$-Modules of $\sh{F}\otimes_{\sh{O}_Y}\sh{G}$ is injective.
\end{proposition}

\begin{proof}
To prove the last assertion, that is to say that for two quasi-coherent sub-$\sh{O}_Y$-Modules $\sh{G}'$, $\sh{G}''$ of $\sh{G}$, the relation $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'=\sh{F}\otimes_{\sh{O}_Y}\sh{G}''$ implies $\sh{G}'=\sh{G}''$, we can reduce to the case where $\sh{G}'\subset\sh{G}''$, and it then suffices to apply the second assertion of the statement to the injection $u:\sh{G}'\to\sh{G}''$.
\end{proof}

\begin{corollary}[2.2.8]
\label{IV.2.2.8}
Let $f:X\to Y$ be a faithfully flat morphism.
For every quasi-coherent $\sh{O}_Y$-Module $\sh{G}$, the canonical map
\begin{equation}
\label{IV.2.2.8.1}
  \Gamma(Y,\sh{G}) \to \Gamma(X,f^*(\sh{G}))
\end{equation}
is injective.
\end{corollary}

\begin{proof}
Indeed, $\Gamma(Y,\sh{G})$ is canonically identified with $\Hom_{\sh{O}_Y}(\sh{O}_Y,\sh{G})$ $(\mathbf{0_I}$, 5.1.1) and similarly $\Gamma(X,f^*(\sh{G}))$ with $\Hom_{\sh{O}_X}(\sh{O}_X,f^*(\sh{G}))$.
By virtue of \sref{IV.2.2.1} and \sref{IV.2.2.4}, the hypothesis that the functor $\sh{G}\mapsto f^*(\sh{G})$ is exact and faithful in the category of $\sh{O}_Y$-Modules implies that a homomorphism $u:\sh{O}_Y\to\sh{G}$ is zero if and only if $f^*(u):f^*(\sh{O}_Y)=\sh{O}_X\to f^*(\sh{G})$ is zero.
\end{proof}

\begin{remark}[2.2.9]
\label{IV.2.2.9}
The results of \sref{IV.2.2.7} and \sref{IV.2.2.8} are still true when the $\sh{O}_Y$-Modules $\sh{G}'$, $\sh{G}$, $\sh{G}''$ that appear there are arbitrary (not necessarily quasi-coherent).
Indeed, for every $y\in Y$, there exists $x\in f^{-1}(y)$ such that $\sh{F}_x$ is a faithfully flat $\sh{O}_y$-module
\oldpage[IV-2]{12}
and consequently the functor $\sh{G}_y\mapsto\sh{G}_y\otimes_{\sh{O}_y}\sh{F}_x$ is faithful; since moreover for every $x\in f^{-1}(y)$ the functor $\sh{G}_y\mapsto\sh{G}_y\otimes_{\sh{O}_y}\sh{F}_x$ is exact, we immediately deduce the conclusion.
\end{remark}

\begin{proposition}[2.2.10]
\label{IV.2.2.10}
Let $f:X\to Y$, $g:Y'\to Y$, $h:Y'\to Z$ be three morphisms of preschemes, $X'=X\times_Y Y'$, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module faithfully flat relative to $Y$, $\sh{G}$ a quasi-coherent $\sh{O}_{Y'}$-Module faithfully flat relative to $Z$.
For $\sh{G}$ to be plat \emph{(}resp. faithfully flat\emph{)} relative to $Z$, it is necessary and sufficient that $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ be a flat $\sh{O}_{X'}$-Module \emph{(}resp. faithfully flat\emph{)} relative to $Z$.
\end{proposition}

\begin{proof}
We already know that if $\sh{G}$ is $Z$-flat, the same is true of $\sh{F}\otimes_{\sh{O}_Y}\sh{G}'$ \sref{IV.2.1.5}.
Consider an arbitrary base change $Z''\to Z$ and set $X''=X'\times_Z Z''$; if $\sh{G}$ is faithfully flat relative to $Z$, the functor
\begin{equation}
\label{IV.2.2.10.1}
  \sh{H}''\mapsto\sh{H}''\otimes_{\sh{O}_Z}\sh{G}'\to(\sh{H}''\otimes_{\sh{O}_Z}\sh{G}')\otimes_{\sh{O}_Y}\sh{F}=\sh{H}''\otimes_{\sh{O}_Z}(\sh{G}'\otimes_{\sh{O}_Y}\sh{F})
\end{equation}
from the category of $\sh{O}_{X''}$-Modules quasi-coherent to that of $\sh{O}_{X''}$-Modules quasi-coherent is composed of two exact and faithful functors, so is exact and faithful.
Conversely, if this composed functor is exact (resp. exact and faithful), the same is true of $\sh{H}''\mapsto\sh{H}''\otimes_{\sh{O}_Z}\sh{G}'$, since the functor $\sh{H}'\mapsto\sh{H}'\otimes_{\sh{O}_Y}\sh{F}$ (from the category of quasi-coherent $\sh{O}_{Y'}$-Modules to that of quasi-coherent $\sh{O}_{X'}$-Modules) is exact and faithful by hypothesis.
\end{proof}

\begin{corollary}[2.2.11]
\label{IV.2.2.11}
\begin{enumerate}[\emph{(i)}]
  \item Let $f:X\to Y$, $g:Y'\to Y$ be two morphisms, $X'=X\times_Y Y'$, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
  If $\sh{F}$ is faithfully flat relative to $Y$, $\sh{F}'=\sh{F}\otimes_{\sh{O}_Y}\sh{O}_{Y'}$ is faithfully flat relative to $Y'$.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module faithfully flat relative to $Y$.
  For $g$ to be a faithfully flat morphism, it is necessary and sufficient that $\sh{F}$ be faithfully flat relative to $Z$.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms, $\sh{G}$ a quasi-coherent $\sh{O}_Y$-Module.
  Suppose the morphism $f$ is faithfully flat.
  For $\sh{G}$ to be flat \emph{(}resp. faithfully flat\emph{)} relative to $Z$, it is necessary and sufficient that $f^*(\sh{G})$ be flat \emph{(}resp. faithfully flat\emph{)} relative to $Z$.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms, $x$ a point of $X$.
  Suppose $f$ is flat at the point $x$.
  For $g\circ f$ to be flat at the point $x$, it is necessary and sufficient that $g$ be flat at the point $f(x)$.
\end{enumerate}
\end{corollary}

\begin{proof}
To show (i), apply \sref{IV.2.2.10} replacing $Z$ by $Y'$ and $\sh{G}'$ by $\sh{O}_{Y'}$.
To show (ii), apply \sref{IV.2.2.10} taking $Y'\to Y$ the identity and replacing $\sh{G}'$ by $\sh{O}_Y$.
To show (iii), apply \sref{IV.2.2.10} by replacing $Y'\to Y$ by the identity, $\sh{F}$ by $\sh{O}_X$ and $\sh{G}'$ by $\sh{G}$.
Finally, (iv) follows from (ii) by replacing $\sh{F}$ by $\sh{O}_X$ and $\sh{G}$ by $\sh{O}_Y$, replacing $X$ by $\Spec(\sh{O}_x)$, $\sh{F}$ by $\sh{O}_X$, $Y$ by $\Spec(\sh{O}_{f(x)})$ and $Z$ by $\Spec(\sh{O}_{g(f(x))})$, and taking into account $(\mathbf{0_I}$, 6.6.2).
\end{proof}

\begin{corollary}[2.2.12]
\label{IV.2.2.12}
Let $Y$ be an affine scheme, $f:X\to Y$ a quasi-compact morphism, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module.
If $\sh{F}$ is faithfully flat relative to $f$, there exists an affine scheme $X'$ and a surjective local isomorphism $g:X'\to X$ such that $g^*(\sh{F})$ is faithfully flat relative to $f\circ g$.
\end{corollary}

\begin{proof}
Since $X$ is quasi-compact, and therefore a finite union of affine opens $X_i$; it suffices to take for $X'$ the affine scheme \emph{sum} of the preschemes $X_i$, $f$ being the canonical morphism \sref[I]{I.3.1},
\oldpage[IV-2]{13}
the hypothesis implies that $g^*(\sh{F})$ is faithfully flat relative to $f\circ g$, by virtue of \sref{IV.2.2.11}~(iii).
\end{proof}

\begin{corollary}[2.2.13]
\label{IV.2.2.13}
\begin{enumerate}[\emph{(i)}]
  \item Let $f:X\to Y$, $g:Y'\to Y$ be two morphisms, $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$.
  If $f$ is a faithfully flat morphism, the same is true of $f'$.
  \item If $f:X\to X'$, $g:Y\to Y'$ are two faithfully flat $S$-morphisms,
  \[
    f\times_S g : X\times_S Y \to X'\times_S Y'
  \]
  is faithfully flat.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms such that $f$ is faithfully flat \emph{(}resp. faithfully flat\emph{)}.
  For $g$ to be a flat \emph{(}resp. faithfully flat\emph{)} morphism, it is necessary and sufficient that $g\circ f$ be a flat \emph{(}resp. faithfully flat\emph{)} morphism.
\end{enumerate}
\end{corollary}

\begin{proposition}[2.2.14]
\label{IV.2.2.14}
Let $f:X\to Y$ be a faithfully flat and quasi-compact morphism.
If $X$ is locally noetherian, the same is true of $Y$.
\end{proposition}

\begin{proof}
The question being local on $Y$, we can suppose $Y=\Spec(A)$ affine; since $f$ is quasi-compact, we can also restrict to the case where $X=\Spec(B)$ is affine.
Then $B$ is a noetherian ring by hypothesis, and a faithfully flat $A$-module \sref{IV.2.2.3}; therefore $A$ is noetherian $(\mathbf{0_I}$, 6.5.2).
\end{proof}

\begin{proposition}[2.2.15]
\label{IV.2.2.15}
Let $f:X\to Y$ be a faithfully flat morphism.
If $\mathfrak{S}(X)$ and $\mathfrak{S}(Y)$ denote respectively the sets of sub-preschemes of $X$ and of $Y$, the map $Z\mapsto f^{-1}(Z)$ from $\mathfrak{S}(Y)$ to $\mathfrak{S}(X)$ is injective.
\end{proposition}

\begin{proof}
Since $f$ is surjective, we have $f(f^{-1}(Z))=Z$.
On the other hand, if $U$ is an open of $Y$ containing $Z$ and in which $Z$ is closed, $f^{-1}(Z)$ is closed in $f^{-1}(U)$, and the restriction $f^{-1}(U)\to U$ of $f$ is faithfully flat.
We can therefore restrict to considering only closed sub-preschemes of $Y$.
Now, if $Z$ is a sub-prescheme of $Y$ corresponding to a quasi-coherent ideal $\sh{J}$ of $\sh{O}_Y$, we know that $f^{-1}(Z)$ corresponds to the quasi-coherent ideal $f^*(\sh{J})\sh{O}_X$ of $\sh{O}_X$ \sref[I]{I.4.4.5}, and since $f$ is flat, $f^*(\sh{J})\sh{O}_X$ is identified with $f^*(\sh{J})$.
But the map $\sh{J}\hookrightarrow f^*(\sh{J})$ from the set of quasi-coherent ideals of $\sh{O}_Y$ to the set of quasi-coherent ideals of $\sh{O}_X$ is injective \sref{IV.2.2.7}, whence the conclusion.
\end{proof}

\begin{corollary}[2.2.16]
\label{IV.2.2.16}
Let $X$, $Y$ be two $S$-preschemes; if $S'\to S$ is a faithfully flat morphism, the map $f\mapsto f_{(S')}$ from $\Hom_S(X,Y)$ to $\Hom_{S'}(X_{(S')},Y_{(S')})$ is injective.
\end{corollary}

\begin{proof}
Since $X_{(S')}\times_{S'}Y_{(S')}=(X\times_S Y)_{(S')}$ \sref[I]{I.3.3.10}, the morphism projection $X_{(S')}\times_{S'}Y_{(S')}\to X\times_S Y$ is faithfully flat \sref{IV.2.2.13}.
The elements of $\Hom_S(X,Y)$ correspond bijectively to the sub-preschemes of $X\times_S Y$ which are the \emph{graphs} of these $S$-morphisms \sref[I]{I.5.3.11}, and if $Z_f$ is the graph of $f\in\Hom_S(X,Y)$, then $Z_{f(S')}=g^{-1}(Z_f)$ \sref[I]{I.5.3.12}.
It therefore suffices to apply $g$ to Proposition~\sref{IV.2.2.15}.
\end{proof}

\begin{proposition}[2.2.17]
\label{IV.2.2.17}
Let $A$ be a ring, $B$ an $A$-algebra such that $B$ is a faithfully flat $A$-module of finite presentation.
Then the structural homomorphism $\varphi:A\to B$ is an isomorphism of the $A$-module $A$ onto a direct summand of the $A$-module $B$.
If $A$ is a local ring, $B$ is a free $A$-module of finite type, and there exists a basis of this module containing the unit element.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{14}
By virtue of Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection3, no.~3, prop.~12, it suffices to prove the proposition when $A$ is a local ring; one then has (\emph{loc.~cit.}, no.~2, cor.~2 of prop.~5) that $B$ is a free $A$-module of finite type, and the conclusion follows from \emph{loc.~cit.}, prop.~5.
\end{proof}

\subsection{Topological properties of flat morphisms}
\label{subsection:IV.2.3}

\begin{lemma}[2.3.1]
\label{IV.2.3.1}
Let $f:X\to Y$ be a quasi-compact and quasi-separated morphism, $g:Y'\to Y$ a flat morphism; set $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$.
Then, for every quasi-coherent $\sh{O}_X$-Module $\sh{F}$, the canonical homomorphism
\begin{equation}
\label{IV.2.3.1.1}
  g^*(f_*(\sh{F})) \to f'_*(\sh{F}\otimes_{\sh{O}_Y}\sh{O}_{Y'})
\end{equation}
is bijective.
\end{lemma}

\begin{proof}
This is a particular case of \sref[III]{III.1.4.15} (improved in \sref{IV.1.7.21}).
\end{proof}

\begin{proposition}[2.3.2]
\label{IV.2.3.2}
Let $S$ be a prescheme, $f:X\to Y$ an $S$-morphism quasi-compact and quasi-separated, $j:Z\to Y$ the closed image of $X$ by $f$ \sref[I]{I.9.5.3} and \sref{IV.1.7.8}, and let $g:X\to Z$ be a morphism such that $f=j\circ g$, where $g:X\to Z$ is a morphism \emph{(}loc.~cit.\emph{)}.
Let $h:S'\to S$ be a flat morphism, and set $f'=f_{(S')}:X_{(S')}\to Y_{(S')}$; then $j'=j_{(S')}:Z_{(S')}\to Y_{(S')}$, the closed image of $X_{(S')}$ by $f'$.
\end{proposition}

\begin{proof}
As the morphism $Y_{(S')}\to Y$ is flat \sref{IV.2.1.4}, we can restrict to the case where $S=Y$ \sref[I]{I.3.3.11}; we can then set $f'=(\psi,\theta)$, on sees easily (for example by reducing to the case where $Y$ and $Y'$ are affine and using \sref[I]{I.2.2.4}) that $\theta':{\sh{O}_{S'}}\to f'_*({\sh{O}_{X'}})$ is composed of the canonical homomorphism \sref{IV.2.3.1}~$h^*(f_*(\sh{O}_X))\to f'_*(\sh{O}_{X'})$ and of $h^*(\theta):h^*(\sh{O}_Y)\to h^*(f_*(\sh{O}_X))$; the conclusion thus follows from \sref{IV.2.3.1} and from the fact that $\Ker(\theta)$, being an $\sh{O}_Y$-Module quasi-coherent, commutes with flat base change $(\mathbf{0_I}$, 6.7.8).
\end{proof}

\begin{env}[2.3.3]
\label{IV.2.3.3}
We will say that a morphism $f:X\to Y$ is \emph{quasi-flat} if there exists a quasi-coherent $\sh{O}_X$-Module $\sh{F}$ of finite type which is $f$-flat and whose support is equal to $X$.
We will say that $f$ is \emph{quasi-faithfully flat} if it is quasi-flat and surjective.
Every flat morphism plat (resp. faithfully flat) morphism is quasi-flat (resp. quasi-faithfully flat), since then $\sh{F}=\sh{O}_X$ satisfies the preceding conditions.

It follows immediately from \sref{IV.2.1.4} and from \sref[I]{I.9.1.13}, that if $f$ is quasi-flat, then for every morphism $Y'\to Y$, the morphism $f_{(Y')}:X\times_Y Y'\to Y'$ is quasi-flat.
Similarly (taking into account \sref[I]{I.3.5.2}), if $f$ is quasi-faithfully flat, the same is true of $f_{(Y')}$.
\end{env}

\begin{proposition}[2.3.4]
\label{IV.2.3.4}
Let $f:X\to Y$ be a quasi-flat morphism \sref{IV.2.3.3}.
Then $f$ has the following properties \emph{(}which are equivalent by virtue of \sref{IV.1.10.4}\emph{)}:
\begin{enumerate}[\emph{(i)}]
  \item For every $x\in X$ and every generisation $y'$ of $y=f(x)$, there exists a generisation $x'$ of $x$ such that $f(x')=y'$.
  \item For every $x\in X$, the image by $f$ of $\Spec(\sh{O}_{X,x})$ is $\Spec(\sh{O}_{Y,y})$.
\oldpage[IV-2]{15}
  \item For every closed irreducible part $Y'$ of $Y$, every irreducible component of $f^{-1}(Y')$ dominates $Y'$.
\end{enumerate}
\end{proposition}

\begin{proof}
It suffices, for example, to prove (ii).
By hypothesis, there is a quasi-coherent $\sh{O}_X$-Module of finite type $\sh{F}$, which is $f$-flat and such that $\Supp(\sh{F})=X$.
For every $x\in X$, $\sh{F}_x$ is then a finitely generated $\sh{O}_x$-module, non-reduced to $0$, and moreover $\sh{F}_x$ is a flat $\sh{O}_y$-module \emph{flat}, and consequently the homomorphism $\rho:\sh{O}_y\to\sh{O}_x$.
Since the latter is local and $\sh{F}_x\neq 0$, the Nakayama lemma shows that $\sh{F}_x\otimes_{\sh{O}_y}k(y)\neq 0$, so $\sh{F}_x$ is a faithfully flat $\sh{O}_y$-module $(\mathbf{0_I}$, 6.4.1).
It follows that for every prime ideal $\mathfrak{q}$ of $\sh{O}_y$, there exists a prime ideal $\mathfrak{p}$ of $\sh{O}_x$ such that $\mathfrak{q}=\rho^{-1}(\mathfrak{p})$ $(\mathbf{0_I}$, 6.5.1), which proves (ii).
\end{proof}

\begin{corollary}[2.3.5]
\label{IV.2.3.5}
Let $f:X\to Y$ be a morphism satisfying the equivalent conditions (i), (ii), (iii) of \sref{IV.2.3.4} \emph{(}in particular a quasi-flat or open morphism \sref{IV.1.10.4}\emph{)}.
\begin{enumerate}[\emph{(i)}]
  \item Let $Z$, $Z'$ be two closed irreducible parts of $Y$ such that $Z\subset Z'$, and let $T$ be an irreducible component of $f^{-1}(Z)$; then there exists an irreducible component $T'$ of $f^{-1}(Z')$ containing $T$ \emph{(}and dominating $Z'$\emph{)}.
  \item For every irreducible component $Y'$ of $Y$, $\overline{f(T)}$ is an irreducible component of $Y$.
  \item Suppose $Y$ irreducible, and, if $y$ is its generic point, suppose that $f^{-1}(y)$ is irreducible.
  Then $X$ is irreducible.
\end{enumerate}
\end{corollary}

\begin{proof}
(i) It suffices to apply \sref{IV.2.3.4}~(i) by taking for $x$ the generic point of $T$  (the generic point of $Z$) and for $y'$ the generic point of $Z'$.

(ii) It is clear that $\overline{f(T)}=Z$ is irreducible, and by virtue of (i), $Z$ cannot be strictly contained in a closed irreducible component $Z'$ of $Y$.

(iii) By virtue of (ii), every irreducible component of $X$ dominates $Y$, so every such component meets $f^{-1}(y)$; the conclusion then follows from $(\mathbf{0_I}$, 2.1.8).
\end{proof}

\begin{proposition}[2.3.6]
\label{IV.2.3.6}
Let $Y$ be a prescheme whose set of irreducible components is locally finite \emph{(}which is the case if the underlying space of $Y$ is locally noetherian \emph{(}cf.~\sref[I]{I.6.1.9}\emph{))}.
\begin{enumerate}[\emph{(i)}]
  \item Every stable closed part $W$ of $Y$ by generisation $(\mathbf{0_I}$, 2.1.2) is open.
  In particular, every connected component of $Y$ is open.
  \item Let $f:X\to Y$ be a closed morphism satisfying moreover the equivalent conditions \emph{(i)}, \emph{(ii)}, \emph{(iii)} of \sref{IV.2.3.4} \emph{(}which will be the case if $f$ is quasi-flat or open\emph{)}.
  Then the image by $f$ of every connected component $C$ of $X$ is a connected component of $Y$.
  \item Let $f:X\to Y$ be a morphism satisfying the equivalent conditions \emph{(i)}, \emph{(ii)}, \emph{(iii)} of \sref{IV.2.3.4} and such that moreover for every $y\in Y$, the set $f^{-1}(y)$ is finite \emph{(}which will be the case if $f$ is quasi-finite or radiciel\emph{)}.
  Then the set of irreducible components of $X$ is locally finite.
\end{enumerate}
\end{proposition}

\begin{proof}
\oldpage[IV-2]{16}
(i) If $y\in W$, the generic points $\eta_i$ ($1\leq i\leq m$) of the irreducible components $Y_i$ of $Y$ containing $y$ belong by hypothesis to $W$, so, since $W$ is closed, these components are themselves contained in $W$; since there is by hypothesis a neighbourhood $U$ of $y$ such that $U$ is a union of the $U\cap Y_i$ $(\mathbf{0_I}$, 2.1.6), then $U\subset W$, so $W$ is open.
Since for every $y\in Y$, $\overline{\{y\}}$ is connected, a connected component of $Y$ is stable by generisation, so the second assertion follows immediately from the first.

(ii) Since $C$ is closed in $X$, $f(C)$ is closed in $Y$ by hypothesis.
Moreover, as $C$ is stable by generisation, the hypothesis on $f$ implies that $f(C)$ is stable by generisation; therefore, by virtue of (i), $f(C)$ is open and closed, and since it is connected, it is a connected component of $Y$.

(iii) Let $x\in X$; by hypothesis, there is a neighbourhood $U$ of $y=f(x)$ which intersects only a finite number of irreducible components $Y_i$ of $Y$ ($1\leq i\leq n$); let $y_i$ be the generic point of $Y_i$ \sref{IV.2.3.4}.
Since each of these sets is finite by hypothesis, this proves our assertion.
\end{proof}

\begin{proposition}[2.3.7]
\label{IV.2.3.7}
Let $f:X\to Y$ be a morphism, $g:Y'\to Y$ a quasi-flat morphism, $f'=f_{(Y')}:X'\to Y'$.
\begin{enumerate}[\emph{(i)}]
  \item If $f$ is quasi-compact and dominant, the same is true of $f'$.
  \item If every irreducible component of $X$ dominates an irreducible component of $Y$, then every irreducible component of $X'$ dominates an irreducible component of $Y'$.
\end{enumerate}
\end{proposition}

\begin{proof}
Denote by $g':X'\to X$ the canonical projection, which is a quasi-flat morphism \sref{IV.2.3.3}.

(i) We already know \sref{IV.1.1.2} that $f'$ is quasi-compact; moreover, if $y'$ is a maximal point \sref{IV.1.1.4} of $Y'$, $y=g(y')$ is a maximal point of $Y$, as it follows from \sref{IV.2.3.4}~(iii).
By hypothesis \sref{IV.1.1.5}, there exists $x\in X$ such that $f(x)=y$; therefore \sref[I]{I.3.4.7}, there exists $x'\in X'$ such that $f'(x')=y'$.

(ii) Let $x'$ be a maximal point of $X'$, and let $x=g'(x')$; it follows from \sref{IV.2.3.4}~(ii) that $x$ is a maximal point of $X$, and by hypothesis $y=f(x)$ is a maximal point of $Y$.
Set $y'=f'(x')$, so that $g(y')=y$; it suffices to show that $y'$ is a maximal point of $Y'$.
By virtue of \sref[I]{I.5.1.7} and \sref{IV.2.3.3}, we can restrict to the case where $X$ and $Y$ are \emph{reduced}, so $\sh{O}_x$ and $\sh{O}_y$ are fields, and by virtue of \sref[I]{I.2.4.4} applied twice and of \sref[I]{I.3.6.5}, we can restrict to the case where $Y=\Spec(\sh{O}_y)$ and $Y'=\Spec(\sh{O}_y)$.
Then $f$ is a flat morphism since $\sh{O}_y$ is a field \sref{IV.2.1.2}, and it follows likewise from \sref{IV.2.1.4} that $y'=f'(x')$ is a maximal point of $Y'$; therefore it follows from \sref{IV.2.3.4}~(ii) that $y'=f'(x')$ is a maximal point.
\end{proof}

\begin{corollary}[2.3.8]
\label{IV.2.3.8}
\textup{(Zariski).} ---
Let $A$, $B$ be two noetherian local rings, $\mathfrak{m}$, $\mathfrak{n}$ their maximal ideals respectively, $\varphi:A\to B$ a local homomorphism.
Suppose the following hypotheses are satisfied:
\begin{enumerate}
  \item[$1^\circ$] $B$ is an $A$-algebra essentially of finite type \sref{IV.1.3.8}.
  \item[$2^\circ$] The completion $\widehat{A}$ of $A$ for the $\mathfrak{m}$-adic topology is integral.
  \item[$3^\circ$] $\varphi$ is injective.
\end{enumerate}
Then the $\mathfrak{m}$-adic topology on $A$ is induced by the $\mathfrak{n}$-adic topology of $B$.
\end{corollary}

\begin{proof}
\oldpage[IV-2]{17}
Set $B'=B\otimes_A\widehat{A}$, where $C$ is an $A$-algebra of finite type and $S$ is a multiplicative part of $C$, so $B'$ is an $A$-algebra of finite type.
Since $A$ is identified with a subring of $\widehat{A}$ $(\mathbf{0_I}$, 7.3.5), $A$ is integral by $2^\circ$.
Hypothesis $3^\circ$ implies that there exists a prime ideal $\mathfrak{a}$ of $B$ inducing the ideal $\mathfrak{o}$ of $A$ $(\mathbf{0_I}$, 1.5.8), and therefore the local homomorphism $A\to B/\mathfrak{a}$ is injective.
We can therefore restrict to proving \sref{IV.2.3.8} by adding the hypothesis that $B$ is a local \emph{integral} ring.
Let us apply \sref{IV.2.3.7}~(ii) to $Y=\Spec(A)$, $X=\Spec(B)$, $Y'=\Spec(\widehat{A})$ and $X'=\Spec(B')$; since the morphism $Y'\to Y$ is flat and $X$ dominates $Y$, every irreducible component of $X'$ dominates $Y'$; since by hypothesis $Y'$ is integral, $X'$ is also integral.
Identifying $A$ and $\widehat{A}$ as subrings of $\sh{O}_{x'}$, and denoting by $\mathfrak{r}$ the maximal ideal of $\sh{O}_{x'}$, the intersection of the ideals $\mathfrak{r}^k\cap\widehat{A}$ are open in $\widehat{A}$ $(\mathbf{0_I}$, 7.3.5).
Moreover, the $\mathfrak{n}$-preadic topology of $B$ induces on $A$ a topology that is \emph{finer} than the $\mathfrak{m}$-preadic topology of $A$ $(\mathbf{0_I}$, 7.3.5).
On the other hand, $\mathfrak{n}^k\cap\mathfrak{r}^k\cap A$, so the $\mathfrak{n}$-preadic topology of $B$ induces on $A$ the $\mathfrak{m}$-preadic topology; but $\mathfrak{m}^k\subset\mathfrak{n}^k\cap A$, so these two topologies are \emph{identical}.
\end{proof}

\begin{remark}[2.3.9]
\label{IV.2.3.9}
We will see further on \sref[IV]{IV.7.8.3}~(vii), that for the most common noetherian local rings $A$ in algebraic geometry, the hypothesis that $A$ is integral and integrally closed implies that $\widehat{A}$ is integral.
This is why, in algebraic geometry over a base field, one generally states \sref{IV.2.3.8} under the hypothesis that $A$ is \emph{integral and integrally closed}.
\end{remark}

\begin{theorem}[2.3.10]
\label{IV.2.3.10}
Let $f:X\to Y$ be a quasi-flat morphism \sref{IV.2.3.3}.
Then, for every pro-constructible part \sref{IV.1.9.4} $Z$ of $Y$, we have $f^{-1}(\overline{Z})=\overline{f^{-1}(Z)}$.
\end{theorem}

\begin{proof}
Since $f$ is continuous, we have $f^{-1}(\overline{Z})\supset\overline{f^{-1}(Z)}$ and it suffices to prove that for every $x\in X$ such that $f(x)\in\overline{Z}$, $x$ is adherent to $f^{-1}(Z)$; it is clear that the question is local on $Y$, so we can suppose $Y$ affine.
By virtue of the hypothesis, there exists an affine scheme $Y'$ and a morphism $g:Y'\to Y$ such that $g(Y')=Z$ \sref{IV.1.9.5}~(ix).
Let $Y_1$ be the image by $f$ of $Y'$ \sref[I]{I.9.5.3}, and let $X_1$ be the closed sub-prescheme $f^{-1}(Y_1)$ of $X$; if $f_1:X_1\to Y_1$ is the restriction of $f$ to $X_1$, we know that $f_1$ is quasi-flat \sref{IV.2.3.3}; we can therefore replace $X$, $Y$ by $X_1$, $Y_1$ respectively, in other words suppose that $g$ is \emph{dominant}.
Set then $X'=X\times_Y Y'$, and let $f'$ and $g'$ be the projections of $X'$ to $Y'$ and $X$ respectively, so that we have the commutative diagram
\[
\xymatrix{
  X & X' \ar[l]_{g'} \\
  Y \ar[u]^f & Y' \ar[l]^g \ar[u]_{f'}
}
\]
Since $f$ is quasi-flat, $g$ quasi-compact and dominant, it follows from \sref{IV.2.3.7} (where the roles of $f$ and $g$ are interchanged) that $g'$ is a dominant morphism, which proves the theorem.
\end{proof}

\begin{corollary}[2.3.11]
\label{IV.2.3.11}
Let $f$ be a quasi-flat and quasi-compact morphism, $F$ a closed part of $X$ such that $F=f^{-1}(f(F))$; then $F=f^{-1}(\overline{f(F)})$.
\end{corollary}

\begin{proof}
\oldpage[IV-2]{18}
Let $Y'$ be the reduced sub-prescheme of $X$ having $F$ as underlying space \sref[I]{I.5.2.1} and let $j:Y'\to X$ be the canonical injection; then $f\circ j$ is quasi-compact \sref{IV.1.1.2},
so $Z=f(F)$ is pro-constructible in $Y$ \sref{IV.1.9.5}~(vii), and the corollary follows from the fact that $F$ is closed.
\end{proof}

We can also write the result of \sref{IV.2.3.11} in the form $F=f^{-1}(f(X)\cap\overline{f(F)})$, in other words, the closed subsets of the subspace $f(X)$ of $Y$ are the parts $Z\subset f(X)$ such that $f^{-1}(Z)$ is closed in $X$; this also means that \emph{the topology induced by $Y$ on $f(X)$ is the quotient of that of $X$ by the equivalence relation defined by $f$}.
In particular:

\begin{corollary}[2.3.12]
\label{IV.2.3.12}
Let $f:X\to Y$ be a quasi-faithfully flat morphism \sref{IV.2.3.3} and quasi-compact.
Then the topology of $Y$ is the quotient of that of $X$ by the equivalence relation defined by $f$ \emph{(}in other words, for $Z\subset Y$ to be open \emph{(}resp. closed\emph{)} in $Y$, it is necessary and sufficient that $f^{-1}(Z)$ be open \emph{(}resp. closed\emph{)} in $X$\emph{)}.
\end{corollary}

\begin{proof}
Indeed, we then have $f(X)=Y$.
\end{proof}

\begin{corollary}[2.3.13]
\label{IV.2.3.13}
Let $X$, $Y$ be two $S$-preschemes, $f:X\to Y$ an $S$-morphism faithfully flat and quasi-compact.
For $Y$ to be separated over $S$, it is necessary and sufficient that the canonical immersion $j:X\times_Y X\to X\times_S X$ \sref[I]{I.5.3.10} be a closed immersion.
\end{corollary}

\begin{proof}
Note for this that we have the commutative diagram \sref[I]{I.5.3.5}
\[
\xymatrix{
  X\times_Y X \ar[r]^j \ar[d]_\pi & X\times_S X \ar[d]^{f\times_S f} \\
  Y \ar[r]^-{\Delta_Y} & Y\times_S Y
}
\]
identifying $X\times_Y X$ with the product of the $(Y\times_S Y)$-preschemes $Y$ and $X\times_S X$.
Since $f$ is surjective, the same is true of $\pi$ and of $f\times_S f$, so the diagonal $\Delta_Y(Y)$ has as inverse image by $f\times_S f$ the image $j(X\times_Y X)$ \sref[I]{I.3.4.8}.
Since $f\times_S f$ is faithfully flat and quasi-compact \sref{IV.1.1.2} and \sref{IV.2.2.13}, it suffices to apply \sref{IV.2.3.12} to this morphism.
\end{proof}

\begin{corollary}[2.3.14]
\label{IV.2.3.14}
Let $f:X\to Y$ be a quasi-faithfully flat and quasi-compact morphism, and let $Z$ be a part of $Y$.
For $Z$ to be a locally closed pro-constructible part of $Y$, it is necessary and sufficient that $f^{-1}(Z)$ be a locally closed pro-constructible part of $X$.
\end{corollary}

\begin{proof}
We already know \sref{IV.1.9.12} that, for $Z$ to be a pro-constructible part of $Y$, it is necessary and sufficient that $f^{-1}(Z)$ be a pro-constructible part of $X$.
The condition is evidently necessary.
To prove that it is sufficient, consider the closed reduced sub-prescheme $Y_1$ of $Y$ having $\overline{Z}$ as underlying space and, let $X_1$ be its inverse image $f^{-1}(Y_1)$, which has $\overline{f^{-1}(Z)}$ as underlying space by virtue of \sref{IV.2.3.10}.
Since $f^{-1}(Z)$ is locally closed in $X$, it is open in $\overline{f^{-1}(Z)}$, and therefore in $X_1$; or, the morphism $f_1:X_1\to Y_1$ induced by $f$ by restriction is quasi-faithfully flat and quasi-compact \sref{IV.1.1.2} and \sref{IV.2.3.3}; we deduce from \sref{IV.2.3.12} that $Z$ is open in $\overline{Z}$, which shows that $Z$ is locally closed in $Y$.
\end{proof}

\begin{remark}[2.3.15]
\label{IV.2.3.15}
It does not suffice, in \sref{IV.2.3.12}, to suppose only that $f$ is \emph{faithfully flat}.
For example, take for $Y$ a reduced irreducible algebraic curve
\oldpage[IV-2]{19}
\sref[II]{II.7.4.2}, for $X$ the prescheme \emph{sum} of the schemes $\Spec(\sh{O}_y)$, where $y$ ranges over $Y$ \sref[I]{I.3.1}, and for $f$ the canonical morphism, which is faithfully flat \sref[I]{I.2.4.2}; if $\eta$ denotes the generic point of $Y$, $Z=\{\eta\}$ is not open in $Y$ \sref[II]{II.7.4.3}, but $f^{-1}(Z)=\Spec(\sh{O}_\eta)$ since $\sh{O}_\eta$ is a field, and $f^{-1}(Z)$ is open in $X$.
\end{remark}

\subsection{Universally open morphisms and flat morphisms}
\label{subsection:IV.2.4}

\begin{env}[2.4.1]
\label{IV.2.4.1}
We have already defined \sref[II]{II.5.4.9} the notion of a \emph{universally closed} morphism; in the same manner, we introduce the following definitions:
\end{env}

\begin{definition}[2.4.2]
\label{IV.2.4.2}
We say that a morphism of preschemes $f:X\to Y$ is \emph{universally open} (resp. \emph{universally bicontinuous}, resp. a \emph{universal homeomorphism}) if, for every morphism $g:Y'\to Y$, the morphism $f_{(Y')}:X\times_Y Y'\to Y'$ is open (resp. a homeomorphism onto its image, resp. a homeomorphism onto $Y'$).
\end{definition}

We will see further on \sref[IV]{IV.14.3.2} that when $Y$ is locally noetherian, the definition of universally open morphism given here is equivalent to Definition~\sref[III]{III.4.3.9} for morphisms of finite type; the reader will be able to verify that we do not use this last definition before \textsection14.

\begin{proposition}[2.4.3]
\label{IV.2.4.3}
\begin{enumerate}[\emph{(i)}]
  \item An immersion \emph{(}resp. an open immersion, resp. a closed immersion\emph{)} is universally bicontinuous \emph{(}resp. universally open, resp. universally closed\emph{)}.
  \item The composite of two universally open \emph{(}resp. universally closed, resp. universally bicontinuous, resp. universal homeomorphisms\emph{)} is so as well.
  \item If $f:X\to Y$ is an $S$-morphism universally open \emph{(}resp. universally closed, resp. universally bicontinuous, resp. a universal homeomorphism\emph{)}, the same is true of $f_{(S')}:X_{(S')}\to Y_{(S')}$ for every base change $S'\to S$.
  \item If $f:X\to X'$, $g:Y\to Y'$ are two $S$-morphisms universally open \emph{(}resp. universally closed, resp. universally bicontinuous, resp. universal homeomorphisms\emph{)}, the same is true of $f\times_S g:X\times_S Y\to X'\times_S Y'$.
  \item Let $f:X\to Y$, $g:Y\to Z$ be two morphisms such that $f$ is surjective; if $g\circ f$ is universally open \emph{(}resp. universally closed, resp. universally bicontinuous, resp. a universal homeomorphism\emph{)}, the same is true of $g$.
  \item For a morphism $f$ to be universally open \emph{(}resp. universally closed, resp. universally bicontinuous, resp. a universal homeomorphism\emph{)}, it is necessary and sufficient that $f_{\mathrm{red}}$ be so.
  \item Let $(U_\alpha)$ be an open cover of $Y$. For $f:X\to Y$ to be universally open \emph{(}resp. universally closed, resp. universally bicontinuous, resp. a universal homeomorphism\emph{)}, it is necessary and sufficient that for every $\alpha$, its restriction $f^{-1}(U_\alpha)\to U_\alpha$ be so.
\end{enumerate}
\end{proposition}

\begin{proof}
\oldpage[IV-2]{20}
Assertion (i) follows from \sref[I]{I.4.3.2}.
Assertion (ii) also follows from the definitions, and it is the same for (iii) by reducing to the case where $Y=S$, $Y'=S'$, which one can do using \sref[I]{I.3.3.11}; one knows that (iv) follows from (ii) and (iii) \sref[I]{I.3.5.1}.
To prove (v), note that for every morphism $Z'\to Z$, $f_{(Z')}:X_{(Z')}\to Y_{(Z')}$ is surjective \sref[I]{I.3.5.2}, and it is then a purely topological question.
For the case where $g\circ f$ is open (resp. closed), the fact that $g$ is then open (resp. closed) follows from Bourbaki, \emph{Top.~g\'en.}, chap.~I, 3\textsuperscript{e} \'ed., \textsection5, no.~1, prop.~1; for the two other cases, we can reduce to supposing that $g(f(X))=g(Y)=Z$, in other words that $g\circ f$ is a homeomorphism from $X$ to $Z$; since $f$ is surjective, $g$ is necessarily bijective, and since $g$ is a continuous open map by what precedes, $g$ is indeed a homeomorphism from $Y$ to $Z$.

To prove (vi), note that to say that a morphism $g$ is open (resp. closed, resp. a homeomorphism onto its image, resp. a homeomorphism bijective) is equivalent to saying that $g_{\mathrm{red}}$ has the same property.
On the other hand \sref[I]{I.5.1.8}, for every morphism $Y'\to Y$, one has $(X_{\mathrm{red}}\times_{Y_{\mathrm{red}}}Y'_{\mathrm{red}})_{\mathrm{red}}=(X\times_Y Y')_{\mathrm{red}}$, so the previous remark shows that if $f_{\mathrm{red}}$ is universally open (resp. universally closed, resp. universally bicontinuous, resp. a universal homeomorphism), the same is true of $f$.
The converse is proved in the same way, noting here that for every morphism $Y''\to Y_{\mathrm{red}}$, one has $(X_{\mathrm{red}}\times_{Y_{\mathrm{red}}}Y'')_{\mathrm{red}}=(X\times_Y Y'')_{\mathrm{red}}$ \sref[I]{I.5.1.8}.

Finally, the necessity of (vii) follows immediately from (iii).
Conversely, suppose (vii) is fulfilled, and let $g:Y'\to Y$ be a morphism; then the $g^{-1}(U_\alpha)=U'_\alpha$ form an open cover of $Y'$, and if we denote $f_\alpha$ the restriction $f^{-1}(U_\alpha)\to U_\alpha$ of $f$, and $f'$ the morphism $f_{(Y')}$, the restriction $f'^{-1}(U'_\alpha)\to U'_\alpha$ is none other than $(f_\alpha)_{(U'_\alpha)}$.
We can therefore restrict to proving that $f$ is open (resp. closed, resp. a homeomorphism onto its image, resp. a homeomorphism onto $Y$), which is immediate.
\end{proof}

\begin{proposition}[2.4.4]
\label{IV.2.4.4}
A universally bicontinuous morphism $f:X\to Y$ is radiciel \emph{(}hence separated \sref{IV.1.7.7.1}\emph{)}.
\end{proposition}

\begin{proof}
Indeed, $f$ is universally injective by hypothesis \sref[I]{I.3.5.11}.
\end{proof}

\begin{proposition}[2.4.5]
\label{IV.2.4.5}
\begin{enumerate}[\emph{(i)}]
  \item A morphism $f:X\to Y$ which is \emph{integral}, \emph{surjective} and \emph{radiciel} is a universal homeomorphism.
  \item Inversely, suppose $Y$ locally noetherian.
  Then, if a morphism of finite type $f:X\to Y$ is a universal homeomorphism, it is finite, surjective and radiciel.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) It suffices to observe that the three properties of $f$ are conserved by base change \sref[I]{I.3.5.2}, \sref[I]{I.3.5.7} and \sref[II]{II.6.1.5}, and as an integral morphism is closed \sref[II]{II.6.1.10}, it is clear that $f$ is a universal homeomorphism from $X$ to $Y$.

(ii) Since $f$ is of finite type and closed by hypothesis and separated by \sref{IV.2.4.4}, it is proper \sref[II]{II.5.4.4}, and for every $y\in Y$, $f^{-1}(y)$ is reduced to a single element; therefore \sref[III]{III.4.4.2} $f$ is finite; it is clear that $f$ is surjective, and it is radiciel since it is universally injective \sref[I]{I.3.5.11}.
\end{proof}

\begin{theorem}[2.4.6]
\label{IV.2.4.6}
Let $f:X\to Y$ be a quasi-flat morphism \sref{IV.2.3.3} and locally of finite presentation.
Then $f$ is universally open.
In particular, a flat morphism locally of finite presentation is universally open.
\end{theorem}

\begin{proof}
We know that for every base change $Y'\to Y$, $f_{(Y')}$ is quasi-flat \sref{IV.2.3.3} and locally of finite presentation \sref{IV.1.4.3}~(iii).
It therefore suffices to prove that $f$ is an open
\oldpage[IV-2]{21}
morphism.
But this follows from the criterion \sref{IV.1.10.4} for morphisms locally of finite presentation, the conditions b), b'), and c) of \sref{IV.1.10.4} being none other than the conditions (i), (ii), (iii) of \sref{IV.2.3.4}.
\end{proof}

\begin{corollary}[2.4.7]
\label{IV.2.4.7}
For every prescheme $Y$, the structural morphism $\mathbf{V}_Y^n\to Y$ \emph{(}where $\mathbf{V}_Y^n=Y\otimes_{\bb{Z}}\Spec(\bb{Z}[T_1,\ldots,T_n])$, also written $Y[T_1,\ldots,T_n]$\emph{)} is universally open.
\end{corollary}

\begin{proof}
Indeed, for $Y=\Spec(A)$, $Y[T_1,\ldots,T_n]=\Spec(A[T_1,\ldots,T_n])$, and $A[T_1,\ldots,T_n]$ is a free $A$-algebra of finite presentation.
\end{proof}

\begin{remarks}[2.4.8]
\label{IV.2.4.8}
(i) Note that a morphism $f:X\to Y$ faithfully flat and quasi-compact \emph{need not be universally open}, even when $X$ and $Y$ are noetherian.
Take for example for $Y$ a reduced irreducible algebraic curve \sref[II]{II.7.4.2} of generic point $y$, and let $X$ be the prescheme \emph{sum} $Y\amalg\Spec(k(y))$, $f$ being the canonical morphism; it is clear that $f$ is flat and surjective, hence faithfully flat, and quasi-compact, but the image by $f$ of the open part $\Spec(k(y))$ of $X$ is the singleton $\{y\}$, which is not open in $Y$ \sref[II]{II.7.4.3}.

(ii) For every prescheme $X$, the canonical morphism $f:X_{\mathrm{red}}\to X$ is a closed immersion and a \emph{universal homeomorphism} \sref{IV.2.4.3}~(vi); but when $X$ is locally noetherian, $f$ is not \emph{flat} unless $X$ is reduced, that is $f=1_X$ \sref{IV.2.2.17}.
\end{remarks}

\begin{proposition}[2.4.9]
\label{IV.2.4.9}
Let $Y$ be a discrete prescheme.
Then every morphism $f:X\to Y$ is universally open.
\end{proposition}

\begin{proof}
The question being local on $Y$ \sref{IV.2.4.3}~(vii), we can reduce to the case where the underlying space of $Y$ is a point; replacing $f$ by $f_{\mathrm{red}}$ \sref{IV.2.4.3}~(vi), we can suppose moreover that $Y$ is the spectrum of a field $k$; on the other hand, for every base change $Y'\to Y$, the open sets of $X'=X\times_Y Y'$ are the inverse images of open affine sets of $X'$, so we can suppose $X=\Spec(B)$ affine, $B$ being a $k$-algebra.
It suffices then to prove that for every $k$-algebra $A'$, if we set $B'=B\otimes_k A'$, the image by $f':\Spec(B')\to\Spec(A')$ of every open set of the form $U=D(t)$ ($t\in B'$) is open in $\Spec(A')$.
Now, $B$ is the inductive limit of the family of its sub-$k$-algebras of finite type $B_\alpha$, so $B'=\varinjlim(B_\alpha\otimes_k A')$; there exists an index $\alpha$ such that $t$ is the image in $B'$ of an element $t_\alpha\in B'_\alpha=B_\alpha\otimes_k A'$.
But since $k$ is a field, $B_\alpha$ is a $k$-algebra of finite presentation and a flat $k$-module, hence $B'_\alpha$ is an $A'$-algebra of finite presentation and a flat $A'$-module; we conclude from \sref{IV.2.4.6} that the image by $f'_\alpha:\Spec(B'_\alpha)\to\Spec(A')$ is \emph{open} in $\Spec(A')$.
It suffices to check that for every point $y'\in f'_\alpha(U_\alpha)$, the intersection $V=U\cap f'^{-1}(y')$ is non-empty.
But by definition of $y'$, $V_\alpha=U_\alpha\cap f_\alpha'^{-1}(y')$ is non-empty; that is, $V_\alpha$ is a non-empty open subset of $\Spec(B'_\alpha\otimes_{k}k(y'))$ and $V$ is its inverse image by the morphism $v:\Spec(B\otimes_k k(y'))\to\Spec(B_\alpha\otimes_k k(y'))$.
As $B_\alpha$ is a sub-algebra of $B$ and $k$ is a field, the homomorphism $B_\alpha\otimes_k k(y')\to B\otimes_k k(y')$
\oldpage[IV-2]{22}
is injective, so the morphism $v$ is dominant \sref[I]{I.1.2.7}, which concludes the proof.
\end{proof}

\begin{corollary}[2.4.10]
\label{IV.2.4.10}
Let $k$ be a field, $X$, $Y$ two $k$-preschemes; then the projection morphism $X\times_k Y\to X$ is universally open.
In particular, for every extension $K$ of $k$, the projection morphism $X_{(K)}\to X$ is universally open.
\end{corollary}

\begin{proof}
It suffices to apply \sref{IV.2.4.9} to the structural morphism $Y\to\Spec(k)$.
\end{proof}

\begin{remark}[2.4.11]
\label{IV.2.4.11}
If $f:X\to Y$ is an \emph{open} morphism, we know that, for every part $E$ of $Y$, one has $f^{-1}(\overline{E})=\overline{f^{-1}(E)}$ (Bourbaki, \emph{Top.~g\'en.}, chap.~I, 3\textsuperscript{e} \'ed., \textsection5, no.~4, prop.~7).
This remark applies for example when $f$ is a flat morphism \emph{locally of finite presentation} \sref{IV.2.4.6}, or a morphism projection $X\times_k Y\to X$ where $X$, $Y$ are preschemes over a field $k$ \sref{IV.2.4.10}, and generalises then \sref{IV.2.3.10}.
\end{remark}

\subsection{Permanence of properties of Modules by faithfully flat descent}
\label{subsection:IV.2.5}

\begin{proposition}[2.5.1]
\label{IV.2.5.1}
Let $f:X\to Y$ be a morphism, $\sh{F}$ a quasi-coherent $\sh{O}_X$-Module, $g:Y'\to Y$ a faithfully flat morphism, $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$, $\sh{F}'=\sh{F}\otimes_{\sh{O}_Y}\sh{O}_{Y'}$.
For $\sh{F}$ to be flat \emph{(}resp. faithfully flat\emph{)} relative to $f$, it is necessary and sufficient that $\sh{F}'$ be flat \emph{(}resp. faithfully flat\emph{)} relative to $f'$.
\end{proposition}

\begin{proof}
It suffices to apply \sref{IV.2.2.10} replacing $X$, $Y$, $Z$, $\sh{F}$, $\sh{G}'$ by $Y'$, $X$, $Y$, $\sh{O}_{Y'}$, $\sh{F}$ respectively; we conclude that for $\sh{F}$ to be flat (resp. faithfully flat) relative to $f$, it is necessary and sufficient that $\sh{F}'$ be flat (resp. faithfully flat) relative to $g\circ f'$.
But since $g':X'\to X$ is the canonical projection, $g'$ is faithfully flat \sref{IV.2.2.13} and we have $g\circ f'=f\circ g'$; therefore for $\sh{F}'$ to be flat (resp. faithfully flat) relative to $f\circ g'$, it is necessary and sufficient that $\sh{F}$ be so relative to $f$ \sref{IV.2.2.11}~(iii).
\end{proof}

\begin{proposition}[2.5.2]
\label{IV.2.5.2}
Let $f:X'\to X$ be a faithfully flat and quasi-compact morphism, $\sh{F}'=f^*(\sh{F})$.
Consider, for a quasi-coherent Module, the property of being:
\begin{enumerate}[\emph{(i)}]
  \item of finite type;
  \item of finite presentation;
  \item locally free of finite type;
  \item locally free of rank $n$.
\end{enumerate}
Then, if $\mathbf{P}$ denotes one of the preceding properties, for $\sh{F}$ to have property $\mathbf{P}$, it is necessary and sufficient that $\sh{F}'$ have it.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{23}
Note that the question is local on $X$, and we can therefore suppose $X$ affine; we then know \sref{IV.2.2.12} that there is an affine scheme $X''$ and a faithfully flat morphism $g:X''\to X'$ which is a surjective local isomorphism; by hypothesis, the question is local on $X$, and taking into account \sref{IV.2.2.3} and \sref[II]{II.6.1.4.1}, it therefore suffices to prove the

\begin{lemma}[2.5.3]
\label{IV.2.5.3}
Let $A$ be a ring, $A'$ a faithfully flat $A$-algebra, $M$ an $A$-module; $M'=M\otimes_A A'$.
For $M$ to be of finite type \emph{(}resp. of finite presentation\emph{)}, it is necessary and sufficient that $M'$ be so.
\end{lemma}

The proof follows from Bourbaki, \emph{Alg.~comm.}, chap.~I, \textsection3, no.~6, prop.~11.
\end{proof}

\begin{remark}[2.5.4]
\label{IV.2.5.4}
The assertions of \sref{IV.2.5.2} for properties (i) and (ii) are still valid when one supposes only $f$ \emph{quasi-faithfully flat} \sref{IV.2.3.3} and quasi-compact.
This is because one is reduced to Bourbaki, \emph{loc.~cit.} in this way.
\end{remark}

\label{IV.2.5.4.1}\emph{Lemma} \sref{IV.2.5.4.1}. ---
\emph{Let $\rho:A\to A'$ be a homomorphism of rings, such that the corresponding morphism $f:\operatorname{Spec}(A')\to\operatorname{Spec}(A)$ is surjective; suppose that there exists an $A'$-module $N'$ of finite type, which is A-flat and has $\operatorname{Spec}(A')$ as support.
Then, if $u:P\to Q$ is a homomorphism of A-modules such that $u\otimes 1:P\otimes_A A'\to Q\otimes_A A'$ is surjective, $u$ is surjective.}

Indeed, one deduces first that the homomorphism $u\otimes 1_{N'}:(P\otimes_A A')\otimes_{A'}N'\to(Q\otimes_A A')\otimes_{A'}N'$ is surjective.
Let $\mathfrak{q}$ be a prime ideal of $A'$, and let $\mathfrak{p}=\rho^{-1}(\mathfrak{q})$; the corresponding homomorphism $(P\otimes_A N')_\mathfrak{q}\to(Q\otimes_A N')_\mathfrak{q}$ is surjective, and can be written $u_\mathfrak{p}\otimes 1:P_\mathfrak{p}\otimes_{A_\mathfrak{p}}N'_\mathfrak{q}\to Q_\mathfrak{p}\otimes_{A_\mathfrak{p}}N'_\mathfrak{q}$ ($\mathbf{0_I}$, 1.5.4).
By hypothesis we have $N'_\mathfrak{q}\neq 0$ and $N'_\mathfrak{q}$ is a flat $A_\mathfrak{p}$-module ($\mathbf{0_I}$, 6.3.1); by virtue of Nakayama's lemma, we have $\mathfrak{q} N'_\mathfrak{q}\neq N'_\mathfrak{q}$, and \emph{a fortiori} $\mathfrak{p} N'_\mathfrak{q}\neq N'_\mathfrak{q}$, hence $N'_\mathfrak{q}$ is a faithfully flat $A_\mathfrak{p}$-module ($\mathbf{0_I}$, 6.4.1).
It follows that $u_\mathfrak{p}$ is surjective ($\mathbf{0_I}$, 6.4.1), and this holding for every $\mathfrak{p}\in\operatorname{Spec}(A)$, since $f$ is surjective, one concludes finally that $u$ is surjective (Bourbaki, \emph{Alg.\ comm.}, chap.~II, \textsection~3, n\textsuperscript{o}~3, th.~1).

\begin{proposition}[2.5.5]
\label{IV.2.5.5}
Let $f:X\to Y$ be a morphism, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent of finite type and $f$-flat, $\sh{E}$ an $\sh{O}_Y$-Module quasi-coherent of finite type; for every $y\in Y$, set $\sh{F}_y=\sh{F}\otimes_{\sh{O}_y}k(y)$.
For a point $x\in X$ to be a maximal point of $\Supp(\sh{E}\otimes_Y\sh{F})$, it is necessary and sufficient that $y=f(x)$ be a maximal point of $\Supp(\sh{E})$, and that $x$ be a maximal point of $\Supp(\sh{F}_y)$ in $f^{-1}(y)$.
If this is so, then we have
\begin{equation}
\label{IV.2.5.5.1}
  \operatorname{long}((\sh{E}\otimes_Y\sh{F})_x) = \operatorname{long}(\sh{E}_y)\cdot\operatorname{long}((\sh{F}_y)_x).
\end{equation}
\end{proposition}

\begin{proof}
\oldpage[IV-2]{24}
It is clear that $f(\Supp(\sh{E}\otimes_Y\sh{F}))\subset\Supp(\sh{E})$ ($\mathbf{0_I}$, 5.2.2); the image by $f$ of every irreducible component of $\Supp(\sh{E}\otimes_Y\sh{F})$ is hence contained in an irreducible component of $\Supp(\sh{E})$.
One can restrict to the case where $\Supp(\sh{F})=X$.
Indeed ($\mathbf{Err_{III}}$, 30), since $\sh{F}$ is of finite type, there exists a closed sub-prescheme $X'$ of $X$ having $\Supp(\sh{F})$ as underlying space and an $\sh{O}_{X'}$-Module quasi-coherent of finite type $\sh{G}$ such that, if $j:X'\to X$ is the canonical injection, we have $\sh{F}=j_*(\sh{G})$.
If one sets $f'=f\circ j$, it is clear that $\sh{G}$ is $f'$-flat and that $\Supp(\sh{E}\otimes_Y\sh{F})=\Supp(\sh{E}\otimes_Y\sh{G})$.

Suppose therefore $\Supp(\sh{F})=X$; if $Z$ is an irreducible component of $\Supp(\sh{E})$, one has $f^{-1}(Z)\subset\Supp(\sh{E}\otimes_Y\sh{F})$ ($\mathbf{I}$, 9.1.13), and it results from \sref{IV.2.3.4} that every irreducible component of $f^{-1}(Z)$ dominates $Z$; in other words, if $x$ is the generic point of an irreducible component of $\Supp(\sh{E}\otimes_Y\sh{F})$ contained in $f^{-1}(Z)$, $y=f(x)$ is the generic point of $Z$; moreover ($\mathbf{0_I}$, 2.1.8), $x$ is the generic point of an irreducible component of $f^{-1}(y)=\Supp(\sh{F}_y)$ ($\mathbf{I}$, 9.1.13), and reciprocally every generic point of one of these components is the generic point of an irreducible component of $f^{-1}(Z)$.

It remains to prove \sref{IV.2.5.5.1}; one has $(\sh{E}\otimes_Y\sh{F})_x=\sh{E}_y\otimes_{\sh{O}_y}\sh{F}_x$ ($\mathbf{I}$, 9.1.12) and $(\sh{F}_y)_x=\sh{F}_x/\mathfrak{m}_y\sh{F}_x$; one is hence reduced to proving the

\medskip
\label{IV.2.5.5.2}\emph{Lemma} \sref{IV.2.5.5.2}. ---
\emph{Let $A$, $B$ be two local rings, $\rho:A\to B$ a local homomorphism, $N$ a B-module which is a faithfully flat A-module and such that $N/\mathfrak{m} N$ is a B-module of finite length; then one has}
\begin{equation}
\tag{2.5.5.3}
\label{IV.2.5.5.3}
  \operatorname{long}_B(M\otimes_A N)=\operatorname{long}_A(M)\cdot\operatorname{long}_B(N/\mathfrak{m} N).
\end{equation}

If $M$ is of infinite length, the same is true of $M\otimes_A N$, for every strictly increasing sequence of $n$ submodules $M_i$ of $M$ $(1\leq i\leq n)$ yields $n$ distinct submodules $M_i\otimes_A N$ of $M\otimes_A N$; since $N\neq\mathfrak{m} N$ since $N$ is a faithfully flat A-module, the formula \sref{IV.2.5.5.3} is true in this case.
Suppose therefore $M$ of finite length.
If $M=0$ the two sides of \sref{IV.2.5.5.3} are zero.
One can hence suppose $M\neq 0$; if $\mathfrak{m}^k M\neq 0$, it follows from Nakayama's lemma that $\mathfrak{m}^{k+1}M\neq\mathfrak{m}^k M$; since the submodules $\mathfrak{m}^k M$ are of finite length, there necessarily exists an integer $r$ such that $\mathfrak{m}^r M=0$.
The $\mathfrak{m}^k M\otimes_A N$ are hence sub-B-modules of $M\otimes_A N$, $(\mathfrak{m}^k M\otimes_A N)/(\mathfrak{m}^{k+1}M\otimes_A N)$ being also isomorphic to $(\mathfrak{m}^k M/\mathfrak{m}^{k+1}M)\otimes_{A/\mathfrak{m}}(N/\mathfrak{m} N)$ $(0\leq k\leq r-1)$; the length of the latter, considered as a B-module, is hence the product of $\operatorname{long}_B(N/\mathfrak{m} N)$ and of the rank of the $(A/\mathfrak{m})$-vector space $\mathfrak{m}^k M/\mathfrak{m}^{k+1}M$, rank which is also the length of the A-module $\mathfrak{m}^k M/\mathfrak{m}^{k+1}M$.
By addition (for $0\leq k\leq r-1$), one deduces also the formula \sref{IV.2.5.5.3}.
\end{proof}

\begin{remark}[2.5.5.4]
\label{IV.2.5.5.4}
We note that when $N$ is an $A$-module of finite type, saying that $N$ is a faithfully flat $A$-module is equivalent to saying that $N$ is a flat $A$-module; indeed, Nakayama's lemma then shows that $\mathfrak{m}N\neq N$.
\end{remark}

\begin{lemma}[2.5.6]
\label{IV.2.5.6}
Let $B$ be a ring \emph{(}not necessarily commutative\emph{)}, $V$, $W$ two isomorphic left $B$-modules, $C$ the ring $\operatorname{End}_B(V)$, and let $M=\Hom_B(V,W)$, equipped with its canonical structure of right $C$-module.
Then $M$ is a $C$-module isomorphic to $C_d$; moreover, for every $u\in M$, the following conditions are equivalent:
\begin{enumerate}[\rm a)]
  \item $\{u\}$ is a basis of the $C$-module $M$.
  \item $u$ is an isomorphism from $V$ to $W$.
\end{enumerate}
\end{lemma}

\begin{proof}
If $u$ is an isomorphism from $V$ to $W$, the map $v\mapsto u\circ v$ from $C$ to $M$ is evidently a bijection, hence b) implies a).
Conversely, suppose that $\{u\}$ is a basis of the $C$-module $M$.
By hypothesis, there exists an isomorphism $u'$ from $V$ to $W$, and $\{u'\}$ is therefore a basis of $M$; there is thus an invertible element $w$ of $C$ (that is, an automorphism of $V$) such that $u=u'\circ w$, which implies that $u$ is an isomorphism from $V$ to $W$.
\end{proof}

\begin{corollary}[2.5.7]
\label{IV.2.5.7}
The hypotheses on $B$, $V$ and $W$ being those of \sref{IV.2.5.6}, suppose moreover that one of the following conditions is satisfied:
\begin{enumerate}[\rm(i)]
  \item $V$ and $W$ are Noetherian $B$-modules;
  \item $V$ and $W$ are modules of finite presentation over a commutative sub-ring of $B$.
\end{enumerate}
Then the conditions \emph{a)} and \emph{b)} of \sref{IV.2.5.6} are also equivalent to the following:
\begin{enumerate}
  \item[\rm a')] $u$ is a generator of the $C$-module $M$.
  \item[\rm b')] $u$ is an epimorphism from $V$ to $W$.
\end{enumerate}
\end{corollary}

\begin{proof}
One knows indeed that an epimorphism of an $A$-module $E$ onto itself is \emph{bijective} in the two following cases: 1\textsuperscript{o}~$E$ is a Noetherian $A$-module (Bourbaki, \emph{Alg.}, chap.~VIII, \textsection2, no.~2, lemma~3); 2\textsuperscript{o}~$A$ is commutative and $E$ is an $A$-module of finite presentation \sref{IV.8.9.3}\footnote{The reader can verify that \sref{IV.2.5.7} and \sref{IV.2.5.8} are not used before \textsection9.}; hence b) and b') are equivalent.

\oldpage[IV-2]{25}

On the other hand, if $u$ generates $M$ and if $\{u'\}$ is a basis of $M$, there exists $v\in C$ such that $u'=u\circ v$, which proves that $u$ is surjective; hence a') implies b'), and since a) evidently implies a'), this finishes proving the corollary.
\end{proof}

\begin{proposition}[2.5.8]
\label{IV.2.5.8}
Let $A$ be a commutative semi-local ring, $B$ an $A$-algebra \emph{(}not necessarily commutative\emph{)}, $V$ and $W$ two $B$-modules.
Let $A'$ be a commutative $A$-algebra which is a faithfully flat $A$-module; set $B'=B\otimes_A A'$, $V'=V\otimes_A A'$, $W'=W\otimes_A A'$, so that $B'$ is an $A'$-algebra, $V'$ and $W'$ are $B'$-modules.
Suppose moreover that one of the following conditions is satisfied:
\begin{enumerate}[\rm(i)]
  \item $A$ and $A'$ are Noetherian, $V$ and $W$ are $A$-modules of finite type.
  \item $B$ is an $A$-module of finite type, $V$ is a projective $B$-module of finite type and an $A$-module of finite presentation.
\end{enumerate}
Then, if $V'$ and $W'$ are isomorphic $B'$-modules, $V$ and $W$ are isomorphic $B$-modules.
\end{proposition}

\begin{proof}
We note that in case (ii), $W'$ being $A'$-isomorphic to $V'$, is an $A'$-module of finite type, whence it results that $W$ is an $A$-module of finite type (Bourbaki, \emph{Alg.~comm.}, chap.~I, \textsection3, no.~6, prop.~11); hence in all cases $V$ and $W$ are $A$-modules of finite type.
Moreover:

\medskip
\label{IV.2.5.8.1}\textbf{(2.5.8.1)}
\emph{Under either of the hypotheses} (i), (ii), $\Hom_B(V,W)$ \emph{is an $A$-module of finite type.}
\medskip

This is evident in case (i), since $\Hom_A(V,W)$ is then an $A$-module of finite type, and $\Hom_B(V,W)$ is a sub-$A$-module of $\Hom_A(V,W)$.
In case (ii), $V$ is a direct factor of a free $B$-module $B^p$, hence $\Hom_B(V,W)$ is a direct factor of $\Hom_B(B^p,W)=W^p$, and since $W$ is an $A$-module of finite type, the same is true of $\Hom_B(V,W)$.

Set $C=\operatorname{End}_B(V)$, $M=\Hom_B(V,W)$, which are $A$-modules of finite type in cases (i) and (ii).
One knows that under either of the conditions (i), (ii), the canonical homomorphism
\begin{equation}
\label{IV.2.5.8.2}\tag{2.5.8.2}
  \Hom_A(V,W)\otimes_A A'\to\Hom_{A'}(V',W')
\end{equation}
is bijective (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection2, no.~10, prop.~11).
As $A'$ is a flat $A$-module, $\Hom_B(V,W)\otimes_A A'$ identifies canonically with a sub-$A'$-module of $\Hom_A(V,W)\otimes_A A'$.
The image of this submodule by the homomorphism \sref{IV.2.5.8.2} is contained in $\Hom_{B'}(V',W')$, for if $u\in\Hom_B(V,W)$ and $a'\in A'$, the image of $u\otimes a'$ by \sref{IV.2.5.8.2} is the homomorphism $u':V'\to W'$ such that $u'(x\otimes 1)=u(x)\otimes a'$; for every $b\in B$, we therefore have $u'((b\otimes 1)(x\otimes 1))=u'(bx\otimes 1)=u(bx)\otimes a'=bu(x)\otimes a'=(b\otimes 1)(u(x)\otimes a')$, whence our assertion.
This being so:

\medskip
\label{IV.2.5.8.3}\textbf{(2.5.8.3)}
\emph{Under either of the hypotheses} (i), (ii), \emph{the homomorphism}
\begin{equation}
\label{IV.2.5.8.4}\tag{2.5.8.4}
  \Hom_B(V,W)\otimes_A A'\to\Hom_{B'}(V',W')
\end{equation}
\emph{is bijective.}
\medskip

For every $b\in B$, denote by $h(b)$ (resp.\ $h'(b)$) the homothety $x\mapsto bx$ in $V$ (resp.\ $W$), which is an $A$-endomorphism.
Let $(b_\alpha)_{\alpha\in I}$ be a system of generators of the $A$-algebra $B$; the map
\[
  u\mapsto(h'(b_\alpha)\circ u-u\circ h(b_\alpha))_{\alpha\in I}
\]
from $\Hom_A(V,W)$ to $(\Hom_A(V,W))^I$ is $A$-linear, and by definition, its kernel is none other than $\Hom_B(V,W)$; in other words, we have an exact sequence
\[
  0\to\Hom_B(V,W)\to\Hom_A(V,W)\to(\Hom_A(V,W))^I.
\]
The same reasoning applies by replacing $A$, $B$, $V$, $W$ by $A'$, $B'$, $V'$, $W'$; moreover, we have a diagram
\begin{equation}
\label{IV.2.5.8.5}\tag{2.5.8.5}
\xymatrix@C=0.5em{
0 \ar[r] & \Hom_B(V,W){\otimes_A}A' \ar[r] \ar[d]^r & \Hom_A(V,W){\otimes_A}A' \ar[r] \ar[d]^s & {(\Hom_A(V,W))^I}{\otimes_A}A' \ar[d]^t \\
0 \ar[r] & \Hom_{B'}(V',W') \ar[r] & \Hom_{A'}(V',W') \ar[r] & {(\Hom_{A'}(V',W'))^I}
}
\end{equation}
where $r$ is the homomorphism \sref{IV.2.5.8.4}, $s$ the homomorphism \sref{IV.2.5.8.2}, and $t$ the composed homomorphism
\[
  (\Hom_A(V,W))^I\otimes_A A'\xrightarrow{w}(\Hom_A(V,W)\otimes_A A')^I\xrightarrow{s^I}(\Hom_{A'}(V',W'))^I
\]
$w$ being the canonical homomorphism (Bourbaki, \emph{Alg.}, chap.~II, 3\textsuperscript{e}~éd., \textsection3, no.~7).
One verifies immediately that the diagram \sref{IV.2.5.8.5} is commutative, and since $A'$ is a flat $A$-module, its rows are exact.
Finally, we have seen

\oldpage[IV-2]{26}

that $s$ is an isomorphism, and it is therefore the same for $s^I$; in case (ii), one can take $I$ finite, and one knows then that $w$ is bijective (Bourbaki, \emph{loc.~cit.}, prop.~7); in case (i), one notes that if $\tilde{B}$ (resp.\ $\tilde{B}'$) is the image of $B$ under $h$ (resp.\ $h'$) in $\operatorname{End}_A(V)$ (resp.\ $\operatorname{End}_A(W)$), $\tilde{B}$ and $\tilde{B}'$ are $A$-modules of finite type, hence one can again take $I$ finite.
Thus, in all cases, $t$ is bijective, and one therefore concludes that the same is true of $r$.

It follows therefore from \sref{IV.2.5.8.4} that if one sets $C'=C\otimes_A A'$, $M'=M\otimes_A A'$, one has canonical bijections
\begin{equation}
\label{IV.2.5.8.6}\tag{2.5.8.6}
  C'\xrightarrow{\sim}\operatorname{End}_{B'}(V'),\quad M'\xrightarrow{\sim}\Hom_{B'}(V',W')
\end{equation}
of which the first is an isomorphism of $A'$-algebras, the second constituting with the first a di-isomorphism of right $C'$-modules.

\medskip
\label{IV.2.5.8.7}\textbf{(2.5.8.7)}
\emph{Reduction to the case $V=B_s$.}
The hypothesis that $V'$ and $W'$ are isomorphic $B'$-modules implies that $C'_d$ and $M'$ are isomorphic right $C'$-modules.
Let us show that in order to prove \sref{IV.2.5.8}, it suffices to find an element $u\in M$ that is a generator of $M$ as a right $C$-module; indeed, in case (i), $V'$ and $W'$ are $A'$-modules of finite type, hence Noetherian since $A'$ is Noetherian; in case (ii), $V'$ and $W'$ are (isomorphic) $A'$-modules of finite presentation.
One can therefore apply \sref{IV.2.5.7} to $A'$, $B'$, $V'$ and $W'$ and one concludes that $u'=u\otimes 1$ is a $B'$-isomorphism from $V'$ to $W'$.
But since $A'$ is faithfully flat over $A$, this implies that $u$ is bijective ($\mathbf{0_I}$, 6.4.1), that is, the conclusion of \sref{IV.2.5.8}.
If one notes that in case (i), $C$ and $M$ are $A$-modules of finite type, and that in case (ii), $C$ is (as was seen in \sref{IV.2.5.8.1}) a direct factor of $V^n$, hence a projective $A$-module of finite type, one sees that one is reduced (by changing notations) to proving \sref{IV.2.5.8} in the particular case where $V=B_s$, and that it suffices to prove the existence of a \emph{generator} of the $B$-module $W$.
One notes that then $B$ is an $A$-module \emph{of finite type}.

\medskip
\label{IV.2.5.8.8}\textbf{(2.5.8.8)}
\emph{Reduction to the case where $A$ and $A'$ are fields, $B$ a simple algebra with centre $A$.}
Let $\mathfrak{r}$ be the radical of the semi-local ring $A$; it suffices to prove that $W/\mathfrak{r}W$ is a monogenic $(B/\mathfrak{r}B)$-module, for if there exists a surjective homomorphism $B_s/\mathfrak{r}B_s\to W/\mathfrak{r}W$, it gives by composition a homomorphism $B_s\to B_s/\mathfrak{r}B_s\to W/\mathfrak{r}W$, which itself (since $B_s$ is a free $B$-module) can be written $B_s\to W\to W/\mathfrak{r}W$, so that the surjective homomorphism considered is $f\otimes 1:B_s\otimes_A(A/\mathfrak{r})\to W\otimes_A(A/\mathfrak{r})$.
As $W$ is an $A$-module of finite type, Nakayama's lemma shows that $f$ is surjective (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection3, no.~2, cor.~1 of prop.~4).
Setting $A_1=A/\mathfrak{r}$, $A'_1=A'\otimes_A A_1=A'/\mathfrak{r}A'$, $B_1=B/\mathfrak{r}B=B\otimes_A A_1$, $W_1=W/\mathfrak{r}W=W\otimes_A A_1$, the hypotheses (i) (resp.\ (ii)) remain satisfied when one replaces $A$, $A'$, $B$, $V$ by $A_1$, $A'_1$, $B_1$, $V_1=(B_1)_s$, $W_1$ respectively; moreover, $V'_1=V'\otimes_A A_1=V_1\otimes_{A_1}A'_1$ and $W'_1=W'\otimes_{A_1}A'_1$ are $B'_1$-isomorphic (with $B'_1=B'\otimes_A A_1=B_1\otimes_{A_1}A'_1$) and $A'_1$ is a faithfully flat $A_1$-module.
One can therefore suppose for the proof of \sref{IV.2.5.8} that $A$ is a \emph{finite product of fields} (commutative).
As $B$ is an $A$-module of finite type, it is an \emph{artinian} ring; let $\mathfrak{N}$ be its radical.
It will now suffice to prove that $W/\mathfrak{N}W$ is a monogenic $(B/\mathfrak{N})$-module, for by using Nakayama's lemma, this implies that $W$ is a monogenic $B$-module; on the other hand, $B'/\mathfrak{N}B'=(B\otimes_A A')\otimes_B(B/\mathfrak{N})=(B/\mathfrak{N})\otimes_A A'$ and similarly $W'/\mathfrak{N}W'=(W/\mathfrak{N}W)\otimes_A A'$.
One can therefore suppose moreover that $\mathfrak{N}=0$, that is, that $B$ is a \emph{semi-simple} $A$-algebra.

Let us now note that since $A$ is a finite product of fields $k_i$ ($1\leq i\leq n$), $A'$ is a direct product of $k_i$-algebras $A'_i$ ($1\leq i\leq n$), each of the $A'_i$ being annihilated by the $k_j$ of index $j\neq i$; the hypothesis that $A'$ is a faithfully flat $A$-module implies that the $A'_i\neq 0$; there is therefore a quotient $A''_i$ of $A'_i$ that is a field, and $A''$, the direct product of the $A''_i$, is a faithfully flat $A$-module and a quotient of $A'$.
Considering then the ring $B''=B'\otimes_{A'}A''=B\otimes_A A''$, $W''=W'\otimes_{A'}A''=W\otimes_A A''$ is a $B''$-module isomorphic to $B''_s$; one can therefore restrict to proving \sref{IV.2.5.8} by replacing $A'$ by $A''$, that is, one can also suppose that $A'$ is a \emph{finite product of fields}.

Let $Z$ be the centre of $B$, which is a finite product of fields and an $A$-module of finite type; note that $B$ and $W$ are $Z$-modules of finite type, projective as any $Z$-module; moreover, $B'=B\otimes_Z Z'$ and $W'=W\otimes_Z Z'$ on setting $Z'=Z\otimes_A A'$, and $Z'$ is a faithfully flat $Z$-module.
One can therefore replace $A$ by $Z$ in the hypothesis, that is, suppose that $A$ is the \emph{centre} of $B$, $B$ being semi-simple and $A'$ a finite product of fields.
If $k_i$ ($1\leq i\leq n$) are the component fields of $A$, $B$ is a direct product of simple rings $B_i$, $k_i$ being the centre of $B_i$, and $W$ is a direct sum of submodules $W_i$ ($1\leq i\leq n$), $W_i$ being annihilated by the $B_j$ of index $j\neq i$; moreover, the reasoning made above shows that one can suppose that $A'$ is a product of fields $k'_i$ ($1\leq i\leq n$), $k'_i$ being an extension of $k_i$ and being

\oldpage[IV-2]{27}

annihilated by the $k_j$ of index $j\neq i$.
The hypothesis that $(B')_s$ and $W'$ are isomorphic $B'$-modules implies then that, for every $i$, $(B_i\otimes_{k_i}k'_i)_s$ and $W_i\otimes_{k_i}k'_i$ are isomorphic $(B_i\otimes_{k_i}k'_i)$-modules; it therefore suffices to prove \sref{IV.2.5.8} when $n=1$, that is, in the case where $A$ and $A'$ are \emph{fields}, $B$ a \emph{simple algebra with centre}~$A$.

\medskip
\label{IV.2.5.8.9}\textbf{(2.5.8.9)}
\emph{End of the proof.}
One knows then (Bourbaki, \emph{Alg.}, chap.~VIII, \textsection5, prop.~6 and~8) that every $B$-module is a direct sum of modules isomorphic to a minimal ideal of $B$, and two $B$-modules of finite rank over $A$ are therefore isomorphic if and only if they have \emph{the same rank} over $A$.
But by hypothesis, $[W':A']=[B'_s:A']=[B_s:A]$; hence $[W:A]=[B_s:A]$, which finishes the proof.
\end{proof}

\subsection{Permanence of set-theoretic and topological properties of morphisms by faithfully flat descent}
\label{subsection:IV.2.6}

\begin{proposition}[2.6.1]
\label{IV.2.6.1}
Let $f:X\to Y$ be an $S$-morphism of $S$-preschemes, $g:S'\to S$ a surjective morphism, $X'=X_{(S')}$, $Y'=Y_{(S')}$, $f'=f_{(S')}:X'\to Y'$.
Consider, for a morphism, the property of being:
\begin{enumerate}[\emph{(i)}]
  \item surjective;
  \item injective;
  \item with finite fibres \emph{(}as sets\emph{)};
  \item bijective;
  \item radiciel.
\end{enumerate}
Then, if $\mathbf{P}$ denotes one of the preceding properties and if $f'$ has property $\mathbf{P}$, the same is true of $f$.
\end{proposition}

\begin{proposition}[2.6.2]
\label{IV.2.6.2}
With the notations being those of \sref{IV.2.6.1}, suppose the morphism $g:S'\to S$ faithfully flat and quasi-compact.
Consider for a morphism the property of being:
\begin{enumerate}[\emph{(i)}]
  \item open;
  \item closed;
  \item quasi-compact and universally bicontinuous onto its image;
  \item a universal homeomorphism;
\end{enumerate}
Then, if $\mathbf{P}$ denotes one of the preceding properties and if $f'$ has property $\mathbf{P}$, the same is true of $f$.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{28}
Since the morphism $Y'\to Y$ is faithfully flat and quasi-compact \sref{IV.2.2.13} and \sref{IV.1.1.2}, one can, by virtue of \sref[I]{I.3.3.11}, reduce to the case where $Y=S$, $Y'=S'$.
If $g'$ is the projection $X'\to X$, we know that for every part $M$ of $X$, $g^{-1}(f(M))=f'(g'^{-1}(M))$ \sref[I]{I.3.4.8}.
If $f'$ is an open (resp. closed) morphism, then, for every open (resp. closed) part $M$ of $X$, $f'(g'^{-1}(M))$ is open (resp. closed) in $Y'$, and since $g$ is faithfully flat and quasi-compact, we conclude by virtue of \sref{IV.2.3.12} that $f(M)$ is open (resp. closed) in $Y$.
This proves the proposition for cases (i) and (ii).
Let us prove it in case (iii) (which will imply the conclusion in case (iv), taking into account \sref{IV.2.6.1}~(iv)); by virtue of \sref{IV.2.6.1}~(ii), $f$ is injective, and it remains to prove that $f$, considered as a map from $X$ onto $f(X)$, is a quasi-compact and \emph{open} map.
Since $f'$ is quasi-compact, the same is true of $f$ \sref{IV.1.1.4}.
It then suffices for every closed set $Z$ of $X$ to have $Z=f^{-1}(\overline{f(Z)})$; since $g'$ is surjective, this relation is equivalent to $g'^{-1}(Z)=g'^{-1}(f^{-1}(\overline{f(Z)}))$ or to $g'^{-1}(Z)=f'^{-1}(g^{-1}(\overline{f(Z)}))$.
Now, since $f$ is quasi-compact, its composite with the canonical injection $Z\to X$ (where $Z$ is the reduced sub-prescheme of $X$ having $Z$ as underlying space) is also quasi-compact, and one concludes that $f(Z)$ is pro-constructible.
Applying \sref{IV.2.3.10} to $f(Z)$ in $Y$ (image of the morphism $f|Z$), we obtain $g^{-1}(\overline{f(Z)})=\overline{g^{-1}(f(Z))}$; the formula to prove is therefore equivalent to: $Z'=f'^{-1}(\overline{f'(Z')})$, where $Z'=g'^{-1}(Z)$; but this last formula follows from the hypothesis that $f'$ is a homeomorphism onto $f'(X')$.
\end{proof}

\begin{remark}[2.6.3]
\label{IV.2.6.3}
In cases (i) and (ii), the conclusions of \sref{IV.2.6.2} are still valid when one supposes only $g$ \emph{quasi-faithfully flat} \sref{IV.2.3.3} and \emph{quasi-compact}; indeed, by virtue of \sref{IV.2.1.4}, \sref[I]{I.3.5.2} and \sref[I]{I.9.1.13.1}, one can still reduce to the case where $Y=S$, $Y'=S'$; the conclusion then follows from \sref{IV.2.3.12}.
In cases (iii) and (iv), the conclusions remain valid when one supposes only $g$ \emph{quasi-faithfully flat}; indeed, one then uses \sref{IV.2.3.10} and the fact that $g$ is surjective.
Finally, if $g$ is faithfully flat and locally of finite presentation, or if $g$ is surjective and $S$ discrete, the conclusion of \sref{IV.2.6.2} is valid when $\mathbf{P}$ is the property:

(iii bis) \emph{of being a homeomorphism onto a sub-space image};

this follows indeed from the proof given in \sref{IV.2.6.2} and from Remark~\sref{IV.2.4.11}.
\end{remark}

\begin{corollary}[2.6.4]
\label{IV.2.6.4}
With the notations being those of \sref{IV.2.6.1}, suppose the morphism $g:S'\to S$ faithfully flat and quasi-compact.
Consider for a morphism the property of being:
\begin{enumerate}[\emph{(i)}]
  \item universally open;
  \item universally closed;
  \item quasi-compact and universally bicontinuous;
  \item a universal homeomorphism;
  \item quasi-compact;
  \item quasi-compact and dominant.
\end{enumerate}
Then, if $\mathbf{P}$ denotes one of the preceding properties, for $f$ to have property $\mathbf{P}$, it is necessary and sufficient that $f'$ have it.
\end{corollary}

\begin{proof}
\oldpage[IV-2]{29}
Properties (v) and (vi) are included for the record, being consequences of \sref{IV.1.1.4}, \sref{IV.1.1.6} and \sref{IV.2.3.7}.
For the others, the condition is necessary by virtue of \sref{IV.2.4.3}.
Conversely, suppose for example that $f'$ is universally open, and let $Y_1\to Y$ be an arbitrary morphism; set $X_1=X\times_Y Y_1$ and $f_1=f_{(Y_1)}$.
On the other hand, set $Y'_1=Y_1\times_Y Y'$, $X'_1=X'\times_{Y'}Y'_1=X_1\times_{Y_1}Y'_1$, $f'_1=f'_{(Y'_1)}=(f_1)_{(Y'_1)}:X'_1\to Y'_1$; since $g'=g_{(Y'_1)}:Y'_1\to Y'$ is faithfully flat and quasi-compact \sref{IV.2.2.13} and \sref{IV.1.1.2}, and that $f'$ is universally open, $f'_1$ is open, and it follows from \sref{IV.2.6.2} that $f_1$ is universally open.
The same reasoning applies for the other cases.
\end{proof}

\subsection{Permanence of various properties of morphisms by faithfully flat descent}
\label{subsection:IV.2.7}

\begin{proposition}[2.7.1]
\label{IV.2.7.1}
Let $f:X\to Y$ be an $S$-morphism of $S$-preschemes, $g:S'\to S$ a faithfully flat and quasi-compact morphism, $X'=X_{(S')}$, $Y'=Y_{(S')}$, $f'=f_{(S')}:X'\to Y'$.
Consider, for a morphism, the property of being:
\begin{enumerate}[\emph{(i)}]
  \item separated;
  \item quasi-separated;
  \item locally of finite type;
  \item locally of finite presentation;
  \item of finite type;
  \item of finite presentation;
  \item proper;
  \item an isomorphism;
  \item a monomorphism;
  \item an open immersion;
  \item a quasi-compact immersion;
  \item a closed immersion;
  \item affine;
  \item quasi-affine;
  \item finite;
  \item quasi-finite;
  \item integral.
\end{enumerate}
Then, if $\mathbf{P}$ denotes one of the preceding properties, for $f$ to have property $\mathbf{P}$, it is necessary and sufficient that $f'$ have it.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{30}
It has been shown in Chapters I, II and in Chapter IV, \textsection1, that if $f$ has one of the properties $\mathbf{P}$ above, the same is true of $f'$ (without hypothesis on the morphism $g:S'\to S$).
It therefore remains to prove the converse; since the projection $Y'\to Y$ is a faithfully flat and quasi-compact morphism \sref{IV.2.2.13} and \sref{IV.1.1.2}, one can reduce to the case where $S=Y$, $S'=Y'$ by virtue of \sref[I]{I.3.3.11}.

(i) To say that $f$ is separated means that the diagonal morphism $\Delta_f:X\to X\times_Y X$ is \emph{closed}; since $\Delta_{f'}=(\Delta_f)_{(Y')}$ \sref[I]{I.5.3.4}, if $\Delta_{f'}$ is closed, the same is true of $\Delta_f$ by virtue of \sref{IV.2.6.2}.

(ii) has already been proved under weaker hypotheses \sref{IV.1.2.5}.

(iii) and (iv): The question is evidently local on $X$ and $Y$, and, taking into account \sref{IV.2.2.12}, it therefore suffices to prove the

\begin{lemma}[2.7.1.1]
\label{IV.2.7.1.1}
Let $A$ be a ring, $B$ an $A$-algebra, $A'$ a faithfully flat $A$-algebra and $B'=B\otimes_A A'$.
For $B$ to be an $A$-algebra of finite type \emph{(}resp. of finite presentation\emph{)}, it is necessary and sufficient that $B'$ be an $A'$-algebra of finite type \emph{(}resp. of finite presentation\emph{)}.
\end{lemma}

We already know that the condition is necessary without hypothesis on $A'$ \sref{IV.1.3.4}, \sref{IV.1.3.6}, \sref{IV.1.4.3} and \sref{IV.1.4.6}.
Suppose that $B'$ is an $A'$-algebra of finite type; let $(B_\alpha)_{\alpha\in I}$ be the increasing filtering family of $A$-sub-algebras of $B$ of finite type, so that $B=\varinjlim B_\alpha$, and consequently $B'=\varinjlim(B_\alpha\otimes_A A')$, the tensor product commuting with inductive limits; if $(x'_i)$ is a finite system of generators of the $A'$-algebra $B'$, there exists an index $\alpha$ such that all the $x'_i$ belong to the sub-algebra $B_\alpha\otimes_A A'$ of $B'$, whence $B'=B_\alpha\otimes_A A'$, and since $A'$ is faithfully flat, $B=B_\alpha$ $(\mathbf{0_I}$, 6.4.1).

Suppose now that $B'$ is an $A'$-algebra of finite presentation; we already know from the preceding that $B$ is an $A$-algebra of finite type, so there exists a surjective homomorphism $C=A[T_1,\ldots,T_n]\to B$ of $A$-algebras and an $A$-homomorphism $C\to B$; let $\mathfrak{J}$ be the kernel of this homomorphism, so that one has an exact sequence $0\to\mathfrak{J}\to C\to B'\to 0$ (since $A'$ is $A$-flat), $\mathfrak{J}'=\mathfrak{J}\otimes_A A'$ (identified with an ideal of $C'$).
Since $B'$ is an $A'$-algebra of finite presentation, $\mathfrak{J}'$ is a $C'$-module of finite type \sref{IV.1.4.4}; but $\mathfrak{J}=\mathfrak{J}\otimes_C C'$ and $C'$ is a faithfully flat $C$-module \sref{IV.2.2.13} and \sref{IV.2.2.3}; we know then that $\mathfrak{J}$ is a $C$-module of finite type (Bourbaki, \emph{Alg.~comm.}, chap.~I, \textsection3, no.~6, prop.~11), hence $B$ is an $A$-algebra of finite presentation.

(v) follows from (iii) and from \sref{IV.2.6.2}~(v) by virtue of \sref{IV.1.5.2}.

(vi) follows similarly from (iv), (v) and (ii) by virtue of \sref{IV.1.6.1}.

(vii) follows from (i), (v) and \sref{IV.2.6.4}~(ii) \sref[II]{II.5.4.1}.

\oldpage[IV-2]{31}
(viii) First note that since $f'$ is an isomorphism, it is a universal homeomorphism, hence the same is true of $f$ \sref{IV.2.6.4}; we conclude already that $f$ is quasi-compact and separated \sref{IV.2.4.4}; it remains to prove that $\theta:\sh{O}_Y\to f_*(\sh{O}_X)$ is an isomorphism of $\sh{O}_Y$-Modules.
Now, if we set $f'=(\psi',\theta')$, $\theta':\sh{O}_{Y'}\to f'_*(\sh{O}_{X'})$ is composed of the canonical homomorphism $g^*(f_*(\sh{O}_X))\to f'_*(\sh{O}_{X'})$ and of $g^*(\theta)$ \sref{IV.2.3.2}; but the first of these two homomorphisms is bijective by virtue of the hypothesis on $g$ \sref{IV.2.3.1}, so if $\theta'$ is bijective, $\theta$ is bijective \sref{IV.2.2.7}, which proves (viii).

(ix) The proposition follows from (viii), from \sref[I]{I.5.3.4} and from \sref[I]{I.5.3.8} which reduces the monomorphisms to isomorphisms.

(x) If $f'$ is an open immersion, $f'(X')$ is open in $Y'$, and we have $f'(X')=g^{-1}(f(X))$ \sref[I]{I.3.4.8}; it follows from \sref{IV.2.3.12} that $f(X)$ is open.
One can then replace $Y$ (resp. $Y'$) by the sub-prescheme induced on the open $f(X)$ (resp. $f'(X')$), taking into account \sref{IV.1.1.2} and \sref{IV.2.2.13}; then $f'$ becomes an isomorphism, so the same is true of $f$ by (viii), and this establishes (x).

(xi) If $f'$ is a quasi-compact immersion, $f'$ is a quasi-compact morphism and quasi-separated \sref[II]{II.5.1.1}, hence the same is true of $f$ by (ii) and \sref{IV.2.6.2}~(v).
Let $Z$ be the closed sub-prescheme of $Y$ closed image of $X$ by $f$ \sref{IV.1.7.8}, and set $f=j\circ g'$ with $j'=j_{(Y')}$, $g'=g_{(Y')}$, and we know that $j'$ is the canonical closed immersion $Z'\to Y'$ of the sub-prescheme $Z'$ of $Y'$, closed image of $X'$ by $f'$ \sref{IV.2.3.2}.

and that $h'\circ u'=f'$, it follows from \sref[II]{II.1.2.7} that $u'=u_{(Z')}$.

This being the case, the hypothesis on $f'$ means that $u'$ is an isomorphism (resp. an open immersion) \sref[II]{II.5.1.6}; it follows then from (viii) (resp. (x)) that $u$ is an isomorphism (resp. an open immersion), whence (xiii) (resp. (xiv)).

\begin{corollary}[2.7.2]
\label{IV.2.7.2}
With the hypotheses and notations being those of \sref{IV.2.7.1}, suppose $f$ quasi-compact; let $\sh{L}$ be a quasi-coherent $\sh{O}_X$-Module inversible, $\sh{L}'=\sh{L}\otimes_{\sh{O}_Y}\sh{O}_{Y'}$, its inverse image.
For $\sh{L}$ to be ample \emph{(}resp. very ample\emph{)} for $f$, it is necessary and sufficient that $\sh{L}'$ be ample \emph{(}resp. very ample\emph{)} for $f'$.

\begin{remarks}[2.7.3]
\label{IV.2.7.3}
(i) It follows from \sref{IV.2.6.1}, \sref{IV.2.6.4} and \sref{IV.2.5.4.1} that the conclusions of \sref{IV.2.7.1} are valid in cases (i), (iii), (v), (vii) and (xvi) when one supposes only that $g$ is quasi-compact and \emph{quasi-faithfully flat} \sref{IV.2.3.3}; we have already remarked that \sref{IV.2.7.1} is valid in case (ii) only assuming $g$ surjective and quasi-compact.

Let then $Y'=Y_1\amalg Y_2$ be the sum prescheme; it is clear that the canonical morphism $g:Y'\to Y$ coinciding with the canonical injections into $Y_1$ and $Y_2$, is faithfully flat and quasi-compact by virtue of \sref[I]{I.5.5.10}; however, although $f':X\times_Y Y'\to Y'$ (coinciding with $f_i$ on each of the $Y_i$) is projective \sref[II]{II.5.5.6}, it is not of the same type as $f$.

\subsection{Preschemes over a regular base of dimension \texorpdfstring{$1$}{1}; closure of a closed subprescheme of the generic fibre}
\label{subsection:IV.2.8}

\begin{proposition}[2.8.1]
\label{IV.2.8.1}
Let $Y$ be a locally Noetherian, regular, irreducible prescheme of dimension $1$, with generic point $\eta$, $f:X\to Y$ a morphism, $X_\eta=f^{-1}(\eta)$ the fibre at the generic point, $i:X_\eta\to X$ the canonical morphism.
Let $\sh{F}$ be a quasi-coherent $\sh{O}_X$-Module, $\sh{F}_\eta=i^*(\sh{F})$, $\sh{G}'$ a quotient $\sh{O}_{X_\eta}$-Module of $\sh{F}_\eta$, and let $\overline{\sh{G}'}$ be the $\sh{O}_X$-Module image of $\sh{F}$ by the composed homomorphism $(\mathbf{0_I}$, 4.4.3.2)
\[
  \sh{F} \xrightarrow{\rho_\sh{F}} i_*(i^*(\sh{F})) \to i_*(\sh{G}').
\]
Then $\overline{\sh{G}'}$ is a quotient $\sh{O}_X$-Module of $\sh{F}$, quasi-coherent and $f$-flat, such that $i^*(\overline{\sh{G}'})=\sh{G}'$, and it is the only quotient $\sh{O}_X$-Module of $\sh{F}$ having these properties.
\end{proposition}

\begin{proof}
\oldpage[IV-2]{34}
Since $i_*$ is quasi-compact and quasi-separated \sref{IV.1.1.2} and \sref{IV.1.2.2}, it follows from \sref{IV.1.7.4} that for every quasi-coherent $\sh{O}_{X_\eta}$-Module $\sh{H}'$, $i_*(\sh{H}')$ is a quasi-coherent $\sh{O}_X$-Module; moreover, for every open subset $U$ of $X$, we have $(i_*(\sh{H}'))|U=(i|U\cap X_\eta)_*(\sh{H}'|U\cap X_\eta)$ by definition $(\mathbf{0_I}$, 3.4.1).
If we prove the proposition when $X$ and $Y$ are affine, it will follow by gluing in the general case, given the uniqueness assertion valid in the affine case.
In other words, we are reduced to proving the following:
\end{proof}

\begin{lemma}[2.8.1.1]
\label{IV.2.8.1.1}
Let $A$ be a regular Noetherian ring $(\mathbf{0_{IV}}$, 17.3.6), integral of dimension~$1$, $K$ its field of fractions, $M$ an $A$-module, $N'$ a quotient $K$-module of $M_{(K)}=M\otimes_A K$ by a sub-$K$-module $P'$, $\overline{N'}$ the image of $M$ by the composed homomorphism $M\to M_{(K)}\to N'$.
Then $\overline{N'}$ is a flat $A$-module, and is the unique quotient module $N$ of $M$ which is a flat $A$-module and such that the kernel of the surjective homomorphism $M_{(K)}\to N_{(K)}$ is equal to $P'$.
\end{lemma}

\begin{proof}
Since for every maximal ideal $\mathfrak{m}$ of $A$, $A_\mathfrak{m}$ is a regular local ring of dimension~$1$, hence a discrete valuation ring, it amounts to the same to say that an $A$-module $N$ is flat or that it is without torsion $(\mathbf{0_I}$, 6.3.4).
Since $N'$ is a $K$-vector space, it is an $A$-module without torsion, hence so is $\overline{N'}$, being a submodule of $N'$; moreover, it is immediate to verify that $\overline{N'}_{(K)}$ identifies with $N'$.
Conversely, if $N$ is a quotient $A$-module of $M$ having the properties stated, the fact that $N$ is a flat $A$-module implies that the canonical homomorphism $N\to N_{(K)}=N\otimes_A K$ is \emph{injective}.
Since $N_{(K)}$ identifies with $N'$, the conclusion follows from the commutativity of the diagram
\[
  \xymatrix{
    M\ar[r]\ar[d] & N\ar[d] \\
    M_{(K)}\ar[r] & N_{(K)}
  }
\]
\end{proof}

\begin{corollary}[2.8.2]
\label{IV.2.8.2}
Under the conditions of \sref{IV.2.8.1}, for $\sh{F}$ to be $f$-flat, it is necessary and sufficient that the canonical homomorphism $\sh{F}\to i_*(\sh{F}_\eta)=i_*(i^*(\sh{F}))$ be injective.
\end{corollary}

\begin{env}[2.8.3]
\label{IV.2.8.3}
The formation of the $\sh{O}_X$-Module $\overline{\sh{G}'}$ is functorial in $\sh{F}$ and $\sh{G}'$: more precisely, if $\sh{F}_1$, $\sh{F}_2$ are two quasi-coherent $\sh{O}_X$-Modules, $u:\sh{F}_1\to\sh{F}_2$ an $\sh{O}_X$-homomorphism, $\sh{G}'_i$ a quotient $\sh{O}_{X_\eta}$-Module of $(\sh{F}_i)_\eta$ ($i=1,2$) and $v:\sh{G}'_1\to\sh{G}'_2$ a homomorphism making the diagram
\[
  \xymatrix{
    (\sh{F}_1)_\eta\ar[r]^{i^*(u)}\ar[d] & (\sh{F}_2)_\eta\ar[d] \\
    \sh{G}'_1\ar[r]^{v} & \sh{G}'_2
  }
\]
commutative (homomorphism determined in a unique way (when it exists) by this property), then the diagram
\[
  \xymatrix{
    \sh{F}_1\ar[r]^{u}\ar[d] & \sh{F}_2\ar[d] \\
    \overline{\sh{G}'_1}\ar[r]^{w} & \overline{\sh{G}'_2}
  }
\]
\oldpage[IV-2]{35}
is commutative, and there is therefore a unique homomorphism $w:\overline{\sh{G}'_1}\to\overline{\sh{G}'_2}$ making the diagram
\[
  \xymatrix{
    \sh{F}_1\ar[r]^{u}\ar[d] & \sh{F}_2\ar[d] \\
    \overline{\sh{G}'_1}\ar[r]^{w} & \overline{\sh{G}'_2}
  }
\]
commutative.
\end{env}

\begin{proposition}[2.8.4]
\label{IV.2.8.4}
The hypotheses on $Y$ being those of \sref{IV.2.8.1}, let $X_1$, $X_2$ be two $Y$-preschemes, $\sh{F}_i$ a quasi-coherent $\sh{O}_{X_i}$-Module, $\sh{G}'_i$ a quotient $\sh{O}_{(X_i)_\eta}$-Module of $(\sh{F}_i)_\eta$ ($i=1,2$).
Then we have
\begin{equation}
\label{IV.2.8.4.1}
  \overline{\sh{G}'_1}\otimes_{\sh{O}_Y}\overline{\sh{G}'_2}=\overline{\sh{G}'_1\otimes_{k(\eta)}\sh{G}'_2}.\tag{2.8.4.1}
\end{equation}
\end{proposition}

\begin{proof}
Indeed, setting $X=X_1\times_Y X_2$, the first member of \sref{IV.2.8.4.1} is a quasi-coherent $\sh{O}_X$-Module which is $Y$-flat $(\mathbf{0_I}$, 6.2.1), whose inverse image in $X_\eta$ is $\sh{G}'_1\otimes_{k(\eta)}\sh{G}'_2$ \sref[I]{I.9.1.5} and which is a quotient of $\sh{F}_1\otimes_{\sh{O}_Y}\sh{F}_2$; the conclusion therefore follows from the uniqueness property of \sref{IV.2.8.1}.
\end{proof}

\begin{proposition}[2.8.5]
\label{IV.2.8.5}
The hypotheses on $X$ and $Y$ being those of \sref{IV.2.8.1}, let $Z'$ be a closed subprescheme of $X_\eta$.
There then exists a unique closed subprescheme $\overline{Z'}$ of $X$ which is $Y$-flat and such that $i^{-1}(\overline{Z'})=Z'$.
\end{proposition}

\begin{proof}
If $\sh{J}'$ is the quasi-coherent Ideal of $\sh{O}_{X_\eta}$ defining $Z'$, it suffices to apply \sref{IV.2.8.1} to the case where $\sh{F}=\sh{O}_X$ and $\sh{G}'=\sh{O}_{X_\eta}/\sh{J}'$; if $\overline{\sh{G}'}=\sh{O}_X/\sh{J}$, we have indeed $\sh{J}'=i^*(\sh{J})\sh{O}_{X_\eta}$, hence $i^{-1}(\overline{Z'})=Z'$ \sref[I]{I.4.4.5}.
\end{proof}

We note that the prescheme $\overline{Z'}$ is the \emph{closed image} of $Z'$ by the composed morphism $Z'\to X_\eta\xrightarrow{i}X$, where the first arrow is the canonical injection \sref[I]{I.9.5.3}; its underlying space is the \emph{closure} in $X$ of $Z'$ \sref[I]{I.9.5.4}, which justifies the notation adopted.
We also say that $\overline{Z'}$ is the subprescheme of $X$ which is the \emph{closure} of $Z'$.

\begin{corollary}[2.8.6]
\label{IV.2.8.6}
Let $X_1$, $X_2$ be two $Y$-preschemes, $Z'_i$ a closed subprescheme of $(X_i)_\eta$ ($i=1,2$).
Then we have
\begin{equation}
\label{IV.2.8.6.1}
  \overline{Z'_1\times_{k(\eta)}Z'_2}=\overline{Z'_1}\times_Y\overline{Z'_2}.\tag{2.8.6.1}
\end{equation}
\end{corollary}

\begin{proof}
This follows from \sref{IV.2.8.4} and \sref{IV.2.8.5}.
\end{proof}
