\section{Smooth morphisms, unramified (or net) morphisms, and \'etale morphisms.}
\label{section:IV.17}

In this paragraph, we revisit the concepts studied in (\textbf{0}\textsubscript{III},~\hyperref[section:0.9]{9}), expressed in the geometric language of schemes from a global point of view, for preschemes locally of finite presentation over a given base.

Most of the results (except~\hyperref[subsection:IV.17.7]{17.7},~\hyperref[subsection:IV.17.8]{17.8},~\hyperref[subsection:IV.17.9]{17.9},~\hyperref[subsection:IV.17.13]{17.13}, and~\hyperref[subsection:IV.17.16]{17.16}) are reduced to various properties already encountered in (\textbf{0}\textsubscript{III},~\hyperref[section:0.9]{9}).

For more specific results on \'etale morphisms, the reader should consult~\textsection\hyperref[section:IV.18]{18}.

\subsection{Formally smooth morphisms, formally unramified morphisms, formally \'etale morphisms.}
\label{subsection:IV.17.1}

\begin{definition}[17.1.1]
\label{IV.17.1.1}
Let $f:X\to Y$ be a morphism of preschemes.
We say that $f$ is \emph{formally smooth} (resp. \emph{formally unramified}, resp. \emph{formally \'etale}) if, for all affine schemes $Y'$, all closed subschemes $Y_0'$ of $Y'$ defined by a nilpotent ideal $\sh{J}$ of $\sh{O}_{Y'}$, and every morphism $Y'\to Y$, the map 
\[
\label{IV.17.1.1.1}
  \Hom_Y(Y',X)\to\Hom_Y(Y_0',X)
  \tag{17.1.1.1}
\]
induced by the canonical map $Y_0'\to Y'$, is \emph{surjective} (resp. \emph{injective}, resp. \emph{bijective}).

One also says that $X$ is \emph{formally smooth} (resp. \emph{formally unramified}, resp. \emph{formally \'etale}) over $Y$.

It is clear that for $f$ to be formally \'etale, it is necessary and sufficient for $f$ to be formally smooth and formally unramified.
\end{definition}

\begin{remark}[17.1.2]
\label{IV.17.1.2}
\medskip\noindent
\begin{enumerate}
  \item[(i)] Suppose that $Y=\Spec(A)$ and $X=\Spec(B)$ are affine, so that $f$ comes from a homomorphism of rings $\vphi:A\to B$. 
    According to \sref[0]{0.19.3.1} and \sref[0]{0.19.10.1}, saying that $f$ is formally smooth (resp. formally unramified, resp. formally \'etale) means that, via $\vphi$, $B$ is a \emph{formally smooth} (resp. \emph{formally unramified}, resp. \emph{formally \'etale}) $A$-algebra, for the \emph{discrete} topologies on $A$ and $B$.
  \item[(ii)] To verify that $f$ is formally smooth (resp. formally unramified, resp. formally \'etale), we can, in Definition~\sref{IV.17.1.1}, restrict to the case where $\sh{J}^2=0$.
  To see this, if $f$ satisfies the corresponding condition of Definition~\sref{IV.17.1.1} in the particular case $\sh{J}^2=0$, and if we have $\sh{J}^n=0$, then we consider the closed subscheme $Y_j'$ of $Y'$ defined by the sheaf of ideals $\sh{J}^{j+1}$ for $0\leq j\leq n-1$, so that $Y_j'$ is a closed subscheme of $Y_{j+1}'$ defined by a square-zero sheaf of ideals;
the hypotheses imply that each of the maps
\[
  \Hom_Y(Y_{j+1}',X)\to\Hom_Y(Y_j',X)\quad(0\leq j\leq n-1) 
\]
\oldpage[IV]{57}
is surjective (resp. injective, resp. bijective);
by composition, we conclude that the same holds for \sref{IV.17.1.1.1}.
\item[(iii)] Note that the properties of the morphism $f$ defined in \sref{IV.17.1.1} are properties of the \emph{representable functor} \sref[\textbf{0}\textsubscript{III}]{0.8.1.8}
\[
  Y'\mapsto\Hom_Y(Y',X) 
\]
from the category of $Y$-preschemes to the category of sets;
they keep a meaning for \emph{any} contravariant functor with the same domain and codomain, representable or not.
\item[(iv)] Assume that the morphism $f$ is formally unramified (resp. formally \'etale);
consider an \emph{arbitrary} $Y$-prescheme $Z$ and a closed subprescheme $Z_0$ of $Z$ defined by a \emph{locally nilpotent} sheaf of ideals $\sh{J}$ of $\sh{O}_Z$. 
Then the map
\[
\label{IV.17.1.2.1}  
  \Hom_Y(Z,X)\to\Hom_Y(Z_0,X)
  \tag{17.1.2.1}
\]
induced by the canonical injection $Z_0\to Z$, is still injective (resp. bijective).
To see this, let $(U_\alpha)$ be an affine open covering of $Z$ such that the sheaves of ideals $\sh{J}|U_\alpha$ are nilpotent, and for each $\alpha$, let $U_\alpha^0$ be the inverse image of $U_\alpha$ in $Z_0$, which is the closed subprescheme of $U_\alpha$ defined by $\sh{J}|U_\alpha$.
Let $f_0:Z_0\to X$ by a $Y$-morphism;
by hypothesis, for each $\alpha$, there is at most one (resp. one and only one) $Y$-morphism $f_\alpha:U_\alpha\to X$ whose restriction to $Z_0$ coincides with $f_0|U_\alpha$.
We immediately conclude that if $f_\alpha$ and $f_\beta$ are defined, then, for each affine open $V\subset U_\alpha\cap U_\beta$, we have $f_\alpha|V=f_\beta|V$, as the restrictions of these morphisms to the inverse image $V_0$ of $V$ in $Z_0$ coincide.
There is therefore at most one (resp. one and only one) $Y$-morphism $f:Z\to X$ whose restriction to $Z_0$ coincides with $f_0$.
\end{enumerate}
\end{remark}

\begin{proposition}[17.1.3]
\label{IV.17.1.3}
\medskip\noindent
\begin{enumerate}
  \item[{\rm(i)}] A monomorphism of preschemes is formally unramified;
    an open immersion is formally \'etale.
  \item[{\rm(ii)}] The composition of two formally smooth (resp. formally unramified, resp. formally \'etale) morphisms is formally smooth (resp. formally unramified, resp. formally \'etale).
  \item[{\rm(iii)}] If $f:X\to Y$ is a formally smooth (resp. formally unramified, resp. formally \'etale) $S$-morphism, then so is $f_{(S')}:X_{(S')}\to Y_{(S')}$ for any base extension $S'\to S$.
  \item[{\rm(iv)}] If $f:X\to X'$ and $g:Y\to Y'$ are two formally smooth (resp. formally unramified, resp. formally \'etale) $S$-morphisms, then so is $f\times_S g:X\times_S Y\to X'\times_S Y'$.
  \item[{\rm(v)}] Let $f:X\to Y$ and $g:Y\to Z$ be two morphisms;
  if $g\circ f$ is formally unramified, then so is $f$.
  \item[{\rm(vi)}] If $f:X\to Y$ is a formally unramified morphism, then so is $f_\red:X_\red\to Y_\red$.
\end{enumerate}
\end{proposition}

\begin{proof}
According to \sref[I]{I.5.5.12}, it suffices to prove (i), (ii), and (iii). 
The assertions in (i) are both trivial.
To prove (ii), consider two morphisms $f:X\to Y$, $g:Y\to Z$, an affine scheme $Z'$, a closed subscheme $Z_0'$ of $Z$ defined by a nilpotent ideal and a morphism $Z'\to Z$. 
Suppose that $f$ and $g$ formally smooth, and consider a $Z$-morphism
\oldpage[IV]{58}
$u_0:Z_0'\to X$;
the hypothesis  on $g$ implies that there exists a $Z$-morphism $v:Z'\to Y$ such that $f\circ u_0=v\circ j$ (where $j:Z_0'\to Z$ is the canonical injection);
the hypothesis on $f$ then implies that there exists a morphism $u:Z'\to X$ such that $f\circ u=v$ and $u\circ j=u_0$, therefore $(g\circ f)\circ u$ is equal to the given morphism $Z'\to Z$ and $u\circ j=u_0$, which proves that $g\circ f$ is formally smooth;
we argue the same way when we suppose that $f$ and $g$ are formally unramified.

Finally, to prove (iii), let $X'=X_{S'}$, $Y'=Y_{S'}$, $f'=f_{S'}$;
consider an affine scheme $Y''$, a closed subscheme $Y_0''$ defined by a nilpotent sheaf of ideals, and a morphism $g:Y''\to Y'$ making $Y''$ a $Y'$-prescheme;
we then know by \sref[I]{I.3.3.8} that $\Hom_{Y'}(Y'',X')$ is canonically identified with $\Hom_Y(Y'',X)$, and $\Hom_{Y'}(Y_0'',X')$ with $\Hom_Y(Y_0'',X)$, and the conclusion follows immediately from Definition~\sref{IV.17.1.1}.
\end{proof}

We note that a \emph{closed immersion} is not necessarily formally smooth.
\begin{proposition}[17.1.4]
\label{IV.17.1.4}
Let $f:X\to Y$ and $g:Y\to Z$ be two morphisms, and suppose that $g$ is formally unramified.
Then, if $g\circ f$ is formally smooth (resp. formally \'etale), so is $f$.
\end{proposition}

\begin{proof}
Let $Y'$ be an affine scheme, $Y_0'$ a closed subscheme of $Y'$ defined by a nilpotent sheaf of ideals, $h:Y'\to Y$ a morphism, $j:Y_0'\to Y'$ the canonical injection, $u_0:Y_0'\to Y$ a $Y$-morphism, such that $f\circ u_0=h\circ j$. 
Suppose that $g\circ f$ is formally smooth;
then there exists a morphism $u:Y'\to X$ such that $u\circ j=u_0$ and $(g\circ f)\circ u=g\circ h$. 
But these two relations imply that $f\circ u$ and $h$ are $Z$-morphisms from $Y'$ to $Y$ such that $(f\circ u)\circ j=h\circ j$;
by virtue of the hypothesis that $g$ is formally unramified, we get that $f\circ u=h$, in other words that $u$ is a $Y$-morphism;
thus $f$ is formally smooth.
Taking into account \sref{IV.17.1.3}[(v)], this proves the proposition.
\end{proof}

\begin{corollary}[17.1.5]
\label{IV.17.1.5}
Suppose that $g$ is formally \'etale;
then, for $g\circ f$ to be formally smooth (resp. formally unramified, resp. formally \'etale), it is necessary and sufficient that $f$ is.
\end{corollary}

\begin{proof}
This follows from \sref{IV.17.1.4} and \sref{IV.17.1.3}[(ii) and (iv)].
\end{proof}

\begin{proposition}[17.1.6]
\label{IV.17.1.6}
Let $f:X\to Y$ be a morphism of preschemes.
\begin{enumerate}
  \item[{\rm(i)}] Let $(U_\alpha)$ be an open covering of $X$ and, for each $\alpha$, let $i_\alpha:U_\alpha\to X$ be the canonical injection.
    For $f$ to be formally smooth (resp. formally unramified, resp. formally \'etale), it is necessary and sufficient that each $f\circ i_\alpha$ is.
  \item[{\rm(ii)}] Let $(V_\lambda)$ be an open covering of $Y$.
    For $f$ to be formally smooth (resp. formally unramified, resp. formally \'etale), it is necessary and sufficient that each of the restrictions $f^{-1}(V_\lambda)\to V_\lambda$ of $f$ is.
\end{enumerate}
\end{proposition}

\begin{proof}
First note that (ii) is a consequence of (i): if $j_\lambda:V_\lambda\to Y$ and $i_\lambda:f^{-1}(V_\lambda)\to X$ are the canonical injections, then the restriction $f_\lambda:f^{-1}(V_\lambda)\to V_\lambda$ of $f$ is such that $j_\lambda\circ f_\lambda=f\circ i_\lambda$;
if $f$ is formally smooth (resp. formally unramified), then so is $f\circ i_\lambda$ since $i_\lambda$ is formally \'etale \sref{IV.17.1.3};
but since $j_\lambda$ is formally \'etale, this means that $f_\lambda$ is formally smooth (resp. formally unramified), by virtue of \sref{IV.17.1.5}.
Conversely, if all the $f_\lambda$ are formally smooth (resp. formally unramified), the same applies to $j_\lambda\circ f_\lambda$ \sref{IV.17.1.3}, so also to $f$ in virtue of (i).

\oldpage[IV]{59}
If we take into account that the $i_\alpha$ are formally \'etale, everything comes down to proving that if the $f\circ i_\alpha$ are formally smooth (resp. formally unramified), then the same applies to $f$.

Therefore let $Y'$ be an affine scheme, $Y_0'$ a closed subscheme of $Y'$ defined by a nilpotent ideal $\sh{J}$, which we may assume to satisfy $\sh{J}^2=0$ \sref{IV.17.1.2}[(ii)], and finally let $g:Y'\to Y$ be a morphism. 
Suppose we are given a $Y$-morphism $u_0:Y_0'\to X$;
denote by $W_\alpha$ (resp. $W_\alpha^0$) the prescheme induced by $Y'$ (resp. $Y_0'$) on the open subset $u_0^{-1}(U_\alpha)$ (we recall that $Y'$ and $Y_0'$ share the \emph{same underlying topological space}). 
Let us first suppose that the $f\circ i_\alpha$ are \emph{formally unramified}, and show that, if $u'$ and $u''$ are two $Y$-morphisms from $Y'$ to $X$ whose restrictions to $Y_0'$ coincide, then we have $u'=u''$. 
Indeed, taking into account \sref{IV.17.1.2}[(iv)], the hypothesis that the $f\circ i_\alpha$ are formally unramified implies that for all $\alpha$, we have $u'|W_\alpha=u''|W_\alpha$, since the restrictions of both $Y$-morphisms to $W_\alpha^0$ coincide. 
Hence the conclusion follows.

Now suppose that the $f\circ i_\alpha$ are \emph{formally smooth} and prove the existence of a $Y$-morphism $u:Y'\to X$ whose restriction to $Y_0'$ is $u_0$.
Now, since $Y'$ is an \emph{affine scheme}, we can apply \sref{IV.16.5.17}, the hypotheses of which are satisfied, and the conclusion of which precisely proves the existence of $u$.
\end{proof}

We can therefore say that the notions introduced in \sref{IV.17.1.1} are \emph{local} on $X$ and $Y$, which always allows, in virtue of \sref{IV.17.1.2}[(i)], to be reduced to the study of formally smooth (resp. formally unramified, resp. formally \'etale) \emph{algebras}.

\subsection{General properties of differentials}
\label{subsection:IV.17.2}

\begin{proposition}[17.2.1]
\label{IV.17.2.1}
For a morphism $f:X\to Y$ to be formally unramified, it is necessary and sufficient that $\Omega_f^1=0$ (what we still write $\Omega_{X/Y}^1=0$ \sref{IV.16.3.1}).  
\end{proposition}

\begin{proof}
Taking into account \sref{IV.17.1.6}, we reduce to the case where $Y=\Spec(A)$ and $X=\Spec(B)$ are affine, and the conclusion then follows from \sref[0]{0.20.7.4} and the interpretation of $\Omega_{X/Y}^1$ in this case \sref{IV.16.3.7}.
\end{proof}

\begin{corollary}[17.2.2]
\label{IV.17.2.2}
Let $f:X\to Y$ and $g:Y\to Z$ be two morphisms. 
For $f$ to be formally unramified, it is necessary and sufficient that the canonical morphism \sref{IV.16.4.19}
\[
  f^*(\Omega^1_{Y/Z})\to \Omega^1_{X/Z}
\] is surjective.
\end{corollary}

\begin{proof}
This is an immediate consequence of \sref{IV.17.2.1} and the exact sequence \sref{IV.16.4.19.1}.
\end{proof}

\begin{proposition}[17.2.3]
\label{IV.17.2.3}
Let $f:X\to Y$ be a formally smooth morphism.
\begin{enumerate}
  \item[{\rm(i)}] The $\sh{O}_X$-module $\Omega_{X/Y}^1$ is locally projective \sref{IV.16.10.1}.
    If $f$ is locally of finite type, then $\Omega_{X/Y}^1$ is locally free and of finite type.
  \item[{\rm(ii)}] For all morphisms $g:Y\to Z$, the sequence \sref{IV.16.4.19} of $\sh{O}_X$-modules
\[
\label{IV.17.2.3.1}
  0\to f^*(\Omega_{Y/Z}^1)\to\Omega_{X/Z}^1\to\Omega_{X/Y}^1\to 0
  \tag{17.2.3.1}
\]
is exact; moreover, for each $x\in X$, there exists an open neighbourhood $U$ of $x$ such that the restrictions to $U$ of the homomorphisms in \sref{IV.17.2.3.1} form a \emph{split} exact sequence.
\end{enumerate}
\end{proposition}

\oldpage[IV]{60}
\begin{proof}
\medskip\noindent
\begin{enumerate}
  \item[(i)] We know \sref{IV.16.3.9} that if $f$ is locally of finite type, then $\Omega_f^1$ is an $\sh{O}_X$-module of finite type. 
    To prove that, in all cases, it is locally projective, we can reduce, by virtue of \sref{IV.17.1.6}, to the case where $Y=\Spec(A)$ and $X=\Spec(B)$ are affine, and the result follows from the hypothesis on $f$ and from \sref[0]{0.20.4.9} and \sref[0]{0.19.2.1}.
  \item[(ii)] Again, we can restrict to the case where $X$, $Y$, and $Z$ are affine \sref{IV.17.1.6}, and the conclusion in this case follows from the interpretation of the sheaves of modules in the sequence \sref{IV.17.2.3.1} and from \sref[0]{0.20.5.7}.
\end{enumerate}
\end{proof}

\begin{corollary}[17.2.4]
\label{IV.17.2.4}
If $f:X\to Y$ is formally \'etale, then, for all morphisms $g:Y\to Z$, the canonical homomorphism of $\sh{O}_X$-modules
\[
  f^*(\Omega_{Y/Z}^1)\to\Omega_{X/Z}^1
\]
is bijective.
\end{corollary}

\begin{proof}
This follows from the exactness of the sequence \sref{IV.17.2.3.1} and from the fact that we then have $\Omega_{X/Y}^1=0$ \sref{IV.17.2.1}.
\end{proof}

\begin{proposition}[17.2.5]
\label{IV.17.2.5}
Let $f:X\to Y$ be a morphism, $X'$ a subprescheme of $X$ such that the composite morphism $X'\xrightarrow{j}X\xrightarrow{f}Y$ (where $j$ is the canonical injection) is formally smooth.
Then the sequence of $\sh{O}_X$-modules \sref{IV.16.4.21}
\[
\label{IV.17.2.5.1}
  0\to\sh{N}_{X'/X}\to\Omega_{X/Y}^1\otimes_{\sh{O}_X}\sh{O}_{X'}\to\Omega_{X'/Y}^1\to 0
\tag{17.2.5.1}
\]
is exact; moreover, for each $x\in X$, there exists an open neighbourhood $U$ of $x$ such that the restrictions to $U$ of the homomorphisms in \sref{IV.17.2.5.1} form a \emph{split} exact sequence.
\end{proposition}

\begin{proof}
By virtue of \sref{IV.17.1.6}, we reduce to the case where $Y=\Spec(A)$ and $X=\Spec(B)$ are affine, and $X'=\Spec(B/\mathfrak{J})$, where $\mathfrak{J}$ is an ideal of $B$. 
The conormal sheaf $\sh{N}_{X'/X}$ then corresponds to the $B$-module $\mathfrak{J}/\mathfrak{J}^2$ \sref{IV.16.1.3}, and the conclusion follows from \sref[0]{0.20.5.14}.
\end{proof}

\begin{proposition}[17.2.6]
\label{IV.17.2.6}
Let $X$ and $Y$ be two preschemes, $f:X\to Y$ a morphism locally of finite type. 
The following conditions are equivalent:
\begin{enumerate}
  \item[{\rm(a)}] $f$ is a monomorphism.
  \item[{\rm(b)}] $f$ is radicial and formally unramified.
  \item[{\rm(c)}] For each $y\in Y$, the fibre $f^{-1}(y)$ is empty or $\kres(y)$-isomorphic to $\Spec(\kres(y))$ (in other words, it is reduced to a single point $z$ such that $\kres(y)\to\sh{O}_z/\mathfrak{m}_y\sh{O}_z$ is an isomorphism).
\end{enumerate}
\end{proposition}

\begin{proof}
The fact that (a) implies (c) follows from \sref{IV.8.11.5.1}. 
It is clear that (c) implies that $f$ is radicial;
let us prove that it also follows from (c) that $\Omega_{X/Y}^1=0$, which will prove that (c) implies (b) \sref{IV.17.2.1}.
Note that the $\sh{O}_X$-module $\Omega_{X/Y}^1$ is quasi-coherent of finite type \sref{IV.16.3.9}.
It follows from \sref[I]{I.9.1.13.1} that, for $(\Omega_{X/Y}^1)_x=0$, it is necessary and sufficient that if we set $Y_1=\Spec(\kres(y))$, $X_1=f^{-1}(y)=X\times_Y Y_1$, then we have $(\Omega_{X_1/Y_1}^1)_x=0$;
but as the morphism $f_1:X_1\to Y_1$ induced by $f$ is formally unramified by virtue of the hypothesis (c) \sref{IV.17.1.3}, the conclusion follows from \sref{IV.17.2.1}.
Finally, let us prove that (b) implies (a);
for this, consider the diagonal morphism $g=\Delta_f:X\to X\times_Y X$;
since $f$ is radicial, $g$ is surjective \sref{IV.1.8.7.1};
on the other hand, $\Omega_{X/Y}^1$ is by definition the conormal sheaf $\shGr_1(g)$ of the immersion $g$ \sref{IV.16.3.1}, and to say that $f$ is formally unramified therefore means that
\oldpage[IV]{61}
$\shGr_1(g)=0$ \sref{IV.17.2.1}. 
In addition, $g$ is locally of finite presentation \sref{IV.1.4.3.1};
therefore the hypothesis $\shGr_1(g)=0$ implies that $g$ is an open immersion \sref{IV.16.1.10};
being surjective, this immersion is an isomorphism, hence $f$ is a monomorphism \sref[I]{I.5.3.8}.
\end{proof}

\subsection{Smooth morphisms, unramified morphisms, \'etale morphisms}
\label{subsection:IV.17.3}

\begin{definition}[17.3.1]
\label{IV.17.3.1}
We say that a morphism $f:X\to Y$ is \emph{smooth} (resp. \emph{unramified}, or \emph{net}
\footnote{The words ``net'' and ``formally net'' seem more preferable to the terminology used in ``unramified'' (resp. formally unramified'') and will be used almost exclusively in Chapter~V.
In this chapter, we have kept the old terminology so as not to conflict with \hyperref[subsection:IV.19.10]{\textbf{0}, 19.10}.}
resp. \emph{\'etale})
if it is locally of finite presentation and formally smooth (resp. formally unramified, resp. formally \'etale).
\end{definition}

We then also say that $X$ is \emph{smooth} (resp. \emph{unramified}, resp. \emph{\'etale}) \emph{over $Y$}.

We will see later \sref{IV.17.5.2} that this definition of a smooth morphism coincides with the definition already given in \sref{IV.6.8.1};
until then, we will exclusively use definition \sref{IV.17.3.1}.

It is clear that saying that $f$ is \'etale means that it is \emph{both} smooth and unramified.

\begin{remark}[17.3.2]
\label{IV.17.3.2}
\medskip\noindent
\begin{enumerate}
  \item[(i)] Note that definition \sref{IV.17.3.1} can be phrased using only the functor
    \[
      Y'\mapsto\Hom_Y(Y',X)
    \]
    considered in \sref{IV.17.1.2}[(iii)] because to say that $f$ is locally of finite presentation is equivalent to saying that the preceding functor \emph{commutes with projective limits of affine schemes} \sref{IV.8.14.2}.
  \item[(ii)] Let $A$ be a ring and $B$ an $A$-algebra. 
   We say that $B$ is a \emph{smooth} (resp. \emph{unramified}, resp. \emph{\'etale}) $A$-algebra if the corresponding morphism $\Spec(B)\to\Spec(A)$ is smooth (resp. unramified, resp. \'etale).
   It is equivalent to saying that $B$ is an $A$-algebra \emph{of finite presentation} \sref{IV.1.4.6} that is furthermore formally smooth (resp. formally unramified, resp. formally \'etale) for the discrete topologies.
  \item[(iii)] It follows from \sref{IV.17.1.6} and the definition of a morphism locally of finite presentation \sref{IV.1.4.2} that the notion of a smooth (resp. unramified, resp. \'etale) morphism is \emph{local on $X$ and on $Y$}.
\end{enumerate}
\end{remark}

\begin{proposition}[17.3.3]
\label{IV.17.3.3}
\medskip\noindent
\begin{enumerate}
  \item[{\rm(i)}] An open immersion is \'etale.
    For an immersion to be unramified, it is necessary and sufficient for it to be locally of finite presentation.
  \item[{\rm(ii)}] The composition of two smooth (resp. unramified, resp. \'etale) morphisms is smooth (resp. unramified, resp. \'etale).
  \item[{\rm(iii)}] If $f:X\to Y$ is a smooth (resp. unramified, resp. \'etale) $S$-morphism, then so is $f_{(S')}:X_{(S')}\to Y_{(S')}$ for any base extension $S'\to S$. 
  \item[{\rm(iv)}] If $f:X\to X'$ and $g:Y\to Y'$ are smooth (resp. unramified, resp. \'etale) $S$-morphisms, then so is $f\times_S g:X\times_S Y\to X'\times_S Y'$.
\oldpage[IV]{62}
  \item[{\rm(v)}] Let $f:X\to Y$ and $g:Y\to Z$ be two morphisms;
    if $g$ is locally of finite type and if $g\circ f$ is unramified, then $f$ is unramified.
\end{enumerate}
\end{proposition}

\begin{proof}
This follows from \sref{IV.1.4.3} and \sref{IV.17.1.3}.
\end{proof}

\begin{proposition}[17.3.4]
\label{IV.17.3.4}
Let $f:X\to Y$ and $g:Y\to Z$ be two morphisms, and suppose that $g$ is unramified.
Then, if $g\circ f$ is smooth (resp. unramified, resp. \'etale), so is $f$.
\end{proposition}

\begin{proof}
As $g$ and $g\circ f$ are locally of finite presentation, so is $f$ \sref{IV.1.4.3}[(v)];
the conclusion thus follows from \sref{IV.17.1.4} and \sref{IV.17.1.3}[(v)].
\end{proof}

\begin{corollary}[17.3.5]
\label{IV.17.3.5}
Suppose that $g$ is \'etale;
then, for $f$ to be smooth (resp. unramified, resp. \'etale) it is necessary and sufficient that $g\circ f$ is.
\end{corollary}

\begin{proof}
This follows from \sref{IV.17.3.4} and \sref{IV.17.3.3}[(ii)].
\end{proof}

\begin{proposition}[17.3.6]
\label{IV.17.3.6}
Let $g:Y\to S$ and $h:X\to S$ be two morphisms locally of finite presentation.
For an $S$-morphism $f:X\to Y$ to be unramified, it is necessary and sufficient that the canonical homomorphism \sref{IV.16.4.19}
\[
  f^*(\Omega_{Y/S}^1)\to\Omega_{X/S}^1
\]
is surjective.
\end{proposition}

\begin{proof}
As $f$ is locally of finite presentation \sref{IV.1.4.3}[(v)], the proposition follows from \sref{IV.17.2.2}.
\end{proof}

\begin{definition}[17.3.7]
\label{IV.17.3.7}
Let $f:X\to Y$ be a morphism.
We say that $f$ is \emph{smooth} (resp. \emph{unramified}, resp. \emph{\'etale}) at a point $x\in X$, if there exists an open neighbourhood $U$ of $x$ in $X$ such that the restriction $f|U$ is a smooth (resp. unramified, resp. \'etale) morphism from $U$ to $Y$.
\end{definition}

We then also say that $X$ is \emph{smooth} (resp. \emph{unramified}, resp. \emph{\'etale}) \emph{over $Y$ at the point $x$}.

Taking into account remark \sref{IV.17.3.2}[(iii)], it is equivalent to saying that $f$ is smooth (resp. unramified, resp. \'etale) and to say that $f$ is smooth (resp. unramified, resp. \'etale) at all points of $X$.

It is clear that the set of points of $X$ at which the morphism $f:X\to Y$ is smooth (resp. unramified, resp. \'etale) is \emph{open} in $X$.

\begin{proposition}[17.3.8]
\label{IV.17.3.8}
For all preschemes $Y$ and all locally free $\sh{O}_Y$-modules $\sh{E}$ of finite type, the vector bundle prescheme $\bb{V}(\sh{E})$ \sref[II]{II.1.7.8} associated to $\sh{E}$ is a smooth $Y$-prescheme.
\end{proposition}

\begin{proof}
Indeed \sref{IV.17.3.2}[(iii)], we can restrict ourselves to the case where $Y=\Spec(A)$ is affine and $\bb{V}(\sh{E})=\Spec(A[T_1,\dots,T_r])$;
as $A[T_1,\dots,T_r]$ is a formally smooth $A$-algebra for the discrete topologies \sref[0]{0.19.3.2}, and of finite presentation, this proves the proposition \sref{IV.17.3.2}[(ii)] 
\end{proof}

\begin{corollary}[17.3.9]
\label{IV.17.3.9}
Under the hypotheses of \sref{IV.17.3.8}, the projective prescheme $\bb{P}(\sh{E})$ \sref[II]{II.4.1.1} is a smooth $Y$-prescheme.
\end{corollary}

\begin{proof}
We can still restrict to the case where $Y=\Spec(A)$ is affine and $\bb{P}(\sh{E})=\bb{P}_Y^r$.
We then know \sref[II]{II.2.3.14} that we have a finite open cover of $\bb{P}^r_{A}$ by the $D_+(T_i)$ ($0\leq i\leq r$) respectively equal to the spectrum of the ring $S_{(f)}$, where we wrote $S$ for $A[T_1,\dots,T_r]$ and $f$ for $T_i$;
but it follows immediately from the definition of $S_{(f)}$ \sref[II]{II.2.2.1}, that this ring, in this case, is isomorphic to $A[T_0,\dots,T_{i-1},T_{i+1},\dots,T_r]$;
hence the corollary follows by \sref{IV.17.3.8}.
\end{proof}

\subsection{Characterizations of unramified morphisms.}
\label{subsection:IV.17.4}

\oldpage[IV]{63}

\begin{theorem}[17.4.1]
\label{IV.17.4.1}
Let $f:X\to Y$ be a morphism locally of finite presentation, $x$ a point of $X$.
The following properties are equivalent:
\begin{enumerate}[label=(\alph*)]
  \item $f$ is non ramified at the point $x$.
  \item The diagonal morphism $\Delta_f:X\to X\times_Y X$ is an isomorphism local at the point $x$.
  \item[b')] If one sets $\Delta_f=(\psi,\theta)$, $Z=X\times_Y X$ and $z=\psi(x)$, the homomorphism $\theta_z^\#:\sh{O}_{Z,z}\to\sh{O}_{X,x}$ is bijective.
  \item[b'')] For every morphism $g:Y'\to Y$, and every point $y'\in Y'$ above $y=f(x)$, every $Y'$-section $s'$ of $X'=X\times_Y Y'$ such that $x'=s'(y')$ is above $x$ is an isomorphism local at the point $y'$.
  \item $(\Omega^1_{X/Y})_x=0$.
  \item The $k(y)$-prescheme $f^{-1}(y)$ is non ramified on $k(y)$ at the point $x$.
  \item[d')] The point $x$ is isolated in $X_y=f^{-1}(y)$ (equivalently (\textbf{Err}$_\textbf{III}$,~20), the morphism $f$ is quasi-finite at the point $x$) and the ring $\sh{O}_{X_y,x}$ is a field, extension finite and separable of $k(y)$.
  \item[d'')] The ring $\sh{O}_{X_y,x}=\sh{O}_{X,x}/\mathfrak{m}_y\sh{O}_{X,x}$ is a field, extension finite separable of $k(y)$.
  \item The ring $\sh{O}_{X,x}$ is an $\sh{O}_{Y,y}$-algebra formally non ramified for the discrete topologies.
\end{enumerate}
\end{theorem}

\begin{proof}
Since $f$ is locally of finite type, the $\sh{O}_X$-Module $\Omega^1_{X/Y}$ is of finite type (16.3.9), hence it amounts to the same to say that $(\Omega^1_{X/Y})_x=0$ or that there exists a neighbourhood open $U$ of $x$ such that $\Omega^1_{X/Y}|U=0$.
This proves the equivalence of $a)$ and $c)$.
On the other hand, if one sets $A=\sh{O}_{Y,y}$, $B=\sh{O}_{X,x}$, then $(\Omega^1_{X/Y})_x=\Omega^1_{B/A}$ (16.4.15), and the equivalence of $c)$ and $e)$ follows from ($\mathbf{0}$,~20.7.4).

Since $d')$ only involves properties of the morphism $X_y\to\Spec(k(y))$, the equivalence of $a)$ and $d')$ will imply that of $d)$ and $d')$.
On the other hand, $d')$ and $d'')$ are equivalent, since it amounts to the same to say that $\sh{O}_{X_y,x}$ is a $k(y)$-algebra \emph{finite}, or that $x$ is an isolated point of $X_y$, since $X_y$ is a $k(y)$-prescheme locally of finite type (\sref[I]{I.6.4.4}).

We now prove the equivalence of $b)$ and $b')$.
We may restrict to the case where $Y=\Spec(R)$ and $X=\Spec(S)$ are affine and $f$ of finite presentation; then $Z=\Spec(S\otimes_R S)$ and $\Delta_f$ corresponds to the canonical surjective homomorphism $S\otimes_R S\to S$, and the hypothesis that $\theta_z^\#:\sh{O}_{Z,z}\to\sh{O}_{X,x}$ is bijective implies that, replacing $X$ by a suitable open neighbourhood of $x$, the homomorphism $\theta:\sh{O}_Z\to\psi_*(\sh{O}_X)$ is itself bijective ($\mathbf{0}_\mathbf{I}$,~5.2.7).
This shows that $b')$ implies $b)$; the converse is obvious.

On the other hand, the equivalence of $b)$ and $b'')$ follows from (\sref[I]{I.5.3.7}) without hypothesis of finitude on $f$: the data of a $Y'$-section $s':Y'\to X'$ is equivalent to that of a $Y$-morphism $h=g'\circ s':Y'\to X$ (where $g':X'\to X$ is the canonical projection), so that $s'=(1_{Y'},h)_X$, and then the diagram
\[
\tag{17.4.1.1}
\xymatrix{
Y' \ar[r]^-{s'} \ar[d]_h & X'=Y'\times_Y X \ar[d]^{h\times 1_X}\\
X \ar[r]_-{\Delta_f} & X\times_Y X
}
\]
identifies $Y'$ with the $(X\times_Y X)$-prescheme product of $X$ and $X'$.
Hence (\sref[I]{I.4.3.2}) if $\Delta_f$ is an isomorphism local at the point $x$, $s'$ is an isomorphism local at the point $y'$ (since $x=h(y')$), which proves that $b)$ implies $b'')$.
The converse is obtained by applying $b'')$ to the case $Y'=X$, $y'=x$, $g=f$, and $s'=\Delta_f$.

To finish the proof of (17.4.1), it suffices to prove the implications $d'')\Rightarrow c)\Rightarrow b)\Rightarrow d'')$.

$d'')\Rightarrow c)$: Since $\Omega^1_{X/Y}$ is an $\sh{O}_X$-Module of finite type, it follows from the lemma of Nakayama that condition $c)$ is equivalent to $(\Omega^1_{X/Y})_x/\mathfrak{m}_y(\Omega^1_{X/Y})_x=0$, that is to say (16.4.5) to $(\Omega^1_{X_y/\Spec(k(y))})_x=0$.
One is thus reduced to the case where $Y$ is the spectrum of a field $k$ and $X$ a $k$-prescheme algebraic.
The hypothesis that $\sh{O}_{X,x}$ is a field $k'$, finite extension of $k$, implies first that $x$ is a \emph{closed} point in $X$ (\sref[I]{I.6.4.2}), then that $x$ is a \emph{maximal} point, hence is an isolated point of $X$.
Replacing $X$ by the open $\{x\}$ of $X$, one may suppose $X=\Spec(k')$; but then the hypothesis that $k'$ is a finite separable extension of $k$ implies $\Omega^1_{k'/k}=0$ ($\mathbf{0}$,~20.6.20), which proves $c)$.

$c)\Rightarrow b)$: We saw above that one then has $\Omega^1_{X/Y}|U=0$ for a neighbourhood open $U$ of $x$ in $X$; the assertion $b)$ follows then from the definition of $\Omega^1_{X/Y}$ (16.3.1) and (16.1.9).

$b)\Rightarrow d'')$: Replacing $X$ by a neighbourhood open of $x$, one may suppose that $\Delta_f$ is an open immersion; if one denotes by $f_y:X_y\to\Spec(k(y))$ the morphism deduced by base change, $\Delta_{f_y}$ is then also an open immersion (\sref[I]{I.5.3.4}), and since condition $d'')$ only concerns the prescheme $X_y$, one can reduce to the case where $Y$ is the spectrum of a field $k$, $X$ the spectrum of a $k$-algebra $A$ of finite type; the property to be established is that $A$ is a $k$-algebra \emph{finite and separable}; such an algebra is a composed direct of finite separable extensions of $k$.
Let $K$ be an algebraically closed extension of $k$.
Since $k(x)=k$ since $k$ is algebraically closed, there is a $Y$-section $s$ of $X$ such that $s(Y)=\{x\}$, and by (17.4.1), $\{x\}$ is the inverse image of the diagonal $\Delta_X(X)$ by a morphism $X\to X\times_Y X$, hence $\{x\}$ is open in $X$ by the hypothesis $b)$.
One has thus shown that $A$ is a \emph{finite} $k$-algebra.
To express that $\Delta_f$ is an open immersion, one can thus reduce to the case where $A$ is a \emph{local} finite $k$-algebra; the residue field of $A$, being a finite extension of $k$, is necessarily identical to $k$, and hence (\sref[I]{I.3.4.9}) $X\times_k X$ is reduced to a single point and $\Delta_f$ is necessarily an \emph{isomorphism}.
But since $A$ is a $k$-algebra, the canonical homomorphism $A\otimes_k A\to A$ can only be bijective if $A=k$.
\end{proof}

\begin{remark}[17.4.2]
\label{IV.17.4.2}
Suppose only that $f$ is \emph{locally of finite type}.
Then $\Omega^1_{X/Y}$ is still an $\sh{O}_X$-Module of finite type (16.3.9), and $\Delta_f$ a locally of finite presentation immersion (1.4.3.1).
The entire demonstration of (17.4.1) remains valid, subject to replacing $a)$ by: \emph{the restriction of $f$ to a suitable neighbourhood of $x$ is a morphism formally non ramified}.
One sees moreover that in this case the restriction of $f$ to a suitable neighbourhood of $x$ is a morphism \emph{locally quasi-finite}.
\end{remark}

\begin{corollary}[17.4.2]
\label{IV.17.4.2.cor}
Let $f:X\to Y$ be a morphism locally of finite presentation.
The following properties are equivalent:
\begin{enumerate}[label=(\alph*)]
  \item $f$ is non ramified.
  \item The diagonal morphism $\Delta_f:X\to X\times_Y X$ is an open immersion.
  \item[b')] For every morphism $Y'\to Y$, every $Y'$-section of $X'=X\times_Y Y'$ is an open immersion.
  \item $\Omega^1_{X/Y}=0$.
  \item For every $y\in Y$, the $k(y)$-prescheme $f^{-1}(y)$ is non ramified on $k(y)$.
  \item[d')] For every $y\in Y$, the $k(y)$-prescheme $f^{-1}(y)$ is isomorphic to a prescheme of the form $\coprod_{\lambda\in L}\Spec(K_\lambda)$, where for every $\lambda\in L$, $K_\lambda$ is a finite extension finite and separable of $k(y)$.
  \item For every $x\in X$, the ring $\sh{O}_{X,x}$ is an $\sh{O}_{Y,f(x)}$-algebra formally non ramified for the discrete topologies.
\end{enumerate}
\end{corollary}

\begin{corollary}[17.4.3]
\label{IV.17.4.3}
If $f:X\to Y$ is non ramified, then $f$ is locally quasi-finite (\textbf{Err}$_\textbf{III}$,~20).
\end{corollary}

\begin{proof}
Indeed, by (\sref{IV.17.4.1}, $d'$)), for every $y\in Y$, the fibre $f^{-1}(y)$ is isomorphic to a prescheme of the form $\coprod_{\lambda\in L}\Spec(K_\lambda)$, where the $K_\lambda$ are finite separable extensions of $k(y)$; hence $f^{-1}(y)$ is a discrete topological space.
\end{proof}

\begin{proposition}[17.4.4]
\label{IV.17.4.4}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, $y=f(x)$.
Set $A=\sh{O}_{Y,y}$, $B=\sh{O}_{X,x}$, which are local Noetherian rings, and let $k$ be the residue field of $A$.
Then the conditions equivalent $a)$ to $e)$ of theorem (17.4.1) are also equivalent to each of the following:
\begin{enumerate}
  \item[f)] $\hat{B}\otimes_{\hat{A}}k$ is a field, finite separable extension of $k$ (which implies that $\hat{B}$ is an $\hat{A}$-algebra finite).
  \item[f')] $\hat{B}$ is an $\hat{A}$-algebra formally non ramified for the adic topologies.
\end{enumerate}
If moreover $k(x)=k(y)$, or if $k$ is separably closed, these conditions are also equivalent to:
\begin{enumerate}
  \item[f'')] The homomorphism $\hat{A}\to\hat{B}$ is surjective.
\end{enumerate}
\end{proposition}

\begin{env}[17.4.5]
\label{IV.17.4.5}
Given an $S$-prescheme $Y$ and two $S$-morphisms $f:X\to Y$, $g:X\to Y$, one canonically deduces an $S$-morphism $(f,g)_S:X\to Y\times_S Y$.
We call the \emph{prescheme of coincidences} of $f$ and $g$ the inverse image by $(f,g)_S$ of the diagonal $\Delta_{Y|S}$; it is therefore a sub-prescheme of $X$, which is a \emph{closed} $S$-scheme when $Y$ is an $S$-scheme (\sref[I]{I.5.4.1}).
\end{env}

\begin{proposition}[17.4.6]
\label{IV.17.4.6}
Let $h:Y\to S$ be a non ramified morphism, and let $f:X\to Y$, $g:X\to Y$ be two $S$-morphisms.
Then the prescheme $C$ of coincidences of $f$ and $g$ is a sub-prescheme induced on an open of $X$; if moreover $Y$ is an $S$-scheme (\sref[I]{I.5.4.1}), $C$ is a closed sub-prescheme of $X$.
\end{proposition}

\begin{proof}
Indeed, since $\Delta_h:Y\to Y\times_S Y$ is an open immersion (17.4.2), the inverse image by $(f,g)_S$ of $\Delta_{Y|S}$ is a sub-prescheme induced on an open of $X$ (\sref[I]{I.4.4.1}).
The last assertion follows from (17.4.5).
\end{proof}

\begin{corollary}[17.4.7]
\label{IV.17.4.7}
Under the hypotheses of (17.4.6), let $x$ be a point of $X$ such that the two composed morphisms $\Spec(k(x))\to X\xrightarrow{f} Y$ and $\Spec(k(x))\to X\xrightarrow{g} Y$ are equal.
Then there exists an open neighbourhood $U$ of $x$ such that $f|U=g|U$.
If moreover $Y$ is an $S$-scheme, there exists a closed open neighbourhood $X'$ of $x$ in $X$ such that $f|X'=g|X'$.
If finally $X$ is connected, then $f=g$.
\end{corollary}

This follows from (17.4.6) and (\sref[I]{I.5.3.17}).

\begin{corollary}[17.4.8]
\label{IV.17.4.8}
Under the hypotheses of (17.4.6), suppose that the structural morphism $\varphi=h\circ f=h\circ g$ of $X$ in $S$ is closed.
Let $s$ be a point of $S$; let $X_s$ be the $k(s)$-prescheme $\varphi^{-1}(s)$ and suppose that the two morphisms composed $X_s\to X\xrightarrow{f} Y$ and $X_s\to X\xrightarrow{g} Y$ are equal.
Then there exists a neighbourhood open $V$ of $s$ in $S$ such that $f|\varphi^{-1}(V)=g|\varphi^{-1}(V)$.
If moreover $\varphi$ is open, one can take $V$ open and closed.
If finally $S$ is connected, $f=g$.
\end{corollary}

\begin{proposition}[17.4.9]
\label{IV.17.4.9}
Let $Y$ be a connected prescheme, $f:X\to Y$ a non ramified and separated morphism.
Then every $Y$-section $g$ of $X$ is an isomorphism of $Y$ onto a connected component open of $X$, and the application $g\mapsto g(Y)$ is a bijection of $\Gamma(X/Y)$ onto the set of connected components $Z$ of $X$ (necessarily \emph{open} in $X$) such that the restriction of $f$ to $Z$ is an isomorphism.
In particular, if $g'$ and $g''$ are two $Y$-sections of $X$ such that $g'(y)=g''(y)$ for some $y\in Y$, then $g'=g''$.
\end{proposition}

\begin{proof}
It follows from (17.4.1,~$b'')$) that a $Y$-section $s$ of $X$ is also an open immersion, and since $X$ is a $Y$-scheme, $s$ is also a closed immersion (\sref[I]{I.5.4.7}); it follows that $s$ is an isomorphism of $Y$ onto a sub-prescheme of $X$ that is both open and closed, hence a connected component open of $X$, and since $s(Y)$ is connected, it is necessarily a connected component.
The rest of the proposition is immediate.
\end{proof}

\subsection{Characterizations of smooth morphisms.}
\label{subsection:IV.17.5}

\oldpage[IV]{67}

\begin{theorem}[17.5.1]
\label{IV.17.5.1}
Let $f:X\to Y$ be a morphism locally of finite presentation, $x$ a point of $X$, $y=f(x)$.
The following conditions are equivalent:
\begin{enumerate}[label=(\alph*)]
  \item $f$ is smooth at the point $x$.
  \item $f$ is flat at the point $x$ and the $k(y)$-prescheme $f^{-1}(y)$ is smooth on $k(y)$ at the point $x$ (6.8.1).
  \item[b')] $f$ is regular at the point $x$ (6.8.1).
  \item $\sh{O}_{X,x}$ is an $\sh{O}_{Y,y}$-algebra formally smooth for the discrete topologies.
\end{enumerate}
\end{theorem}

\begin{proof}
One may reduce to the case where $Y=\Spec(A)$, $X=\Spec(C)$, where $C=B/\mathfrak{J}$, $B=A[T_1,\ldots,T_n]$ being a polynomial algebra and $\mathfrak{J}$ an ideal \emph{of finite type}.
The equivalence of $a)$ and $c)$ follows from ($\mathbf{0}$,~22.6.4).
On the other hand, applying this result to the morphism locally of finite type $f^{-1}(y)\to\Spec(k(y))$, one sees that the equivalence of $b)$ and $b')$ follows from the equivalence of $a)$ and $b)$ in (6.8.6).
It remains to show the equivalence of $a)$ and $b)$.

Let us show first that $a)$ implies $b)$.
The hypothesis $a)$ implies first that $f^{-1}(y)$ is smooth over $k(y)$ at the point $x$ by (17.3.3), and it suffices to show that $C_\mathfrak{p}=\sh{O}_{X,x}$ is an $A_\mathfrak{r}$-module \emph{flat}.
The condition (17.5.1.1) implies that the canonical images $v_i$ of the $u_i$ in the maximal ideal $\mathfrak{m}$ of $B_\mathfrak{q}/\mathfrak{r} B_\mathfrak{q}$ are \emph{linearly independent} mod $\mathfrak{m}^2$: otherwise there would exist polynomials $w_i\in B$ not all in $\mathfrak{q}$ such that $\sum w_i(\partial u_i/\partial T_{j_k})\in\mathfrak{q}$ for $1\leqslant k\leqslant r$, which contradicts (17.5.1.1) since $\mathfrak{q}$ is prime.
One concludes from ($\mathbf{0}$,~17.1.7) that $(v_i)$ is a \emph{regular sequence} in $B_\mathfrak{q}/\mathfrak{r} B_\mathfrak{q}$.
But since $B$ is a morphism locally of finite presentation, it follows from (11.3.8) that the canonical images $u_i'$ of the $u_i$ in $B_\mathfrak{q}$ also form a regular sequence and that $B_\mathfrak{q}/(\sum_i u_i' B_\mathfrak{q})$ is then an $A_\mathfrak{r}$-module \emph{flat}.
It follows from Nakayama's lemma that $\sum_i u_i' B_\mathfrak{q}=\mathfrak{J}_\mathfrak{q}$, and $C_\mathfrak{p}=B_\mathfrak{q}/\mathfrak{J}_\mathfrak{q}$ is thus indeed an $A_\mathfrak{r}$-module flat.

Let us prove finally that $b)$ implies $a)$.
With the same notation, the hypothesis that $B_\mathfrak{q}$ is an $A_\mathfrak{r}$-module flat implies that the canonical injection $\mathfrak{J}_\mathfrak{q}/\mathfrak{r}\mathfrak{J}_\mathfrak{q}\to B_\mathfrak{q}/\mathfrak{r} B_\mathfrak{q}$ is injective ($\mathbf{0}_\mathbf{I}$,~6.1.2), so that $\mathfrak{J}_\mathfrak{q}/\mathfrak{r}\mathfrak{J}_\mathfrak{q}$ identifies with an ideal of $B_\mathfrak{q}/\mathfrak{r} B_\mathfrak{q}$.
Since $B_\mathfrak{q}/\mathfrak{r} B_\mathfrak{q}$ is formally smooth, the Jacobian criterion ($\mathbf{0}$,~22.6.4), together with ($\mathbf{0}$,~19.1.12), proves the existence of $r$ polynomials $v_i\in k(y)[T_1,\ldots,T_n]$ and $r$ indices $j_i$ ($1\leqslant i\leqslant r$) such that their images in $\mathfrak{J}_\mathfrak{q}/\mathfrak{J}_\mathfrak{q}^2$ engender this $B_\mathfrak{q}$-module and $\det(\partial v_i/\partial T_{j_k})\notin\mathfrak{q}$.
If one designates by $u_i$ an element of $\mathfrak{J}$ whose image is $v_i$, by Nakayama the $u_i'$ in $\mathfrak{J}_\mathfrak{q}$ engender $\mathfrak{J}_\mathfrak{q}$, and $C_\mathfrak{p}=B_\mathfrak{q}/\mathfrak{J}_\mathfrak{q}$; the Jacobian criterion ($\mathbf{0}$,~22.6.4) together with ($\mathbf{0}$,~19.1.12), proves that $C_\mathfrak{p}=B_\mathfrak{q}/\mathfrak{J}_\mathfrak{q}$ is an $A_\mathfrak{r}$-algebra formally smooth for the discrete topologies.
C.Q.F.D.
\end{proof}

\begin{corollary}[17.5.2]
\label{IV.17.5.2}
Let $f:X\to Y$ be a morphism locally of finite presentation.
For $f$ to be smooth (in the sense of (17.3.1)), it is necessary and sufficient that $f$ be regular (6.8.1), i.e.\ that $f$ be flat and that for every $y\in Y$, $f^{-1}(y)$ be a $k(y)$-prescheme geometrically regular (6.7.6).
\end{corollary}

One has thus established the equivalence of the two definitions of ``smooth morphism'' given in (6.8.1) and (17.3.1).

\begin{proposition}[17.5.3]
\label{IV.17.5.3}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, $y=f(x)$.
Set $A=\sh{O}_{Y,y}$, $B=\sh{O}_{X,x}$, which are local Noetherian rings.
Then the conditions equivalent $a)$ to $c)$ of (17.5.1) are also equivalent to each of the following:
\begin{enumerate}
  \item[d)] $B$ is an $A$-algebra formally smooth for the preadic topologies.
  \item[d')] $\hat{B}$ is an $\hat{A}$-algebra formally smooth for the adic topologies.
\end{enumerate}
If moreover $k(x)=k(y)$, these conditions are also equivalent to:
\begin{enumerate}
  \item[d'')] $\hat{B}$ is an $\hat{A}$-algebra isomorphic to a formal power series algebra $\hat{A}[[T_1,\ldots,T_n]]$.
\end{enumerate}
\end{proposition}

\begin{remark}[17.5.4]
\label{IV.17.5.4}
Suppose that $Y$ is a locally Noetherian prescheme, and $f:X\to Y$ a morphism locally of finite type.
The criterion (17.5.3,~$d)$), together with ($\mathbf{0}$,~22.1.4) shows that for proving $f$ is smooth, one can apply the definition (17.1.1), \emph{restricting} to the case where the affine scheme $Y'$ is the spectrum of a local Artinian ring.
\end{remark}

\begin{proposition}[17.5.5]
\label{IV.17.5.5}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, $y=f(x)$.
Suppose that $Y$ is reduced at the point $y$.
Then, for $f$ to be smooth at the point $y$, it is necessary and sufficient that $f$ be universally open in a neighbourhood of $x$ in $f^{-1}(y)$ and that $f^{-1}(y)$ be a $k(y)$-prescheme geometrically regular at the point $x$.
\end{proposition}

\begin{proof}
By (17.5.1), everything reduces to seeing that, if $f^{-1}(y)$ is a $k(y)$-prescheme geometrically regular at the point $x$, it is equivalent to say that $f$ is flat or universally open in a neighbourhood of $x$ in $f^{-1}(y)$.
But if $f$ is flat at the point $x$, it is in a neighbourhood of $x$ in $X$ (11.1.1) and hence universally open in this neighbourhood (2.4.6).
Reciprocally, the hypothesis that $f$ is universally open in a neighbourhood of $x$ in $f^{-1}(y)$ and that $f^{-1}(y)$ is a $k(y)$-prescheme geometrically regular at the point $x$ implies that $f$ is flat at the point $x$, since $\sh{O}_{Y,y}$ is reduced (15.2.2).
\end{proof}

\begin{corollary}[17.5.6]
\label{IV.17.5.6}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, $y=f(x)$.
Suppose that $Y$ is reduced and geometrically unibranch (6.15.1) at the point $y$.
Then, for $f$ to be smooth at the point $x$, it is necessary and sufficient that $f$ be equidimensional at the point $x$ and that $f^{-1}(y)$ be a $k(y)$-prescheme geometrically regular at the point $x$.
\end{corollary}

This follows from \sref{IV.17.5.5} and the Chevalley criterion (14.4.4).

\begin{proposition}[17.5.7]
\label{IV.17.5.7}
Let $f:X\to Y$ be a morphism locally of finite presentation, smooth at a point $x\in X$; set $y=f(x)$.
Then, for $\sh{O}_{X,x}$ to be reduced (resp.\ integrally closed, resp.\ geometrically unibranch), it is necessary and sufficient that $\sh{O}_{Y,y}$ be so.
\end{proposition}

This has been proved in (11.3.13) and (11.3.14) completed by \textbf{Err}$_\textbf{IV}$,~30.

\begin{proposition}[17.5.8]
\label{IV.17.5.8}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, smooth at a point $x\in X$; set $y=f(x)$.
Then:
\begin{enumerate}[label=(\roman*)]
  \item $\dim(\sh{O}_{X,x})=\dim(\sh{O}_{Y,y})+\dim(\sh{O}_{X,x}\otimes_{\sh{O}_{Y,y}}k(y))$.
  \item $\operatorname{coprof}(\sh{O}_{X,x})=\operatorname{coprof}(\sh{O}_{Y,y})$.

These are special cases of (6.1.2), (6.3.2), (6.4.1) and (6.5.3).


\subsection{Characterizations of \'etale morphisms.}
\label{subsection:IV.17.6}

\oldpage[IV]{70}

\begin{theorem}[17.6.1]
\label{IV.17.6.1}
Let $f:X\to Y$ be a morphism locally of finite presentation, $x$ a point of $X$, $y=f(x)$.
The following conditions are equivalent:
\begin{enumerate}[label=(\alph*)]
  \item $f$ is \'etale at the point $x$.
  \item[a')] $f$ is smooth at the point $x$ and non ramified at the point $x$.
  \item $f$ is smooth at the point $x$ and locally quasi-finite (\textbf{Err}$_\textbf{III}$,~20).
  \item $f$ is flat at the point $x$ and non ramified at the point $x$.
  \item[c')] $f$ is flat at the point $x$ and the ring $\sh{O}_{X,x}/\mathfrak{m}_y\sh{O}_{X,x}$ is a field, extension finite separable of $k(y)$.
  \item[c'')] $f$ is flat, and for every $y\in Y$ and every algebraically closed extension $k'$ of $k(y)$, the ``geometric fibre'' $f^{-1}(y)\otimes_{k(y)}k'$ is a sum of spectra of fields isomorphic to $k'$.
  \item $\sh{O}_{X,x}$ is an $\sh{O}_{Y,y}$-algebra formally \'etale for the discrete topologies.
\end{enumerate}
\end{theorem}

\begin{proof}
The equivalence of $a)$ and $a')$ follows immediately from the definitions; that of $a)$ and $d)$ follows from the equivalence of $a)$ and $d)$ in (17.4.1) and (17.5.1).
The equivalence of $c)$ and $c')$ follows from $c)$ in (17.4.1); inversely, if $c')$ is verified, $f$ is regular (hence smooth by (17.5.1)) at the point $x$, for if $K$ is any finite extension of a field $k$, $\Spec(K\otimes_k k')$ is regular, being a finite sum of spectra of fields.
The fact that $a')$ implies $b)$ follows from (17.4.1,~$d')$) and (17.5.1,~$b)$).
It remains to show that $b)$ implies $c)$, and since one already knows that $f$ is flat at the point $x$ (by (17.5.1)), it suffices to show that $f^{-1}(y)$ is a $k(y)$-prescheme non ramified on $k(y)$; in other words, one is reduced to proving that $b)$ implies $c)$ when $Y=\Spec(k)$ is the spectrum of a field $k$, $X$ the spectrum of a $k$-algebra $A$ of finite type.
By hypothesis $b)$, $A$ is a $k$-algebra formally smooth for the discrete topologies (for the topologies preadic, which coincide with the discrete topologies here); hence ($\mathbf{0}$,~19.6.5), $A$ is a regular local ring, hence a field since it is Artinian, and it then follows from ($\mathbf{0}$,~19.6.5.1) that $A$ must be a finite separable extension of $k$, which is not possible unless $A$ is a field, finite separable extension of $k(y)$ (4.6.1).
\end{proof}

\begin{corollary}[17.6.2]
\label{IV.17.6.2}
Let $f:X\to Y$ be a morphism locally of finite presentation.
The following conditions are equivalent:
\begin{enumerate}[label=(\alph*)]
  \item $f$ is \'etale.
  \item[a')] $f$ is smooth and non ramified.
  \item $f$ is smooth and locally quasi-finite (\textbf{Err}$_\textbf{III}$,~20).
  \item $f$ is flat and non ramified.
  \item[c')] $f$ is flat, and every fibre $f^{-1}(y)$ is a $k(y)$-scheme sum of spectra of fields, extensions finite and separable of $k(y)$.
  \item[c'')] $f$ is flat, and for every $y\in Y$ and every algebraically closed extension $k'$ of $k(y)$, the ``geometric fibre'' $f^{-1}(y)\otimes_{k(y)}k'$ is a sum of spectra of fields isomorphic to $k'$.
\end{enumerate}
\end{corollary}

\begin{proposition}[17.6.3]
\label{IV.17.6.3}
Let $Y$ be a locally Noetherian prescheme, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, $y=f(x)$.
Set $A=\sh{O}_{Y,y}$, $B=\sh{O}_{X,x}$, which are local Noetherian rings, and let $k$ be the residue field of $A$.
Then the conditions equivalent $a)$ to $d)$ of (17.6.1) are also equivalent to each of the following:
\begin{enumerate}
  \item[e)] $\hat{B}$ is an $\hat{A}$-algebra formally \'etale for the adic topologies.
  \item[e')] $\hat{B}$ is an $\hat{A}$-module free and $\hat{B}\otimes_{\hat{A}}k$ is a field, extension finite and separable of $k$ (which implies that $\hat{B}$ is an $\hat{A}$-algebra finite).
\end{enumerate}
If moreover $k(x)=k(y)$, or if $k$ is separably closed, these conditions are also equivalent to:
\begin{enumerate}
  \item[e'')] The canonical homomorphism $\hat{A}\to\hat{B}$ is bijective.
\end{enumerate}
\end{proposition}

\begin{proposition}[17.6.4]
\label{IV.17.6.4}
Under the hypotheses of (17.6.3), if $f$ is \'etale at the point $x$, then $\dim(\sh{O}_{X,x})=\dim(\sh{O}_{Y,y})$.
\end{proposition}

This is a particular case of (17.5.8,~(i)) since $x$ is isolated in its fibre $f^{-1}(y)$.


\subsection{Properties of descent, passage to the limit, and constructibility.}
\label{subsection:IV.17.7}

\oldpage[IV]{72}

\begin{proposition}[17.7.1]
\label{IV.17.7.1}
Let $f:X\to Y$ be a morphism locally of finite presentation, $g:Y'\to Y$ a morphism, $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$ and $g':X'\to X$ the canonical projections.
Let $x'$ be a point of $X'$ and set $x=g'(x')$, $y'=f'(x')$.
\begin{enumerate}[label=(\roman*)]
  \item If $f'$ is non ramified at the point $x'$, then $f$ is non ramified at the point $x$.
  \item Suppose moreover that $g$ is flat at the point $y'$.
  Then, if $f'$ is smooth (resp.\ \'etale) at the point $x'$, $f$ is smooth (resp.\ \'etale) at the point $x$.
\end{enumerate}
\end{proposition}

\begin{proof}
Set $y=f(x)=g(y')$, so that $f'^{-1}(y')=f^{-1}(y)\otimes_{k(y)}k(y')$; note that $f'$ is locally of finite presentation.

(i) Since the property for a morphism (locally of finite presentation) of being non ramified at a point only involves the fibre of the morphism at that point (17.4.1,~$d$)), one may reduce to the case where $Y$ and $Y'$ are spectra of fields.
But then (17.4.1,~$d'$)) it suffices to say that if $(f')$ is non ramified at $x$ (resp.\ $x'$), or that it is \'etale at $x$ (resp.\ $x'$).

(ii) Since $f'$ is flat at $x'$ (17.5.1), the hypothesis that $g$ is flat at $y'$ implies $f$ is flat at $x$ (by transitivity, 2.2.11,~(iv)).
The fact that $f$ is smooth (resp.\ \'etale) at $x$ then follows from (17.5.1) (resp.\ (17.6.1)) since the fibre $f'^{-1}(y')=f^{-1}(y)\otimes_{k(y)}k(y')$ and being smooth (resp.\ \'etale) at $x'$ implies $f^{-1}(y)$ is geometrically regular (resp.\ a sum of spectra of fields) at $x$ (6.7.8) and this implies that $X$ is a $k(y)$-prescheme geometrically regular at $x$, hence $f$ is smooth at $x$ by (17.5.1).
\end{proof}

\begin{corollary}[17.7.2]
\label{IV.17.7.2}
\begin{enumerate}[label=(\roman*)]
  \item With the notation of (17.7.1), let $U$ (resp.\ $U'$) be the set of points of $X$ (resp.\ $X'$) where $f$ (resp.\ $f'$) is non ramified; then $U'=g'^{-1}(U)$.
  \item Suppose moreover that $g$ is flat, and let $V$ (resp.\ $V'$) be the set of points of $X$ (resp.\ $X'$) where $f$ (resp.\ $f'$) is smooth; then $V'=g'^{-1}(V)$.
\end{enumerate}
\end{corollary}

\begin{proof}
\emph{(i).} Since the property of being non ramified only involves the fibre (\sref{IV.17.4.1}, $d$)), and the fibre of $f'$ at $x'$ is $f^{-1}(y)\otimes_{k(y)}k(y')$, the result follows from the fact that non-ramification is stable under base change of fields (\sref{IV.6.7.8}).

\emph{(ii).} By (i), one has $U'\cap V'=g'^{-1}(U\cap V)$. Since $f'$ is smooth at $x'$ means $f'$ is flat at $x'$ and $f'$ is non ramified at $x'$ on the fibre, the result follows from the fact that flatness descends by flat morphisms (\sref{IV.2.2.11}, (iv)) and from (\sref{IV.17.7.1}, (ii)).
\end{proof}

\begin{corollary}[17.7.3]
\label{IV.17.7.3}
\begin{enumerate}[label=(\roman*)]
  \item Suppose $g$ surjective; then, for $f$ to be non ramified, it is necessary and sufficient that $f'$ be so.
  \item Let $f:X\to Y$ be an $S$-morphism, $g:S'\to S$ a morphism faithfully flat, or that $g$ be quasi-compact.
  Suppose that $f$ is locally of finite presentation, or that $g$ is locally of finite presentation.
  Then, for $f$ to be smooth (resp.\ non ramified, resp.\ \'etale), it is necessary and sufficient that $f'$ be so.
\end{enumerate}
\end{corollary}

\begin{proof}
\emph{(i).} The sufficiency follows from base change (\sref{IV.17.3.3}, (ii)). For the necessity, by (\sref{IV.17.7.2}, (i)), $U'=g'^{-1}(U)$; if $f'$ is non ramified, then $U'=X'$, and since $g$ is surjective, $g'$ is surjective, hence $U=X$.

\emph{(ii).} The sufficiency is again (\sref{IV.17.3.3}, (ii)). For the necessity, the base change of a smooth (resp.\ non ramified, resp.\ \'etale) morphism is smooth (resp.\ non ramified, resp.\ \'etale) (\sref{IV.17.3.3}, (ii)); for the converse, use (\sref{IV.17.7.1}) and the fact that a faithfully flat quasi-compact morphism is surjective (\sref{IV.2.2.13}).
\end{proof}

\begin{proposition}[17.7.4]
\label{IV.17.7.4}
Let $f:X\to Y$ be a morphism, $g:Y'\to Y$ a morphism flat and locally of finite presentation, $X'=X\times_Y Y'$, $f'=f_{(Y')}:X'\to Y'$ and $g':X'\to X$ the canonical projections.
Let $V$ (resp.\ $V'$) be the set of $x\in X$ (resp.\ $x'\in X'$) where $f$ (resp.\ $f'$) has one of the following properties:
\begin{enumerate}[label=(\roman*)]
  \item locally of finite type;
  \item locally of finite presentation;
  \item flat;
  \item non ramified;
  \item smooth;
  \item \'etale.
\end{enumerate}
Then $V'=g'^{-1}(V)$ (in other words, for $f'$ to have the property considered at a point $x'$, it is necessary and sufficient that $f$ have this property at $x$).
\end{proposition}

\begin{lemma}[17.7.5]
\label{IV.17.7.5}
Let $f:X\to Y$ be a faithfully flat morphism and locally of finite presentation, $g:Y\to Z$ a morphism such that $g\circ f:X\to Z$ has one of the following properties: being
\begin{enumerate}[label=(\roman*)]
  \item locally of finite type;
  \item locally of finite presentation;
  \item of finite type.
\end{enumerate}
Then $g$ has the same property.
If moreover $f$ is quasi-compact or $g$ quasi-separated, the same conclusion holds for the property:
\begin{enumerate}
  \item[(iv)] of finite presentation.
\end{enumerate}
\end{lemma}

\begin{remark}[17.7.6]
\label{IV.17.7.6}
Regarding the assertion relative to property (iv), one cannot suppress the hypothesis that $f$ is quasi-compact; without this, it would imply that every morphism $g:Y\to Z$ quasi-compact and locally of finite presentation would be of finite presentation, a conclusion known to be erroneous \sref[I]{I.6.4}.
Indeed, one can only restrict to the case where $Z$ is affine, hence $Y$ quasi-compact; there is then a finite covering of $Y$ by affine opens $U_i$ such that the restrictions $g|U_i$ are of finite presentation; it would then suffice to take for $X$ the prescheme \emph{sum} of the $U_i$, for $f:X\to Y$ the canonical morphism, which is evidently faithfully flat and locally of finite presentation \sref[I]{I.6.5}; $g\circ f$ would be of finite presentation by virtue of the choice of the $U_i$ and of \sref[I]{I.6.5}, whence our assertion.
\end{remark}

\begin{proposition}[17.7.7]
\label{IV.17.7.7}
Let $f:Y\to S$ and $h:X\to S$ be two morphisms locally of finite presentation, $g:X\to Y$ an S-morphism, $x$ a point of $X$, $y=g(x)$.
Suppose that $g$ is flat at the point $x$.
Then, if $h$ is lisse (resp.\ non ramified, resp.\ \'etale) at the point $x$, $f$ is lisse (resp.\ non ramified, resp.\ \'etale) at the point $y$.
\end{proposition}

\begin{proof}
Set $s=h(x)=f(y)$.
To say that $h$ is non ramified at the point $x$ (resp.\ that $f$ is non ramified at the point $y$) means that $h^{-1}(s)$ is \'etale on $k(s)$ at the point $x$ (resp.\ that $f^{-1}(s)$ is \'etale on $k(s)$ at the point $y$) \sref{IV.17.4.1} and \sref{IV.17.6.1}; as the morphism $g_s:h^{-1}(s)\to f^{-1}(s)$ deduced from $g$ is flat at the point $x$, one can reduce to proving the proposition when $h$ is lisse or \'etale.
Moreover, as $h$ is then flat at $x$ \sref{IV.17.5.1}, $f$ is flat at the point $y$ (by transitivity, \sref{IV.2.2.11},~(iv)).
It therefore suffices to say that $f^{-1}(s)$ is lisse (resp.\ \'etale) on $k(s)$ at $y$, and one is thus reduced to the case where $S=\operatorname{Spec}(k)$ is the spectrum of a field.

\emph{(i) Case of smooth morphisms.}
As $g$ is a morphism locally of finite presentation \sref[I]{I.4.3},~(v)), there is a neighbourhood $U$ of $x$ in $X$ in which $g$ is flat \sref{IV.11.3.1} and $h$ lisse.
Moreover $g(U)$ is an open neighbourhood of $y$ in $Y$ \sref{IV.2.4.6}; replacing $X$ by $U$ and $Y$ by $g(U)$, one can suppose that $g$ is \emph{faithfully flat and lisse}, and one is reduced to proving that $f$ is then lisse.
If $k'$ is an algebraically closed extension of $k$, $X\otimes_k k'$ is then lisse on $k'$ \sref{IV.17.7.3},~(ii)); it suffices to prove that $Y\otimes_k k'$ is lisse on $k'$ (since $\operatorname{Spec}(k')\to\operatorname{Spec}(k)$ is faithfully flat); one can therefore restrict to the case where $k$ is \emph{algebraically closed}.
The set of points of $Y$ rational on $k$ is then very dense in $Y$ \sref{IV.10.4.8} and the set of points of $Y$ where $Y$ is lisse on $k$ is open, so it suffices to prove that $Y$ is lisse on $k$ at each rational point $y$; but at such a point, saying that $Y$ is lisse on $k$ is equivalent to saying that $Y$ is \emph{regular} at $y$ \sref{IV.17.5.1} and \sref{IV.6.7.8}).
One has $y=g(x)$ for some $x\in X$ and by hypothesis \sref{IV.17.5.1} $X$ is regular at the point $x$; as $X$ and $Y$ are then locally Noetherian and $g$ is flat, $Y$ is indeed regular at the point $y$ \sref{IV.6.5.1},~(i)).

\oldpage[IV]{75}

\emph{(ii) Case of \'etale morphisms.}
By (i) one already knows that $f$ is \'etale at the point $y$; by virtue of \sref{IV.17.6.1}, it suffices to show that $f$ is \emph{quasi-finite} at the point $y$.
Or, as $\sh{O}_{X,x}$ is an $\sh{O}_{Y,y}$-module faithfully flat ($\mathbf{0}_I$, \sref{IV.6.6.2}), $\sh{O}_{Y,y}$ is identified with a sub-$k$-module of $\sh{O}_{X,x}$ and as $\sh{O}_{X,x}$ is by hypothesis a $k$-algebra finite, so is $\sh{O}_{Y,y}$.
\end{proof}

\begin{proposition}[17.7.8]
\label{IV.17.7.8}
With the notations being those of \sref{IV.8.8.1}, suppose $X_\alpha$ and $Y_\alpha$ locally of finite presentation on $S_\alpha$.
Let $f_\alpha:X_\alpha\to Y_\alpha$ be an $S_\alpha$-morphism, $f:X\to Y$ the corresponding S-morphism.
\begin{enumerate}[label=(\roman*)]
  \item Let $x$ be a point of $X$, $x_\lambda$ its canonical projection in $X_\lambda$.
  For $f$ to be lisse (resp.\ non ramified, resp.\ \'etale) at the point $x$, it is necessary and sufficient that there exist $\lambda\geq\alpha$ such that $f_\lambda$ is lisse (resp.\ non ramified, resp.\ \'etale) at the point $x_\lambda$.
  \item Suppose moreover that $X_\alpha$ is quasi-compact.
  For $f$ to be lisse (resp.\ non ramified, resp.\ \'etale), it is necessary and sufficient that there exist $\lambda\geq\alpha$ such that $f_\lambda$ is lisse (resp.\ non ramified, resp.\ \'etale).
\end{enumerate}
\end{proposition}

\begin{proof}
(i) By hypothesis, for every $\lambda$, $f^{-1}(y)=f_\lambda^{-1}(y_\lambda)\otimes_{k(y_\lambda)}k(y)$; the part of the statement concerning non ramified morphisms is thus clear, and it suffices to consider the case of smooth morphisms.
As $f$ and $f_\lambda$ are locally of finite presentation, it reduces to the same by \sref{IV.17.5.1} to saying that $f^{-1}(y)$ is geometrically regular at the point $x$ or that $f_\lambda^{-1}(y_\lambda)$ is geometrically regular at the point $x_\lambda$ \sref{IV.6.7.8}.
The proposition then results from \sref{IV.17.5.1} and \sref{IV.11.2.6}.

(ii) For every $\lambda$, let $U_\lambda$ be the open set of $x_\lambda\in X_\lambda$ where $f_\lambda$ is lisse (resp.\ non ramified, resp.\ \'etale); let $V_\lambda$ be its inverse image in $X$.
By hypothesis, for every $x\in X$, there exists a $\lambda$ such that $f_\lambda$ is lisse (resp.\ non ramified, resp.\ \'etale) at the point $x_\lambda$.
As, by \sref{IV.17.3.3}, for $\lambda\leq\mu$ we have $V_\lambda\subset V_\mu$, and as $X$ is quasi-compact, there exists an index $\mu$ such that $X=V_\mu$.
Furthermore \sref{IV.17.7.3},~(ii)), one concludes that for some $\nu\geq\mu$, $U_\nu=X_\nu$, which means $f_\nu$ is lisse (resp.\ non ramified, resp.\ \'etale) by \sref{IV.17.3.3}.
\end{proof}

\begin{corollary}[17.7.9]

Let $S=\operatorname{Spec}(A)$ be an affine scheme, $f:X\to S$ a morphism.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] $f$ is a morphism of finite presentation and lisse (resp.\ non ramified, resp.\ \'etale).
  \item[b)] There exists an affine Noetherian scheme $S_0=\operatorname{Spec}(A_0)$, a morphism of finite type $f_0:X_0\to S_0$ and a morphism $S\to S_0$ such that the $S$-prescheme $X_0\otimes_{S_0} S$ is S-isomorphic to $X$ and that $f_0$ is lisse (resp.\ non ramified, resp.\ \'etale).
  \item[c)] The conditions of $b$) are satisfied, and moreover $A_0$ is a sub-$\bb{Z}$-algebra of finite type of $A$, and the morphism $S\to S_0$ corresponds to the canonical injection $A_0\to A$.
\end{enumerate}
\end{corollary}

\begin{proof}
The proof from \sref{IV.17.7.8} is the same as that of \sref{IV.11.2.7} from \sref{IV.11.2.6}.
\end{proof}

\begin{proposition}[17.7.10]
\label{IV.17.7.10}
\oldpage[IV]{76}
Let $f:Y\to S$, $h:X\to S$ be two morphisms locally of finite presentation, $g:X\to Y$ an S-morphism, $x$ a point of $X$, $y=g(x)$.
Suppose that $h$ is flat at the point $x$ and that $f$ is non ramified at the point $y$.
Then $f$ is \'etale at the point $y$ and $g$ is flat at the point $x$.
\end{proposition}

\begin{proof}
(One will see further on \sref{IV.18.4.9} that one can in fact dispense with the hypothesis that $h$ is locally of finite presentation.)

The question being local on $S$, $X$ and $Y$, one can suppose that $S$, $X$ and $Y$ are affine, $f$, $g$, $h$ are morphisms of finite presentation, $h$ flat and $f$ non ramified; taking into account \sref{IV.11.2.7} and \sref{IV.17.7.9}, one can furthermore suppose that $S$, $X$ and $Y$ are \emph{Noetherian}; finally one can reduce to the case where $S=\operatorname{Spec}(\sh{O}_{S,s})$ (where $s=f(y)=h(x)$).
The ring $A=\sh{O}_{S,s}$ being a local Noetherian ring, there exists a complete local Noetherian ring $B$, of residue field algebraically closed, and a local homomorphism $A\to B$ making $B$ an $A$-module faithfully flat $(\mathbf{0_{III}}$, \sref{IV.10.3.1}).
Replacing $X$ and $Y$ by $X\times_S\operatorname{Spec}(B)$ and $Y\times_S\operatorname{Spec}(B)$, one concludes by \sref{IV.2.5.1} and \sref{IV.17.7.1} that one can reduce to the case where $A$ is complete and with residue field algebraically closed.
The hypothesis that $f$ is non ramified at the point $y$ implies \sref{IV.17.4.4} that $\sh{O}_{Y,y}$ is an $\sh{O}_{S,s}$-algebra finite, and as the residue field of $\sh{O}_{S,s}$ is algebraically closed, the homomorphism $\sh{O}_{S,s}\to\sh{O}_{Y,y}$ is \emph{surjective} \sref{IV.17.4.4}.
But on the other hand, the hypothesis that $h$ is flat at the point $x$ implies that the composed homomorphism $\sh{O}_{S,s}\to\sh{O}_{Y,y}\to\sh{O}_{X,x}$ is injective ($\mathbf{0}_I$, \sref{IV.6.5.1}), whence $\sh{O}_{S,s}\to\sh{O}_{Y,y}$ is bijective; this shows that $f$ is \'etale at the point $y$ \sref{IV.17.6.3}; furthermore, $\sh{O}_{X,x}$ is an $\sh{O}_{Y,y}$-module flat, and $g$ is flat at the point $x$.
\end{proof}

\begin{proposition}[17.7.11]
\label{IV.17.7.11}
Let $S$ be a prescheme, $X$, $Y$ two S-preschemes locally of finite presentation on $S$.
For every $s\in S$, denote $X_s$, $Y_s$, $f_s$ the preschemes and morphism deduced from $X$, $Y$, $f$ by the change of base $\operatorname{Spec}(k(s))\to S$.
Then:
\begin{enumerate}[label=(\roman*)]
  \item The set of $x\in X$ such that, if $s$ is the image of $x$ in $S$, $f_s:X_s\to Y_s$ is lisse (resp.\ non ramified, resp.\ \'etale, resp.\ differentially lisse) at the point $x$, is locally constructible.
  \item Suppose moreover that $f$ is of finite presentation.
  The set of $y\in Y$ such that, if $s$ is the image of $y$ in $S$, $f_s$ is lisse (resp.\ non ramified, resp.\ \'etale, resp.\ differentially lisse) at all the points of $f_s^{-1}(y)$, is locally constructible.
  \item Suppose $X$ and $Y$ of finite presentation on $S$.
  The set of $s\in S$ such that $f_s$ is lisse (resp.\ non ramified, resp.\ \'etale, resp.\ differentially lisse) is locally constructible.
\end{enumerate}
\end{proposition}

\oldpage[IV]{77}

\begin{proof}
Let $E$ be the set of $x\in X$ satisfying the property considered in (i).
Denoting by $g:X\to S$ and $h:Y\to S$ the structural morphisms, one can first replace $S$ by the reduced sub-prescheme $S'$ of $S$ having $\overline{\{g(x)\}}$ as underlying space, $X$ by $g^{-1}(S')$, $Y$ by $h^{-1}(S')$; the fibres of $X$ and $X'$ (resp.\ $Y$ and $Y'$) at the points of $S'$ being the same.
Now let us restrict to the case where $S$ is \emph{integral} and where $\eta=g(x)=h(y)$ (where $y=f(x)$) is its generic point.

$1\textsuperscript{o}$
Suppose first that $x\in E$.
The local rings $\sh{O}_{X_{\eta},x}$ and $\sh{O}_{Y_{\eta},y}$ are respectively equal to $\sh{O}_{X,x}$ and $\sh{O}_{Y,y}$; as the property of smoothness of a morphism of finite presentation at a point depends only on the local ring of the morphism at that point and on the local ring of its image \sref{IV.17.5.1}, one sees that the hypothesis $x\in E$ means the morphism $f$ is lisse at the point $x$; it then has this property at the points of an open neighbourhood of $x$ in $X$, and it suffices to apply \sref{IV.17.3.3},~(iii)) to obtain the conclusion.

$2\textsuperscript{o}$
Suppose second that $x\in X-E$, and that the morphism $f_\eta$ is \emph{not flat} at the point $x$.
The conclusion results from the following lemma which specifies \sref{IV.11.2.8}:

\begin{lemma}[17.7.11.1]
\label{IV.17.7.11.1}
Let $g:X\to S$, $h:Y\to S$ be two morphisms locally of finite presentation, $f:X\to Y$ an S-morphism, $\sh{F}$ an $\sh{O}_X$-Module quasi-coherent.
Then the set $E$ of $x\in X$ such that $\sh{F}_{g(x)}$ is $f_{g(x)}$-flat at the point $x$ is locally constructible.
\end{lemma}

$3\textsuperscript{o}$
Suppose now that $x\in X-E$, that the morphism $f_\eta$ is \emph{flat} at the point $x$, but that $f_\eta$ does not lisse at the point $x$.
Note that to say $f_\eta$ is flat at the point $x$ means that $f$ itself is flat at the point $x$ and replacing $X$ by a neighbourhood of $x$, \sref{IV.11.1.1}; one concludes that it is also true of $f_s$ for every $s\in S$, and since for every $y\in Y$, $f^{-1}(y)=f_s^{-1}(y)$, it comes back to the same to say that $f_{g(x)}$ is lisse at the point $x'$ or to say that $f$ is lisse at the point $x'$.
But the set of $x'\in X$ where $f$ is lisse is open \sref{IV.17.5.1}, and since it does not contain $x$ by hypothesis, it also does not contain $\overline{\{x\}}$, which achieves the proof for the first property considered in (i).

\oldpage[IV]{78}

We prove secondly in (i) when the property in question is that of being \emph{\'etale}.
Note that this property for $f_s$ at the point $x$ means that $f_s$ is at the same time \emph{lisse} and \emph{quasi-finite} at the point $x$ \sref{IV.17.6.1}.
Or, it comes back to the same to say that $f_{g(x)}$ is quasi-finite, or that $f$ is itself quasi-finite at this point; it results from \sref{IV.13.1.4} that the set of points $x$ such that $f_{g(x)}$ is quasi-finite at the point $x$ is open in $X$, and \emph{a fortiori} locally constructible; the conclusion thus results from the fact that the set of $x$ such that $f_{g(x)}$ is lisse is also locally constructible.

Let us pass to the proof of (i) when the property in question is that of being \emph{differentially lisse}.
Let $p:X\times_Y X\to X$ be the second canonical projection; the second projection $X_s\times_{Y_s}X_s\to X_s$ is just $p_s$ for every $s\in S$, and it results from \sref{IV.17.12.5} that for $f_{g(x)}$ to be differentially lisse at the point $x$, it is necessary and sufficient that $p_{g(x)}$ be lisse at $\Delta_f(x)$.
As $p$ is locally of finite presentation, the set of points $z\in X\times_Y X$ such that $p_{g(p(z))}$ is lisse at the point $z$ is locally constructible, and the intersection of this set with the locally closed set $\Delta_f(X)$ is also locally constructible; the restriction of $p$ to $\Delta_f(X)$ being an isomorphism on $X$, one sees that the set of $x\in X$ such that $f_{g(x)}$ is differentially lisse at the point $x$ is locally constructible.

Finally, consider the property of being non ramified; note that the diagonal morphism $\Delta_f:X\to X\times_Y X$ is an immersion locally of finite presentation \sref[I]{I.4.3.1} and for every $s\in S$, the diagonal morphism $\Delta_{f_s}:X_s\to X_s\times_{Y_s}X_s$ is just $(\Delta_f)_s$; saying that $f_{g(x)}$ is non ramified at the point $x$ means that $(\Delta_f)_{g(x)}$ is an isomorphism local at the point $x$ \sref{IV.17.4.1}, and since this is an immersion locally of finite presentation, it suffices from \sref{IV.17.9.1} to say that $(\Delta_f)_{g(x)}$ is \'etale at the point $x$; it suffices therefore to apply what has been seen above for the property of being \'etale.

The assertions (ii) and (iii) of \sref{IV.17.7.11} follow from (i), the theorem of Chevalley on images of constructible sets \sref{IV.1.8.4}, and (when $h$ is the structural morphism) similar arguments.
\end{proof}


\subsection{Criteria for smoothness and non-ramification by fibres.}
\label{subsection:IV.17.8}

\oldpage[IV]{79}

\begin{proposition}[17.8.1]
\label{IV.17.8.1}
Let $g:Y\to S$, $h:X\to S$ be two morphisms locally of finite presentation.
For an S-morphism $f:X\to Y$ to be non ramified, it is necessary and sufficient that for every $s\in S$, the morphism $f_s:h^{-1}(s)\to g^{-1}(s)$ deduced from $f$ by the change of base $\operatorname{Spec}(k(s))\to S$, be non ramified.
\end{proposition}

\begin{proof}
This follows immediately from the fact that, for a morphism locally of finite presentation, being non ramified is a property of the \emph{fibres} of this morphism \sref{IV.17.4.1},~$d$)), and from the fact that for every $y\in Y$, one has $f^{-1}(y)=f_s^{-1}(y)$ where $s=g(y)$.
\end{proof}

\begin{proposition}[17.8.2]
\label{IV.17.8.2}
Let $g:Y\to S$, $h:X\to S$ be two morphisms locally of finite presentation; suppose moreover that $h$ is flat.
For an S-morphism $f:X\to Y$ to be lisse (resp.\ \'etale), it is necessary and sufficient that for every $s\in S$, the morphism $f_s:h^{-1}(s)\to g^{-1}(s)$ be lisse (resp.\ \'etale).
When this is the case, the morphism $g$ is \'etale and flat at all points of $f(X)$.
\end{proposition}

\begin{proof}
One knows in fact \sref{IV.11.3.10} that for every $s\in S$, $f_s$ is flat at all points of $f(X)$, and it suffices that for every $s\in S$, $f_s$ be lisse at all points of $f(X)$; but for a morphism locally of finite presentation plat, the fact of being lisse is a property of the \emph{fibres} of this morphism \sref{IV.17.5.1},~$b$)), and for every $y\in Y$, $f^{-1}(y)=f_s^{-1}(y)$ where $s=g(y)$.
\end{proof}

\begin{remark}[17.8.3]
\label{IV.17.8.3}
The preceding proofs show (taking into account \sref{IV.11.3.10}) that if the hypotheses on $g$ and $h$ are the same as above, then, for $f$ to be non ramified (resp.\ lisse, resp.\ \'etale) \emph{at a point} $x\in X$, it suffices that, if one sets $s=h(x)$, $f_s$ be non ramified (resp.\ lisse, resp.\ \'etale) at the point $x$.
\end{remark}


\subsection{\'Etale morphisms and open immersions.}
\label{subsection:IV.17.9}

\begin{theorem}[17.9.1]
\label{IV.17.9.1}
Let $f:X\to Y$ be a morphism.
The following properties are equivalent:
\begin{enumerate}
  \item[a)] $f$ is an open immersion.
  \item[b)] $f$ is a flat monomorphism locally of finite presentation.
  \item[c)] $f$ is \'etale and radiciel.
\end{enumerate}
\end{theorem}

\begin{proof}
It results from \sref[I]{I.4.3},~(i)) that $a)$ implies $b)$.
The condition $b)$ implies that for every $y\in Y$, the fibre $f^{-1}(y)$ is either empty or isomorphic to $\operatorname{Spec}(k(y))$ \sref{IV.8.11.5.1}, hence $b)$ implies $c)$, by virtue of \sref{IV.17.6.2},~$c$)).
It remains to show that $c)$ implies $a)$.
The question being local on $X$ and on $Y$ (since $f$ is injective), one can reduce to the case where $Y$ is affine and $f$ of finite presentation.
As $f$ is flat, this is an \emph{open} morphism \sref{IV.2.4.6}, hence, replacing $Y$ by $f(X)$, one can suppose that $f$ is \emph{surjective}.
For every morphism $Y'\to Y$, $f'=f_{(Y')}:X_{(Y')}\to Y'$ is again \'etale, radiciel, surjective and of finite presentation; in other words $f$ is a \emph{universal homeomorphism}, that is to say $f$ is of finite type and is separated \sref[I]{I.8.7.1},~(ii)).
The hypothesis that $f$ is radiciel and of finite type implies that $f$ is quasi-finite; hence \sref{IV.8.11.1} $f$ is a finite morphism.
To prove that $f$ is an isomorphism, one can reduce to the case where $Y=\operatorname{Spec}(A)$, $A$ being a local ring.
As $f$ is of finite presentation, on has $X=\operatorname{Spec}(B)$, where $B$ is an A-module flat of finite presentation \sref[I]{I.4.7}, hence \emph{free} (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection~5, n\textsuperscript{o}~2, cor.~2 of th.~1).
Moreover, if $\mathfrak{m}$ is the maximal ideal of $A$ and $k$ its residue field, $B/\mathfrak{m}B$ is by hypothesis a field, which is both radiciel extension and \'etale extension of $k$ \sref{IV.17.6.1}; hence $B/\mathfrak{m}B$ is isomorphic to $k$, and as $B$ is a free A-module, $B$ is isomorphic to $A$.
C.Q.F.D.
\end{proof}

\begin{corollary}[17.9.2]
\label{IV.17.9.2}
Let $X$ be a connected prescheme; if $f:X\to Y$ is a closed \'etale immersion, then $f$ is an isomorphism of $X$ onto a connected component of $Y$.
\end{corollary}

\begin{proof}
Indeed $f$ is \'etale and radiciel, hence an isomorphism from $X$ onto a sub-prescheme of $Y$; but by hypothesis $f(X)$ is closed in $Y$, hence since open and closed, and since $f(X)$ is connected, it is a connected component of $Y$.
\end{proof}

\oldpage[IV]{80}

\begin{corollary}[17.9.3]
\label{IV.17.9.3}
Let $f:X\to Y$ be a morphism \'etale (resp.\ \'etale and separated).
Then every Y-section $g:Y\to X$ of $X$ is an open immersion (resp.\ open and closed); moreover the application $g\mapsto g(Y)$ is a bijection of the set $\Gamma(X/Y)$ of Y-sections of $X$ to the set of open (resp.\ open and closed) parts $Z$ of $X$ such that the restriction of $f$ to $Z$ is a surjective morphism and radiciel of $Z$ onto $Y$.
\end{corollary}

\begin{proof}
Indeed, the fact that $g$ is an open immersion follows already from the fact that $f$ is non ramified \sref{IV.17.4.1},~$b''$)), and the restriction of $f$ to the open $g(Y)$ of $X$ is an isomorphism.
Inversely, if $Z$ is an open of $X$ such that $f|Z$ is a surjective morphism and radiciel, since $f$ is \'etale, $f|Z$ is an isomorphism by virtue of \sref{IV.17.9.1}, which proves the corollary.
\end{proof}

\begin{corollary}[17.9.4]
\label{IV.17.9.4}
Let $Y$ be a connected prescheme and separated, $f:X\to Y$ a morphism \'etale.
Then every Y-section $g$ of $X$ is an isomorphism of $Y$ onto a connected component of $X$, and the application $g\mapsto g(Y)$ is a bijection of $\Gamma(X/Y)$ to the set of connected components of $X$ such that the restriction of $f$ to $Z$ is a surjective morphism and radiciel of $Z$ onto $Y$.
\end{corollary}

\begin{corollary}[17.9.5]
\label{IV.17.9.5}
Let $g:Y\to S$, $h:X\to S$ be two morphisms locally of finite presentation; suppose moreover that $h$ is flat.
For an S-morphism $f:X\to Y$ to be an open immersion (resp.\ isomorphism), it is necessary and sufficient that for every $s\in S$, the morphism $f_s:h^{-1}(s)\to g^{-1}(s)$ deduced from $f$ by the change of base $\operatorname{Spec}(k(s))\to S$, be an open immersion (resp.\ isomorphism).
\end{corollary}

\begin{proof}
Indeed, if $f_s$ is an open immersion for every $s\in S$, it results from \sref{IV.17.8.3} that $f$ is \'etale, and as for every $y\in Y$, $f^{-1}(y)=f_s^{-1}(y)$ with $s=g(y)$, $f$ is radiciel; hence $f$ is an open immersion by virtue of \sref{IV.17.9.1}.
If moreover $f_s$ is surjective for every $s\in S$, $f$ is surjective, hence an isomorphism.
\end{proof}

The following proposition specifies \sref{IV.10.4.11}:
\begin{proposition}[17.9.6]
\label{IV.17.9.6}
Let $S$ be a prescheme, $X$ an S-prescheme of finite presentation.
Every S-endomorphism of $X$ which is a monomorphism is an automorphism of $X$.
\end{proposition}

\begin{proof}
Let $f:X\to S$ be the structural morphism, $g$ the S-endomorphism considered.
The question being local on $S$, one can suppose $S$ affine.
Using \sref{IV.8.9.1} and \sref{IV.8.10.5},~(i \emph{bis})), one is reduced to the case where $S=\operatorname{Spec}(A)$, where $A$ is a $\bb{Z}$-algebra of finite type, and hence $X$ is a $\bb{Z}$-prescheme of finite type.
It follows already from \sref{IV.10.4.11} that $g$ is a bijective morphism, since a monomorphism is radiciel (\sref{IV.8.11.5.1}); it will thus suffice to show that $g$ is an open immersion, and since $g$ is radiciel, it will suffice, by virtue of \sref{IV.17.9.1}, to prove that $g$ is \'etale.
Moreover, as the set of points where $g$ is \'etale is open and $X$ is a Jacobson prescheme (\sref{IV.10.4.7}), it suffices to show that $g$ is \'etale at every \emph{closed} point $z$ of $X$ (\sref{IV.10.3.1}).
Set $z'=g(z)$; it follows from \sref{IV.10.4.11},~(i) that $z'$ is also a closed point of $X$, and since $g$ is a monomorphism, the map $k(z')\to k(z)$ deduced from $g$ is an isomorphism.
For $g$ to be \'etale at the point $z$, it is thus necessary and sufficient that the canonical homomorphism $\hat{\sh{O}}_{X,z'}\to\hat{\sh{O}}_{X,z}$ be bijective (\sref{IV.17.6.3},~$e''$)).
We will show for this purpose that for every integer $n$, the canonical homomorphism $\sh{O}_{X,z'}/\mathfrak{m}_{z'}^{n+1}\to\sh{O}_{X,z}/\mathfrak{m}_z^{n+1}$ is bijective, from which the conclusion will immediately follow.
One can suppose that $z$ belongs to the finite set $T_{p,d}$ of closed points $t$ of $X$ such that $k(t)$ is an extension of $\bb{F}_p$ whose degree divides $d$; in which case the same holds for $z'$.
Denote by $\sh{J}$ the coherent Ideal of $\sh{O}_X$ such that $\sh{J}_x=\sh{O}_x$ for $x\notin T_{p,d}$, and $\sh{J}_x=\mathfrak{m}_x^{n+1}$ for $x\in T_{p,d}$.
Let $T_{p,d,n}$ be the closed sub-prescheme of $X$ defined by $\sh{J}$, $j:T_{p,d,n}\to X$ the canonical injection; the composed morphism $g\circ j:T_{p,d,n}\to X$ maps the set $T_{p,d}$ into itself, and for every $x\in T_{p,d,n}$, the homomorphism $\sh{O}_{X,g(x)}\to\sh{O}_{X,x}/\mathfrak{m}_x^{n+1}$ factors as
\[
  \sh{O}_{X,g(x)} \to \sh{O}_{X,g(x)}/\mathfrak{m}_{g(x)}^{n+1} \to \sh{O}_{X,x}/\mathfrak{m}_x^{n+1}.
\]
By (\sref[I]{I.4.9}), there therefore exists a unique endomorphism $g'$ of $T_{p,d,n}$ such that $j\circ g'=g\circ j$.
As $j$ is a monomorphism, and $g$ is also one by hypothesis, one deduces that $g'$ is also a monomorphism, and the proposition will be proven if one shows that $g'$ is an \emph{automorphism} of $T_{p,d,n}$.
But, for every $x\in T_{p,d,n}$, we have seen that $k(x)$ is a finite field, and as $\sh{O}_{X,x}$ is Noetherian, each of the $\mathfrak{m}_x^h/\mathfrak{m}_x^{n+1}$ is a $k(x)$-vector space of finite rank; from which one concludes immediately that the local ring $\sh{O}_{X,x}/\mathfrak{m}_x^{n+1}$ at the point $x$ has a finite number of elements, and \emph{a fortiori} is a $\bb{Z}$-module of finite type.
As $T_{p,d,n}$ is the sum of the preschemes $\operatorname{Spec}(\sh{O}_{X,x}/\mathfrak{m}_x^{n+1})$ in finite number, it is a finite $\bb{Z}$-prescheme; the endomorphism $g'$ is therefore also finite (\textbf{II},~6.1.5,~(v)), hence proper.
But a proper monomorphism is a closed immersion (\sref{IV.8.11.5}); if $T_{p,d,n}=\operatorname{Spec}(B)$, $g'$ corresponds to a surjective endomorphism $\varphi:B\to B$ of the ring $B$.
But the set $B$ is finite, hence $\varphi$ is necessarily bijective.
\end{proof}


\subsection{Relative dimension of a smooth prescheme over another.}
\label{subsection:IV.17.10}

\oldpage[IV]{81}

\begin{definition}[17.10.1]
\label{IV.17.10.1}
Let $f:X\to Y$ be a morphism locally of finite type.
The \emph{relative dimension} of $f$ at a point $x\in X$ (or the \emph{dimension of $X$ relative to $Y$ at the point $x$}) is the positive integer $\dim_x f=\dim_x(f^{-1}(f(x)))$.
\end{definition}

To say that $f$ is \emph{quasi-finite} at the point $x$ ($\mathbf{Err_{III}}$, 20) is equivalent to saying that $\dim_x f=0$.
One has seen \sref{IV.13.1.3} that the function $x\mapsto\dim_x f$ is upper semi-continuous.
One notes that, even when the morphism $f$ has the property $(S_1)$ (that is to say \sref{IV.6.8.1} is flat and such that its fibres have no embedded associated prime cycle), the function $x\mapsto\dim_x f$ is not necessarily continuous, as shown by the example where $Y=\operatorname{Spec}(k)$, where $k$ is a field, and $X=\operatorname{Spec}(k[\mathrm{U},\mathrm{V},\mathrm{W}]/\mathfrak{p}\mathfrak{q})$, where $\mathfrak{p}=(\mathrm{W})$ and $\mathfrak{q}=(\mathrm{U})+(\mathrm{V}-\mathrm{W})$, the prime ideals of $k[\mathrm{U},\mathrm{V},\mathrm{W}]$ ($X$ being thus the reunion, in the affine space of $3$ dimensions, of a plane and of a line non parallel to this plane).

To say that a morphism locally of finite presentation $f:X\to Y$ is \'etale at a point $x\in X$ means that $f$ is lisse at the point $x$ and that $\dim_x f=0$ \sref{IV.17.6.1}.

\begin{proposition}[17.10.2]
\label{IV.17.10.2}
Let $f:X\to Y$ be a smooth morphism.
For every $x\in X$, the $\sh{O}_{X,x}$-Module locally free $\Omega^1_f$ \sref{IV.17.2.3} has rank equal to $\dim_x f$ in $x$ (which implies that $x\mapsto\dim_x f$ is a \emph{continuous} function on $X$).
\end{proposition}

\begin{proof}
Indeed, if $y=f(x)$, $X_y=f^{-1}(y)$ is lisse over $k(y)$, and if $f_y:X_y\to\operatorname{Spec}(k(y))$ is the structural morphism, one has $\Omega^1_{f_y}=\Omega^1_f\otimes_{\sh{O}_x} k(y)$ \sref{IV.16.4.5}; one can thus reduce to the case where $Y=\operatorname{Spec}(k)$ is the spectrum of a field.
But \sref{IV.16.4.12} $(\Omega^1_{X/k})_x\otimes_{\sh{O}_x} k(x)$ is then $k$-isomorphic to $\mathfrak{m}_x/\mathfrak{m}_x^2$, and since $\sh{O}_x$ is a regular local ring \sref{IV.17.5.1}, $\operatorname{rg}_k(\mathfrak{m}_x/\mathfrak{m}_x^2)=\dim(\sh{O}_x)$ ($\mathbf{0}$, \sref{IV.17.1.1}); hence $\operatorname{rg}_{\sh{O}_x}(\Omega^1_{X/k})_x=\dim(\sh{O}_x)$.
But since $k(x)=k$, one has $\dim_x f=\dim(\sh{O}_x)$ \sref{IV.5.2.3}.
C.Q.F.D.
\end{proof}

We will prove a reciprocal of this result further on \sref{IV.17.15.5}.

\oldpage[IV]{82}

\begin{corollary}[17.10.3]
\label{IV.17.10.3}
Let $f:X\to Y$, $g:Y\to Z$ be two smooth morphisms.
Then for every $x\in X$, one has
\[
\dim_x(g\circ f) = \dim_x f + \dim_{f(x)} g.
\]
\end{corollary}

\begin{proof}
Indeed, $g\circ f$ is smooth \sref{IV.17.3.3}, hence the three $\sh{O}_X$-Modules $\Omega^1_{X/Y}$, $\Omega^1_{X/Z}$ and $f^*(\Omega^1_{Y/Z})$ are locally free (\sref{IV.17.2.3} and ($\mathbf{0_I}$, 5.4.5)); moreover the rank at $x$ of $f^*(\Omega^1_{Y/Z})$ is equal to the rank at $y$ of $\Omega^1_{Y/Z}$.
The equality \sref{IV.17.10.3} is thus a consequence of \sref{IV.17.10.2} and of the exactness of the suite \sref{IV.17.2.3}.
\end{proof}

\begin{corollary}[17.10.4]
\label{IV.17.10.4}
Let $f:X\to Y$ be a smooth morphism, $X'$ a sub-prescheme of $X$ such that the composed morphism $X'\xrightarrow{j}X\xrightarrow{f}Y$ (where $j$ is the canonical injection) be lisse.
Then the conormal sheaf $\sh{N}_{X'/X}$ is an $\sh{O}_{X'}$-Module locally free, and for every $x\in X'$, one has
\begin{equation}
\label{IV.17.10.4.1}
\tag{17.10.4.1}
  \dim_x f = \dim_x(f\circ j) + \operatorname{rg}_{\sh{O}_{X',x}}(\sh{N}_{X'/X})_x.
\end{equation}
\end{corollary}

\begin{proof}
Indeed, $\Omega^1_{X/Y}\otimes_{\sh{O}_x} \sh{O}_{X'}$ and $\Omega^1_{X'/Y}$ are then locally free and the exact sequence \sref{IV.17.2.5.1} is locally split in a neighbourhood of each point of $X'$, hence $\sh{N}_{X'/X}$ is also locally free, and the relation \sref{IV.17.10.4.1} results from the exactness of the suite \sref{IV.17.2.5.1}.
\end{proof}


\subsection{Smooth morphisms of smooth preschemes.}
\label{subsection:IV.17.11}

\begin{theorem}[17.11.1]
\label{IV.17.11.1}
Let $f:Y\to S$ and $h:X\to S$ be two morphisms locally of finite presentation, $g:X\to Y$ an S-morphism, $x$ a point of $X$; set $y=g(x)$, $s=f(y)=h(x)$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] $f$ is lisse at the point $y$ and $g$ is lisse at the point $x$.
  \item[b)] $g$ and $h$ are lisse at the point $x$.
  \item[c)] $h$ is lisse at the point $x$, and the canonical homomorphism \sref{IV.16.4.18}
\end{enumerate}
\begin{equation}
\label{IV.17.11.1.1}
\tag{17.11.1.1}
  (g^*(\Omega^1_{Y/S}))_x \to (\Omega^1_{X/S})_x
\end{equation}
is left invertible (in other words is an isomorphism onto a direct factor of $(\Omega^1_{X/S})_x$).
\begin{enumerate}
  \item[c')] $h$ is lisse at the point $x$, and the canonical homomorphism
\end{enumerate}
\begin{equation}
\label{IV.17.11.1.2}
\tag{17.11.1.2}
  ((\Omega^1_{Y/S})_y\otimes_{\sh{O}_{Y,y}} k(y))\otimes_{k(y)} k(x) \to (\Omega^1_{X/S})_x\otimes_{\sh{O}_{X,x}} k(x)
\end{equation}
is injective.

Suppose moreover that the homomorphism $k(y)\to k(x)$ is bijective.
Then the preceding conditions are also equivalent to the following:
\begin{enumerate}
  \item[d)] $h$ is lisse at the point $x$, and the canonical application $\mathrm{T}_{X/S}(x)\to\mathrm{T}_{Y/S}(y)$ \sref{IV.16.5.12} from the tangent vector space at $x$ to the tangent vector space at $y$ to $Y$ is surjective.
\end{enumerate}
\end{theorem}

\begin{proof}
The fact that $a)$ implies $b)$ results trivially from \sref{IV.17.3.3},~(ii)); $b)$ implies $c)$ by application of \sref{IV.17.2.3},~(ii)).
To see that $c)$ is equivalent to $c'$), note that by virtue of \sref{IV.17.2.3},~(i)), $\Omega^1_{X/S}$ is an $\sh{O}_X$-Module locally free of finite type; it suffices to apply ($\mathbf{0}$, \sref{IV.19.1.12}) to the ring $\sh{O}_{X,x}$ and the homomorphism \sref{IV.17.11.1.1} of $\sh{O}_{X,x}$-modules of finite type.
When $k(x)=k(y)$, the tangent linear application $\mathrm{T}_{X/S}(x)\to\mathrm{T}_{Y/S}(y)$ is the transpose of \sref{IV.17.11.1.2}, by \sref{IV.16.5.12}, hence the equivalence of $c')$ and $d)$ in this case.

\oldpage[IV]{83}

It remains to prove that $c)$ implies $a)$.
One can restrict to the case where $S=\operatorname{Spec}(A)$, $Y=\operatorname{Spec}(B)$, $X=\operatorname{Spec}(C)$ are affine.
The hypothesis $c)$ implies (\sref{IV.17.2.3},~(i)) that $(\Omega^1_{X/S})_x=\Omega^1_{C_\mathfrak{m}/A}$ is a $C_\mathfrak{m}$-module locally free: indeed \sref{IV.16.10.6} there exist elements $t_i$ $(1\leq i\leq r)$ of $B_\mathfrak{p}=\sh{O}_{Y,y}$ such that the differentials $d_{B/A}(t_i)$ engender the $B_\mathfrak{p}$-module $\Omega^1_{B/A_\mathfrak{p}}$ and their images in $\Omega^1_{C_\mathfrak{m}/A}$ are part of a base of this $C_\mathfrak{m}$-module.
As $\Omega^1_{Y/S}$ (resp.\ $\Omega^1_{X/S}$) is an $\sh{O}_Y$-Module (resp.\ $\sh{O}_X$-Module) of finite presentation \sref{IV.16.4.22}, one can, replacing when needed $X$ and $Y$ by suitable open affine neighbourhoods of $x$ and $y$ respectively, suppose that $\Omega^1_{C/A}$ is a $C$-module libre and that the $t_i$ are images of elements $s_i$ $(1\leq i\leq r)$ of $B$ such that the $d_{B/A}(s_i)$ are part of a base of the $C$-module $\Omega^1_{C/A}$.
Let $\varphi$ be the A-homomorphism from $B'=A[\mathrm{T}_1,\ldots,\mathrm{T}_r]$ to $B$ such that $\varphi(\mathrm{T}_i)=s_i$ for every $i$; the di-homomorphism corresponding $\Omega^1_{B'/A}\to\Omega^1_{B/A}$ ($\mathbf{0}$, \sref{IV.20.5.2}) transforms the $d_{B'/A}(\mathrm{T}_i)$, which form a base of $\Omega^1_{B'/A}$ ($\mathbf{0}$, \sref{IV.20.4.13}), into the $d_{B/A}(s_i)$ and is hence \emph{surjective}; if $Y'=\operatorname{Spec}(B')$ and if $u:Y\to Y'$ is the S-morphism corresponding to $\varphi$, one concludes that $u$ is \emph{non ramified} \sref{IV.17.2.2}.
If one proves that the composed morphism $u\circ g:X\to Y'$ is \emph{\'etale} at the point $y$, then from \sref{IV.17.3.5} that $g$ is lisse at the point $x$; finally, as the structural morphism $f':Y'\to S$ is lisse \sref{IV.17.3.8}, $f=f'\circ u$ will be lisse at the point $y$ by \sref{IV.17.3.5}.
That for the proof that $c)$ implies $a)$, one sees thus that one can restrict to the case where $Y'=Y$, and hence suppose that $f$ is a smooth morphism \emph{lisse}.

By virtue of \sref{IV.17.8.2}, one can reduce to the case where $S=\operatorname{Spec}(k)$ is the spectrum of a field; moreover, by \sref{IV.16.4.5} and \sref{IV.17.7.1},~(ii)), one can suppose that $k$ is algebraically closed, and one can suppose that $f$ and $h$ are both lisse, and that the canonical homomorphism
\[

\]
is left invertible ($\mathbf{0}$, \sref{IV.19.1.12}).
Then the set of points of $X$ rational on $k$ is very dense in $X$ \sref{IV.10.4.8}, hence, to prove that $g$ is lisse, it suffices to prove that $g$ is lisse at every point $x\in X$ \emph{rational on $k$}, or equivalently that at such a point, $\sh{O}_{X,x}$ is an $\sh{O}_{Y,y}$-module flat and $\sh{O}_{X,x}/\mathfrak{m}_y\sh{O}_{X,x}$ a regular ring \sref{IV.17.5.1}.
Or, $y=g(x)$ is the image of $x$, B$=\sh{O}_{Y,y}$, the residue field of $B$ is equal to $k$, it is the same as that of $A$, hence $x$ (resp.\ $y$) is a closed point of $X$ (resp.\ $Y$) ($\mathbf{I}$, \sref{IV.6.4.2}), and one has $\dim(A)=\dim(B)=n$.
Since $f$ is non ramified \sref{IV.17.7.1}, the homomorphism $\hat{A}\to\hat{B}$ is surjective \sref{IV.17.4.4},~$f''$)); but as $\dim(\hat{A})=\dim(\hat{B})=n$, and $\hat{A}$ being a regular ring, integral, the homomorphism $\hat{A}\to\hat{B}$ is also \emph{injective} ($\mathbf{0}$, \sref{IV.16.3.10}).
This homomorphism is thus bijective, which achieves the proof that $f$ is \'etale at the point $x$ \sref{IV.17.6.3},~$e''$)).
\end{proof}

\begin{corollary}[17.11.3]

\end{corollary}

\begin{proof}
It suffices to apply \sref{IV.17.11.1} (resp.\ \sref{IV.17.11.2}) taking for $f$ the structural morphism $S[\mathrm{T}_1,\ldots,\mathrm{T}_r]\to S$.
\end{proof}

\begin{corollary}[17.11.4]
\label{IV.17.11.4}
For a morphism $h:X\to S$ to be lisse at a point $x\in X$, it is necessary and sufficient that there exist an open neighbourhood $U$ of $x$, an integer $r$, and an S-morphism \'etale $U\to S[\mathrm{T}_1,\ldots,\mathrm{T}_n]$.
\end{corollary}

\begin{proof}
The condition is evidently sufficient since the structural morphism $S[\mathrm{T}_1,\ldots,\mathrm{T}_n]\to S$ is lisse \sref{IV.17.3.8}.
To show that it is necessary, note that since $h$ is lisse at the point $x$, there is an open neighbourhood $U$ of $x$ such that $\Omega^1_{X/S}|U$ is locally free \sref{IV.16.10.6} for obtaining (restricting to $U$) sections $s_i$ of $\sh{O}_X$ above $U$ such that the $d_{X/S}(s_i)$ form a base of $\Omega^1_{X/S}|U$; one concludes by applying \sref{IV.17.11.3}.
\end{proof}

\begin{proposition}[17.11.5]
\label{IV.17.11.5}
Let $f:Y\to S$, $h:X\to S$ be two smooth morphisms.
For an S-morphism $g:X\to Y$ to be an open immersion, it is necessary and sufficient that $g$ be a monomorphism of preschemes and that for every $x\in X$, if one sets $y=g(x)$, one has $\dim_y(f)=\dim_x(h)$ (for a generalisation of this proposition, see \sref{IV.18.10.5}).
\end{proposition}

\begin{proof}
The condition is evidently necessary; let us prove that it is sufficient.
By virtue of \sref{IV.17.9.5}, one is immediately reduced to the case where $S=\operatorname{Spec}(k)$ is the spectrum of a field, and by virtue of \sref{IV.2.7.1},~(x)), one can suppose $k$ algebraically closed.
By virtue of \sref{IV.17.9.1}, it suffices to prove that $g$ is \'etale at the \emph{closed} points (or equivalently, rational on $k$) of $X$ \sref{IV.10.4.8}.
So let $x$ be such a point, and set $y=g(x)$, which is also rational on $k$ and hence closed.
Set $A=\sh{O}_{Y,y}$, $B=\sh{O}_{X,x}$, which are regular local rings, and let $k$ be the residue field of $A$.
Let $d=\dim(A)$ and $(t_i)_{1\leq i\leq d}$ a regular system of parameters for $A$.
Set $B=\sh{O}_{X,x}$, $C=B/\mathfrak{m}_y B$.
As $g$ is a monomorphism, so is the morphism $\operatorname{Spec}(C)\to\operatorname{Spec}(k(y))=\operatorname{Spec}(k)$ deduced from $g$ by base change ($\mathbf{I}$, \sref{IV.3.3.12}); but this means the corresponding homomorphism $u:k\to C$ is \emph{surjective} (hence bijective), since $u$ admits a left inverse $v:C\to k$, and $u\circ v$ and the identity of $k\to C$ give the same morphism, which, composed with $u$, gives the identity of $k\to C$.
As by hypothesis $B$ is a regular ring of dimension $d$, the images of the $t_i$ in $B$ form a regular system of parameters for $B$ ($\mathbf{0}$, \sref{IV.17.1.7}); the condition \sref{IV.17.6.3},~$e''$)) is thus verified by $g$ at the point $x$, which achieves the proof.
\end{proof}

\begin{theorem}[17.12.1]
\label{IV.17.12.1}
Let $f:X\to S$, $h:Y\to S$ be two morphisms locally of finite presentation, $j:Y\to X$ an immersion, $y$ a point of $Y$, $x=j(y)$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] $h$ is lisse at the point $y$ and $f$ is lisse at the point $x$.
  \item[b)] $f$ is lisse at the point $x$, and the canonical homomorphism \sref{IV.16.4.21}
\[
(\sh{N}_{Y/X})_y \to (j^*(\Omega^1_{X/S}))_y
\]
is left invertible.
  \item[c)] $h$ is lisse at the point $y$ and there exists an open neighbourhood $U$ of $y$ in $Y$ such that $j|U:U\to X$ is a regular immersion \sref{IV.16.9.2}.
\end{enumerate}
\end{theorem}

\begin{corollary}[17.12.2]
\label{IV.17.12.2}
With the notations of \sref{IV.17.12.1}, suppose that $f$ is lisse at the point $x$, and let $Y$ be a closed sub-prescheme of $X$ defined by an Ideal $\sh{J}$ of $\sh{O}_X$, and S-lisse at the point $x$.
Let $(g_i)_{1\leq i\leq r}$ be sections of $\sh{J}$ above $X$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] The $(g_i)_x$ form a system of generators of the $\sh{O}_{X,x}$-module $\sh{J}_x$ whose number of elements is the smallest possible.
  \item[b)] If $g'_i$ is the canonical image of $g_i$ in $\Gamma(X,\sh{J}/\sh{J}^2)$, the $(g'_i)_x$ form a base of the $(\sh{O}_{X,x}/\sh{J}_x)$-module $\sh{J}_x/\sh{J}_x^2$.
  \item[c)] The images of the $d_{X/S} g_i$ in $(\Omega^1_{X/S})_x\otimes_{\sh{O}_{X,x}} k(x)$ are linearly independent over $k(x)$, and the $(g_i)_x$ engender $\sh{J}_x$.
  \item[d)] There exists an open neighbourhood $U$ of $x$ in $X$ and sections $g_j$ $(r+1\leq j\leq n)$ of $\sh{O}_X$ above $U$ such that the S-morphism $u:U\to S[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ corresponding to the homomorphism $\sh{O}^n_S\to f_*(\sh{O}_U)$ defined by the sections $g_i|U$ $(1\leq i\leq r)$ and $g_j$ $(r+1\leq j\leq n)$ is an \'etale morphism, and such that the reciprocal image of the sub-prescheme $Y'=S[\mathrm{T}_{r+1},\ldots,\mathrm{T}_n]$ is the open $U\cap j(Y)$.
\end{enumerate}
\end{corollary}

\begin{proof}
As $(g'_i)_x$ is the canonical image of $(g_i)_x$ and $b)$ results from \sref{IV.17.10.4} (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection~3, n\textsuperscript{o}~2, prop.~5).
By virtue of \sref{IV.17.12.1},~$b$)), $\sh{J}_x/\sh{J}_x^2$ is a free $(\sh{O}_{X,x}/\sh{J}_x)$-module libre of rank $r$; and the equivalence of $b)$ and $c)$ results from Bourbaki, \emph{loc.~cit.}
Moreover, if $a)$ is verified, $(g'_i)_x$ is the image of $(g_i)_x$ in $\sh{O}_{X,x}/\sh{J}_x$; the equivalence of $a)$ and $b)$ follows from the fact that $\sh{J}_x$ is of finite type and $\sh{J}_x/\sh{J}_x^2$ free, by virtue of the Nakayama lemma.

\oldpage[IV]{86}

Moreover, if $a)$ is verified, the images of the $d_{X/k} g_i$ engender $(\Omega^1_{X/k})_x\otimes 1$ for $1\leq i\leq r$; the $d_i$ for $r+1\leq i\leq n$ already engender $\Omega^1_{k(x)/k}$.
As $(\Omega^1_{X/S})_x$ is free of rank $n$, it results from Bourbaki, \emph{loc.~cit.}, that replacing $X$ by an open neighbourhood of $x$, one can suppose that there exist $n-r$ sections $g_i$ $(r+1\leq i\leq n)$ of $\sh{O}_X$ above $X$ such that the $d_{X/S} g_i$ for $1\leq i\leq n$ form a base of $(\Omega^1_{X/S})_x$; the fact that the corresponding \'etale morphism $u$ is \'etale at the point $x$ results from \sref{IV.17.11.3}; it is then \'etale in a neighbourhood of $x$; replacing $X$ by this neighbourhood, $u$ can be supposed \'etale.
Moreover, it is immediate that $Y$ (identified, by restriction of $j(Y)$ to $X$) is a closed sub-prescheme of $u^{-1}(Y')$, and by virtue of \sref{IV.17.2.5} and \sref{IV.17.11.2}, the restriction to $Y$ of $u$ can be supposed \'etale.
One deduces that for every $y'\in Y'$, $Y\cap u^{-1}(y')$ is an open immersion \sref{IV.17.9.6}), whence $Y\to u^{-1}(Y')$ is an open immersion, and we conclude by \sref{IV.17.11.3} that this proves that $a)$ implies $d)$.
The reciprocal $d)\Rightarrow a)$ follows from \sref{IV.17.11.3}.
\end{proof}

\begin{corollary}[17.12.3]

Let $f:X\to S$ be a morphism locally of finite presentation, $u:S\to X$ an S-section of $X$.
For $f$ to be lisse at a point $x\in X$, it is necessary and sufficient that $u$ be a quasi-regular immersion in a neighbourhood of $s=f(x)$.
\end{corollary}

\begin{proposition}[17.12.4]
\label{IV.17.12.4}
Every smooth morphism $f:X\to S$ is differentially smooth (in other words \sref{IV.16.10.5} $\Delta_f:X\to X\times_S X$ is a quasi-regular immersion).
In particular, the $\sh{P}^n_f=\sh{P}^n_{X/S}$ and the $\operatorname{gr}_n(\sh{P}_{X/S})$ are $\sh{O}_X$-Modules locally free of finite type.
\end{proposition}

\begin{proof}
It suffices to note that the structural morphism $p_1:X\times_S X\to X$ is lisse \sref{IV.17.3.3},~(iii)) and that $\Delta_f$ is an X-section for $p_1$; the conclusion results from \sref{IV.17.12.3}.
\end{proof}

\begin{proposition}[17.12.5]
\label{IV.17.12.5}
Let $f:X\to Y$ be a morphism locally of finite presentation.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] $f$ is differentially smooth.
  \item[b)] For every morphism $g:Y'\to Y$, if one sets $X'=X\times_Y Y'$ and $f'=f_{(Y')}:X'\to Y'$, for every Y'-section $s'$ of $X'$, $f'$ is lisse at all points of $s'(Y')$.
  \item[c)] The second projection $p_2:X\times_Y X\to X$ is lisse at all the points of the diagonal $\Delta_f(X)$.
\end{enumerate}
\end{proposition}

\oldpage[IV]{87}

\begin{proof}
The condition $c)$ is a particular case of $b)$: it suffices to take $Y'=X$ in $b)$, $f'$ is then the second projection $p_2$, and $\Delta_f$ is an $X$-section for $p_2$.
In second place, $c)$ implies $a)$, since $p_2$ is a morphism locally of finite presentation, and if $p_2$ is lisse at the points of $\Delta_f(X)$, $\Delta_f$ is a quasi-regular immersion \sref{IV.17.12.3}, hence $f$ is differentially smooth.
It remains to show that $a)$ implies $b)$.
One can suppose that $Y'=Y$, and $f'=f$, and hence show that if $f$ is differentially smooth, then for every Y-section $s$ of $X$, $f$ is lisse at the points of $s(Y)$.
Or, if $m:G\times_S G\to G$ is the morphism defining the group prescheme structure, $m\circ(s\times 1_G):S\times_S G\to G\times_S G\to G$ is an S-isomorphism of $G$, ``left translation'' $\tau_s$, transforming the identity section of $G$ into the points of $s(S)$; hence one concludes that $G$ is lisse on $S$ at the points of $s(S)$.
\end{proof}

\begin{corollary}[17.12.6]
\label{IV.17.12.6}
With the notations being those of \sref{IV.8.8.1}, suppose $X_\alpha$ quasi-compact, and $f_\alpha:X_\alpha\to S_\alpha$ locally of finite presentation.
For $f:X\to S$ to be differentially lisse, it is necessary and sufficient that there exist $\lambda\geq\alpha$ such that $f_\lambda:X_\lambda\to S_\lambda$ be differentially lisse (in which case $f_\mu:X_\mu\to S_\mu$ is differentially lisse for $\mu\geq\lambda$).
\end{corollary}

\begin{proof}
The sufficiency of the condition and the last assertion result from \sref{IV.16.10.4}.
To prove that the condition is necessary, note that $\Delta_f:X\to X\times_S X$ and $p_2:X\times_S X\to X$ are deduced by change of base from $\Delta_{f_\lambda}:X_\lambda\to X_\lambda\times_{S_\lambda} X_\lambda$ and $p_{2,\lambda}:X_\lambda\times_{S_\lambda}X_\lambda\to X_\lambda$.
By virtue of \sref{IV.17.12.5}, for every $z\in X\times_S X$ in $\Delta_f(X)$, there exists an affine open $U(z)$ of $z$ in $X\times_S X$ such that $p_2$ is lisse in $U(z)$; by virtue of \sref{IV.8.2.11} and \sref{IV.17.7.8}, there exists an index $\lambda(z)$ and a neighbourhood $U_{\lambda(z)}$ of $z$ in $X_{\lambda(z)}\times_{S_{\lambda(z)}} X_{\lambda(z)}$ such that $p_{2,\lambda(z)}$ is lisse in $U_{\lambda(z)}$.
As $X$ is quasi-compact, one can cover $\Delta_f(X)$ by a finite number of such neighbourhoods; by virtue of \sref{IV.8.3.4}, there exists a $\lambda$ superior to all the $\lambda(z_i)$ such that the inverse images of the $U_{\lambda(z_i)}$ in $X_\lambda\times_{S_\lambda}X_\lambda$ form a covering; it results then from \sref{IV.17.12.5} that this index $\lambda$ answers the question.
\end{proof}

\begin{example}[17.12.7]
\label{IV.17.12.7}
Let $S$ be a prescheme, $G$ an S-\emph{prescheme in groups}, locally of finite presentation on $S$.
For $G$ to be differentially smooth on $S$, it is necessary and sufficient that $G$ be lisse on $S$ at all the points of the ``identity section'' $e$ of $G$.
The condition is indeed necessary by \sref{IV.17.12.5}.
Inversely, suppose it is fulfilled; for every morphism $S'\to S$, $G'=G\times_S S'$ is then an $S'$-prescheme in groups locally of finite presentation on $S'$; on can therefore, by virtue of \sref{IV.17.12.5}, restrict to proving that for every S-section $s$ of $G$, $f$ is lisse at the points of $s(S)$.
But if $m:G\times_S G\to G$ is the morphism that defines the group prescheme structure, $m\circ(s\times 1_G):S\times_S G\to G\times_S G\to G$ is an S-isomorphism of $G$, the ``left translation'' $\tau_s$, transforming the identity section of $G$ into the points of $s(S)$; whence one concludes that $G$ is lisse on $S$ at the points of $s(S)$.
\end{example}

\begin{remark}[17.12.8]
\label{IV.17.12.8}
An S-prescheme in groups $G$, of finite type (and even finite) on a locally Noetherian prescheme $S$, can be differentially smooth on $S$ without being lisse (or even \emph{flat}) on $S$.
Consider for example for $S$ the spectrum of the dual number ring $D=k[\mathrm{T}]/\mathrm{T}^2 k[\mathrm{T}]$ over a field $k$, for $G$ the spectrum of the $D$-algebra direct composition $E=D\oplus k$.
One defines on $G$ a structure of $S$-scheme in groups by defining a ``diagonal application'' $\delta$, homomorphism of $E$ in $E\otimes_D E$: if $e'$, $e''$ are the idempotents, canonical images of $1$ of $D$ in the factors $D$ and $k$ of $E$, one easily verifies that such an application is obtained by taking $\delta(e')=e'\otimes e'+e''\otimes e''$ and $\delta(e'')=e'\otimes e''+e''\otimes e'$.
The identity section of the group scheme $G$ corresponds to the homomorphism $E\to D$ which is the identity in $D$ and $0$ in $k$; it is an isomorphism from $S$ onto a connected component of $G$, and \emph{a fortiori} $G$ is differentially lisse on $S$ \sref{IV.17.12.6}, but it is clear that $G$ is not S-flat.
\end{remark}

where the lower line is the exact suite \sref{IV.16.4.21} applied to $j$, the upper line comes from the same exact suite for $i$, by application of the exact functor to the right $g^*$ (which thus leaves it exact); $\operatorname{gr}_1(g)$ is defined in \sref{IV.16.2.1}, and the commutativity of the two squares results from ($\mathbf{0}$, \sref{IV.20.5.7.3}) and ($\mathbf{0}$, \sref{IV.20.5.11.3}).

\begin{proposition}[17.13.2]
\label{IV.17.13.2}
With the notations of \sref{IV.17.13.1}, let $x'$ be a point of $Y'$, $x=g(x')$ its image in $Y$; suppose $X$ and $Y$ lisse on $S$ at the point $x$, $X'$ lisse on $S$ at the point $x'$.
Let $m$, $m-c$ be the relative dimensions of $X$ and $Y$ on $S$ at the point $x$ \sref{IV.17.10.1}, $n$ that of $X'$ on $S$ at the point $x'$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] $Y'$ is lisse on $S$ at the point $x'$ and of relative dimension $n-c$.
  \item[b)] The homomorphism $\alpha\otimes 1:g^*(\sh{N}_{Y/X})\otimes_{\sh{O}_{Y'}} k(x') \to \Omega^1_{X'/S}\otimes_{\sh{O}_{X'}} k(x')$ deduced from the homomorphism $\alpha$ of \sref{IV.17.13.1.3} is injective.
\end{enumerate}
Let $s$ be the image of $x$ in $S$, and suppose that $x'$ is rational on $k(s)$ ($s$ being the image of $x'$ in $S$).
Then the conditions $a)$ and $b)$ are also equivalent to the following:
\begin{enumerate}
  \item[b')] The transposed homomorphism of $\alpha\otimes 1$
\[
\mathrm{T}_{X'/S}(x') \to (\sh{N}_{Y/X}\otimes_{\sh{O}_Y} k(x'))^\vee = \mathrm{T}_{X/S}(x)/\mathrm{T}_{Y/S}(x)
\]
(cf.\ \sref{IV.16.5.12}) is surjective.
\end{enumerate}
\end{proposition}

\begin{definition}[17.13.3]
\label{IV.17.13.3}
With the notations of \sref{IV.17.13.1}, we say that the morphism $f$ is \emph{transversal} to $Y$ at the point $x'$, relatively to $S$, if $X$ and $Y$ are lisse on $S$ at the point $x$, $X'$ lisse on $S$ at the point $x'$, and the equivalent conditions $a)$, $b)$ of \sref{IV.17.13.2} are satisfied.
When there is no confusion to fear, we suppress the mention of the prescheme $S$ and simply say that $f$ is transversal to $Y$ at the point $x'$.
\end{definition}

\begin{remarks}[17.13.4]
\label{IV.17.13.4}
(i)
Suppose that $X$, $Y$ and $X'$ are flat and locally of finite presentation on $S$.
For every $s\in S$, denote $X_s$, $Y_s$, $X'_s$, $Y'_s$ the fibres of $X$, $Y$, $X'$, $Y'$ at the point $s$, $f_s:X'_s\to X_s$ the morphism deduced from $f$ by change of base.
It results then from \sref{IV.17.5.1} that for $f$ to be a transversal morphism to $Y$ at the point $x'$, it is necessary and sufficient that, if $s$ is the image of $x$ in $S$, $f_s$ be transversal to $Y_s$ at the point $x'$, relatively to $k(s)$.\footnote{[Trans.] The original reference (17.5.9) is an erratum in the original; there is no item~17.5.9. The correct reference is (17.5.1).}

(ii)
One has seen in \sref{IV.17.13.2} that the set of points $x'\in Y'$ where $f$ is transversal is open in $Y'$; if $Y'$ is moreover \emph{proper} on $S$, one deduces that the set of $s\in S$ such that $f_s$ is transversal to $Y$ at \emph{all} the points of $Y'_s$, relatively to $S$, is \emph{open} in $S$.

(iii)
The property of being lisse in a point, as well as the notion of relative dimension on $S$ in a point, being stable by every change of base $S'\to S$ (\sref{IV.17.3.3} and \sref{IV.4.2.7}), the same holds for the property of a morphism $f:X'\to X$ of being transversal to a sub-prescheme of $X$ at a point, relatively to $S$.

(iv)
The condition $b)$ of \sref{IV.17.13.2} also expresses that the homomorphism $\alpha:g^*(\sh{N}_{Y/X})\to\Omega^1_{X/S}\otimes_{\sh{O}_x}\sh{O}_{Y'}$ is \emph{universally injective} with respect to $Y'$ \sref{IV.11.9.18}.
\end{remarks}

\begin{corollary}[17.13.6]
\label{IV.17.13.6}
Let $S$ be a prescheme, $X$, $Y$, $Z$ three $S$-preschemes, $f:Y\to X$, $g:Z\to X$ two $S$-morphisms; set $T=Y\times_X Z$, and let $p:T\to Y$, $q:T\to Z$ be the canonical projections.
Let $t$ be a point of $T$, $y=p(t)$, $z=q(t)$, $x=f(y)=g(z)$.
Suppose that $X$ is smooth over $S$ at the point $x$, of relative dimension $m$, $Y$ (resp.\ $Z$) smooth over $S$ at the point $y$ (resp.\ $z$), of relative dimension $m+a$ (resp.\ $m+b$), $a$ and $b$ being positive or negative integers.
Then the following conditions are equivalent:
\begin{enumerate}[label=\alph*)]
  \item $T$ is smooth over $S$ at the point $t$, of relative dimension $m+a+b$.
  \item The homomorphism
  \[
    \rho\otimes 1: \Omega^1_{X/S}\otimes_{\sh{O}_x}k(t) \to (\Omega^1_{Y/S}\otimes_{\sh{O}_y}k(t))\oplus(\Omega^1_{Z/S}\otimes_{\sh{O}_z}k(t))
  \]
  where $\rho$ is given by \sref{IV.17.13.5.4}, is injective.
  \item The morphism $T=Y\times_S Z\to X\times_S X$ is transversal to the diagonal of $X\times_S X$ at the point $t$.
\end{enumerate}
If $t$ is rational over the residue field of its image $s$ in $S$, these conditions are also equivalent to the following:
\begin{enumerate}
  \item[b${}'$)] The homomorphism
  \[
  \label{IV.17.13.6.1}
  \tag{17.13.6.1}
    \mathrm{T}_y(f)-\mathrm{T}_s(g): \mathrm{T}_{Y/S}(y)\oplus\mathrm{T}_{Z/S}(z) \to \mathrm{T}_{X/S}(x)
  \]
  (cf.\ \sref{IV.16.5.12.5}) is surjective.
\end{enumerate}
Moreover, when conditions \emph{a)} and \emph{b)} are verified at $t$, they are so in a neighbourhood of $t$ in $T$, and restricting $T$ to such a neighbourhood, the sequence
\[
\label{IV.17.13.6.2}
\tag{17.13.6.2}
  0 \to \Omega^1_{X/S}\otimes_{\sh{O}_x}\sh{O}_T \xrightarrow{\rho} (\Omega^1_{Y/S}\otimes_{\sh{O}_y}\sh{O}_T)\oplus(\Omega^1_{Z/S}\otimes_{\sh{O}_z}\sh{O}_T) \xrightarrow{\sigma} \Omega^1_{T/S} \to 0
\]
(where $\sigma$ is given by \sref{IV.17.13.5.3}) is exact.

The only point remaining to prove in \sref{IV.17.13.6} concerns $b')$, and results from the fact that, under the hypothesis that $t$ is rational over $k(s)$, the homomorphism $\rho\otimes 1$ is the \emph{transpose} of $\mathrm{T}_y(f)-\mathrm{T}_s(g)$, by \sref{IV.17.13.5.4}.

When the equivalent conditions of \sref{IV.17.13.6} are satisfied, one says that $f$ and $g$ form a \emph{couple of transversal morphisms at the point $t$, relatively to} $S$; here again, one often suppresses the mention of $S$.
\end{corollary}

\begin{proposition}[17.13.8]
\label{IV.17.13.8}
Let $S$ be a prescheme, $X$ an $S$-prescheme locally of finite presentation over $S$, $Y$, $Z$ two sub-preschemes of $X$, $T=\inf(Y,Z)$ the sub-prescheme ``intersection'' of $Y$ and $Z$, $x$ a point of $T$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] The canonical injection $i:Y\to X$ is a transversal morphism to $Z$ at the point $x$, relatively to $S$ \sref{IV.17.13.3}.
  \item[a${}'$)] The canonical injection $j:Z\to X$ is a transversal morphism to $Y$ at the point $x$, relatively to $S$.
  \item[a${}''$)] $Y$ and $Z$ cut transversally at the point $x$, relatively to $S$.
  \item[b)] The homomorphism
  \[
    \rho\otimes 1: \Omega^1_{X/S}\otimes_{\sh{O}_x}k(x) \to (\Omega^1_{Y/S}\otimes_{\sh{O}_y}k(x))\oplus(\Omega^1_{Z/S}\otimes_{\sh{O}_z}k(x))
  \]
  is injective.
\end{enumerate}
When $x$ is rational over the residue field $k(s)$ ($s$ image of $x$ in $S$), these conditions are also equivalent to the following:
\begin{enumerate}
  \item[b${}'$)] The homomorphism
  \[
    \mathrm{T}_x(i)-\mathrm{T}_x(j): \mathrm{T}_{Y/S}(x)\oplus\mathrm{T}_{Z/S}(x) \to \mathrm{T}_{X/S}(x)
  \]
  is surjective.
\end{enumerate}
Moreover, when conditions \emph{a)} and \emph{b)} are verified at $x$, they are so in a neighbourhood of $x$ in $T$, and restricting $X$ to a neighbourhood of $x$, the sequence \sref{IV.17.13.6.2} is exact, and one has a canonical isomorphism
\[
\label{IV.17.13.8.1}
\tag{17.13.8.1}
\sh{N}_{T/X} \xrightarrow{\sim} (\sh{N}_{Y/X}\otimes_{\sh{O}_y}\sh{O}_T)\oplus(\sh{N}_{Z/X}\otimes_{\sh{O}_z}\sh{O}_T).
\]
\end{proposition}

\begin{proof}
\oldpage[IV]{95}
The conditions $a)$, $a')$, $a'')$ all imply that $X$, $Y$, $Z$ are smooth over $S$ at the point $x$.
Let moreover $m$, $m-a$, $m-b$ be the relative dimensions of $X$, $Y$, $Z$ over $S$ at the point $x$.
It results then from \sref{IV.17.13.2} applied replacing $X'$ by $Z$ and $Y'$ by $T=Y\times_X Z$, that the conditions $a)$, $a')$ are both equivalent to saying that $T$ is smooth over $S$ at the point $x$ and of relative dimension $m-a-b$ at this point; but by \sref{IV.17.13.7}, this means precisely that condition $a'')$ is verified, whence the equivalence of $a)$, $a')$ and $a'')$.
The equivalence of $a'')$ and $b)$ (or $b')$ when $x$ is rational over $k(s)$) has been proved in \sref{IV.17.13.6}, as well as the fact that if these conditions are satisfied at the point $x$, they are so in a neighbourhood of $x$, and the exactness of the sequence \sref{IV.17.13.6.2} in such a neighbourhood.
It remains to define the canonical isomorphism \sref{IV.17.13.8.1}.
Let us denote $\alpha$ and $-\beta$ the homomorphisms appearing in the second member of \sref{IV.17.13.5.4}, $\gamma$ and $\delta$ those appearing in the second member of \sref{IV.17.13.5.2}.
Saying that the sequence \sref{IV.17.13.6.2} is exact signifies that, in the category of $\sh{O}_T$-Modules, $\Omega^1_{X/S}\otimes_{\sh{O}_x}\sh{O}_T$ identifies canonically with the \emph{fibre product} of $\Omega^1_{Y/S}\otimes_{\sh{O}_y}\sh{O}_T$ and $\Omega^1_{Z/S}\otimes_{\sh{O}_z}\sh{O}_T$ over $\Omega^1_{T/S}$, for the homomorphisms $\gamma$ and $\delta$.
The same reasoning as in ($\mathbf{0}$, 18.1.2 and 18.1.3), where one replaces rings and ideals respectively by Modules and sub-Modules, furnishes a commutative diagram where the 3rd and 4th rows and columns are exact by the smoothness hypotheses.
The fact that the composed homomorphism $\varepsilon=\gamma\circ\alpha=\delta\circ\beta$ is the homomorphism corresponding to the canonical injection $T\to X$ results from ($\mathbf{0}$, 20.5.2.7).
As the diagonal of the preceding diagram is exact, $\operatorname{Ker}(\varepsilon)$ identifies canonically with $\sh{N}_{T/X}$, taking into account that $T$ is smooth over $S$ \sref{IV.17.2.5}; hence one has a canonical isomorphism \sref{IV.17.13.8.1} which is the reciprocal of $\iota$.
\end{proof}

\begin{corollary}[17.13.10]

  (where $\delta$ is the diagonal application) is surjective.

It suffices to note that $X^I$ is smooth over $S$ at the point $x$ and of relative dimension $d\cdot\operatorname{Card}(I)$, and $Y$ smooth over $S$ at the point $t$ and of relative dimension $d\cdot\operatorname{Card}(I)+\sum_{i\in I} c_i$.

\label{IV.17.13.11.1}
\tag{17.13.11.1}
\codim_x(\mathrm{T}_s,\mathrm{X}_s) = \sum_{i\in I}\codim_x((\mathrm{Y}_i)_s,\mathrm{X}_s).
\]

Moreover, one has the following property which extends \sref{IV.17.13.8.1}, and of which we give another demonstration when $I$ has $2$ elements:
\end{env}

\begin{corollary}[17.13.12]
\label{IV.17.13.12}
When the $Y_i$ cut transversally at the point $x$, one has a canonical isomorphism
\[
\label{IV.17.13.12.1}
\tag{17.13.12.1}
\sh{N}_{T/X} \xrightarrow{\sim} \bigoplus_{i\in I}(\sh{N}_{Y_i/X}\otimes_{\sh{O}_{y_i}}\sh{O}_T).
\]
\end{corollary}

\begin{proof}
One can reduce to the case where the $Y_i$ are closed sub-preschemes of $X$, defined by quasi-coherent Ideals $\sh{J}_i$, so that $T$ is defined by the Ideal $\sh{J}=\sum_i\sh{J}_i$.
By definition of the conormal sheaf of an immersion \sref{IV.16.1.3}, the canonical homomorphism $\bigoplus_i\sh{J}_i\to\sh{J}$ gives, by passage to quotients, a \emph{surjective} homomorphism
\[
\label{IV.17.13.12.2}
\tag{17.13.12.2}
\bigoplus_{i\in I}(\sh{N}_{Y_i/X}\otimes_{\sh{O}_{y_i}}\sh{O}_T) \to \sh{N}_{T/X}.
\]
\oldpage[IV]{98}
But here the $\sh{O}_T$-Modules on both sides of \sref{IV.17.13.12.2} are locally free and of the same rank (since $c_i$ is the rank of $\sh{N}_{Y_i/X}$), by \sref{IV.17.2.5} and the condition $a)$ of \sref{IV.17.13.10}; one concludes then from Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection3, n\textsuperscript{o}~2, cor.\ de la prop.~6, that \sref{IV.17.13.12.2} is bijective, and \sref{IV.17.13.12.1} is the reciprocal isomorphism.
\end{proof}

\subsection{Local and infinitesimal characterisations of smooth morphisms, of non ramified morphisms and of \'etale morphisms.}
\label{subsection:IV.17.14}

\oldpage[IV]{98}

\begin{proposition}[17.14.1]
\label{IV.17.14.1}
Let $f:X\to Y$ be a morphism locally of finite presentation, $x$ a point of $X$, $y=f(x)$.
For $f$ to be lisse (resp.\ non ramified, resp.\ \'etale) at the point $x$, it is necessary and sufficient that the following condition be satisfied:

For every local scheme $Y'=\operatorname{Spec}(A')$ of closed point $y'$, every morphism $h:Y'\to Y$ such that $h(y')=y$, every closed sub-scheme $Y'_0=\operatorname{Spec}(A'/\mathfrak{J})$ of $Y'$, where $\mathfrak{J}$ is an Ideal of square zero, and every Y-morphism $g_0:Y'_0\to X$ such that $g_0(y')=x$, there exists \emph{at least one} (resp.\ \emph{at most one}, resp.\ \emph{a unique}) Y-morphism $g:Y'\to X$ whose restriction to $Y'_0$ is $g_0$.
\end{proposition}

\begin{proof}
Taking into account the definitions \sref{IV.17.1.1} and \sref{IV.17.3.1}, it suffices to show that the condition is sufficient in the cases of smooth morphisms and non ramified morphisms respectively.

\emph{(i) Case of smooth morphisms.}
One can restrict to the case where $Y=\operatorname{Spec}(A)$ and $X=\operatorname{Spec}(C)$ are affine, with $B=A[\mathrm{T}_1,\ldots,\mathrm{T}_n]$ an $A$-algebra of polynomials and $C$ a quotient $B/\mathfrak{R}$, $\mathfrak{R}$ being a finitely generated ideal of $B$.
Using \sref{IV.17.7.2},~(ii)) and \sref{IV.17.7.9}, one can further suppose $S$, $X$ and $Y$ are \emph{Noetherian}; finally by virtue of ($\mathbf{0_{III}}$, \sref{IV.9.2.3}), one is reduced to the Noetherian case.
It suffices to establish that $\sh{O}_{X,x}$ is an $\sh{O}_{Y,y}$-algebra formally smooth for the \emph{pre-adic topologies}.
But as $\mathfrak{m}'$ is nilpotent, one sees that the condition of the statement is satisfied when one replaces the condition $\mathfrak{m}'\mathfrak{J}=0$ by $\mathfrak{J}\subset\mathfrak{m}'$; and considering the rings $A'/\mathfrak{m}'^i\mathfrak{J}$ and reasoning by recurrence on $j$, this identity results from the hypothesis applied, by recurrence on $n$, to $A'=C/\mathfrak{r}^n$ and $\mathfrak{J}'=(r^n+(\mathfrak{J}/\mathfrak{J}^2))/\mathfrak{r}^n$.

\emph{(ii) Case of non ramified morphisms.}
We take up the notations of the proof of \sref{IV.17.14.1},~(ii)); to prove that $\mathfrak{J}/\mathfrak{J}^2$ of $C$ is null, note that $C$ is then a local Noetherian ring, being the local ring of an affine open sub-prescheme of $X\times_Y X$, which is locally of finite type on $X$, hence on $Y$.
The hypothesis, applied to $A'=C$ and $\mathfrak{J}'=\mathfrak{J}/\mathfrak{J}^2$, shows that the two applications $B\xrightarrow{p_1}C\to C/\mathfrak{r}^n$ and $B\xrightarrow{p_2}C\to C/\mathfrak{r}^n$ are identical; but as $C/\mathfrak{r}^n$ is artinian, this identity results from the hypothesis applied to $A'=C/\mathfrak{r}^n$, hence $\Omega^1_{B/A}=0$ by ($\mathbf{0}$, \sref{IV.20.4.7}).
\end{proof}

\begin{proposition}[17.14.2]
\label{IV.17.14.2}
Let $Y$ be a prescheme locally Noetherian, $f:X\to Y$ a morphism locally of finite type, $x$ a point of $X$, $y=f(x)$.
For $f$ to be lisse (resp.\ non ramified, resp.\ \'etale) at the point $x$, it is necessary and sufficient that it satisfies the condition of \sref{IV.17.14.1} where one supposes that the local ring $A'$ is artinian, that $\mathfrak{m}'\mathfrak{J}=0$, where $\mathfrak{m}'$ is the maximal ideal of $A'$, and that the residue field $A'/\mathfrak{m}'$ of $A'$ is equal to $k(x)$.
\end{proposition}

\begin{remark}[17.14.3]
\label{IV.17.14.3}
One notes that, under the hypotheses of \sref{IV.17.14.2}, if moreover the residue field $k(x)$ is a \emph{finite} extension of $k(y)$, then the $A'$-modules $\mathfrak{m}'^j/\mathfrak{m}'^{j+1}$ are $k(y)$-vector spaces of finite rank, hence \emph{a fortiori} $A'$ is an $\sh{O}_{Y,y}$-algebra finite.
\end{remark}

\begin{proposition}[17.15.1]
\label{IV.17.15.1}
Let $k$ be a field, $X$ a prescheme locally of finite type on $k$, $x$ a point of $X$.
For $X$ to be smooth at a point $x$, it is necessary and sufficient that for every finite radiciel extension $k'$ of $k$ (or only for every finite radiciel extension $k'$ of $k$), the local ring $\sh{O}_{X,x}\otimes_k k'$ be regular.

\begin{corollary}[17.15.2]
\label{IV.17.15.2}
Under the conditions of \sref{IV.17.15.1}, for $X$ to be lisse on $k$, it is necessary and sufficient that $X$ be geometrically regular on $k$ \sref{IV.6.7.6}, which implies that $X$ is reduced.
If $k$ is perfect, $X$ is lisse on $k$ if and only if $X$ is regular.
\end{corollary}

\begin{proposition}[17.15.3]
\label{IV.17.15.3}
Let $k$ be a field, $X$ a prescheme locally of finite type on $k$, $x$ a point of $X$, $n=\dim_x(X)$.
Let $(g_i)_{1\leq i\leq n}$ be a family of $n$ sections of $\sh{O}_X$ above $X$, and let $f:X\to Y=\operatorname{Spec}(k[\mathrm{T}_1,\ldots,\mathrm{T}_n])$ be the morphism corresponding to the $k$-homomorphism $k[\mathrm{T}_1,\ldots,\mathrm{T}_n]\to\Gamma(X,\sh{O}_X)$ sending $\mathrm{T}_i$ to $g_i$.
The following conditions are equivalent (and imply that $X$ is lisse on $k$ at the point $x$ and hence regular at the point $x$):
\begin{enumerate}
  \item[a)] $f$ is \'etale at the point $x$.
  \item[b)] The images $d_{X/k} g_i$ form a base of the $\sh{O}_{X,x}$-module $(\Omega^1_{X/k})_x$.
  \item[c)] The images $d_{X/k} g_i$ engender the $\sh{O}_{X,x}$-module $(\Omega^1_{X/k})_x$.
\end{enumerate}
If $f$ is \'etale at the point $x$, $X$ is lisse on $k$ at the point $x$ since $Y=\operatorname{Spec}(k[\mathrm{T}_1,\ldots,\mathrm{T}_n])$ is lisse on $k$ \sref{IV.17.3.8}; the fact that $a)$ implies $b)$ is thus a particular case of \sref{IV.17.11.3}.
As $b)$ trivially implies $c)$, it remains to see that $c)$ implies $a)$.
\end{proposition}

\begin{proof}
Note first that hypothesis $c)$ entails that the homomorphism $(f^*(\Omega^1_{Y/k}))_x\to(\Omega^1_{X/k})_x$ is surjective, hence, replacing $X$ by an open neighbourhood of $x$, one can suppose that the homomorphism $f^*(\Omega^1_{Y/k})\to\Omega^1_{X/k}$ is surjective (Bourbaki, \emph{Alg.~comm.}, chap.~II, \textsection~5, n\textsuperscript{o}~1, prop.~2); hence $f$ is non ramified (\sref{IV.17.2.2}).
Let us first show that one can reduce to the case where $x$ is rational over $k$.
Indeed, if one sets $k'=k(x)$ and $X'=X\otimes_k k'$, $Y'=Y\otimes_k k'=\Spec(k'[\mathrm{T}_1,\ldots,\mathrm{T}_n])$, there exists a point $x'\in X'$ above $x$ such that $k(x')=k'$.
To prove that $f$ is \'etale at the point $x$, it suffices to show that $f'=f_{(k')}:X'\to Y'$ is \'etale at the point $x'$ (\sref{IV.17.7.1}, (ii)); moreover $f'$ is non ramified (\sref{IV.17.3.3}, (iii)) and one has $\dim_{x'}(X')=n$ (\sref{IV.4.2.7}).
In the same way, replacing $k'$ by an algebraically closed extension of $k'$, one can suppose that $k$ is algebraically closed.
Set then $y=f(x)$, $A=\sh{O}_{Y,y}$, $B=\sh{O}_{X,x}$; as the residue field of $B$ is equal to $k$, the same is true of that of $A$, hence $x$ (resp.\ $y$) is a closed point of $X$ (resp.\ $Y$) $(\sref[I]{I.6.4.2})$ and one has therefore $\dim(A)=\dim(B)=n$.
Since $f$ is non ramified, the homomorphism $\hat{A}\to\hat{B}$ is surjective (\sref{IV.17.4.4}, $f''$)); but as $\dim(\hat{A})=\dim(\hat{B})=n$, and $\hat{A}$, being a regular ring, is integral, the homomorphism $\hat{A}\to\hat{B}$ is also injective ($\mathbf{0}$, 16.3.10).
This homomorphism is therefore bijective, which completes the proof that $f$ is \'etale at the point $x$ (\sref{IV.17.6.3}, $e''$)).
\end{proof}

\oldpage[IV]{100}

\begin{corollary}[17.15.4]
\label{IV.17.15.4}
Under the general hypotheses of \sref{IV.17.15.3}, suppose moreover that $k(x)$ is a finite and separable extension of $k$ and that the germs $(g_i)_x$ belong to $\mathfrak{m}_x$.
Then the conditions $a)$, $b)$, $c)$ of \sref{IV.17.15.3} are also equivalent to each of the following:
\begin{enumerate}
  \item[d)] The $n$ germs $(g_i)_x$ engender the maximal ideal $\mathfrak{m}_x$ of $\sh{O}_{X,x}$.
  \item[d')] The ring $\sh{O}_{X,x}$ is regular and the $(g_i)_x$ form a regular system of parameters for this ring ($\mathbf{0}$, \sref{IV.17.1.6}).
\end{enumerate}
\end{corollary}

\begin{proof}
Indeed, $d')$ trivially implies $d)$.
On the other hand, as $k(x)$ is a finite extension and separable of $k$, one has $\Omega^1_{k(x)/k}=0$ ($\mathbf{0}$, \sref{IV.20.6.20}) and $k(x)=\sh{O}_{X,x}/\mathfrak{m}_x$ is a $k$-algebra formally lisse for the discrete topologies ($\mathbf{0}$, \sref{IV.19.6.1}), hence the exact suite ($\mathbf{0}$, \sref{IV.20.5.14.1}) applies with $A=k$, $B=\sh{O}_{X,x}$, $\mathfrak{R}=\mathfrak{m}_x$, and furnishes an isomorphism canonical
\[
\delta:\mathfrak{m}_x/\mathfrak{m}_x^2\xrightarrow{\sim} (\Omega^1_{X/k})_x\otimes_{\sh{O}_{X,x}} k(x).
\]
From this one first deduces the equivalence of conditions $c)$ and $d)$, by virtue of the Nakayama lemma.
On the other hand, if $f$ is \'etale at the point $x$, the ring $\sh{O}_{X,x}$ is regular and of dimension $n$ ($\mathbf{I}$, \sref{IV.6.4.2}), and $n$ elements that engender this ideal maximal form necessarily a regular system of parameters for $\sh{O}_{X,x}$ ($\mathbf{0}$, \sref{IV.17.1.6}), which proves that $a)$ implies $d')$.
\end{proof}

\begin{proposition}[17.15.5]
\label{IV.17.15.5}
Let $k$ be a field, $X$ a prescheme locally of finite type on $k$, $x$ a point of $X$, $n=\dim_x(X)$.
The following conditions are equivalent:
\begin{enumerate}
  \item[a)] $X$ is lisse on $k$ at the point $x$.
  \item[b)] $X$ is differentially smooth on $k$ at the point $x$.
  \item[c)] $(\Omega^1_{X/k})_x$ is an $\sh{O}_{X,x}$-module libre of rank $n$.
  \item[d)] $(\Omega^1_{X/k})_x$ is an $\sh{O}_{X,x}$-module admitting a system of $n$ generators.
\end{enumerate}
\end{proposition}

\begin{proof}
We have already proved that $a)$ implies $c)$ \sref{IV.17.10.2}; $c)$ implies $d)$ trivially; $c)$ also implies $b)$ \sref{IV.17.12.4}.
The fact that $d)$ implies $a)$ results from ($\mathbf{0}$, \sref{IV.20.4.7} and \sref{IV.17.15.3},~$c$)).
The fact that $b)$ implies $a)$ is shown in the proof of \sref{IV.17.15.3}.
\end{proof}

\begin{corollary}[17.15.6]
\label{IV.17.15.6}
Let $X$ be a prescheme of finite type on a field $k$.
For $X$ to be lisse on $k$, it is necessary and sufficient that the $\sh{O}_X$-Module $\Omega^1_{X/k}$ be locally free, and that the local rings of $X$ at maximal points be fields, separable extensions of $k$ (the latter condition being automatically satisfied if $k$ is a perfect field and $X$ a \emph{reduced} prescheme).
\end{corollary}

\begin{proof}
The conditions are necessary, for if $X$ is lisse on $k$, it results from \sref{IV.17.10.2} that $\Omega^1_{X/k}$ is locally free; on the other hand, $X$ is regular, hence \emph{a fortiori} reduced, hence at every maximal point $x$ of $X$, $\sh{O}_{X,x}$ is a field, which must be a $k$-algebra formally smooth for the discrete topologies (\sref{IV.17.5.1}), hence a separable extension of $k$ ($\mathbf{0}$, 19.6.1).

The conditions are sufficient.
One can indeed reduce to the case where $X$ is connected, hence $\Omega^1_{X/k}$ is locally free of constant rank $n$.
For every maximal point $x$ of $X$, one then has $\operatorname{rg}_{k(x)}\Omega^1_{k(x)/k}=n$ since $\sh{O}_{X,x}=k(x)$; as by hypothesis $k(x)$ is a separable extension of $k$, one has $\Upsilon_{k(x)/k}=0$ ($\mathbf{0}$, 20.6.3), hence $\deg.\operatorname{tr}_k k(x)=n$ by Cartier's equality ($\mathbf{0}$, 21.7.1).
All the irreducible components of $X$ therefore have the same dimension $n$ (\sref{IV.5.2.1}), and one concludes that $X$ is lisse on $k$ at every point by virtue of the fact that $c)$ implies $a)$ in (\sref{IV.17.15.5}).
\end{proof}

\begin{proof}

\end{proof}

\begin{remarks}[17.15.10]

\begin{enumerate}[label=(\roman*)]

\begin{proposition}[17.15.12]
\label{IV.17.15.12}
Let $k$ be a field, $X$ a prescheme locally of finite type over $k$.
For there to exist an everywhere dense open subset $U$ of $X$ which is lisse over $k$, it is necessary and sufficient that for every maximal point $x$ of $X$, $X$ be reduced at this point and that $k(x)$ be a separable extension of $k$.
For there to exist an everywhere dense open subset $U$ of $X$ such that $U_{\mathrm{red}}$ be lisse over $k$, it is necessary and sufficient that for every maximal point $x$ of $X$, $k(x)$ be a separable extension of $k$.
\end{proposition}

\begin{proof}
The second assertion results evidently from the first.
To say that there exists an everywhere dense open $U$ lisse over $k$ signifies that $X$ is lisse over $k$ at each of its maximal points.
It is necessary for this that $\sh{O}_{X,x}$ be regular \sref{IV.17.15.1} and \emph{a fortiori} reduced, hence a \emph{field}, equal to $k(x)$; moreover \sref{IV.17.15.1}, $k(x)\otimes_k k'$ must be regular for every radiciel extension $k'$ of $k$, hence $k(x)$ must be a separable extension of $k$ \sref{IV.4.3.5}; the reciprocal is immediate, by virtue of \sref{IV.17.15.1}.
\end{proof}

\end{corollary}

\begin{proof}
\oldpage[IV]{104}

The statements of this number complement those of \sref{IV.14.5}, by means of hypotheses of flatness.

\begin{proposition}[17.16.1]
\label{IV.17.16.1}
Let $f:X\to S$ be a morphism flat locally of finite presentation.
Let $s\in S$, $x$ a closed point of $X_s$ such that $\sh{O}_{X_s,x}$ is a Cohen-Macaulay ring; then there exists an open neighbourhood $U$ of $s$ in $S$ and a sub-prescheme $X'\subset f^{-1}(U)$ of $X$ such that $x\in X'$ and the morphism $X'\to U$, restriction of $f$, is flat, quasi-finite and of finite presentation.
\end{proposition}

\begin{corollary}[17.16.2]
\label{IV.17.16.2}
Let $f:X\to S$ be a morphism faithfully flat and locally of finite presentation.
Then there exists a morphism $g:S'\to S$, faithfully flat, locally of finite presentation and locally quasi-finite, such that there exists an S-morphism $S'\to X$ (in other words, such that there exists an $S'$-section of $X'=X\times_S S'$ ($\mathbf{I}$, \sref{IV.3.3.14})).
If $S$ is quasi-compact (resp.\ quasi-compact and quasi-separated), one can suppose $S'$ affine (resp.\ the morphism $g$ of finite presentation and quasi-finite).
\end{corollary}

\begin{corollary}[17.16.3]
\label{IV.17.16.3}
(i) Let $f:X\to S$ be a smooth morphism lisse.
Let $s\in S$, $x$ a closed point of $X_s$ such that the residue field $k(x)$ is separable on $k(s)$; then, in the conclusion of \sref{IV.17.16.1}, one can take the morphism $X'\to U$, restriction of $f$, to be \'etale.

(ii)
Let $f:X\to S$ be a smooth morphism lisse surjective.
Then, in the conclusions of \sref{IV.17.16.2}, one can further suppose that $g:S'\to S$ be \'etale.
\end{corollary}

\begin{proposition}[17.16.4]
\label{IV.17.16.4}
Let $S$ be a prescheme quasi-compact and quasi-separated, $f:X\to S$ a morphism of finite presentation such that, for every $s\in S$, $X_s=f^{-1}(s)$ is a prescheme proper on $k(s)$.
Then there exists a finite family $(S_i)_{i\in I}$ of sub-preschemes affines of $S$, of finite presentation on $S$, pairwise disjoint, of reunion $S$, and such that, for every $i\in I$, there exist two morphisms of finite presentation $S''_i\xrightarrow{g_i} S'_i\xrightarrow{h_i}S_i$, where $h_i$ is \'etale and $g_i$ flat and radiciel, and an S-morphism $S''_i\to X$ (in other words, such that there exists an $S''_i$-section of $X''_i=X\times_S S_i''\to S_i$).
\end{proposition}

\begin{corollary}[17.16.5]
\label{IV.17.16.5}
Let $f:X\to S$ be a surjective morphism locally of finite presentation.
Then there exists a morphism $g:S'\to S$ surjective, locally of finite presentation and locally quasi-finite, such that there exists an S-morphism $S'\to X$ (in other words, such that there exists an $S'$-section of $X'=X\times_S S'$).
If $S$ is quasi-compact (resp.\ quasi-compact and quasi-separated), one can suppose $S'$ affine (resp.\ $S'$ affine and the morphism $g$ of finite presentation and quasi-finite).
\end{corollary}

\begin{proposition}[17.16.6]
\label{IV.17.16.6}
Let $S$ be a prescheme quasi-compact and quasi-separated, $f:X\to S$ a morphism of finite presentation such that, for every $s\in S$, $X_s=f^{-1}(s)$ is a prescheme proper on $k(s)$.
Then there exists a finite family $(S_i)_{i\in I}$ of sub-preschemes affines of $S$, of finite presentation on $S$, pairwise disjoint, of reunion $S$, and such that, for every $i$, the morphism $X_i=X\times_S S_i\to S_i$ is proper and flat.
If moreover, for every $s\in S$, $X_s$ is finite on $k(s)$, one can take the $S_i$ such that each of the morphisms $X_i\to S_i$ factors as
\[
X_i \xrightarrow{u_i} X'_i \xrightarrow{h_i} S_i
\]
where $u_i$ and $h_i$ are finite and locally free, $u_i$ is \'etale and surjective, and $h_i$ is flat.
\end{proposition}
