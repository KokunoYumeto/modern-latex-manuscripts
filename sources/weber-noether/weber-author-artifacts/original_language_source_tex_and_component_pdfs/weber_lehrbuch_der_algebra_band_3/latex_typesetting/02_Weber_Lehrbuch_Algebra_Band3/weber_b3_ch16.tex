\begin{align}
\tag{7}
\frac{2\varkappa'K}{\pi}\,\frac{1}{\cn2Ku}
&= \frac{1}{\cos\pi u} - 4\cos\pi u\,\sum\frac{(-1)^{\frac{m}{2}}q^{\frac{n}{2}}(1+q^m)}{1+2q^m\cos2\pi u+q^{2m}} \notag\\
&= \frac{1}{\cos\pi u} - 4\sum(-1)^{\frac{n-1}{2}}\frac{q^n}{1+q^n}\cos n\pi u.
\end{align}

\begin{align}
\tag{8}
\frac{2\varkappa'K}{\pi}\,\frac{\sn Ku}{\cn 2Ku}
&= \tg\pi u + 4\sin2\pi u\,\sum\frac{(-1)^{\frac{m}{2}}q^m}{1-q^m\cos2\pi u+q^{2m}} \notag\\
&= \tg\pi u + 4\sum(-1)^{\frac{m}{2}}\frac{q^m}{1+q^m}\sin m\pi u.
\end{align}

\begin{align}
\tag{9}
\frac{2K}{\pi}\,\frac{\dn2Ku}{\cn2Ku}
&= \frac{1}{\cos\pi u} + 4\cos\pi u\,\sum\frac{q^{\frac{m}{2}}(1+q^m)}{1+2q^m\cos2\pi u+q^{2m}} \notag\\
&= \frac{1}{\cos\pi u} + 4\sum(-1)^{\frac{n-1}{2}}\frac{q^n}{1-q^n}\cos n\pi u.
\end{align}

\begin{align}
\tag{10}
\frac{2\varkappa'K}{\pi}\,\frac{1}{\dn2Ku}
&= 1 - 4\sum(-1)^{\frac{n-1}{2}}\frac{q^n(q^n+\cos2\pi u)}{1+2q^n\cos2\pi u+q^{2n}} \notag\\
&= 1 + 4\sum(-1)^{\frac{m}{2}}\frac{q^{\frac{m}{2}}}{1+q^m}\cos m\pi u.
\end{align}

\begin{align}
\tag{11}
\frac{2\varkappa\varkappa'K}{\pi}\,\frac{\sn2Ku}{\dn2Ku}
&= 4\sin\pi u\,\sum\frac{(-1)^{\frac{n-1}{2}}q^{\frac{n}{2}}(1-q^n)}{1+2q^n\cos2\pi u+q^{2n}} \notag\\
&= 4\sum\frac{q^{\frac{n}{2}}}{1-q^n}\sin n\pi u.
\end{align}

\begin{align}
\tag{12}
\frac{2\varkappa K}{\pi}\,\frac{\cn2Ku}{\dn2Ku}
&= 4\cos\pi u\,\sum\frac{(-1)^{\frac{n-1}{2}}q^{\frac{n}{2}}(1+q^n)}{1+2q^n\cos2\pi u+q^{2n}} \notag\\
&= 4\sum\frac{q^{\frac{n}{2}}}{1+q^n}\cos n\pi u.

\end{equation}
Man setze nun
\begin{alignat*}{2}
-g + \alpha u &= a,\qquad & g + \alpha u &= a'\\
\alpha\delta &= a' - a &&= 2g
\end{alignat*}
und erh"alt:
\begin{equation}
\tag{15}
a < \Sigma g_s\lambda_s < a'.
\end{equation}
Ersetzt man $u$ durch $u' = u + \delta$, so geht $a$ in $a'$ "uber (das w"are bei der Formulierung des Textes nicht der Fall), und wenn man $a'' = a' + \alpha\delta = g + \alpha u'$ setzt, und wenn $\lambda_s'$ ebenso von $u'$ abh"angt wie $\lambda_s$ von $u$, so ergibt sich
\[a' < \Sigma g_s\lambda_s' < a''.\]

\[\eta,\; \eta',\; \eta'',\; \eta''',\; \dots\]
