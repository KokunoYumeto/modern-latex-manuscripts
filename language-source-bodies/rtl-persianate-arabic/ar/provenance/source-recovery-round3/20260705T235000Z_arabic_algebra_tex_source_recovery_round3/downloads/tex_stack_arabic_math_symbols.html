this works with XeLaTeX

\documentclass[2pt,a4paper]{article}
\usepackage{amsmath,amsfonts,amssymb,mathrsfs,tikz,fancyhdr}
\usepackage{fontspec} % For loading OpenType fonts
\usepackage{unicode-math} % For setting the math font
\usepackage{polyglossia}
\usepackage{bidi}

\setdefaultlanguage[calendar=gregorian]{arabic}  % from polyglossia
\setmainfont[Script=Arabic,Mapping=arabicdigits]{Amiri}
\setmathfont{XITS Math} % Set the math font to XITS Math

% Custom integral with Arabic text on the left
\newcommand{\intr}[3]{%
\reflectbox{\ensuremath{\displaystyle\int_{\reflectbox{$\scriptstyle\text{#1}$}}^{\reflectbox{$\scriptstyle\text{#2}$}}{\reflectbox{\textnormal{#3}}}}}%
}

% Custom superscript
\newcommand{\os}[2]{%
\reflectbox{\ensuremath{\reflectbox{\text{#1}}^{\reflectbox{$\scriptstyle\text{#2}$}}}}%
}

% Custom subscript
\newcommand{\sub}[2]{%
\reflectbox{\ensuremath{\reflectbox{\text{#1}}_{\reflectbox{$\scriptstyle\text{#2}$}}}}%
}

% Custom square root with Arabic text
\newcommand{\arsqrt}[2]{%
\reflectbox{\ensuremath{\sqrt[\reflectbox{$\scriptstyle\text{#1}$}]{\reflectbox{\textnormal{#2}}}}}%
}

% Custom summation
\newcommand{\asum}[3]{%
\reflectbox{\ensuremath{\displaystyle\sum_{\reflectbox{$\scriptstyle\text{#1}$}}^{\reflectbox{$\scriptstyle\text{#2}$}}{\reflectbox{\text{#3}}}}}%
}

% Custom Limits with subscript directly beneath, with arrow
\newcommand{\alim}[3]{%
\reflectbox{\ensuremath{\underset{\reflectbox{$\scriptstyle\text{#1 $\leftarrow$ #2}$}}{\reflectbox{\text{نهـــــــا     }} \reflectbox{\text{#3}}}}}%
}

% Custom fraction
\newcommand{\afrac}[2]{%
\reflectbox{$\frac{\reflectbox{\text{#1}}}{\reflectbox{\text{#2}}}$}%
}

\begin{document}

\title{الرموز الرياضية العربية}
\date{}
\maketitle

\section{التكاملات}
كود التكامل بجانبه ثلاثة أقواس وهم الحد الأدنى للتكامل، الحد الأعلى للتكامل، والدالة التي سيتم تكاملها. إذا تركت قوساً فارغاً لن يحدث مشكلة.

\begin{itemize}
    \item \intr{}{}{} $\Leftarrow$  علامة تكامل فارغة

    \item \intr{س}{ص}{} $\Leftarrow$  حدود التكامل بها حرف واحد فقط

    \item \intr{\sub{س}{أ}}{\sub{س}{ب}}{} $\Leftarrow$ كل حد من حدود التكامل به حرفين وغالبًا يكون الحرف الثاني (أ، ب) أسفل الأول لذا استخدمنا كود الـ subscript بداخل أقواس كل حد

    \item \intr{}{}{\os{س}{2} دس} $\Leftarrow$ تكامل غير محدد

    \item \intr{\sub{س}{0}}{\sub{س}{1}}{\os{س}{2} دس } $\Leftarrow$ تكامل محدد

    \item \intr{\sub{س}{أ}}{\sub{س}{ب}}{س + ص} $\Leftarrow$ تكامل مع دالة متغيرتين
\end{itemize}

\section{الأسس}
كود الأسس يحتوي على قوسين فقط: الأول للأساس والثاني للأس أو القوة.

\begin{itemize}
    \item \os{ع}{2} $\Leftarrow$ الأساس "ع" والأس "2"
    \item \os{ص}{\os{2}{3}} = \os{ص}{6} $\Leftarrow$ يمكن إنشاء أسس مرفوعة لبعض الأسس الأخرى
    \item \os{س}{$\infty$} $\Leftarrow$ استخدام الرموز اللاتينية في الأسس
    \item \os{س}{\os{ص}{2$$}} $\Leftarrow$ أس مع متغيرين
\end{itemize}

\section{الجذور}
كود الجذور يحتوي على قوسين: الأول للأساس (درجة الجذر) والثاني للمحتوى.

\begin{itemize}
    \item \arsqrt{2}{ص + 3} $\Leftarrow$ جذر تربيعي لدالة
    \item \arsqrt{\os{ن}{3}}{س + 5} $\Leftarrow$ جذر نوني مع أسس
\end{itemize}

\section{التجميع}
كود التجميع يحتوي على ثلاثة أقواس: الأول للحرف المتغير، الثاني للحد الأدنى، والثالث للحد الأعلى.

\begin{itemize}
    \item \asum{ك = 0}{ن}{\sub{س}{ك}} = \sub{س}{0}،\sub{س}{1}،...،\sub{س}{ن} $\Leftarrow$ تجميع لحدود مختلفة
    \item \asum{ك = 1}{ن}{\os{ع}{2}} $\Leftarrow$ تجميع للأسس
\end{itemize}

\section{الكسور}
كود الكسر يحتوي على قوسين فقط: الأول للبسط والثاني للمقام.

\begin{itemize}
    \item \afrac{ص + 2س}{س} $\Leftarrow$ كسر عادي
    \item \afrac{\os{س}{2} + \os{ص}{3}}{\os{ع}{2}} $\Leftarrow$ كسر مع أسس
    \item \afrac{\arsqrt{3}{س + 1}}{ص} $\Leftarrow$ كسر مع جذور
\end{itemize}

\section{النهايات}
كود النهاية يحتوي على ثلاثة أقواس: الأول للمتغير، الثاني للحد الذي يقترب منه المتغير، والثالث للدالة.

\begin{itemize}
    \item \alim{س}{0}{\afrac{جا(س)}{س}} $\Leftarrow$ النهاية للدالة \afrac{جا(س)}{س} عندما تقترب س من الصفر.
    \item \alim{س}{$\infty$}{\afrac{1}{س}} $\Leftarrow$ النهاية للدالة \afrac{1}{س} عندما تقترب س من اللانهاية.
\end{itemize}

\end{document}
