\section{Termín 27. ledna 2025}

\subsection{Tečná rovina}
K ploše
\[
    x^3 + e^{xyz} = 2 \cos(y)
\]
v bodě $A = (1, 0, 1)$ určete
\begin{enumerate}[(a),noitemsep]
    \item tečnou rovinu,
    \item kolmou přímku.
\end{enumerate}

\subsection{Stacionární body}
Určete největší a nejmenší hodnotu funkce
\[
    f(x,y) = -xy + y^2 + 2x - y
\]
na ploše trojúhelníku $M$ s vrcholy $(0, 0)$, $(1, 0)$ a $(0, 1)$.

\subsection{Dvojný integrál}
Přepište následující integrál
\[
    \int_{0}^{1} \int_{-\sqrt{x}}^{x} f(x, y) \dif y \dif x
\]
\begin{enumerate}[a), noitemsep]
    \item v opačném pořadí integrace,
    \item v polárních souřadnicích se středem v počátku v pořadí $\dif \varrho \dif \varphi$.
\end{enumerate}

\subsection{Křivkový integrál}
Najděte délku křivky $C$ s parametrizací
\[
    \varphi(t) = 
    \begin{pmatrix}
        e^t, & e^t \sin t, & e^t \cos t
    \end{pmatrix} \text{pro } 0 \leq t \leq 2 \pi.
\]

\subsection{Gaussova věta}
Pomocí Gaussovy věty určete tok pole
\[
    \vec{F}(x, y, z) =
    \begin{pmatrix}
        xz, & z^2, & y^2
    \end{pmatrix}
\]
povrchem tělesa
\[
    E \text{: } x^2 + y^2 + z^2 \leq 1 \text{, } x \geq 0 \text{, } z \geq 0
\]
s vnější orientací.
