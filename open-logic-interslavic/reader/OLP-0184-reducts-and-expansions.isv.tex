\section{Redukty i razširjenja}

Često jest koristno ili potrěbno sravnjivati jezyky, ktore imajut
obće simboly, kako i !!{structure}s tih jezykov. Najčestějši slučaj
nastupaje, kogda vse simboly v !!a{language}~$\Lang{L}$ takože sųt
čestju !!a{language}~$\Lang{L'}$, t.j. $\Lang{L} \subseteq
\Lang{L'}$. Taku $\Lang{L}$-!!{structure}~$\Struct{M}$ možno vsegda
razširiti do $\Lang{L'}$-!!{structure} dodanjem interpretacij
dodatnyh simbolov, pri tom ostavljajuči interpretacije obćih simbolov
jednakymi. S drugoj strany, iz
$\Lang{L'}$-structure~$\Struct{M'}$ možemo dostati
$\Lang{L}$-structure prosto „zabyvajuči” interpretacije simbolov,
ktore ne pojavjajut se v~$\Lang{L}$.

\begin{defn}
\ollabel{defn:reduct}
Prědpoložimo $\Lang L \subseteq \Lang L'$, $\Struct M$ jest
$\Lang L$-!!{structure}, a $\Struct M'$ jest $\Lang L'$-!!{structure}.
$\Struct M$ jest \emph{reduktom} struktury $\Struct M'$ do $\Lang L$,
a $\Struct M'$ jest \emph{razširjenjem} struktury $\Struct M$ do
$\Lang L'$ togda i samo togda, kogda
\begin{enumerate}
\item $\Domain{M} = \Domain{M'}$
\item Dla vsakoj !!{constant}~$c \in \Lang L$, $\Assign{c}{M} =
  \Assign{c}{M'}$.
\item Dla vsakoj !!{function}~$f \in \Lang L$, $\Assign{f}{M} =
  \Assign{f}{M'}$.
\item Dla vsakogo !!{predicate}~$P \in \Lang L$, $\Assign{P}{M} =
  \Assign{P}{M'}$.
\end{enumerate}
\end{defn}

\begin{prop}
\ollabel{prop:reduct}
Ako $\Lang{L}$-!!{structure}~$\Struct{M}$ jest reduktom
$\Lang{L'}$-!!{structure}~$\Struct{M'}$, togda dla vsih
$\Lang{L}$-!!{sentence}s~$!A$,
\[
\Sat{M}{!A} \text{ togda i samo togda, kogda } \Sat{M'}{!A}.
\]
\end{prop}

\begin{proof}
  Zadanje.
\end{proof}

\begin{prob}
Dokažite \olref[mod][bas][red]{prop:reduct}.
\end{prob}

\begin{defn}
Kogda imajemo $\Lang{L}$-structure $\Struct{M}$, a $\Lang{L'} =
\Lang{L} \cup \{P\}$ jest razširjenje jezyka $\Lang{L}$, dostano
dodanjem jedinogo $n$-městnogo !!{predicate}~$P$, i
$R \subseteq \Domain{M}^n$ jest $n$-městno odnošenje, togda pišemo
$\Expan{M}{R}$ za razširjenje~$\Struct{M'}$ struktury~$\Struct{M}$ s
$\Assign{P}{M'} = R$.
\end{defn}
